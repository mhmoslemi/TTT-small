import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with bounds
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_integral}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts
    n_restarts = 20
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start with random perturbation of best known solution
        h0 = best_h + np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1) 
        h0 = h0 * (n_points/2) / np.sum(h0)
        
        try:
            res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_test = res.x
                c5_test = objective(h_test)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
        except:
            pass
    
    # Refine with CVXPY if time allows
    if time() - start_time < budget_s * 0.8:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add a few key constraint based on best solution
            h_test = best_h.copy()
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            max_k_idx = np.argmax(corr)
            
            # Add constraints to prevent bad overlaps
            # Sample 5 top lags for constraints
            top_lags = np.argsort(corr)[-5:]
            for k in top_lags:
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            pass
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
