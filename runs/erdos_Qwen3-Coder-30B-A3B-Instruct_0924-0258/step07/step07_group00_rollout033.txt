import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient descent approach for better local optimization
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-9
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Update best if improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Identify which lag causes maximum overlap
        # The full correlation has length 2*n_points - 1
        # The middle element corresponds to lag 0
        lag = max_lag_idx - (n_points - 1)
        
        # Gradient of overlap w.r.t h for this lag
        # ∂/∂h[i] (h[i] * (1 - h[i+lag])) = 1 - h[i+lag] - h[i]
        gradient = np.zeros(n_points)
        
        # Handle the finite difference carefully for all valid i
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j] - h[i]
        
        # Update via gradient descent
        h_new = h - step_size * gradient
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if improvement is minimal, then decrease step_size
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if abs(new_c5 - max_overlap) < tolerance:
            step_size *= 0.5
            if step_size < 1e-8:
                break
        else:
            h = h_new
            
    # Final refinement with SLSQP if possible
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
