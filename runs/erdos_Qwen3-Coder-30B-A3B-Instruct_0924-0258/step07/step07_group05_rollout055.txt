import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
        if fit < best_c5:
            best_h = h.copy()
            best_c5 = fit

    # Main GA loop
    generation = 0
    stagnant_generations = 0
    best_score_history = [best_c5]
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = sorted_pop[:elite_size]
        new_fit = sorted_fit[:elite_size]
        
        # Tournament selection
        while len(new_pop) < pop_size:
            # Select parents
            indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fit = [fitness[i] for i in indices]
            parent1_idx = indices[np.argmin(tournament_fit)]
            
            indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fit = [fitness[i] for i in indices]
            parent2_idx = indices[np.argmin(tournament_fit)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, population[parent1_idx], population[parent2_idx])
            else:
                child = population[parent1_idx].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.03)
                child = np.clip(child, 0, 1)
            
            # Repair constraint
            sum_h = np.sum(child)
            if abs(sum_h - n_points/2) > 1e-6:
                child = child * (n_points/2) / sum_h
            
            # Evaluate
            child_fit = evaluate_erdos_solution(child, dx)
            
            new_pop.append(child)
            new_fit.append(child_fit)
            
            if child_fit < best_c5:
                best_h = child.copy()
                best_c5 = child_fit
                stagnant_generations = 0
            else:
                stagnant_generations += 1
        
        population = new_pop
        fitness = new_fit
        
        # Periodic convex relaxation every 20 generations
        if generation % 20 == 0:
            try:
                # Select top candidates
                sorted_indices = np.argsort(fitness)[:10]
                for idx in sorted_indices:
                    h_candidate = population[idx].copy()
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add some overlap constraints
                    for k in range(1, n_points//2):
                        if k < n_points:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-5)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass
                
        # Early stopping
        if stagnant_generations > 50:
            break
            
        best_score_history.append(best_c5)
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
