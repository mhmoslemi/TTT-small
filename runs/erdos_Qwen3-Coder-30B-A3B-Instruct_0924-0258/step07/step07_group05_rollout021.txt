import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis functions approach
    m = 10  # number of basis functions
    # Create basis functions as piecewise linear
    basis_functions = []
    x_grid = np.linspace(0, 2, n_points)
    
    # Initialize with simple triangular basis
    for i in range(m):
        basis = np.zeros(n_points)
        center = i * 2.0 / (m - 1) if m > 1 else 1.0
        half_width = 0.4 / (m + 1)
        for j in range(n_points):
            x = x_grid[j]
            if abs(x - center) <= half_width:
                basis[j] = 1.0 - abs(x - center) / half_width
        basis_functions.append(basis)
    
    # Convert basis to matrix form
    basis_matrix = np.array(basis_functions).T
    # Normalize basis to ensure proper scaling
    basis_matrix = basis_matrix / (np.max(np.sum(basis_matrix, axis=0)) or 1.0)
    
    # Initialize coefficients
    coeffs = np.random.rand(m) * 0.5
    h_current = basis_matrix @ coeffs
    
    # Ensure initial h satisfies constraints
    scale_factor = (n_points / 2) / np.sum(h_current)
    h_current = h_current * scale_factor
    h_current = np.clip(h_current, 0, 1)
    
    # Optimization loop
    max_iter = 2000
    patience = 0
    max_patience = 100
    perturbation_magnitude = 0.05
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current solution
        h_current = basis_matrix @ coeffs
        h_current = np.clip(h_current, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_current)
        h_current = h_current * scale_factor
        
        # Evaluate current c5
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        # Perturbation check
        if patience > max_patience:
            coeffs += np.random.normal(0, perturbation_magnitude, m)
            perturbation_magnitude *= 0.5
            patience = 0
            
        # Trust-region step
        # Compute gradient estimation (simplified)
        eps = 1e-6
        grad = np.zeros(m)
        for i in range(m):
            coeffs_p = coeffs.copy()
            coeffs_p[i] += eps
            h_p = basis_matrix @ coeffs_p
            h_p = np.clip(h_p, 0, 1)
            scale_p = (n_points / 2) / np.sum(h_p)
            h_p = h_p * scale_p
            c5_p = evaluate_erdos_solution(h_p, dx)
            grad[i] = (c5_p - current_c5) / eps
            
        # Simple gradient descent step (trust region like)
        step_size = 0.01
        new_coeffs = coeffs - step_size * grad
        new_coeffs = np.clip(new_coeffs, -1, 1)  # Bound coefficients
        coeffs = new_coeffs
        
    # Final refinement with local optimizer
    try:
        def objective(c):
            h = basis_matrix @ c
            h = np.clip(h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h)
            h = h * scale_factor
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(c):
            h = basis_matrix @ c
            h = np.clip(h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h)
            h = h * scale_factor
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(-10, 10) for _ in range(m)]
        
        res = minimize(objective, coeffs, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        
        if res.success:
            h_final = basis_matrix @ res.x
            h_final = np.clip(h_final, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h_final)
            h_final = h_final * scale_factor
            c5_final = objective(res.x)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
