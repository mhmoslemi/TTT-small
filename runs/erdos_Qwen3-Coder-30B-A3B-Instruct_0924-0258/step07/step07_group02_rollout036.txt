import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_iter = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    tournament_size = 5
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            # Latin hypercube sampling approach
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Hall of fame
    hall_of_fame = [(best_h.copy(), best_c5)]
    
    # Evolution loop
    no_improvement_count = 0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        if sorted_fit[0] < hall_of_fame[0][1]:
            hall_of_fame = [(sorted_pop[0].copy(), sorted_fit[0])]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 80:
            break
            
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(sorted_pop[i].copy())
            new_fitness.append(sorted_fit[i])
        
        # Selection and reproduction
        while len(new_population) < pop_size:
            # Tournament selection
            selected_indices = np.random.choice(pop_size, tournament_size, replace=False)
            selected_fit = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fit)]
            
            parent = population[winner_idx].copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Gaussian mutation with decaying variance
                decay_factor = max(0.1, 0.1 * (1 - iter_num / max_iter))
                mutation = np.random.normal(0, decay_factor, n_points)
                parent = parent + mutation
                parent = np.clip(parent, 0, 1)
                parent = parent * (n_points/2) / np.sum(parent)
            
            # Crossover (with probability)
            if np.random.random() < crossover_rate and len(new_population) + 1 < pop_size:
                # Select second parent
                selected_indices2 = np.random.choice(pop_size, tournament_size, replace=False)
                selected_fit2 = [fitness[i] for i in selected_indices2]
                winner_idx2 = selected_indices2[np.argmin(selected_fit2)]
                parent2 = population[winner_idx2].copy()
                
                # Arithmetic crossover
                alpha = np.random.uniform(0.2, 0.8)
                child1 = alpha * parent + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent + alpha * parent2
                
                # Repair and normalize
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                new_fitness.append(evaluate_erdos_solution(child1, dx))
                
                if len(new_population) < pop_size:
                    new_population.append(child2)
                    new_fitness.append(evaluate_erdos_solution(child2, dx))
            else:
                new_population.append(parent)
                new_fitness.append(evaluate_erdos_solution(parent, dx))
        
        population = new_population
        fitness = new_fitness
    
    # Final SQP refinement of top candidates
    top_candidates = []
    top_fitness = []
    for i in range(min(5, len(population))):
        top_candidates.append(population[i].copy())
        top_fitness.append(fitness[i])
    
    for i in range(len(top_candidates)):
        if time() - start_time > budget_s * 0.95:
            break
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add objective
        try:
            prob = cp.Problem(cp.Minimize(cp.sum(h_opt)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
