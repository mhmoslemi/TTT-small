def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, dx, active_k):
    n = len(h)
    grad = np.zeros(n)
    
    # For active k values, compute gradient contribution
    for k in active_k:
        if k < n:
            # Gradient for overlap term (1-h[i]) * h[i+k] 
            for i in range(n-k):
                grad[i] += (1 - h[i+k])
            for i in range(n-k):
                grad[i+k] -= h[i]
    
    # Normalize and clip
    grad = grad * dx / n
    return grad
