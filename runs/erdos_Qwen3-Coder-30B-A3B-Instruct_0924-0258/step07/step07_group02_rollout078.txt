import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm
    pop_size = 50
    max_generations = 200
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = 5
    
    # Initialize population
    population = []
    fitness = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            noise = np.random.normal(0, 0.05, n_points)
            h = h_start + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # EA loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = []
        
        # Elite preservation
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate, dx, n_points)
            child2 = mutate(child2, mutation_rate, dx, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Evaluate new population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Adaptive mutation rate
        mutation_rate = max(0.005, mutation_rate * 0.99)
    
    # Final SQP refinement with warm start from best solution so far
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        # Add the main constraint based on best
        h_test = best_h.copy()
        corr = np.correlate(h_test, 1 - h_test, mode='full')
        max_k_idx = np.argmax(corr)
        max_lag = max_k_idx - (n_points - 1)
        # Just add a few key constraints
        sample_lags = np.argsort(corr)[-5:]
        for k in sample_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, tournament_size):
    selected_idx = np.random.choice(len(population), tournament_size)
    selected_fitness = [fitness[i] for i in selected_idx]
    winner_idx = selected_idx[np.argmin(selected_fitness)]
    return population[winner_idx].copy()

def crossover(parent1, parent2):
    # Blend crossover (BLX-0)
    alpha = 0.5
    child1 = np.zeros_like(parent1)
    child2 = np.zeros_like(parent2)
    for i in range(len(parent1)):
        a = min(parent1[i], parent2[i])
        b = max(parent1[i], parent2[i])
        range_val = b - a
        lower = a - alpha * range_val
        upper = b + alpha * range_val
        child1[i] = np.random.uniform(lower, upper)
        child2[i] = np.random.uniform(lower, upper)
    return child1, child2

def mutate(h, mutation_rate, dx, n_points):
    # Gaussian mutation
    noise = np.random.normal(0, mutation_rate * 0.1, n_points)
    h = h + noise
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h
