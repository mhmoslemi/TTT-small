import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more carefully designed initial point
        x = np.linspace(0, 2, n_points)
        # Create a piecewise constant with some variation
        h_start = np.zeros(n_points)
        h_start[:n_points//3] = 0.7
        h_start[n_points//3:2*n_points//3] = 0.3
        h_start[2*n_points//3:] = 0.6
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # EA parameters
    pop_size = 50
    generations = 200
    elite_size = 5
    
    # Initialize population
    population = []
    fitnesses = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Perturb initial solution
            noise = np.random.normal(0, 0.05, n_points)
            h = h_start + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness = evaluate_erdos_solution(h, dx)
        fitnesses.append(fitness)
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Selection probability based on fitness
    def tournament_selection():
        idxs = np.random.choice(pop_size, size=3, replace=False)
        best_of_tour = np.argmin([fitnesses[i] for i in idxs])
        return population[idxs[best_of_tour]]
    
    # Crossover of two parents
    def crossover(p1, p2):
        # Blend crossover
        alpha = np.random.rand()
        child = alpha * p1 + (1 - alpha) * p2
        return child
    
    # Mutation
    def mutate(h, generation, max_gen):
        # Adaptive mutation standard deviation
        mut_std = 0.1 * (1 - generation / max_gen)
        noise = np.random.normal(0, mut_std, n_points)
        h_m = h + noise
        h_m = np.clip(h_m, 0, 1)
        # Renormalize
        h_m = h_m * (n_points/2) / np.sum(h_m)
        return h_m
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            # Crossover
            child = crossover(parent1, parent2)
            # Mutation
            child = mutate(child, gen, generations)
            new_population.append(child)
        
        # Update population
        population = new_population[:pop_size]
        
        # Recalculate fitnesses
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final refinement with scipy minimize for top solutions
    top_candidates = []
    top_fitnesses = []
    for i in range(min(10, len(population))):
        h = population[i]
        # Run L-BFGS-B
        try:
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 50}
            )
            if res.success:
                h_refined = res.x
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                refined_fitness = evaluate_erdos_solution(h_refined, dx)
                top_candidates.append(h_refined)
                top_fitnesses.append(refined_fitness)
        except Exception:
            pass
    
    # Check all top candidates
    if top_candidates:
        best_refined_idx = np.argmin(top_fitnesses)
        best_h = top_candidates[best_refined_idx]
        best_c5 = top_fitnesses[best_refined_idx]
    
    # Final cleanup and verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
