import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create an initial guess with a structured approach
        h_start = np.ones(n_points) * 0.5
        # Slightly perturb to break symmetry
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Normalize the sum constraint
        h_norm = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h_norm, dx)
    
    # Define constraint for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Differential evolution optimization
    de_options = {
        'popsize': 50,
        'maxiter': 500,
        'mutation': (0.5, 1.0),
        'recombination': 0.9,
        'strategy': 'best1bin',
        'constraints': ({'type': 'eq', 'fun': constraint_sum},),
        'seed': seed
    }
    
    try:
        # Run differential evolution
        de_result = differential_evolution(
            objective, 
            bounds, 
            **de_options,
            callback=lambda x, convergence: None,
            maxiter=1000 if budget_s > 300 else 500
        )
        
        if de_result.success:
            h_de = de_result.x
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except Exception as e:
        pass
    
    # Local refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective_slsqp(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum_slsqp(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum_slsqp}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        res = minimize(
            objective_slsqp, 
            h_refined, 
            method='SLSQP', 
            bounds=bounds_slsqp, 
            constraints=cons_sum, 
            options={'maxiter': 1000, 'ftol': 1e-10}
        )
        
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective_slsqp(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception as e:
        pass
    
    # Final validation and cleanup
    try:
        # Ensure final values satisfy constraints
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        # Final C5 computation
        final_c5 = evaluate_erdos_solution(best_h, dx)
        best_c5 = final_c5
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
