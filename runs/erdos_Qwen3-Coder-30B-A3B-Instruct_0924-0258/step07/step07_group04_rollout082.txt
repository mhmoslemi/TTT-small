import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Create periodic initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        # Normalize to correct integral
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multiresolution refinement
    resolutions = [20, 40, 80]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to higher resolution
        if res > n_points:
            x_full = np.linspace(0, 2, res)
            x_coarse = np.linspace(0, 2, n_points)
            h_interp = np.interp(x_full, x_coarse, best_h)
            n_points = res
            dx = 2.0 / n_points
            h_start = h_interp
        else:
            h_start = best_h.copy()
            n_points = res
            dx = 2.0 / n_points
            
        # Local search with adaptive annealing
        current_h = h_start.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Annealing parameters
        temp = 0.5
        temp_decay = 0.995
        max_iter = 200 if res == 80 else 150 if res == 40 else 100
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Randomly select points to perturb
            num_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            
            # Perturb selected points
            h_candidate = current_h.copy()
            for i in indices:
                # Adaptive perturbation size
                pert_size = max(0.05, 0.15 * temp)
                delta = np.random.uniform(-pert_size, pert_size) * h_candidate[i]
                h_candidate[i] += delta
            
            # Enforce bounds and normalization
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
            
            # Evaluate new solution
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject based on Metropolis criterion
            if c5_candidate < current_c5 or np.random.random() < np.exp(-(c5_candidate - current_c5) / (temp + 1e-10)):
                current_h = h_candidate
                current_c5 = c5_candidate
            
            # Update global best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
            # Cool down temperature
            temp *= temp_decay
    
    # Final refinement with SLSQP
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
