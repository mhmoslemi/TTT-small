import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction if available

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Construct step function with two intervals
    # Based on strategy: h = 1 on [a, a+0.5] and [b, b+0.5] with a=0.65, b=0.85
    h = np.zeros(n_points)
    
    # Define positions in terms of indices
    a_pos = 0.65
    b_pos = 0.85
    interval_length = 0.5
    
    a_idx = int(a_pos / dx)
    b_idx = int(b_pos / dx)
    
    # Half interval length in indices
    half_length = int(interval_length / dx)
    
    # Create intervals [a, a+0.5] and [b, b+0.5] in terms of indices
    start1 = max(0, a_idx)
    end1 = min(n_points, start1 + half_length)
    start2 = max(0, b_idx)
    end2 = min(n_points, start2 + half_length)
    
    # Ensure total area is 1
    h[start1:end1] = 1.0
    h[start2:end2] = 1.0
    
    # Normalize to sum = n_points/2
    current_sum = np.sum(h)
    h = h * (n_points/2) / current_sum
    
    # Evaluate
    c5 = evaluate_erdos_solution(h, dx)
    
    return (list(h), c5, n_points)
