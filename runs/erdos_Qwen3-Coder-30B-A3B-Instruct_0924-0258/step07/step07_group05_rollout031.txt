import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    convex_interval = 50
    max_generations = 2000
    
    # Initialize population
    population = []
    fitness = []
    
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Genetic Algorithm Loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Tournament selection
        new_population = []
        new_fitness = []
        
        for _ in range(pop_size):
            # Tournament
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Clone winner
            parent = population[winner_idx].copy()
            
            # Crossover with another random parent
            if np.random.rand() < crossover_rate:
                other_parent_idx = np.random.choice(pop_size)
                other_parent = population[other_parent_idx].copy()
                
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent, other_parent)
            else:
                child = parent
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Mutate selected positions
                mutation_positions = np.random.choice(n_points, 
                    size=int(n_points * mutation_rate), replace=False)
                for pos in mutation_positions:
                    child[pos] += np.random.normal(0, 0.03)
                child = np.clip(child, 0, 1)
                # Rescale to meet constraint
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            c5 = evaluate_erdos_solution(child, dx)
            new_fitness.append(c5)
            
            if c5 < best_c5:
                best_h = child.copy()
                best_c5 = c5
        
        population = new_population
        fitness = new_fitness
        
        # Periodic convex relaxation
        if gen % convex_interval == 0 and gen > 0:
            # Select top 10% of population
            top_indices = np.argsort(fitness)[:max(1, pop_size // 10)]
            for idx in top_indices:
                try:
                    h_candidate = population[idx]
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add a few key constraints
                    k_vals = np.linspace(0.05, 1.95, min(15, n_points//3))
                    for k_val in k_vals:
                        k = int(k_val * n_points / 2)
                        if k < n_points and k > 0:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=300)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
                except Exception:
                    continue
    
    # Final local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
