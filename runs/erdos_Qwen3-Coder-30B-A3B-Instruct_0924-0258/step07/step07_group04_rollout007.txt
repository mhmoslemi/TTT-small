import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initialize with periodic function
        n_points = 20
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement at different resolutions
    resolutions = [20, 40, 80]
    iter_per_res = [100, 150, 200]
    
    for res_idx, (n_points, max_iter) in enumerate(zip(resolutions, iter_per_res)):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate solution from previous resolution
        if res_idx > 0:
            h_prev = best_h
            h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_prev)), h_prev)
            # Add small random noise
            noise_level = 0.05
            h_start += np.random.normal(0, noise_level, n_points)
            h_start = np.clip(h_start, 0, 1)
            # Normalize
            h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Neighborhood search parameters
        perturbation_size = 0.15 if res_idx == 0 else 0.1
        T = 0.1 if res_idx == 0 else 0.05
        T_decay = 0.99
        iter_count = 0
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Select random points to perturb
            n_perturb = max(1, int(0.2 * n_points))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            
            # Create candidate
            h_candidate = best_h.copy()
            for i in indices:
                delta = np.random.uniform(-perturbation_size, perturbation_size) * h_candidate[i]
                h_candidate[i] += delta
            
            # Enforce bounds
            h_candidate = np.clip(h_candidate, 0, 1)
            # Normalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate candidate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept or reject (simulated annealing style)
            if c5_candidate < best_c5 or np.random.random() < np.exp(-(c5_candidate - best_c5) / T):
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
            
            iter_count += 1
            if iter_count % 50 == 0:
                T *= T_decay
                perturbation_size *= 0.95
                
            if best_c5 < 0.3807:  # Early stopping if good enough
                break
    
    # Final refinement with SLSQP
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        if time() - start_time < budget_s - 30:
            res = minimize(objective, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                best_c5 = objective(h_final)
        
        # Ensure final normalization
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        best_c5 = evaluate_erdos_solution(h_final, dx)
        best_h = h_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
