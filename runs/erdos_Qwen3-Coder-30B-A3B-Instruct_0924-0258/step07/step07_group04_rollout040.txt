import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initialization that tends to work well
        h_start = np.zeros(n_points)
        # Use a symmetric, piecewise constant shape with some variation
        mid = n_points // 2
        h_start[:mid] = 0.5
        h_start[mid:] = 0.3
        # Add small random variation to break symmetry
        h_start += np.random.normal(0, 0.02, n_points)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize sum constraint: sum(h) = n_points / 2
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define bounds for differential evolution
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints for the optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Constraint for SLSQP
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Simple Differential Evolution optimization for global search
    # This stage uses a relatively small population and fewer iterations to be fast
    # Use time tracking to budget the first phase
    de_start = time()
    de_maxiter = 100
    de_popsize = 50
    
    # Adaptive DE parameters
    def de_objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    try:
        result = differential_evolution(
            de_objective,
            bounds,
            maxiter=de_maxiter,
            popsize=de_popsize,
            seed=seed,
            strategy='best1bin',
            disp=False,
            tol=1e-6
        )
        
        if not result.success:
            # Fallback to the original h_start if DE fails
            pass
        else:
            h_de = result.x
            h_de = np.clip(h_de, 0, 1)
            h_de = h_de * (n_points/2) / np.sum(h_de)
            c5_de = evaluate_erdos_solution(h_de, dx)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
                
    except:
        # If DE fails, keep the starting point
        pass
    
    # Local refinement with SLSQP if time allows
    # We use a more sophisticated optimization approach here
    if time() - start_time > budget_s * 0.8:
        # Stop early if we're running out of time
        return (list(best_h), best_c5, n_points)
    
    try:
        # Refine with SLSQP
        # Ensure we're working with a feasible point
        h_refined = np.clip(best_h, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = objective(h_final)
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
