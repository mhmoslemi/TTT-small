import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient descent approach
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    patience = 50
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Calculate all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # If we improve, keep going
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            # Early stopping if no improvement for a while
            break
            
        # Compute gradient for the maximum overlap lag
        # Lag is in the middle of the full correlation
        lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient of the overlap with respect to h
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i + lag < len(h):
                grad[i] = 2 * (1 - h[i + lag] - h[i]) * dx
        
        # Update h using gradient descent
        h_new = h - step_size * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy constraint sum(h) * dx = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check convergence
        if np.linalg.norm(h_new - h) < tolerance:
            break
            
        h = h_new
        
    # Final validation and refinement
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Try local optimization on the best solution found
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-9})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < final_c5:
                final_c5 = c5_refined
                best_h = h_refined
    except Exception:
        pass
    
    return (list(best_h), final_c5, n_points)
