import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a simple construction
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.4
        h_start[1] = 0.6
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm approach
    pop_size = 100
    generations = 300
    elite_size = pop_size // 10
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def mutate(h, generation):
        # Adaptive mutation
        sigma = 0.1 * (0.9 ** (generation // 50))
        if sigma < 1e-5:
            sigma = 1e-5
        
        mutated = h + np.random.normal(0, sigma, n_points)
        return repair(mutated)
    
    def crossover(h1, h2):
        alpha = np.random.uniform(0.2, 0.8)
        child = alpha * h1 + (1 - alpha) * h2
        return repair(child)
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        noise = np.random.uniform(-0.1, 0.1, n_points)
        h = best_h + noise
        h = repair(h)
        population.append(h)
    
    # Hall of fame
    hall_of_fame = []
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update hall of fame
        for i in range(min(elite_size, len(sorted_population))):
            if len(hall_of_fame) < elite_size or sorted_fitness[i] < hall_of_fame[-1][0]:
                hall_of_fame.append((sorted_fitness[i], sorted_population[i].copy()))
                hall_of_fame.sort(key=lambda x: x[0])
                if len(hall_of_fame) > elite_size:
                    hall_of_fame.pop()
        
        # Check for improvement
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Selection (tournament)
        new_population = []
        tournament_size = 3
        
        # Keep elite
        new_population.extend([h.copy() for _, h in hall_of_fame])
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection for parents
            parent1_idx = np.random.choice(len(sorted_population), tournament_size)
            parent1 = sorted_population[np.argmin([sorted_fitness[i] for i in parent1_idx])]
            
            parent2_idx = np.random.choice(len(sorted_population), tournament_size)
            parent2 = sorted_population[np.argmin([sorted_fitness[i] for i in parent2_idx])]
            
            # Crossover
            if np.random.rand() < 0.7 - 0.4 * (gen / generations):
                child = crossover(parent1, parent2)
            else:
                # Mutation
                child = mutate(parent1, gen)
            
            new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final refinement with numerical optimization
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1,
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
