import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    # Use a periodic function as starting point
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 60
        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    T = 1.0  # Initial temperature
    alpha = 0.998  # Cooling rate
    tabu_size = 20
    tabu_list = []
    
    # Parameters for perturbation
    max_perturb = 0.15
    min_perturb = 0.05
    
    iteration = 0
    stagnation_count = 0
    max_stagnation = 200
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Generate candidate solution
        candidate_h = current_h.copy()
        
        # Select random indices to perturb (10-20% of points)
        n_perturb = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb selected indices
        for i in indices:
            perturb = np.random.uniform(min_perturb, max_perturb)
            if np.random.random() < 0.5:
                perturb = -perturb
            candidate_h[i] += perturb * (1 - 2 * np.random.random())
            
        # Clip and renormalize
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check if candidate is in tabu list
        is_tabu = False
        for tabu_h in tabu_list:
            if np.allclose(candidate_h, tabu_h, atol=1e-6):
                is_tabu = True
                break
        
        # Accept or reject based on SA criteria
        accept = False
        if candidate_c5 < current_c5:
            accept = True
        elif not is_tabu and np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            accept = True
            
        if accept:
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(candidate_h.copy())
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cooling schedule
        T *= alpha
        
        # Stagnation check
        if stagnation_count > max_stagnation:
            # Reset by reinitializing
            current_h = initialize_h(n_points)
            current_h = current_h * (n_points/2) / np.sum(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            stagnation_count = 0
            T = 1.0  # Reset temperature
            
    # Final refinement using SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 1000, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
