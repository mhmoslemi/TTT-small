import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + max_lag < n_points:
                grad[i] = 1 - 2 * h_current[i] - h_current[i + max_lag] if max_lag >= 0 else 1 - 2 * h_current[i] - h_current[i - max_lag]
        
        # Gradient descent step
        h_new = h_current - step_size * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check constraint satisfaction
        if np.abs(np.sum(h_new) - n_points/2) < tolerance:
            h_current = h_new
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
        else:
            # Constraint violation, reduce step size
            step_size *= 0.5
            if step_size < 1e-10:
                break
    
    # Final refinement with constrained optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
