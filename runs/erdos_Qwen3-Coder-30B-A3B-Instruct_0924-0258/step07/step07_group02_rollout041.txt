import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a piecewise constant function
        h_start = np.ones(n_points) * 0.5
        h_start[:n_points//3] = 1.0
        h_start[2*n_points//3:] = 0.0
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert to cosine basis representation
    n_terms = 12  # Number of cosine terms to use
    x = np.linspace(0, 2, n_points)
    # Precompute cosine basis functions
    Phi = np.array([[np.cos(np.pi * k * xi) for xi in x] for k in range(n_terms)])
    
    # Initialize coefficients
    c_init = np.random.rand(n_terms) * 0.5
    
    # Try convex optimization approach to improve the solution
    # We will use a grid search + gradient-based optimization
    
    def project_to_feasible(c_coeffs):
        h_val = np.dot(Phi, c_coeffs)
        h_val = np.clip(h_val, 0, 1)
        h_val = h_val * (n_points/2) / np.sum(h_val)
        return h_val
    
    # Objective function to minimize
    def objective(c_coeffs):
        h_val = project_to_feasible(c_coeffs)
        return evaluate_erdos_solution(h_val, dx)
    
    # Make a few concentrated attempts with different starting points
    max_attempts = 10
    for attempt in range(max_attempts):
        if time() - start_time > budget_s * 0.95:
            break
            
        np.random.seed(seed + attempt * 17)  # Change seed for diversity
        c_trial = np.random.rand(n_terms) * 1.0 + 0.1
        
        # Try optimization approaches
        try:
            # Try gradient descent approach with bounds
            from scipy.optimize import minimize as scimin
            
            def obj(c):
                h_val = project_to_feasible(c)
                return evaluate_erdos_solution(h_val, dx)
            
            # Constrained optimization
            bounds = [(0, 10) for _ in range(n_terms)]
            res = scimin(obj, c_trial, bounds=bounds, method='L-BFGS-B', 
                         options={'maxiter': 2000, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if res.success:
                c_opt = res.x
                h_test = project_to_feasible(c_opt)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
                    
        except Exception:
            pass
    
    # Try alternating directions with synthetic information to refine
    # Perform steepest descent on the projected solution
    c_best = np.random.rand(n_terms) * 0.5
    best_h = project_to_feasible(c_best)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # More focused coordinate descent
    for i in range(50):
        if time() - start_time > budget_s * 0.95:
            break
        for k in range(n_terms):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Try a small perturbation in one dimension
            c_trial = c_best.copy()
            step = 0.1 * (np.random.rand() - 0.5)
            c_trial[k] += step
            
            h_trial = project_to_feasible(c_trial)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                c_best = c_trial.copy()
    
    # Final refinement through a local search
    try:
        # Reconstruct h from basis coefficients with tightened variable bounds
        # Using CVXPY for final constrained optimization if time allows
        if time() - start_time < budget_s * 0.9:
            # Create variables in a convex program
            c_vars = cp.Variable(n_terms)
            h_vars = Phi @ c_vars
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Dummy objective since we just want feasibility
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, max_iters=500, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                c_final = c_vars.value
                h_final = project_to_feasible(c_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
    except Exception:
        pass

    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
