import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis function construction with triangular pulses
    def create_basis(n, m):
        # Create m triangular basis functions with compact support
        basis = np.zeros((n, m))
        x = np.linspace(0, 2, n)
        for i in range(m):
            center = i * 2.0 / (m - 1) if m > 1 else 1.0
            width = 0.5
            for j in range(n):
                dist = abs(x[j] - center)
                basis[j, i] = max(0, 1 - dist / width)
            # Normalize
            basis[:, i] /= np.sum(basis[:, i]) * n_points / 2
        return basis
    
    # Initial basis
    m = 10
    basis = create_basis(n_points, m)
    
    # Current coefficients
    coeffs = np.linalg.lstsq(basis, best_h, rcond=None)[0]
    current_h = basis @ coeffs
    
    def basis_objective(coeffs):
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h, dx)
    
    def basis_constraint(coeffs):
        h = basis @ coeffs
        return np.sum(h) - n_points / 2
    
    # Trust-region optimization
    max_iter = 5000
    trust_radius = 0.1
    patience = 0
    max_patience = 50
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_h = basis @ coeffs
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            # Perturb coefficients
            coeffs += np.random.normal(0, 0.05, m)
            patience = 0
            
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros(m)
        for i in range(m):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = basis @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = basis @ coeffs_minus
            h_minus = np.clip(h_minus, 0, 1)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update with trust region step
        try:
            step = -grad * trust_radius / np.linalg.norm(grad)
            new_coeffs = coeffs + step
            
            # Keep step within bounds
            new_h = basis @ new_coeffs
            if np.all(new_h >= 0) and np.all(new_h <= 1):
                new_coefficients = new_coeffs
                new_c5 = basis_objective(new_coefficients)
                
                if new_c5 < current_c5:
                    coeffs = new_coefficients
                    trust_radius *= 2.0
                else:
                    trust_radius *= 0.5
            else:
                # Reject step
                trust_radius *= 0.5
                
        except:
            trust_radius *= 0.5

    # Final refinement with local optimization
    try:
        def objective(coeffs):
            h = basis @ coeffs
            h = np.clip(h, 0, 1)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(coeffs):
            h = basis @ coeffs
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(None, None) for _ in range(m)]
        res = minimize(objective, coeffs, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            final_coeffs = res.x
            final_h = basis @ final_coeffs
            final_h = np.clip(final_h, 0, 1)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
