import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize a, s for alternating optimization
    a = np.ones(n_points) * 0.5  # amplitudes
    s = np.ones(n_points)        # support (0 or 1)
    
    # Set s to initial values or a random configuration
    if 'initial_h_values' in globals():
        s = (initial_h_values > 1e-6).astype(float)
    else:
        # Initialize s evenly with 1/2 of points active
        k_active = n_points // 2
        s = np.zeros(n_points)
        active_indices = np.random.choice(n_points, k_active, replace=False)
        s[active_indices] = 1.0
    
    # Ensure correct support sum
    s_sum = np.sum(s)
    if s_sum > 0:
        a = a * (n_points / 2) / s_sum
    
    h = a * s
    current_c5 = evaluate_erdos_solution(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Alternating optimization
    max_iter = 1000
    stagnation_count = 0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fix s, optimize a using CVXPY or fallback SLSQP
        a_optimized = a.copy()
        try:
            # Convex Optimization for fixed s
            a_var = cp.Variable(n_points)
            h_var = cp.multiply(a_var, s)
            # Simplified estimation via quadratic approximation
            # Compute correlation for fixed s and find a that minimizes max overlap via optimization
            # Use direct gradient descent to solve this subproblem
            def opt_objective(a_val):
                h_temp = a_val * s
                return evaluate_erdos_solution(h_temp, dx)
            
            def opt_constraint(a_val):
                return np.sum(a_val * s) - n_points/2
            
            cons = {'type': 'eq', 'fun': opt_constraint}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(opt_objective, a.copy(), method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                a_optimized = res.x
            else:
                # fallback
                a_optimized = a.copy()
        except:
            a_optimized = a.copy()
        
        # Ensure sum(a * s) = n_points / 2
        old_sum = np.sum(a_optimized * s)
        if old_sum > 0:
            a_optimized = a_optimized * (n_points / 2) / old_sum
        
        # Update a
        a = a_optimized
        
        # Fix a, optimize s by top-k selection
        s_old = s.copy()
        k_active = int(n_points / 2)
        topk_indices = np.argsort(a)[::-1][:k_active]  # Indices of top-k a values
        
        s_new = np.zeros(n_points)
        s_new[topk_indices] = 1.0
        
        # Repair sum
        s_sum = np.sum(s_new)
        if s_sum > 0:
            factor = (n_points / 2) / s_sum
            s_new = s_new * factor
        
        # Update s
        s = s_new
        
        # Update h
        h = a * s
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    h_refined = best_h.copy()
    h_refined = np.clip(h_refined, 0, 1)
    h_refined_sum = np.sum(h_refined)
    if h_refined_sum > 0:
        h_refined = h_refined * (n_points / 2) / h_refined_sum
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_refined, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
