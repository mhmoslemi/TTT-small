import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Construct a step function with two equal intervals
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two intervals of length 0.5 each
        # Center them to maximize distance and minimize overlap
        a = 0.65
        b = 0.85
        a_idx = int(a / dx)
        b_idx = int(b / dx)
        interval_length = int(0.5 / dx)
        
        # Set first interval [a, a + 0.5]
        end_a = min(a_idx + interval_length, n_points)
        h_start[a_idx:end_a] = 1.0
        
        # Set second interval [b, b + 0.5]  
        end_b = min(b_idx + interval_length, n_points)
        h_start[b_idx:end_b] = 1.0
        
        # Normalize to integral = 1
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try local search around this construction
    # Perform gradient-based optimization with bounds
    from scipy.optimize import minimize
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try refinement with different optimizers
    h_refined = best_h.copy()
    
    try:
        # Try L-BFGS-B first
        res = minimize(objective, h_refined, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 200})
        if res.success:
            h_candidate = res.x
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
    except Exception as e:
        pass
    
    # Try multiple random restarts
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the current best
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='L-BFGS-B',

                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception as e:
            continue
    
    return (list(best_h), best_c5, n_points)
