import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Basis functions approach
    m = 10  # number of basis functions
    # Initialize basis functions as triangular pulses
    x = np.linspace(0, 2, n_points)
    
    # Create basis functions
    basis = []
    for i in range(m):
        # triangular basis with support [i*2/m, (i+1)*2/m] 
        pos = i * 2.0 / m
        width = 2.0 / m
        basis_func = np.maximum(0, 1 - np.abs((x - pos - width/2) / (width/2)))
        basis_func = basis_func / np.max(basis_func) if np.max(basis_func) > 0 else basis_func
        basis.append(basis_func)
    basis = np.array(basis).T  # shape (n_points, m)
    
    # Initialize coefficients
    coeffs = np.random.rand(m) * 0.5
    h_current = basis @ coeffs
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    # Optimization using trust-region
    max_iter = 5000
    patience = 0
    max_patience = 100
    perturbation_magnitude = 0.05
    trust_radius = 0.1
    
    def obj_and_grad(coeffs):
        # Compute h from coeffs
        h = basis @ coeffs
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        cost = evaluate_erdos_solution(h, dx)
        # Estimate gradient by finite differences
        eps = 1e-6
        grad = np.zeros(m)
        for i in range(m):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = basis @ coeffs_plus
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            cost_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (cost_plus - cost) / eps
        return cost, grad
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current loss and gradient
        try:
            cost, grad = obj_and_grad(coeffs)
            if cost < best_c5:
                best_c5 = cost
                best_h = basis @ coeffs
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points/2) / np.sum(best_h)
                
            # Update using trust-region method (simplified)
            # Compute step using gradient
            step = -grad * trust_radius
            new_coeffs = coeffs + step
            
            # Ensure feasibility and re-scale
            h_new = basis @ new_coeffs
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_cost = evaluate_erdos_solution(h_new, dx)
            
            if new_cost < cost:
                coeffs = new_coeffs.copy()
                trust_radius = min(2.0 * trust_radius, 1.0)
                patience = 0
            else:
                trust_radius = max(trust_radius / 2, 1e-6)
                patience += 1
                
            if patience > max_patience:
                # Perturb coefficients
                coeffs += np.random.normal(0, perturbation_magnitude, m)
                patience = 0
                perturbation_magnitude *= 0.5
        except Exception:
            # Fallback
            if iter_num % 100 == 0:
                coeffs = np.random.rand(m) * 0.5
                
    # Final refinement with local optimization
    try:
        def objective(c):
            h = basis @ c
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
        
        cons = [{'type': 'eq', 'fun': lambda c: np.sum(basis @ c) - n_points / 2.0}]
        bounds = [(0, 1) for _ in range(m)]
        res = minimize(objective, coeffs, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = basis @ res.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
