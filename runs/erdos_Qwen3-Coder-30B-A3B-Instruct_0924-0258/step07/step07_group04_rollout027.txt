import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_h(n_points):
    x = np.linspace(0, 2, n_points)
    h_initial = 0.5 * (1 + np.sin(2 * np.pi * x))
    h_initial = np.clip(h_initial, 0, 1)
    normalization = (n_points / 2) / np.sum(h_initial)
    h_initial *= normalization
    return h_initial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

        n_points = len(h_start)
    else:

        h_start = initialize_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search with simulated annealing parameters
    tabu_list = []
    tabu_size = 20
    T = 1.0
    alpha = 0.998
    max_iter = 10000
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = current_h.copy()
        num_changes = max(1, int(0.15 * n_points))
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        
        # Perturb selected indices
        for i in indices:
            perturbation = np.random.uniform(-0.15, 0.15)
            candidate_h[i] += perturbation
            candidate_h[i] = np.clip(candidate_h[i], 0, 1)
        
        # Renormalize to meet constraint
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check if candidate is tabu
        tabu = False
        for prev_h in tabu_list:
            if np.allclose(prev_h, candidate_h, atol=1e-5):
                tabu = True
                break
                
        # Accept or reject based on SA criteria
        if candidate_c5 < current_c5:
            # Accept better solution
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                last_improvement = iteration
        elif not tabu and np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            # Accept worse solution with probability
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
        # Update tabu list
        tabu_list.append(current_h.copy())
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
            
        # Cooling schedule
        T *= alpha
        
        # Early stopping if no improvement for long
        if iteration - last_improvement > 500:
            stagnation_count += 1
            if stagnation_count > 50:  # Stop if stagnated too long
                break
    
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
