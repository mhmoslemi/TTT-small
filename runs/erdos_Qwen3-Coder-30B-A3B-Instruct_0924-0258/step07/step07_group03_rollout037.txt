import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    
    # Initial solution
    h0 = h_start.copy()
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (n_points/2) / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, dx)
    
    # Convex relaxation approach
    # Introduce a matrix variable Z = h*h^T
    # We want to minimize max_k \int h(x)(1-h(x+k)) dx
    
    # Let's use a more direct approach with NLP
    # Try gradient-based optimization with constraints
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Try to refine with different algorithms
    # Starting with a more robust approach
    n_iter = 0
    max_iter = 2000
    
    def solve_with_method(method='SLSQP'):
        h_current = h0.copy()
        try:
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_current, method=method,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                return h_refined, c5_refined
        except Exception:
            pass
        return h_current, evaluate_erdos_solution(h_current, dx)

        except Exception:
            continue
    
    # Additional refinement: local search from multiple points
    for i in range(10):

        # Random start near the best solution
        h_rand = best_h + np.random.normal(0, 0.05, n_points)

        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        
        try:
            cons = {'type': 'eq', 'fun': constraint_sum}

    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
