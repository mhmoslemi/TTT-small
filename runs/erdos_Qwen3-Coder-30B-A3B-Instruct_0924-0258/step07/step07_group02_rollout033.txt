import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point based on known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 100
    max_generations = 300
    elite_size = pop_size // 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = []
    fitness_scores = []
    
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Initialize with random feasible solutions
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness_scores.append(evaluate_erdos_solution(h, dx))
    
    # Track best solutions
    hall_of_fame = list(zip(fitness_scores, population))
    hall_of_fame.sort()
    best_c5 = hall_of_fame[0][0]
    best_h = hall_of_fame[0][1].copy()
    
    # EA loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Keep elite
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        new_fitness = [fitness_scores[i] for i in range(elite_size)]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[np.random.randint(0, pop_size//2)]
            parent2 = population[np.random.randint(0, pop_size//2)]
            
            # Crossover
            if np.random.random() < crossover_rate:
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Add noise to 20% of elements
                indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                noise = np.random.normal(0, 0.1, len(indices))
                child[indices] += noise
                
            # Repair and ensure feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))
        
        population = new_population
        fitness_scores = new_fitness
        
        # Update hall of fame
        for i in range(pop_size):
            if fitness_scores[i] < hall_of_fame[0][0]:
                hall_of_fame.append((fitness_scores[i], population[i].copy()))
                hall_of_fame.sort()
                hall_of_fame = hall_of_fame[:elite_size]
                if fitness_scores[i] < best_c5:
                    best_c5 = fitness_scores[i]
                    best_h = population[i].copy()
    
    # Local refinement with L-BFGS
    try:
        # Refine top solutions from hall of fame
        for h_refine in hall_of_fame[:5]:
            h_init = h_refine[1]
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points/2
            
            def constraint_bounds(h):
                return np.concatenate([h, 1 - h])
            
            # Use scipy minimize with bounds and constraint
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            result = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 200})
            
            if result.success:
                h_feasible = result.x
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
