import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization
    max_iter = 10000
    step_size = 0.01
    tolerance = 1e-9
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(-len(h)+1, len(h))
        max_overlap = np.max(corr)
        l_max = lags[np.argmax(corr)]
        
        # Compute gradient for the most constraining lag
        # For lag l_max, gradient w.r.t. h[i] is (1 - h[i+l_max]) if l_max >= 0
        # or h[i-l_max] if l_max < 0
        grad = np.zeros(n_points)
        if l_max >= 0:
            for i in range(n_points):
                j = i + l_max
                if j < n_points:
                    grad[i] = 1 - h[j]
        else:
            for i in range(n_points):
                j = i - l_max
                if j < n_points:
                    grad[i] = h[j]
        
        # Update h
        h_new = h - step_size * grad
        
        # Project to feasible region: clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check for improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update h
        h = h_new
        
        iter_count += 1
        
        # Adaptive step size
        if abs(c5_new - best_c5) < tolerance:
            step_size *= 0.5
            if step_size < 1e-6:
                break
    
    # Final local refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        from scipy.optimize import minimize
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try several random starting points for refinement
        best_refined = best_h.copy()
        best_refined_c5 = best_c5
        
        for _ in range(5):
            h_init = best_h + np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            
            res = minimize(objective, h_init, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-12})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_refined_c5:
                    best_refined = h_refined
                    best_refined_c5 = c5_refined
                    
        best_h = best_refined
        best_c5 = best_refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
