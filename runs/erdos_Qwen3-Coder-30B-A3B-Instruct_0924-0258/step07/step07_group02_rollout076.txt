import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 50
    generations = 200
    elite_size = 5
    mutation_rate = 0.05
    crossover_rate = 0.7
    mutation_decay = 0.95
    time_check_interval = 10
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for i in range(pop_size):
        if i == 0:
            h = best_h.copy()
        else:
            # Random perturbation around the best solution
            noise = np.random.normal(0, mutation_rate, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # Main EA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        if gen % time_check_interval == 0:
            # Periodic check
            best_idx = np.argmin(fitness)
            if fitness[best_idx] < best_c5:
                best_h = population[best_idx].copy()
                best_c5 = fitness[best_idx]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        remaining_indices = sorted_indices[elite_size:]
        
        # Create new population
        new_population = [population[i] for i in elite_indices]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1, parent2 = None, None
            for _ in range(5):
                idx1, idx2 = np.random.choice(remaining_indices, 2, replace=False)
                if fitness[idx1] < fitness[idx2]:
                    parent1 = population[idx1]
                else:
                    parent1 = population[idx2]
                break
            
            for _ in range(5):
                idx1, idx2 = np.random.choice(remaining_indices, 2, replace=False)
                if fitness[idx1] < fitness[idx2]:
                    parent2 = population[idx1]
                else:
                    parent2 = population[idx2]
                break
                
            # Crossover
            if np.random.rand() < crossover_rate and parent1 is not None and parent2 is not None:
                # Blend crossover
                alpha = np.random.rand(n_points)
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Mutation
                child = parent1 if parent1 is not None else population[np.random.choice(remaining_indices)]
                noise = np.random.normal(0, mutation_rate, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Repair
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        # Update population and fitness
        population = new_population[:pop_size]
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update global best
        for i, h in enumerate(population):
            if fitness[i] < best_c5:
                best_h = h.copy()
                best_c5 = fitness[i]
        
        # Adapt mutation rate
        mutation_rate *= mutation_decay
    
    # Final SQP refinement for top solutions
    top_indices = np.argsort(fitness)[:10]
    for idx in top_indices:
        h_test = population[idx]
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            continue
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
