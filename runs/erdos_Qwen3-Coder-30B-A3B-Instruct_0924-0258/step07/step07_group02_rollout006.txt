import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct a better starting point
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    pop_size = 50
    max_iter = 200
    mutation_factor = 0.8
    crossover_prob = 0.7
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Add perturbation
            noise = np.random.normal(0, 0.1, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(pop_size):
            # DE mutation
            idxs = [idx for idx in range(pop_size) if idx != i]
            a, b, c = population[np.random.choice(idxs, 3, replace=False)]
            mutant = a + mutation_factor * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points/2) / np.sum(mutant)
            
            # DE crossover
            trial = []
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            trial = np.array(trial)
            
            # Evaluate trial
            c5_trial = evaluate_erdos_solution(trial, dx)
            c5_current = evaluate_erdos_solution(population[i], dx)
            
            # Selection
            if c5_trial < c5_current:
                population[i] = trial
                if c5_trial < best_c5:
                    best_h = trial.copy()
                    best_c5 = c5_trial
    
    # Final SQP refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass

    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
