import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a different optimization approach: gradient-based refinement
    max_iter = 1000
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Initial parameters for optimization
    tolerance = 1e-8
    learning_rate = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient ascent on overlap integral
        # We'll use finite differences to estimate gradients
        gradient = np.zeros_like(current_h)
        epsilon = 1e-5
        
        for i in range(n_points):
            # Perturb h[i]
            h_plus = current_h.copy()
            h_minus = current_h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            
            # Ensure values are in [0,1]
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Normalize the perturbed vectors
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            
            # Compute overlap
            overlap_plus = evaluate_erdos_solution(h_plus, dx)
            overlap_minus = evaluate_erdos_solution(h_minus, dx)
            
            # Gradient approximation
            gradient[i] = (overlap_plus - overlap_minus) / (2 * epsilon)
        
        # Update step with clipping
        current_h = current_h - learning_rate * gradient
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points/2) / np.sum(current_h)
        
        # Evaluate updated value
        new_c5 = evaluate_erdos_solution(current_h, dx)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            best_h = current_h.copy()
            
            # Early stopping if improvement is small
            if (new_c5 <= 0.3807):
                break
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.95
        
        # Verify our solution still meets constraints
        assert 0 <= np.min(current_h) and np.max(current_h) <= 1
        assert abs(np.sum(current_h) - n_points / 2) < 1e-10
    
    # Final refinement using scipy minimize with bounds
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation that our solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
