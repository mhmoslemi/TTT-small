import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based refinement with adaptive learning rate
    h_current = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute correlation and find worst lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_lag_idx = np.argmax(corr)
        current_c5 = corr[max_lag_idx] * dx
        
        # Compute gradient contributions for worst lag
        # This is the derivative of the overlap integral w.r.t. h
        # For lag k, contribution to h[i] is 1 - 2*h[i+k] if i+k < n_points
        # But we need to be more careful about indexing
        gradient = np.zeros(n_points) 
        # Compute the actual contribution from the worst lag
        # At lag k, for h[i], contribution to overlap is h[i]*(1-h[i+k])
        # So gradient w.r.t. h[i] = 1 - 2*h[i+k] for i+k < n_points
        # And gradient w.r.t. h[i+k] = -h[i]
        # We don't compute full gradient, but the quantity needed for update
        
        # Derivative idea: if overlap is large, we want to decrease h where 1-h[i+k] is large
        # or increase h where h[i+k] is large. Simplification: based on shortest correlation
        # Compute for the worst lag specifically
        
        # This is an approximation: move along the gradient of the maximum overlap term
        # But we'll use a simple approximation: 
        # approx grad[i] = (1 - 2*h[i]) * around_lag_contrib
        # We'll compute a crude approximation via finite difference
        
        # Iterating through indices for contributions
        grad_approx = np.zeros(n_points)
        for i in range(n_points):
            # Find the index corresponding to max_lag_idx - this is the 'k'
            # Mu is the position in the correlation - correlation is of length 2*n_points-1
            # We can compute the corresponding actual lags
            # Lag k = max_lag_idx - (n_points - 1)
            lag_pos = max_lag_idx - (n_points - 1)
            # Gradient contribution: for h[i], let's check how a small change affects the
            # term involving h[i] and h[i+lag_pos] (or i-lag_pos) 
            # Let's compute variance-based gradient:
            # If we change h[i] by delta, we affect term: h[i]*(1 - h[i+lag_pos])  
            # Gradient estimate is (1 - 2*h[i+lag_pos]) if i+lag_pos < n_points
            # Or (1 - 2*h[i-lag_pos]) if i >= lag_pos
            
            if lag_pos >= 0 and i + lag_pos < n_points:
                grad_approx[i] += 1 - 2 * h_current[i + lag_pos]
            elif lag_pos < 0 and i >= -lag_pos:
                grad_approx[i] += 1 - 2 * h_current[i - lag_pos]
        
        # Update h with gradient approximation
        h_current = h_current + learning_rate * grad_approx
        # Project to [0,1] box
        h_current = np.clip(h_current, 0, 1)
        # Adjust to meet sum constraint
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Adaptive learning rate
        new_c5 = evaluate_erdos_solution(h_current, dx)
        if new_c5 < current_c5:
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 20:
                learning_rate *= 0.9
                stagnation_count = 0
                
        # Periodic perturbation
        if iteration % 50 == 0 and iteration > 0:
            perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            noise = np.random.uniform(-0.1, 0.1, len(perturb_indices))
            h_current[perturb_indices] += noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Phase 2: SQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
