import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create sine wave initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize to correct integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with periodic perturbations
    max_iter = 2000
    gamma = 0.1  # learning rate
    perturb_freq = 50
    perturb_strength = 0.05
    
    h = h_start.copy()
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Get the maximum correlation index
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        
        # Compute the lag (in samples) for the maximum overlap
        lag = max_idx - (len(h) - 1)
        
        # Create direction step: aim to reduce max correlation
        # by moving in direction that reduces the overlap
        direction = np.zeros_like(h)
        if lag >= 0:
            # Positive lag: 1-h shifted forward
            direction[lag:] += 1
            direction[:lag] = 0
        else:
            # Negative lag: 1-h shifted backward
            direction[:abs(lag)] += 1
            direction[abs(lag):] = 0
            
        # Apply direction (Frank-Wolfe update)
        h_new = h - gamma * direction
        h_new = np.clip(h_new, 0, 1)
        
        # Project back to sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check if we're stagnating
        if stagnation_count > 100:
            # Perturb
            h_perturbed = h + np.random.normal(0, perturb_strength, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                h = h_perturbed.copy()
            else:
                h = h_new.copy()
        else:
            h = h_new.copy()
            
        # Periodic perturbation
        if iteration % perturb_freq == 0 and iteration > 0:
            h = h + np.random.normal(0, perturb_strength, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
