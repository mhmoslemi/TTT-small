import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Initialize history
    prev_c5 = float('inf')
    iteration = 0
    
    # Main optimization loop
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Compute full overlap correlations
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr_full)
        k_worst = max_idx - (n_points - 1)  # Convert from full index to shift k
        
        # Compute overlap for this shift using direct calculation
        if k_worst >= 0:
            overlap = np.sum(best_h[:-k_worst] * (1 - best_h[k_worst:])) * dx
        else:
            k_abs = -k_worst
            overlap = np.sum(best_h[k_abs:] * (1 - best_h[:-k_abs])) * dx
        
        # Add constraint to specify this overlap is bounded by current best
        # For now we simply improve by gradient-based local search
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
        
        # Perturb slightly to escape local minima
        if iteration % 10 == 0:
            perturbation = perturbation_mag * np.random.randn(n_points)
            best_h += perturbation
            best_h = np.clip(best_h, 0, 1)
            # Rescale
            best_h = best_h * (n_points/2) / np.sum(best_h)
            new_c5 = evaluate_erdos_solution(best_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
        
        # Reduce perturbation magnitude slightly
        perturbation_mag *= perturbation_decay
        
        # Check convergence
        if abs(prev_c5 - best_c5) < convergence_tol:
            break
        prev_c5 = best_c5
        iteration += 1
    
    # Final polishing
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
