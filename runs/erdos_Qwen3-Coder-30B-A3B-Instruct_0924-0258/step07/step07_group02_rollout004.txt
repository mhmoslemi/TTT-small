import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a diversified starting point using smooth oscillations
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    initial_c5 = best_c5
    
    # Evolutionary Algorithm with adaptive mutation and crossover
    pop_size = 50
    max_generations = 200
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Add perturbations
            noise = np.random.normal(0, 0.05, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evolutionary process
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population and sort
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select top 30% for breeding
        elite_size = max(1, pop_size // 3)
        elite = population[:elite_size]
        elite_fitness = fitness[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = [elite[0].copy(), elite[1].copy()]  # Keep two best
        
        # Crossover: blend between elite members
        while len(new_population) < pop_size:
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            
            # Blend with alpha
            alpha = np.random.rand()
            child = alpha * elite[parent1_idx] + (1 - alpha) * elite[parent2_idx]
            
            # Add Gaussian mutation
            mutation_strength = 0.05 * (1 - generation/max_generations)
            noise = np.random.normal(0, mutation_strength, n_points)
            child += noise
            
            # Repair for constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
    
    # Final L-BFGS-B refinement of top individuals
    top_candidates = population[:10]
    for h_candidate in top_candidates:
        try:
            # Set up optimization problem for this candidate
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Use a slightly relaxed objective with feasibility restoration
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
