import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 200
    mutation_rate = 0.3
    elite_size = int(0.2 * population_size)
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    best_generation = 0
    
    # Start GA loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness (lower C5 is better)
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            best_generation = generation
        
        # Early stopping if no improvement
        if generation - best_generation > 50:
            break
            
        # Select elite (top 20%)
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite[:]
        
        while len(new_population) < population_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament_size = 5
                tournament_indices = np.random.choice(len(population), tournament_size)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_index = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(population[winner_index])
            
            # Arithmetic crossover
            alpha = 0.5
            child = alpha * parents[0] + (1 - alpha) * parents[1]
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final local optimization with SLSQP on top few candidates
    top_candidates = population[:5]
    for h_candidate in top_candidates:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_candidate, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            continue
    
    return (list(best_h), best_c5, n_points)
