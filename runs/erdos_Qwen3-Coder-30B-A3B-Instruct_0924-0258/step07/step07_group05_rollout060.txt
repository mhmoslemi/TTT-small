import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Augmented Lagrangian with adaptive penalty
    max_iter = 5000
    convergence_tol = 1e-6
    
    # Initialize Lagrange multipliers and penalty
    lambda_eq = 0.0
    mu = 100.0
    alpha = 0.1
    trust_region = 0.1
    
    # Store previous values for convergence check
    prev_c5 = float('inf')
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap and gradient
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient w.r.t. h
        # For overlap at shift k, gradient is (1-dx) * [1 - 2*h] for active points
        gradient = np.zeros(n_points)
        if max_k < n_points:
            # overlapping region from h[0] to h[n_points - max_k - 1]
            for i in range(n_points - max_k):
                gradient[i] += (1 - best_h[i + max_k]) - best_h[i]
        else:
            # overlapping region from h[max_k - n_points + 1] to h[n_points - 1]
            offset = max_k - n_points + 1
            for i in range(n_points - offset):
                gradient[i + offset] += (1 - best_h[i]) - best_h[i + offset]
                
        # Add augmented Lagrangian terms
        gradient += lambda_eq
        gradient += mu * (best_h - 1)  # penalty for h > 1
        gradient -= mu * best_h        # penalty for h < 0
        
        # Update h with trust region bounds
        h_new = best_h - alpha * gradient
        
        # Projection to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to maintain sum constraint
        sum_h = np.sum(h_new)
        if abs(sum_h - n_points/2) > 1e-6:
            h_new = h_new * (n_points/2) / sum_h
            
        # Backtracking line search
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better, or with probability based on merit function
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            alpha = min(alpha * 1.1, 0.5)  # expand trust region
        else:
            # Try smaller step
            alpha *= 0.5
        
        # Update penalty parameter
        # Check constraints
        violation_sum = abs(np.sum(best_h) - n_points/2)
        if violation_sum > 1e-3:
            mu = min(mu * 2, 1e6)
        else:
            mu *= 0.9
            
        # Update Lagrange multiplier
        lambda_eq += mu * (np.sum(best_h) - n_points/2)
        
        # Check convergence
        if abs(prev_c5 - best_c5) < convergence_tol:
            break

                    return evaluate_erdos_solution(h, dx)

                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]

            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
