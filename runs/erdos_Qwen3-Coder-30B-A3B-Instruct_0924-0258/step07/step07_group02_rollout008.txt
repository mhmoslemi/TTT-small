import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    pop_size = 100
    population = []
    fitness = []
    
    # Generate initial population
    for _ in range(pop_size):
        # Random initialization
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Track best solutions
    hall_of_fame = sorted(zip(population, fitness), key=lambda x: x[1])[:10]
    best_h = hall_of_fame[0][0].copy()
    best_c5 = hall_of_fame[0][1]
    
    # Track improvement over generations
    last_improvement = 0
    generation = 0
    max_generations = 300
    
    # EA loop
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Sort by fitness
        sorted_pop = sorted(zip(population, fitness), key=lambda x: x[1])
        top_pop = [x[0] for x in sorted_pop[:pop_size//2]]
        top_fitness = [x[1] for x in sorted_pop[:pop_size//2]]
        
        # Create new population
        new_population = []
        new_fitness = []
        
        # Keep top 10%
        elite_count = pop_size // 10
        for i in range(elite_count):
            new_population.append(top_pop[i].copy())
            new_fitness.append(top_fitness[i])
        
        # Crossover and mutation
        for _ in range(pop_size - elite_count):
            # Selection
            parent1 = top_pop[np.random.randint(0, len(top_pop))]
            parent2 = top_pop[np.random.randint(0, len(top_pop))]
            
            # Crossover
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            mutation_rate = 0.2
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Evaluate
            child_fitness = evaluate_erdos_solution(child, dx)
            
            new_population.append(child)
            new_fitness.append(child_fitness)
        
        population = new_population
        fitness = new_fitness
        
        # Update hall of fame
        all_solutions = list(zip(population, fitness))
        all_solutions.sort(key=lambda x: x[1])
        hall_of_fame = all_solutions[:10]
        
        if hall_of_fame[0][1] < best_c5:
            best_h = hall_of_fame[0][0].copy()
            best_c5 = hall_of_fame[0][1]
            last_improvement = generation
        
        # Early stopping
        if generation - last_improvement > 80:
            break
    
    # Final refinement with local optimization
    try:
        # Use best solution so far to initialize L-BFGS-B
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        # Constraint: sum(h) = n_points/2
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points/2}
        
        # Local optimization
        result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 1000})
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
