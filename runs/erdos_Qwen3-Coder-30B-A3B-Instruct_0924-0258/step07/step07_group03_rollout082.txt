import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    patience = 50
    best_iter = 0
    
    # Manual SGD with residual-based gradient
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        
        # Compute gradient based on worst lag
        grad = np.zeros(n_points)
        # For a correlation of h * (1 - h_shifted) at lag k,
        # the gradient w.r.t. h[i] is (1 - 2*h[i]) if h_shifted[k+i] contributes
        # Let's compute this as a convolution of h with -1+2*delta(k-shift)
        # More directly, for lag k, which indices contribute?
        # h[i] contributes to correlation at lag k if h_shifted[k+i] = h[i-k] exists
        # So grad[i] = -2 * (sum over valid j: h[j] with j = i - k if valid) 
        # which is same as h[i-k] terms (i.e., convolve h with -2*shifted delta)
        # Let's compute based on the actual overlapping lags more precisely
        # This is a bit tricky - emperically compute Jacobian and use_pseudo_inv estimation
        
        # Simpler approach: compute simplified gradient for single lag k
        # The spread of this lag is mostly over a contiguous window of similar size
        # Take approximate method: most significant contribution at worst lag
        
        if max_lag_idx < n_points:
            # lag is 0 to n_points-1 (forward shift of h by k to match 1-h)
            offset = max_lag_idx
            # Gradient at current i: -2 * h[i+offset]  
            for i in range(n_points):
                j = (i - offset) % n_points
                if j < n_points:
                    grad[i] -= 2 * h[j]
        else:
            # lag is n_points to 2*n_points-2 (negative shift)
            offset = max_lag_idx - n_points + 1
            # h[i+offset] contribution for lag k diff
            for i in range(n_points):
                j = (i + offset) % n_points
                if j < n_points:
                    grad[i] -= 2 * h[j]
        
        # Apply gradient with learning rate
        h += learning_rate * grad

        # Project to feasible region with constraint ∫h dx = 1 (i.e. sum = n_points/2)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (n_points / 2) / current_sum
        
        # Evaluate current point
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_iter = i
        elif i - best_iter > patience:
            learning_rate *= 0.5
            if learning_rate < 1e-6:
                break
        
        # Add periodic perturbations
        if i % 50 == 0 and i > 0:
            change_mask = np.random.rand(n_points) < 0.1
            delta = np.random.uniform(-0.1, 0.1, n_points)
            h[change_mask] += delta[change_mask]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Phase 2: Refinement with L-BFGS
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'ftol': 1e-12})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
