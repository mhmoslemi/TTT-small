import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization with adaptive learning rate
    max_iter = 2000
    learning_rate = 0.1
    patience = 50
    best_iter = 0
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all correlations and find the worst one
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Compute gradient: for each position, how much to adjust
        # This gradient is derived from the contribution to max_corr
        grad = np.zeros_like(h_current)
        
        # The correlation at position max_idx is sum h[i] * (1 - h[i + max_idx - (n_points-1)])
        # But need to handle indices properly
        # Let's compute it directly:
        # The cross-correlation element at lag k is 
        # sum_{i=0}^{n_points-1} h[i] * (1 - h[i + k]) = sum h[i] - sum h[i] * h[i + k]
        # so the derivative w.r.t. h[i] is: 
        # -1 + (1 if i is in support of the k-th lag) * (1 if i+k is in bounds)
        
        # Compute contribution to the gradient
        lag_index = max_idx - (n_points - 1)
        for i in range(n_points):
            grad[i] = -1.0  # from linear term 
            # Add influence: how often does h[i+k] participate in the sum?
            if 0 <= i - lag_index < n_points:  # if the corresponding index is valid
                # This is a simplified approach - we add where there's a contribution to max_corr
                # Consider all valid combinations
                if 0 <= i < n_points and 0 <= i - lag_index < n_points:
                    grad[i] += 1.0  # Because h[i] multiplies (1-h[i]), the term is h[i] - h[i]^2
                else:
                    grad[i] += 0.0
                    
        # Simplified: Direct computation of contributions for max_correlation
        # If we want to reduce max_corr, typically we:
        #   reduce values that appear in the product
        # But a better approach:
        # At lag k, the contribution of h[i] to C_k is h[i] * (1 - h[i + k]) where k is lag_index  
        # So the gradient w.r.t. h[i] at that position is - h[i+k] for i+k < n_points, 0 otherwise  
        # But since we're taking derivative of sum of these terms, sum up the contributions
        
        # Better gradient assuming we take the lag with highest overlap:
        strategy_h = h_current.copy()
        max_lag = max_idx - (n_points - 1)
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] -= (1 - h_current[j])  # derivative of h[i] * (1 - h[i+k])
            # No change needed for other terms in that one lag
        grad -= np.sum(grad) / n_points  # Subtract mean for sum constraint

        # Normalize gradient
        grad = grad / (np.linalg.norm(grad) + 1e-12)

        # Line search helper
        def line_search_step(alpha):
            h_trial = h_current - alpha * grad
            # Ensure within bounds and proper sum
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            return evaluate_erdos_solution(h_trial, dx)
        
        # Backtracking line search
        step_size = learning_rate
        trial_c5 = line_search_step(step_size)
        tries = 0
        while trial_c5 >= current_c5 and tries < 10:
            step_size *= 0.5
            trial_c5 = line_search_step(step_size)
            tries += 1
        if trial_c5 < current_c5:
            h_current = h_current - step_size * grad
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            current_c5 = trial_c5

        # Apply adaptive learning rate
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            best_iter = iteration
            learning_rate = min(learning_rate * 1.1, 1.0)
        else:
            if iteration - best_iter > patience:
                learning_rate *= 0.8
                if learning_rate < 1e-6:
                    break
        # Periodic utility check
        if iteration % 100 == 0:
            if time() - start_time > budget_s * 0.8:
                break
    
    # Phase 2: Local optimization with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try a few restarts from random starts to avoid local minima
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = np.random.uniform(0.1, 0.9, n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            res = minimize(objective, h_restart, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except:
        pass

    return (list(best_h), best_c5, n_points)
