import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better initial point using a piecewise construction
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Use a step function with a discontinuity near the center
        for i in range(n_points):
            x = i * dx
            if x < 0.8:
                h_start[i] = 0.0
            elif x < 1.2:
                h_start[i] = 0.7
            else:
                h_start[i] = 0.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with trust-constr
    # Use a few restarts with different starting points
    restarts = 5
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the best solution slightly
        h_perturb = best_h + np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        
        # Use scipy minimize with trust-constr
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points/2
            
        def constraint_bounds_l(h):
            return h
            
        def constraint_bounds_u(h):
            return 1 - h
            
        cons = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_bounds_l},
            {'type': 'ineq', 'fun': constraint_bounds_u}
        ]
        
        try:
            res = minimize(
                obj, 
                h_perturb, 
                method='trust-constr',
                constraints=cons,
                options={'maxiter': 500, 'disp': False}
            )
            if res.success and res.fun < best_c5:
                best_h = res.x.copy()
                best_c5 = res.fun
        except:
            pass
    
    # A simple local search to slightly refine the solution
    # Try a few neighbor constructions
    for trial in range(50):
        if time() - start_time > budget_s * 0.95:
            break
        # Small random perturbations
        h_trial = best_h + np.random.normal(0, 0.03, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
