import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 50
    max_gen = 100
    mut_rate = 0.05
    cross_rate = 0.7
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, mut_rate, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolutionary Loop
    for gen in range(max_gen):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_pop = [population[0].copy()]  # Elitism
        
        # Crossover and mutation
        while len(new_pop) < pop_size:
            if random.random() < cross_rate and len(new_pop) + 1 < pop_size:
                # Select parents
                p1, p2 = random.sample(population[:20], 2)
                
                # Blend crossover
                alpha = random.random()
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
                
                # Mutation
                child1 += np.random.normal(0, mut_rate, n_points)
                child2 += np.random.normal(0, mut_rate, n_points)
                
                # Repair
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_pop.append(child1)
                if len(new_pop) < pop_size:
                    new_pop.append(child2)
            else:
                # Just mutate
                parent = population[random.randint(0, 19)]
                child = parent + np.random.normal(0, mut_rate, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_pop.append(child)
        
        population = new_pop
        
        # Adaptive mutation rate
        if gen % 10 == 0:
            mut_rate = max(0.005, mut_rate * 0.9)
    
    # Final refinement with Nelder-Mead
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    # Try several refinements
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        try:
            h_refine = best_h + np.random.normal(0, 0.01, n_points)
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            
            res = minimize(obj, h_refine, method='Nelder-Mead', options={'maxiter': 100})
            
            if res.success:
                c5_refined = obj(res.x)
                if c5_refined < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_refined
        except:
            pass
    
    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
