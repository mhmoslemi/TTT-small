import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    basis_functions = []

        for j in range(n_points):
            x = x_basis[j]
            if abs(x - center) <= width:
                b[j] = 1.0 - abs(x - center) / width
        basis_functions.append(b)

    coeffs = np.random.rand(n_basis) * 0.5

    perturbation_magnitude = 0.05
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute h from coefficents
        h_current = B @ coeffs
        h_current = np.clip(h_current, 0, 1)
        
        # Adjust for constraint
        sum_h = np.sum(h_current)
        if abs(sum_h - n_points/2) > 1e-6:
            h_current = h_current * (n_points/2) / sum_h

        current_c5 = evaluate_erdos_solution(h_current, dx)

            best_c5_value = current_c5
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1
            
        # Perturb coefficients if stuck
        if patience > patience_limit:

            perturbation_magnitude *= 0.95  # Reduce perturbation
            patience = 0

        # Simple gradient estimation using finite differences

        grad = np.zeros(n_basis)
        eps = 1e-4

    try:

        def objective(c):
            h_vals = B @ c
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points/2) / np.sum(h_vals)

            h_vals = B @ c

            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5_value:
                best_h = h_final
                best_c5_value = c5_final
                
    except Exception:
        pass
    
    return (list(best_h), best_c5_value, n_points)
