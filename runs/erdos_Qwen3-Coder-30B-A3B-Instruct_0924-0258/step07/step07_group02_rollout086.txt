import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # EA parameters
    pop_size = 100
    max_generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10
    tournament_size = 5
    
    # Create initial population
    population = []
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            h = np.clip(h, 0, 1)
        population.append(h)
    
    # Hall of fame
    hall_of_fame = []
    
    # EA loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update hall of fame
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:elite_size]
        for idx in top_indices:
            h = population[idx]
            c5 = fitness[idx]
            if not any(np.allclose(h, h_f) for h_f, _ in hall_of_fame):
                hall_of_fame.append((h.copy(), c5))
        hall_of_fame.sort(key=lambda x: x[1])
        hall_of_fame = hall_of_fame[:elite_size]
        
        # Check for improvement
        if fitness[sorted_indices[0]] < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = fitness[sorted_indices[0]]
        
        # Create next generation
        new_pop = []
        
        # Elite retention
        for i in range(elite_size):
            new_pop.append(population[top_indices[i]].copy())
        
        # Selection, crossover, mutation
        while len(new_pop) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            selected_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Apply crossover with probability
            if np.random.rand() < crossover_rate and len(new_pop) + 1 < pop_size:
                # Select second parent
                tournament_indices2 = np.random.choice(pop_size, tournament_size, replace=False)
                tournament_fitness2 = [fitness[i] for i in tournament_indices2]
                selected_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
                
                # Arithmetic crossover with alpha in [0.2, 0.8]
                alpha = np.random.uniform(0.2, 0.8)
                h1 = population[selected_idx]
                h2 = population[selected_idx2]
                child1 = alpha * h1 + (1 - alpha) * h2
                child2 = (1 - alpha) * h1 + alpha * h2
                
                # Repair and add to population
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_pop.append(child1)
                new_pop.append(child2)
            else:
                # Clone and mutate
                h_clone = population[selected_idx].copy()
                # Gaussian mutation
                mutation_magnitude = 0.1 * (0.9 ** (gen // 50))
                noise = np.random.normal(0, mutation_magnitude, n_points)
                h_clone += noise
                # Bit-flip mutation (randomly flip 20% of values)
                if np.random.rand() < 0.1:  # 10% chance to do bit-flip
                    flip_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
                    for i in flip_indices:
                        h_clone[i] = 1 - h_clone[i]
                
                # Repair
                h_clone = np.clip(h_clone, 0, 1)
                h_clone = h_clone * (n_points/2) / np.sum(h_clone)
                new_pop.append(h_clone)
        
        # Trim to exact pop size
        population = new_pop[:pop_size]
    
    # Final refinement using gradient-based method on best solutions
    try:
        # Perform local optimization on top fine solutions
        max_refinements = min(5, len(hall_of_fame))
        for i in range(max_refinements):
            h_init = hall_of_fame[i][0]
            c5_init = hall_of_fame[i][1]
            # Define objective function for L-BFGS
            def objective(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return -np.max(corr) * dx  # negative because minimize
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            # Solve with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(objective, h_init, method='L-BFGS-B', bounds=bounds,
                              constraints={'fun': constraint_sum, 'type': 'eq'},
                              options={'maxiter': 1000})
            
            if result.success:
                h_refined = result.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except Exception:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final convex feasibility check
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add key constraints
        corr = np.correlate(best_h, 1 - best_h, mode='full') 
        max_k_idx = np.argmax(corr)
        # Focus on critical lags
        critical_lags = np.sort(np.argsort(corr)[-15:])  # Top 15 lags
        for k in critical_lags:
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
