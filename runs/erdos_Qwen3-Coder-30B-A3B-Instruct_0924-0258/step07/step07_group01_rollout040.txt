import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Initialize with initial_h_values if available

        n_points = len(h_start)
    else:

    best_c5 = evaluate_erdos_solution(h_start, dx)

    s = np.ones(n_points)

        s = (np.array(initial_h_values) > 1e-6).astype(float)
    else:

        s = np.ones(n_points)
        s[n_points//2:] = 0

    a = a * (n_points/2) / np.sum(a * s)
    a = np.clip(a, 0, 1)
    
    # Track best
    best_a, best_s = a.copy(), s.copy()
    best_c5 = evaluate_erdos_solution(a * s, dx)
    
    # Iteration variables
    stagnation_count = 0
    max_stagnation = 100
    
    def project_sum(a, s):
        """Ensure sum(a * s) = n_points / 2"""
        total = np.sum(a * s)
        if abs(total - n_points/2) < 1e-10:
            return a, s
        scale = (n_points / 2) / total
        a = a * scale
        a = np.clip(a, 0, 1)
        return a, s
    
    def optimize_a_fixed_s(a, s):
        """Minimize overlap with fixed support s"""
        # Only optimize a on active support
        active = s > 0.5
        if not np.any(active):
            return a
            
        # Use CVXPY to solve the convex subproblem
        try:
            a_var = cp.Variable(n_points)
            
            # Construct correlation matrix or use a simpler approach 
            # since active entries may be fewer
            def compute_overlap(a_vec):
                h = a_vec * s
                corr = np.correlate(h, 1 - h, mode='full')
                return np.max(corr) * dx
            
            # Use scipy for easier handling
            def obj(lam_a):
                a_test = lam_a * s
                a_test = np.clip(a_test, 0, 1)
                return compute_overlap(a_test)
            
            # Equality constraint: sum(a*s) = n_points/2
            def constraint_sum(lam_a):
                a_test = lam_a * s
                return np.sum(a_test) - n_points / 2
            
            # Use SLSQP for this
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]

            a_scaled = a_scaled * (n_points/2) / np.sum(a_scaled * s)
            a_scaled = np.clip(a_scaled, 0, 1)
            
            res = minimize(obj, np.sum(a_scaled * s), 
                           method='SLSQP', 
                           bounds=[(0, 1e6)], 
                           constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-8})
            
            if res.success:
                # Recompute a values
                scale = res.x[0] / np.sum(a_scaled * s)
                a_out = a_scaled * scale
                a_out = np.clip(a_out, 0, 1)

        return a

        res = minimize(objective, final_h,

                       bounds=bounds, 
                       options={'maxiter': 100})
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
        else:
            # fallback
            best_h = final_h
    except Exception as e:
        pass
    
    return (list(best_h), best_c5, n_points)
