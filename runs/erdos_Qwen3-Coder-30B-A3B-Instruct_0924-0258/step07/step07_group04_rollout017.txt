import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def barrier_penalty(h, dx, penalty_factor=1e6):
    # Penalty for sum constraint
    sum_violation = np.abs(np.sum(h) - (len(h) / 2.0))
    sum_penalty = penalty_factor * sum_violation**2
    
    # Penalty for bounds
    bound_penalty = penalty_factor * (np.sum(np.maximum(0, -h)) + 
                                      np.sum(np.maximum(0, h - 1)))
    
    return sum_penalty + bound_penalty

def lagrangian_objective(h, dx, penalty_factor=1e6):
    # Objective: maximize overlap integral (minimize its negative)
    overlap = evaluate_erdos_solution(h, dx)
    penalty = barrier_penalty(h, dx, penalty_factor)
    return overlap + penalty

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with simpler pattern
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE parameters
    popsize = 50
    maxiter = 1000
    mutation_rate = 0.5
    crossover_rate = 0.9
    
    # Ensure feasible starting point
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for DE
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=int(maxiter * 0.7),
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=crossover_rate,
        seed=seed,
        disp=False
    )
    
    if de_result.success:
        h_de = de_result.x
        c5_de = objective(h_de)
        if c5_de < best_c5:
            best_h = h_de
            best_c5 = c5_de
    
    # Local refinement with SLSQP
    try:
        # Ensure bounds
        h_refined = np.clip(best_h, 0, 1)
        # Ensure sum constraint
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds_slsqp = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds_slsqp, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    except Exception:
        pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
