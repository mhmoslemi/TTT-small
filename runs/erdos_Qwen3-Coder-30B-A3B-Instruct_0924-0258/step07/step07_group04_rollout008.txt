import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        # Initialize with periodic function to encourage low overlap
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))  # 5 cycles over [0,2]
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
        # Add small random perturbation
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive perturbation with multiresolution refinement
    resolutions = [20, 40, 80]  # Coarse to fine
    current_n = resolutions[0]
    
    # For each resolution level
    for res_idx, n_res in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate or interpolate
        if n_res == current_n:
            h = best_h.copy()
        else:
            # Interpolate from previous resolution
            x_old = np.linspace(0, 2, current_n)
            x_new = np.linspace(0, 2, n_res)
            h = np.interp(x_new, x_old, best_h)
            # Add random noise to improve exploration
            h += np.random.normal(0, 0.05, n_res)
            h = np.clip(h, 0, 1)
            normalization = (n_res / 2) / np.sum(h)
            h *= normalization
            current_n = n_res
            
        dx_res = 2.0 / n_res
        
        # Search parameters
        max_iter = 200 if n_res == 80 else 150 if n_res == 40 else 100
        perturbation_size = 0.15 if n_res == 20 else 0.1 if n_res == 40 else 0.05
        temp = 1.0
        temp_decay = 0.99
        tabu_list = []
        tabu_size = 20
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate candidate
            h_candidate = h.copy()
            indices = np.random.choice(n_res, size=max(1, int(0.2 * n_res)), replace=False)
            for i in indices:
                delta = np.random.uniform(-perturbation_size, perturbation_size) * h_candidate[i]
                h_candidate[i] += delta
            
            # Enforce bounds and normalization
            h_candidate = np.clip(h_candidate, 0, 1)
            normalization = (n_res / 2) / np.sum(h_candidate)
            h_candidate *= normalization
            
            # Evaluate
            c5_candidate = evaluate_erdos_solution(h_candidate, dx_res)
            
            # Accept or reject
            accept = False
            if c5_candidate < best_c5:
                accept = True
            else:
                # Simulated annealing acceptance probability
                if np.random.random() < np.exp((best_c5 - c5_candidate) / (temp + 1e-10)):
                    accept = True
            
            if accept:
                h = h_candidate.copy()
                if c5_candidate < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_candidate
                
                # Update tabu list
                tabu_list.append(h.copy())
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
            
            # Update temperature
            temp *= temp_decay
            
        # Final polishing if finest resolution
        if n_res == 80:
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx_res)
                
                def constraint_sum(h):
                    return np.sum(h) - n_res / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_res)]
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 500, 'ftol': 1e-12})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                pass
        else:
            # Update the current best for next resolution
            best_h = h.copy()
            best_c5 = evaluate_erdos_solution(h, dx_res)
            current_n = n_res
    
    # Verify constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "h not in [0,1]"
    assert abs(np.sum(best_h) - n_points/2) < 1e-6, "sum constraint violated"
    
    return (list(best_h), best_c5, n_points)
