import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = np.zeros(n_points)
        # Place a peak in the middle and decay on both sides
        h_start[n_points//4:3*n_points//4] = 0.5 + 0.3 * np.sin(np.pi * (np.arange(n_points//4, 3*n_points//4) - n_points//4) / (n_points//2))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: exploiting convex structure for fine-tuning
    # Start with direct optimization using simulated annealing
    # based on local improvements
    
    # Construct a sequence of smaller optimizations to avoid local minima
    # First, try a local optimizer on a random perturbation
    current_h = best_h.copy()
    iter_count = 0
    max_iter = 1000
    
    # Use scipy.optimize with L-BFGS-B for local search
    def objective(h_vals):
        h = np.array(h_vals)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_fun(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_fun}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Random restarts within the search
    num_restarts = 50
    for restart in range(num_restarts):
        if time() - start_time > budget_s * 0.95:
            break

        h_rand = current_h + np.random.normal(0, 0.05, n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        
        try:
            result = minimize(objective, h_rand, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 50})
            if result.success:
                h_candidate = np.clip(result.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
            
        # Occasionally, perturb the best solution and try to improve
        if restart % 5 == 0:
            # Perturb best solution
            perturbation = np.random.normal(0, 0.02, n_points)
            temp_h = best_h + perturbation
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = temp_h * (n_points / 2) / np.sum(temp_h)

    # Try repairing best solution using convex program if still in time
    if time() - start_time < budget_s * 0.9:

            prob = cp.Problem(cp.Minimize(0), constraints)
            # Small chance to avoid solver instability

            if prob.status == cp.OPTIMAL:

                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_feasible
        except Exception:
            # If solver fails, continue with best found
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
