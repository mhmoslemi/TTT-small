import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization with adaptive learning rate
    max_iter = 1000
    learning_rate = 0.1
    stagnation_count = 0
    last_improvement = 0
    
    h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
        
        # Compute current overlap and lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_overlap = corr[max_lag_idx] * dx
        
        # Compute gradient of overlap w.r.t. h
        # For the maximum lag, get the contribution to gradient
        n_corr = len(corr)
        lag = max_lag_idx - (n_points - 1)
        
        # Gradient contribution from correlation kernel
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + lag
            if 0 <= j < n_points:
                grad[i] += (1 - 2 * h[i])  # From derivative of (h[i] * (1 - h[j]))
        
        # Update with learning rate
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to satisfy sum constraint
        h_sum = np.sum(h_new)
        if abs(h_sum) > 1e-12:
            h_new = h_new * (n_points / 2) / h_sum
        
        # Check if we improved
        h = h_new
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if iteration - last_improvement > 50:
            learning_rate *= 0.9
            last_improvement = iteration
        
        # Perturb occasionally
        if iteration % 50 == 0 and iteration > 0:
            indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[indices] += np.random.uniform(-0.1, 0.1, len(indices))
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-12:
                h = h * (n_points / 2) / h_sum
    
    # Phase 2: Local SQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use best solution from phase 1 as starting point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10, 'eps': 1e-6})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
