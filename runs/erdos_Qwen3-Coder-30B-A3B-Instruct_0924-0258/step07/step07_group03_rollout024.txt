import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing
    T0 = 0.1
    cooling_rate = 0.001
    max_iterations = 1000
    perturbation_fraction = 0.15
    
    h_current = h_start.copy()
    c5_current = best_c5
    T = T0
    stagnation_count = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb h_current
        h_proposal = h_current.copy()
        num_perturb = int(n_points * perturbation_fraction)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-0.1, 0.1, num_perturb)
        h_proposal[indices] += delta
        h_proposal = np.clip(h_proposal, 0, 1)
        
        # Normalize
        h_proposal = h_proposal * (n_points/2) / np.sum(h_proposal)
        
        # Evaluate
        c5_proposal = evaluate_erdos_solution(h_proposal, dx)
        
        # Accept or reject
        delta_c5 = c5_proposal - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_proposal
            c5_current = c5_proposal
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Cool down
        T = T0 * np.exp(-cooling_rate * iteration)
        
        if stagnation_count > 100:
            break
    
    # Gradient Descent Refinement
    learning_rate = 0.1
    max_gd_iterations = 500
    gd_stagnation = 0
    prev_c5 = best_c5
    
    h_refined = best_h.copy()
    
    for _ in range(max_gd_iterations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient with respect to h for lag k_max
        grad = np.zeros(n_points)
        if k_max < n_points:
            for i in range(n_points - k_max):
                grad[i] += (1 - h_refined[i + k_max]) * dx
        else:
            for i in range(k_max - n_points + 1, n_points):
                grad[i] += (1 - h_refined[i - k_max]) * dx
        
        # Update h
        h_new = h_refined - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        if new_c5 < prev_c5:
            h_refined = h_new
            prev_c5 = new_c5
            gd_stagnation = 0
        else:
            gd_stagnation += 1
            if gd_stagnation > 50:
                learning_rate *= 0.5
                gd_stagnation = 0
                
        if learning_rate < 1e-8:
            break

    # Final SLSQP Refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
