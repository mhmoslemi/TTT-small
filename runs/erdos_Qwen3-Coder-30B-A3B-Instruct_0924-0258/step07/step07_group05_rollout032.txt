import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 50
    num_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # Fitness function for GA
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Tournament selection
    def tournament_selection(pop, fitnesses):
        indices = np.random.choice(len(pop), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx]
    
    # Crossover function (uniform)
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        mask = np.random.rand(len(parent1)) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        return child1, child2
    
    # Mutation function
    def mutate(h):
        mask = np.random.rand(len(h)) < mutation_rate
        noise = np.random.normal(0, 0.03, len(h))
        h_mut = h + noise * mask
        h_mut = np.clip(h_mut, 0, 1)
        # Rescale to maintain constraint
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        return h_mut
    
    # Convex relaxation helper
    def convex_relaxation(h_candidate, c5_bound):
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            # Sample a few k values to create constraints
            k_vals = np.linspace(0.1, 1.9, min(15, n_points//3))
            for k_val in k_vals:
                k = int(k_val * n_points / 2)
                if k < n_points and k > 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= c5_bound * dx + 1e-8)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                return h_feasible, c5_feasible
        except Exception:
            pass
        return h_candidate, evaluate_erdos_solution(h_candidate, dx)
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        
        # Find best in population
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Selection and reproduction
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population
        
        # Every 50 generations, do convex relaxation on top candidates
        if gen % 50 == 0 and gen > 0:
            # Sort by fitness and apply convex relaxation to top 10%
            sorted_indices = np.argsort([fitness(h) for h in population])
            top_indices = sorted_indices[:pop_size // 10]
            
            for i in top_indices:
                h_refined, c5_refined = convex_relaxation(population[i], best_c5)
                if c5_refined < fitnesses[i]:
                    population[i] = h_refined.copy()
                    fitnesses[i] = c5_refined
        
        # Early stopping if no improvement in last 100 generations
        if gen > 100 and all(fitnesses[i] >= best_c5 for i in range(pop_size)):
            break
    
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass

    return (list(best_h), best_c5, n_points)
