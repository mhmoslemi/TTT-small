import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based descent with perturbations
    max_iter = 1000
    lr = 0.1
    patience = 50
    patience_counter = 0
    
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps and find worst lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        l_max = np.argmax(corr)
        current_c5 = corr[l_max] * dx
        
        # Compute gradient contributions
        n_corr = len(corr)
        grad = np.zeros_like(h_current)
        
        # For the worst lag, compute gradient contribution
        for i in range(len(h_current)):
            for k in range(len(h_current)):
                j = (i + k) % len(h_current)
                if j == i and k == l_max - (len(h_current) - 1):
                    grad[i] += (1 - 2 * h_current[i])
        
        # Update h with gradient
        h_current += lr * grad
        h_current = np.clip(h_current, 0, 1)
        
        # Project to correct sum
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Perturb periodically
        if iteration % 50 == 0 and iteration > 0:
            perturb_indices = np.random.choice(len(h_current), size=int(0.1 * len(h_current)), replace=False)
            for i in perturb_indices:
                h_current[i] += np.random.uniform(-0.1, 0.1)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h_current, dx)
        if new_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience:
                lr *= 0.5
                patience_counter = 0
    
    # Phase 2: SQP refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use last iteration's solution as starting point for SQP
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
