import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    pop_size = 100
    generations = 250
    mutation_rate = 0.2
    crossover_rate = 0.7
    elite_size = 10

        population.append(h)
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)

            best_h = h.copy()
            best_c5 = fit
    
    # Hall of fame
    hall_of_fame = [best_h.copy()]
    hall_of_fame_fitness = [best_c5]
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]

        if fitness[0] < hall_of_fame_fitness[0]:

            hall_of_fame_fitness[0] = fitness[0]

        while len(new_population) < pop_size:
            # Tournament selection (size 3)
            idx1, idx2 = np.random.choice(len(population), 3, replace=False)

            idx3, idx4 = np.random.choice(len(population), 3, replace=False)
            parent2 = population[idx3] if fitness[idx3] < fitness[idx4] else population[idx4]

            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                h = alpha * parent1 + (1 - alpha) * parent2
            else:
                h = parent1.copy()

                noise = np.random.normal(0, 0.1, n_points)

                    best_h = h_feasible
                    best_c5 = c5_feasible

        pass
    
    # Final verification with L-BFGS
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])

    cons = [{'type': 'eq', 'fun': constraint_sum}]

    try:
        res = minimize(objective, hall_of_fame[0], method='L-BFGS-B',

        if res.success:

            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)

                best_c5 = c5_opt
    except Exception:
        pass
    
    # Ensure feasibility and final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
