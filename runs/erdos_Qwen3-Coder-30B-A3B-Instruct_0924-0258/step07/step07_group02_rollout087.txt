import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Create a smooth initial construction

        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 100
    max_generations = 250
    elite_count = 10
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    repair_attempts = 5
    
    # Population initialization
    population = []
    fitness = []
    
    # Create initial population
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            # Use the initial construction as the first individual
            h = h_start.copy()
        else:
            # Generate random individuals
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        f = evaluate_erdos_solution(h, dx)
        fitness.append(f)
        if f < best_c5:
            best_h = h.copy()
            best_c5 = f
    
    # Evolutionary loop
    generation = 0
    no_improvement_count = 0
    
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= 80:
                break
        
        # Keep elite
        new_population = population[:elite_count]
        new_fitness = fitness[:elite_count]
        
        # Generate offspring using tournament selection and crossover
        while len(new_population) < pop_size:
            # Tournament selection
            idx1 = np.random.choice(tournament_size, replace=False)
            idx2 = np.random.choice(tournament_size, replace=False)
            parent1 = population[np.argmin([fitness[i] for i in idx1])]
            parent2 = population[np.argmin([fitness[i] for i in idx2])]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Add noise to some elements
                noise_level = 0.1 * np.exp(-generation / 50.0)
                mask = np.random.rand(n_points) < 0.2
                child += np.random.normal(0, noise_level, n_points) * mask
                child = np.clip(child, 0, 1)
                
            # Repair and normalize
            for _ in range(repair_attempts):

                if np.abs(np.sum(child) - n_points / 2) < 1e-10 and np.all(child >= 0) and np.all(child <= 1):
                    break
            
            # Evaluate and add to new population
            child_fitness = evaluate_erdos_solution(child, dx)
            new_population.append(child)
            new_fitness.append(child_fitness)
        
        population = new_population
        fitness = new_fitness
        generation += 1

    try:

        top_individuals = [population[i] for i in range(min(5, len(population)))]
        best_refined = best_h
        best_refined_c5 = best_c5

            # Create a function for optimization
            def obj(h):
                corr = np.correlate(h, 1 - h, mode='full')
                return -np.max(corr) * dx
            
            # Constraints
            constr = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimize with L-BFGS-B
            result = minimize(obj, h_temp, method='L-BFGS-B', bounds=bounds, constraints=constr, options={'maxiter': 100})
            
            if result.success:
                h_opt = result.x
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_refined_c5:
                    best_refined = h_opt
                    best_refined_c5 = c5_opt
                    
        best_h = best_refined
        best_c5 = best_refined_c5
    except Exception:
        pass
    
    # Final verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
