import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based optimization
    max_iter = 1000
    lr = 0.1
    patience = 0
    patience_limit = 50
    
    h_current = h_start.copy()
    target_sum = n_points / 2.0
    
    # For gradient descent with line search
    def compute_gradient(h, k_max):
        grad = np.zeros_like(h)
        # Compute correlation for this specific lag
        corr = np.correlate(h, 1 - h, mode='full')
        # Identify where the maximum occurs
        if k_max < len(h):
            # Forward case
            for i in range(len(h)):
                if i + k_max < len(h):
                    grad[i] += 1.0  # 1 for h[i] contributing to overlap with 1-h[i+k_max]
        else:
            # Backward case  
            k = len(h) - 1 - (k_max - len(h) + 1)
            for i in range(k + 1, len(h)):
                grad[i] += 1.0  # contribution from h[i] to 1-h[i-k_max+len(h)-1]  
        return grad

    def line_search(h_current, grad, lr, dx):
        # Reduce lr until constraint satisfied or reach small lr
        temp_lr = lr
        for _ in range(10):
            h_new = h_current - temp_lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            if np.sum(h_new) > 0.9 * target_sum and np.sum(h_new) < 1.1 * target_sum:
                return h_new
            temp_lr /= 2.0
        return h_current

    for i in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all overlaps and find max
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_h = h_current.copy()
            best_c5 = max_overlap

        # Compute gradient
        grad = compute_gradient(h_current, k_max)
        
        # Line search to find good step
        h_new = line_search(h_current, grad, lr, dx)
        
        # Update current solution
        h_current = h_new
        
        # Adjust learning rate based on stagnation
        patience += 1
        if patience > patience_limit:
            lr /= 2.0
            patience = 0
            if lr < 1e-5:
                break


    # Phase 2: Local refinement with L-BFGS-B
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
