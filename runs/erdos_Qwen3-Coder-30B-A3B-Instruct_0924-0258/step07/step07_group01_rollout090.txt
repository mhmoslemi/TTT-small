def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    h = h_start.copy()
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)

    a = np.ones(n_points)

    h = a * s

    # Alternating optimization
    iterations = 0
    stagnation_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Step 1: Fix s, optimize a
        try:
            # Constraint that sum(a*s) = n_points/2
            # We'll optimize a with fixed s
            # s is fixed. Our variables are a
            
            def optimize_a_fixed_s():
                # The objective is to minimize the overlap integral 
                # Maximize about lag in convolution (for the maximum overlap)
                # Simple idea: project to correct support size, then scale.
                h_trial = a * s
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                return evaluate_erdos_solution(h_trial, dx)
            
            # For s fixed, we optimize a such that:
            # sum(a * s) = n_points/2, 0 <= a <= 1
            # minimize max convolution
            # This is equivalent to: we want to 'flatten' the overlap profile.
            
            # Heuristic: make a dense and uniform on s, but adjusted
            # Weight each active point evenly to satisfy constraint
            # and leave the a values to be as uniform as possible
            
            # Total support
            total_active = np.sum(s)
            if total_active > 0:
                # All a's on s should sum to n_points/2
                a = np.ones(n_points)
                # Normalize
                a = a * (n_points / 2) / np.sum(a)
                # Apply support
                a = a * s
            else:
                # In case safe fallback
                a = np.ones(n_points) * 0.5
                
            # Now keep a bounded
            a = np.clip(a, 0, 1)
            a = a * (n_points / 2) / np.sum(a)
            
        except Exception:

        # Step 2: Fix a, optimize s

            a = a * s  # This might be redundant: already is 0 outside support
            a = a * (n_points / 2) / np.sum(a)
            
        except Exception:
            pass
        
        # refinement step
        h = a * s

    # Final SLSQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        # Use current best as start
        h_refined = best_h.copy()
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
