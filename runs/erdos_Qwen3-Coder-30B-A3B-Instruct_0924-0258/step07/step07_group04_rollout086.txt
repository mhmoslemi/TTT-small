import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with periodic function or use initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with Tabu Search
    # Parameters
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_list_size = 20
    max_iter = 1000
    perturbation_size = 0.1
    
    # Tabu list
    tabu_list = []
    
    # Initial solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    stagnation_counter = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
        
        for i in indices:
            delta = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += delta
        
        # Apply bounds and normalization
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = candidate_h * (n_points/2) / np.sum(candidate_h)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Check tabu list
        tabu_accept = True
        for tabu_h in tabu_list:
            if np.allclose(tabu_h, candidate_h, atol=1e-8):
                tabu_accept = False
                break
                
        # Acceptance criterion
        if candidate_c5 < current_c5 or tabu_accept:
            # Accept the candidate
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            # Update tabu list
            tabu_list.append(current_h.copy())
            if len(tabu_list) > tabu_list_size:
                tabu_list.pop(0)
                
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                last_improvement = iteration
                
        elif np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            # Accept worse solution with probability
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
        # Update temperature
        T = max(T * alpha, T_min)
        
        # Stagnation check
        if iteration - last_improvement > 50:
            # Perturb best solution
            noise = np.random.normal(0, 0.05, n_points)
            current_h = best_h + noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points/2) / np.sum(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            last_improvement = iteration
            
        # Adaptively adjust perturbation size based on progress
        if iteration > 50 and iteration % 10 == 0:
            # Decrease perturbation if we're converging
            if iteration > 100 and (iteration % 25 == 0):
                perturbation_size *= 0.95
    
    # Final refinement with SLSQP
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
