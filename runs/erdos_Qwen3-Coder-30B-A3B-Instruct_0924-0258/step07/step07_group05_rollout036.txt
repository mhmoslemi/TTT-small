import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhanced gradient-based approach
    max_iter = 10000
    convergence_tol = 1e-6
    
    # Adaptive parameters
    alpha = 0.1
    penalty = 100.0
    trust_radius = 0.1
    max_trust_radius = 1.0
    
    # Track history for convergence
    history = []
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient using finite differences
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        current_c5 = objective(best_h)
        history.append(current_c5)
        
        if len(history) > 5 and abs(history[-1] - history[-2]) < convergence_tol:
            break
            
        # Gradient approximation via finite differences
        grad = np.zeros(n_points)
        epsilon = 1e-4
        for i in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            if best_h[i] + epsilon > 1:
                h_plus[i] = 1.0
            else:
                h_plus[i] += epsilon
                
            if best_h[i] - epsilon < 0:
                h_minus[i] = 0.0
            else:
                h_minus[i] -= epsilon
                
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        
        # Gradient-based update with trust region
        update = -alpha * grad
        
        # Project update to respect bounds
        h_new = best_h + update
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points/2) / sum_h
        
        # Evaluate new solution
        new_c5 = objective(h_new)
        
        # Accept or reject based on improvement
        if new_c5 < current_c5:
            best_h = h_new
            best_c5 = new_c5
            # Increase trust radius
            trust_radius = min(trust_radius * 1.1, max_trust_radius)
            alpha = min(alpha * 1.1, 1.0)
        else:
            # Decrease trust radius
            trust_radius *= 0.5
            alpha *= 0.5
            if trust_radius < 1e-8:
                break
                
        # Regularize with penalty
        penalty *= 0.99995
        
        # Periodic constraint enforcement
        if iter_num % 50 == 0:
            sum_h = np.sum(best_h)
            if abs(sum_h - n_points/2) > 1e-6:
                best_h = best_h * (n_points/2) / sum_h
            
    # Final refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
