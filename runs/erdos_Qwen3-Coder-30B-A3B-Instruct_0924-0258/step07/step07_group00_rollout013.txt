import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    h = h_start.copy()
    lr = 0.01
    stagnation_count = 0
    max_stagnation = 50
    
    # Main optimization loop
    iterations = 0
    while time() - start_time < budget_s * 0.95:
        # Evaluate current state
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > max_stagnation:
            break
            
        # Find the most constraining lag k
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient for this lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            gradient[i] = 2 * (1 - h[j] - h[i])
            
        # Update step
        h_new = h - lr * gradient
        
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if update was beneficial
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5:
            h = h_new
            # Increase learning rate if beneficial step
            lr = min(lr * 1.05, 0.1)
        else:
            # Decrease learning rate if step was bad
            lr *= 0.5
            
        # Reset learning rate periodically
        if iterations % 100 == 0 and iterations > 0:
            lr = 0.01
            
        iterations += 1
    
    # Final refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 500, 'ftol': 1e-9})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
