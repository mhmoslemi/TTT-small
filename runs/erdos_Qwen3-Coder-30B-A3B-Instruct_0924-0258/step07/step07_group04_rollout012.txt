import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Periodic initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / 5))
        h_start = np.clip(h_start, 0, 1)
        normalization = (n_points / 2) / np.sum(h_start)
        h_start *= normalization
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    T_min = 1e-5
    tabu_list = []
    tabu_size = 20
    stagnation_count = 0
    max_stagnation = 50
    perturbation_size = 0.1
    
    # Simulated annealing with tabu search
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        # Create candidate solution
        candidate_h = best_h.copy()
        indices = np.random.choice(n_points, int(n_points * 0.15), replace=False)
        for i in indices:
            perturbation = np.random.uniform(-perturbation_size, perturbation_size)
            candidate_h[i] += perturbation
        
        # Enforce bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Normalize to sum constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Evaluate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Tabu check
        is_tabu = False
        for tabu_h in tabu_list:
            if np.allclose(candidate_h, tabu_h, atol=1e-6):
                is_tabu = True
                break
        
        # Accept or reject
        if candidate_c5 < best_c5:
            # Always accept improvements
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            stagnation_count = 0
        elif not is_tabu:
            # Accept with probability based on temperature
            delta = candidate_c5 - best_c5
            if np.random.random() < np.exp(-delta / T):
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                stagnation_count = 0
        else:
            # Tabu - avoid cycling
            pass
        
        # Update tabu list
        if len(tabu_list) >= tabu_size:
            tabu_list.pop(0)
        tabu_list.append(best_h.copy())
        
        # Temperature update
        T = max(T * alpha, T_min)
        
        # Stagnation check
        if stagnation_count >= max_stagnation:
            # Perturb best solution to escape
            perturb_indices = np.random.choice(n_points, int(n_points * 0.1), replace=False)
            for i in perturb_indices:
                best_h[i] += np.random.uniform(-0.05, 0.05)
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            stagnation_count = 0
        
        stagnation_count += 1
        
        # Decrease perturbation size over iterations
        if iteration % 50 == 0:
            perturbation_size = max(perturbation_size * 0.95, 0.05)
    
    # Final refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation
    final_sum = np.sum(best_h)
    assert abs(final_sum - n_points / 2.0) < 1e-6, f"Sum constraint violated: {final_sum}"
    assert all(0 <= h <= 1 for h in best_h), "Bounds violated"
    
    return (list(best_h), best_c5, n_points)
