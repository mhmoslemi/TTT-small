import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Leverage a known good structure
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Algorithm parameters
    pop_size = 60
    generations = 200
    elite_size = 5
    mutation_rate = 0.1
    crossover_rate = 0.7
    adaptive_mutation = True
    
    # Initialize population
    population = []
    fitness = []
    
    # Add elite from initial solution
    population.append(h_start)
    fitness.append(best_c5)
    
    # Generate rest of population
    for i in range(pop_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # Sort population by fitness
    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]
    fitness = [fitness[i] for i in sorted_indices]
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection: tournament
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, 5, replace=False)
            winner = min(tournament, key=lambda x: fitness[x])
            selected.append(population[winner])
        
        # Crossover and mutation
        new_pop = population[:elite_size]  # Elitism
        
        for i in range(elite_size, pop_size, 2):
            if np.random.rand() < crossover_rate:
                parent1, parent2 = selected[i], selected[i+1]
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = selected[i], selected[i+1]
                
            # Mutate
            if adaptive_mutation:
                mutation_rate = max(0.01, mutation_rate * 0.995)
                
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.02)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.02)
            
            # Repair
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_pop.extend([child1, child2])
        
        # Evaluate new population
        population = new_pop[:pop_size]
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    # Final L-BFGS refinement of best solutions
    top_candidates = []
    for i in range(min(5, len(population))):
        h = population[i].copy()
        top_candidates.append(h)
    
    for h in top_candidates:
        # Test with scipy minimize
        try:
            res = minimize(evaluate_erdos_solution, h, args=(dx,), method='L-BFGS-B', 
                           bounds=[(0, 1) for _ in range(n_points)],
                           options={'maxiter': 50})
            if res.success:
                h_refined = res.x
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
        except:
            pass
    
    # Final safety check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
