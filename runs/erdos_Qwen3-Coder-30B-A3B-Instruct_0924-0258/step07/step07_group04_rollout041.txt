import numpy as np
from scipy.optimize import differential_evolution, minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a low-amplitude sine wave to avoid local optima
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.1 * np.sin(2 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    h_start = h_start * (n_points/2) / np.sum(h_start)  # normalize sum constraint
    
    # Define bounds for each variable (0 to 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Differential Evolution parameters
    # Use a larger population and more iterations to improve global search
    de_maxiter = 500
    de_popsize = 50
    
    result_de = differential_evolution(
        objective,
        bounds,
        maxiter=de_maxiter,
        popsize=de_popsize,
        seed=seed,
        disp=False,
        polish=True,
        init='latinhypercube'
    )
    
    # Extract the best result from DE
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Local refinement with SLSQP for better precision
    # Add explicit constraint for sum(h) = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    # Refine with SLSQP using DE solution as starting point
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds_slsqp, constraints=cons_sum,
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    # Final validation and ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
