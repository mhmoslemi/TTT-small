import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive step size
    max_iterations = 500
    step_size = 0.01
    tolerance = 1e-6
    prev_c5 = float('inf')
    
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient direction
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step
        h_new = h - step_size * gradient
        
        # Enforce constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step_size = min(0.01, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.5)
            
        h = h_new.copy()
        
        # Early stopping condition
        if abs(prev_c5 - max_overlap) < tolerance:
            break
        prev_c5 = max_overlap
    
    # Final local refinement with L-BFGS
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        return max_overlap, gradient.flatten()
    
    try:
        from scipy.optimize import fmin_l_bfgs_b
        best_h_flat = best_h.flatten()
        bounds = [(0, 1) for _ in range(n_points)]
        refined_h, _, _ = fmin_l_bfgs_b(objective_and_gradient, best_h_flat,
                                        bounds=bounds, maxiter=200)
        refined_h = refined_h.reshape(n_points)
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = refined_h * (n_points/2) / np.sum(refined_h)
        c5_refined = evaluate_erdos_solution(refined_h, dx)
        if c5_refined < best_c5:
            best_h = refined_h
            best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
