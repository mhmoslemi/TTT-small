import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a carefully designed initial construction
        x = np.linspace(0, 2, n_points)
        h_start = np.zeros(n_points)
        # Create a piecewise constant profile
        for i in range(n_points):
            if 0.5 <= x[i] <= 1.5:
                h_start[i] = 1.0
            else:
                h_start[i] = 0.0
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimize using Nelder-Mead combined with local refinement
    max_iter = 1000
    eval_count = 0
    
    def objective(h):
        nonlocal eval_count
        eval_count += 1
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
        
    def constraint_bounds(h):
        return np.concatenate([h, 1.0 - h])
    
    # Use a combination of bounds and simplex optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample from the current best using perturbation
        h_trial = best_h.copy()
        # Add small random perturbations
        noise_scale = 0.05
        noise = np.random.normal(0, noise_scale, n_points)
        h_trial += noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        
        # Evaluate
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
            
        # Occasionally run a local minimizer
        if iter_num % 50 == 0 and iter_num > 0:
            # Create a slightly different starting point for local search
            h_local_start = np.random.uniform(0, 1, n_points)
            h_local_start = h_local_start * (n_points/2) / np.sum(h_local_start)
            
            try:
                res = minimize(
                    objective, 
                    h_local_start, 
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50}
                )
                if res.success:
                    c5_local = evaluate_erdos_solution(res.x, dx)
                    if c5_local < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_local
            except:
                pass
                
    # Final SQP refinement with CVXPY if needed
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Introduce a proxy for maximization
        C5 = cp.Variable()
        for i in range(n_points):
            # For k = 0 to n_points-1, compute correlation terms
            corr_term = cp.sum(cp.multiply(h_opt[:n_points-i], 1 - h_opt[i:]))
            constraints.append(corr_term <= C5 * dx + 1e-8)
            
        prob = cp.Problem(cp.Minimize(C5), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Check final feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
