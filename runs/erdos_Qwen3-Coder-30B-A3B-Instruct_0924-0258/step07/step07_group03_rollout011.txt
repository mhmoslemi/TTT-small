import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()

    # Convex relaxation using SDP
    def solve_convex_relaxation(h_init, n_points, dx):
        # Prepare SDP formulation
        h = cp.Variable(n_points)

        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
            Z == h @ h.T,

        # The objective will be more complex to handle the full correlation
        # For now let's use a simplified closest approximation approach
        # This is a simplified formulation, a proper SDP for the exact alignment

        return h, constraints
