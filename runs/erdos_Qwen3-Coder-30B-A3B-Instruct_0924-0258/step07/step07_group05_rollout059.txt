import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    generation = 0
    stall_count = 0
    max_stall = 20
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        if generation % 100 == 0:
            if stall_count > max_stall:
                # Restart with initial construction
                population = []
                for i in range(pop_size):
                    if 'initial_h_values' in globals():
                        h = h_start + np.random.normal(0, 0.05, n_points)
                    else:
                        h = np.ones(n_points) * 0.5 + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                    population.append(h)
                stall_count = 0
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            best_in_tournament = min(tournament, key=lambda x: fitness[x])
            selected.append(population[best_in_tournament].copy())
        
        # Crossover and Mutation
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Create offspring
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(pop_size, 2, replace=False)
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, selected[parent1], selected[parent2])
            else:
                parent = np.random.choice(pop_size)
                child = selected[parent].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
                for i in mut_indices:
                    child[i] += np.random.normal(0, 0.03)
                child = np.clip(child, 0, 1)
                # Rescale to maintain sum constraint
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Periodic convex relaxation
        if generation % 50 == 0:
            try:
                # Find top candidates
                top_indices = np.argsort(fitness)[:10]
                for idx in top_indices:
                    h_candidate = population[idx]
                    # Solve convex relaxation using CVX
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add some constraints to improve relaxation
                    for k in range(1, n_points//2):
                        if k < n_points:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                    
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
            except Exception:
                pass

    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
