from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    max_overlap_initial = 0.38085
    tighten_factor = 0.995
    max_iter = 50

    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
    
    # Use cvxpy for optimization with bisection
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create variables
        h_vars = cp.Variable(n_points)

        # Using a relaxed feasible region for the bilinear products
        # For now, we just solve to see if we can decrease max overlap
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,

        ]
