import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h = h * (n_points/2) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient-based optimization with adaptive step
    max_iter = 200
    step_size = 0.01
    tolerance = 1e-6
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            k = (j + l_max) % n_points
            gradient[j] = -2 * dx * (1 - h[k] - h[j])
        
        # Update
        h_new = h - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            step_size = min(0.015, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.5)
            
        h = h_new
        
        if step_size < 1e-6:
            break
    
    # Local refinement with L-BFGS-B
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(n_points)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        gradient = np.zeros(n_points)
        for j in range(n_points):
            k = (j + l_max) % n_points
            gradient[j] = -2 * dx * (1 - h[k] - h[j])
        
        return max_overlap, gradient.flatten()
    
    try:
        from scipy.optimize import fmin_l_bfgs_b
        best_h_flat, _, _ = fmin_l_bfgs_b(
            objective_and_gradient, 
            best_h.flatten(),
            bounds=[(0, 1) for _ in range(n_points)],
            maxiter=200
        )
        best_h = best_h_flat.reshape(n_points)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
