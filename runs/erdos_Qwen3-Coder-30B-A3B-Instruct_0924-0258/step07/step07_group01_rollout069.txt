import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with best known construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize alternating optimization
    a = np.ones(n_points) * 0.5  # amplitude
    s = np.ones(n_points)       # support (binary)
    s[:n_points//2] = 0         # initialize with first half active
    
    # Normalize initial support
    a = a * (n_points/2) / np.sum(a)
    h = a * s
    
    # Alternating optimization
    max_iter = 500
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fix s, optimize a
        try:
            # Use CVXPY for convex optimization
            a_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                cp.sum(cp.multiply(a_var, s)) == n_points/2,
                a_var >= 0,
                a_var <= 1
            ]
            
            # Objective: minimize max overlap
            # For simplicity in CVXPY, we'll use a heuristic proxy
            # Instead, we'll compute overlap directly in numpy and fine-tune
            # This is done via gradient descent approach below for better integration
            
            # Refine current a using SLSQP
            def obj_a(a_test):
                h_test = a_test * s
                return evaluate_erdos_solution(h_test, dx)
                
            def con_sum_a(a_test):
                return np.sum(a_test * s) - n_points/2
                
            # SLSQP optimization
            bounds = [(0,1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum_a}
            
            res = minimize(obj_a, a, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                a = res.x.copy()
                
        except Exception:
            pass
            
        # Enforce constraints on a
        a = np.clip(a, 0, 1)
        a = a * (n_points/2) / np.sum(a * s) if np.sum(s) > 0 else a
        
        # Fix a, optimize s
        # Set s to top n_points/2 entries of a
        top_k = np.argsort(a)[::-1][:n_points//2]
        s_new = np.zeros(n_points)
        s_new[top_k] = 1
        
        # Adjust to satisfy sum constraint
        h_new = a * s_new
        h_adjusted = h_new.copy()
        current_sum = np.sum(h_adjusted)
        if abs(current_sum - n_points/2) > 1e-5:
            # Linear adjustment to match integral
            diff = n_points/2 - current_sum
            # Scale up the top k elements to match the sum
            sorted_indices = np.argsort(a)[::-1]
            adjusted_indices = sorted_indices[:n_points//2]
            total_a = np.sum(a[adjusted_indices])
            if total_a > 0:
                scale_factor = (n_points/2) / total_a
                h_adjusted[adjusted_indices] = a[adjusted_indices] * scale_factor
            else:
                h_adjusted = a * s_new
                
        # Ensure h is valid
        h_adjusted = np.clip(h_adjusted, 0, 1)
        a_new = h_adjusted.copy()
        a_new = a_new / np.max(a_new) if np.max(a_new) > 0 else a_new
        s_new = np.zeros(n_points)
        s_new[np.argsort(a_new)[::-1][:n_points//2]] = 1
        
        # Re-update a and s consistently
        a = a_new
        s = s_new
        
        # Final update of h after re-computing
        h = a * s
        
        # Check if we need to update
        c5_current = evaluate_erdos_solution(h, dx)
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
            
    # Final refinement with SLSQP
    h_final = best_h.copy()
    h_final = h_final * (n_points/2) / np.sum(h_final)  # ensure integral
    h_final = np.clip(h_final, 0, 1)
        
    def objective(h_test):
        return evaluate_erdos_solution(h_test, dx)
            
    def constraint_sum(h_test):
        return np.sum(h_test) - n_points / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res = minimize(objective, h_final, method='SLSQP',
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 500, 'ftol': 1e-10})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
