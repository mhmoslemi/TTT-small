import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Phase 1: Gradient-based refinement with adaptive learning rate
    h = h_start.copy()
    learning_rate = 0.1
    max_iter = 1000
    stagnation = 0
    best_c5_phase1 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        max_lag_idx = np.argmax(corr)
        max_lag = lags[max_lag_idx]
        
        if max_lag_idx < n_points - 1:
            # Right-side correlation
            k = max_lag_idx
            gradient = np.zeros(n_points)
            for i in range(n_points):
                if i + max_lag >= 0 and i + max_lag < n_points:
                    gradient[i] = 1 - 2 * h[i + max_lag]
        else:
            # Left-side correlation
            k = max_lag_idx
            gradient = np.zeros(n_points)
            for i in range(n_points):
                if i - max_lag >= 0 and i - max_lag < n_points:
                    gradient[i] = 1 - 2 * h[i - max_lag]
        
        # Update h
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint via scaling
        h = h * (n_points / 2) / np.sum(h)
        
        # Perturbation every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            perturb_idx = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h[perturb_idx] += np.random.uniform(-0.1, 0.1, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5_phase1:
            best_c5_phase1 = current_c5
            h_best_phase1 = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 100:
                learning_rate *= 0.5
                stagnation = 0
        
        # Early exit if improvement is too slow
        if learning_rate < 1e-8:
            break
    
    # Adjust final h to best from phase 1 
    h = h_best_phase1.copy() if 'h_best_phase1' in locals() else h
    
    # Phase 2: SQP refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Apply SLSQP with callbacks
    try:
        res = minimize(objective, h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
