import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    # Start with initial solution and mutated versions
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    last_improvement = 0
    generation = 0
    max_generations = 200
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Evaluate fitness of each individual
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            last_improvement = generation
        
        # Select top 20% as parents
        num_parents = max(5, population_size // 5)
        parents = population[:num_parents]
        
        # Generate offspring through crossover and mutation
        offspring = []
        while len(offspring) < population_size - num_parents:
            # Select two random parents
            parent1, parent2 = random.sample(parents, 2)
            # Arithmetic crossover
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if random.random() < 0.3:  # 30% mutation rate
                noise = np.random.normal(0, 0.03, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            
            # Normalize to correct integral
            child = child * (n_points/2) / np.sum(child)
            
            offspring.append(child)
        
        # Replace worst individuals with offspring
        population[num_parents:] = offspring
        
        generation += 1
        
        # Early stopping if no improvement for a while
        if generation - last_improvement > 50:
            break
    
    # Local refinement with SLSQP for top candidates
    top_candidates = population[:5]
    for h_candidate in top_candidates:
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
