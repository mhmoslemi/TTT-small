import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Augmented Lagrangian parameters
    max_iter = 5000
    convergence_tol = 1e-6
    alpha = 0.1
    mu = 100.0
    trust_region_factor = 1.1
    
    # Gradient-based optimization with ALM
    h_current = best_h.copy()
    prev_c5 = best_c5
    lagrange_mult = 0.0
    
    def compute_gradient_and_max_overlap(h):
        # Compute all overlaps and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        
        # Compute gradient w.r.t. active constraint (this is simplified)
        n = len(h)
        grad = np.zeros(n)
        
        # Only the active overlap contributes to gradient
        # Extract the difference pattern from correlation
        if max_idx < n:  # k <= 0 case
            # This is a simplified gradient calculation based on the peak overlap
            # For each position i, the derivative with respect to h_i is affected by:
            # .. (1 - h_{i+k}) for i+k < n .. 
            k = max_idx  # This is simplified, in practice we need more precise computation
            for i in range(n):
                if i + k < n:
                    grad[i] += 1 - h[i + k]  # derivative of h_i * (1 - h_{i+k})
                if i - k >= 0:
                    grad[i] += -h[i - k]  # derivative of h_{i-k} * (1 - h_i)
            grad = grad / n
        else:  # k > 0 case
            k = max_idx - n + 1
            for i in range(n):
                if i + k < n:
                    grad[i] += 1 - h[i + k]
                # No need to also include negative contribution for convenience
        return grad, max_overlap * dx
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current gradient and max overlap
        grad, current_c5 = compute_gradient_and_max_overlap(h_current)
        
        # ALM step
        lagrangian_grad = grad + lagrange_mult + mu * (h_current - 1.0)
        
        # Backtracking line search
        step_size = alpha
        for _ in range(10):
            h_proposed = h_current - step_size * lagrangian_grad
            h_proposed = np.clip(h_proposed, 0, 1)
            h_proposed = h_proposed * (n_points / 2) / np.sum(h_proposed)  # Scale to sum constraint
            
            proposed_c5 = evaluate_erdos_solution(h_proposed, dx)
            
            # Accept if improvement or within tolerance
            if proposed_c5 < current_c5 or step_size < 1e-6:
                h_current = h_proposed
                break
            else:
                step_size *= 0.5
                
        # Update Lagrange multiplier and penalty
        constraint_violation = np.sum(h_current) - n_points / 2
        lagrange_mult += mu * constraint_violation
        if abs(constraint_violation) > 1e-3:
            mu *= 2.0
        else:
            mu *= 0.9
            
        # Trust-region adaptation
        if abs(current_c5 - prev_c5) < convergence_tol or step_size < 1e-6:
            alpha *= trust_region_factor
            if alpha > 1.0:
                alpha = 1.0
        else:
            alpha /= trust_region_factor
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        prev_c5 = current_c5
        
        # Occasionally do final polishing with higher penalty
        if iter_num % 500 == 0 and iter_num > 0:
            mu = 1e4
            for _ in range(10):
                grad, _ = compute_gradient_and_max_overlap(h_current)
                h_proposed = h_current - 0.05 * grad
                h_proposed = np.clip(h_proposed, 0, 1)
                h_proposed = h_proposed * (n_points / 2) / np.sum(h_proposed)
                h_current = h_proposed
                
    # Final refinement with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
        
    return (list(best_h), best_c5, n_points)
