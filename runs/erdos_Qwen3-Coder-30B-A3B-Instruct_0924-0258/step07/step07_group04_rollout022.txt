import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth starting function
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with perturbations
    max_iter = 1000
    gamma = 0.1  # learning rate
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Focus on central region for better accuracy
        mid = len(corr) // 2
        # Maximum overlap in a reasonable lag range
        window_start = max(0, mid - 50)
        window_end = min(len(corr), mid + 50)
        max_lag_idx = np.argmax(corr[window_start:window_end]) + window_start
        
        # Direction: gradient direction
        # Direction to move: (1 - h) - diagonal element due to max correlation
        direction = np.zeros(n_points)
        # Create the diagonal matrix for the maximum lag index
        lag = max_lag_idx - mid
        if lag >= 0:
            # correlation is right shift by 'lag' of h with (1-h) 
            direction[lag:] = 1 - best_h[:-lag]
        else:
            direction[:lag] = 1 - best_h[-lag:]
            
        # Frank-Wolfe step
        new_h = best_h - gamma * direction
        # Clip to feasible region
        new_h = np.clip(new_h, 0, 1)
        # Project to correct sum
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(new_h, dx)
        
        # Accept if improvement
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        else:
            # Occasionally perturb to escape local minima
            if iteration % 50 == 0:
                perturb = np.random.normal(0, 0.02, n_points)
                new_h = best_h + perturb
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                c5_new = evaluate_erdos_solution(new_h, dx)
                if c5_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = c5_new

    # Refinement with SLSQP to polish the solution
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 1000, 'ftol': 1e-12})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
