import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points/2) / np.sum(h)
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization with adaptive step size
    max_iterations = 500
    step_size = 0.01
    tolerance = 1e-6
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags (including negative shifts)
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        l_max = np.argmax(overlaps)
        
        # Compute gradient at the maximum overlap lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + l_max) % n_points
            gradient[i] = -2 * dx * (1 - h[j] - h[i])
        
        # Update step
        h_new = h - step_size * gradient
        
        # Enforce bounds and constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept the step if it improves
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            step_size = min(0.015, step_size * 1.2)
        else:
            step_size = max(1e-6, step_size * 0.5)
            
        # Early stopping
        if abs(c5_new - best_c5) < tolerance:
            break
    
    # Final refinement with L-BFGS-B
    try:
        from scipy.optimize import minimize
        
        def objective_and_grad(h):
            h = h.reshape(n_points)
            overlaps = np.correlate(h, 1 - h, mode='full') * dx
            max_overlap = np.max(overlaps)
            l_max = np.argmax(overlaps)
            
            # Gradient computation
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = (i + l_max) % n_points
                gradient[i] = -2 * dx * (1 - h[j] - h[i])
                
            return max_overlap, gradient.flatten()
        
        # Optimize with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective_and_grad, best_h, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            best_h = res.x.reshape(n_points)
            best_h = best_h * (n_points/2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
