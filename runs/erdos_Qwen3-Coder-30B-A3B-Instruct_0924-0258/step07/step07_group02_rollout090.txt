import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Construct better starting point from known good structures
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(2 * np.pi * x / (2 / n_points)))
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization: try multiple randomized initializations
    for _ in range(10):  # Try 10 random restarts
        if time() - start_time > budget_s * 0.95:
            break
            
        # Slightly perturb initial solution
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local optimization using L-BFGS-B
        def objective(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(
                objective,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 500}
            )
            if res.success:
                c5_new = res.fun
                if c5_new < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Final convex optimization refinement
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Compute approximate max correlation lag from current best
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_lags = np.argsort(corr)[-10:]
        
        # Add constraints to bound specific correlations (simplified version)
        for k in top_lags:
            if k < n_points:
                # Simplified correlation sum constraint based on the fixed lag
                left = np.arange(0, min(n_points - k, n_points))
                right = np.arange(k, min(n_points, n_points + k))
                if len(left) > 0 and len(right) > 0:
                    corr_sum = cp.sum(cp.multiply(h_opt[left], 1 - h_opt[right]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
        
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception as e:
        pass

    # Final normalization to ensure sum(h) = n_points/2
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
