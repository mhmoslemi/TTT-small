import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Benders decomposition approach
    max_iter = 5000
    convergence_tol = 1e-6
    perturbation_mag = 0.05
    perturbation_decay = 0.95
    
    # Initialize
    h_current = best_h.copy()
    c5_current = best_c5
    previous_c5 = float('inf')
    
    # Track constraints
    constraints_added = []
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Extract the relevant overlaps (shifts from 0 to n_points-1)
        overlaps = corr[n_points-1:] * dx  # This gives us all shifts k = 0 to n_points-1
        
        # Find the worst shift
        k_worst = np.argmax(overlaps)
        max_overlap = overlaps[k_worst]
        
        # Add constraint if not already present
        if k_worst not in constraints_added:
            constraints_added.append(k_worst)
            
            # Create new constraint for this k
            new_constraint = {
                'type': 'ineq',
                'fun': lambda h, k=k_worst: max_overlap - np.sum(h[:n_points-k_worst] * (1 - h[k_worst:])) * dx
            }
            
        # Solve with convex relaxation
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add existing constraints
            for k in constraints_added:
                if k < n_points:
                    constraint_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(constraint_sum <= best_c5 * dx + 1e-6)
            
            # Set objective to minimize a small perturbation to satisfy the constraint
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            
            if prob.status == cp.OPTIMAL and prob.value < 1e10:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    h_current = h_feasible
                    c5_current = c5_feasible
                    best_h = h_feasible
                    best_c5 = c5_feasible
            else:
                # Fallback to just adding the constraint one way
                pass
        except Exception:
            pass
        
        # Perturb current solution
        h_perturbed = h_current + perturbation_mag * np.random.randn(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Rescale to meet constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate and potentially update
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            h_current = h_perturbed
            c5_current = c5_perturbed
            best_h = h_perturbed
            best_c5 = c5_perturbed
            
        # Update perturbation magnitude
        perturbation_mag *= perturbation_decay
        
        # Check convergence
        if abs(c5_current - previous_c5) < convergence_tol:
            break
        previous_c5 = c5_current
    
    # Final polishing with local optimization
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
