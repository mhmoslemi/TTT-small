import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_feasible(h_start.copy(), n_points)
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.02
    penalty_weight = 1e-3
    max_iter = 5000
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Annealing main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Find dominant lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_lag = np.argmax(corr)
        
        # Gradient computation
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + dominant_lag) % n_points
            grad[i] = -(1 - h_current[j] - h_current[i])
        
        # Penalty gradient
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(h_current)
        penalty_grad += 2 * penalty_weight * (h_sum - n_points / 2)
        penalty_grad += penalty_weight * (h_current > 1) - penalty_weight * (h_current < 0)
        
        # Candidate
        h_candidate = h_current + eta * (-grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Accept or reject
        new_c5 = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.random():
            h_current = h_candidate
        else:
            # Add random perturbation to escape local minima
            perturbation = np.random.normal(0, 0.02, n_points)
            h_current = project_feasible(h_current + perturbation, n_points)
            
        T *= alpha
        
        # Random restart if no improvement for a while
        if no_improvement_count > max_no_improvement:
            h_current = project_feasible(np.random.random(n_points) * 0.5, n_points)
            no_improvement_count = 0
            
    # Multi-resolution refinement
    resolutions = [60, 120]
    for n_points_refine in resolutions:
        if time() - start_time > budget_s * 0.98:
            break
            
        # Interpolate
        x_orig = np.linspace(0, 2, len(h_current))
        x_new = np.linspace(0, 2, n_points_refine)
        h_interpolated = np.interp(x_new, x_orig, h_current)
        h_refined = project_feasible(h_interpolated, n_points_refine)
        dx_refined = 2.0 / n_points_refine
        
        # Local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx_refined)
            
        # Use L-BFGS-B
        bounds = [(0, 1) for _ in range(n_points_refine)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_refine / 2}]
        
        try:
            res = minimize(objective, h_refined, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-6})
            if res.success:
                h_opt = project_feasible(res.x, n_points_refine)
                c5_opt = evaluate_erdos_solution(h_opt, dx_refined)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
    
    if best_h is None:
        best_h = h_current
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    return (list(best_h), best_c5, len(best_h))
