import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Optimization using a trust-region approach
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 5000
    iteration = 0
    
    # Create a history of improvement
    last_improvement = 0
    patience = 10
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Adaptive sampling: smaller steps for local refinement
        noise_magnitude = max(0.001 * np.exp(-iteration / 100), 1e-5)
        noise = noise_magnitude * np.random.normal(0, 1, n_points)
        h_noisy = h + noise
        h_noisy = project_h(h_noisy)
        
        # Local optimization
        current_c5 = compute_c5(h)
        new_c5 = compute_c5(h_noisy)
        
        if new_c5 < current_c5:
            h = h_noisy
        else:
            # If no improvement, try again with smaller noise
            noise_small = 0.0001 * np.random.normal(0, 1, n_points)
            h_small = h + noise_small
            h_small = project_h(h_small)
            small_c5 = compute_c5(h_small)
            if small_c5 < current_c5:
                h = h_small
        
        # Update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Convergence check
        if last_improvement > patience:
            # Try a larger jump in case of getting stuck
            h = project_h(h + 0.01 * np.random.normal(0, 1, n_points))
            last_improvement = 0
            
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
