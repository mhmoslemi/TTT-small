import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    
    # Ensure final sum is exact
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_clamp = 0.05
    
    # Get initial solution
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Generate initial swarm around h0
    for i in range(swarm_size):
        if i == 0:
            h = h0.copy()
        else:
            # Perturb the best solution found so far
            h = h0 + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points / 2.0)
        
        # Ensure valid sum
        h = project_to_feasible(h, n_points / 2.0)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Store personal bests
    pbest = [swarm[i].copy() for i in range(swarm_size)]
    pbest_values = [evaluate(h, 2.0/n_points) for h in pbest]
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            v_new = (omega * velocities[i] + 
                    c1 * r1 * (pbest[i] - swarm[i]) + 
                    c2 * r2 * (gbest - swarm[i]))
            
            # Clamp velocity
            v_max = velocity_clamp * (1.0 - np.min(swarm[i]))
            v_min = velocity_clamp * np.min(swarm[i])
            v_new = np.clip(v_new, v_min, v_max)
            
            # Update position
            swarm[i] += v_new
            swarm[i] = project_to_feasible(swarm[i], n_points / 2.0)
            
            # Evaluate
            current_value = evaluate(swarm[i], 2.0/n_points)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = swarm[i].copy()
                    gbest_value = current_value
                    
            velocities[i] = v_new.copy()
            
        iteration += 1
    
    return (gbest.tolist(), float(gbest_value), n_points)
