import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search setup
    C_low = 0.37
    C_high = 0.38085
    best_feasible_h = None
    best_feasible_C = C_high
    
    # Binary search iterations
    iterations = 0
    max_iterations = 40
    
    # Solver setup
    solver = 'ECOS'
    
    while iterations < max_iterations and (time() - start_time) < budget_s * 0.9:
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate CVXPY problem
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add constraint for each k
        for k in range(1 - n_points, n_points):
            # Compute circular convolution constraint
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= C_mid * n_points / 2)
        
        # Minimize zero (feasibility check)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=solver, feastol=1e-6, verbose=False)
            if prob.status == 'optimal':
                h_opt = h_var.value
                if h_opt is not None:
                    c5_val = evaluate_erdos_solution(h_opt, dx)
                    if c5_val < best_feasible_C:
                        best_feasible_C = c5_val
                        best_feasible_h = h_opt.copy()
                    C_high = C_mid
                else:
                    C_low = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
        
        iterations += 1
    
    # If we found a feasible solution, refine it with gradient descent
    if best_feasible_h is not None:
        best_h = best_feasible_h
        
        # Refinement with gradient descent
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def proj(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            return h
        
        # First project the feasible solution to ensure feasibility
        best_h = proj(best_h)
        
        # Gradient descent refinement
        for i in range(1000):  # reasonable number of iterations
            if time() - start_time > budget_s * 0.95:
                break
                
            # Simple gradient approximation
            eps = 1e-5
            grad = np.zeros_like(best_h)
            base_val = obj(best_h)
            
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = proj(h_plus)
                grad[j] = (obj(h_plus) - base_val) / eps
            
            # Update with small step
            step_size = 1e-3
            best_h = best_h - step_size * grad
            best_h = proj(best_h)
            
            # Re-evaluate
            new_c5 = obj(best_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
    
    return (list(best_h), best_c5, n_points)
