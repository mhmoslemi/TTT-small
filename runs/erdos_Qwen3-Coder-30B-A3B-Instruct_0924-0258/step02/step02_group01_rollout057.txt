import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hyperparameters
    pop_size = 30
    max_iter = 100
    perturb_rate = 0.2
    noise_std = 0.1
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # Evolutionary optimization
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        new_population = []
        new_fitness = []
        
        for i, h in enumerate(population):
            # Perturb
            h_pert = h.copy()
            indices = np.random.choice(n_points, size=int(perturb_rate * n_points), replace=False)
            h_pert[indices] += np.random.normal(0, noise_std, len(indices))
            h_pert = project_to_feasible(h_pert, n_points)
            
            # Optimize
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    new_population.append(h_opt)
                    new_fitness.append(c5_opt)
                else:
                    new_population.append(h_pert)
                    new_fitness.append(evaluate_erdos_solution(h_pert, dx))
            except:
                new_population.append(h_pert)
                new_fitness.append(evaluate_erdos_solution(h_pert, dx))
        
        # Selection: keep best individuals
        sorted_indices = np.argsort(new_fitness)
        population = [new_population[i] for i in sorted_indices[:pop_size]]
        fitness = [new_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= 20:
            break
    
    # Resolution refinement
    resolutions = [50, 70, 100]
    for res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
        if res <= n_points:
            continue
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_to_feasible(h_interp, res)
        dx_new = 2.0 / res
        best_h = h_interp
        n_points = res
        best_c5 = evaluate_erdos_solution(best_h, dx_new)
    
    return (list(best_h), best_c5, n_points)
