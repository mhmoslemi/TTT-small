import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction or use default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Global best
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization
    population_size = 30
    population = [project_to_feasible(np.random.rand(n_points), n_points) 
                  for _ in range(population_size)]
    population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Find best in population
    min_idx = np.argmin(population_c5)
    if population_c5[min_idx] < best_c5:
        best_h = population[min_idx].copy()
        best_c5 = population_c5[min_idx]
    
    # Optimization parameters
    max_iterations = 200
    perturbation_rate = 0.2
    noise_std = 0.1
    patience = 20
    
    # Optimization loop
    iterations_no_improvement = 0
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        new_population = []
        for h in population:
            # Apply perturbation
            h_pert = h.copy()
            indices = np.random.choice(n_points, 
                                      size=int(n_points * perturbation_rate),
                                      replace=False)
            h_pert[indices] += np.random.normal(0, noise_std, len(indices))
            h_pert = project_to_feasible(h_pert, n_points)
            
            # Local optimization with SLSQP
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            try:
                res = minimize(obj, h_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                else:
                    h_opt = h_pert
                    c5_opt = evaluate_erdos_solution(h_pert, dx)
            except:
                h_opt = h_pert
                c5_opt = evaluate_erdos_solution(h_pert, dx)
            
            new_population.append(h_opt)
            
        # Evaluate the new population
        new_c5 = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Update best
        min_idx = np.argmin(new_c5)
        if new_c5[min_idx] < best_c5:
            best_h = new_population[min_idx].copy()
            best_c5 = new_c5[min_idx]
            iterations_no_improvement = 0
        else:
            iterations_no_improvement += 1
            
        # Replace population
        population = new_population
        population_c5 = new_c5
        
        if iterations_no_improvement >= patience:
            break
    
    # Refinement: Increase resolution
    if len(best_h) < 500 and time() - start_time < budget_s * 0.9:
        fine_n = min(1000, len(best_h) * 2)
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, fine_n)
        h_fine = np.interp(x_new, x_orig, best_h)
        h_fine = project_to_feasible(h_fine, fine_n)
        
        # Final optimization at higher resolution
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, 2.0 / len(h_vec))
        
        bounds = [(0, 1) for _ in range(fine_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - fine_n / 2}]
        try:
            res = minimize(obj, h_fine, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_fine = project_to_feasible(res.x, fine_n)
                c5_fine = evaluate_erdos_solution(h_fine, 2.0 / fine_n)
                if c5_fine < best_c5:
                    best_h = h_fine
                    best_c5 = c5_fine
                    n_points = fine_n
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
