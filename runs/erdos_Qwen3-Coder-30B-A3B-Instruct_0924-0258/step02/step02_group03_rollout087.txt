import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

def trust_region_step(h, dx, grad, trust_radius, target_sum):
    """Perform trust region step with projected gradient descent."""
    # Create a projected gradient descent step
    step = -grad * dx
    
    # Clip step to trust region
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    
    new_h = h + step
    # Project to feasible region
    new_h = project_to_feasible(new_h, target_sum)
    return new_h

def compute_gradient(h, dx, max_k_index):
    """Compute gradient of C5 at h."""
    n = len(h)
    grad = np.zeros(n)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # find which k achieves max
    k_offset = n - 1  # the center
    max_k = max_k_index - k_offset  # k in range [-n+1, n-1]
    
    # Gradient is simpler to compute for specific k:
    # For k-th element in corr, we need pos k in h, and pos k in 1-h
    # where function value is h[i]*(1-h[i+k])
    # grad wrt h[i] is (1 - h[i+k]) if k > 0, -h[i+k] if k < 0
    # Actually for correlation, it's direct from computing gradient of
    # sum over all i and k of h[i]*(1-h[i+k])
    
    # For i=0 to n-1, contribution to d/dh_i of correlation k is
    # when current i position is compared to i+k position
    # for k=-n+1 to n-1, there's only k for which i+k is valid
    
    # We compute
    # grad[i] = sum over k of  d/dh_i [h[i]*(1-h[i+k])] for correlation k
    #         = 1 - h[i+k] if k=i+k0 and 0 o.w.
    # So for each i we scan which k leads to valid h[i+k]
    
    # Special case of setting a finite difference
    for i in range(n):
        if max_k > 0:  # i+k = i+max_k, h[i+k] exists
            if i + max_k < n:
                grad[i] = -(1 - h[i + max_k])  # or = -1 + h[i + max_k]
        elif max_k < 0:  # i+k = i+max_k, h[i+k] exists
            if i + max_k >= 0:
                grad[i] = h[i + max_k]
        else:
            grad[i] = 0.0  # just skip
    
    # return gradient for the direction of max_k
    return grad

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    h_current = h0.copy()
    h_current = project_to_feasible(h_current, target_sum)
    
    trust_radius = 0.2
    max_trust_radius = 1.0
    min_trust_radius = 0.01
    best_h = None

    while time.time() - start_time < budget_s - 2:

        max_k_index = np.argmax(corr)
        # Access correlation value at that index
        # This is easiest to compute; the actual gradient is in different way
        # We now tune a problem that computes gradients (not gradient descent,
