import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Perform blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def gaussian_mutation(individual, mutation_rate, std=0.1):
    """Apply Gaussian mutation to an individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        mutated = individual + noise
        return mutated
    return individual

    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8

    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Calculate initial fitness
    fitness = []
    for individual in population:
        dx = 2.0 / n_points
        fit = evaluate_erdos_solution(individual, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    gen = 0
    stagnant = 0
    max_stagnant = 20
    
    while time() - start_time < budget_s * 0.95:
        gen += 1
        if gen * time_per_gen > (budget_s - (time() - start_time)):
            break
            
        # Create new population
        new_population = []
        new_fitness = []
        
        # Keep top 25% from current population
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size//4]
        for i in top_indices:
            new_population.append(population[i].copy())
            new_fitness.append(fitness[i])
        
        # Generate rest through evolution
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitness, tournament_k)
            parent2 = tournament_selection(population, fitness, tournament_k)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
                
            # Mutation
            offspring1 = gaussian_mutation(offspring1, mutation_rate)
            offspring2 = gaussian_mutation(offspring2, mutation_rate)

            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            # Evaluate
            dx = 2.0 / n_points
            fit1 = evaluate_erdos_solution(offspring1, dx)
            fit2 = evaluate_erdos_solution(offspring2, dx)

            new_fitness.extend([fit1, fit2])
            
        # Keep best individuals
        combined_fitness = new_fitness[:population_size]

            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnant = 0
        else:
            stagnant += 1
            
        if stagnant >= max_stagnant:
            break
            
    return (list(best_h), best_c5, n_points)
