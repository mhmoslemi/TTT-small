import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize a simplex of particles"""
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    particles = [project_to_feasible(h0, n_points)]
    
    # Generate m-1 additional points by moving along random directions
    for _ in range(m - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h_new = project_to_feasible(particles[0] + noise, n_points)
        particles.append(h_new)
    
    return particles

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia = 0.7
    cogn_bias = 2.0
    soc_bias = 2.0
    shake_threshold = 50
    grad_refine_interval = 200
    max_shake_attempts = 10
    
    # Initialize swarm
    particles = initialize_simplex(n_points, swarm_size)
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Evaluate initial particles
    fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Track best
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    iteration = 0
    no_improvement = 0
    
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            cogn_bias * r1 * (pbest[i] - particles[i]) + 
                            soc_bias * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            new_fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best if needed
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
                    no_improvement = 0
            else:
                no_improvement += 1
        
        # Apply shaking if no improvement
        if no_improvement > shake_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            
            # Reset counter
            no_improvement = 0
            
            # Expand swarm slightly
            if swarm_size < 50:
                swarm_size += 5
                while len(particles) < swarm_size:
                    # Expand by adding new random particles
                    noise = np.random.normal(0, 0.1, n_points)
                    new_particle = project_to_feasible(gbest + noise, n_points)
                    particles.append(new_particle)
                    velocities.append(np.zeros(n_points))
                    pbest.append(new_particle)
                    pbest_fitness.append(evaluate_erdos_solution(new_particle, dx))
        
        # Periodically apply gradient refinement
        if iteration % grad_refine_interval == 0 and iteration > 0:
            # Refine top 10 particles
            top_indices = np.argsort(pbest_fitness)[:min(10, len(pbest_fitness))]
            for idx in top_indices:
                h_refined = pbest[idx].copy()
                
                # Define objective for local optimization
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                # Optimize with bounds and constraint
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
                
                try:
                    res = minimize(obj, h_refined, method='L-BFGS-B', 
                                bounds=bounds, constraints=cons, 
                                options={'maxiter': 100})
                    if res.success:
                        h_opt = project_to_feasible(res.x, n_points)
                        new_fitness = evaluate_erdos_solution(h_opt, dx)
                        if new_fitness < pbest_fitness[idx]:
                            pbest[idx] = h_opt.copy()
                            pbest_fitness[idx] = new_fitness
                            
                            if new_fitness < gbest_fitness:
                                gbest = h_opt.copy()
                                gbest_fitness = new_fitness
                except:
                    pass
        
        iteration += 1
    
    # Final refinement on the best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 500})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_fitness = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_fitness, n_points)
