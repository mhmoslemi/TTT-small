import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_frank_wolfe_step(g, n_points):
    # Solve: min_s g·s subject to sum(s) = 0, 0 <= s <= 1
    # This is a constrained linear optimization problem
    # We can write s = t * (x - y) for some x, y in [0,1]^n
    # We have constraints sum(x) - sum(y) = 0 and x,y >= 0, x,y <= 1
    # Using the fact that this is a polytope and the optimal solution  
    # should be at a vertex with some simplification
    
    # Trick: we want to minimize g·s, where s = x - y, sum(x) = sum(y) = n/2, 
    # 0 <= x, y <= 1. One way to do this is to set s = x - y, where x is the 
    # vector with max g components and y is the vector with min g components.
    # But we must satisfy sum(x) = sum(y) = n/2.
    
    # Simpler approach: set s = argmin g·s subject to sum(s) = 0, 0 <= s <= 1
    # This is linear programming.
    # But note: sum(s) = 0 is a weird constraint. What we want might be:
    # the update direction s such that sum(s) = 0 but we try to minimize g·s.
    # So, assume we have a vector g, want to find s with sum(s) = 0 such that 
    # g·s is minimized.
    # We can do this without constraints as s = - (sum(g) / n) * ones + given_vector
    # That's not right either. Let's fix interpretation.
    # To make an update, usually what we do is take a direction s where sum(s) = 0
    # and update h_next = h + gamma * s. Therefore: sum(s) = sum(h + gamma * s) - sum(h) = 0.
    # This leads to sum(s) = 0 as an equality constraint.
    
    # Proper Frank-Wolfe: minimize g·s under sum(s) = 0, 0 <= s <= 1.
    # We can find a vertex of this polyhedron by setting some components
    # to the bounds and deducing others. This is computationally complex.
    # Simpler heuristic: set the largest g components to 1 and others to 0 and adjust
    # to make the sum 0.
    
    s = np.zeros(n_points)
    # Compute a descending sort indices for g
    idx_desc = np.argsort(-g)
    idx_asc = np.argsort(g)
    
    # Estimate required number of +1/-1 components:
    # Consider sum(g) = sum(g+) - sum(g-) for positive and negative g entries
    # AAHH. wait. we're not constrained by sum(s) = 0 in the canonical sense.
    # We are trying to find a feasible descent direction of the dual problem.
    # Paper reference: eliminate projected gradient and use Frank-Wolfe correctly 
    
    # Actually we seek s that satisfies the constrained optimization (Frank-Wolfe or its worst-case)
    # Because we want h_{new} = h + gamma * s where sum(s) = 0 and 0 <= h + gamma * s <= 1
    # This is difficult to do in general, so try a simpler version:
    # Let s = 1 where g is negative, -1 where g is positive, then scale to satisfy sum(s) = 0.
    # This is a reasonable estimate.
    s = np.zeros(n_points)
    # Place large negative values of g: 
    sorted_g = np.sort(g)[::-1]
    negative_g_idx = np.argsort(-g)
    
    negative_idx = []
    positive_idx = []
    
    for i in range(n_points):
        if g[i] < 0:
            negative_idx.append(i)
        else:
            positive_idx.append(i)
    
    # Move part of negative parts to positive to balance sums:
    # Approximate solution
    s = np.zeros(n_points)
    # This is O(n) approximation but region-specific
    # Just to proceed:
    s[negative_idx[:n_points//2]] = 1
    s[positive_idx[:n_points//2]] = -1

    return s

def frank_wolfe_step(h, n_points, dx, max_iter=80):
    c5_prev = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = c5_prev
    
    for i in range(max_iter):
        if i % 10 == 0 and i > 0:
            # Optional periodic improvement
            pass
            
        # Compute overlap for all shifts
        corr = np.correlate(h, 1-h, mode='full')
        # Find where the maximum occurs
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)  # Corresponding shift for index offset
        c5 = corr[max_idx] * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute subgradient: dC/dh[i] = (1 - h[(i-k)%n]) - h[i]
        # Only critical subgradients where the maximum is achieved
        # For simplicity, compute it for the maximum k
        g = np.zeros(n_points)
        for i in range(n_points):
            g[i] = (1 - h[(i - k) % n_points]) - h[i]
            
        # Solve Frank-Wolfe optimization step
        s = solve_frank_wolfe_step(g, n_points)
        
        # Try different γ in [0.1, 0.9] using backtrack
        best_gamma = 0.5
        best_here_c5 = c5
        for gamma in np.linspace(0.1, 0.9, 9):
            h_new = h + gamma * s
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_here_c5:
                best_here_c5 = c5_new
                best_gamma = gamma
        h = h + best_gamma * s
        h = project_to_feasible(h, n_points)
        if evaluate_erdos_solution(h, dx) < best_c5:
            best_h = h.copy()
            best_c5 = evaluate_erdos_solution(h, dx)
    
    return best_h, best_c5
