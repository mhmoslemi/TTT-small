import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population with variations around initial_h_values or simple pattern"""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base)), base)
    else:
        base = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        perturbation = np.random.normal(0, 0.1, n_points)
        individual = project(base + perturbation, n_points)
        population.append(individual)
    
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    pop_size = 50
    n_points = 30
    dx = 2.0 / n_points
    max_generations = 1000
    stagnation_limit = 200
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    
    best_fitness = min(fitness)
    best_individual = population[np.argmin(fitness)].copy()
    
    # Track stagnation
    last_improvement = 0
    mutation_rate = 0.2
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover
        offspring = []
        for i in range(0, len(selected), 2):
            if i+1 < len(selected) and np.random.rand() < 0.7:
                child1 = crossover(selected[i], selected[i+1])
                child2 = crossover(selected[i+1], selected[i])
                offspring.extend([child1, child2])
            else:
                offspring.append(selected[i])
                if i+1 < len(selected):
                    offspring.append(selected[i+1])
        
        # Mutation and project
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i], mutation_rate, 0.1)
        
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(ind, dx) for ind in offspring]
        
        # Replace worst individuals
        combined_population = population + offspring
        combined_fitness = fitness + offspring_fitness
        
        # Sort by fitness and keep top pop_size
        sorted_indices = np.argsort(combined_fitness)[:pop_size]
        population = [combined_population[i] for i in sorted_indices]
        fitness = [combined_fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best_fitness = min(fitness)
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[np.argmin(fitness)].copy()
            last_improvement = generation
            mutation_rate = max(0.1, mutation_rate * 0.99)
        else:
            mutation_rate = min(0.5, mutation_rate * 1.01)
        
        # Check for stagnation
        if generation - last_improvement > stagnation_limit:
            # Reset mutation rate and possibly reset population
            mutation_rate = 0.2
            population = initialize_population(n_points, pop_size)
            fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    
    return (list(best_individual), best_fitness, n_points)
