import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5 bound
    c_low = 0.37
    c_high = 0.38085
    tolerance = 1e-5
    max_iterations = 10
    
    # Feasibility check function using CVX
    def check_feasibility(h_init, c_target):
        # Create variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            if k == 0:
                continue  # Skip zero shift
            # Create overlap sum for shift k
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h[i] * (1 - h[i + k])
            overlap_sum *= dx
            constraints.append(overlap_sum <= c_target)
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return prob.status == cp.OPTIMAL
        except:
            return False
    
    # Refinement using gradient descent
    def refine_solution(h_init, c_target):
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            # Add penalty to make sure it stays within bounds
            penalty = 0
            if max_corr > c_target:
                penalty = 1e6 * (max_corr - c_target)
            return max_corr + penalty
        
        # Project to feasible region
        def project(h_proj):
            h_proj = np.clip(h_proj, 0, 1)
            h_sum = np.sum(h_proj)
            if h_sum > 0:
                h_proj = h_proj * (n_points / 2) / h_sum
            return h_proj
        
        # Simple gradient descent refinement
        h = h_init.copy()
        for _ in range(500):
            if time() - start_time > budget_s - 5:
                break
            h = project(h)
            grad = np.zeros_like(h)
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            # Approximate gradient using finite differences
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                max_plus = np.max(corr_plus)
                grad[i] = (max_plus - np.max(corr)) / eps
            
            h = h - 0.001 * grad
            h = project(h)
            c5 = evaluate_erdos_solution(h, dx)

            refined_h = refine_solution(best_h, c_mid)

                best_h = refined_h.copy()
                best_c5 = refined_c5
        else:

    # Final refinement
    final_h = refine_solution(best_h, c_high)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
