import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation for upsample
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    # Initialization
    current_h = h.copy()
    best_c5 = compute_overlap(current_h)
    best_h = current_h.copy()
    
    # Convex-concave procedure (CCCP)
    max_iter = 50
    tolerance = 1e-5
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve a convex subproblem using CVXPY
        # Set up convex optimization problem
        h_vars = cp.Variable(n_points)
        
        # Formulate the overlap constraints
        # We approximate max over all k via a single constraint
        # Compute correlation with current h (fixed) to build matrix
        h_current = current_h
        
        # Compute all correlations to set up quadratic problem
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        mid = len(corr_full) // 2
        corr_vals = corr_full[mid-n_points+1:mid+n_points-1] * dx
        
        # Find the maximum correlation value and the indices
        max_idx = np.argmax(corr_vals)
        max_val = corr_vals[max_idx]
        
        # For now, approximate by using the constraint that maximum overlap
        # cannot exceed some value that we try to minimize
        # We treat this as finding min C such that C >= all overlaps
        
        # Build QP formulation
        C = cp.Variable()
        
        # Constraints: each h[i] in [0,1]
        constrs = [h_vars[i] >= 0 for i in range(n_points)]
        constrs += [h_vars[i] <= 1 for i in range(n_points)]
        
        # Constraint on sum
        constrs += [cp.sum(h_vars) == target_sum]
        
        # Constraint for the maximum overlap
        # Approximate by ensuring C is at least the upper bound we compute
        # Also add constraints: for each k in range(2*n_points-1):
        # integral of h(x) * (1 - h(x+k)) dx <= C
        # This is approximately used in the convex formulation
        
        # Make it more precise: we solve the convex subproblem using the full correlation matrix
        # Build the matrix B such that the objective is to minimize max_i (sum_j B[i,j]*h[j])
        # and in the convex problem, we fix h_current at current iteration and solve for new h
        # But we approximate as: min C s.t. sum_ij B[i,j] h[j] <= C for all i
        
        # This approach uses that we can write:
        # For k = n_points-1 + i (with i = 0,1,...,2*n_points-2)
        # integral h(x) * (1 - h(x+k)) dx = sum_j h[j] * (1 - h[j+i])
        # If indexed, sorted by j = 0,1,...,n_points-1 and zipped with j+i, some will be rotated
        # Represent the actual correlations explicitly as above
        
        # Since we're doing CCCP - we build matrix of overlaps 
        # But simpler approach: 
        # We minimize over h that satisfies the constraints and maximizes the overlap integral
        
        # Simplification: Let's directly compute the actual C5 on current h
        # and improve using first-order method with gradient descent on a smooth proxy
        # of the max
        
        # Instead: build matrix of correlations rows, map to element in 2n_points-1 array
        # Matrix [a_{i,j}] where over a_{i,j} = h_current[i]* (1 - h_current[j-k]) 
        # For each k we compute all these overlaps
        # So we solve
        # min h:  max_k integral h(x)*(1-h(x+k)) dx
        # subject to: h[i] in [0,1], sum h == target_sum
        
        # This is the reason to directly use a n^2 formulation
        # We convert to a convex-approximate problem by replacing in the overlap integral
        # \int h(x) (1 - h(x+\tau)) dx  by a suitable convex function handling max.
        # But this is complex. Less so, we’ll refine with gradient descent or convex relaxation
        # approach where we transfer the constraint: max_k \int h(x)(1-h(x+k))dx <= C
        
        # Let's rebuild and start with solving convex subproblem (iteratively simple method):
        # We create one increasing sequence. Focusing on good graph for performance and restarts
        
        # Relaxation and optimization approach: try simple gradient-based method for better convergence
        # First, there are up to 2*n_points - 1 distinct overlaps. 
        # We'll maintain gradient estimates with small finite differences
        # Fast iterative descent with project to feasible set
        
        # Dropout steps and then regular descent
        h_temp = current_h.copy()
        learning_rate = 0.1
        
        # Gradient descent loop 
        for step in range(200):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Approximate gradient as finite difference in the resulting overlap function
            gradient = np.zeros_like(h_temp)
            eps = 1e-4
            
            base_val = compute_overlap(h_temp)
            
            for i in range(n_points):
                h_plus = h_temp.copy()
                h_plus[i] += eps
                h_plus = project_feasible(h_plus)
                val_plus = compute_overlap(h_plus)
                
                h_minus = h_temp.copy()
                h_minus[i] -= eps
                h_minus = project_feasible(h_minus)
                val_minus = compute_overlap(h_minus)
                
                gradient[i] = (val_plus - val_minus) / (2 * eps)
                
            # Update step
            h_temp -= learning_rate * gradient
            h_temp = project_feasible(h_temp)
            
            # Check stopping condition
            new_val = compute_overlap(h_temp)
            if abs(new_val - base_val) < tolerance:
                break
                
        current_h = h_temp.copy()
        new_c5 = compute_overlap(current_h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Subsample to increase diversity (fewer iterations for speed)
        if iteration % 3 == 0 and iteration > 0:
            # Resample part of solution to break potential local minima
            indices = np.random.choice(n_points, size=n_points//3, replace=False)
            for i in indices:
                current_h[i] = np.random.random() * 0.5 + 0.25
        
        # Early stopping when no significant improvement
        if last_improvement > 5:
            break
    
    return (best_h.tolist(), best_c5, n_points)
