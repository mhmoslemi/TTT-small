import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to meet target sum
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    n_particles = 100
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, target_sum)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize particles around h0 with random noise
    particles = []
    for _ in range(n_particles):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h)
    
    # Initialize
    best_h = None
    best_c5 = float('inf')
    temperature = 1.0
    proposal_sigma = 0.1
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s and last_improvement < 50:
        iteration += 1
        
        # Importance sampling
        weights = []
        new_particles = []
        acceptance_count = 0
        
        for particle in particles:
            # Sample new particle
            h_new = particle + np.random.normal(0, proposal_sigma, n_points)
            h_new = project_to_feasible(h_new, target_sum)
            
            # Compute weight (importance sampling)
            c5_new = evaluate(h_new, dx)
            weight = np.exp(-c5_new / temperature)
            weights.append(weight)
            
            new_particles.append(h_new)
            
            if c5_new < evaluate(particle, dx):
                acceptance_count += 1
        
        # Normalize weights
        if sum(weights) > 0:
            weights = np.array(weights) / sum(weights)
        else:
            weights = np.ones(len(weights)) / len(weights)
        
        # Resampling (top 50% by weight)
        indices = np.argsort(weights)[::-1][:n_particles//2]
        particles = [new_particles[i] for i in indices]
        
        # Add mutated particles
        for i in range(len(particles)):
            for _ in range(2):  # 2 mutations per particle
                # Global mutation
                h_mut = particles[i] + np.random.normal(0, proposal_sigma, n_points)
                h_mut = project_to_feasible(h_mut, target_sum)
                particles.append(h_mut)
        
        # Keep best particles
        particle_scores = [evaluate(h, dx) for h in particles]
        best_indices = np.argsort(particle_scores)[:n_particles//10]
        particles = [particles[i] for i in best_indices]
        
        # Update best solution
        min_c5 = min(particle_scores)
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = particles[np.argmin(particle_scores)].copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Adapt temperature and proposal sigma
        temperature *= 0.99
        if acceptance_count > 0.3 * n_particles:
            proposal_sigma *= 1.05
        elif acceptance_count < 0.1 * n_particles:
            proposal_sigma *= 0.95
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h0, target_sum)
        final_c5 = evaluate(final_h, dx)
        return (final_h.tolist(), float(final_c5), n_points)
