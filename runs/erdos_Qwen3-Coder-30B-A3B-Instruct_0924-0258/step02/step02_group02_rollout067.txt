import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Initial coarse grid evaluation to get a starting point
    max_shifts = 10
    shift_indices = np.linspace(0, n_points-1, max_shifts, dtype=int)
    
    # Estimate initial worst case
    worst_shift = 0
    best_c5 = float('inf')
    for i in shift_indices:
        # indices in the full correlation
        s = i
        if s >= n_points:
            continue
        corr = np.sum(h * (1 - np.roll(h, s)))
        corr *= dx
        if corr > best_c5:
            best_c5 = corr
            worst_shift = i
    
    # Refined grid around worst case for cutting plane
    fine_grid_size = 100
    coarse_shifts = np.linspace(max(0, worst_shift - fine_grid_size//2), 
                                min(n_points-1, worst_shift + fine_grid_size//2), 
                                fine_grid_size, dtype=int)
    all_constraints = []
    
    # CVX optimization setup
    h_var = Variable(n_points)
    eta = Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, sum(h_var) == target_sum, eta >= 0]
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Main optimization loop
    iteration = 0
    max_iter = 500  # reasonable limit
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Current h
        current_h = h_var.value if h_var.value is not None else h
        
        # Evaluate numerous shifts to find worst case
        shifts_to_check = coarse_shifts[coarse_shifts < n_points]
        max_c_k = -1
        worst_k = 0
        
        for k in shifts_to_check:
            # Shifted h
            shifted_h = np.roll(current_h, k)
            corr = np.sum(current_h * (1 - shifted_h)) * dx
            if corr > max_c_k:
                max_c_k = corr
                worst_k = k
                
        # Add constraint for actual worst shift found
        if worst_k >= n_points:
            break
            
        shift_vec = np.zeros(n_points)
        shift_vec[worst_k] = 1
        shifted_h = np.roll(current_h, worst_k)
        current_overlap = np.sum(current_h * (1 - shifted_h)) * dx
        
        # Formulate and solve the CVX problem
        if current_overlap <= eta.value:
            # If this constraint already satisfied, just refine
            continue
            
        # Solve optimization with new worst case constraint
        constraints_new = constraints + [eta >= current_overlap]
        prob = Problem(Minimize(eta), constraints_new)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
        except:
            # fallback to basic projection
            break
            
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            new_h = h_var.value
            new_eta = eta.value
            h = project_h(new_h)
            if new_eta < best_c5:
                best_c5 = new_eta
                best_h = h.copy()
        
        iteration += 1
        if iteration > max_iter:
            break
            
    if best_h is not None:
        final_h = project_h(best_h)
        final_c5 = compute_c5(final_h)
        return (final_h.tolist(), final_c5, n_points)
    else:
        final_c5 = compute_c5(h)
        return (h.tolist(), final_c5, n_points)
