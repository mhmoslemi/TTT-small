import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, gamma_min=0.1, gamma_max=0.9):
    """Frank-Wolfe optimization step for the Erdős problem"""
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute subgradient from this maximum overlap
        # The overlap at shift k is sum h[i] * (1 - h[(i + k) % n])
        # So d/dh[i] = (1 - h[(i - k) % n]) - h[(i - k) % n] 
        #  = 1 - 2 * h[(i - k) % n]  
        # For i fixed, this is 1 - 2 * h[(i - k) % n]
        # but we want to find the derivative w.r.t. h[i]
        # Special case approach: different i give different terms
        # Let shift k be the index of maximum overlap
        k = max_idx - (n_points - 1)  # Map indices back to actual shifts
        # Rule: overlap = sum_{i=0}^{n-1} h[i] * (1 - h[(i+k)%n]) 
        # d/dh[i] = (1 - h[(i+k)%n]) 
        # This derivative might contribute to other h values via reflection...
        # We can compute total subgradient for elements that contribute
        subgradient = np.zeros_like(h)
        
        for i in range(n_points):
            # Difference is sum_{j} h[j] * (1 - h[(j+k)%n]) 
            # Take derivative w.r.t. h[i]
            # I only contribute if j = i, so: 
            subgradient[i] = 1 - h[(i + k) % n]
        
        # This time we should be more careful about adding 1
        # What we actually want is s = arg min (<g,s> : s >= 0, sum(s) = sum(h))
        # The subgradient is dC/dh[i] = 1 - 2 * h[(i-k)%n] for the k_th overlap
        
        # So for the current maximum shift k, the derivative w.r.t. h[i] is:
        #  1 - h[(i - k) % n]  where this is from h[i] term in overlap
        # Let's compute one correctly:
        # Shift k generates overlap = sum_{i} h[i] * (1 - h[(i+k)%n])
        # dC/dh[i] = 1 - h[(i+k)%n] if the term h[i] * (1-h[(i+k)%n]) is included
        # This gives: dC/dh[i] = 1 - h[(i-k) % n] 
        # But better approach:
        # Let's make a new array to correctly map the overlap contributions:
        subgradient = np.zeros_like(h)
        max_overlap_indices = np.where(corr == max_val)[0]
        # For robustness, consider top few shifts if there are ties
        top_k_percent = max(1, len(max_overlap_indices) // 10)  # Take top 10%
        chosen_indices = max_overlap_indices[-top_k_percent:]  # Use highest indices (larger k shift)
        for key_shift in chosen_indices:
            shift = key_shift - (n_points - 1)
            for i in range(n_points):
                subgradient[i] += 1.0 - h[(i - shift) % n_points]
        
        # Normalize subgradient to comparable scale
        subgradient /= len(chosen_indices)
        
        # Solve constrained linear program: min <g,s> subject to s >= 0, sum(s) = sum(h)
        # The direction of descent is given by the solution to this:
        # s = argmin_{s >= 0, sum(s) = n/2} <g,s>
        # IR keeps at least one of the components of the gradient. 
        # The formula for the Frank-Wolfe direction:
        # s_* = argmin_{s} <g, s> s.t. 0 <= s <= 1, sum(s) = n/2
        # Then the direction is: s - h
        # min_{0 <= s <= 1, sum(s) = n/2} g.T * s
        # This is min_{s} g.T * s with constraint sum s = n/2, s_i in [0,1]  
        # This can be solved by: set s_i = 0 if g[i] > 0 and s_i = 1 if g[i] < 0 
        # then adjust to satisfy sum(s) = n/2
        # Yes, it's min_0 <= s <= 1, sum s = n/2
        # And min g.T * s
        # So similar to argmin_{x \in [0,1]^n} <g, x> s.t. sum x = n/2
        # The solution is s_i = 1 if g[i] < 0, 0 if g[i] > 0, then project to sum = n/2
        
        direction = np.zeros_like(h)
        for i in range(n_points):
            direction[i] = 1.0 if subgradient[i] < 0 else 0.0
            
        # project to sum = n/2
        current_sum = np.sum(direction)
        if current_sum > 0:
            direction = direction * (n_points / 2) / current_sum
        else:
            direction = np.ones_like(h) * (n_points / 2) / n_points
        
        # NewFrank-Wolfe update 
        # In our case, h_new = h + gamma * (s - h)  
        # Or equivalently h_new = (1-gamma) * h + gamma * s 
        # The step size gamma can be tuned via linesearch to find best descent
        
        # Try a simple backtracking
        best_gamma = gamma_min
        best_h_new = h.copy()
        best_c5_new = best_c5
                
        for gamma in np.linspace(gamma_min, gamma_max, 5):
            h_new = (1 - gamma) * h + gamma * direction
            # Project and normalize
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5_new:
                best_c5_new = c5_new
                best_h_new = h_new.copy()
                best_gamma = gamma
                
        h = best_h_new
        if best_c5_new < best_c5:
            best_c5 = best_c5_new
            best_h = h.copy()
        
    return best_h, best_c5

    # Try to avoid local minima with multiple restarts

        n_points = 25

            h_start[i] = 0.5 if (i % 4) < 2 else 0.25
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Try different resolutions
    # Try to improve by using different discretizations
    resolutions = [25, 30, 35]
    
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            # Interpolate from previous resolution (or start with initial)
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            
        # Use Frank-Wolfe strategy for optimization
        # Need to try multiple random restarts at each resolution
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Small perturbation
            h_pert = h_start + np.random.normal(0, 0.05, n_points)
            h_pert = project_to_feasible(h_pert, n_points)
                
            h_opt, c5_opt = frank_wolfe_step(h_pert, n_points, dx_try, max_iter=30)
                
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
                
        h_start = best_h.copy()
    
    # Optional fine tune at highest resolution
    if time() - start_time < budget_s * 0.9:
        # Try with several more iterations at highest resolution
        dx_finetune = 2.0 / best_n_points
        h_finetune, c5_finetune = frank_wolfe_step(best_h, best_n_points, dx_finetune, max_iter=50)
        if c5_finetune < best_c5:
            best_h = h_finetune.copy()
            best_c5 = c5_finetune
    
    return (list(best_h), best_c5, best_n_points)
