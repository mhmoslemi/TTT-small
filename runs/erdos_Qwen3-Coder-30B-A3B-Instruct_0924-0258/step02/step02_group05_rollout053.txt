import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    # Project to feasible region
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2.0) / np.sum(h_vec)
        return h_vec
    
    # Compute C5 overlap function
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Use convolution from scipy
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Extract the relevant part
        mid = len(conv) // 2
        overlaps = conv[mid - n_points + 1 : mid + n_points - 1]
        return np.max(overlaps) * (2.0 / n_points)
    
    # Differential Evolution parameters
    pop_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 200
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        population.append(project_feasible(individual))
    
    # Evaluate initial population
    fitness = [compute_C5(ind) for ind in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_C5 = fitness[best_idx]
    
    # Evolution loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        new_population = []
        new_fitness = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Differential mutation
            mutant = a + mutation_factor * (b - c)
            mutant = project_feasible(mutant)
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            
            trial = np.array(trial)
            
            # Evaluate trial
            trial_C5 = compute_C5(trial)
            
            # Selection
            if trial_C5 < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_C5)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_C5:
            best_C5 = fitness[best_idx]
            best_h = population[best_idx].copy()
    
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
