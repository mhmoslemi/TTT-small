import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, backtracking_steps=10):
    """Frank-Wolfe optimization step with backtracking line search"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        if iteration % 10 == 0:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        max_shifts = np.where(corr == np.max(corr))[0]
        # Get the indices of maximum shifts
        max_k = max_shifts[0] if len(max_shifts) > 0 else 0
        
        # Construct subgradient for the maximum overlap
        gradient = np.zeros_like(h)
        for k in max_shifts[:min(10, len(max_shifts))]:  # Take top few
            k_idx = k % n_points
            for i in range(n_points):
                gradient[i] += (1 - h[(i - k_idx) % n_points]) - h[i]
        gradient /= len(max_shifts[:min(10, len(max_shifts))])
        
        # Solve linearized problem: min g^T s subject to sum(s) = 0, 0 <= s <= 1
        # This is equivalent to min g^T s subject to sum(s) = 0, s_i in [0,1]
        
        # Simple approach: find direction that decreases gradient most
        # Using v = argmin (g^T s | s in convex hull of extreme points)
        # Since extrema are coordinate vectors, we try gradient descent direction
        # But more carefully: use Frank-Wolfe approach
        
        # For simple Frank-Wolfe, we'll use a projected gradient step  
        s = np.zeros_like(h)
        # We want s such that s_i - h_i has sign matching gradient, within bounds
        # A simple approach: directional gradient method but project step
        step_size = 0.1
        trial_s = h - step_size * gradient
        # Project trial_s to [0,1] range, then rescale to sum to n_points/2
        trial_s = np.clip(trial_s, 0, 1)
        trial_s = project_to_feasible(trial_s, n_points)
        
        # Backtracking line search to find best step along direction (trial_s - h)
        best_trial = h
        best_trial_c5 = evaluate_erdos_solution(h, dx)
        for i in range(backtracking_steps):
            gamma = 0.9 ** i
            new_h = h + gamma * (trial_s - h)
            new_h = project_to_feasible(new_h, n_points)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_trial_c5:
                best_trial_c5 = new_c5
                best_trial = new_h.copy()
        
        # Update h
        h = best_trial.copy()
    
    return h, best_c5

    best_n_points = n_points
    
    # Try different small resolutions
    resolutions = [25, 30, 35]
    
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)),

            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Multiple restarts
        for restart in range(2):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slight perturbation of current best
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            
            # Apply Frank-Wolfe optimization
            h_opt, c5_opt = frank_wolfe_step(h_restart, n_points, dx_try, max_iter=40, backtracking_steps=5)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points

    # Final refinement at full resolution if needed
    if time() - start_time < budget_s * 0.9:
        # Try one final optimization at highest resolution  
        dx_fine = 2.0 / best_n_points
        h_opt, c5_opt = frank_wolfe_step(best_h, best_n_points, dx_fine, max_iter=30, backtracking_steps=5)

            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    return (list(best_h), best_c5, best_n_points)
