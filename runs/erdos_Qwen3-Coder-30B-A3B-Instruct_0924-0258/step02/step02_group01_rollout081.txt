import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(49):
        h = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iter = 100
    patience = 20
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        new_population = []
        for h in population:
            # Perturb with Gaussian noise on 20% of elements
            h_perturbed = h.copy()
            indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
            h_perturbed[indices] += np.random.normal(0, 0.1, len(indices))
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Optimize with SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        no_improvement_count = 0
                        if best_c5 < 0.3807:
                            return (list(best_h), best_c5, n_points)
                    new_population.append(h_opt)
                else:
                    new_population.append(h)
            except:
                new_population.append(h)
        
        population = new_population
        no_improvement_count += 1
        if no_improvement_count >= patience:
            break
    
    # Resolution refinement
    current_h = best_h.copy()
    current_n = n_points
    while current_n < 1000 and time() - start_time < budget_s * 0.95:
        current_n *= 2
        x_orig = np.linspace(0, 2, len(current_h))
        x_new = np.linspace(0, 2, current_n)
        current_h = np.interp(x_new, x_orig, current_h)
        current_h = project_to_feasible(current_h, current_n)
        
        dx = 2.0 / current_n
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(current_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - current_n / 2}]
        
        try:
            res = minimize(obj, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                current_h = project_to_feasible(res.x, current_n)
                current_c5 = evaluate_erdos_solution(current_h, dx)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    if best_c5 < 0.3807:
                        return (list(best_h), best_c5, current_n)
        except:
            pass
    
    return (list(best_h), best_c5, len(best_h))
