import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    eta = 0.01  # Learning rate
    penalty_weight = 1e-3
    
    # Optimization loop
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95:
        # Compute current C5 and find dominant lag
        current_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Simple gradient calculation (simplified for speed)
        # For C5 = max_k ∫ h(x)(1 - h(x+k)) dx, estimate gradient for dominant k
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        dominant_k = np.argmax(corr)
        
        # Gradient approximation: ∇h C5 ≈ (1 - h[(i+k) % n_points] - h[i]) * dx
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + dominant_k) % n_points
            grad[i] = (1 - best_h[j] - best_h[i]) * dx
        
        # Local update with gradient ascent
        h_candidate = best_h - eta * grad
        
        # Add penalty for constraint violation
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(best_h)
        penalty_grad += 2 * penalty_weight * (h_sum - n_points/2)
        
        h_candidate = h_candidate - eta * penalty_grad
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on SA criterion
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h = h_candidate.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Update temperature
        T *= alpha
        
        # Random perturbation if no improvement for too long
        if no_improvement_count > max_no_improvement:
            noise = np.random.normal(0, 0.05, n_points)
            h_candidate = project_feasible(best_h + noise, n_points)
            best_h = h_candidate.copy()
            no_improvement_count = 0

        # Occasionally refine resolution if time allows
        if iterations % 50 == 0 and time() - start_time < budget_s * 0.8:

            n_points_new = min(n_points * 2, 60)
            if n_points_new > n_points:

                h_interp = np.interp(x_new, x_old, best_h)
                h_interp = project_feasible(h_interp, n_points_new)
                best_h = h_interp.copy()
                n_points = n_points_new
    
    # Final evaluation and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
