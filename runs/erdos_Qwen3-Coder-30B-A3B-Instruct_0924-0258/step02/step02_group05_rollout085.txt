import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution from parent construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO Parameters
    swarm_size = 40
    max_iter = 2000
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    vel_max = 0.15
    
    # Initialize swarm
    swarm = []
    for i in range(swarm_size):
        if i == 0:
            particle = h.copy()
        else:
            # Random perturbation around initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = project_feasible(particle)
        swarm.append(particle)
    
    # Track best solutions
    particle_best = [p.copy() for p in swarm]
    particle_best_scores = [compute_overlap(p) for p in particle_best]
    global_best = particle_best[np.argmin(particle_best_scores)].copy()
    global_best_score = min(particle_best_scores)
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(swarm_size):
            particle = swarm[i]
            pbest = particle_best[i]
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            v = w * (particle - pbest) + c1 * r1 * (global_best - particle) + c2 * r2 * (pbest - particle)
            
            # Clip velocity
            v = np.clip(v, -vel_max, vel_max)
            
            # Update position
            new_pos = particle + v
            new_pos = project_feasible(new_pos)
            
            # Evaluate new position
            new_score = compute_overlap(new_pos)
            
            # Update personal best
            if new_score < particle_best_scores[i]:
                particle_best[i] = new_pos.copy()
                particle_best_scores[i] = new_score
                
                # Update global best
                if new_score < global_best_score:
                    global_best = new_pos.copy()
                    global_best_score = new_score
                    
            # Update swarm particle
            swarm[i] = new_pos.copy()
    
    # Final evaluation
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
