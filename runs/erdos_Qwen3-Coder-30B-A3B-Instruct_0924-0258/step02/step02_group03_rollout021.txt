import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    # Annealing parameters
    T0 = 0.5
    alpha = 0.995
    T = T0
    max_iter = 50000
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate(best_h, 2.0 / n_points)
    no_improvement_count = 0
    
    # Acceptance rate tracking
    acceptance_count = 0
    total_attempts = 0
    
    # Main annealing loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate candidate
        h_new = h0 + np.random.normal(0, T, n_points)
        h_new = project_to_feasible(h_new, n_points / 2.0)
        
        # Evaluate
        c5_new = evaluate(h_new, 2.0 / n_points)
        c5_old = evaluate(h0, 2.0 / n_points)
        
        # Accept or reject based on Metropolis criterion
        total_attempts += 1
        if c5_new < c5_old or np.random.random() < np.exp(-(c5_new - c5_old) / T):
            h0 = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            acceptance_count += 1
        else:
            no_improvement_count += 1
            
        # Update temperature
        T = max(T * alpha, 1e-6)
        
        # Adaptive cooling based on acceptance rate
        if total_attempts % 50 == 0 and total_attempts > 0:
            acceptance_rate = acceptance_count / total_attempts
            if acceptance_rate < 0.2:
                alpha = max(alpha * 0.99, 0.99)
            elif acceptance_rate > 0.5:
                alpha = min(alpha * 1.01, 0.999)
            acceptance_count = 0
            total_attempts = 0
            
        # Reset if stuck
        if no_improvement_count > 1000:
            T = min(T * 2.0, 0.5)
            no_improvement_count = 0
    
    final_c5 = evaluate(best_h, 2.0 / n_points)
    return (best_h.tolist(), float(final_c5), n_points)
