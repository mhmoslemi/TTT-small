import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_shifted_overlap(h_vec, shift):
        # Calculate overlap for specific shift
        if shift >= 0:
            return np.sum(h_vec[:-shift] * (1 - h_vec[shift:])) * dx
        else:
            return np.sum(h_vec[-shift:] * (1 - h_vec[:-shift])) * dx

    # Initialize variables
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    learning_rate = 0.01
    max_iter = 10000
    iteration = 0
    last_improvement = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Randomly sample shifts to optimize
        shifts = np.arange(n_points)
        np.random.shuffle(shifts)
        
        # Update h to minimize the maximum overlap
        for shift in shifts[:min(10, len(shifts))]:  # Limit to first 10 shifts
            if shift == 0:
                continue
                
            # Compute gradient of overlap w.r.t h
            grad = np.zeros_like(h)
            if shift > 0:
                # Gradient for h[i] term when computing overlap with shift
                # h[i](1-h[i+shift]) contribution to overlap
                shift_start = shift
                shift_end = len(h) - shift
                grad[shift_start:shift_end] += (1 - h[shift_start+shift:shift_end+shift]) * dx
                grad[0:shift_end-shift] -= h[shift:shift_end] * dx
            else:
                shift_start = -shift
                shift_end = len(h) + shift
                grad[shift_start:shift_end] += (1 - h[shift_start-shift:shift_end-shift]) * dx
                grad[0:shift_end-shift] -= h[-shift:shift_end] * dx
                
            # Gradient descent step
            h_new = h - learning_rate * grad
            h = project_h(h_new)
            
        # Check for improvement
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Adaptive learning rate
        if iteration - last_improvement > 500:
            learning_rate *= 0.1
            last_improvement = iteration
            
        iteration += 1
        if iteration > max_iter:
            break
        
        # Small random perturbation to escape local minima
        if iteration % 20 == 0:
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
