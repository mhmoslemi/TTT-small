        # Use the exact same approach but store the result correctly
        A_eq = np.array([np.ones(n_points)])
        b_eq = np.array([target_sum])
        
        A_ub = []
        b_ub = []
        
        for k in range(-n_points + 1, n_points):

                if 0 <= j < n_points:
                    # From h[i] * (1 - h[j]) dpdx, contributions
