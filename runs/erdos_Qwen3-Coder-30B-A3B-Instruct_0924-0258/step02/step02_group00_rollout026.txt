import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import cvxpy as cp
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_feasibility(h_start, n_points, dx, C):
    """Solve convex feasibility problem for given C"""
    h = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        h >= 0,
        h <= 1
    ]
    
    # Add overlap constraints for all shifts
    for k in range(n_points):
        if k == 0:
            continue  # Skip shift 0
        # For shift k: sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) * dx <= C
        # But with the way numpy is indexing it, for shift k:
        # the indices in h that overlap with h[0] are [0, 1, ..., n-k-1] 
        # the indices in h that overlap with h[k] are [k, k+1, ..., n-1] 
        # h[0] * (1 - h[k]) + h[1] * (1 - h[k+1]) + ... + h[n-k-1] * (1 - h[n-1])
        # which is sum h[i] * (1 - h[i+k]) for i = 0 to n-k-1
        if k < n_points:
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h[i] * (1 - h[i + k])
            if overlap_sum.is_affine():
                constraints.append(overlap_sum * dx <= C)
    
    # Objective: minimize anything (feasibility problem)
    objective = cp.Minimize(0)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C
    C_low = 0.37
    C_high = 0.38085
    C_tolerance = 1e-6
    max_iterations = 10
    
    for _ in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        C_mid = (C_low + C_high) / 2.0
        h_feasible = solve_convex_feasibility(h_start, n_points, dx, C_mid)
        
        if h_feasible is not None:
            # Refine with gradient descent
            refined_h = project_to_feasible(h_feasible, n_points)
            # Evaluate to verify
            c5 = evaluate_erdos_solution(refined_h, dx)
            if c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = c5
            C_high = C_mid
        else:
            C_low = C_mid
    
    # Final refinement
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Use a local solver for further improvement using scipy minimize
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Minimize repeatedly to see if improvement is possible
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        h_perturb = project_to_feasible(final_h + np.random.normal(0, 0.02, n_points), n_points)
        res = minimize(objective, h_perturb, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 30, 'ftol': 1e-6})
        
        if res.success:
            h_opt = project_to_feasible(res.x, n_points)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < final_c5:
                final_h = h_opt.copy()
                final_c5 = c5_opt
    
    return (list(final_h), final_c5, n_points)
