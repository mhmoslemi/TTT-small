import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    
    # Initialize best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [25, 30, 35, 40, 45]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = np.clip(h_interp, 0, 1)
            h_start = h_start * (n_points_try / 2) / np.sum(h_start)
            n_points = n_points_try
        
        # Refine using optimization
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 80)
        
        # Optimization with restarts
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = np.clip(h_start + np.random.normal(0, 0.05, n_points), 0, 1)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            # Use scipy minimize
            def objective(h):
                return evaluate_erdos_solution(h, dx_try)
            
            def constraint_func(h):
                return np.sum(h) - n_points / 2
            
            constraints = {'type': 'eq', 'fun': constraint_func}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_restart, method='SLSQP', 
                          bounds=bounds, constraints=constraints, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx_try)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt

            # Small perturbation to escape local optima
            h_start = np.clip(best_h + np.random.normal(0, 0.02, n_points), 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
