import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Starting point: use initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085
    max_iterations = 10
    best_solution = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use CVXPY for convex formulation
    def is_feasible(h_guess, c_val):
        # Formulate convex problem:
        # minimize 0 s.t. constraints
        # Variables: h[i] for i in [0, n_points)
        h_vars = cp.Variable(n_points)
        constraints = []
        
        # bounds
        constraints += [h_vars[i] >= 0 for i in range(n_points)]
        constraints += [h_vars[i] <= 1 for i in range(n_points)]
        
        # sum constraint
        constraints += [cp.sum(h_vars) == n_points / 2]
        
        # overlap constraints for each k
        # k ranges from 0 to n_points-1 (but correlates only k < n_points)
        # Full correlation uses all shifts
        # np.correlate gives full convolution of length 2*n_points - 1
        # We want the correlation at each shift up to n_points - 1, then another n_points-1
        # But we only care about valid overlaps
        
        for i in range(n_points):
            corr_sum = 0
            for j in range(n_points - i):
                corr_sum += h_vars[j] * (1 - h_vars[j + i])
            # We actually should be constructing the full correlation correctly.
            # Let's recompute our constraints according to actual correlation
            corr_sum = cp.sum(cp.multiply(h_vars[:n_points-i], 1 - h_vars[i:]))
            constraints += [corr_sum * dx <= c_val]
        
        # Actually simplify this - since x correlation of h with 1-h  
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return True
            else:
                return False
        except:
            return False
    
    # Try to find better candidate via gradient descent refitting
    def refine_solution(h_guess, target_c):
        # Refine by minimizing C5 directly using local search
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        def jac(h):
            # Simple numerical gradient for objective (C5) w.r.t h
            eps = 1e-6
            grad = np.zeros_like(h)
            base_obj = objective(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus *= (n_points / 2) / np.sum(h_plus)  # project back to feasible
                grad[i] = (objective(h_plus) - base_obj) / eps
            return grad
        
        try:
            # SLSQP is good for bounds
            res = minimize(objective, h_guess, method='SLSQP', bounds=bounds, 
                           constraints=cons, jac=jac, options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                # Only return if we actually improved it
                if c5_refined < target_c:
                    return h_refined, c5_refined
        except:
            pass
        return h_guess, target_c
    
    # Binary search refinement
    prev_c = None
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        c_mid = (c_low + c_high) / 2.0
        if prev_c is None or abs(c_mid - prev_c) > 1e-6:
            if is_feasible(best_solution, c_mid):
                c_high = c_mid
                # Refine solution
                refined_h, refined_c = refine_solution(best_solution, c_mid)
                if refined_c < best_c5:
                    best_c5 = refined_c
                    best_solution = refined_h.copy()
            else:
                c_low = c_mid
        else:
            break
        prev_c = c_mid
    
    # Final numerical refinement to ensure best local solution
    final_h, final_c = refine_solution(best_solution, c_high)
    if final_c < best_c5:
        best_c5 = final_c
        best_solution = final_h.copy()
    
    return (list(best_solution), best_c5, n_points)
