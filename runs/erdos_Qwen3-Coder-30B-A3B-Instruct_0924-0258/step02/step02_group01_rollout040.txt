import numpy as np
from time import time

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def compute_gradient(h, k, n_points, dx):
    """Compute gradient for the k-th lag in correlation"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = -(1 - h[j] - h[i]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    h = project_feasible(h, n_points)
    
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_pen = 1e-3
    no_improve_count = 0
    max_no_improve = 50
    
    # Resolution sweep
    resolutions = [30, 40, 50, 60, 70, 80]
    current_res = 0
    
    while time() - start_time < budget_s * 0.95:
        if current_res < len(resolutions):
            new_n_points = resolutions[current_res]
            if new_n_points != n_points:
                # Interpolate
                x_orig = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, new_n_points)
                h = np.interp(x_new, x_orig, h)
                n_points = new_n_points
                dx = 2.0 / n_points
            current_res += 1
            
        # Simulated Annealing loop
        for _ in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Find dominant lag
            corr = np.correlate(h, 1 - h, mode='full')
            k = np.argmax(corr) - (n_points - 1)
            
            # Compute gradient for dominant lag
            grad = compute_gradient(h, k, n_points, dx)
            
            # Compute penalty gradient
            penalty_grad = np.zeros(n_points)
            h_sum = np.sum(h)
            penalty_grad += 2 * lambda_pen * (h_sum - n_points / 2)
            
            # Perturb in gradient direction
            h_candidate = h - eta * (grad + penalty_grad)
            
            # Project to feasible region
            h_candidate = project_feasible(h_candidate, n_points)
            
            # Evaluate change
            new_c5 = evaluate_erdos_solution(h_candidate, dx)
            delta_c5 = new_c5 - current_c5
            
            # Accept or reject
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                h = h_candidate
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
                
            # Cooling
            T *= alpha
            
            # Random perturbation if no improvement
            if no_improve_count > max_no_improve:
                h += np.random.randn(n_points) * 0.05
                h = project_feasible(h, n_points)
                T *= 0.9
                no_improve_count = 0
                
    return (list(best_h), best_c5, n_points)
