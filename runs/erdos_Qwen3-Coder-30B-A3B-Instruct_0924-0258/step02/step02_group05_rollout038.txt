import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    velocity_limit = 0.15
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # One particle from initial construction
    swarm.append(h.copy())
    velocities.append(np.zeros_like(h))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h_new = np.random.rand(n_points) * 0.5 + 0.25  # More centered
        h_new = project_feasible(h_new)
        swarm.append(h_new)
        velocities.append(np.random.rand(n_points) * velocity_limit * 2 - velocity_limit)
    
    # Track best solutions
    personal_best = [s.copy() for s in swarm]
    personal_best_scores = [compute_overlap(s) for s in swarm]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) +
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project onto feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            score = compute_overlap(swarm[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = score
                
                # Check if global best
                if score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = score
                    
    return (global_best.tolist(), global_best_score, n_points)
