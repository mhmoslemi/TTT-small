import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization
    population_size = 30
    population = [project_to_feasible(np.random.normal(0.5, 0.1, n_points), n_points) for _ in range(population_size)]
    population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_idx = np.argmin(population_c5)
    if population_c5[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = population_c5[best_idx]
    
    # Evolutionary optimization
    max_iter = 100
    patience = 20
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        new_population = []
        for h in population:
            # Perturb
            h_perturbed = h.copy()
            mask = np.random.rand(n_points) < 0.2
            h_perturbed[mask] += np.random.normal(0, 0.1, np.sum(mask))
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization
            def obj(h_local):
                return evaluate_erdos_solution(h_local, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    new_population.append(h_opt)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        no_improvement_count = 0
                else:
                    new_population.append(h_perturbed)
            except:
                new_population.append(h_perturbed)
                
        population = new_population
        population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
        
        no_improvement_count += 1
        
        # Reseed if no improvement
        if no_improvement_count > patience:
            population = [project_to_feasible(np.random.normal(0.5, 0.1, n_points), n_points) for _ in range(population_size)]
            population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
            no_improvement_count = 0
            
    # Refinement: Try higher resolution
    if n_points < 1000:
        resolution_factor = 2
        while n_points * resolution_factor < 1000 and time() - start_time < budget_s * 0.9:
            n_points_new = n_points * resolution_factor
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_to_feasible(h_interp, n_points_new)
            dx_new = 2.0 / n_points_new
            
            # Optimize at new resolution
            def obj(h):
                return evaluate_erdos_solution(h, dx_new)
                
            bounds = [(0, 1) for _ in range(n_points_new)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_new / 2}]
            
            try:
                res = minimize(obj, h_interp, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = project_to_feasible(res.x, n_points_new)
                    c5_refined = evaluate_erdos_solution(h_refined, dx_new)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        n_points = n_points_new
                else:
                    break
            except:
                break
                
            resolution_factor = 2
            
    return (list(best_h), best_c5, n_points)
