import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use an alternating optimization approach
    # Initialize variables
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 100000
    iteration = 0
    last_improvement = 0
    learning_rate = 0.01
    
    # Create shift indices
    shift_indices = list(range(n_points))  # 0 to n_points-1
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Shuffled order of shifts
        np.random.shuffle(shift_indices)
        
        # Update based on shifts one by one
        for s in shift_indices:
            # Create the shifted version
            h_shifted = np.roll(h, s)
            
            # Compute the current overlap for this shift
            overlap = np.sum(h * (1 - h_shifted)) * dx
            
            # Compute gradients for current shift
            # df/dh with respect to shift s: 
            # This is an approximation for gradient
            # For the shift s, we compute contribution of h[i] to overlap
            # Which depends on h[i] and h[i+s] for valid i
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + s < len(h):
                    grad[i] -= (1 - h[i+s])
                if i - s >= 0:
                    grad[i] -= (1 - h[i-s])
            # Proper gradient and update direction (simplified)
            new_h = h - learning_rate * grad
            new_h = project_h(new_h)
            
            # Check if this improved the overlap for shift s
            # Update h if better
            temp_h = h.copy()
            h = new_h.copy()
            new_overlap = np.sum(h * (1 - np.roll(h, s))) * dx
            if new_overlap < overlap or iteration % 100 == 0:
                # A small probability to escape local minima
                if np.random.rand() < 0.05:
                    h = temp_h.copy()
            
        # Re-compute C5 and update best
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
        else:
            # Decrease learning rate if no improvement
            if iteration - last_improvement > 500:
                learning_rate *= 0.9
                last_improvement = iteration
                
        iteration += 1
        if iteration >= max_iter:
            break

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
            
    return (best_h.tolist(), final_c5, n_points)
