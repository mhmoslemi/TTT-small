import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from existing construction or default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    h_start = project_feasible(h_start, n_points)
    dx = 2.0 / n_points
    
    # Parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    max_iter = 10000
    
    # Initialize Adam optimizer
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    h = h_start.copy()
    last_improvement = 0
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        t += 1
        
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for the dominant lag k
        grad = np.zeros(n_points)
        for j in range(n_points):
            grad[j] = (1 - h[(j + k) % n_points]) - h[j]
        grad *= dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update parameters
        h -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h = project_feasible(h, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        
        # Track best
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = i
        
        # Adaptive learning rate adjustment
        if i - last_improvement > 500:
            learning_rate *= 0.9
            last_improvement = i
            
        # Occasionally restart to escape local minima
        if i % 2000 == 0 and i > 0:
            h = project_feasible(h + np.random.normal(0, 0.05, n_points), n_points)
    
    # Final optimization refinement
    h_final = best_h.copy()
    for _ in range(5):
        if time() - start_time > budget_s * 0.98:
            break
        # Local optimization with scipy
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        res = minimize(obj, h_final, method='SLSQP', bounds=bounds, 
                       constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = project_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    
    return (list(best_h), best_c5, n_points)
