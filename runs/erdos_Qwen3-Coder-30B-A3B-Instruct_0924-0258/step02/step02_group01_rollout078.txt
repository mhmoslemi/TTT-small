import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def crossover_blend(parent1, parent2, alpha=0.5):
    """Blend crossover: average with some randomness"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate_ind(individual, mutation_rate=0.2, mutation_strength=0.1):
    """Add Gaussian noise to selected elements"""
    mask = np.random.random(len(individual)) < mutation_rate
    if not np.any(mask):
        idx = np.random.randint(0, len(individual))
        mask[idx] = True
    noise = np.random.normal(0, mutation_strength, len(individual))
    mutated = individual + mask * noise
    return mutated

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    n_generations = 60
    tournament_k = 5
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Create initial population
    population = []
    population.append(h_start.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points) * 0.5
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic Algorithm
    generation = 0
    stagnation_count = 0
    best_c5_history = []
    
    while time() - start_time < budget_s * 0.95 and generation < n_generations:
        generation += 1
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Keep top 25% (elitism)
        elite_size = pop_size // 4
        new_population = sorted_pop[:elite_size]
        new_fitness = sorted_fit[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitness, tournament_k)
            parent2 = tournament_selection(population, fitness, tournament_k)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = crossover_blend(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate_ind(offspring1, mutation_rate)
            offspring2 = mutate_ind(offspring2, mutation_rate)
            
            # feasibility
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            # Evaluate
            fit1 = evaluate_erdos_solution(offspring1, dx)
            fit2 = evaluate_erdos_solution(offspring2, dx)
            
            new_population.extend([offspring1, offspring2])
            new_fitness.extend([fit1, fit2])
        
        # Keep only pop_size solutions
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Update best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        best_c5_history.append(best_c5)
        
        if stagnation_count >= 20:
            break
    
    # Ensure final solution is feasible
    best_h = project_feasible(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
