import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial_h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic algorithm parameters
    population_size = 100
    num_generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    # Best solution tracking
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness of all individuals
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative for minimization
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i + 1] if i + 1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Simple uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=np.random.randint(1, 3), replace=False)
                for idx in indices:
                    child1[idx] += np.random.normal(0, 0.05)
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=np.random.randint(1, 3), replace=False)
                for idx in indices:
                    child2[idx] += np.random.normal(0, 0.05)
            
            offspring.append(project_h(child1))
            offspring.append(project_h(child2))
        
        # Replace population
        population = offspring[:population_size]
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
