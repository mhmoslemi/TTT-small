import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Binary search on C_5
    c_low = 0.37
    c_high = 0.38085
    c_mid = (c_low + c_high) / 2
    best_feasible_h = None
    max_iter = 40
    iter_count = 0
    
    # Precompute shifted indices for efficiency
    shift_indices = {}
    for k in range(-(n_points - 1), n_points):
        shift_indices[k] = np.arange(n_points)
        shift_indices[k] = np.roll(shift_indices[k], -k)
    
    # Binary search loop
    while iter_count < max_iter and (time() - start_time) < (budget_s * 0.9):
        iter_count += 1
        c_mid = (c_low + c_high) / 2
        
        # Convex feasibility problem
        h_cvx = cp.Variable(n_points)
        constraints = [
            cp.sum(h_cvx) == n_points / 2,
            h_cvx >= 0,
            h_cvx <= 1
        ]
        
        # Add correlation constraints
        for k in range(-(n_points - 1), n_points):
            # Compute the correlation term efficiently
            shift_idx = shift_indices[k]
            corr_sum = cp.sum(cp.multiply(h_cvx, 1 - h_cvx[shift_idx]))
            constraints.append(corr_sum <= c_mid * (n_points / 2))
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INFEASIBLE:
                if best_feasible_h is None or c_mid < best_c5:
                    best_feasible_h = h_cvx.value.copy()
                    best_c5 = c_mid
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
        if (time() - start_time) > (budget_s * 0.95):
            break
    
    # If we found a feasible solution, refine it with gradient descent
    if best_feasible_h is not None:
        h_refined = best_feasible_h.copy()
        dx_ref = 2.0 / n_points
        best_c5 = evaluate_erdos_solution(h_refined, dx_ref)
        
        # Now perform gradient-based refinement using scipy
        def obj(h):
            return evaluate_erdos_solution(h, dx_ref)
            
        def grad(h):
            # Approximate gradient via finite differences
            eps = 1e-5
            grad_val = np.zeros_like(h)
            base_val = obj(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                grad_val[i] = (obj(h_plus) - base_val) / eps
                
            return grad_val
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        # Try a few restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_init = project_to_feasible(h_refined + np.random.normal(0, 0.01, n_points), n_points)
            try:
                res = minimize(obj, h_init, method='SLSQP', bounds=bounds, 
                               constraints=cons, jac=grad, options={'maxiter': 50})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx_ref)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        h_refined = h_opt.copy()
            except:
                pass
        best_h = h_refined
        
    return (list(best_h), best_c5, n_points)
