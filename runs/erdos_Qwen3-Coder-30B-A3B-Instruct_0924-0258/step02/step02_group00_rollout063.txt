import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def compute_correlation_gradient(h, n_points, dx):
    """Compute gradient of the smooth max correlation with respect to h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    
    # Compute gradient using the chain rule
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Compute contribution to gradient through correlation
        for k in range(n_points):
            if (i + k) % n_points < n_points:
                # This is a simplified gradient
                grad[i] += (1 - h[(i + k) % n_points]) * dx
    
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    """Optimize using gradient descent with adaptive learning rate"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive learning rate
    learning_rate = 0.05
    decay_rate = 0.95
    iteration = 0
    
    while time() - start_time < time_limit:
        iteration += 1
        
        if iteration % 50 == 0:
            learning_rate *= decay_rate
            
        # Compute approximate gradient
        corr = np.correlate(h_start, 1 - h_start, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)  # offset for full correlation
        
        # Simplified gradient: derivative of the maximum term
        grad = np.zeros_like(h_start)
        for i in range(n_points):
            grad[i] = -(1 - h_start[(i + max_k) % n_points])
        
        # Update h
        h_new = h_start - learning_rate * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h_start = h_new
        
        # Regular restarts to avoid local optima
        if iteration % 100 == 0:
            h_start = project_to_feasible(h_start + np.random.normal(0, 0.01, n_points), n_points)
            
    return best_h, best_c5

    for n_points_try in resolutions:

        if n_points_try != n_points:
