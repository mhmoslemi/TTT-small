import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    
    try:
        h = np.array(initial_h_values)
    except NameError:
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Differential Evolution parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.random() < 0.7:
            # Mutate initial solution
            pert = np.random.normal(0, 0.1, n_points)
            new_h = h + pert
        else:
            # Random initialization
            new_h = np.random.rand(n_points) * 0.5
        population.append(project_feasible(new_h))
    
    best_individual = None
    best_fitness = float('inf')
    
    # DE main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = []
        for individual in population:
            fit = compute_overlap(individual)
            fitnesses.append(fit)
            if fit < best_fitness:
                best_fitness = fit
                best_individual = individual.copy()
        
        # Create new population
        new_population = []
        for i, individual in enumerate(population):
            # Select three different individuals
            idxs = list(range(population_size))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.random() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(individual[j])
            trial = np.array(trial)
            
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Selection
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
            else:
                new_population.append(individual)
                
        population = new_population
    
    # Final check
    final_C5 = compute_overlap(best_individual)
    
    return (best_individual.tolist(), final_C5, n_points)
