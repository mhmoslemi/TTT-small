import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, population_size):
    """Initialize population with feasible solutions"""
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            h_base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_base)), h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(population_size):
        perturbed = h_base + np.random.normal(0, 0.05, n_points)
        population.append(project(perturbed, n_points))
    
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Perform two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    max_generations = 1000
    stagnation_limit = 200
    dx = 2.0 / n_points
    
    # Initialize
    population = initialize_population(n_points, population_size)
    fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    stagnation_count = 0
    
    generation = 0
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Check for improvement
        current_best = np.min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Termination condition
        if stagnation_count > stagnation_limit:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create offspring
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            
            if np.random.rand() < crossover_rate:
                child1 = crossover(parent1, parent2)
                child2 = crossover(parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            offspring.extend([mutate(child1, mutation_rate), 
                             mutate(child2, mutation_rate)])
        
        # Replace worst individuals
        pop_fitness = [(i, fitness[i]) for i in range(len(population))]
        pop_fitness.sort(key=lambda x: x[1])
        worst_indices = [x[0] for x in pop_fitness[-(len(offspring)):]]
        
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Update mutation rate based on stagnation
        if stagnation_count > 100:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
        
        generation += 1
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
