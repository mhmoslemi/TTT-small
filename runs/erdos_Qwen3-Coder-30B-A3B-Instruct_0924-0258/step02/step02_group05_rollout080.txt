import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Simplified optimization approach using cvxpy
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Use convex-concave procedure for optimization
    max_iterations = 50
    improvement_tolerance = 1e-6
    
    # Initialize variables for quadratic programming
    h_var = cp.Variable(n_points)
    # Optimization parameters
    max_time = budget_s - 2  # Leave some margin
    
    for iter_count in range(max_iterations):
        if time.time() - start_time > max_time:
            break
            
        # Fix current h to get a QP subproblem
        h_current = best_h.copy()
        
        # Our approach: formulate as a QP problem
        # Minimize max overlap
        # Since maximum of linear expressions can be handled via big-M trick
        # Let's instead try one direct approach using cvxpy
        # We'll build the maximum constraint directly
        # Ceilings and floors need different approach - we try an iterative one
        
        # We'll try to model this as an optimization:
        # minimize C5 subject to C5 >= sum(h[i] * (1 - h[i+k])) * dx for all valid k
        # But that would be nonlinear otherwise
        
        # Let's try a different approach - we'll make C5 a variable and add
        # constraints that it's at least each overlap
        
        # Create constraints for all k
        # Overlaps are from k = 0 to 2*dx, step dx
        # We'll work with the full correlation coefficients correctly
        
        # Rebuild overlaps from conv - but less computational effort,
        # We'll keep the small implementation from before:
        # Still prefer idea of CCP but avoiding the complexity of full Laplace hyperspace
        
        # Here's a simplification:
        # Project current h to feasible set
        h_projected = project_feasible(h_current)
        
        # Now optimize the weights with a simple local method
        # Use scipy's minimize with L-BFGS-B
        def objective(h_vec):
            return compute_overlap(h_vec)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        # Optimize with bound constraints only
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = optimize.minimize(
                objective, 
                h_projected, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'disp': False}
            )
            
            if res.success:
                new_h = res.x
                if np.all(new_h >= 0) and np.all(new_h <= 1):
                    new_h = project_feasible(new_h)
                    new_C5 = compute_overlap(new_h)
                    
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        best_h = new_h.copy()
        except:
            pass
    
    # Make sure we return valid solution
    final_h = project_feasible(best_h)
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
