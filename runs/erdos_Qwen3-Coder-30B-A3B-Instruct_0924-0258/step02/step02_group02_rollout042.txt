import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Variable, Minimize, Problem, Maximum, sum, multiply, Constraint, norm
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Build a more structured optimization approach using CVXPY for better convergence
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Refine using a sequence of projections with increasing constraint set
    max_iterations = 2000
    iteration = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:  # Leave some margin for finalization
            break
            
        # Formulate convex optimization problem
        h_vars = Variable(n_points)
        eta = Variable()
        
        # Constraints for h_vars
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == target_sum,
            eta >= 0
        ]
        
        # Generate a sampling of shifts to add constraints
        shift_indices = np.linspace(0, n_points - 1, min(50, n_points), dtype=int)
        
        for i in shift_indices:
            # Compute convolution with shifted 1-h at this index
            k = i * dx
            shifted_ones = np.zeros_like(h)
            if i < n_points:
                shifted_ones[i:] = 1 - h[:-i] if i > 0 else 1 - h
            else:
                shifted_ones = np.zeros_like(h)
            
            # Create overlap constraint:
            # int h(x)(1 - h(x+k)) dx \leq eta
            overlap_constraint = cp.sum(cp.multiply(h_vars, shifted_ones)) * dx <= eta
            constraints.append(overlap_constraint)
        
        # Objective: minimize eta
        objective = Minimize(eta)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=1000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_vars.value).flatten()
                h_new = project_h(h_new)
                new_c5 = compute_c5(h_new)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
                h = h_new
            else:
                # Fallback to simple gradient ascent for exploration
                h = project_h(h + 0.01 * np.random.randn(n_points))
        except Exception:
            # Fallback to simple gradient ascent
            h = project_h(h + 0.01 * np.random.randn(n_points))
        
        iteration += 1
        if iteration > max_iterations:
            break

    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
