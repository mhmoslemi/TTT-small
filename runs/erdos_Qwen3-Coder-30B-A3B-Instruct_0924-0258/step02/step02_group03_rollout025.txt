import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    # Clamp again after scaling
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent or create default
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # PSO parameters
    swarm_size = 40
    max_iter = 10000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.05  # 5% of the range
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest_idx = 0
    gbest_val = float('inf')
    
    for i in range(swarm_size):
        # Initialize particle around h0 with small perturbations
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        val = evaluate(h, 2.0 / n_points)
        pbest.append(h.copy())
        pbest_vals.append(val)
        if val < gbest_val:
            gbest_val = val
            gbest_idx = i
    
    # PSO loop
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            vel_update = (inertia * velocities[i] +
                          c1 * r1 * (pbest[i] - particles[i]) +
                          c2 * r2 * (pbest[gbest_idx] - particles[i]))
            # Clip velocity
            vel_update = np.clip(vel_update, -v_max, v_max)
            velocities[i] = vel_update
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points / 2.0)
            
            # Evaluate new position
            new_val = evaluate(particles[i], 2.0 / n_points)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest_val = new_val
                    gbest_idx = i
        
        iteration += 1
    
    # Return best solution
    best_h = particles[gbest_idx]
    final_c5 = evaluate(best_h, 2.0 / n_points)
    return (best_h.tolist(), float(final_c5), n_points)
