import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize m particles using simplex method"""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base)), base)
    else:
        base = np.ones(n_points) * 0.5
    
    particles = [project_to_feasible(base, n_points)]
    
    # Generate m-1 additional particles by perturbing different directions
    for i in range(m-1):
        pert = base + np.random.normal(0, 0.1, n_points)
        particles.append(project_to_feasible(pert, n_points))
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Settings
    n_points = 30
    dx = 2.0 / n_points
    swarm_size = 10
    max_iter = 2000
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    max_shake_particles = 5
    
    # Initialize swarm
    particles = initialize_simplex(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # Track stagnation
    last_improvement = 0
    best_h = gbest.copy()
    best_c5 = gbest_value
    best_n_points = n_points
    
    iteration = 0
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Update velocities and positions
        for i in range(swarm_size):
            # Random weights
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_weight * r1 * (pbest[i] - particles[i]) +
                social_weight * r2 * (gbest - particles[i])
            )
            
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
        
        # Evaluate new positions
        new_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
        
        # Update personal bests
        improved = new_values < pbest_values
        pbest[improved] = particles[improved]
        pbest_values[improved] = new_values[improved]
        
        # Update global best
        new_gbest_idx = np.argmin(pbest_values)
        if pbest_values[new_gbest_idx] < gbest_value:
            gbest = pbest[new_gbest_idx].copy()
            gbest_value = pbest_values[new_gbest_idx]
            last_improvement = iteration
            if gbest_value < best_c5:
                best_h = gbest.copy()
                best_c5 = gbest_value
                best_n_points = n_points
            
        # Check for stagnation and apply shaking
        if iteration - last_improvement > shaking_threshold:
            # Select some particles to shake
            shake_indices = np.random.choice(swarm_size, size=min(max_shake_particles, swarm_size), replace=False)
            for i in shake_indices:
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            last_improvement = iteration
            
        # Periodically apply local refinement
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Sort indices by fitness
            idxs = np.argsort(pbest_values)
            top_indices = idxs[:min(10, len(idxs))]
            for idx in top_indices:
                try:
                    def obj(h):
                        return evaluate_erdos_solution(h, dx)
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(obj, pbest[idx], method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_value = evaluate_erdos_solution(refined, dx)
                        if refined_value < pbest_values[idx]:
                            pbest[idx] = refined
                            pbest_values[idx] = refined_value
                            if refined_value < gbest_value:
                                gbest = refined.copy()
                                gbest_value = refined_value
                                last_improvement = iteration
                                if gbest_value < best_c5:
                                    best_h = gbest.copy()
                                    best_c5 = gbest_value
                                    best_n_points = n_points
                except:
                    pass
    
    # Final refinement of best solution
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        if res.success:
            refined = project_to_feasible(res.x, n_points)
            refined_value = evaluate_erdos_solution(refined, dx)
            if refined_value < best_c5:
                best_h = refined
                best_c5 = refined_value
    except:
        pass
    
    return (list(best_h), best_c5, best_n_points)
