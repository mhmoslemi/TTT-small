import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def gradient_smooth_max(h, n_points, dx):
    """Gradient of the smooth max of the correlation"""
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    grad_corr = alpha * exp_corr / sum_exp
    
    # Reconstruct gradient w.r.t. h from convolution gradient
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad[i] = 0
        for k in range(len(h)):
            if i - k >= 0:
                grad[i] += grad_corr[k] * (1 - h[i-k])
            else:
                grad[i] += grad_corr[k] * (1 - h[i-k + len(h)])
    return grad * dx

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    """Optimize using gradient descent with restarts"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5_history = [best_c5]

    # Parameters
    lr = 0.01
    patience = 0
    patience_limit = 20
    min_lr = 1e-6

    for iteration in range(10000):
        if time() - start_time > time_limit:
            break
        
        # Compute gradient and update
        try:
            corr = np.correlate(h_start, 1 - h_start, mode='full')
            smooth_val = smooth_max(corr, alpha=100)
            
            # Approximate gradient using finite differences for safety
            eps = 1e-6
            grad = np.zeros_like(h_start)
            for i in range(len(h_start)):
                h_plus = h_start.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                smooth_plus = smooth_max(corr_plus, alpha=100)
                grad[i] = (smooth_plus - smooth_val) / eps
            
            # Update
            h_new = h_start - lr * grad
            h_new = project_to_feasible(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                best_c5_history.append(best_c5)
                patience = 0
            else:
                patience += 1

            # Decay learning rate
            if patience > 10:
                lr = max(lr * 0.9, min_lr)
                patience = 0

            h_start = h_new
        except:
            lr = max(lr * 0.5, min_lr)
            h_start = best_h.copy()
            patience = 0

    return best_h, best_c5

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points 
    
    # Test multiple resolutions
    resolutions = [25, 30, 35, 40, 45]
    
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Multiple restarts
        for restart in range(4):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slight perturbation
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed = project_to_feasible(h_start + noise, n_points)
            
            # Time per restart
            time_per_res = (budget_s - (time() - start_time)) / (len(resolutions) * 4)
            
            h_opt, c5_opt = optimize_with_gradient_descent(h_perturbed, n_points, dx_try, time_per_res)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
        
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
