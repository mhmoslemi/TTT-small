import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_overlaps(h_vec):
        # Compute all overlaps for different shifts
        overlaps = []
        for s in range(n_points):
            overlap = 0.0
            for i in range(n_points - s):
                overlap += h_vec[i] * (1 - h_vec[i + s]) * dx
            overlaps.append(overlap)
        return np.array(overlaps)

    # Alternating optimization approach
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Parameters
    max_iter = 5000
    lr = 0.01
    patience = 0
    patience_limit = 50
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlaps
        overlaps = compute_overlaps(h)
        max_overlap = np.max(overlaps)
        
        # Update h to reduce the maximum overlap
        new_h = h.copy()
        gradient = np.zeros(n_points)
        
        # Compute gradient for each position
        for i in range(n_points):
            for s in range(n_points - i):
                if i + s < n_points:
                    # Gradient contribution from shift s
                    grad = (1 - h[i + s]) * dx
                    if i + s < len(gradient):
                        gradient[i] += grad
        
        # Update h with learning rate
        new_h = h - lr * gradient
        
        # Project and rescale
        new_h = project_h(new_h)
        
        # Check if we improved
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping condition
        if patience >= patience_limit and iteration > 500:
            break
            
        # Adaptive learning rate
        if patience > 0 and iteration % 200 == 0:
            lr *= 0.9
            
        h = new_h.copy()
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
