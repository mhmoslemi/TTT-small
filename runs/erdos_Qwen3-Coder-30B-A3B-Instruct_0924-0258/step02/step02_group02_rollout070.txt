import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial_h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic Algorithm parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Track best solution
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Main GA loop
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            fitness = -compute_c5(h)
            fitness_scores.append(fitness)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Keep top 50%
        survivors = population[:population_size//2]
        new_population = survivors[:]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            selected_indices = []
            for _ in range(2):
                tournament = np.random.choice(population_size//2, tournament_size, replace=False)
                best_in_tournament = tournament[np.argmin([fitness_scores[i] for i in tournament])]
                selected_indices.append(best_in_tournament)
            
            parent1, parent2 = population[selected_indices[0]], population[selected_indices[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Mutate 1 or 2 positions
                    num_mutations = np.random.randint(1, 3)
                    mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                    for idx in mutation_indices:
                        child[idx] = np.random.rand()
                    
                    # Project to valid range and rescale
                    child = project_h(child)
                
                new_population.append(child)
        
        # Trim to population size
        population = new_population[:population_size]
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
