import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    n_points = 35
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    mutation_strength = 0.1
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        n_points = len(population[0])
    else:
        population = []
    
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Find best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    gen = 0
    while time() - start_time < budget_s * 0.98:
        gen += 1
        # Selection using tournament
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size]
            
            # Crossover
            if np.random.random() < crossover_rate:
                alpha = np.random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child1 += noise
                child1 = project_feasible(child1, n_points)
                
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child2 += noise
                child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Evaluate new population
        new_fitness = []
        for h in new_population[:pop_size]:
            fit = evaluate_erdos_solution(h, 2.0 / n_points)
            new_fitness.append(fit)
        
        # Survival - keep best individuals + some mutated copies of best
        combined_population = new_population[:pop_size]
        combined_fitness = new_fitness[:pop_size]
        
        # Keep best individuals
        sorted_indices = np.argsort(combined_fitness)
        elite_count = pop_size // 4
        survivors = [combined_population[i] for i in sorted_indices[:elite_count]]
        elite_fitness = [combined_fitness[i] for i in sorted_indices[:elite_count]]
        
        # Add mutated versions of elite
        for i in range(elite_count, pop_size):
            parent_idx = np.random.choice(elite_count)
            child = survivors[parent_idx].copy()
            noise = np.random.normal(0, mutation_strength, n_points)
            child += noise
            child = project_feasible(child, n_points)
            survivors.append(child)
            elite_fitness.append(evaluate_erdos_solution(child, 2.0 / n_points))
        
        population = survivors
        fitness = elite_fitness
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    return (list(best_h), best_c5, n_points)
