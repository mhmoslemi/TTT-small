import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    h0 = project_to_feasible(h0, n_points/2.0)
    
    # MSPF parameters
    n_particles = 100
    particles = []
    weights = []
    
    # Initialize particles around h0
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.1, n_points)
            h = project_to_feasible(h, n_points/2.0)
        particles.append(h)
    
    # Main optimization loop
    best_h = None
    best_c5 = float('inf')
    temp = 1.0
    proposal_std = 0.1
    acceptance_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Importance sampling step
        new_particles = []
        new_weights = []
        
        for i, h in enumerate(particles):
            # Propose new particle
            h_proposed = h + np.random.normal(0, proposal_std, n_points)
            h_proposed = project_to_feasible(h_proposed, n_points/2.0)
            
            # Compute importance weight
            c5_proposed = evaluate(h_proposed, 2.0/n_points)
            weight = np.exp(-c5_proposed / temp)
            
            new_particles.append(h_proposed)
            new_weights.append(weight)
            
            # Track acceptance
            if i == 0 or c5_proposed < evaluate(h, 2.0/n_points):
                acceptance_count += 1
            
            # Update best solution
            if c5_proposed < best_c5:
                best_c5 = c5_proposed
                best_h = h_proposed.copy()
        
        # Normalize weights
        total_weight = sum(new_weights)
        if total_weight == 0:
            # Reset if weights are degenerate
            weights = [1.0] * n_particles
        else:
            weights = [w / total_weight for w in new_weights]
        
        # Resampling step with top 50% particles retained
        indices = np.argsort(weights)[::-1]
        selected_indices = indices[:n_particles//2]
        selected_particles = [new_particles[i] for i in selected_indices]
        selected_weights = [weights[i] for i in selected_indices]
        
        # Mutation step
        mutated_particles = []
        for h in selected_particles:

            h_new1 = h + np.random.normal(0, proposal_std, n_points)
            h_new1 = project_to_feasible(h_new1, n_points/2.0)
            mutated_particles.append(h_new1)
            
            # Local exploitation mutation  
            h_new2 = h + np.random.normal(0, 0.05, n_points)

            mutated_particles.append(h_new2)
