import numpy as np
import cvxpy as cp
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    max_iterations = 10
    tolerance = 1e-5
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Define variables for convex optimization
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # sum(h) * dx = 1
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all shifts
        max_k = n_points - 1
        for k in range(max_k):
            # Compute correlation terms for shift k
            corr_terms = []
            for i in range(n_points - k):
                corr_terms.append(h[i] * (1 - h[i + k]))
            overlap_sum = cp.sum(corr_terms)
            constraints.append(overlap_sum <= C_mid * (n_points - k) * dx)
        
        # Create problem and solve
        objective = cp.Minimize(0)  # Dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try solving with ECOS solver
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, feastol=1e-8, abstol=1e-8)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h.value
                # Ensure feasibility
                h_feasible = np.clip(h_solution, 0, 1)
                h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_c5 = c5_feasible
                    best_h = h_feasible.copy()
                
                # Move to smaller C
                C_high = C_mid
            else:
                # Infeasible, increase C
                C_low = C_mid
        except:
            C_low = C_mid
    
    # Final refinement with gradient-based method
    if time() - start_time < budget_s * 0.9:
        # Try using BFGS with a slightly perturbed version of the best solution
        h_refined = best_h.copy()
        dx_refined = 2.0 / n_points
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_refined
        
        # Simple local search to refine
        current_c5 = objective(h_refined)
        
        # Try small perturbations
        for _ in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            h_new = h_refined.copy()
            # Small random perturbations 
            noise = np.random.normal(0, 0.02, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = objective(h_new)
            if new_c5 < current_c5:
                h_refined = h_new.copy()
                current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    
    return (list(best_h), best_c5, n_points)
