import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, tol=1e-6):
    """Frank-Wolfe optimization for minimizing max overlap"""
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        # Compute all correlations to find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        if iteration > 0 and abs(max_overlap - best_c5) < tol:
            break
            
        # Identify contributing indices for the maximum overlap
        max_shift = max_idx - (n_points - 1)
        # Approximate subgradient based on the maximum shift(s)
        g = np.zeros_like(h)
        for i in range(n_points):
            k = (i - max_shift) % n_points
            g[i] = (1 - h[k]) - h[i]
        
        # Solve linearized problem to find descent direction
        # s such that min g·s subject to sum(s) = 0, 0 <= s <= 1
        s = np.zeros_like(h)
        # We want to minimize g·s subject to s constraints
        # Since sum(s) = 0, we fix one element and set others to minimize g·s
        # Simple approach: set s as maximizing -g (clipped to [0,1])
        s = np.clip(-g, 0, 1)
        # Ensure sum(s) = 0 by shifting
        s_mean = np.mean(s)
        s = s - s_mean  # This ensures sum(s) = 0
        
        # Project back to feasible region and rescale
        s = project_to_feasible(s, n_points)
        
        # Line search for step size in [0.1, 0.9]
        best_gamma = 0.1
        best_h_trial = h.copy()
        for gamma in [0.1, 0.3, 0.5, 0.7, 0.9]:
            h_trial = h + gamma * (s - h)
            h_trial = project_to_feasible(h_trial, n_points)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_gamma = gamma
                best_h_trial = h_trial.copy()
        
        h = best_h_trial.copy()
        if best_c5 < evaluate_erdos_solution(best_h, dx):
            best_h = h.copy()
    
    return best_h, best_c5

    resolutions = [25, 30, 35]
    best_h = None
    best_c5 = float('inf')

    else:
        # Build a simple starting vector
        current_n = 25
        current_h = np.ones(current_n) * 0.5
    
    # Refine at each resolution 
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        if current_n != n_points:
            # Interpolate from previous resolution
            old_x = np.linspace(0, 2, current_n)
            new_x = np.linspace(0, 2, n_points)
            h_interp = np.interp(new_x, old_x, current_h)
            current_h = project_to_feasible(h_interp, n_points)
            current_n = n_points
            
        # Multiple restarts for better exploration
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb the current solution slightly for restart
            h_perturbed = current_h + np.random.normal(0, 0.02, n_points)
            h_feasible = project_to_feasible(h_perturbed, n_points)
            
            # Run Frank-Wolfe
            h_opt, c5_opt = frank_wolfe_step(h_feasible, n_points, dx, max_iter=40)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n_points = n_points
                
        current_h = best_h.copy()  # Use best so far for next iteration
        
    # Final optimization if needed
    if best_h is not None and time() - start_time < budget_s * 0.9:
        dx = 2.0 / best_n_points
        h_final, c5_final = frank_wolfe_step(best_h.copy(), best_n_points, dx, max_iter=20)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
