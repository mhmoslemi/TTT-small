import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to achieve target sum
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial h to satisfy sum constraint
    h0 = project_to_feasible(h0, n_points / 2.0)
    
    # Multi-swarm particle filter parameters
    n_particles = 100
    n_top = 10
    n_mutations = 2
    temp_init = 1.0
    temp_decay = 0.99
    init_sigma = 0.1
    target_acceptance = 0.3
    
    # Initialize swarms
    particles = [h0.copy()]
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, init_sigma, n_points)
        particles.append(project_to_feasible(h, n_points / 2.0))
    
    # Evaluate initial particles
    particle_values = [evaluate(h, 2.0/n_points) for h in particles]
    
    best_c5 = min(particle_values)
    best_h = particles[np.argmin(particle_values)].copy()
    
    temp = temp_init
    sigma = init_sigma
    no_improvement_count = 0
    
    # Tracking for acceptance rate adaptation
    accepted_count = 0
    total_proposals = 0
    
    while time.time() - start_time < budget_s - 1:
        # Importance sampling step
        new_particles = []
        new_values = []
        weights = []
        
        for h in particles:
            current_val = evaluate(h, 2.0/n_points)
            # Sample new particles
            for _ in range(n_mutations):
                h_new = h + np.random.normal(0, sigma, n_points)
                h_new = project_to_feasible(h_new, n_points / 2.0)
                new_val = evaluate(h_new, 2.0/n_points)
                total_proposals += 1
                
                # Accept or reject based on weight
                weight = np.exp(-new_val / temp)
                weights.append(weight)
                
                if np.random.random() < weight:
                    new_particles.append(h_new)
                    new_values.append(new_val)
                    accepted_count += 1
                else:
                    new_particles.append(h)
                    new_values.append(current_val)
        
        # Resampling using stratified resampling
        if len(new_particles) > 0:
            weights = np.array(weights) + 1e-10  # Avoid zero weights
            weights = weights / np.sum(weights)
            
            # Stratified resampling
            n_selected = n_particles  # Keep all particles
            indices = np.random.random(n_selected) * np.sum(weights)
            cumulative = np.cumsum(weights)
            selected_indices = np.searchsorted(cumulative, indices)
            
            # Sample particles based on selected indices
            particles = [new_particles[i] for i in selected_indices]
            particle_values = [new_values[i] for i in selected_indices]
            
            # Update best
            min_val = min(particle_values)
            if min_val < best_c5:
                best_c5 = min_val
                best_h = particles[np.argmin(particle_values)].copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Adapt temperature and noise
            temp *= temp_decay
            if total_proposals > 0:
                acceptance_rate = accepted_count / total_proposals
                if acceptance_rate > target_acceptance * 1.1:
                    sigma *= 1.1
                elif acceptance_rate < target_acceptance * 0.9:
                    sigma /= 1.1
                # Reset counters
                accepted_count = 0
                total_proposals = 0
        
        # Occasionally, perform a blast mutation from top performers
        if no_improvement_count % 10 == 0:
            sorted_indices = np.argsort(particle_values)
            top_indices = sorted_indices[:n_top]
            for idx in top_indices:
                h = particles[idx]
                # Add exploration mutations
                for _ in range(n_mutations):
                    h_mut = h + np.random.normal(0, sigma*2, n_points)
                    h_mut = project_to_feasible(h_mut, n_points / 2.0)
                    particles.append(h_mut)
                    particle_values.append(evaluate(h_mut, 2.0/n_points))
        
        if no_improvement_count > 50:
            break
    
    # Final evaluation
    final_c5 = evaluate(best_h, 2.0/n_points)
    return (best_h.tolist(), float(final_c5), n_points)
