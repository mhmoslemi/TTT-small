import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        h0 = np.full(45, 1.0 / 45)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h0 satisfies the constraint
    h0 = h0 * target_sum / np.sum(h0)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5

    # Initialize population
    population = []
    population.append(h0.copy())
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_h(h)
        population.append(h)
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(-c5)  # Negative for minimization

        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmax([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create new population through crossover and mutation
        new_population = [h0.copy()]  # Keep best individual
        
        for i in range(1, population_size, 2):
            if i + 1 < population_size and np.random.random() < crossover_rate:
                parent1 = selected[i]
                parent2 = selected[i + 1]
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Mutation
                if np.random.random() < mutation_rate:
                    idx = np.random.randint(0, n_points)
                    child1[idx] = np.random.uniform(0, 1)
                    child1 = project_h(child1)
                
                if np.random.random() < mutation_rate:
                    idx = np.random.randint(0, n_points)
                    child2[idx] = np.random.uniform(0, 1)
                    child2 = project_h(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just copy the selected individuals
                new_population.append(selected[i])
                if len(new_population) < population_size and i + 1 < population_size:
                    new_population.append(selected[i + 1])
        
        # Ensure correct size
        while len(new_population) < population_size:
            h = np.random.uniform(0, 1, n_points)
            h = project_h(h)
            new_population.append(h)
        
        # Update population
        population = new_population[:population_size]
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
