import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.15
    
    # Initialize swarm
    swarm = []
    personal_best = []
    personal_best_values = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        swarm.append(h_init.copy())
        personal_best.append(h_init.copy())
        personal_best_values.append(compute_overlap(h_init))
    else:
        h_init = np.ones(n_points) * 0.5
        swarm.append(h_init)
        personal_best.append(h_init.copy())
        personal_best_values.append(compute_overlap(h_init))
    
    # Add random particles
    for i in range(swarm_size - len(swarm)):
        h_rand = np.random.rand(n_points) * 0.5 + 0.25  # [0.25, 0.75]
        h_rand = project_feasible(h_rand)
        swarm.append(h_rand)
        personal_best.append(h_rand.copy())
        personal_best_values.append(compute_overlap(h_rand))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            v = (w * (swarm[i] - personal_best[i]) + 
                 c1 * r1 * (personal_best[i] - swarm[i]) + 
                 c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_pos = swarm[i] + v
            
            # Project to feasible region
            new_pos = project_feasible(new_pos)
            
            # Evaluate
            new_value = compute_overlap(new_pos)
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_best[i] = new_pos.copy()
                personal_best_values[i] = new_value
                
                # Update global best
                if new_value < global_best_value:
                    global_best = new_pos.copy()
                    global_best_value = new_value
                    global_best_idx = i
        
        # Update swarm
        for i in range(swarm_size):
            swarm[i] = personal_best[i].copy()
    
    # Final check
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
