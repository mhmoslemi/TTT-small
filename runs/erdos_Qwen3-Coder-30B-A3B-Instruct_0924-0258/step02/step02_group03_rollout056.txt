import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Initialize mutation strengths
    sigma = np.full(pop_size, 0.05)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')

    generation = 0

        # Evaluate population
        fitness = [evaluate(h, 2.0/n_points) for h in population]
        
        # Update best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            last_improvement = generation
        
        # Check for stagnation (early exit)
        if generation - last_improvement > 50:
            break
            
        # Selection (rank-based)
        ranked_indices = np.argsort(fitness)
        elite_size = max(1, int(elite_ratio * pop_size))
        elites = [population[i] for i in ranked_indices[:elite_size]]
        
        # Generate offspring
        offspring = []
        for i in range(pop_size - elite_size):
            if np.random.rand() < mutation_rate:
                # Tournament selection
                parent_idx = ranked_indices[np.random.randint(0, pop_size // 2)]
                parent = population[parent_idx]
                s = sigma[parent_idx]
                
                # Mutate

                # Copy elite
                elite_idx = np.random.randint(0, elite_size)
                offspring.append(elites[elite_idx].copy())
