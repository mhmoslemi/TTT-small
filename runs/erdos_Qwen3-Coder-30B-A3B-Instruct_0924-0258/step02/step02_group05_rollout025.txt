import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Differential Evolution parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.random() < 0.5:
            individual = h + np.random.normal(0, 0.05, n_points)
        else:
            individual = np.random.uniform(0, 1, n_points)
        individual = project_feasible(individual)
        population.append(individual)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fit = compute_overlap(ind)
        fitness.append(fit)
    
    best_index = np.argmin(fitness)
    best_h = population[best_index].copy()
    best_fitness = fitness[best_index]
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    max_stagnation = 50
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three different individuals for mutation
            indices = list(range(population_size))
            indices.remove(i)
            a, b, c = np.random.choice(indices, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.random() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Evaluate trial
            trial_fitness = compute_overlap(trial)
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = new_population
        fitness = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= max_stagnation:
            # Random restart
            for j in range(population_size):
                population[j] = np.random.uniform(0, 1, n_points)
                population[j] = project_feasible(population[j])
            fitness = [compute_overlap(ind) for ind in population]
            current_best_idx = np.argmin(fitness)
            if fitness[current_best_idx] < best_fitness:
                best_fitness = fitness[current_best_idx]
                best_h = population[current_best_idx].copy()
            stagnation_count = 0
            
        generation += 1
    
    # Final validation
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
