import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:

        h0 = np.ones(n_points) * 0.5
    
    best_h = None
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    max_iter = 50000
    iter_count = 0
    acceptance_count = 0
    accept_threshold = 50
    no_improvement_count = 0
    max_no_improvement = 100
    
    current_h = h0.copy()
    current_c5 = evaluate(current_h, 2.0/n_points)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Generate candidate
        candidate_h = current_h + np.random.normal(0, T, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points/2.0)
        candidate_c5 = evaluate(candidate_h, 2.0/n_points)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            acceptance_count += 1
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Update temperature
        T = max(T * alpha, 1e-6)
        
        # Adaptive cooling
        if no_improvement_count > max_no_improvement:
            T = min(T * 1.1, 0.5)
            no_improvement_count = 0
            
        iter_count += 1
    
    if best_h is not None:
        final_c5 = evaluate(best_h, 2.0/n_points)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h0, n_points/2.0)
        final_c5 = evaluate(final_h, 2.0/n_points)
        return (final_h.tolist(), float(final_c5), n_points)
