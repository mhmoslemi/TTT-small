import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize a simplex with m points"""
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    simplex = [project_to_feasible(h0, n_points)]
    
    # Generate m-1 additional points
    for i in range(m-1):
        perturb = np.random.normal(0, 0.1, n_points)
        h_i = project_to_feasible(simplex[0] + perturb, n_points)
        simplex.append(h_i)
    
    return simplex

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    n_points = 30
    dx = 2.0 / n_points
    swarm_size = 10
    max_iter = 2000
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    shake_threshold = 50
    lbfgs_interval = 200
    max_shake = 50
    
    # Initialize swarm
    simplex = initialize_simplex(n_points, swarm_size)
    swarm = [h.copy() for h in simplex]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Evaluate initial swarm
    costs = [evaluate_erdos_solution(h, dx) for h in swarm]
    
    # Track best
    best_idx = np.argmin(costs)
    gbest = swarm[best_idx].copy()
    gbest_cost = costs[best_idx]
    
    # Personal best tracking
    pbest = [h.copy() for h in swarm]
    pbest_costs = costs.copy()
    
    # Iteration counter and stagnation tracking
    iteration = 0
    stagnation_count = 0
    
    # Main PSO loop
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Apply L-BFGS-B to top 10 particles
            top_indices = np.argsort(costs)[:10]
            for idx in top_indices:
                h0 = swarm[idx].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                # Only optimize if not already very good
                if costs[idx] < 0.4:
                    try:
                        bounds = [(0, 1) for _ in range(n_points)]
                        res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, 
                                     options={'maxiter': 10})
                        if res.success:
                            h_opt = project_to_feasible(res.x, n_points)
                            c5_opt = evaluate_erdos_solution(h_opt, dx)
                            if c5_opt < costs[idx]:
                                swarm[idx] = h_opt.copy()
                                costs[idx] = c5_opt
                                if c5_opt < gbest_cost:
                                    gbest = h_opt.copy()
                                    gbest_cost = c5_opt
                    except:
                        pass
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (pbest[i] - swarm[i]) + 
                           social_weight * r2 * (gbest - swarm[i]))
            
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate
            new_cost = evaluate_erdos_solution(swarm[i], dx)
            costs[i] = new_cost
            
            # Update personal best
            if new_cost < pbest_costs[i]:
                pbest[i] = swarm[i].copy()
                pbest_costs[i] = new_cost
                
                # Check if global best improves
                if new_cost < gbest_cost:
                    gbest = swarm[i].copy()
                    gbest_cost = new_cost
                    stagnation_count = 0
        
        # Stagnation handling
        if iteration > 0 and iteration % shake_threshold == 0:
            if stagnation_count > 0:
                for i in range(swarm_size):
                    noise = np.random.normal(0, 0.1, n_points)
                    swarm[i] = project_to_feasible(swarm[i] + noise, n_points)
                    costs[i] = evaluate_erdos_solution(swarm[i], dx)
                    if costs[i] < pbest_costs[i]:
                        pbest[i] = swarm[i].copy()
                        pbest_costs[i] = costs[i]
                        if costs[i] < gbest_cost:
                            gbest = swarm[i].copy()
                            gbest_cost = costs[i]
            stagnation_count += 1
            
            # Expand swarm if necessary
            if swarm_size < 50:
                old_size = swarm_size
                swarm_size = min(swarm_size + 5, 50)
                for i in range(old_size, swarm_size):
                    perturb = np.random.normal(0, 0.1, n_points)
                    h_new = project_to_feasible(gbest + perturb, n_points)
                    swarm.append(h_new)
                    velocities.append(np.zeros(n_points))
                    costs.append(evaluate_erdos_solution(h_new, dx))
                    pbest.append(h_new.copy())
                    pbest_costs.append(costs[-1])
        
        iteration += 1
        if stagnation_count > max_shake:
            break
    
    # Final refinement
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 20})
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < gbest_cost:
                gbest = final_h
                gbest_cost = final_c5
    except:
        pass
    
    return (list(gbest), gbest_cost, n_points)
