import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, dx, n_points, max_iter=40, tolerance=1e-6):
    """Frank-Wolfe optimization for the Erdős constant minimization"""
    h = project_to_feasible(h.copy(), n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        if iteration % 10 == 0:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Compute correlations to find max overlap shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_shift = np.argmax(corr)
        max_val = corr[max_shift]
        
        # Find all shifts that achieve maximum value (within tolerance)
        max_shifts = np.where(np.abs(corr - max_val) < tolerance)[0]
        
        # Compute subgradient
        g = np.zeros(n_points)
        for shift in max_shifts:
            # For correlation at shift k, derivative w.r.t. h[i] is (1-h[(i-k)%n])
            # But also subtract h[i] (from penalty term in convolution)
            for i in range(n_points):
                k = shift - i
                if k < 0:
                    k += n_points
                g[i] += (1 - h[k]) - h[i]
        
        # Average the subgradients
        g /= len(max_shifts)
        
        # Solve linear subproblem: minimize g^T s
        # subject to s >= 0, sum(s) = n/2, sum(h) = n/2
        # Equivalent to minimize g^T s with s_i >= 0, sum(s) = 0 (difference form)
        # This measures the angle of depreciated values
        # From this perspective, we're minimizing uphill direction along g
        # Use projected gradient approach or simple choice:
        s = np.zeros(n_points)
        # Select step direction by picking index with highest negative gradient
        idx = np.argmin(g) if np.min(g) < 0 else np.argmax(g)
        s[idx] = 1.
        s = project_to_feasible(s, n_points) - h
        
        # Line search for step size
        gamma = 0.9
        for _ in range(20):
            h_new = h + gamma * s
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new <= best_c5 or c5_new < c5:
                h = h_new
                c5 = c5_new
                break
            gamma *= 0.5
        
        # Mid-check break condition
        if (iteration > 0 and abs(c5 - best_c5) < 1e-5 * best_c5):
            break

    return h, best_c5

    np.random.seed(seed)
    start_time = time()

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
        h_start[::2] = 0.0
        h_start[1::2] = 1.0
    
    dx = 2.0 / n_points
    h_best, c5_best = h_start.copy(), evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points

    resolutions = [25, 30, 35]
    for n_points_current in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_current = 2.0 / n_points_current
        if n_points_current != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_current), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_current)
            n_points = n_points_current

        # Single optimization step with Frank-Wolfe
        time_remaining = budget_s - (time() - start_time)
        time_per_res = min(time_remaining / len(resolutions), 100)
        
        # Multiple restarts
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = project_to_feasible(
                h_start + np.random.normal(0, 0.03, n_points), n_points)
            h_opt, c5_opt = frank_wolfe_step(h_restart, dx_current, n_points, 
                                            max_iter=40, tolerance=1e-6)
            if c5_opt < c5_best:
                h_best = h_opt.copy()
                c5_best = c5_opt
                best_n_points = n_points
        h_start = h_best.copy()
        
    return (list(h_best), c5_best, best_n_points)
