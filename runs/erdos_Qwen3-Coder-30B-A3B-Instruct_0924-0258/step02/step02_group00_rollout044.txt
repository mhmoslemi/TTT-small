import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_subgradient(h, max_shifts, n_points):
    """Compute subgradient for current maximum overlap shifts"""
    subgrad = np.zeros_like(h)
    for k in max_shifts:
        for i in range(n_points):
            j = (i + k) % n_points
            subgrad[i] += (1 - h[j])
    return subgrad

def frank_wolfe_step(h, subgrad, n_points, dx):
    """Frank-Wolfe step to find descent direction s"""
    # Linear problem: minimize subgrad^T * s subject to sum(s) = 0, 0 <= s <= 1
    # We can solve this by finding direction that minimizes the dot product
    # Let's solve: s = argmin s^T * g subject to sum(s) = 0, 0 <= s <= 1
    
    # For Frank-Wolfe, we'll try to find an extreme point of the feasible set
    # that gives most negative dot product with g
    # To do this, we want to shift h in a direction that reduces the objective
    # We use a mirror descent-like step by identifying most positive component
    # of subgrad and defining an update direction
    
    # Choose s that is the direction which could improve the result
    # Solve for s such that s is the extreme point of the constraint set
    # that gives the most negative inner product with subgrad
    
    # This is a simplified version: find index with max subgrad, try to move it down
    # and others up, while satisfying the sum constraint
    
    # Find the position that would be good to decrease
    s = np.zeros_like(h)
    positive_idx = np.argmax(subgrad)
    s[positive_idx] = -1.0
    # Now we want sum(s) = 0 -> we also need to add 1 to some other element
    negative_idx = np.argmin(subgrad)
    if negative_idx != positive_idx:
        s[negative_idx] = 1.0
    else:
        # Need to find another one if same
        other_idx = (negative_idx + 1) % n_points
        s[other_idx] = 1.0
    
    return s

def adaptive_frank_wolfe(h_start, n_points, dx, max_iter=100, tolerance=1e-6):
    """Main Frank-Wolfe optimization loop"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        
        # Find all indices that give max value
        max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
        
        # Use a threshold to take top 10% of shifts for stability
        # but in any case at least 1
        max_shifts = max_indices[-max(1, len(max_indices) // 10):]
        
        # Compute subgradient
        subgrad = compute_subgradient(h, max_shifts, n_points)
        
        # Optimize direction s using Frank-Wolfe approach
        s = frank_wolfe_step(h, subgrad, n_points, dx)
        
        # Backtracking line search on gamma
        gamma = 0.9
        for trial in range(20):
            h_new = h + gamma * s
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            if c5_new < max_val * dx - 1e-8:
                break
            gamma *= 0.7
        
        h = h_new.copy()
        if abs(best_c5 - max_val * dx) < tolerance:
            h = best_h.copy()
            break
    
    return best_h, best_c5
