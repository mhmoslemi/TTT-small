import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a binary-like sequence with 1s and 0s arranged to minimize autocorrelation
        # This is a heuristic based on LABS principles
        h = np.zeros(n_points)
        ones_count = n_points // 2
        # Place 1s approximately evenly
        indices = np.linspace(0, n_points-1, ones_count, dtype=int)
        h[indices] = 1.0
        # Slightly perturb to avoid perfect periodicity
        np.random.shuffle(indices)
        h[indices[:ones_count//2]] = 0.0
        h[indices[ones_count//2:]] = 1.0
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts with deterministic perturbations
        h_current = h_start.copy()
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Optimizer function
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            # Optimization constraints and bounds
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            # Try optimization with L-BFGS-B
            try:
                res = minimize(objective, h_current, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except:
                pass
            
            # Perturb with structured noise for next restart
            noise_strength = 0.05 / (1 + restart)
            noise = np.random.normal(0, noise_strength, n_points)
            h_current = project_to_feasible(h_current + noise, n_points)
    
    # Final refinement at highest resolution if needed
    if best_h is None or best_n_points < 75:
        n_points = 75
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(objective, h_start, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
        except:
            pass
    
    # If still no solution, use the starting configuration from best resolution
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h = h_start
        best_c5 = evaluate_erdos_solution(h_start, dx)
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
