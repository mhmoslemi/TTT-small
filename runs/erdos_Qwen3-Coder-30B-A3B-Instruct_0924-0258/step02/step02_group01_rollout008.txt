import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Generate population of candidates
    population_size = 50
    population = []
    
    # Add initial solution
    population.append(h_start.copy())
    
    # Add random perturbations of initial solution
    for _ in range(population_size - 1):
        h_perturb = h_start.copy()
        indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        h_perturb[indices] += np.random.normal(0, 0.1, len(indices))
        h_perturb = project_to_feasible(h_perturb, n_points)
        population.append(h_perturb)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 20
    
    while time() - start_time < budget_s * 0.95 and no_improvement_count < max_no_improvement:
        iteration += 1
        new_population = []
        
        for h in population:
            # Apply perturbation
            h_perturb = h.copy()
            indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            h_perturb[indices] += np.random.normal(0, 0.1, len(indices))
            h_perturb = project_to_feasible(h_perturb, n_points)
            
            # Local optimization with SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturb, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        no_improvement_count = 0
            except:
                h_opt = h_perturb
                c5_opt = evaluate_erdos_solution(h_opt, dx)
            
            new_population.append(h_opt)
        
        population = new_population
        no_improvement_count += 1
        
        # Occasionally rescale search if no improvements for a while
        if no_improvement_count > 5:
            for i in range(len(population)):
                h = population[i]
                # Add diversity by slightly altering the population
                if np.random.rand() < 0.3:
                    indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                    h[indices] += np.random.normal(0, 0.05, len(indices))
                    h = project_to_feasible(h, n_points)
                    population[i] = h

    # Refinement at higher resolution
    current_h = best_h.copy()
    current_n = n_points
    
    while (time() - start_time < budget_s * 0.97) and (current_n < 1000):
        current_n *= 2
        dx_new = 2.0 / current_n
        x_old = np.linspace(0, 2, len(current_h))
        x_new = np.linspace(0, 2, current_n)
        current_h = np.interp(x_new, x_old, current_h)
        current_h = project_to_feasible(current_h, current_n)
        
        def obj(h):
            return evaluate_erdos_solution(h, dx_new)
            
        bounds = [(0, 1) for _ in range(current_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - current_n / 2}]
        
        try:
            res = minimize(obj, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                current_h = project_to_feasible(res.x, current_n)
                new_c5 = evaluate_erdos_solution(current_h, dx_new)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()
        except:
            pass
    
    return (list(best_h), best_c5, len(best_h))
