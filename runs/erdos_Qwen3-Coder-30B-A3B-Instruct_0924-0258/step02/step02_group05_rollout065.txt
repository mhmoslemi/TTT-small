import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Fourier representation
    def fourier_to_h(coeffs, n):
        """Convert Fourier coefficients to step function values"""
        # coeffs: [a0, a1, b1, a2, b2, ...]
        a0 = coeffs[0]
        a_coeffs = coeffs[1::2]
        b_coeffs = coeffs[2::2]
        
        x = np.linspace(0, 2, n)
        h = a0 / 2.0
        
        for i, (a, b) in enumerate(zip(a_coeffs, b_coeffs)):
            k = i + 1
            h += a * np.cos(2 * np.pi * k * x) + b * np.sin(2 * np.pi * k * x)
        
        return np.clip(h, 0, 1)
    
    def compute_overlap(h_vec):
        """Compute overlap integral C5"""
        # Convolution of h with (1-h)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        """Project to feasible region (0-1, sum = target_sum)"""
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def gradient_descent_optimize():
        # Fourier coefficients: a0 (1), a1,b1 (2), a2,b2 (2), ...
        # Start with few components
        n_components = 5
        total_coeffs = 1 + 2 * n_components
        coeffs = np.zeros(total_coeffs)
        
        # Initialize from initial_h_values if available
        if 'h_init' in locals():
            h_sampled = h_init
            # Compute approximate Fourier coeffs
            x = np.linspace(0, 2, n_points)
            for k in range(1, n_components + 1):
                coeffs[1 + 2*(k-1)] = 2 * np.sum(h_sampled * np.cos(2*np.pi*k*x)) / n_points
                coeffs[2 + 2*(k-1)] = 2 * np.sum(h_sampled * np.sin(2*np.pi*k*x)) / n_points
            coeffs[0] = 2 * np.sum(h_sampled) / n_points
            
        # Optimizer params
        alpha = 0.01
        beta1 = 0.9
        beta2 = 0.999
        epsilon = 1e-8
        m = np.zeros_like(coeffs)
        v = np.zeros_like(coeffs)
        
        best_c5 = float('inf')
        best_coeffs = coeffs.copy()
        
        iter_count = 0
        last_improvement = 0
        
        while time.time() - start_time < budget_s - 1 and iter_count < 50000:
            iter_count += 1
            
            # Forward pass
            h = fourier_to_h(coeffs, n_points)
            h = project_feasible(h)
            current_c5 = compute_overlap(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_coeffs = coeffs.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
            if last_improvement > 5000:
                break
            
            # Compute gradients using finite differences
            eps = 1e-4
            grad = np.zeros_like(coeffs)
            
            for i in range(len(coeffs)):
                coeffs_plus = coeffs.copy()
                coeffs_plus[i] += eps
                h_plus = fourier_to_h(coeffs_plus, n_points)
                h_plus = project_feasible(h_plus)
                c_plus = compute_overlap(h_plus)
                
                coeffs_minus = coeffs.copy()
                coeffs_minus[i] -= eps
                h_minus = fourier_to_h(coeffs_minus, n_points)
                h_minus = project_feasible(h_minus)
                c_minus = compute_overlap(h_minus)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            
            m_hat = m / (1 - beta1**iter_count)
            v_hat = v / (1 - beta2**iter_count)
            
            coeffs = coeffs - alpha * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Occasionally increase resolution
            if iter_count % 1000 == 0 and n_components < 10:
                n_components += 1
                total_coeffs = 1 + 2 * n_components
                # Extend coeffs with zeros
                coeffs = np.pad(coeffs, (0, total_coeffs - len(coeffs)), 'constant', constant_values=0)
        
        # Return best solution
        final_h = fourier_to_h(best_coeffs, n_points)
        final_h = project_feasible(final_h)
        final_c5 = compute_overlap(final_h)
        
        return final_h.tolist(), final_c5, n_points
    
    h_values, c5_bound, n_points = gradient_descent_optimize()
    
    return h_values, c5_bound, n_points
