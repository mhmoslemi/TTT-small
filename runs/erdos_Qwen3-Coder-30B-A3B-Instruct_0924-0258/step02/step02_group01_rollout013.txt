import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    iterations = 0
    last_improvement = 0
    patience = 100
    
    while time() - start_time < budget_s * 0.95:
        # Compute full correlation to find the maximum overlap lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for the dominant lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - best_h[j]) - best_h[i]
        grad *= dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        t = iterations + 1
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update h
        best_h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        best_h = project_feasible(best_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(best_h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            last_improvement = iterations
        
        # Check for improvement stall - restart with random perturbation
        if iterations - last_improvement > patience:
            perturbation_strength = 0.05
            noise = np.random.normal(0, perturbation_strength, n_points)
            h_restart = best_h + noise
            best_h = project_feasible(h_restart, n_points)
            # Reset Adam moments
            m = np.zeros(n_points)
            v = np.zeros(n_points)
            last_improvement = iterations
        
        iterations += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Try increasing resolution if time allows
    if time() - start_time < budget_s * 0.8:
        resolutions = [50, 70, 100]
        for res in resolutions:
            if time() - start_time > budget_s * 0.95:
                break
            h_interp = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            h_interp = project_feasible(h_interp, res)
            c5_test = evaluate_erdos_solution(h_interp, 2.0 / res)
            if c5_test < final_c5:
                final_c5 = c5_test
                best_h = h_interp
                n_points = res
    
    return (list(best_h), final_c5, n_points)
