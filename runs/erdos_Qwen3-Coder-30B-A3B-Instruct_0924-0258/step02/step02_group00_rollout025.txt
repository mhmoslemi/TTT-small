import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for the optimal C5
    c_low = 0.37
    c_high = 0.38085
    c_mid = (c_low + c_high) / 2.0
    max_iterations = 10
    iteration = 0
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Formulate convex feasibility problem
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2.0,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints for all possible shifts
        for k in range(n_points):
            if k == 0:
                # For k=0, we have h[i] * (1-h[i]) overlap
                shift_sum = cp.sum(cp.multiply(h_vars, 1 - h_vars))
            else:
                # Sum over valid indices
                corr_sum = 0
                for i in range(n_points - k):
                    corr_sum += h_vars[i] * (1 - h_vars[i + k])
                shift_sum = corr_sum
            
            constraints.append(shift_sum <= c_mid)
        
        # Create and solve the convex problem
        objective = cp.Minimize(0)  # Dummy objective
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if problem.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                if h_feasible is not None:
                    # Refine with local optimization
                    def obj(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    # Project to feasible region
                    h_proj = np.clip(h_feasible, 0, 1)
                    h_proj = h_proj * (n_points / 2.0) / np.sum(h_proj)
                    
                    # Use scipy for local refinement
                    def objective_func(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    res = minimize(objective_func, h_proj, method='L-BFGS-B',
                                   bounds=[(0, 1) for _ in range(n_points)],
                                   options={'maxiter': 50})
                    
                    if res.success:
                        refined_h = np.clip(res.x, 0, 1)
                        refined_h = refined_h * (n_points / 2.0) / np.sum(refined_h)
                        refined_c5 = evaluate_erdos_solution(refined_h, dx)
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

        c_mid = (c_low + c_high) / 2.0
        iteration += 1
        
    return (list(best_h), best_c5, n_points)
