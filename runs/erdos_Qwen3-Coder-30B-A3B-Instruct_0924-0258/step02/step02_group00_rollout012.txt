import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max function"""
    exp_x = np.exp(alpha * x)
    return exp_x / (np.sum(exp_x))

def compute_overlap_gr_and_hessian(h, n_points, dx):
    """Compute gradients and Hessian for overlap calculation"""
    # Compute correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Smooth max with alpha=100
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    smooth_max_val = np.log(sum_exp) / alpha
    grad_corr = alpha * exp_corr / sum_exp
    
    # Gradients for h
    grad = np.zeros(n_points)
    # Manually compute gradient using finite differences
    eps = 1e-8
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        smooth_plus = np.log(np.sum(np.exp(alpha * corr_plus))) / alpha
        grad[i] = (smooth_plus - smooth_max_val) / eps
    return smooth_max_val, grad

def optimize_with_adam(h_start, n_points, dx, time_limit):
    """Optimize using Adam-like method"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    momentum = np.zeros_like(h_start)
    velocity = np.zeros_like(h_start)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Adam-like update loop
    for t in range(1000):
        if time() - start_time > time_limit:
            break
            
        # Compute gradient approximations with finite difference
        c5_current = evaluate_erdos_solution(h_start, dx)
        # Update with small perturbations
        grad = np.zeros_like(h_start)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_start.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update momentum and velocity
        momentum = beta1 * momentum + (1 - beta1) * grad
        velocity = beta2 * velocity + (1 - beta2) * (grad ** 2)
        mhat = momentum / (1 - beta1 ** (t + 1))
        vhat = velocity / (1 - beta2 ** (t + 1))
        
        # Update weights
        h_start = h_start - lr * mhat / (np.sqrt(vhat) + epsilon)
        h_start = project_to_feasible(h_start, n_points)
        
        c5 = evaluate_erdos_solution(h_start, dx)
        if c5 < best_c5:
            best_h = h_start.copy()
            best_c5 = c5
            
    return best_h, best_c5

        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:

            h_start = project_to_feasible(h_interp, n_points_try)
