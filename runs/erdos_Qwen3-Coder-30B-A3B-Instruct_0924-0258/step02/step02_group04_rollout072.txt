import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use a single peak pattern known to work well
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-1:mid+2] = 1.0
        h = project_to_feasible(h, n_points)
    return h

def optimize_fourier_spectrum(h_start, n_points, dx, time_limit):
    """Optimize h using Fourier-based approach to minimize overlap"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Ensure we have initial feasible solution
    h_best = project_to_feasible(h_best, n_points)
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # For Fourier optimization, focus on spectrum shaping
    freqs = np.fft.fftfreq(n_points, dx)
    # Use a simple gradient update on the power spectrum
    max_iter = 1000
    step_size = 1e-3
    penalty_weight = 0.1
    
    # Keep track of stagnant improvement
    last_improvement = 0
    last_c5 = c5_best
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
        
        # Convert h to frequency domain
        H = np.fft.fft(h_best)
        # Power spectrum
        P = np.abs(H)**2
        
        # Objective: penalize high power at specific frequencies
        # We want to encourage flat spectrum to reduce autocorrelation peaks
        penalty = np.sum(P**2) * penalty_weight
        
        # Compute gradient of penalty w.r.t. H
        # dP/dH = 2 * H * np.conj(H) * H = 2 * H * |H|**2 
        # Actually, take derivative of |H|^4 w.r.t. H = 4 * H * |H|^2
        d_penalty = 4 * H * P
        
        # Update H using gradient descent (in frequency domain)
        H_new = H - step_size * d_penalty
        
        # Convert back to time domain
        h_new = np.real(np.fft.ifft(H_new))
        
        # Apply feasibility constraints
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check if improved
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adjust step size if stagnated
        if no_improvement_count > 50:
            step_size *= 0.9
        
        # Occasionally do a more aggressive jump
        if iteration % 100 == 0 and iteration > 0:
            h_jump = h_best + np.random.normal(0, 0.05, n_points)
            h_jump = project_to_feasible(h_jump, n_points)
            c5_jump = evaluate_erdos_solution(h_jump, dx)
            if c5_jump < c5_best:
                h_best = h_jump
                c5_best = c5_jump
        
        # Early exit if too stagnated
        if iteration - last_improvement > 200 and iteration > 100:
            break
    
    # Final cleanup with local search
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, h_best, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < c5_best:
                h_best = h_final
                c5_best = c5_final
    except:
        pass
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use progressive refinement
    resolutions = [15, 30, 45, 60]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation per resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 150)
        
        # Perform optimization at this resolution
        try:
            h_opt, c5_opt = optimize_fourier_spectrum(h_start, n_points, dx, time_per_res)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
        except Exception as e:
            continue
    
    # If nothing was found, return initial solution
    if best_h is None:
        n_points = 15
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_fourier_spectrum(h_start, n_points, dx, budget_s)
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
