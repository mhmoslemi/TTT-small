import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Precompute the correlation matrix for all shifts
    def compute_overlaps(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Convex-concave procedure
    best_h = h.copy()
    best_C5 = compute_overlaps(best_h)
    
    # Initialize with existing solution if available
    if 'initial_h_values' in locals() or 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        best_h = h.copy()
        best_C5 = compute_overlaps(best_h)
    
    max_iter = 100
    improvement_tolerance = 1e-6
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find which shift contributes most to C5
        conv = np.correlate(best_h, 1 - best_h, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        max_idx = np.argmax(overlaps)
        
        # Use CVXPY to solve a convex subproblem
        # We want to minimize max_k integral(h(x)(1-h(x+k))dx)
        # This is equivalent to minimizing max_k sum(h[i](1-h[i+k]))*dx
        # This is done by solving a quadratic program
        
        # Create a variable for h
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Objective: minimize C5
        objective = cp.Minimize(C5_var)
        
        constraints = []
        # Constraints from shifts that contribute to the maximum
        for k in range(-n_points+1, n_points):
            # For all shifts k, add constraint
            if abs(k) < n_points:
                if k >= 0:
                    # sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k]) * dx <= C5
                    idx1 = list(range(n_points - k))
                    idx2 = list(range(k, n_points))
                    expr = cp.sum(cp.multiply(h_var[idx1], 1 - h_var[idx2])) * dx
                else:
                    idx1 = list(range(-k, n_points))
                    idx2 = list(range(n_points + k))
                    expr = cp.sum(cp.multiply(h_var[idx1], 1 - h_var[idx2])) * dx
                constraints.append(expr <= C5_var)
        
        # Feasibility constraints
        constraints.extend([
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ])
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == "optimal":
                new_h = h_var.value
                new_h = project_feasible(new_h)
                C5 = compute_overlaps(new_h)
                
                if C5 < best_C5:
                    best_h = new_h.copy()
                    best_C5 = C5
                    last_improvement = 0
                else:
                    last_improvement += 1
            else:
                # If optimization failed, fallback to projected gradient descent
                pass
        except:
            # If CVXPY fails, fall back to simple i.i.d. optimization
            pass
            
    # Final computation
    final_C5 = compute_overlaps(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
