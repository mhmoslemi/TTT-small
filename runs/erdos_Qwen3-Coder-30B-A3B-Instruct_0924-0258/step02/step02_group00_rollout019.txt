import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40):
    # Find current maximum overlap and corresponding shift(s)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_indices = np.where(corr == max_val)[0]
    
    # Subgradient computation
    subgradient = np.zeros_like(h)
    for idx in max_indices:
        # idx corresponds to shift k, so we add derivative for h[i] based on h[(i-k)%n] 
        # For the correlation at shift k: overlap = sum h[i] * (1 - h[i+k]) 
        # So d/dh[i] = (1 - h[i+k]) 
        # But due to periodicity, for overlap at shift k, we compute conv h, 1-h
        # The derivative wrt h[i] for this shift is 1-h[i-k]
        # Remember conv gives corr[k] = sum h[i] * (1 - h[i-k]), so d/dh[i] = (1 - h[i-k])
        for i in range(n_points):
            k = idx - i
            k = k % n_points
            subgradient[i] += (1 - h[k])
            
    # Average subgradients from all max shifts
    subgradient /= len(max_indices)
    
    # Linear optimization problem: min s in feasible region: g·s = sum_i g[i]*s[i]
    # subject to sum(s) = n/2, 0 <= s[i] <= 1
    # We want to minimize g·s, so we want s to be as negative as possible
    # s = argmin_g·s subject to sum(s) = n/2, 0 <= s <= 1
    # So we set s = min(1, max(0, s_base)) where s_base = -g + lambda (to maintain sum) 
    # But it's easier to solve this with Frank-Wolfe-style approx.
    # s = argmin_g·s subject to sum(s) = n/2, 0 <= s <= 1
    # Use the fact that it's solved by setting s = project_to_feasible[-g] but  
    # we must preserve the sum. Simplify: find s such that sum(s) = n/2 and s[i] = max(0, min(1, -g[i]))
    # compute an s that respects sum constraint
    s = np.clip(-subgradient, 0, 1)
    s_sum = s.sum()
    if s_sum > 0:
        s = s * (n_points / 2) / s_sum
    else:
        s = np.ones_like(s) * (n_points / 2) / n_points
    
    # Backtracking line search for step size
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    gamma = 0.9
    for _ in range(20):
        h_new = h + gamma * (s - h)
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            gamma *= 0.7
        if gamma < 1e-5:
            break
    return best_h, best_c5

    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available

        n_points = len(h_start)
    else:
        n_points = 25
        h_start = np.ones(n_points) * 0.5
        # Slight perturbation to break symmetry
        h_start += np.random.normal(0, 0.05, n_points)  
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            # Interpolate to new resolution
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Apply Frank-Wolfe optimization with small number of iterations
        current_h = h_start.copy()
        current_c5 = evaluate_erdos_solution(current_h, dx_try)
        
        # Use more intensive optimization for last resolution
        steps = 40 if n_points_try == resolutions[-1] else 20
        
        for i in range(steps):
            if time() - start_time > budget_s * 0.95:
                break
                
            new_h, new_c5 = frank_wolfe_step(current_h, n_points, dx_try)
            if new_c5 < current_c5:
                current_h = new_h.copy()
                current_c5 = new_c5
                
            # Occasionally reinitialize with a slightly different approach
            if i % 10 == 0 and i > 0:
                perturbation = np.random.normal(0, 0.02, n_points)
                current_h = project_to_feasible(current_h + perturbation, n_points)
                current_c5 = evaluate_erdos_solution(current_h, dx_try)
                
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            best_n_points = n_points
            
        h_start = current_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
