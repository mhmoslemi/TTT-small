import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iterations = 1000
    inertia_start = 0.9
    inertia_end = 0.4
    c1, c2 = 2.0, 2.0
    velocity_max = 0.15
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        swarm.append(h.copy())
        velocities.append(np.zeros_like(h))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros_like(h))
    
    # Add random solutions
    for _ in range(swarm_size - len(swarm)):
        h_rand = np.random.rand(n_points) * 0.5 + 0.25
        h_rand = project_feasible(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.rand(n_points) * 0.05 - 0.025)
    
    # PSO main loop
    best_global = None
    best_global_val = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Update inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_iterations
        
        for i in range(swarm_size):
            particle = swarm[i]
            
            # Compute overlap for current particle
            current_val = compute_overlap(particle)
            
            # Update personal best
            if current_val < best_global_val:
                best_global_val = current_val
                best_global = particle.copy()
            
            # Update swarm best
            if i == 0 or current_val < compute_overlap(swarm[0]):
                swarm[0] = particle.copy()
            if i < len(swarm) and current_val < compute_overlap(swarm[i]):
                swarm[i] = particle.copy()
                
        # Update particles
        for i in range(swarm_size):
            if i == 0:
                continue  # Skip updating the master particle
                
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocity = (inertia * velocity + 
                       c1 * r1 * (swarm[0] - particle) + 
                       c2 * r2 * (best_global - particle))
            
            # Clip velocity
            velocity = np.clip(velocity, -velocity_max, velocity_max)
            
            # Update position
            particle = particle + velocity
            
            # Project to feasible region
            particle = project_feasible(particle)
            
            # Update swarm and velocity
            swarm[i] = particle
            velocities[i] = velocity
            
        iteration += 1
    
    if best_global is None:
        best_global = swarm[0]
    
    final_C5 = compute_overlap(best_global)
    
    return (best_global.tolist(), final_C5, n_points)
