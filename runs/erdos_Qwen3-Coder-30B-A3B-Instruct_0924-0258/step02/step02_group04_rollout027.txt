import numpy as np
from scipy.optimize import minimize
from time import time
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    samples = []
    for _ in range(n_samples):
        sample = np.random.uniform(0, 1, n_points)
        samples.append(sample)
    return np.array(samples)

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level=0.1):
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use smaller resolutions for faster exploration
    resolutions = [15, 30, 45, 60]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Multi-start sampling with Latin Hypercube
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Use multiple restarts with Latin hypercube samples
        lhs_samples = latin_hypercube_sample(n_points, 20)
        candidates = [h_start] + [project_to_feasible(h, n_points) for h in lhs_samples]
        
        for i, h_init in enumerate(candidates):
            if time() - start_time > budget_s * 0.95:
                break
            try:
                # Perform optimization using L-BFGS-B
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                # Initialize with current best
                h_opt = h_init.copy()
                c5_val = evaluate_erdos_solution(h_opt, dx)
                if c5_val < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_val
                    best_n_points = n_points
                
                # Local optimization with multiple restarts
                for _ in range(20):
                    if time() - start_time > budget_s * 0.95:
                        break
                    
                    h_perturbed = perturb_h(h_opt, 0.05)
                    try:
                        res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                                      bounds=bounds, options={'maxiter': 30})
                        if res.success:
                            h_sample = project_to_feasible(res.x, n_points)
                            c5_sample = evaluate_erdos_solution(h_sample, dx)
                            if c5_sample < best_c5:
                                best_h = h_sample.copy()
                                best_c5 = c5_sample
                                best_n_points = n_points
                    except:
                        pass
                    
                    # Use adaptive step size
                    h_opt = perturb_h(h_opt, 0.02)
                    c5_val = evaluate_erdos_solution(h_opt, dx)
                    if c5_val < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_val
                        best_n_points = n_points
                        
            except Exception as e:
                continue
    
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = h_start, evaluate_erdos_solution(h_start, dx)
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
