import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Differential Evolution parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 1000
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            individual = h.copy()
        else:
            # Perturb initial solution
            individual = h + np.random.normal(0, 0.05, n_points)
        individual = project_feasible(individual)
        population.append(individual)
    
    best_h = None
    best_val = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            val = compute_overlap(individual)
            fitness.append(val)
            if val < best_val:
                best_val = val
                best_h = individual.copy()
                
        # Evolution step
        new_population = []
        for i in range(population_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Select three different individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Differential mutation
            mutant = a + mutation_factor * (b - c)
            mutant = project_feasible(mutant)
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.random() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            trial = np.array(trial)
            
            # Selection
            trial_val = compute_overlap(trial)
            if trial_val < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = new_population
        
    if best_h is None:
        # If we didn't find a better solution, use the latest best from population
        final_fitness = [compute_overlap(ind) for ind in population]
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        best_val = final_fitness[best_idx]
    
    # Final validation
    final_h = project_feasible(best_h)
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
