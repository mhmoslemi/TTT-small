import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # PSO parameters
        n_particles = 50
        max_iter = 200
        w_start = 0.7
        w_end = 0.4
        c1 = 2.0
        c2 = 2.0
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            pbest.append(h.copy())
            pbest_vals.append(evaluate_erdos_solution(h, dx_try))
            
        # Global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # PSO iteration
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            w = w_start - (w_start - w_end) * iter_num / max_iter
            
            for i in range(n_particles):
                r1, r2 = np.random.random(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                particles[i] = particles[i] + velocities[i]
                particles[i] = project_to_feasible(particles[i], n_points)
                
                current_val = evaluate_erdos_solution(particles[i], dx_try)
                
                if current_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = current_val
                    
                    if current_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = current_val
        
        # Update best if new solution is better
        if gbest_val < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_val
            best_n_points = n_points
        
        h_start = best_h.copy()
    
    # Final refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def gradient(h):
            # Simple finite difference gradient
            eps = 1e-6
            grad = np.zeros_like(h)
            base_val = objective(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                grad[i] = (objective(h_plus) - base_val) / eps
            return grad
        
        # Gradient descent
        h_refine = best_h.copy()
        learning_rate = 0.001
        for i in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            grad = gradient(h_refine)
            h_refine = h_refine - learning_rate * grad
            h_refine = project_to_feasible(h_refine, n_points)
        
        final_c5 = evaluate_erdos_solution(h_refine, dx)
        if final_c5 < best_c5:
            best_h = h_refine.copy()
            best_c5 = final_c5
    
    return (list(best_h), best_c5, best_n_points)
