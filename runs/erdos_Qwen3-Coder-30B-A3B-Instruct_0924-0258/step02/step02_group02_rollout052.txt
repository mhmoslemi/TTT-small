import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_gradients(h_vec, s_min):
        # Compute gradient of A_{s_min} with respect to h
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + s_min < len(h_vec):
                grad[i] += h_vec[i + s_min]
            if i >= s_min:
                grad[i] += h_vec[i - s_min]
        return grad

    def find_min_shift(h_vec):
        # Find the shift with minimum autocorrelation
        min_corr = float('inf')
        min_shift = 0
        for s in range(len(h_vec)):
            # Compute correlation for shift s
            corr = 0.0
            for i in range(len(h_vec) - s):
                corr += h_vec[i] * h_vec[i + s]
            corr *= dx
            if corr < min_corr:
                min_corr = corr
                min_shift = s
        return min_shift, min_corr

    def optimize_step(h_vec, learning_rate=0.01):
        s_min, _ = find_min_shift(h_vec)
        grad = compute_gradients(h_vec, s_min)
        h_new = h_vec + learning_rate * grad
        return project_h(h_new)

    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 500
    learning_rate = 0.01

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Try to improve using gradient-based step
        h_new = optimize_step(best_h, learning_rate)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count % 50 == 0:
                learning_rate *= 0.9
            # Add noise to escape local minima
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h_new = best_h + noise
            h_new = project_h(h_new)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                no_improvement_count = 0
                
        iter_count += 1
        if no_improvement_count >= max_no_improvement:
            break

    return (best_h.tolist(), best_c5, n_points)
