import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Shift the correlation to make it match the form needed
        # Compute the overlap for shift k: integral h(x)(1-h(x+k))
        corr_full = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr_full *= dx
        return np.min(corr_full) if len(corr_full) > 0 else 0

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def gradient_ascent_step(h_vec, learning_rate=0.01):
        # Compute all correlations (overlap with 1-h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        # Find index of minimum correlation
        min_idx = np.argmin(corr)
        # The indices are [0, 1, ..., 2*n_points-2]
        # The shift k is: min_idx - (n_points - 1)
        k = min_idx - (n_points - 1)
        # Create the gradient for this shift
        grad = np.zeros_like(h_vec)
        # Add gradient contribution from h[i] * h[i + k] 
        # This is for the term h[i] * (1 - h[i + k]) in the overlap
        # It's roughly the derivative of that term
        if k >= 0:
            # h[i] * (1 - h[i + k]) contributes: -h[i + k] to d/dh[i]
            grad[k:] += -h_vec[:-k] * dx  # multiplier = dx?
        else:
            # h[i] * (1 - h[i + k]) contributes h[i + k] to d/dh[i] 
            grad[:k] += -h_vec[-k:] * dx
        # Normalize gradient by sum or just use as is
        # Update h
        h_new = h_vec + learning_rate * grad
        return project_h(h_new)

    # Optimized version using simple gradient DESCENT on min autocorr
    # Because maximize min means minimize max, we work with min autocorr = 1 - C5
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Early stopping if no improvement for too long
        if no_improvement_count > 500:
            break
            
        h_new = gradient_ascent_step(h, learning_rate=0.01)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
