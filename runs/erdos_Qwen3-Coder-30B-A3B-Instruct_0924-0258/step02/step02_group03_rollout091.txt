import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    h0 = project_to_feasible(h0, n_points / 2.0)

    n_particles = 100
    particles = []
    weights = []
    n_evals = 0
    
    # Initialize swarm around the good solution
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h)
    
    # Initialize weights
    best_c5 = float('inf')
    best_h = None
    
    # Adaptive parameters
    temperature = 1.0
    proposal_stdev = 0.1
    acceptance_count = 0
    acceptance_window = 10
    last_improvement = 0

    while time.time() - start_time < budget_s and last_improvement < 50:
        # Importance sampling phase
        new_particles = []
        new_weights = []
        total_weight = 0
        
        for i, h in enumerate(particles):
            # Propose new particle
            h_new = h + np.random.normal(0, proposal_stdev, n_points)
            h_new = project_to_feasible(h_new, n_points/2.0)
            
            # Evaluate new particle
            c5 = evaluate(h_new, 2.0/n_points)
            n_evals += 1
            
            # Compute importance weight
            weight = np.exp(-c5 / temperature)
            
            new_particles.append(h_new)
            new_weights.append(weight)
            total_weight += weight
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                last_improvement = 0
            
        # Normalize weights
        if total_weight > 0:
            new_weights = np.array(new_weights) / total_weight
        
        # Resample
        indices = np.random.choice(n_particles, size=n_particles//2, 
                                  p=new_weights, replace=True)
        particles = [new_particles[i] for i in indices]
        
        # Mutation phase
        for i in range(len(particles)):
            # Clone particle
            new_h = particles[i].copy()
            
            # Two mutations: one global, one local
            # Global mutation
            noise = np.random.normal(0, proposal_stdev, n_points)
            new_h1 = new_h + noise
            new_h1 = project_to_feasible(new_h1, n_points/2.0)

            noise_local = np.random.normal(0, 0.05, n_points)
            new_h2 = new_h + noise_local
            new_h2 = project_to_feasible(new_h2, n_points/2.0)

            c5_1 = evaluate(new_h1, 2.0/n_points)
