import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_population(population):
        fitness = []
        for h_vec in population:
            f = compute_overlap(h_vec)
            fitness.append(f)
        return np.array(fitness)
    
    # Differential Evolution parameters
    pop_size = 25
    F = 0.7
    CR = 0.5
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h_vec = h.copy()
        else:
            # Add small random perturbations to initial solution
            h_vec = h + np.random.normal(0, 0.05, n_points)
        h_vec = project_feasible(h_vec)
        population.append(h_vec)
    
    best_individual = None
    best_fitness = float('inf')
    
    # DE loop
    max_generations = 1000
    generation = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        fitness = evaluate_population(population)
        
        # Track best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_fitness:
            best_fitness = fitness[best_idx]
            best_individual = population[best_idx].copy()
        
        # Create new population
        new_population = []
        
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            trial = np.where(np.random.random(n_points) < CR, mutant, population[i])
            
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Selection
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = new_population
        generation += 1
    
    # Final evaluation
    if best_individual is not None:
        final_fitness = compute_overlap(best_individual)
    else:
        # Fallback to last population member if nothing was tracked
        final_fitness = compute_overlap(population[0])
        best_individual = population[0]
        
    return (best_individual.tolist(), float(final_fitness), n_points)
