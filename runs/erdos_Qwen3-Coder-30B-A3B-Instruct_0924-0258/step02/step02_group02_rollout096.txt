import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def mutate(h_vec, rate=0.2):
        h_copy = h_vec.copy()
        if np.random.random() < rate:
            # Mutate 1 or 2 random indices
            n_mutate = np.random.choice([1, 2])
            indices = np.random.choice(n_points, n_mutate, replace=False)
            for i in indices:
                h_copy[i] += np.random.normal(0, 0.05)
        return project_h(h_copy)

    def crossover(h1, h2, rate=0.8):
        if np.random.random() < rate:
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
            return project_h(child)
        else:
            # Return one of the parents
            return project_h(h1 if np.random.random() < 0.5 else h2)

    def evaluate_fitness(h_vec):
        return -compute_c5(h_vec)

    # Initialize population
    population_size = 100
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        # Create random individuals
        h = np.random.random(n_points)
        h = project_h(h)
        population.append(h)

    # Track best solution
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    generation = 0
    max_generations = 500
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        if generation >= max_generations:
            break
            
        # Evaluate population fitness
        fitness_scores = [evaluate_fitness(h) for h in population]
        
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        current_best_fitness = fitness_scores[0]
        if -current_best_fitness < best_c5:
            best_c5 = -current_best_fitness
            best_h = population[0].copy()
            
        # Generate offspring
        new_population = [population[0]]  # Elitism
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitness)]
            
            # Select second parent
            second_parent = population[np.random.choice(population_size)]
            
            # Crossover and mutation
            child = crossover(population[winner_index], second_parent)
            child = mutate(child)
            new_population.append(child)
            
        population = new_population
        generation += 1
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
