def evaluate_erdos_solution(h_values):
    # This function evaluates the C5 value of a given solution
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    conv = np.correlate(h, 1 - h, mode='full')
    mid = len(conv) // 2
    overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
    return np.max(overlaps)
