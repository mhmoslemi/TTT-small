import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_fitness(h_vec):
        return -compute_c5(h_vec)

    # Genetic Algorithm
    population_size = 100
    num_generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        individual = np.random.rand(n_points)
        individual = project_h(individual)
        population.append(individual)
    
    best_solution = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for generation in range(num_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = [evaluate_fitness(ind) for ind in population]
        
        # Track best solution
        current_best_idx = np.argmax(fitness_scores)
        current_best_c5 = -fitness_scores[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_solution = population[current_best_idx].copy()
        
        # Selection with tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament]
            winner_idx = tournament[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = [population[current_best_idx].copy()]
        for i in range(0, population_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Simple average crossover
                child1 = (parent1 + parent2) / 2.0
                child2 = (parent1 + parent2) / 2.0
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                child1[mutation_idx] += np.random.normal(0, 0.01)
                child1[mutation_idx] = max(0, min(1, child1[mutation_idx]))
                
            if np.random.rand() < mutation_rate:
                mutation_idx = np.random.randint(0, n_points)
                child2[mutation_idx] += np.random.normal(0, 0.01)
                child2[mutation_idx] = max(0, min(1, child2[mutation_idx]))
            
            # Project to constraints
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
        
        # Keep population size consistent
        population = new_population[:population_size]
    
    # Final validation
    if best_solution is None:
        best_solution = h.copy()
    
    final_c5 = compute_c5(best_solution)
    
    return (best_solution.tolist(), final_c5, n_points)
