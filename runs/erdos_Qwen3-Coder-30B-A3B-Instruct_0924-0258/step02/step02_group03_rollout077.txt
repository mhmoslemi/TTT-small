import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    
    # Clamp to [0,1] again after scaling
    h = np.clip(h, 0, 1)
    
    return h

        # Start near initial solution or random
        if i == 0 and 'initial_h_values' in globals():
            p = h0.copy()
        else:

        # Ensure initial feasibility (with room for perturbation)
        p = project_to_feasible(p, n_points / 2.0)
        
        particles.append(p.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(p.copy())
        pbest_values.append(evaluate(p, 2.0 / n_points))

    # Global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO main loop
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (omega * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) + 
                            c2 * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            v_max = v_max_ratio * (np.max(particles[i]) - np.min(particles[i]))
            if v_max <= 1e-10:
                v_max = 0.05
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points / 2.0)
            
            # Evaluate
            current_value = evaluate(particles[i], 2.0 / n_points)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest_values[i] = current_value
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < gbest_value:
                    gbest_value = current_value
                    gbest_position = particles[i].copy()

        iteration += 1

    return (gbest_position.tolist(), float(gbest_value), n_points)
