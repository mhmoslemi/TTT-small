import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    inertia = 0.8
    c1, c2 = 2.0, 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros(n_points))
        
    # Add random particles
    for _ in range(swarm_size - len(swarm)):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_feasible(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Track personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_values = [compute_overlap(swarm[i]) for i in range(swarm_size)]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = 0.9 - (0.5 * iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_best[i] - swarm[i])
            social = c2 * r2 * (global_best - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate new solution
            current_value = compute_overlap(swarm[i])
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
                    
    # Final validation
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
