import time

    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_autocorrelations(h_vec):
        """Compute all autocorrelations of h with itself"""
        n = len(h_vec)
        autocorrs = []
        for k in range(n):
            # Compute sum_{i=0}^{n-k-1} h[i] * h[i+k]
            corr = 0.0
            for i in range(n - k):
                corr += h_vec[i] * h_vec[i + k]
            autocorrs.append(corr * dx)
        return np.array(autocorrs)

    def compute_gradient(h_vec, s_min):
        """Compute gradient of autocorrelation at shift s_min"""
        n = len(h_vec)
        grad = np.zeros(n)
        for i in range(n):
            if i + s_min < n:
                grad[i] += h_vec[i + s_min]
            if i >= s_min:
                grad[i] += h_vec[i - s_min]
        return grad

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    # Main optimization loop
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    max_patience = 50
    last_improvement = 0

        if elapsed > budget_s:
            break
            
        # Compute autocorrelations to find the minimal one
        autocorrs = compute_autocorrelations(h)
        s_min = np.argmin(autocorrs)
        min_autocorr = autocorrs[s_min]
        
        # Compute gradient
        grad = compute_gradient(h, s_min)
        
        # Update h
        h = h + learning_rate * grad
        
        # Project back to feasible region
        h = project_h(h)
        
        # Evaluate new C5
        new_c5 = compute_c5(h)
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

            patience = 0
            # Reduce learning rate for stability
            learning_rate *= 0.95

        if iteration % 300 == 0 and iteration > last_improvement:
            h = np.random.rand(n_points) * 0.5  # Explore random region
            h = project_h(h)
            
        # Prevent infinite loops
        if iteration > 10000:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
