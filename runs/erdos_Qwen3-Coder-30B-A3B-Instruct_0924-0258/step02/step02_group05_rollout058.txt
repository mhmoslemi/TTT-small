import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # CCCP approach
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    improvement_tolerance = 1e-5
    max_cccp_iter = 50
    
    for iter_count in range(max_cccp_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create and solve convex subproblem
        h_vars = Variable(n_points)
        
        # This is a simplified version to construct a QP inner approximation
        # The true approach would handle all bilinear terms via convex relaxation
        # For now we'll solve using alternating projections and gradient-based updates
        
        # Setup: residuals for difference between current h and target
        # For each k, we compute h[i] * (1 - h[i+k])
        # We minimize the maximum such residual subject to constraints
                  
        # Create a relaxed QCQP formulation using linearization at current point
        # But since we're simplifying, we'll just try to directly minimize using scipy
        
        # Instead, perform low-cost iterative improvement using projected gradient
        # Update locally to reduce the maximum overlap
        old_C5 = compute_overlap(best_h)
        
        # Very simple gradient descent on C5
        h_current = best_h.copy()
        lr = 1e-3
        decay = 0.999
        for inner_iter in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Estimate gradient using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] = min(h_plus[i] + eps, 1.0)
                h_plus = project_feasible(h_plus)
                val_plus = compute_overlap(h_plus)
                
                h_minus = h_current.copy()
                h_minus[i] = max(h_minus[i] - eps, 0.0)
                h_minus = project_feasible(h_minus)
                val_minus = compute_overlap(h_minus)
                
                grad[i] = (val_plus - val_minus) / (2 * eps)
            
            # Gradient step
            h_current = h_current - lr * grad
            h_current = project_feasible(h_current)
            
            # Adaptive learning rate
            lr *= decay
            
        # Update best solution
        new_C5 = compute_overlap(h_current)
        if new_C5 < best_C5:
            best_h = h_current.copy()
            best_C5 = new_C5
        else:
            # Reduce learning rate if no improvement
            log_improvement = np.log(new_C5 / old_C5)
            if log_improvement > -10:
                lr *= 0.95
            else:
                break  # Converged or no improvement
    
    # Final validation
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
