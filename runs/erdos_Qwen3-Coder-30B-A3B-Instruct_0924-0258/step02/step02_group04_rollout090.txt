import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a binary LABS-inspired pattern
        # Place 1s with equal spacing to minimize autocorrelation
        h = np.zeros(n_points)
        ones_count = n_points // 2
        positions = np.linspace(0, n_points, ones_count + 1, endpoint=False)[1:].astype(int)
        h[positions] = 1.0
        h = project_to_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Define resolution steps to explore
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation for this resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Optimizer loop with random restarts
        h_current = h_start.copy()
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Use L-BFGS-B for local optimization
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Apply optimization with projection
            try:
                res = minimize(objective, h_current, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except:
                pass
            
            # Perturb current solution to avoid local minima
            noise = np.random.normal(0, 0.05, n_points)
            h_current = project_to_feasible(h_current + noise, n_points)
            
            # If we've exhausted time for this resolution, break early
            if time() - start_time > budget_s * 0.95:
                break
    
    # Final check: if we haven't found anything, return something
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h = h_start
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
