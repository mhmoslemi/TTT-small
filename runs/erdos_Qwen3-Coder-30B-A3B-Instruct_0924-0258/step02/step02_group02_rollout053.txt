import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_autocorrelation(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Only keep valid correlations (shifts from -n+1 to n-1)
        mid = len(corr) // 2
        valid_corr = corr[mid+1:mid+n_points]  # exclude shift 0
        return valid_corr

    def compute_grad_i(h_vec, s):
        # Compute gradient of A_s = sum h[i] * h[i+s] with respect to h[i]
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + s < len(h_vec):
                grad[i] += h_vec[i + s]
            if i >= s:
                grad[i] += h_vec[i - s]
        return grad

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Early termination counter
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    iteration = 0
    lr = 0.01
    no_improvement_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Identify the shift with minimal autocorrelation
        auto_corr_vals = compute_autocorrelation(h)
        best_shift = np.argmax(auto_corr_vals)  # Very careful here: we want min, not max
        # Find actual minimal autocorrelation and corresponding shift
        min_idx = np.argmin(auto_corr_vals)
        s_min = min_idx - (len(auto_corr_vals) // 2)  # shift index: -n+1 to n-1
        if s_min < 0:
            s_min = len(auto_corr_vals) + s_min - 1  # this round trips properly
        else:
            s_min = s_min + 1  # +1 to ensure correct index in autocorr
                
        # Actually, let's back off this approach - using scipy optimization
        
        # Instead, we'll use a coordinate descent style approach: 
        # 1. Evaluate all shifts, find min
        # 2. Compute the gradient at that point
        # 3. Do small step in direction that increases that min value
        auto_corr_vals = compute_autocorrelation(h)
        
        shift = np.argmin(auto_corr_vals)
        s_min = shift - (len(auto_corr_vals) // 2) # Convert to actual shift
        
        grad = compute_grad_i(h, s_min if s_min >= 0 else -s_min)
        
        h_new = h - lr * grad
        h_new = project_h(h_new)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count % 100 == 0:
                lr *= 0.5  # Reduce learning rate
            h_new = h + 0.001 * np.random.normal(0, 1, n_points)
            h_new = project_h(h_new)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                no_improvement_count = 0
        
        h = h_new
        
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
