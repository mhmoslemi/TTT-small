import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # ES parameters
    population_size = 30
    elite_size = population_size // 10
    mutation_rate = 0.4
    beta = 0.2
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Mutation strengths for each individual
    sigma = np.ones(population_size) * 0.05
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [evaluate(h, 2.0/n_points) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 50:
            break
            
        # Selection: keep elite
        elites = population[:elite_size]
        
        # Mutation and recombination
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            # Select parents based on rank
            parent_indices = np.random.choice(population_size, size=3, replace=False)
            parent_fitness = [fitness[i] for i in parent_indices]
            parent_idx = parent_indices[np.argmin(parent_fitness)]
            
            # Copy parent
            mutant = population[parent_idx].copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, sigma[parent_idx], n_points)
                mutant += noise
                mutant = project_to_feasible(mutant, n_points/2.0)
            
            new_population.append(mutant)
            
        # Update mutation strength (self-adaptation)
        for i in range(population_size):
            delta = np.random.normal(0, 1)
            sigma[i] *= np.exp(beta * delta)
            sigma[i] = max(0.001, min(0.5, sigma[i]))
        
        population = new_population
    
    if best_h is not None:
        final_c5 = evaluate(best_h, 2.0/n_points)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h0, n_points/2.0)
        final_c5 = evaluate(final_h, 2.0/n_points)
        return (final_h.tolist(), float(final_c5), n_points)
