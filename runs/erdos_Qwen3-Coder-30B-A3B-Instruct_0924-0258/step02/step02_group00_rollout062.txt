import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def compute_gradient(h, n_points, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    grad_corr = alpha * exp_corr / sum_exp
    
    # Gradient of correlation w.r.t. h
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Approximate gradient using finite differences
        eps = 1e-8
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        smooth_max_plus = smooth_max(corr_plus, alpha)
        
        h_base = h.copy()
        corr_base = np.correlate(h_base, 1 - h_base, mode='full')
        smooth_max_base = smooth_max(corr_base, alpha)
        
        grad[i] = (smooth_max_plus - smooth_max_base) / eps
    
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # For 50 iterations, use a decreasing learning rate
    steps = 50
    eta = 0.01
    for i in range(steps):
        if time() - start_time > time_limit:
            break
        
        try:
            grad = compute_gradient(best_h, n_points, dx)
            new_h = best_h - eta * grad
            new_h = project_to_feasible(new_h, n_points)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                
        except:
            pass
        
        # Decrease learning rate periodically  
        if i % 10 == 9:
            eta *= 0.9
    
    return best_h, best_c5
