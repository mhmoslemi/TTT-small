import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    # Manual correlation computation for full mode
    n = len(h)
    max_corr = 0.0
    for k in range(2*n - 1):
        corr = 0.0
        for i in range(n):
            if 0 <= i + k < n:
                corr += h[i] * (1.0 - h[i + k])
        max_corr = max(max_corr, corr)
    return max_corr * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need more sum - distribute to smallest values
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum - distribute from largest values
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    T = 0.5  # Initial temperature

            T = 0.5
            no_improvement_count = 0
        
        # Generate candidate by perturbing h
        candidate_h = h + np.random.normal(0, T, n_points)
        
        # Project candidate to feasible region
        candidate_h = project_to_feasible(candidate_h, n_points / 2.0)
        
        # Evaluate candidate
        candidate_c5 = evaluate(candidate_h, 2.0 / n_points)
        
        # Accept or reject candidate
        delta = candidate_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = candidate_h.copy()

                best_c5 = candidate_c5

        # Cool temperature
