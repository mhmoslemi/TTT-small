import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    # Parameters for simulated annealing
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    eta = 0.02  # Learning rate
    lambda_penalty = 1e-3  # Penalty weight
    max_no_improve = 50  # No improvement threshold
    
    # Initialize
    h_current = project_feasible(h_start.copy(), n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = current_c5
    no_improve_count = 0
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Find dominant lag for gradient computation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_best = np.argmax(corr)
        max_corr = corr[k_best]
        
        # Compute gradient for the dominant lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_best) % n_points
            grad[i] = (1 - h_current[j] - h_current[i]) * dx
            
        # Compute penalty gradient (sum constraint)
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(h_current)
        penalty_grad += 2 * lambda_penalty * (h_sum - n_points / 2)

        h_candidate = h_current - eta * (grad + penalty_grad)

        h_candidate = project_feasible(h_candidate, n_points)

                best_h = h_current.copy()
                best_c5 = current_c5
                no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Annealing
        T *= alpha
        
        # Reset counter if improved or perturb if stuck
        if no_improve_count >= max_no_improve:
            h_perturb = h_current + np.random.normal(0, 0.05, n_points)
            h_current = project_feasible(h_perturb, n_points)
