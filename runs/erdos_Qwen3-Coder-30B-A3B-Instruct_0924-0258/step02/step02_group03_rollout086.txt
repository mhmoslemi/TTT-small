import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, trust_radius, target_sum, dx, n):
    """Perform a trust region step to minimize the objective."""
    # For simplicity, we use a quadratic approximation: 
    # f ≈ f0 + g^T * d + 0.5 * d^T * H * d
    # Minimize with trust region constraint ||d|| ≤ Δ
    
    # Using a simplified approach: compute feasible step in direction -grad
    # subject to trust region and constraints
    
    # Projected gradient descent step (approximation)
    candidate = h - grad * 0.01
    
    # Apply trust region constraint (project to sphere)
    d = candidate - h
    norm_d = np.linalg.norm(d)
    
    if norm_d > trust_radius and norm_d > 1e-10:
        d = d * trust_radius / norm_d
    new_h = h + d
    
    # Clamp to [0,1]
    new_h = np.clip(new_h, 0, 1)
    
    # Scale to satisfy sum constraint
    new_h = project_to_feasible(new_h, target_sum)
    
    return new_h
