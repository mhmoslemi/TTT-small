import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    # Final scaling to ensure exact sum
    if abs(current_sum) > 1e-10:
        h *= target_sum / current_sum
    else:
        h.fill(target_sum / len(h))
    
    return h

    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max_ratio = 0.05
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    gbest_idx = 0
    gbest_value = float('inf')
    
    for i in range(swarm_size):
        # Start near initial solution or random
        if i == 0 and 'initial_h_values' in globals():
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        val = evaluate(h, dx)
        pbest.append(h.copy())
        pbest_values.append(val)
        if val < gbest_value:
            gbest_value = val
            gbest_idx = i
    
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = c1 * r1 * (pbest[i] - particles[i])
            social = c2 * r2 * (pbest[gbest_idx] - particles[i])
            velocities[i] = omega * velocities[i] + cognitive + social
            
            # Apply velocity limit
            v_max = v_max_ratio * (np.max(particles[i]) - np.min(particles[i]))
            if v_max < 1e-10:
                v_max = 0.05
            
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate
            val = evaluate(particles[i], dx)
            
            # Update personal best
            if val < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = val
                if val < gbest_value:
                    gbest_value = val
                    gbest_idx = i
    
        iteration += 1
    
    return (particles[gbest_idx].tolist(), float(gbest_value), n_points)
