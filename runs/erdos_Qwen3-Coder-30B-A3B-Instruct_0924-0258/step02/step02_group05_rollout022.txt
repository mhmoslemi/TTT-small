import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.2
        h[-1] = 0.8
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def solve_feasible_lp(C5_candidate):
        # Construct constraints for the LP
        # Variables are h[0], h[1], ..., h[n_points-1]
        # We're checking: h[i] * (1 - h[i+k]) <= C5_candidate * dx for all k
        # Overlap: sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k]) <= C5_candidate * dx
        # This defines C5_candidate * dx constraints
        
        # Constraint matrix and bounds
        A_eq = np.ones((1, n_points))
        b_eq = np.array([target_sum])
        
        # Inequality constraints
        A_ub = []
        b_ub = []
        
        # Compute all overlaps for specific k values
        k_range = range(-n_points+1, n_points)
        for k in k_range:
            if k == 0: continue
            row = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    row[i] = 1
            A_ub.append(row)
            b_ub.append(C5_candidate * dx)
            
        # Bound constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = linprog(
                c=np.zeros(n_points), 
                A_eq=A_eq, b_eq=b_eq,
                A_ub=A_ub, b_ub=b_ub,
                bounds=bounds,
                method='highs'
            )
            if result.success:
                h_feasible = result.x
                # Project to ensure feasibility if needed
                return project_feasible(h_feasible)
            else:
                return None
        except:
            return None
    
    # Binary search on C5
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    best_h = project_feasible(h)
    best_C5 = compute_overlap(best_h)
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        mid = (low + high) / 2
        
        h_feasible = solve_feasible_lp(mid)
        
        if h_feasible is not None:
            # Solution found - this C5 is feasible
            # Now check the actual computed C5
            actual_C5 = compute_overlap(h_feasible)
            if actual_C5 < best_C5:
                best_C5 = actual_C5
                best_h = h_feasible.copy()
            high = mid
        else:
            # No feasible solution - this C5 is too low
            low = mid
    
    return (best_h.tolist(), best_C5, n_points)
