import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Apply Gaussian mutation to an individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        mutated = individual + noise
        return mutated
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 35
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
        
    # GA parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    max_generations = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Add the initial solution
    h_feasible = project_feasible(h_start, n_points)
    population.append(h_feasible)
    fitness.append(evaluate_erdos_solution(h_feasible, 2.0/n_points))
    
    # Add random individuals
    for i in range(pop_size - 1):
        h_rand = np.random.random(n_points)
        h_feasible = project_feasible(h_rand, n_points)
        population.append(h_feasible)
        fitness.append(evaluate_erdos_solution(h_feasible, 2.0/n_points))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep best 25% and add some diversity
        new_pop = sorted_population[:pop_size//4]
        new_fitness = sorted_fitness[:pop_size//4]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
                
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Feasibility
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            # Evaluate
            new_fitness.append(evaluate_erdos_solution(offspring1, 2.0/n_points))
            new_pop.append(offspring1)
            
            if len(new_pop) < pop_size:
                new_fitness.append(evaluate_erdos_solution(offspring2, 2.0/n_points))
                new_pop.append(offspring2)
                
        # Survival: Keep best population
        combined_pop = new_pop
        combined_fitness = new_fitness
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_pop[i] for i in sorted_indices[:pop_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            
    return (list(best_h), best_c5, n_points)
