import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    swarm_size = 40
    max_iter = 10000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max_ratio = 0.05
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    particles = []
    velocities = []
    
    for _ in range(swarm_size):
        # Initialize around h_init with small noise
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    pbest_positions = [p.copy() for p in particles]
    pbest_values = [evaluate(p, 2.0/n_points) for p in particles]
    
    # Initialize global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            v = (omega * velocities[i] + 
                 c1 * r1 * (pbest_positions[i] - particles[i]) + 
                 c2 * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            v_max = v_max_ratio * (np.max(particles[i]) - np.min(particles[i]))
            if v_max <= 0:
                v_max = 0.1
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_pos = particles[i] + v
            new_pos = project_to_feasible(new_pos, n_points / 2.0)
            
            # Evaluate
            new_val = evaluate(new_pos, 2.0/n_points)
            
            # Update personal best
            if new_val < pbest_values[i]:
                pbest_positions[i] = new_pos.copy()
                pbest_values[i] = new_val
            
            # Update global best
            if new_val < gbest_value:
                gbest_position = new_pos.copy()
                gbest_value = new_val
            
            # Update particle
            particles[i] = new_pos
            velocities[i] = v
            
        iteration += 1
    
    # Return best solution found
    final_c5 = evaluate(gbest_position, 2.0/n_points)
    return (gbest_position.tolist(), float(final_c5), n_points)
