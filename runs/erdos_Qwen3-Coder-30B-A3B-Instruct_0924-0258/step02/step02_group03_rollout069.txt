import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):

    h0 = project_to_feasible(h0, n_points/2.0)
    
    # Parameters for Multi-Swarm Particle Filter
    num_particles = 100
    max_iterations = 10000
    temperature = 1.0
    proposal_sigma = 0.1
    acceptance_rate_target = 0.3
    patience = 50
    best_c5 = float('inf')
    best_h = None
    
    # Initialize particles with perturbed versions of h0
    particles = [project_to_feasible(h0 + np.random.normal(0, proposal_sigma, n_points), n_points/2.0)
                 for _ in range(num_particles)]
    
    # Track acceptance rate for sigma adaptation
    accept_count = 0
    total_proposals = 0
    
    # Main loop
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Importance sampling step
        weights = np.zeros(num_particles)
        new_particles = []
        c5_values = []
        
        for i, h in enumerate(particles):
            # Generate candidate particle
            h_candidate = h + np.random.normal(0, proposal_sigma, n_points)
            h_candidate = project_to_feasible(h_candidate, n_points/2.0)
            
            c5 = evaluate(h_candidate, 2.0/n_points)
            weights[i] = np.exp(-c5 / temperature) if temperature > 0 else 0.0
            
            new_particles.append(h_candidate)
            c5_values.append(c5)
        
        # Normalize weights
        if np.sum(weights) > 0:
            weights /= np.sum(weights)
        else:
            weights = np.ones(num_particles) / num_particles
        
        # Update best solution
        min_c5_idx = np.argmin(c5_values)

            best_c5 = c5_values[min_c5_idx]
            best_h = new_particles[min_c5_idx].copy()
            last_improvement = iteration
        
        # Resampling step with stratified resampling
        # Convert to cumulative distribution
        cum_weights = np.cumsum(weights)
        indices = []
        
        # Stratified resampling
        r = np.random.random()/num_particles
        for _ in range(num_particles):
            for j in range(len(weights)):
                if r <= cum_weights[j]:
                    indices.append(j)
                    break

        temperature *= 0.99

                    proposal_sigma *= 1.05
                elif acceptance_rate < acceptance_rate_target - 0.05:
                    proposal_sigma /= 1.05
                accept_count = 0
                total_proposals = 0

        if iteration - last_improvement > patience:
            break
    
    # Final check for best solution
    if best_h is None:
        # Revert to initial h if we never found a better solution
        # This shouldn't happen given our initial projection,
        # but is a fallback safeguard
        min_c5_idx = np.argmin([evaluate(h, 2.0/n_points) for h in particles])
        best_h = particles[min_c5_idx].copy()
    else:
        # Verify the final best_h is still properly constrained
        best_h = project_to_feasible(best_h, n_points/2.0)
    
    final_c5 = evaluate(best_h, 2.0/n_points)
    return (best_h.tolist(), float(final_c5), n_points)
