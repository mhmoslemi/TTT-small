import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    h_current = project_to_feasible(h0.copy(), n_points/2.0)
    c5_current = evaluate(h_current, 2.0/n_points)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    min_T = 1e-6
    acceptance_count = 0
    total_count = 0
    last_improvement = 0
    patience = 100
    
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Generate candidate
        noise = np.random.normal(0, T, n_points)
        h_candidate = h_current + noise
        h_candidate = project_to_feasible(h_candidate, n_points/2.0)
        c5_candidate = evaluate(h_candidate, 2.0/n_points)
        
        # Accept or reject
        if c5_candidate < c5_current or np.random.random() < np.exp(-(c5_candidate - c5_current) / T):
            h_current = h_candidate
            c5_current = c5_candidate
            acceptance_count += 1
        
        total_count += 1
        
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Cooling schedule
        if T > min_T:
            T *= alpha
            
        # Adaptive cooling
        if total_count % 50 == 0:
            acceptance_rate = acceptance_count / total_count
            if acceptance_rate < 0.2:
                alpha = max(0.99, alpha * 0.99)
            elif acceptance_rate > 0.5:
                alpha = min(0.9999, alpha * 1.01)
            acceptance_count = 0
            total_count = 0
            
        # Reset temperature if no improvement
        if last_improvement > patience:
            T = 0.5
            last_improvement = 0
            
        iteration += 1
        
    return (best_h.tolist(), float(best_c5), n_points)
