import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise use a simple guess
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Differential Evolution parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 500
    
    def compute_overlap(h_vec):
        # Compute correlations
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        # Take only relevant part, scale and return max
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initialize population
    population = []
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        population.append(project_feasible(individual))
    
    best_solution = None
    best_value = float('inf')
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for individual in population:
            val = compute_overlap(individual)
            fitness.append(val)
            if val < best_value:
                best_value = val
                best_solution = individual.copy()
        
        # Evolve population
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation (rand/1/bin)
            mutant = a + mutation_factor * (b - c)
            mutant = project_feasible(mutant)
            
            # Crossover (binomial)
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Selection
            trial_val = compute_overlap(trial)
            if trial_val < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = new_population
        generation += 1
    
    # Final validation and return
    if best_solution is not None:
        final_C5 = compute_overlap(best_solution)
        return (best_solution.tolist(), final_C5, n_points)
    else:
        final_C5 = compute_overlap(h)
        return (h.tolist(), final_C5, n_points)
