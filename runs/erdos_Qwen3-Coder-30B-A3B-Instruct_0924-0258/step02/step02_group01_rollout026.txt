import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project candidate_h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    """Select an individual via tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, noise_std=0.1):
    """Mutate an individual with Gaussian noise"""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(individual))
        individual += noise
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    n_points = 35
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():

        population.append(h_start.copy())
        n_points = len(h_start)
    else:
        h_start = np.ones(n_points) * 0.5
        population.append(h_start.copy())
    
    # Generate random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    dx = 2.0 / n_points
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolution loop
    generation = 0
    stagnation_counter = 0
    stagnation_limit = 20
    
    while time() - start_time < budget_s * 0.98 and stagnation_counter < stagnation_limit:
        generation += 1
        new_population = []
        
        # Elitism: keep best 25% 
        elite_count = population_size // 4
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Feasibility projection
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Survival: keep best individuals
        all_fitnesses = fitnesses + new_fitnesses
        all_population = population + new_population
        combined_indices = np.argsort(all_fitnesses)[:population_size]
        
        population = [all_population[i] for i in combined_indices]
        fitnesses = [all_fitnesses[i] for i in combined_indices]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
            stagnation_counter = 0
        else:
            stagnation_counter += 1
    
    return (list(best_h), best_c5, n_points)
