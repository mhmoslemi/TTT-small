import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, backtracking_attempts=10):
    """Frank-Wolfe optimization step with line search"""

    best_c5 = evaluate_erdos_solution(h, dx)
    h_best = h.copy()
    
    for iteration in range(max_iter):

        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute subgradient for the maximizing shift
        subgrad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - max_k) % n_points
            subgrad[i] = (1 - h[j]) - h[i]
        
        # Frank-Wolfe linear optimization
        # Minimize subgrad^T * s subject to sum(s) = 0, 0 <= s <= 1
        # This is equivalent to finding a feasible s that minimizes the dot product
        
        # Propositional approach: find optimal s
        s = np.zeros_like(h)
        # Find index that maximizes -subgrad for feasible descent
        # We want to move in direction that improves the function
        # We do not exactly solve the constrained problem but 
        # find the feasible s via projection and line search
        s = np.zeros_like(h)
        # Look for vertex in direction of -subgrad that is feasible wrt constraints
        
        # This represents a heuristic for finding the Frank-Wolfe direction
        index_min_grad = np.argmin(subgrad)
        s[index_min_grad] = 1.0
        # In general, we solve constrained quadratic or use more sophisticated
        # Frank-Wolfe with line-search, but we keep simpler here due to discrete nature
        
        # Simple approach: proceed with projection and line search
        for gamma in np.linspace(0.1, 0.9, backtracking_attempts):
            h_new = h + gamma * (s - h)
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                break

    return h, best_c5
