import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    # Initialize from the provided initial construction

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use a more sophisticated optimization approach
    # Gradient-based method with restarts to escape local minima
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try multiple random restarts
    n_restarts = 10
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Random perturbation of the initial solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_current = h + perturbation
        h_current = project_h(h_current)

        for _ in range(1000):
            if time.time() - start_time > budget_s - 1.0:
                break
                
            # Try different optimization methods
            methods = ['L-BFGS', 'TNC', 'SLSQP']
            for method in methods:

                    # Use scipy minimize with different algorithms
                    result = minimize(evaluate_objective, h_current, 
                                    method=method,

                                    options={'maxiter': 10, 'ftol': 1e-8})
                    if result.success:
                        h_current = result.x
                    break  # Use first successful method
                except:
                    continue
                    
            h_current = project_h(h_current)
            new_c5 = compute_c5(h_current)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_current.copy()
            
            # Add noise to escape local minima
            if np.random.random() < 0.1:
                noise = 0.001 * np.random.normal(0, 1, n_points)
                h_current = h_current + noise
                h_current = project_h(h_current)

    # Use a few iterations of refinement with method that works well for the final step
    h_final = best_h.copy()

        # Small gradient descent step
        eps = 1e-5
        grad = np.zeros_like(h_final)

            h_plus = h_final.copy()

            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps

        step_size = 0.001
        h_new = h_final - step_size * grad
        h_new = project_h(h_new)
        
        # Accept if improvement

            h_final = h_new

            h_final = h_final + noise
            h_final = project_h(h_final)
    
    final_c5 = compute_c5(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
