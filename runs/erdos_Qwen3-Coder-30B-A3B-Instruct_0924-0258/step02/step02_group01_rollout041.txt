import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_feasible(h_start, n_points)
    c5_current = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    learning_rate = 0.01
    penalty_weight = 1e-3
    max_no_improve = 50
    no_improve_count = 0
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Gradient computation for overlap integral
    def compute_gradient(h, k):
        # h is length n_points
        # Start with a simple gradient estimation
        # This assumes that the maximum overlap occurs at a specific lag k
        n = len(h)
        grad = np.zeros(n)
        for i in range(n):
            idx = (i + k) % n
            grad[i] = -(1 - h[idx] - h[i]) * dx
        return grad
    
    # Simulated Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Find the dominant lag k
        corr = np.correlate(h_current, 1-h_current, mode='full')
        k = np.argmax(corr) - (len(h_current)-1)
        
        # Compute gradient for this lag
        grad = compute_gradient(h_current, k)
        
        # Compute penalty gradient
        penalty_grad = np.zeros_like(h_current)
        h_sum = np.sum(h_current)
        penalty_grad += 2 * penalty_weight * (h_sum - n_points/2) * np.ones_like(h_current)
        penalty_grad += penalty_weight * np.maximum(0, 1-h_current)
        penalty_grad -= penalty_weight * np.maximum(0, -h_current)
        
        # Propose new point with gradient and penalty
        h_candidate = h_current + learning_rate * (-grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - c5_current
        
        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5/T) > np.random.rand():
            h_current = h_candidate
            c5_current = c5_candidate
            no_improve_count = 0
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        else:
            no_improve_count += 1
        
        # Temperature update
        T *= alpha
        
        # Annihilation
        if no_improve_count >= max_no_improve:
            h_current += np.random.normal(0, 0.05, n_points)
            h_current = project_feasible(h_current, n_points)
            no_improve_count = 0
    
    # Multi-resolution refinement
    resolutions = [60, 120]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        
        # Interpolate
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_new = project_feasible(h_interp, res)
        
        # Local optimization
        def obj(h):
            return evaluate_erdos_solution(h, 2.0/res)
        
        bounds = [(0, 1) for _ in range(res)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - res / 2}]
        
        try:
            res_opt = minimize(obj, h_new, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-6})
            if res_opt.success:
                h_opt = project_feasible(res_opt.x, res)
                c5_opt = evaluate_erdos_solution(h_opt, 2.0/res)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    n_points = res
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
