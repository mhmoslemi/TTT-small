import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use a more robust optimization approach
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters for optimization
    max_iter = 5000
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    
    # Function to compute overlap for all shifts
    def compute_all_overlaps(h_vec):
        overlaps = []
        for shift in range(n_points):
            overlap = 0
            for i in range(n_points - shift):
                overlap += h_vec[i] * (1 - h_vec[i + shift]) * dx
            overlaps.append(overlap)
        return np.array(overlaps)
    
    # Optimization loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlaps
        overlaps = compute_all_overlaps(h)
        
        # Find maximum overlap
        max_overlap = np.max(overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early termination if no improvement for a while
        if patience > patience_limit:
            break
            
        # Compute gradient estimates using finite differences
        gradient = np.zeros_like(h)
        eps = 1e-5
        current_c5 = evaluate_objective(h)
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = project_h(h_eps)
            c5_eps = evaluate_objective(h_eps)
            gradient[i] = (c5_eps - current_c5) / eps
        
        # Update h using gradient descent
        h = h - learning_rate * gradient
        h = project_h(h)
        
        # Adaptive learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.1
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
