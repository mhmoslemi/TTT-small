import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation for upsample
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    # PSO parameters
    swarm_size = 30
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    v_max = 0.15
    particle_positions = []
    particle_velocities = []
    particle_best_positions = []
    particle_best_values = []
    swarm_best_position = None
    swarm_best_value = float('inf')
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        particle_positions.append(h.copy())
    else:
        particle_positions.append(np.ones(n_points) * 0.5)
        
    for i in range(swarm_size - 1):
        # Start with random perturbations around initial solution
        perturbation = np.random.normal(0, 0.1, n_points)
        new_pos = particle_positions[0] + perturbation
        new_pos = project_feasible(new_pos)
        particle_positions.append(new_pos)
    
    for i in range(swarm_size):
        # Initialize velocities
        particle_velocities.append(np.random.uniform(-v_max, v_max, n_points))
        # Initialize personal best
        particle_best_positions.append(particle_positions[i].copy())
        particle_best_values.append(compute_overlap(particle_positions[i]))
        
        if particle_best_values[i] < swarm_best_value:
            swarm_best_value = particle_best_values[i]
            swarm_best_position = particle_positions[i].copy()
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            particle_velocities[i] = (w * particle_velocities[i] + 
                                    c1 * r1 * (particle_best_positions[i] - particle_positions[i]) +
                                    c2 * r2 * (swarm_best_position - particle_positions[i]))
            
            # Clip velocity
            particle_velocities[i] = np.clip(particle_velocities[i], -v_max, v_max)
            
            # Update position
            particle_positions[i] = particle_positions[i] + particle_velocities[i]
            
            # Project onto feasible region
            particle_positions[i] = project_feasible(particle_positions[i])
            
            # Evaluate new position
            current_value = compute_overlap(particle_positions[i])
            
            # Update personal best
            if current_value < particle_best_values[i]:
                particle_best_values[i] = current_value
                particle_best_positions[i] = particle_positions[i].copy()
                
                # Update swarm best
                if current_value < swarm_best_value:
                    swarm_best_value = current_value
                    swarm_best_position = particle_positions[i].copy()
    
    # Final evaluation
    final_h = project_feasible(swarm_best_position)
    final_c5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
