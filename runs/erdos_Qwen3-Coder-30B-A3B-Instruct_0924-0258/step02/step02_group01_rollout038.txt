import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    h_current = project_feasible(h_start.copy(), n_points)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    # Moments for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Optimization loop
    iteration = 0
    last_improvement = 0
    while time() - start_time < budget_s * 0.98:
        iteration += 1
        
        # Compute correlation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for the maximum overlap lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h_current[j]) - h_current[i]
        
        grad *= dx
        
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        t = iteration
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Parameter update
        h_current -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_current = project_feasible(h_current, n_points)
        
        # Evaluate
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            last_improvement = iteration
        
        # Restart if no improvement for a while
        if iteration - last_improvement > 500 and time() - start_time < budget_s * 0.9:
            h_current = project_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
            m.fill(0)
            v.fill(0)
            last_improvement = iteration
    
    return (list(best_h), best_c5, n_points)
