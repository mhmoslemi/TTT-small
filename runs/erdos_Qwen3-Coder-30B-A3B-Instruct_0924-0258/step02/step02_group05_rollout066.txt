import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation for upsample
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    # Optimizer parameters
    adam_lr = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    
    # Multi-resolution optimization
    # Use smaller resolution to start with to get a quick rough solution
    resolutions = [5, 10, 15, 25, 35, n_points]
    current_h = h.copy()
    
    for i, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize at this resolution
        if i == 0:
            h_vec = np.ones(res) * 0.5
        else:
            # Upsample previous solution (if exists)
            h_vec = upsample_h(current_h, res)
            h_vec = project_feasible(h_vec)
            
        # Adam optimizer variables
        m = np.zeros_like(h_vec)
        v = np.zeros_like(h_vec)
        
        # Simple optimization with moment-based updates
        last_improvement = 0
        best_val = compute_overlap(h_vec)
        best_h = h_vec.copy()
        
        # Local optimization at this resolution
        for iter_count in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Finite difference gradient
            gradient = np.zeros_like(h_vec)
            eps = 1e-4
            for j in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[j] += eps
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                h_minus = h_vec.copy()
                h_minus[j] -= eps
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps)
            
            # Adam update with gradient clipping
            m = adam_beta1 * m + (1 - adam_beta1) * gradient
            v = adam_beta2 * v + (1 - adam_beta2) * gradient**2
            m_hat = m / (1 - adam_beta1**(iter_count + 1))
            v_hat = v / (1 - adam_beta2**(iter_count + 1))
            h_vec = h_vec - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Clip gradient to prevent large jumps
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * target_sum / np.sum(h_vec)  # Re-normalize
            
            # Check improvement
            current_val = compute_overlap(h_vec)
            if current_val < best_val:
                best_val = current_val
                best_h = h_vec.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
            if last_improvement > 500:
                break
                
        current_h = best_h.copy()
    
    # Final check with full evaluation
    final_C5 = compute_overlap(current_h)
    
    return (current_h.tolist(), final_C5, n_points)
