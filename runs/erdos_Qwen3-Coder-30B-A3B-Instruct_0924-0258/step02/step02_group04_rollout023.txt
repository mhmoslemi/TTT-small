import numpy as np
from scipy.fftpack import fft, ifft
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise use flat spectrum"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a flat spectrum to encourage minimal overlap
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def optimize_fourier_spectrum(h_start, n_points, dx, time_limit):
    """Optimize h in Fourier domain to minimize overlap"""
    np.random.seed(42)
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    alpha = 1e-4
    lambda_penalty = 0.1
    max_iter = 1000
    patience = 500
    
    # Initialize h
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = c5_best
    last_improvement = 0
    
    # Fourier domain variables
    H = fft(h)
    
    # Frequency indices
    freqs = np.arange(n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Update spectrum to penalize peaks
        H_mag = np.abs(H)
        H_phase = np.angle(H)
        
        # Apply penalty: encourage flat spectrum by penalizing |H|^4
        penalty_grad = -4 * lambda_penalty * H_mag**3
        
        # Adjust spectrum in frequency domain
        H_adjusted = H_mag * np.exp(1j * H_phase) + alpha * penalty_grad * np.exp(1j * H_phase)
        
        # Ensure real signal (Hermitian symmetry)
        H_adjusted = H_adjusted.copy()
        if n_points % 2 == 0:
            H_adjusted[0] = np.real(H_adjusted[0])  # DC component
            H_adjusted[n_points//2] = np.real(H_adjusted[n_points//2])  # Nyquist
            for i in range(1, n_points//2):
                H_adjusted[n_points-i] = np.conj(H_adjusted[i])
        else:
            H_adjusted[0] = np.real(H_adjusted[0])  # DC component  
            for i in range(1, (n_points+1)//2):
                H_adjusted[n_points-i] = np.conj(H_adjusted[i])
        
        # Convert back to time domain
        h_new = np.real(ifft(H_adjusted))
        
        # Project to feasible domain
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update best solution
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iteration
        
        # Adaptive learning rate
        if iteration > 0 and (iteration % 50 == 0):
            if c5_new < c5_best:
                alpha *= 1.1
            else:
                alpha *= 0.9
            
        # Rule out stagnation
        if iteration - last_improvement > patience:
            break
        
        # Update current solution
        c5_best = c5_new
        h = h_new.copy()
        H = fft(h)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration  
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Optimization
        h_opt, c5_opt = optimize_fourier_spectrum(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
    
    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_fourier_spectrum(h_start, n_points, dx, budget_s)
        best_n_points = n_points
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / best_n_points)
    
    return (list(best_h), final_c5, best_n_points)
