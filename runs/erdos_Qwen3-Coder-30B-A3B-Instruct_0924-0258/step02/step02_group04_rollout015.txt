import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a low-correlation binary-like pattern
        h = np.zeros(n_points)
        ones_count = n_points // 2
        for i in range(ones_count):
            pos = (i * n_points // ones_count) % n_points
            h[pos] = 1.0
        # Add some random perturbation to make it more interesting
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts
        h_restart = h_start.copy()
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Optimization with L-BFGS-B
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            # Constraints
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, h_restart, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except:
                pass
                
            # Perturb for next restart
            noise = np.random.normal(0, 0.05, n_points)
            h_restart = project_to_feasible(h_restart + noise, n_points)

    # Final refinement at best resolution
    if best_h is not None:
        dx = 2.0 / best_n_points
        h_opt, c5_opt = minimize(evaluate_erdos_solution, best_h, 
                               method='L-BFGS-B', 
                               bounds=[(0,1) for _ in range(best_n_points)],
                               args=(dx,), 
                               options={'maxiter': 50, 'ftol': 1e-8})
        h_opt = project_to_feasible(h_opt, best_n_points)
        c5_opt = evaluate_erdos_solution(h_opt, dx)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
