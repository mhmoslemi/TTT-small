import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    tolerance = 1e-5
    low = 0.0
    high = 0.5
    
    # Use a fallback solver
    solver = 'highs'  # or 'glpk', 'scipy'
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Binary search
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        C5_candidate = (low + high) / 2
        # Optimization: only solve LP if we're checking feasibility
        # With a simple warm start approach to creating constraints
        # Build constraint matrix for the LP
        # We'll use a simple approach to formulate constraints correctly
        # Initialize constraint matrix
        A_eq = np.zeros((1, n_points))
        A_eq[0] = np.ones(n_points)
        b_eq = np.array([target_sum])
        
        # Collect inequality constraints
        A_ub_list = []
        b_ub_list = []
        
        # For k=-n_points+1 to n_points-1
        # Compute required constraints
        for k in range(-n_points + 1, n_points):
            if time.time() - start_time > budget_s - 1:
                break
            # This is the sum over i : h[i] * (1 - h[i+k]) <= C5_candidate * dx^{-1}
            # Use full convolution, but from the middle position
            A_ub_row = np.zeros(n_points)
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    A_ub_row[i] += 1.0
            A_ub_list.append(A_ub_row)
            b_ub_list.append(C5_candidate / dx)
        
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        # We also need bounds: 0 <= h[i] <= 1
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Since linprog expects to minimize, but we only check feasibility, use zero cost
        c = np.zeros(n_points)
        
        try:
            result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                            bounds=bounds, method=solver, options={'presolve': True})
            if result.success:
                # Our solution is feasible, tighten the bound
                high = C5_candidate
                # If we have a better solution, update best
                if compute_overlap(result.x) < best_C5:
                    best_h = result.x.copy()
                    best_C5 = compute_overlap(result.x)
            else:
                # Not feasible, so we need to increase bound
                low = C5_candidate
        except:
            low = C5_candidate

    if compute_overlap(best_h) < best_C5:
        best_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
