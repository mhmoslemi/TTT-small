import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    dx = 2.0 / n_points
    h = project_to_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Fourier-based optimization
    iter_count = 0
    stagnation = 0
    last_improvement = 0

    freqs = np.fft.fftfreq(n_points, dx)

        iter_count += 1
        
        # Convert h to frequency domain
        H = np.fft.fft(h)
        H_magnitude = np.abs(H)
        
        # Compute power spectrum (|H|^2)
        power_spectrum = H_magnitude ** 2
        
        # Objective: penalize high power in certain frequencies
        # Create a penalty based on power spectrum smoothing
        penalty = np.sum(power_spectrum ** 2)
        
        # Compute gradient of penalty w.r.t. H (simplified gradient)
        grad_power = 2 * H_magnitude * H
        grad_penalty = 2 * grad_power
        
        # Update H: attempt to make spectrum flatter (reduce autocorrelation peaks)
        step_size = 1e-4 * (1.0 + 0.1 * np.random.random())
        H_new = H - step_size * grad_penalty

            best_c5 = c5_new
            stagnation = 0
            last_improvement = iter_count

        # Occasionally reinitialize with perturbed best solution
        if stagnation > stagnation_limit and iter_count - last_improvement > 300:
            noise_scale = 0.1
            h_temp = h + np.random.normal(0, noise_scale, n_points)
            h_temp = project_to_feasible(h_temp, n_points)
            c5_temp = evaluate_erdos_solution(h_temp, dx)

                best_c5 = c5_temp
