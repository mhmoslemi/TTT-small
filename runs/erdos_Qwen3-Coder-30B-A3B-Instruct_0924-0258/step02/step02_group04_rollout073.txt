import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Default initialization:
        # Start with a smooth, balanced step function
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def optimize_with_fourier(h_start, n_points, dx, time_limit):
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Fourier optimization using gradient descent on spectrum
    max_iter = 1000
    step_size = 1e-4
    penalty_weight = 0.1
    
    # Discrete Fourier Transform
    def dft(x):
        return np.fft.fft(x)
    
    def idft(X):
        return np.fft.ifft(X).real
    
    # Initialize with starting h
    h_current = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Get spectrum
        H = dft(h_current)
        H_conj = np.conj(H)
        
        # Estimate autocorrelation in frequency domain
        # (This is a heuristic: we approximate autocorrelation by inverse transform of |H|^2)
        power_spectrum = np.abs(H)**2
        # Apply penalty for peakiness
        penalty_gradient = 4 * np.abs(H)**2 * H_conj
        
        # Update spectrum in frequency domain
        H_new = H - step_size * penalty_weight * penalty_gradient
        # Ensure real signal property
        H_new = (H_new + np.conj(H_new[::-1])) / 2  # Symmetrize
        
        # Inverse transform
        h_new = idft(H_new)
        
        # Fix feasibility
        h_new = project_to_feasible(h_new, n_points)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            
        h_current = h_new
        
        # Adaptive step size
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.95
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Multi-resolution approach
    resolutions = [15, 30, 45, 60]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.99:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), budget_s * 0.4)
        
        h_opt, c5_opt = optimize_with_fourier(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
            
    # Final refinement on best solution
    if best_h is not None:
        dx = 2.0 / best_n_points
        final_h, final_c5 = optimize_with_fourier(best_h, best_n_points, dx, 20)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_with_fourier(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
