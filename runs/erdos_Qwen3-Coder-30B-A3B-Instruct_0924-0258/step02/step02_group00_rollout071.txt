import numpy as np
from scipy.signal import convolve
from time import time

def evaluate_erdos_solution(h, dx):
    corr = convolve(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Use initial_h_values if available

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    inertia_weight = w_start
    
    # Multi-resolution optimization: coarse to fine
    resolutions = [25, 30, 35]
    
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_scores = []
        
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.1, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            score = evaluate_erdos_solution(h, dx_try)
            pbest.append(h.copy())
            pbest_scores.append(score)
        
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        # PSO loop
        for iter_idx in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            inertia_weight = w_start - (w_start - w_end) * (iter_idx / max_iter)
            
            for p in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[p] = (inertia_weight * velocities[p] + 
                                c1 * r1 * (pbest[p] - particles[p]) + 
                                c2 * r2 * (gbest - particles[p]))
                
                # Update position
                particles[p] += velocities[p]
                particles[p] = project_to_feasible(particles[p], n_points)
                
                # Evaluate
                score = evaluate_erdos_solution(particles[p], dx_try)
                
                # Update personal best
                if score < pbest_scores[p]:
                    pbest[p] = particles[p].copy()
                    pbest_scores[p] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[p].copy()
                    gbest_score = score
        
        # Update global best solution
        if gbest_score < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_score
            best_n_points = n_points
        
        h_start = best_h.copy()
    
    # Final refinement with gradient descent
    def grad_descent_step(h, dx, learning_rate=0.001, steps=100):
        for _ in range(steps):
            if time() - start_time > budget_s * 0.95:
                break
            # Gradient approximation using finite differences
            grad = np.zeros_like(h)
            eps = 1e-6
            base_score = evaluate_erdos_solution(h, dx)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, len(h))
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (score_plus - base_score) / eps
            
            # Update
            h -= learning_rate * grad
            h = project_to_feasible(h, len(h))
            
        return h
    
    refined_h = grad_descent_step(best_h.copy(), dx)
    refined_c5 = evaluate_erdos_solution(refined_h, dx)
    
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5
    
    return (list(best_h), best_c5, best_n_points)
