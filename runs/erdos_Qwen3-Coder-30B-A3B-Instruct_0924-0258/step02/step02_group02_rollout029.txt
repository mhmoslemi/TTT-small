import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_autocorrelation(h_vec):
        # Compute all autocorrelations
        conv = np.correlate(h_vec, h_vec, mode='full')
        shifts = np.arange(-len(h_vec) + 1, len(h_vec))
        return conv, shifts

    def get_min_shift_autocorr(h_vec):
        # Find shift that gives minimum autocorrelation with 1-h
        corr_1mh = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr_1mh *= dx
        min_idx = np.argmin(corr_1mh)
        return min_idx, corr_1mh[min_idx]

    def compute_gradient(h_vec, min_shift_idx):
        # Compute gradient of the minimum autocorrelation w.r.t h
        n = len(h_vec)
        grad = np.zeros(n)
        # We need to compute the gradient for the term h[i] * h[i+shift] 
        # or h[i] * h[i-shift] in the autocorrelation sum
        # for the shift that gives the minimum value
        shift = min_shift_idx - (n - 1)
        # For shift > 0, we add h[i+shift] to grad[i]
        # For shift < 0, we add h[i-shift] to grad[i]
        if shift > 0:
            grad[shift:] += h_vec[:-shift]
        elif shift < 0:
            grad[:shift] += h_vec[-shift:]
        else:
            # shift == 0, gradient is 1 - h[i] for 1-h[i] term
            pass
        return grad

    # Early termination counter
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    iteration = 0
    no_improvement_count = 0
    learning_rate = 0.01
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Find shift with minimum autocorrelation
        min_shift_idx, min_autocorr = get_min_shift_autocorr(h)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, min_shift_idx)
        
        # Update h using gradient ascent (we want to maximize min_autocorr)
        h = h + learning_rate * grad
        
        # Project h back to constraints
        h = project_h(h)
        
        # Check improvement
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                # Reduce learning rate if no improvement
                learning_rate *= 0.9
                no_improvement_count = 0
        
        # Add some noise occasionally to escape local minima
        if iteration % 10 == 0 and no_improvement_count > 5:
            noise = 0.0001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
            
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
