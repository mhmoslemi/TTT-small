import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    # Alternating optimization approach
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Initialize learning rate and other parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 0
    patience_limit = 100
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:  # Leave some buffer time
            break
            
        # Compute current overlaps
        overlaps = []
        for k in range(n_points):  # For each shift k
            # Compute the overlap for shift k
            overlap = 0
            for i in range(n_points - k):
                overlap += h[i] * (1 - h[i + k]) * dx
            overlaps.append(overlap)
        
        max_overlap = max(overlaps)
        
        # Update h to reduce max overlap
        # This is approximate gradient descent approach
        # We compute gradient for the max overlap term
        max_idx = np.argmax(overlaps)
        grad_h = np.zeros(n_points)
        for i in range(n_points - max_idx):
            grad_h[i] += (1 - h[i + max_idx]) * dx
            grad_h[i + max_idx] -= h[i] * dx
        
        # Update h
        h -= learning_rate * grad_h
        h = project_h(h)
        
        # adaptive learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.1
        
        # Track best solution
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                # Restart with a small noise if no improvement
                h = project_h(h + 0.001 * np.random.randn(n_points))
                patience = 0
    
    # Final validation of best result
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
