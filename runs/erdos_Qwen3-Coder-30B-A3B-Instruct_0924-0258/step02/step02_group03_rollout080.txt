import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    population = []
    sigmas = []
    for i in range(pop_size):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
        sigmas.append(np.full(n_points, sigma_init))

        generation += 1
        
        # Evaluate all individuals
        fitness = []

            fitness.append(f)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        sigmas = [sigmas[i] for i in sorted_indices]
        
        # Update best if improved
        if fitness[0] < best_c5:

        else:

        new_pop = population[:elite_size]
        new_sigmas = sigmas[:elite_size]

            parent_idx = np.random.choice(pop_size, p=rank_probs)

            parent_h = population[parent_idx].copy()
            parent_sigma = sigmas[parent_idx].copy()

            noise = np.random.normal(0, 1, n_points)
