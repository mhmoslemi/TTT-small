import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents"""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(candidate, mutation_rate=0.2, noise_std=0.1):
    """Apply mutation to a candidate"""
    if np.random.random() > mutation_rate:
        return candidate.copy()
    noise = np.random.normal(0, noise_std, len(candidate))
    mutated = candidate + noise
    return mutated

    dx = 2.0 / n_points
    
    # Generate initial population
    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic algorithm
    tournament_size = 5
    mutation_rate = 0.2
    crossover_rate = 0.8
    elite_size = population_size // 4
    
    while time() - start_time < budget_s * 0.98:
        # Selection - Tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and Mutation
        new_population = []
        
        # Keep elite
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
            child1, child2 = crossover(selected[parent1_idx], selected[parent2_idx], crossover_rate)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Evaluation
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
