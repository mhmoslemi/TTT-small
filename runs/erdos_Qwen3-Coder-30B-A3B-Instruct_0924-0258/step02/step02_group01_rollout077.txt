import numpy as np
from time import time

def project_feasible(candidate_h, n_points):
    """Project candidate_h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover for two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutation(individual, mutation_rate=0.2, mutation_strength=0.1):
    """Apply Gaussian mutation to an individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, mutation_strength, len(individual))
        mutated = individual + noise
        return mutated
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    time_limit = budget_s
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Initialize population
    pop_size = 50
    population = []
    fitness = []
    
    # Add initial solution
    h_feasible = project_feasible(h_start, n_points)
    population.append(h_feasible)
    fitness.append(evaluate_erdos_solution(h_feasible, 2.0 / n_points))
    
    # Add random individuals
    for _ in range(pop_size - 1):
        h_random = np.random.rand(n_points)
        h_feasible = project_feasible(h_random, n_points)
        population.append(h_feasible)
        fitness.append(evaluate_erdos_solution(h_feasible, 2.0 / n_points))
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    gen = 0
    stagnation_count = 0
    stagnation_limit = 20
    
    while time() - start_time < time_limit * 0.95:
        gen += 1
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Keep top 25%
        elite_count = max(1, pop_size // 4)
        new_population = sorted_pop[:elite_count]
        new_fitness = sorted_fit[:elite_count]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.random() < 0.8:  # crossover_rate
                offspring1, offspring2 = blend_crossover(parent1, parent2, alpha=0.5)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutation(offspring1)
            offspring2 = mutation(offspring2)
            
            # Feasible projection
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            # Evaluate
            new_fitness.append(evaluate_erdos_solution(offspring1, 2.0 / n_points))
            new_population.append(offspring1)
            
            if len(new_population) < pop_size:
                new_fitness.append(evaluate_erdos_solution(offspring2, 2.0 / n_points))
                new_population.append(offspring2)
        
        # Replace population
        population = new_population[:pop_size]
        fitness = new_fitness[:pop_size]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early termination if stagnation
        if stagnation_count >= stagnation_limit:
            break
    
    return (list(best_h), best_c5, n_points)
