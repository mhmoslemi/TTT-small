import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize m particles using simplex method"""
    particles = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        # Default to constant function
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_to_feasible(h0, n_points)
    particles.append(h0.copy())
    
    # Generate rest by making random small perturbations
    for i in range(1, min(m, n_points + 1)):
        h = h0.copy()
        # Perturb a few random points
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for idx in indices:
            h[idx] += np.random.normal(0, 0.1)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    n_points = 30  # Standard resolution
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    w = 0.7
    c1 = c2 = 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    max_shake_attempts = 10
    
    # Initialize particles
    particles = initialize_simplex(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    
    # Evaluate initial particles
    fitness = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Track personal bests
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    
    # Global best
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # Track stagnation
    last_improvement = 0
    global_best_history = []
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(len(particles)):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            new_fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
            
            # Update global best
            if new_fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = new_fitness
                last_improvement = iteration
                global_best_history.append((iteration, gbest_fitness))
        
        # Check for stagnation and apply shaking
        if iteration - last_improvement > shaking_threshold:
            # Apply shaking to particles
            for i in range(len(particles)):
                # Add noise to escape local minima
                noise_magnitude = 0.1 + (0.1 * (iteration - last_improvement) / shaking_threshold)
                noise = np.random.normal(0, noise_magnitude, n_points)
                particles[i] += noise
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Re-evaluate
                new_fitness = evaluate_erdos_solution(particles[i], dx)
                if new_fitness < pbest_fitness[i]:
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = new_fitness
                    
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
                    last_improvement = iteration
                    global_best_history.append((iteration, gbest_fitness))
            
            # Reset stagnation counter
            last_improvement = iteration
        
        # Apply L-BFGS-B refinement periodically
        if iteration % lbfgs_interval == 0 and iteration > 0 and len(global_best_history) > 0:
            # Select top particles for refinement
            top_indices = np.argsort(pbest_fitness)[:min(10, len(pbest_fitness))]
            for idx in top_indices:
                # Define objective for refinement
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                # Set bounds (0 to 1 for each element)
                bounds = [(0, 1) for _ in range(n_points)]
                
                # L-BFGS-B optimization
                try:
                    res = minimize(obj, pbest[idx], method='L-BFGS-B', 
                                  bounds=bounds, options={'maxiter': 20, 'ftol': 1e-8})
                    if res.success:
                        refined_h = project_to_feasible(res.x, n_points)
                        refined_fitness = evaluate_erdos_solution(refined_h, dx)
                        if refined_fitness < pbest_fitness[idx]:
                            pbest[idx] = refined_h.copy()
                            pbest_fitness[idx] = refined_fitness
                            
                        if refined_fitness < gbest_fitness:
                            gbest = refined_h.copy()
                            gbest_fitness = refined_fitness
                            last_improvement = iteration
                            global_best_history.append((iteration, gbest_fitness))
                except:
                    pass
    
    # Final refinement of the global best
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', 
                      bounds=bounds, options={'maxiter': 50, 'ftol': 1e-10})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_fitness = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_fitness, n_points)
