import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    # Initialize parameters
    pop_size = 25
    F = 0.7
    CR = 0.5
    max_generations = 500
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_mut = h + np.random.normal(0, 0.05, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * target_sum / np.sum(h_mut)
        population.append(h_mut)
    
    # Helper function to compute overlap
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # Constraint projection
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    best_h = None
    best_val = float('inf')
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for i, indiv in enumerate(population):
            val = compute_overlap(indiv)
            if val < best_val:
                best_val = val
                best_h = indiv.copy()
            fitness.append(val)
        
        # Create new population
        new_population = []
        for i in range(pop_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Differential evolution
            a, b, c = np.random.choice(pop_size, 3, replace=False)
            trial = population[a] + F * (population[b] - population[c])
            
            # Crossover
            for j in range(n_points):
                if np.random.random() > CR:
                    trial[j] = population[i][j]
                    
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Selection
            if compute_overlap(trial) < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = new_population
    
    # Final validation
    if best_h is not None:
        final_C5 = compute_overlap(best_h)
        return (best_h.tolist(), final_C5, n_points)
    else:
        # Fall back to initial solution if no better solution found
        final_C5 = compute_overlap(h)
        return (h.tolist(), final_C5, n_points)
