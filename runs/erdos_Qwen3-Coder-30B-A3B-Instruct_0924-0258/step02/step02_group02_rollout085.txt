import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)
    
    # Start optimization
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Optimization with simulated annealing based approach
    current_h = h.copy()
    temp = 1.0
    min_temp = 1e-8
    cooling_rate = 0.999
    
    iteration = 0
    last_improvement = 0
    patience = 500
    
    while time.time() - start_time < budget_s:
        # Generate neighbor solution
        neighbor_h = current_h.copy()
        
        # Small random perturbation
        perturbation_magnitude = 0.01 * temp
        noise = np.random.normal(0, perturbation_magnitude, n_points)
        neighbor_h += noise
        
        # Project to feasible region
        neighbor_h = project_h(neighbor_h)
        
        # Evaluate neighbor
        neighbor_c5 = compute_c5(neighbor_h)
        current_c5 = compute_c5(current_h)
        
        # Accept or reject based on simulated annealing
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            if neighbor_c5 < best_c5:
                best_c5 = neighbor_c5
                best_h = neighbor_h.copy()
                last_improvement = iteration
        else:
            # Accept with probability based on temperature and difference
            delta = neighbor_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temp):
                current_h = neighbor_h.copy()
        
        # Adaptive cooling
        temp = max(min_temp, temp * cooling_rate)
        
        # Early stopping if no improvement for a while
        if iteration - last_improvement > patience:
            break
            
        iteration += 1
    
    # Final projection and evaluation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
