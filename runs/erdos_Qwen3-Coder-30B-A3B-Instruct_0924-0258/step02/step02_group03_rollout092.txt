import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5

    # Initialize swarm
    n_particles = 100
    particles = [h0.copy()]
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h)
    
    # Parameters
    tau = 1.0
    sigma = 0.1
    accept_count = 0
    accept_total = 0
    last_improvement = 0
    best_h = None
    best_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 1:
                break

        # Importance sampling with proposal
        weights = []
        new_particles = []
        for h in particles:
            # Generate proposal
            proposal = h + np.random.normal(0, sigma, n_points)
            proposal = project_to_feasible(proposal, n_points/2.0)
            c5_proposal = evaluate(proposal, 2.0/n_points)
            
            # Compute importance weight
            weight = np.exp(-c5_proposal / tau)
            weights.append(weight)
            new_particles.append(proposal)
        
        # Normalize weights
        weights = np.array(weights)
        weights = weights / (np.sum(weights) + 1e-10)
        
        # Resampling using stratified sampling
        cumulative_weights = np.cumsum(weights)
        resampled_particles = []
        for _ in range(n_particles // 2):
            r = np.random.random()
            idx = np.searchsorted(cumulative_weights, r)
            resampled_particles.append(new_particles[idx].copy())
        
        # Add random particles for diversity
        for _ in range(n_particles // 2):
            h = h0 + np.random.normal(0, 0.1, n_points)
            h = project_to_feasible(h, n_points/2.0)
            resampled_particles.append(h)
        
        particles = resampled_particles
        
        # Mutation strategy
        mutated_particles = []
        for h in particles[:n_particles//2]:  # Only mutate top 50%
            # Global exploration
            h_mut1 = h + np.random.normal(0, sigma * 2, n_points)
            h_mut1 = project_to_feasible(h_mut1, n_points/2.0)
            mutated_particles.append(h_mut1)
            
            # Local exploitation
            h_mut2 = h + np.random.normal(0, 0.05, n_points)
            h_mut2 = project_to_feasible(h_mut2, n_points/2.0)
            mutated_particles.append(h_mut2)
            
        particles.extend(mutated_particles)
        
        # Update best solution
        for h in particles:
            c5 = evaluate(h, 2.0/n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
        
        # Adaptive temperature and noise
        tau *= 0.99
        if iteration % 5 == 0:
            if accept_count > 0:
                accept_rate = accept_count / max(accept_total, 1)
                if accept_rate < 0.25:
                    sigma *= 0.9
                elif accept_rate > 0.35:
                    sigma *= 1.1
            accept_count = 0
            accept_total = 0
        
        # Early termination if no improvement for too long
        if iteration - last_improvement > 50:
            break
            
    if best_h is not None:
        final_c5 = evaluate(best_h, 2.0/n_points)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h0, n_points/2.0)
        final_c5 = evaluate(final_h, 2.0/n_points)
        return (final_h.tolist(), float(final_c5), n_points)
