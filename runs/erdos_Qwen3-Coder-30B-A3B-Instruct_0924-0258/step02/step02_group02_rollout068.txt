from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):

    h = h * target_sum / np.sum(h)

        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initialize problem variables
    h_vars = Variable(n_points)
    eta = Variable()
    
    # Set initial constraints
    prob_constraints = [0 <= h_vars, h_vars <= 1, sum(h_vars) == target_sum]
    
    # Minimize eta with initial constraints
    objective = Minimize(eta)
    prob = Problem(objective, prob_constraints)
    
    start_time = time.time()
    best_h = h.copy()

    max_iterations = 1000
    iteration = 0

    initial_c5 = compute_c5(h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = h.copy()

            break

            corr_sum = 0.0
            for i in range(n_points - shift_idx):
                corr_sum += best_h[i] * (1 - best_h[i + shift_idx])
            corr_sum *= dx

                    new_h = project_h(new_h)
