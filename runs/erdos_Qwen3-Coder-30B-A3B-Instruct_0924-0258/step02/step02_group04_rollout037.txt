import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m=10):
    """Initialize a simplex of feasible points"""
    simplex = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        if len(h_start) != n_points:
            h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
    else:
        # Simple initial pattern
        h_start = np.ones(n_points) * 0.5
    
    h_start = project_to_feasible(h_start, n_points)
    simplex.append(h_start.copy())
    
    # Generate additional points by perturbing the initial point
    for i in range(1, min(m, n_points)):
        h_perturbed = h_start.copy()
        # Perturb a few random indices
        indices = np.random.choice(n_points, size=min(3, n_points//3), replace=False)
        for idx in indices:
            h_perturbed[idx] += np.random.normal(0, 0.1)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        simplex.append(h_perturbed)
    
    # Fill to m points with random feasible points
    while len(simplex) < m:
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        simplex.append(h)
    
    return np.array(simplex)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    max_iterations = 2000
    shaking_threshold = 50
    gradient_refinement_interval = 200
    min_swarm_size = 10
    max_swarm_size = 50
    
    # Initialize simplex
    particles = initialize_simplex(n_points, min_swarm_size)
    n_particles = len(particles)
    
    # Track best
    best_h = None
    best_c5 = float('inf')
    
    # Personal bests and velocities
    pbest = particles.copy()
    pbest_c5 = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Velocities
    velocities = np.zeros_like(particles)
    
    # Stagnation tracking
    stagnation_count = 0
    last_improvement = 0
    
    # PSO loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_weight * r1 * (pbest[i] - particles[i]) +
                social_weight * r2 * (gbest - particles[i])
            )
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
                    if iteration - last_improvement > shaking_threshold:
                        last_improvement = iteration
                        
        # Periodic gradient refinement
        if iteration % gradient_refinement_interval == 0 and iteration > 0:
            # Select top particles
            top_indices = np.argsort(pbest_c5)[:min(10, n_particles)]
            for idx in top_indices:
                h_init = pbest[idx].copy()
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                # Apply local optimization
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
                
                try:
                    res = minimize(obj, h_init, method='L-BFGS-B', 
                                 bounds=bounds, constraints=cons, 
                                 options={'maxiter': 100})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < gbest_c5:
                            gbest = h_refined
                            gbest_c5 = c5_refined
                except:
                    pass
        
        # Stagnation check and adaptation
        if iteration - last_improvement > shaking_threshold:
            # Apply shaking
            for i in range(n_particles):
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
                
            # Increase swarm size if needed
            if n_particles < max_swarm_size:
                n_particles += int(0.5 * n_particles)  # 50% growth
                n_particles = min(n_particles, max_swarm_size)
                # Extend particles array
                extra_particles = initialize_simplex(n_points, n_particles - len(particles))
                particles = np.vstack([particles, extra_particles])
                velocities = np.vstack([velocities, np.zeros((n_particles - len(velocities), n_points))])
                pbest = np.vstack([pbest, extra_particles])
                pbest_c5 = np.concatenate([pbest_c5, np.array([evaluate_erdos_solution(p, dx) for p in extra_particles])])
                
    # Final refinement on global best
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', 
                     bounds=bounds, constraints=cons, 
                     options={'maxiter': 500})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_c5 = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_c5, n_points)
