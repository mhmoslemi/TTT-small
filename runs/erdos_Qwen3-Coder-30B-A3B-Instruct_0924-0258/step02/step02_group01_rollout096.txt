import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project candidate_h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    start_time = time()

    population_size = 50

    # Initialize population
    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fit = evaluate_erdos_solution(ind, dx)
        fitness.append(fit)
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Evolve
    num_generations = int(budget_s * 10)  # rough estimate
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
                child1 = project_feasible(child1, n_points)
            
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
                child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Keep only population_size individuals
        new_population = new_population[:population_size]
        
        # Evaluate new population
        new_fitness = []
        for ind in new_population:
            fit = evaluate_erdos_solution(ind, dx)
            new_fitness.append(fit)
            
            if fit < best_c5:
                best_h = ind.copy()
                best_c5 = fit
                
        population = new_population
        fitness = new_fitness
    
    return (list(best_h), best_c5, n_points)
