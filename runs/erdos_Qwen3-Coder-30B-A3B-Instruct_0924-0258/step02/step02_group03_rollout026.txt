import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # PSO parameters
    swarm_size = 40
    max_iter = 10000
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    v_max = 0.05  # velocity max as % of range
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        # Start near initial solution
        h = h0 + np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        pbest_positions.append(h.copy())
        pbest_scores.append(evaluate(h, 2.0/n_points))
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    # PSO main loop
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            
            cognitive = c1 * r1 * (pbest_positions[i] - h)
            social = c2 * r2 * (gbest_position - h)
            v = w * v + cognitive + social
            
            # Clip velocity
            v_max_abs = v_max * (np.max(h) - np.min(h))
            v = np.clip(v, -v_max_abs, v_max_abs)
            
            # Update position
            h_new = h + v
            h_new = project_to_feasible(h_new, n_points/2.0)
            
            # Evaluate
            new_score = evaluate(h_new, 2.0/n_points)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbest_positions[i] = h_new.copy()
                pbest_scores[i] = new_score
            
            # Update global best
            if new_score < gbest_score:
                gbest_position = h_new.copy()
                gbest_score = new_score
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        iteration += 1
    
    return (gbest_position.tolist(), float(gbest_score), n_points)
