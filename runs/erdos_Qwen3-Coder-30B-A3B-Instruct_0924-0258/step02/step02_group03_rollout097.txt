import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    # Initialize
    h_current = project_to_feasible(h0.copy(), n_points / 2.0)
    current_c5 = evaluate(h_current, 2.0 / n_points)
    
    best_h = h_current.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    max_iter = 100000
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 1000
    accept_count = 0
    total_accept = 0
    
    # Acceptance rate tracking
    recent_accept = []
    recent_threshold = 50
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Generate candidate by perturbing current solution
        candidate = h_current + np.random.normal(0, T, n_points)
        
        # Project to feasible region
        candidate = project_to_feasible(candidate, n_points / 2.0)
        
        # Evaluate candidate
        candidate_c5 = evaluate(candidate, 2.0 / n_points)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h_current = candidate
            current_c5 = candidate_c5
            accept_count += 1
            total_accept += 1
        else:
            recent_accept.append(0)
        
        recent_accept.append(1)
        if len(recent_accept) > recent_threshold:
            recent_accept.pop(0)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Gradual cooling
        T = max(T * alpha, 1e-6)
        
        # Adaptive cooling based on acceptance rate
        if iter_count % recent_threshold == 0:
            accept_rate = sum(recent_accept) / len(recent_accept)
            if accept_rate < 0.2:
                # Too few accepts, increase temperature
                T = min(T * 1.05, 1.0)
                recent_accept = []
            elif accept_rate > 0.5:
                # Too many accepts, cool more aggressively
                T = max(T * 0.995, 1e-6)
                recent_accept = []
        
        # Reset temperature if stuck
        if no_improve_count > max_no_improve:
            T = min(T * 1.5, 1.0)
            no_improve_count = 0
        
        # Occasionally add more randomness to escape local minima
        if np.random.random() < 0.001:
            T = min(T * 1.2, 1.0)
    
    return (best_h.tolist(), float(best_c5), n_points)
