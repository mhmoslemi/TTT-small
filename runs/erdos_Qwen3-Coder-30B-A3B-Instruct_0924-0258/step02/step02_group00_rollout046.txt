import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    omega_start = 0.7
    omega_end = 0.4
    c1, c2 = 2.0, 2.0
    inertia_decrease = (omega_start - omega_end) / max_iter
    
    # Multi-resolution search
    resolutions = [25, 30, 35]
    
    # Time tracking
    start_time = time()
    
    # Initialize best solution
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
        dx = 2.0 / n_points
        best_h = h_current.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points
    else:
        n_points = 30
        dx = 2.0 / n_points
        h_current = np.ones(n_points) * 0.5
        best_h = h_current.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points

    # Begin multi-resolution search
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate from previous resolution if needed
        if res_idx > 0:
            prev_h = best_h
            prev_n = best_n_points
            x_old = np.linspace(0, 2, prev_n)
            x_new = np.linspace(0, 2, n_points)
            h_current = np.interp(x_new, x_old, prev_h)
            h_current = project_to_feasible(h_current, n_points)
            
        # Initialize particles
        particles = []
        velocities = []
        pbest_h = []
        pbest_c5 = []
        
        # Create initial population around current best
        for i in range(n_particles):
            # Start with a version of current best with noise
            h = h_current + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points)
            
            # Clamp velocities
            v = np.random.uniform(-0.1, 0.1, n_points)
            
            particles.append(h)
            velocities.append(v)
            pbest_h.append(h.copy())
            
            c5 = evaluate_erdos_solution(h, 2.0/n_points)
            pbest_c5.append(c5)
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                best_n_points = n_points
                
        # PSO loop
        omega = omega_start
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        for iter_count in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update particles
            gbest_idx = np.argmin(pbest_c5)
            gbest_h = pbest_h[gbest_idx]
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (pbest_h[i] - particles[i]) + 
                                c2 * r2 * (gbest_h - particles[i]))
                
                # Update position
                particles[i] = particles[i] + velocities[i]
                
                # Project to feasible region
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                c5 = evaluate_erdos_solution(particles[i], 2.0/n_points)
                
                # Update personal best
                if c5 < pbest_c5[i]:
                    pbest_c5[i] = c5
                    pbest_h[i] = particles[i].copy()
                
                # Update global best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particles[i].copy()
                    best_n_points = n_points
                    
            # Reduce inertia weight
            omega = max(omega_end, omega - inertia_decrease)
            
            # Early stopping if improvement is minimal
            if iter_count > 10 and abs(pbest_c5[gbest_idx] - best_c5) < 1e-6:
                break
            
        # Final evaluation of best solution
        best_c5 = evaluate_erdos_solution(best_h, 2.0/best_n_points)
    
    # Refinement using gradient-based optimization
    if 'initial_h_values' in globals():
        # Try to improve using optimization around the best found solution
        try:
            dx = 2.0 / len(best_h)
            # Perform a few optimization iterations
            for _ in range(50):
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def grad(h):
                    eps = 1e-5
                    g = np.zeros_like(h)
                    for i in range(len(h)):
                        h_plus = h.copy()
                        h_plus[i] += eps
                        h_plus = project_to_feasible(h_plus, len(h))
                        obj_plus = obj(h_plus)
                        obj_base = obj(h)
                        g[i] = (obj_plus - obj_base) / eps
                    return g
                
                res = minimize(obj, best_h, method='L-BFGS-B', 
                               bounds=[(0,1) for _ in range(len(best_h))], 
                               jac=grad, maxiter=10)
                if res.success:
                    best_h = project_to_feasible(res.x, len(best_h))
                    best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass

    return (list(best_h), best_c5, best_n_points)
