import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population using initial_h_values or create variations"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        h = project(base_h + np.random.normal(0, 0.1, n_points), n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Apply mutation to an individual"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Apply crossover between two individuals"""
    if np.random.rand() < 0.7:
        idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
        offspring = np.copy(h1)
        offspring[idx1:idx2] = h2[idx1:idx2]
        return project(offspring, len(offspring))
    else:
        return np.copy(h1)

def tournament_selection(population, fitness, k=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), k, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Settings
    n_points = 30
    pop_size = 50
    mutation_prob = 0.2
    sigma = 0.1
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Track best before optimization
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_c5 = fitness[best_idx]
        best_h = population[best_idx].copy()
    
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 200
    
    # Optimization loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(selected), 2):
            h1 = selected[i]
            h2 = selected[min(i+1, len(selected)-1)]
            child1 = crossover(h1, h2)
            child1 = mutate(child1, mutation_prob, sigma)
            offspring.append(child1)
            
            if len(offspring) < pop_size:
                child2 = crossover(h2, h1)
                child2 = mutate(child2, mutation_prob, sigma)
                offspring.append(child2)
        
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Replace worst individuals
        all_individuals = population + offspring
        all_fitness = fitness + offspring_fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(all_fitness)
        population = [all_individuals[i] for i in sorted_indices[:pop_size]]
        fitness = [all_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adapt mutation rate if stagnation
        if stagnation_count > 100:
            mutation_prob = min(0.5, mutation_prob * 1.1)
        
        # Reset if stagnation too high
        if stagnation_count >= max_stagnation:
            population = initialize_population(n_points, pop_size)
            fitness = [evaluate_erdos_solution(h, dx) for h in population]
            stagnant_count = 0
            mutation_prob = 0.2
            
        # Periodic check for termination
        if gen % 100 == 0:
            elapsed = time() - start_time
            if elapsed > budget_s * 0.9:
                break
    
    if best_h is None:
        # Fallback to simple initialization and evaluation
        h_start = initialize_population(n_points, 1)[0]
        best_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start
    
    return (list(best_h), best_c5, n_points)
