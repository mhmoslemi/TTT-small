import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with a low resolution for fast initial exploration
    n_points = 15
    dx = 2.0 / n_points
    
    # Initialize
    h = initialize_h(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    # Fourier optimization approach
    max_iter = 1000
    iter_count = 0
    
    # Gradient step size
    step_size = 1e-4
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Optimization with different perturbations
        h_new = h.copy()
        
        # Add small random noise to explore
        noise = np.random.normal(0, 0.01, n_points)
        h_new += noise
        h_new = project_to_feasible(h_new, n_points)
        
        # Try gradient-like step if possible
        try:
            # Estimate gradient using finite differences
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                f_plus = obj(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, n_points)
                f_minus = obj(h_minus)
                
                grad[i] = (f_plus - f_minus) / (2 * eps)
            
            # Apply gradient step
            h_new = h - step_size * grad
            h_new = project_to_feasible(h_new, n_points)
            
        except:
            pass
        
        # Accept or reject based on improvement
        new_c5 = obj(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
    
    # Medium resolution refinement
    if (time() - start_time) < budget_s * 0.7 and n_points < 30:
        n_points = 30

        while iter_count < 500 and (time() - start_time) < budget_s * 0.95:
            iter_count += 1
            h_new = h.copy()
            
            # Add noise for exploration
            noise = np.random.normal(0, 0.01, n_points)
            h_new += noise

            except:
