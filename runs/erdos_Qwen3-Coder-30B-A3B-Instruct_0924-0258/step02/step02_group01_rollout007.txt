import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def perturb_solution(h, noise_level=0.1):
    """Apply random perturbation to the function"""
    h_pert = h.copy()
    n = len(h)
    # Perturb about 20% of elements
    indices = np.random.choice(n, size=int(0.2 * n), replace=False)
    noise = np.random.normal(0, noise_level, len(indices))
    h_pert[indices] += noise
    return project_to_feasible(h_pert, len(h))

    populations = []

        h_rand = np.random.rand(n_points)
        h_rand = project_to_feasible(h_rand, n_points)
        populations.append(h_rand)
    
    # Optimization parameters
    max_iterations = 100

                return evaluate_erdos_solution(h_vec, dx)
