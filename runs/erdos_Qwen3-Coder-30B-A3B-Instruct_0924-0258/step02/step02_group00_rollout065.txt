import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_subgradient(h, n_points, max_shifts):
    """Compute subgradient based on current max overlap shifts"""
    # Correlation between h and 1-h for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    
    # Build the subgradient vector from the shifts that give max overlap
    subgrad = np.zeros(n_points)
    for k in max_shifts:
        # For shift k, contribution to gradient at index i is (1-h[i-k]) - h[i]
        # Handle wrapping
        for i in range(n_points):
            j = (i - k) % n_points
            subgrad[i] += (1 - h[j]) - h[i]
            
    return subgrad

def frank_wolfe_step(h, n_points, dx, max_shifts, gamma=0.1):
    """Perform one Frank-Wolfe step to minimize the maximum overlap"""
    g = compute_subgradient(h, n_points, max_shifts)
    
    # Solve linearized problem: min_{s in feasible} g^T s
    # subject to: sum s = n_points/2, 0 <= s <= 1
    # This is equivalent to selecting s that minimizes g^T s
    # under the constraint that it is in the feasible region

    # Initialize with s that minimizes g^T s subject to bounds
    s = np.zeros(n_points)
    # We want to minimize g^T s subject to sum s = n_points/2, 0<=s<=1
    # This can be done by taking s to be 1 where g is most negative, 
    # and 0 where g is most positive, then rescaling
    
    # Sort indices by g values (descending)
    idxs = np.argsort(g)[::-1]
    s_sum = 0.0
    for i in idxs:
        if s_sum < n_points/2:
            # Determine how much more we can add
            max_add = min(1.0, n_points/2 - s_sum)
            s[i] = max_add
            s_sum += max_add
        else:
            s[i] = 0.0
    
    # Ensure we have the right sum by scaling
    if s_sum > 0:
        s = s * (n_points / 2) / s_sum
    
    h_new = h + gamma * (s - h)
    h_new = project_to_feasible(h_new, n_points)
    return h_new

    # Run optimization at different resolutions
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        
        # Interpolate if needed
        if n_points != len(h):
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h)), 
                                h)
            h = project_to_feasible(h_interp, n_points)
        
        dx = 2.0 / n_points
        
        # Perform a series of Frank-Wolfe steps at this resolution
        current_c5 = evaluate_erdos_solution(h, dx)
