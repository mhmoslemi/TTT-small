import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization with multiple restarts
    population_size = 50
    max_iter = 100
    patience = 20
    
    # Generate population
    population = [project_to_feasible(np.random.dirichlet(np.ones(n_points)) * (n_points / 2), n_points) 
                  for _ in range(population_size)]
    population.append(best_h)
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        improved = False
        for i in range(len(population)):
            if time() - start_time > budget_s * 0.95:
                break
                
            h = population[i].copy()
            
            # Apply perturbation
            perturb_mask = np.random.random(n_points) < 0.2
            noise = np.random.normal(0, 0.1, n_points)
            h_perturbed = h + noise * perturb_mask
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        improved = True
                    if c5_opt < evaluate_erdos_solution(population[i], dx):
                        population[i] = h_opt.copy()
            except:
                pass
        
        if not improved:
            patience -= 1
            if patience <= 0:
                break
    
    # Resolution refinement
    current_h = best_h.copy()
    current_n = n_points
    current_c5 = best_c5
    
    while time() - start_time < budget_s * 0.98:
        new_n = min(current_n * 2, 1000)
        if new_n <= current_n:
            break
            
        x_old = np.linspace(0, 2, current_n)
        x_new = np.linspace(0, 2, new_n)
        h_interp = np.interp(x_new, x_old, current_h)
        h_interp = project_to_feasible(h_interp, new_n)
        
        dx_new = 2.0 / new_n
        
        # Local optimization at higher resolution
        def obj(h):
            return evaluate_erdos_solution(h, dx_new)
            
        bounds = [(0, 1) for _ in range(new_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - new_n / 2}]
        
        try:
            res = minimize(obj, h_interp, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, new_n)
                c5_opt = evaluate_erdos_solution(h_opt, dx_new)
                if c5_opt < current_c5:
                    current_h = h_opt.copy()
                    current_c5 = c5_opt
                    current_n = new_n
                else:
                    break
            else:
                break
        except:
            break
            
    return (list(current_h), current_c5, current_n)
