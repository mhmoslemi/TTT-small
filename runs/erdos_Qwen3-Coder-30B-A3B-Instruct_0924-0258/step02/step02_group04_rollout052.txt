import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use a different starting approach: piecewise constant pattern
        h = np.ones(n_points) * 0.5
        # Add some variation to break symmetry
        for i in range(n_points):
            h[i] += 0.1 * np.sin(i * 2 * np.pi / n_points)
    return project_to_feasible(h, n_points)

def latin_hypercube_sampling(n_points, n_samples=50):
    """Generate Latin Hypercube Samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        h = np.random.rand(n_points)
        # Make it more structured but varied
        for i in range(n_points):
            h[i] += 0.05 * np.sin(i * 3.14159 / n_points)
        samples.append(project_to_feasible(h, n_points))
    return samples

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Pre define resolution to sample from
    resolutions = [15, 30, 45, 60, 75]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Allow 20% for initial setup and final tuning
    time_remaining = budget_s * 0.8
    
    for n_points in resolutions:
        if time() - start_time > time_remaining:
            break
            
        dx = 2.0 / n_points
        samples = latin_hypercube_sampling(n_points, 50)
        
        # Evaluate and optimize the samples
        for h_start in samples:
            if time() - start_time > time_remaining:
                break 
                
            # Local optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 50, 'ftol': 1e-7})
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
            except:
                pass
                
    # If no good solution was found, run a final detailed optimization
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        best_h = project_to_feasible(res.x, n_points)  
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
