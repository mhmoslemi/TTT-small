import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use structured initialization with low autocorrelation
        h = np.zeros(n_points)
        ones_positions = np.linspace(0, n_points, n_points//2 + 1, endpoint=False, dtype=int)[1:]
        h[ones_positions] = 1.0
        h = project_to_feasible(h, n_points)
    return h

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts
        h_restart = h_start.copy()
        restarts = 3 if n_points == 15 else 2
        for _ in range(restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Use L-BFGS-B for optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            
            # First do a coarse optimization
            h_opt = h_restart.copy()
            for _ in range(30):
                if time() - start_time > budget_s * 0.95:
                    break
                try:
                    res = minimize(obj, h_opt, method='L-BFGS-B', bounds=bounds, 
                                  options={'maxiter': 10, 'ftol': 1e-6})
                    if res.success:
                        h_opt = project_to_feasible(res.x, n_points)
                    else:
                        # Fallback to simple perturbation if optimization fails
                        h_opt = perturb_h(h_opt, 0.1)
                except:
                    h_opt = perturb_h(h_opt, 0.1)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
                    
            # Add a bit of perturbation and continue
            h_restart = perturb_h(h_opt, 0.05)
    
    # Final refinement at highest resolution if we didn't get there
    if best_n_points < max(resolutions) and time() - start_time < budget_s * 0.9:
        n_points = max(resolutions)
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Final optimization with more iterations at highest resolution
        h_opt = h_start.copy()
        for iter_num in range(50):
            if time() - start_time > budget_s * 0.95:
                break
            try:
                res = minimize(obj, h_opt, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                else:
                    h_opt = perturb_h(h_opt, 0.02)
            except:
                h_opt = perturb_h(h_opt, 0.02)
                
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
