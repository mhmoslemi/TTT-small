import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max function"""
    exp_x = np.exp(alpha * x)
    return exp_x / (np.sum(exp_x))

def compute_overlap_gradient(h, n_points, dx):
    """Compute gradient of overlap with respect to h"""
    # Compute correlation for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Find which shift has maximum overlap
    max_idx = np.argmax(corr)
    # Compute gradient of that specific correlation term
    grad = np.zeros_like(h)
    # For correlation at shift k, we have h[i] * (1 - h[(i+k)%n])
    # Partial derivatives:
    # d/dh[i] = (1 - h[(i+k)%n]) for i-th element
    # d/dh[(i+k)%n] = -h[i] for shifted element
    for i in range(n_points):
        shift = (i - max_idx + n_points) % n_points
        grad[i] += 1 - h[shift]
        grad[shift] -= h[i]
    return grad * dx

def optimize_step(h_start, n_points, dx, time_limit):
    """Single optimization step using gradient descent"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple gradient descent with projection
    learning_rate = 0.01
    for i in range(1000):  # max iterations
        if time() - start_time > time_limit:
            break
        
        # Compute gradient of overlap
        corr = np.correlate(h_start, 1 - h_start, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx
        grad = np.zeros_like(h_start)
        for j in range(n_points):
            k = (j - shift + n_points) % n_points
            grad[j] += 1 - h_start[k]
            grad[k] -= h_start[j]
        grad *= dx
        
        # Apply gradient step
        h_new = h_start - learning_rate * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_new, dx)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            learning_rate = min(learning_rate * 1.05, 0.1)
        else:
            learning_rate *= 0.99
            
        h_start = h_new.copy()
        
    return best_h, best_c5

    # Trying different resolutions
    resolutions = [25, 30, 35, 40]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try),

            h_start = project_to_feasible(h_interp, n_points_try)

        for restart in range(5):

                break
                
            # Small random perturbation
            h_perturb = project_to_feasible(h_start + np.random.normal(0, 0.01, n_points), n_points)
