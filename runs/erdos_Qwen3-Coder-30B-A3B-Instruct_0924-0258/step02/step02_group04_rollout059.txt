import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population using initial_h_values if available"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for i in range(pop_size):
        if i == 0:
            h = base_h.copy()
        else:
            noise = np.random.normal(0, 0.1, n_points)
            h = base_h + noise
        population.append(project(h, n_points))
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate a solution with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Perform two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, k=5):
    """Select one individual via tournament selection"""
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Settings
    n_points = 30
    pop_size = 50
    max_generations = 1000
    mutation_prob = 0.2
    crossover_prob = 0.7
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    best_n_points = n_points
    
    gen_no_improve = 0
    mutation_rate = mutation_prob
    
    # GA Loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            gen_no_improve = 0
        else:
            gen_no_improve += 1
            
        # Adapt mutation rate
        if gen_no_improve > 100:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        elif gen_no_improve > 50:
            mutation_rate = min(0.5, mutation_rate * 1.05)
            
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_prob:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutation
            child = mutate(child, mutation_rate, 0.1)
            
            new_population.append(child)
            
        population = new_population[:pop_size]
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
    return (list(best_h), best_c5, best_n_points)
