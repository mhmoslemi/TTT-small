import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Prepare for Adam optimization
    dx = 2.0 / n_points
    h = project_feasible(h_start.copy(), n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Adam parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    max_iter = int(budget_s * 10)  # Rough estimation
    no_improvement_count = 0
    last_best = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute new correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for this specific lag k
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k) % n_points
            grad[j] = (1 - h[idx]) - h[j]
        
        grad *= dx
        
        # Adam updates
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible set
        h = project_feasible(h, n_points)
        
        # Evaluate and track best
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improvement_count = 0
            last_best = c5
        else:
            no_improvement_count += 1
            
        # Restart if no improvement for a while
        if no_improvement_count > 1000:
            h = project_feasible(h + np.random.normal(0, 0.05, n_points), n_points)
            no_improvement_count = 0
            
        # Adaptive resolution refinement
        if i % 5000 == 0 and i > 0:
            if n_points < 100:
                old_h = h.copy()
                n_points *= 2
                dx = 2.0 / n_points
                x_old = np.linspace(0, 2, len(old_h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_old, old_h)
                h = project_feasible(h, n_points)
                best_h = h.copy()
                best_c5 = evaluate_erdos_solution(h, dx)
            elif n_points >= 100:
                break  # Stop if too high resolution
                
    return (list(best_h), best_c5, n_points)
