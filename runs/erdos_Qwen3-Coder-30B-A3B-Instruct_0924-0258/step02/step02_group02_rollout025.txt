import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

        n_points = 45

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute autocorrelations for all shifts and return minimum
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Extract valid correlations (excluding padding)
        mid = len(corr) // 2
        valid_corr = corr[mid:mid + len(h_vec)]
        return np.min(valid_corr)

    def compute_gradient_min_shift(h_vec):
        # Compute gradient with respect to the minimal autocorrelation shift
        corr = np.correlate(h_vec, h_vec, mode='full')
        mid = len(corr) // 2
        valid_corr = corr[mid:mid + len(h_vec)]
        min_shift_idx = np.argmin(valid_corr)
        min_shift = min_shift_idx
        
        # Compute gradient for this shift
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + min_shift < len(h_vec):
                grad[i] += h_vec[i + min_shift]
            if i >= min_shift:
                grad[i] += h_vec[i - min_shift]
        return grad

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    # Use a more sophisticated optimization approach:
    # gradient-based descent with adaptive learning rate
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    learning_rate = 0.01
    no_improvement_count = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Simple gradient ascent on the minimal autocorrelation
        min_autocorr = compute_min_autocorr(h)
        grad = compute_gradient_min_shift(h)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Update using gradient ascent
        h_new = h + learning_rate * grad
        h_new = project_h(h_new)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 100:
                learning_rate *= 0.5
                no_improvement_count = 0
                
        h = h_new

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
