import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_feasible(h_start.copy(), n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.02
    penalty_weight = 1e-3
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Gradient-based optimization with simulated annealing
    while time() - start_time < budget_s * 0.9:
        # Find dominant lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_lag = np.argmax(corr)
        
        # Compute gradient for dominant lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + dominant_lag) % n_points
            grad[i] = -(1 - h_current[j] - h_current[i])
        
        # Penalty gradient (constraint violations)
        penalty_grad = np.zeros(n_points)
        sum_violation = np.sum(h_current) - n_points / 2
        for i in range(n_points):
            if h_current[i] > 1:
                penalty_grad[i] -= penalty_weight * 2 * (h_current[i] - 1)
            if h_current[i] < 0:
                penalty_grad[i] -= penalty_weight * 2 * h_current[i]
        penalty_grad += penalty_weight * 2 * sum_violation
        
        # Gradient step
        h_candidate = h_current + eta * (-grad + penalty_grad)
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Cool temperature
        T *= alpha
        
        # Random perturbation if no improvement
        if no_improvement_count >= max_no_improvement:
            h_current += np.random.randn(n_points) * 0.05
            h_current = project_feasible(h_current, n_points)
            current_c5 = evaluate_erdos_solution(h_current, dx)
            no_improvement_count = 0
            
        # Resolution refinement
        if time() - start_time > budget_s * 0.8 and n_points < 100:
            new_n = min(n_points * 2, 100)
            if new_n > n_points:
                x_old = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, new_n)
                h_interp = np.interp(x_new, x_old, h_current)
                h_current = project_feasible(h_interp, new_n)
                current_c5 = evaluate_erdos_solution(h_current, 2.0 / new_n)
                n_points = new_n
                if current_c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = current_c5
    
    return (list(best_h), best_c5, n_points)
