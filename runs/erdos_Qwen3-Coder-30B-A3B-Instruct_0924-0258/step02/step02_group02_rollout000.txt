import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Extract valid shifts (from -n+1 to n-1)
        mid = len(corr) // 2
        autocorr = corr[mid - (n_points - 1):mid + n_points]
        # Compute the actual overlapping integral (h(x) * (1 - h(x+k)))
        # For shift k, it's sum of h[i] * (1 - h[i+k])
        # This is equal to: sum(h[i] * 1) - sum(h[i] * h[i+k]) = sum(h) - autocorr(k)
        # But we work with full correlation
        # This should be corrected: 
        # We want sum(h[i] * (1 - h[i+k])) = sum(h[i]) - sum(h[i]*h[i+k])
        # So: sum(h) - autocorr(k) where we deal with same indexing
        # Actually easier: use correlate of h, 1-h
        corrs = np.correlate(h_vec, 1 - h_vec, mode='full')
        corrs *= dx
        return np.min(corrs)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_gradient(h_vec, s_min):
        # Compute gradient of sum(h[i] * (1 - h[i+s_min])) for shift s_min
        # This is: sum of -(h[i+s_min]) for i = 0..n-s_min-1
        grad = np.zeros_like(h_vec)
        if s_min > 0:
            grad[:-s_min] -= h_vec[s_min:]
        else:
            grad[:] = 0  # Actually we want to handle it correctly
        return grad

    # Modified version using direct gradient-based approach
    start_time = time.time()
    
    # Initial setup
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_value = best_c5
    
    # Gradient-based iteration
    iteration = 0
    stagnation_count = 0
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # First get the current autocorr
        # Compute all autocorrelations for 1-h
        corr_h = np.correlate(h, 1 - h, mode='full')
        corr_h *= dx
        
        # Find the minimal value
        min_corr_idx = np.argmin(corr_h)
        min_corr_val = np.min(corr_h)
        
        # Detect shift with minimal correlation (we will compute its index)
        mid = len(corr_h) // 2  # corresponds to no shift
        s_min = min_corr_idx - mid  # shift value (negative means earlier)
        
        # Compute gradient
        grad = np.zeros_like(h)
        if s_min > 0:
            # Gradient w.r.t. h[i] is contribution from h[i+s_min] 
            grad[s_min:] -= h[:-s_min]
        elif s_min < 0:
            # Gradient w.r.t. h[i] is contribution from h[i+s_min] 
            grad[:-abs(s_min)] -= h[abs(s_min):]
        
        # Update using gradient
        h_new = h + learning_rate * grad
        
        # Project back to valid space
        h_new = project_h(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new)
        
        # Accept if better
        if new_c5 < c5_value:
            h = h_new
            c5_value = new_c5
            stagnation_count = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stagnation_count += 1
            if stagnation_count > 200:
                learning_rate *= 0.9
                stagnation_count = 0
        
        if iteration % 500 == 0:
            # Occasionally reset with slight perturbation
            h = project_h(h + 0.001 * np.random.normal(0, 1, n_points))

    return (best_h.tolist(), best_c5, n_points)
