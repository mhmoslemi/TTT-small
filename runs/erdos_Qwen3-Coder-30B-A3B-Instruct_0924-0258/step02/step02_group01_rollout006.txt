import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    population_size = 50
    perturbation_rate = 0.2
    max_iterations = 100
    patience = 20
    
    # Initialize population
    population = [project_to_feasible(h_start.copy(), n_points)]
    for _ in range(population_size - 1):
        h_random = np.random dirichlet(np.ones(n_points)) * (n_points / 2)
        population.append(project_to_feasible(h_random, n_points))
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # Optimization loop
    iter_count = 0
    no_improve_count = 0
    
    while iter_count < max_iterations and time() - start_time < budget_s * 0.95:
        new_population = []
        for h in population:
            # Perturbation
            h_perturbed = h.copy()
            num_perturb = max(1, int(perturbation_rate * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            h_perturbed[indices] += np.random.normal(0, 0.1, num_perturb)
            
            # Projection
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization using SLSQP
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
                
                res = minimize(
                    lambda x: evaluate_erdos_solution(x, dx),
                    h_perturbed,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-6}
                )
                
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    new_population.append(h_opt)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        no_improve_count = 0
                else:
                    new_population.append(h_perturbed)
            except:
                new_population.append(h_perturbed)
            
            no_improve_count += 1
            
        population = new_population
        iter_count += 1
        
        if no_improve_count >= patience:
            break

    # Resolution refinement
    while n_points < 1000 and time() - start_time < budget_s * 0.95:
        n_points *= 2
        dx = 2.0 / n_points
        
        # Interpolate
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, best_h)
        
        # Project and optimize
        h_refined = project_to_feasible(h_interp, n_points)
        
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            res = minimize(
                lambda x: evaluate_erdos_solution(x, dx),
                h_refined,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass

    return (list(best_h), best_c5, n_points)
