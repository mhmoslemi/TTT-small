```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the maximum overlap integral"""
    corr = np.correlate(h, 1 - h, mode='full')
    corr *= dx
    return np.max(corr)

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h = h * target_sum / np.sum(h)
    
    # Initial evaluation
    start_time = time.time()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Coarse grid to find worst case shift
    coarse_shifts = np.linspace(0, 2, 10)
    
    # Cutting plane approach
    c5_constraints = []
    
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Compute current overlaps for coarse shifts
        current_c5s = []
        for k in coarse_shifts:
            s = int(round(k / dx))
            if s < n_points and s >= 0:
                corr = np.correlate(h, 1 - h, mode='full')
                corr *= dx
                c5_val = corr[s + n_points - 1]  # Adjust for full correlation
                current_c5s.append((c5_val, k))
        
        # Find worst case shift
        if current_c5s:
            worst_case_c5, worst_k = max(current_c5s, key=lambda x: x[0])
            c5_constraints.append((worst_k, worst_case_c5))
        
        # Try to improve by solving a constrained optimization using CVXPY
        # This is a simplified model using linear approximations
        
        # Create CVX variables
        h_vars = cp.Variable(n_points)
        eta = cp.Variable()
        
        # Define constraints
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1,
            eta >= 0  # eta is non-negative
        ]
        
        # Add current constraints
        if len(c5_constraints) > 0:
            max_k = c5_constraints[0][0]
            if max_k < 2:
                s = int(round(max_k / dx))
                if s < n_points and s >= 0:
                    # Compute overlap constraint manually for now
                    corr = np.correlate(best_h, 1 - best_h, mode='full')
                    corr *= dx
                    current_max = corr[s + n_points - 1]
                    constraints.append(eta >= current_max)
        
        # Objective: minimize eta
        objective = cp.Minimize(eta)
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.UNBOUNDED:
                new_h = h_vars.value
                new_h = project_h(new_h, target_sum)
                new_c5 = compute_c5(new_h, dx)
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
            
        except Exception as e:
            # If optimization fails, fall back to simple descent
            pass
        
        # Add small noise to escape local minima
        noise = 0.001 * np.random.normal(0, 1, n_points)
        h = h + noise
        h = project_h(h, target_sum)
        
        # Try to refine the optimization with cutoff on large overlaps
        # Recompute current best
        c5_here = compute_c5(h, dx)
        if c5_here < best_c5:
            best_c5 = c5_here
            best_h = h.copy()
    
    # Finalize
    final_h = project_h(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
