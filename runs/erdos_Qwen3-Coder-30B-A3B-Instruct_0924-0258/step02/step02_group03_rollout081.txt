import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points

    # Evolution Strategy parameters
    population_size = 30
    elite_size = population_size // 10
    mutation_rate = 0.4
    beta = 0.2
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        population.append(h)
    
    # Initialize mutation strengths
    sigma = np.full(population_size, 0.05)
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    last_best_c5 = float('inf')
    
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [evaluate(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best = sorted_fitness[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = sorted_population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= 50:
                break
        
        # Selection: keep elites
        elites = sorted_population[:elite_size]
        
        # Mutation and recombination
        new_population = elites[:]
        
        while len(new_population) < population_size:
            # Select parents
            parent_indices = np.random.choice(population_size, 5, replace=False)
            parents = [sorted_population[i] for i in parent_indices]
            
            # Simple crossover (arithmetic)
            child = np.mean(parents, axis=0)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, sigma[parent_indices[0]], n_points)
                child += noise
            
            # Project to feasible space
            child = project_to_feasible(child, target_sum)
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation strengths
        sigma = sigma * np.exp(beta * np.random.normal(0, 1, population_size))
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate(best_h, dx)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
