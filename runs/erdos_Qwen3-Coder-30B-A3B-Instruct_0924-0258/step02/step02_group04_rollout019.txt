import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h using LABS-inspired approach"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # LABS-inspired construction
        h = np.zeros(n_points)
        ones_positions = np.arange(0, n_points, 2)
        h[ones_positions] = 1.0
        # Ensure exactly n_points/2 ones
        if len(ones_positions) > n_points // 2:
            h[ones_positions[n_points // 2:]] = 0.0
        elif len(ones_positions) < n_points // 2:
            h[ones_positions[-1]] = 0.0
            h[0] = 1.0
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def lbfgsb_optimize(h_start, n_points, dx, time_limit):
    """Optimize using L-BFGS-B with careful constraints"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Optimization wrapper
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    # Main loop
    for iteration in range(1000):
        if time() - start_time > time_limit:
            break
            
        try:
            # Local optimization step
            result = minimize(obj, h_best, method='L-BFGS-B', 
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 50, 'ftol': 1e-8})
            
            if result.success:
                h_opt = project_to_feasible(result.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
        
        # Occasionally perturb to escape local minima
        if iteration % 50 == 0:
            h_perturbed = perturb_h(h_best, 0.05)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < c5_best:
                h_best = h_perturbed
                c5_best = c5_perturbed
                
        # Gradual noise decay
        noise_level = max(0.01, 0.05 * (0.99 ** iteration))

    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts with perturbations
        h_restart = h_start.copy()
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = lbfgsb_optimize(h_restart, n_points, dx, time_per_res / 3)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_restart = perturb_h(h_opt, 0.1)
    
    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = lbfgsb_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
