import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    h_current = h0.copy()
    h_current = project_to_feasible(h_current, n_points / 2.0)
    dx = 2.0 / n_points
    current_c5 = evaluate(h_current, dx)
    
    # Initialize tracker
    best_h = h_current.copy()
    best_c5 = current_c5
    t = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    iterations = 0
    accept_count = 0
    total_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        # Generate a candidate solution
        noise = np.random.normal(0, t, n_points)
        h_candidate = h_current + noise
        h_candidate = project_to_feasible(h_candidate, n_points / 2.0)
        
        # Evaluate candidate
        candidate_c5 = evaluate(h_candidate, dx)
        total_count += 1
        
        # Accept or reject
        if candidate_c5 <= current_c5:
            h_current = h_candidate.copy()
            current_c5 = candidate_c5
            accept_count += 1
            no_improvement_count = 0
        else:
            # Accept with probability
            prob = np.exp(-(candidate_c5 - current_c5) / t)
            if np.random.random() < prob:
                h_current = h_candidate.copy()
                current_c5 = candidate_c5
                accept_count += 1
            else:
                no_improvement_count += 1
        
        # Update best if needed
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        
        # Update temperature
        if no_improvement_count > 50:
            # Reset temperature if stuck
            t = 0.5
            no_improvement_count = 0
        elif t > 1e-6:
            t *= alpha
        
        # Occasionally check if should reset length
        if iterations % 100 == 0 and no_improvement_count > 1000:
            t = min(t * 1.2, 0.5)
        
        iterations += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
