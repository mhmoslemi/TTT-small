import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

def trust_region_step(h, grad, hess, trust_radius, target_sum):

    # Solve trust region subproblem: min g^T d + 0.5 d^T H d, s.t. ||d|| <= trust_radius
    # This is a simplified approximation using identity Hessian
    
    # Step 1: Compute projected gradient
    d = -grad.copy()
    
    # Step 2: Project to trust region
    norm_d = np.linalg.norm(d)
    if norm_d > trust_radius and norm_d > 1e-10:
        d *= trust_radius / norm_d
    
    # Step 3: Apply and project
    new_h = h + d
    new_h = project_to_feasible(new_h, target_sum)
    
    return new_h

    H = np.eye(len(h))
    
    return grad, H
