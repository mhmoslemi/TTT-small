import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population with variations of initial_h_values or random"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for i in range(pop_size):
        if i == 0:
            h = base_h.copy()
        else:
            # Add noise to create diversity
            noise = np.random.normal(0, 0.1, n_points)
            h = base_h + noise
        h = project(h, n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate h with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Perform two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        competitors = [fitness[i] for i in indices]
        winner_idx = indices[np.argmin(competitors)]
        selected.append(population[winner_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    n_points = 30
    pop_size = 50
    max_gens = 1000
    dx = 2.0 / n_points
    
    # Initialize
    population = initialize_population(n_points, pop_size)
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    stagnation_limit = 200
    mutation_rate = 0.2
    
    # Evaluation
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Update best
    min_idx = np.argmin(fitness)
    if fitness[min_idx] < best_c5:
        best_h = population[min_idx].copy()
        best_c5 = fitness[min_idx]
    
    # GA loop
    for generation in range(max_gens):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            if np.random.rand() < 0.7:  # Crossover probability
                child1 = crossover(parent1, parent2)
                child2 = crossover(parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutate
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            offspring.extend([child1, child2])
        
        # Replace worst with offspring
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Combine and select best
        combined_pop = population + offspring
        combined_fitness = fitness + offspring_fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_pop[i] for i in sorted_indices[:pop_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitness[min_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt mutation rate
        if no_improvement_count > 50:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
        
        if no_improvement_count > stagnation_limit:
            break
    
    if best_h is None:
        # Fallback to simpler search
        h_start = initialize_population(n_points, 1)[0]
        h_opt, c5_opt = h_start, evaluate_erdos_solution(h_start, dx)
        best_h = h_opt
        best_c5 = c5_opt
    
    return (list(best_h), best_c5, n_points)
