import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085
    max_iterations = 10
    tolerance = 1e-5
    
    # Convex optimization setup
    h = Variable(n_points)
    
    # Constraints for convex optimization
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
    
    # Define problem instance that might be used in a loop
    prob = Problem(Minimize(0), constraints)
    
    # Binary search
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Build constraints to check feasibility for this C
        new_constraints = constraints[:]
        
        # For each shift k from 0 to n_points-1, add constraint:
        # sum h[i] * (1 - h[i+k]) * dx <= c_mid
        for shift in range(n_points):
            if shift == 0:
                continue
            # Constrain the correlation sum
            corr_sum = 0
            for i in range(n_points - shift):
                corr_sum += h[i] * (1 - h[i + shift])
            new_constraints.append(corr_sum * dx <= c_mid)
            
        # Setup the feasibility problem
        try:
            prob = Problem(Maximize(0), new_constraints)
            prob.solve(solver=ECOS, abstol=1e-6, reltol=1e-6, feastol=1e-6)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                # Reconstruct h from solution
                h_vals = np.array([float(h[i].value) for i in range(n_points)])
                # Ensure constraints are satisfied
                if np.all(h_vals >= 0) and np.all(h_vals <= 1):
                    c5 = evaluate_erdos_solution(h_vals, dx)
                    if c5 < best_c5 and abs(np.sum(h_vals) * dx - 1.0) < 1e-4:
                        best_h = h_vals.copy()
                        best_c5 = c5
                        
                if c5 < c_mid:
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                # Infeasible
                c_low = c_mid
                
        except Exception:
            # If optimization fails, assume infeasible
            c_low = c_mid

    # Refinement with gradient descent
    # Start from the best found solution
    h_refined = best_h.copy()
    c5_refined = best_c5
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def gradient(h):
        # Simple finite difference gradient for gradient descent
        eps = 1e-6
        grad = np.zeros_like(h)
        h_base = h.copy()
        base_obj = objective(h_base)
        for i in range(len(h)):
            h_test = h_base.copy()
            h_test[i] += eps
            h_test = np.clip(h_test, 0, 1)
            h_test_sum = np.sum(h_test)
            if h_test_sum > 0:
                h_test *= (n_points / 2) / h_test_sum
            obj = objective(h_test)
            grad[i] = (obj - base_obj) / eps
        return grad
    
    # Gradient descent step
    learning_rate = 0.01
    max_iter_refinement = 2000
    
    for i in range(max_iter_refinement):
        if time.time() - start_time > budget_s * 0.98:
            break
            
        grad = gradient(h_refined)
        h_new = h_refined - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new *= (n_points / 2) / h_new_sum
        c5_new = objective(h_new)
        if c5_new < c5_refined:
            h_refined = h_new.copy()
            c5_refined = c5_new
        else:
            learning_rate *= 0.99
            if learning_rate < 1e-6:
                break

    return (list(h_refined), c5_refined, n_points)
