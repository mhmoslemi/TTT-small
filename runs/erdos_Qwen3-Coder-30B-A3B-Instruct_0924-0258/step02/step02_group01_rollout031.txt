import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization with multiple restarts
    population_size = 20
    max_iterations = 100
    patience = 20
    improvement_count = 0
    
    # Generate initial population
    population = []
    for _ in range(population_size):
        h = project_to_feasible(np.random.normal(0.5, 0.1, n_points), n_points)
        population.append(h)
    
    # Add the best known solution to the population
    population[0] = best_h.copy()
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        improved = False
        
        # Evaluate all population members
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            improvement_count = 0
        else:
            improvement_count += 1
            if improvement_count >= patience:
                break
                
        # Create new population through mutation and optimization
        new_population = [population[0].copy()]  # Keep best
        
        # Mutation and optimization
        for i in range(1, population_size):
            # Mutation: perturb a subset of values
            h_mut = population[i].copy()
            n_mutate = max(1, int(0.2 * n_points))  # Mutate 20% of elements
            indices = np.random.choice(n_points, n_mutate, replace=False)
            noise = np.random.normal(0, 0.05, n_mutate)
            h_mut[indices] += noise
            h_mut = project_to_feasible(h_mut, n_points)
            
            # Local optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_mut, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    new_population.append(h_opt)
                else:
                    new_population.append(h_mut)
            except:
                new_population.append(h_mut)
                
        population = new_population
    
    # Resolution refinement
    resolutions = [70, 100, 150, 200, 300]  # Doubling doesn't work well, try progressive increase

        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if len(best_h) != n_points_res:
            x_orig = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_points_res)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_start = project_to_feasible(h_interp, n_points_res)
        else:
            h_start = best_h.copy()
            
        dx_res = 2.0 / n_points_res
        
        # Refine with SQP
        def obj(h):
            return evaluate_erdos_solution(h, dx_res)
            
        bounds = [(0, 1) for _ in range(n_points_res)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_res / 2}]
        
        try:
            res = minimize(obj, h_start, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points_res)
                c5_opt = evaluate_erdos_solution(h_opt, dx_res)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    n_points = n_points_res
            else:
                # If optimization fails, continue with the current one
                pass
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
