import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    n_particles = 50
    max_iter = 200
    omega_start = 0.7
    omega_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Try different resolutions
    resolutions = [25, 30, 35]
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = None
    best_n_points = None
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 25
        h_start = np.ones(n_points) * 0.5
    
    # Multi-resolution search
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best_h = []
        personal_best_c5 = []
        
        for _ in range(n_particles):
            if i == 0:
                h_init = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            else:
                h_init = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
            particles.append(h_init.copy())
            velocities.append(np.zeros(n_points))
            c5 = evaluate_erdos_solution(h_init, dx_try)
            personal_best_h.append(h_init.copy())
            personal_best_c5.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
                best_n_points = n_points
        
        # PSO main loop
        omega = omega_start
        for iter_idx in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update omega linearly
            omega = omega_start - (omega_start - omega_end) * (iter_idx / max_iter)
            
            for p in range(n_particles):
                # Generate random numbers
                r1 = np.random.random(n_points)
                r2 = np.random.random(n_points)
                
                # Update velocity
                v_new = (omega * velocities[p] + 
                        c1 * r1 * (personal_best_h[p] - particles[p]) + 
                        c2 * r2 * (best_h - particles[p]))
                velocities[p] = v_new
                
                # Update position
                h_new = particles[p] + velocities[p]
                h_new = project_to_feasible(h_new, n_points)
                particles[p] = h_new
                
                # Evaluate new solution
                c5_new = evaluate_erdos_solution(h_new, dx_try)
                
                # Update personal best
                if c5_new < personal_best_c5[p]:
                    personal_best_h[p] = h_new.copy()
                    personal_best_c5[p] = c5_new
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        best_n_points = n_points
            
            # Early stopping based on improvement
            if iter_idx > 10 and abs(personal_best_c5[0] - best_c5) < 1e-5:
                break
                
        h_start = best_h.copy()
    
    # Final refinement with gradient descent
    if best_h is not None and time() - start_time < budget_s * 0.9:
        refine_start = time()
        h_refined = best_h.copy()
        dx = 2.0 / best_n_points
        lr = 0.001
        for _ in range(100):
            if time() - refine_start > budget_s * 0.1:
                break
            # Simple gradient descent using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            base_c5 = evaluate_erdos_solution(h_refined, dx)
            for i in range(best_n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            h_refined = h_refined - lr * grad
            h_refined = project_to_feasible(h_refined, best_n_points)
            new_c5 = evaluate_erdos_solution(h_refined, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_refined.copy()
    
    return (list(best_h), best_c5, best_n_points)
