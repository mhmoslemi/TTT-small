import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    h_best = project_feasible(h_start.copy(), n_points)
    c5_best = evaluate_erdos_solution(h_best, 2.0 / n_points)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    # Main optimization loop
    t = 0
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute correlation
        corr = np.correlate(h_best, 1 - h_best, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h_best[j]) - h_best[i]
        
        grad *= (2.0 / n_points)  # Scale by dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        # Update h
        h_best -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h_best = project_feasible(h_best, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_best, 2.0 / n_points)
        
        if c5_new < c5_best:
            c5_best = c5_new
            last_improvement = t
        elif t - last_improvement > 100:
            # Random restart
            h_restart = h_best + np.random.normal(0, 0.05, n_points)
            h_restart = project_feasible(h_restart, n_points)
            c5_restart = evaluate_erdos_solution(h_restart, 2.0 / n_points)
            if c5_restart < c5_best:
                h_best = h_restart
                c5_best = c5_restart
                last_improvement = t
        
        t += 1
    
    # Try refinement with higher resolution if time allows
    if time() - start_time < budget_s * 0.9:
        n_points_refined = min(100, n_points * 2)
        if n_points_refined > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_refined)
            h_interp = np.interp(x_new, x_orig, h_best)
            h_interp = project_feasible(h_interp, n_points_refined)
            
            # Local optimization at higher resolution
            for _ in range(100):
                corr = np.correlate(h_interp, 1 - h_interp, mode='full')
                k = np.argmax(corr)
                grad = np.zeros(n_points_refined)
                for i in range(n_points_refined):
                    j = (i + k) % n_points_refined
                    grad[i] = (1 - h_interp[j]) - h_interp[i]
                grad *= (2.0 / n_points_refined)
                
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad**2
                
                m_hat = m / (1 - beta1**(t+1))
                v_hat = v / (1 - beta2**(t+1))
                
                h_interp -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
                h_interp = project_feasible(h_interp, n_points_refined)
                t += 1
                
            c5_best = evaluate_erdos_solution(h_interp, 2.0 / n_points_refined)
            h_best = h_interp
            n_points = n_points_refined
    
    return (list(h_best), c5_best, n_points)
