import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Scale to target sum
    h0 = project_to_feasible(h0, n_points / 2.0)
    
    # Particle filter parameters
    n_particles = 100
    n_iterations = 1000
    n_top_keep = 10
    resample_percent = 50
    n_mutation_per_particle = 2
    temperature_start = 1.0
    noise_initial = 0.1
    accept_rate_target = 0.3
    cooling_rate = 0.99
    decay_rate = 0.95
    
    # Initialize particles
    particles = []
    for _ in range(n_particles):
        h = h0 + np.random.normal(0, 0.1, n_points)
        particles.append(project_to_feasible(h, n_points / 2.0))
    
    # Track best
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    # Temperature and noise adaptation
    temperature = temperature_start
    noise_sigma = noise_initial
    
    iteration = 0
    while time.time() - start_time < budget_s and iteration < n_iterations:
        # Importance sampling
        weights = []
        for p in particles:
            c5 = evaluate(p, 2.0 / n_points)
            w = np.exp(-c5 / temperature)
            weights.append(w)
        
        # Normalize weights
        weights = np.array(weights)
        if np.sum(weights) > 0:
            weights = weights / np.sum(weights)
        else:
            weights = np.ones_like(weights) / len(weights)
        
        # Update best
        min_idx = np.argmin([evaluate(p, 2.0 / n_points) for p in particles])
        current_c5 = evaluate(particles[min_idx], 2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = particles[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Resampling
        if no_improvement_count < 50:
            n_keep = max(1, int(len(particles) * resample_percent / 100))
            sorted_indices = np.argsort(weights)[::-1]
            keep_indices = sorted_indices[:n_keep]
            particles = [particles[i] for i in keep_indices]
            
            # Generate new particles by mutation
            new_particles = []
            for _ in range(n_particles - len(particles)):
                parent = particles[np.random.choice(len(particles))]
                # Global mutation
                mutated = parent + np.random.normal(0, noise_sigma, n_points)
                new_particles.append(project_to_feasible(mutated, n_points / 2.0))
            
            particles.extend(new_particles)
        else:
            # Reset optimization if no progress
            particles = []
            for _ in range(n_particles):
                h = h0 + np.random.normal(0, 0.1, n_points)
                particles.append(project_to_feasible(h, n_points / 2.0))
            no_improvement_count = 0
        
        # Mutation strategy
        if iteration % 10 == 0:
            new_particles = []
            for p in particles:
                # Global and local mutations
                for _ in range(n_mutation_per_particle):
                    if np.random.random() < 0.5:
                        # Global exploration
                        mutated = p + np.random.normal(0, noise_sigma, n_points)
                    else:
                        # Local exploitation
                        mutated = p + np.random.normal(0, 0.05, n_points)
                    new_particles.append(project_to_feasible(mutated, n_points / 2.0))
            particles.extend(new_particles)
        
        # Adapt temperature and noise
        temperature *= cooling_rate
        if iteration % 20 == 0:
            acceptance_rate = 0.3  # Estimation for now
            if acceptance_rate > accept_rate_target * 1.1:
                noise_sigma *= decay_rate
            elif acceptance_rate < accept_rate_target * 0.9:
                noise_sigma /= decay_rate
        
        if iteration % 50 == 0:
            # Periodic reinitialization
            if len(particles) < n_particles:
                for _ in range(n_particles - len(particles)):
                    h = h0 + np.random.normal(0, 0.1, n_points)
                    particles.append(project_to_feasible(h, n_points / 2.0))
        
        iteration += 1
    
    if best_h is None:
        # Fallback to the best last chance if no improvement
        min_idx = np.argmin([evaluate(p, 2.0 / n_points) for p in particles])
        best_h = particles[min_idx]
    
    final_c5 = evaluate(best_h, 2.0 / n_points)
    return (best_h.tolist(), float(final_c5), n_points)
