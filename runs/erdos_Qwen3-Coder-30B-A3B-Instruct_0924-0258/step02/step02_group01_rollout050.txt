import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), k)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index]

def genetic_algorithm_optimize(h_start, n_points, dx, time_limit):
    """Optimize using genetic algorithm approach"""
    start_time = time()
    
    # GA parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    num_generations = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Start with initial solution if available
    if len(h_start) == n_points:
        population.append(h_start.copy())
        fitness.append(evaluate_erdos_solution(h_start, dx))
    else:
        population.append(np.ones(n_points) * 0.5)
        fitness.append(evaluate_erdos_solution(population[0], dx))
        
    # Generate rest of the population
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand, n_points)
        population.append(h_rand)
        fitness.append(evaluate_erdos_solution(h_rand, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < time_limit and generation < num_generations:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        new_fitness = []
        
        # Keep top 25% (elitism)
        elite_count = population_size // 4
        for i in range(elite_count):
            new_population.append(sorted_population[i])
            new_fitness.append(sorted_fitness[i])
            
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring1 = offspring1 + noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring2 = offspring2 + noise
                
            # Project to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.append(offspring1)
            new_fitness.append(evaluate_erdos_solution(offspring1, dx))
            
            if len(new_population) < population_size:
                new_population.append(offspring2)
                new_fitness.append(evaluate_erdos_solution(offspring2, dx))
        
        # Update population
        population = new_population[:population_size]
        fitness = new_fitness[:population_size]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early stopping if no improvement
        if stagnation_count > 20:
            break
            
        if time() - start_time > time_limit * 0.99:
            break
            
    return best_h, best_c5
