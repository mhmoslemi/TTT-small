import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    h0 = project_to_feasible(h0, n_points/2.0)
    
    # Simulated Annealing parameters
    temp = 0.5
    cooling_rate = 0.995
    min_temp = 1e-6
    max_iter = 10000
    
    # Track best solution
    best_h = h0.copy()
    best_c5 = evaluate(best_h, 2.0/n_points)
    
    # Main SA loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate move
        candidate = h0 + np.random.normal(0, temp, n_points)
        candidate = project_to_feasible(candidate, n_points/2.0)
        
        # Evaluate
        candidate_c5 = evaluate(candidate, 2.0/n_points)
        
        # Accept or reject
        if candidate_c5 < best_c5:
            h0 = candidate.copy()
            best_c5 = candidate_c5
            best_h = candidate.copy()
        else:
            # Probabilistic acceptance
            delta = candidate_c5 - best_c5
            accept_prob = np.exp(-delta / temp)
            if np.random.random() < accept_prob:
                h0 = candidate.copy()
        
        # Cool down
        temp = max(temp * cooling_rate, min_temp)
    
    return (best_h.tolist(), float(best_c5), n_points)
