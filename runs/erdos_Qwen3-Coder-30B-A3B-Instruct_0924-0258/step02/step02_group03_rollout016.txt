import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize h0 to target sum  
    h0 = project_to_feasible(h0, n_points/2.0)
    
    # Parameters
    n_particles = 100
    n_iter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Initialize particles
    particles = []
    for _ in range(n_particles):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0) 
        particles.append(h)
    
    # Temperature and noise adaptation
    temperature = 1.0
    proposal_std = 0.1
    
    # Iteration loop
    last_improvement = 0
    while time.time() - start_time < budget_s and last_improvement < 50:
        n_iter += 1
        
        # Importance sampling
        weights = np.zeros(n_particles)
        evaluations = []
        
        for i, h in enumerate(particles):
            h_new = h + np.random.normal(0, proposal_std, n_points)
            h_new = project_to_feasible(h_new, n_points/2.0)
            c5 = evaluate(h_new, 2.0/n_points)
            evaluations.append(c5)
            weights[i] = np.exp(-c5 / temperature)
        
        # Normalize weights
        weights /= np.sum(weights)
        
        # Resampling (stratified)
        selected_indices = []
        cumsum = np.cumsum(weights)
        u = np.random.random() / n_particles
        
        for i in range(n_particles // 2):
            idx = np.searchsorted(cumsum, u + i / n_particles)
            selected_indices.append(idx)
        
        # Generate new particles
        new_particles = []
        for idx in selected_indices:
            old_h = particles[idx]
            # Mutation: global exploration
            h1 = old_h + np.random.normal(0, proposal_std, n_points)
            h1 = project_to_feasible(h1, n_points/2.0)
            new_particles.append(h1)
            
            # Mutation: local exploitation  
            h2 = old_h + np.random.normal(0, 0.05, n_points)
            h2 = project_to_feasible(h2, n_points/2.0)
            new_particles.append(h2)
        
        # Keep top performers
        evals = [evaluate(h, 2.0/n_points) for h in new_particles]
        sorted_indices = np.argsort(evals)
        particles = [new_particles[i] for i in sorted_indices[:50]]  # top 50%
        
        # Update best
        best_idx = np.argmin(evals)
        if evals[best_idx] < best_c5:
            best_c5 = evals[best_idx]
            best_h = new_particles[best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Adaptive temperature & noise
        temperature *= 0.99
        acceptance_rate = 0.3  # target
        if np.random.random() < 0.1:  # occasional adjustments
            # Adjust proposal_std to match acceptance rate
            if np.random.random() < 0.5:
                proposal_std *= 1.1
            else:
                proposal_std *= 0.9
    
    # Final refinement
    if best_h is not None:
        final_c5 = evaluate(best_h, 2.0/n_points)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        # fallback to initial
        final_h = project_to_feasible(h0, n_points/2.0)
        final_c5 = evaluate(final_h, 2.0/n_points) 
        return (final_h.tolist(), float(final_c5), n_points)
