import numpy as np
import cvxpy as cp
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    tol = 1e-5
    max_iter = 40
    
    # Warm start with initial solution if available
    if 'initial_h_values' in globals():
        try:
            h_feasible = np.array(initial_h_values)
            h_feasible = np.clip(h_feasible, 0, 1)
            h_sum = np.sum(h_feasible)
            if h_sum > 0:
                h_feasible = h_feasible * (n_points / 2) / h_sum
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible.copy()
                best_c5 = c5_feasible
        except:
            pass
    
    # Solver settings
    solver = 'ECOS'
    solver_opts = {'feastol': 1e-6, 'reltol': 1e-6, 'abstol': 1e-6}
    
    # Run binary search
    iter_count = 0
    while iter_count < max_iter and (C_high - C_low) > tol:
        if time() - start_time > budget_s * 0.95:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Formulate CVXPY problem for feasibility check
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add the constraint for all shift values for current C_mid
        max_shift = n_points - 1
        for k in range(-max_shift, max_shift + 1):
            if k == 0:
                continue  # skip zero shift as it's a constant term
            # Compute the sum for this shift
            shift_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                shift_sum += h[i] * (1 - h[j])
            constraints.append(shift_sum <= C_mid * n_points / 2)
        
        # Create and solve the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=solver, **solver_opts)
        except:
            prob.solve(solver=solver, **solver_opts, verbose=False)
        
        # Check feasibility and update bounds
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_sol = h.value
            if h_sol is not None:
                # Project to feasible region (clip to [0,1], scale to sum to n_points/2)
                h_sol = np.clip(h_sol, 0, 1)
                h_sum = np.sum(h_sol)
                if h_sum > 0:
                    h_sol = h_sol * (n_points / 2) / h_sum
                c5 = evaluate_erdos_solution(h_sol, dx)
                if c5 < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5
                C_high = C_mid
            else:
                C_low = C_mid
        else:
            C_low = C_mid
            
        iter_count += 1
    
    # Refine with gradient descent if feasible solution found
    if time() - start_time < budget_s * 0.95:
        # Create gradient descent optimizer with projected gradient
        def project(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            return h
        
        def evaluate_correlations(h):
            corr = np.correlate(h, 1-h, mode='full')
            return np.max(corr) * dx
        
        # Use projected gradient descent
        h_refined = best_h.copy()
        learning_rate = 0.001
        max_iter_refine = 5000
        for _ in range(max_iter_refine):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_val = evaluate_correlations(h_refined)
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project(h_plus)
                val_plus = evaluate_correlations(h_plus)
                grad[i] = (val_plus - base_val) / eps
            
            # Update step with projection
            h_new = h_refined - learning_rate * grad
            h_refined = project(h_new)
            
            # Check convergence
            new_val = evaluate_correlations(h_refined)
            if abs(new_val - base_val) < 1e-8:
                break
                
        final_c5 = evaluate_correlations(h_refined)
        if final_c5 < best_c5:
            best_h = h_refined
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
