import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex optimization approach
    # We'll do binary search on the value of C5
    C_low = 0.37
    C_high = 0.38085
    best_C = C_high
    C_tolerance = 1e-5
    
    # Start with initial construction from parent if available
    h_best = h_start.copy()
    
    def solve_convex_feasibility(h0, C_val):
        """Solve convex feasibility problem for given C"""
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # Compute correlation for this shift
            if k == 0:
                # No shift case: h(i) * (1 - h(i)) for all i
                expr = cp.sum(cp.multiply(h, 1 - h))
                constraints.append(expr <= C_val * n_points / 2)
            else:
                # For shift k, we have h(i) * (1 - h(i+k)) for valid i
                expr = 0
                for i in range(n_points - k):
                    expr += h[i] * (1 - h[i + k])
                expr *= dx
                if expr != 0:  # avoid adding zero constraint
                    constraints.append(expr <= C_val * dx)
        
        objective = cp.Minimize(0)  # Feasibility problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return None
    
    # Binary search on C5 to minimize it
    for iteration in range(20):
        if time() - start_time > 0.9 * budget_s:
            break
            
        C_mid = (C_low + C_high) / 2.0
        h_feasible = solve_convex_feasibility(h_start, C_mid)
        
        if h_feasible is not None:
            # Check if this is a better solution
            h_feasible = np.clip(h_feasible, 0, 1)
            c5_val = evaluate_erdos_solution(h_feasible, dx)
            
            if c5_val < best_c5:
                best_h = h_feasible
                best_c5 = c5_val
                best_C = C_mid
                
            C_high = C_mid
        else:
            C_low = C_mid
            
        # Also try gradient-based refinement
        if time() - start_time > 0.9 * budget_s:
            break
            
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec, dx)
            
        # Gradient-free local search to improve solution infeasible region
        for _ in range(5):
            if time() - start_time > 0.9 * budget_s:
                break
            noise = np.random.normal(0, 0.02, n_points)
            h_trial = np.clip(best_h + noise, 0, 1)
            # Make sure sum is correct
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = obj_func(h_trial)
            
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
    
    # Final refinement with gradient-based optimization
    if time() - start_time < 0.95 * budget_s:
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            return evaluate_erdos_solution(h_vec, dx)
            
        # Try multiple restarts
        for i in range(3):
            if time() - start_time > 0.95 * budget_s:
                break
            # Add some noise to the best solution
            noise = np.random.normal(0, 0.03, n_points)
            h_restart = np.clip(best_h + noise, 0, 1)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            # Apply small local optimization
            def obj(h0):
                return objective(h0)
                
            h_opt = minimize(obj, h_restart, method='L-BFGS-B', 
                            bounds=[(0, 1) for _ in range(n_points)],
                            options={'maxiter': 50, 'ftol': 1e-6})
            if h_opt.success:
                h_opt_clipped = np.clip(h_opt.x, 0, 1)
                h_opt_clipped = h_opt_clipped * (n_points / 2) / np.sum(h_opt_clipped)
                c5_opt = objective(h_opt_clipped)
                if c5_opt < best_c5:
                    best_h = h_opt_clipped
                    best_c5 = c5_opt
    
    return (list(best_h), best_c5, n_points)
