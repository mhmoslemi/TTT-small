import numpy as np
from scipy.signal import convolve
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values, dtype=float)
        n_points = len(h_initial)
    else:
        n_points = 45
        h_initial = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy constraint
    h = h_initial * target_sum / np.sum(h_initial)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def create_step_function(partitions, values):
        # Create step function from partition points and values
        h_step = np.zeros(n_points)
        for i in range(len(partitions)-1):
            start_idx = int(np.round(partitions[i] * n_points / 2.0))
            end_idx = int(np.round(partitions[i+1] * n_points / 2.0))
            h_step[start_idx:end_idx] = values[i]
        return h_step
    
    def evaluate_step_function(partitions, values):
        h_step = create_step_function(partitions, values)
        return compute_c5(h_step)
    
    def optimize_step_function(partitions, values):
        # Gradient ascent update for interval values
        h_step = create_step_function(partitions, values)
        # Compute gradient using finite differences for approximation
        grad = np.zeros_like(values)
        eps = 1e-4
        base_c5 = compute_c5(h_step)
        
        for i in range(len(values)):
            # Perturb value i
            values_pert = values.copy()
            values_pert[i] += eps
            h_pert = create_step_function(partitions, values_pert)
            pert_c5 = compute_c5(h_pert)
            grad[i] = (pert_c5 - base_c5) / eps
            
        # Update with small step
        lr = 0.01
        new_values = values - lr * grad
        new_values = np.clip(new_values, 0, 1)
        return new_values

    def iteratively_adjust_partitions(partitions, values, max_iter=20):
        best_partitions = partitions.copy()
        best_values = values.copy()
        best_c5 = evaluate_step_function(partitions, values)
        
        for iteration in range(max_iter):
            # Try adding midpoints
            new_partitions = [0.0]
            for i in range(len(partitions)-1):
                new_partitions.append((partitions[i] + partitions[i+1]) / 2.0)
                new_partitions.append(partitions[i+1])
            
            # Evaluate each configuration by subdividing and computing new values
            configs = []
            for i in range(0, len(new_partitions)-1, 2):
                test_partitions = [0.0] + new_partitions[i:i+2] + [2.0]
                test_values = []
                for j in range(len(test_partitions)-1):
                    start_idx = int(np.round(test_partitions[j] * n_points / 2.0))
                    end_idx = int(np.round(test_partitions[j+1] * n_points / 2.0))
                    test_values.append(np.mean(h[start_idx:end_idx]))
                test_values = np.array(test_values)
                test_values = project_h(test_values)
                c5 = evaluate_step_function(test_partitions, test_values)
                configs.append((c5, test_partitions, test_values))
            
            # Select best configuration
            if configs:
                min_c5, best_p, best_v = min(configs, key=lambda x: x[0])
                if min_c5 < best_c5:
                    best_c5 = min_c5
                    best_partitions = best_p
                    best_values = best_v
                    
            # Try reducing number of partitions by merging
            if len(partitions) > 2:
                # Merge adjacent intervals
                for i in range(len(partitions)-2):
                    test_partitions = [partitions[0]]
                    test_values = []
                    for j in range(len(partitions)-1):
                        if j == i or j == i+1:
                            if len(test_partitions) > 1 and test_partitions[-1] != partitions[j]:
                                test_partitions.append(partitions[j])
                        else:
                            test_partitions.append(partitions[j])
                    if test_partitions[-1] != partitions[-1]:
                        test_partitions.append(partitions[-1])
                    
                    test_values = []
                    for k in range(len(test_partitions)-1):
                        start_idx = int(np.round(test_partitions[k] * n_points / 2.0))
                        end_idx = int(np.round(test_partitions[k+1] * n_points / 2.0))
                        test_values.append(np.mean(h[start_idx:end_idx]))
                    test_values = np.array(test_values)
                    test_values = project_h(test_values)
                    c5 = evaluate_step_function(test_partitions, test_values)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_partitions = test_partitions
                        best_values = test_values

        return best_partitions, best_values
    
    # Start with a few partitions
    num_partitions = 5
    partitions = np.linspace(0.0, 2.0, num_partitions + 1)
    values = np.full(num_partitions, 0.5)
    
    # Initialize from current solution
    for i in range(num_partitions):
        start_idx = int(np.round(partitions[i] * n_points / 2.0))
        end_idx = int(np.round(partitions[i+1] * n_points / 2.0))
        values[i] = np.mean(h[start_idx:end_idx])
    
    # Optimization loop
    start_time = time.time()
    best_c5 = np.inf
    best_h = h.copy()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Update values using gradient ascent
        values = optimize_step_function(partitions, values)
        values = project_h(values)
        
        # Test final configuration
        current_c5 = evaluate_step_function(partitions, values)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = create_step_function(partitions, values)
        
        # Try adjusting partitions
        if iteration % 5 == 0:
            partitions, values = iteratively_adjust_partitions(partitions, values)
            values = project_h(values)
        
        iteration += 1
        
        # Occasionally try random perturbation
        if iteration % 100 == 0:
            noise = np.random.normal(0, 1e-3, len(values))
            values = values + noise
            values = project_h(values)
    
    # Final conversion to original discretization
    converted_h = create_step_function(partitions, values)
    final_c5 = compute_c5(converted_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = converted_h
    
    return (best_h.tolist(), best_c5, n_points)
