import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def differential_evolution_step(population, fitnesses, F=0.8, CR=0.7):
        new_population = []
        for i in range(len(population)):
            # Select three distinct parents
            idxs = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Differential mutation
            mutant = a + F * (b - c)
            
            # Apply feasibility projection
            mutant = project_feasible(mutant)
            
            # Binomial crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(len(population[i]))
            for j in range(len(population[i])):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                    
            new_population.append(trial)
        return new_population
    
    # Initialize population
    P = 25  # population size
    population = [project_feasible(h + np.random.normal(0, 0.01, n_points)) for _ in range(P)]
    
    # Evaluate initial fitness
    fitnesses = [compute_overlap(individual) for individual in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_val = fitnesses[best_idx]
    
    # Evolution loop
    max_generations = 500
    generation = 0
    
    while generation < max_generations and time.time() - start_time < budget_s - 1:
        # Get best individual so far
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_val:
            best_val = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Create new population
        new_population = differential_evolution_step(population, fitnesses)
        
        # Evaluate new population
        new_fitnesses = [compute_overlap(individual) for individual in new_population]
        
        # Selection: keep better individuals
        for i in range(len(population)):
            if new_fitnesses[i] < fitnesses[i]:
                population[i] = new_population[i]
                fitnesses[i] = new_fitnesses[i]
            
        generation += 1
    
    # Final check
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
