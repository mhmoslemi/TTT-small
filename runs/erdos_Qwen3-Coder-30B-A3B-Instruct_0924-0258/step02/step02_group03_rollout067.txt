import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to target sum
    h = h * target_sum / current_sum
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):

    num_particles = 100
    particles = []
    for _ in range(num_particles):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h)
    
    # Initialize state estimate
    h_estimate = h0.copy()
    best_c5 = float('inf')
    best_h = None
    
    # Adaptive parameters
    temperature = 1.0
    proposal_std = 0.1
    acceptance_count = 0
    acceptance_target = 0.3
    iterations_no_improvement = 0
    
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Importance sampling step
        weights = []
        new_particles = []
        for h in particles:
            # Propose new particle
            h_proposed = h + np.random.normal(0, proposal_std, n_points)
            h_proposed = project_to_feasible(h_proposed, n_points/2.0)
            c5 = evaluate(h_proposed, 2.0/n_points)
            weight = np.exp(-c5 / temperature)
            weights.append(weight)
            new_particles.append(h_proposed)
        
        # Resampling step
        weights = np.array(weights)
        if np.sum(weights) > 0:
            weights = weights / np.sum(weights)
            indices = np.random.choice(len(new_particles), size=len(new_particles), p=weights)
            particles = [new_particles[i] for i in indices]
        else:
            # Fall back to random sampling if weights are too small
            particles = new_particles

        mutated_particles = []

            h1 = project_to_feasible(h1, n_points/2.0)

            # Local mutation
            h2 = h + np.random.normal(0, 0.05, n_points)
            h2 = project_to_feasible(h2, n_points/2.0)
            mutated_particles.append(h2)
        particles.extend(mutated_particles)
        
        # Update best
        for h in particles:
            c5 = evaluate(h, 2.0/n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                iterations_no_improvement = 0
            else:
                iterations_no_improvement += 1
        
        # Update estimate (top 10% as good as possible)
        particle_c5s = [evaluate(h, 2.0/n_points) for h in particles]
