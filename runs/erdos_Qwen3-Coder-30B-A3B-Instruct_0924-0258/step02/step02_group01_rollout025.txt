import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def gaussian_mutation(individual, mutation_rate, std=0.1):
    """Apply Gaussian mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, std, len(individual))
    individual = individual + mask * noise
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Initialize population
    population = [project_feasible(h_start, n_points)]
    for _ in range(population_size - 1):
        h_rand = np.random.uniform(0, 1, n_points)
        population.append(project_feasible(h_rand, n_points))
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Genetic algorithm main loop
    generation = 0
    stagnation_count = 0
    prev_best = best_c5
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        new_population = []
        
        # Create offspring
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            if np.random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = gaussian_mutation(child1, mutation_rate)
            child2 = gaussian_mutation(child2, mutation_rate)
            
            new_population.append(project_feasible(child1, n_points))
            new_population.append(project_feasible(child2, n_points))
        
        # Survival: keep top 25% from parent and offspring
        all_individuals = population + new_population
        all_fitnesses = fitnesses + [evaluate_erdos_solution(ind, dx) for ind in new_population]
        
        # Sort by fitness
        sorted_indices = np.argsort(all_fitnesses)
        surviving_indices = sorted_indices[:population_size]
        population = [all_individuals[i] for i in surviving_indices]
        fitnesses = [all_fitnesses[i] for i in surviving_indices]
        
        # Update best solution
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early stopping if no improvement
        if stagnation_count >= 20:
            break
            
        # Occasionally add diversity
        if generation % 10 == 0:
            for _ in range(5):
                h_rand = np.random.uniform(0, 1, n_points)
                population.append(project_feasible(h_rand, n_points))
                fitnesses.append(evaluate_erdos_solution(population[-1], dx))
    
    return (list(best_h), best_c5, n_points)
