import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with resolution 15
    n_points = 15
    dx = 2.0 / n_points
    
    # Initialize the step function
    h_start = initialize_h(n_points)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 1000
    reduction_factor = 0.9
    step_size = 1e-3
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Add small random perturbation
        perturbation = np.random.normal(0, step_size, n_points)
        h_perturbed = project_to_feasible(best_h + perturbation, n_points)
        
        # Evaluate perturbed solution
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        
        # Accept if better
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            
        # Adaptive step size adjustment
        step_size *= reduction_factor
    
    # Try to further improve with scipy optimization for best solution
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_opt = project_to_feasible(res.x, n_points)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except:
        pass

    return (list(best_h), best_c5, n_points)
