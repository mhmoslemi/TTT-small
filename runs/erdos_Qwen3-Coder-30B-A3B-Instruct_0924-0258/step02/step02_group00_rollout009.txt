import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085
    best_feasible_c = c_high
    best_feasible_h = best_h.copy()
    
    # Binary search for minimal C
    for iteration in range(40):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Optimization problem setup
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Correlation constraints for all shifts k
        for k in range(-n_points + 1, n_points):
            # Compute correlation sum for shift k
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h_var[i] * (1 - h_var[j])
            constraints.append(corr_sum <= c_mid * n_points / 2)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_var.value
                if h_solution is not None:
                    c5_val = evaluate_erdos_solution(h_solution, dx)
                    if c5_val < best_c5:
                        best_h = h_solution.copy()
                        best_c5 = c5_val
                    best_feasible_h = h_solution.copy()
                    best_feasible_c = c_mid
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
    
    # Refine the best solution using gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5 and best_feasible_c < 0.38085:
        h_refined = best_feasible_h.copy()
        dx = 2.0 / n_points
        
        def project_to_feasible(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            return h
        
        def compute_correlation(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Simple gradient descent refinement
        step_size = 0.01
        for _ in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            corr_values = np.correlate(h_refined, 1 - h_refined, mode='full')
            max_k = np.argmax(corr_values)
            
            # Compute gradient of max correlation w.r.t. h
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Simple finite difference
                h_plus = h_refined.copy()
                h_plus[i] += 1e-4
                h_plus = project_to_feasible(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[max_k]
                
                h_minus = h_refined.copy()
                h_minus[i] -= 1e-4
                h_minus = project_to_feasible(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[max_k]
                
                grad[i] = (corr_plus - corr_minus) * 5e5  # Scale to make gradient steeper
            
            h_refined -= step_size * grad
            h_refined = project_to_feasible(h_refined)
            
            new_c5 = compute_correlation(h_refined)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_refined.copy()
    
    return (list(best_h), best_c5, n_points)
