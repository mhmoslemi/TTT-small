import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Start with initial construction or create a base
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local search with multiple restarts
    max_restarts = 20
    max_iter_per_restart = 100
    
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_current = project_to_feasible(np.random.normal(0.5, 0.1, n_points), n_points)
        
        # Use scipy minimize with SLSQP method for local search
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        try:
            res = minimize(lambda x: evaluate_erdos_solution(x, dx), 
                          h_current, 
                          method='SLSQP',
                          bounds=bounds, 
                          constraints=cons,
                          options={'maxiter': max_iter_per_restart, 'ftol': 1e-6})
            
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
            
    # Refine at higher resolution if time allows
    resolutions = [50, 70, 90]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_to_feasible(h_interp, res)
        
        # Local refinement at new resolution
        dx_new = 2.0 / res
        bounds_new = [(0, 1) for _ in range(res)]
        cons_new = [{'type': 'eq', 'fun': lambda x: np.sum(x) - res / 2}]
        
        try:
            res_refine = minimize(lambda x: evaluate_erdos_solution(x, dx_new), 
                                h_interp,
                                method='SLSQP',
                                bounds=bounds_new, 
                                constraints=cons_new,
                                options={'maxiter': 50, 'ftol': 1e-6})
            
            if res_refine.success:
                h_refined = project_to_feasible(res_refine.x, res)
                c5_refined = evaluate_erdos_solution(h_refined, dx_new)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    n_points = res
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
