import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    start_time = time()
    
    # Initialize population
    population_size = 50

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    # Initialize population
    population = [project_feasible(np.random.rand(n_points), n_points) for _ in range(population_size - 1)]
    population.insert(0, project_feasible(h_start, n_points))
    
    # Evaluate initial population
    dx = 2.0 / n_points
    fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Main GA loop
    generations = 0
    max_generations = int(budget_s / 0.5)  # Rough estimate
    
    while time() - start_time < budget_s * 0.98 and generations < max_generations:
        # Selection via tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism: keep best
        elite_idx = np.argsort(fitness)[:population_size // 4]
        for i in elite_idx:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(population_size, 2, replace=False)
                alpha = np.random.rand()
                child = alpha * selected[parent1] + (1 - alpha) * selected[parent2]
            else:
                parent = np.random.choice(population_size)
                child = selected[parent].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = project_feasible(child, n_points)
            
            new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        generations += 1
    
    return (list(best_h), best_c5, n_points)
