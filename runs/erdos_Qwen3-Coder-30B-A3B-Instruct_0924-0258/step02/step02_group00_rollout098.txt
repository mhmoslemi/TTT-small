import numpy as np
from scipy.signal import convolve
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = convolve(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize particles
    n_particles = 50
    max_iterations = 200
    inertia_weight_start = 0.7
    inertia_weight_end = 0.4
    cognitive_coeff = 2.0
    social_coeff = 2.0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values)
        n_points = len(h_best)
    else:
        n_points = 35
        h_best = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        
        if n_points_try != n_points:
            # Interpolate from previous resolution
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, n_points), 
                                h_best)
            h_best = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_values = []
        
        for _ in range(n_particles):
            # Start particles near the current best
            h = project_to_feasible(h_best + np.random.normal(0, 0.05, n_points_try), n_points_try)
            particles.append(h.copy())
            velocities.append(np.zeros(n_points_try))
            pbest.append(h.copy())
            pbest_values.append(evaluate_erdos_solution(h, dx_try))
        
        # Global best
        gbest_idx = np.argmin(pbest_values)
        gbest = pbest[gbest_idx].copy()
        gbest_value = pbest_values[gbest_idx]
        
        # PSO main loop
        for iteration in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update inertia weight
            inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (iteration / max_iterations)
            
            # Update each particle
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia_weight * velocities[i] +
                                cognitive_coeff * r1 * (pbest[i] - particles[i]) +
                                social_coeff * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply constraints
                particles[i] = project_to_feasible(particles[i], n_points_try)
                
                # Evaluate
                c5_current = evaluate_erdos_solution(particles[i], dx_try)
                
                # Update personal best
                if c5_current < pbest_values[i]:
                    pbest[i] = particles[i].copy()
                    pbest_values[i] = c5_current
                    
                # Update global best
                if c5_current < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = c5_current
        
        # Update best solution
        if gbest_value < c5_best:
            h_best = gbest.copy()
            c5_best = gbest_value
            
    # Final refinement with gradient descent
    h_refined = h_best.copy()
    learning_rate = 0.001
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        corr = convolve(h_refined, 1 - h_refined, mode='full')
        max_idx = np.argmax(corr)
        # Simple gradient step
        grad = np.zeros_like(h_refined)
        for i in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[i] += 1e-5
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5_best) / 1e-5
        
        h_refined -= learning_rate * grad
        h_refined = project_to_feasible(h_refined, n_points)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < c5_best:
            c5_best = c5_refined
            h_best = h_refined.copy()
            
    return (list(h_best), c5_best, n_points)
