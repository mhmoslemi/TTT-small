import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Differential Evolution parameters
    pop_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 200
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if len(h) >= n_points:
            h_sample = h[:n_points].copy()
        else:
            h_sample = np.interp(np.linspace(0, 1, n_points), 
                                np.linspace(0, 1, len(h)), h)
        # Add small random perturbation
        h_sample += np.random.normal(0, 0.01, n_points)
        population.append(project_feasible(h_sample))
    
    best_fitness = float('inf')
    best_individual = None
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitnesses = []
        for individual in population:
            fitness = compute_overlap(individual)
            fitnesses.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = individual.copy()
        
        # Create new population
        new_population = []
        
        for i in range(pop_size):
            # Select three different individuals
            indices = list(range(pop_size))
            indices.remove(i)
            a, b, c = np.random.choice(indices, 3, replace=False)
            
            # Mutation: v = target + F*(a - b)
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover: trial = mutant with crossover_prob or parent
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Selection: keep trial if it's better
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Final validation
    if best_individual is None:
        best_individual = population[0]
    
    final_C5 = compute_overlap(best_individual)
    
    return (best_individual.tolist(), final_C5, n_points)
