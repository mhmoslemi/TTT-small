import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Initial coarse grid of shifts
    shift_grid = np.linspace(0, 2, 10)
    max_iter = 1000
    iteration = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute overlap for all shifts in grid
        overlaps = []
        max_overlap = -np.inf
        worst_shift_idx = 0
        
        # Evaluate overlaps for coarse grid
        for shift in shift_grid:
            s = int(round(shift / dx))
            # Compute overlap for this shift
            corr = np.correlate(h, np.flip(1 - h), mode='full')
            idx = len(corr) // 2 + s
            if 0 <= idx < len(corr):
                overlap = corr[idx] * dx
                overlaps.append(overlap)
                if overlap > max_overlap:
                    max_overlap = overlap
                    worst_shift_idx = s
            else:
                overlaps.append(0)
        
        # Add fine grid around the worst shift
        fine_shifts = np.linspace(max(0, worst_shift_idx*dx - 0.1), 
                                 min(2, worst_shift_idx*dx + 0.1), 100)
        
        # Evaluate overlaps for fine grid around worst shift
        fine_overlaps = []
        fine_shift_indices = []
        for shift in fine_shifts:
            s = int(round(shift / dx))
            corr = np.correlate(h, np.flip(1 - h), mode='full')
            idx = len(corr) // 2 + s
            if 0 <= idx < len(corr):
                overlap = corr[idx] * dx
                fine_overlaps.append(overlap)
                fine_shift_indices.append(s)
            else:
                fine_overlaps.append(0)
                fine_shift_indices.append(s)
        
        # Identify the worst fine shift
        max_fine_overlap = max(fine_overlaps)
        best_fine_shift_idx = fine_shift_indices[fine_overlaps.index(max_fine_overlap)]
        
        # Solution with constraints (simpler approach)
        # We'll use a much simpler algorithm here:
        # 1. Try to slightly nudge values to reduce maximum overlap.
        # 2. Use a local heuristic with small random perturbations.
        
        # Random perturbation with projection
        noise = 0.01 * np.random.normal(0, 1, n_points)
        h_candidate = h + noise
        h_candidate = project_h(h_candidate)
        
        new_c5 = compute_c5(h_candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            h = h_candidate.copy()
        else:
            # Slight gradient-like update using finite differences
            h_plus = h.copy()
            h_minus = h.copy()
            eps = 1e-3
            for i in range(n_points):
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = project_h(h_plus)
                h_minus = project_h(h_minus)
                c5_plus = compute_c5(h_plus)
                c5_minus = compute_c5(h_minus)
                # Approximate gradient
                grad = (c5_plus - c5_minus) / (2 * eps)
                h[i] = max(0, min(1, h[i] - 0.001 * grad))
                
            h = project_h(h)
            # Recompute current C5 to check improvement
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
