import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def gradient_smooth_max(h, n_points, dx):
    """Gradient of the smooth max approximation"""
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    grad_corr = alpha * exp_corr / sum_exp
    
    # Compute gradient properly using convolution properties
    grad = np.zeros_like(h)
    for i in range(len(h)):
        grad[i] = grad_corr[i] * dx
    
    return grad

    """Optimize using gradient descent with projection and restarts"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    h = h_start.copy()
    
    # Running average parameters for adaptive learning rate
    momentum = 0.9
    learning_rate = 0.01
    avg_grad = np.zeros_like(h)
    
    for i in range(2000):  # maximum iterations
        if time() - start_time > time_limit:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        alpha = 100
        exp_corr = np.exp(alpha * corr)
        sum_exp = np.sum(exp_corr)
        grad_corr = alpha * exp_corr / sum_exp
        
        # Gradient with respect to h
        gradient = np.zeros_like(h)
        for k in range(len(h)):
            gradient[k] += grad_corr[k] * dx
        
        # Update with momentum
        avg_grad = momentum * avg_grad + (1 - momentum) * gradient
        h = h - learning_rate * avg_grad
        
        # Project to feasible region
        h = project_to_feasible(h, n_points)
        
        # Update learning rate
        if i % 50 == 0 and i > 0:
            learning_rate *= 0.9
            
        # Check if we improved
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Restart if we're stuck
        if i % 200 == 0 and i > 0:
            h = project_to_feasible(best_h + np.random.normal(0, 0.02, n_points), n_points)
    
    return best_h, best_c5
