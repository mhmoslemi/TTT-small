import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_autocorrelation(h_vec):
        """Compute all autocorrelations of h_vec with itself."""
        n = len(h_vec)
        autocorr = np.zeros(2 * n - 1)
        for i in range(n):
            for j in range(n):
                if i + j < 2 * n - 1:
                    autocorr[i + j] += h_vec[i] * h_vec[j]
        return autocorr * dx

    def compute_min_autocorr_and_grad(h_vec):
        """Return min autocorrelation and gradient w.r.t. that smallest value."""
        n = len(h_vec)
        autocorr = np.zeros(2 * n - 1)
        for i in range(n):
            for j in range(n):
                if i + j < 2 * n - 1:
                    autocorr[i + j] += h_vec[i] * h_vec[j]
        autocorr *= dx
        
        min_idx = np.argmin(autocorr)
        min_val = autocorr[min_idx]
        
        # Gradient for the minimal shift
        grad = np.zeros(n)
        if min_idx < n:
            # h[i] * h[i+min_idx] contributes to this correlation
            for i in range(n):
                if i + min_idx < n:
                    grad[i] += h_vec[i + min_idx]
        else:
            # h[i] * h[i-min_idx] for i >= min_idx
            for i in range(n):
                if i >= min_idx - n + 1:
                    grad[i] += h_vec[i - min_idx + n - 1]
        
        return min_val, grad
    
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    step_size = 0.01
    patience = 0
    max_patience = 50
    
    iters = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Step 1: compute the current minimum autocorrelation and gradient
        min_val, grad = compute_min_autocorr_and_grad(h)
        
        # Step 2: update h to increase the minimal autocorrelation
        h = h + step_size * grad
        h = project_h(h)
        
        # Step 3: check if better solution
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
            step_size = 0.01  # reset learning rate
        else:
            patience += 1
            if patience > max_patience:
                step_size *= 0.5
                patience = 0

        iters += 1

    # Final project
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
