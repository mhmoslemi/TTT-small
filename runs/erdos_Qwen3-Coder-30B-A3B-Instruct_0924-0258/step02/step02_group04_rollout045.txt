import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts
        h_restart = h_start.copy()
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Simple gradient descent based approach
            h_opt = h_restart.copy()
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            
            # Gradient ascent on the objective (negate for minimization)
            for _ in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                # Small perturbation for gradient estimation
                h_plus = h_opt.copy()
                h_minus = h_opt.copy()
                eps = 1e-4
                idx = np.random.randint(len(h_opt))
                h_plus[idx] += eps
                h_minus[idx] -= eps
                
                h_plus = project_to_feasible(h_plus, n_points)
                h_minus = project_to_feasible(h_minus, n_points)
                
                grad = (evaluate_erdos_solution(h_plus, dx) - 
                       evaluate_erdos_solution(h_minus, dx)) / (2*eps)
                
                # Update with small step, and project back
                step_size = 1e-3
                h_opt[idx] -= step_size * grad
                h_opt = project_to_feasible(h_opt, n_points)
                
                new_c5 = evaluate_erdos_solution(h_opt, dx)
                if new_c5 < c5_opt:
                    c5_opt = new_c5
                else:
                    # Try a different search direction
                    h_opt = h_restart.copy()  # Backtrack
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_restart = np.random.rand(n_points) * 0.1 + 0.45
            h_restart = project_to_feasible(h_restart, n_points)
    
    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        h_opt = h_start.copy()
        c5_opt = evaluate_erdos_solution(h_opt, dx)
        
        # Simple optimization loop
        for _ in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
            h_plus = h_opt.copy()
            h_minus = h_opt.copy()
            eps = 1e-4
            idx = np.random.randint(len(h_opt))
            h_plus[idx] += eps
            h_minus[idx] -= eps
            
            h_plus = project_to_feasible(h_plus, n_points)
            h_minus = project_to_feasible(h_minus, n_points)
            
            grad = (evaluate_erdos_solution(h_plus, dx) - 
                   evaluate_erdos_solution(h_minus, dx)) / (2*eps)
            
            step_size = 1e-3
            h_opt[idx] -= step_size * grad
            h_opt = project_to_feasible(h_opt, n_points)
            
            new_c5 = evaluate_erdos_solution(h_opt, dx)
            if new_c5 < c5_opt:
                c5_opt = new_c5
            else:
                # Try a different search direction
                h_opt = h_start.copy()  # Backtrack
                
        best_h = h_opt
        best_c5 = c5_opt
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
