import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Setup initial population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Initialize population
    population = [project_feasible(h_start, n_points)]
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = project_feasible(h_random, n_points)
        population.append(h_random)
    
    # Evaluate initial population
    dx = 2.0 / n_points
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    generations = 0
    max_generations = budget_s * 2  # Rough estimation
    
    while time() - start_time < budget_s * 0.95 and generations < max_generations:
        # Selection via tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size] if i + 1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Arithmetic crossover
                alpha = np.random.rand()
                offspring1 = alpha * parent1 + (1 - alpha) * parent2
                offspring2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring2 += noise
            
            # Project to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
        
        # Evaluation
        new_fitness = [evaluate_erdos_solution(h, dx) for h in new_population[:population_size]]
        population = new_population[:population_size]
        fitness = new_fitness
        
        # Survival: Keep best 25% and replace worst with mutations
        sorted_indices = np.argsort(fitness)
        keep_indices = sorted_indices[:population_size // 4]
        survivors = [population[i] for i in keep_indices]
        survivors_fitness = [fitness[i] for i in keep_indices]
        
        # Fill remaining slots with mutations of best individuals
        for _ in range(population_size - len(survivors)):
            selected_individual = survivors[np.random.choice(len(survivors))]
            mutated = selected_individual + np.random.normal(0, 0.05, n_points)
            mutated = project_feasible(mutated, n_points)
            survivors.append(mutated)
            survivors_fitness.append(evaluate_erdos_solution(mutated, dx))
        
        population = survivors
        fitness = survivors_fitness
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        generations += 1
    
    return (list(best_h), best_c5, n_points)
