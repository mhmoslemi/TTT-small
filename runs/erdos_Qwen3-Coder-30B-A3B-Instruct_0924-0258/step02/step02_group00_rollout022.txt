import numpy as np
from scipy.fft import convolve
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        h_best = h_start.copy()
    else:
        n_points = 30
        h_best = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    c5_best = evaluate_erdos_solution(h_best, dx)
    best_n_points = n_points
    
    # PSO parameters
    omega = 0.7
    omega_min = 0.4
    c1 = 2.0
    c2 = 2.0
    n_particles = 50
    max_iter = 200
    
    # Resolution schedule
    resolutions = [25, 30, 35]
    
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        # Interpolate if needed  
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, n_points), 
                                h_best)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Initialize particles
        particles_h = []
        particles_v = []
        particles_best_h = []
        particles_best_c5 = []
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            particles_h.append(h.copy())
            particles_v.append(np.zeros(n_points))
            c5 = evaluate_erdos_solution(h, dx_try)
            particles_best_h.append(h.copy())
            particles_best_c5.append(c5)
            
        # Swarm best
        swarm_best_idx = np.argmin(particles_best_c5)
        swarm_best_h = particles_best_h[swarm_best_idx].copy()
        swarm_best_c5 = particles_best_c5[swarm_best_idx]
        
        # PSO iterations
        for iter_idx in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update inertia weight
            omega_t = omega_min + (omega - omega_min) * (1.0 - iter_idx / max_iter)
            
            for p_idx in range(n_particles):
                # Velocity update
                r1 = np.random.rand(n_points)
                r2 = np.random.rand(n_points)
                particles_v[p_idx] = (omega_t * particles_v[p_idx] + 
                                    c1 * r1 * (particles_best_h[p_idx] - particles_h[p_idx]) +
                                    c2 * r2 * (swarm_best_h - particles_h[p_idx]))
                
                # Position update
                particles_h[p_idx] += particles_v[p_idx]
                
                # Apply feasibility constraints
                particles_h[p_idx] = project_to_feasible(particles_h[p_idx], n_points)
                
                # Evaluate
                c5 = evaluate_erdos_solution(particles_h[p_idx], dx_try)
                
                # Update particle best
                if c5 < particles_best_c5[p_idx]:
                    particles_best_c5[p_idx] = c5
                    particles_best_h[p_idx] = particles_h[p_idx].copy()
                
                # Update swarm best
                if c5 < swarm_best_c5:
                    swarm_best_c5 = c5
                    swarm_best_h = particles_h[p_idx].copy()
        
        # Update global best
        if swarm_best_c5 < c5_best:
            h_best = swarm_best_h.copy()
            c5_best = swarm_best_c5
            best_n_points = n_points
            
        # Gradient descent refinement (small step)
        last_c5 = c5_best + 1e-5
        lr = 0.001
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            h_prev = h_best.copy()
            h_new = h_best - lr * (h_best - 0.5)  # Simple gradient-like update
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx_try)
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
            else:
                break
    
    return (list(h_best), c5_best, best_n_points)
