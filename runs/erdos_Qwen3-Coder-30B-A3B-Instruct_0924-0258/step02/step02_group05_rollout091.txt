import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_h(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Multi-resolution optimization with Fourier features
    max_resolutions = 5
    max_fourier_components = 15
    learning_rate = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    max_iter_per_res = 3000
    
    # Initialize with a coarse Fourier representation
    n_fourier = 5
    a_coeffs = np.zeros(n_fourier)
    b_coeffs = np.zeros(n_fourier)
    
    # Use initial solution for frequency domain if available
    try:
        # Determine coefficients via FFT of initial function
        h_fft = np.fft.fft(h_init)
        freqs = np.fft.fftfreq(n_points, dx)
        # Take first few components (ignore DC)
        a_coeffs[:n_fourier] = np.real(h_fft[1:n_fourier+1]) * 2 / n_points
        b_coeffs[:n_fourier] = np.imag(h_fft[1:n_fourier+1]) * 2 / n_points
    except:
        pass
    
    def h_from_fourier(a_coeffs, b_coeffs):
        # Evaluate the Fourier series at sample points
        x = np.linspace(0, 2, n_points)
        h = np.ones(n_points) * (np.sum(a_coeffs) / 2)
        for i in range(n_fourier):
            k = i + 1
            h += a_coeffs[i] * np.cos(2 * np.pi * k * x) + \
                 b_coeffs[i] * np.sin(2 * np.pi * k * x)
        return h / n_fourier
    
    # Adam variables
    m_a = np.zeros_like(a_coeffs)
    v_a = np.zeros_like(a_coeffs)
    m_b = np.zeros_like(b_coeffs)
    v_b = np.zeros_like(b_coeffs)
    
    best_c5 = float('inf')
    best_a = a_coeffs.copy()
    best_b = b_coeffs.copy()
    
    for res in range(max_resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Increment fourier components
        if res > 0:
            n_fourier = min(n_fourier * 2, max_fourier_components)
            new_a = np.zeros(n_fourier)
            new_b = np.zeros(n_fourier)
            new_a[:len(a_coeffs)] = a_coeffs
            new_b[:len(b_coeffs)] = b_coeffs
            a_coeffs = new_a
            b_coeffs = new_b
            
        # Prepare optimizer
        m_a = np.zeros_like(a_coeffs)
        v_a = np.zeros_like(a_coeffs)
        m_b = np.zeros_like(b_coeffs)
        v_b = np.zeros_like(b_coeffs)
        
        last_improvement = 0
        early_stop = False
        
        for iter_count in range(max_iter_per_res):
            if time.time() - start_time > budget_s - 1:
                early_stop = True
                break
                
            # Evaluate function
            h_eval = h_from_fourier(a_coeffs, b_coeffs)
            h_eval = project_feasible(h_eval)
            current_c5 = evaluate_h(h_eval)
            
            # Compute gradient estimates
            eps = 1e-4
            grad_a = np.zeros_like(a_coeffs)
            grad_b = np.zeros_like(b_coeffs)
            
            for i in range(n_fourier):
                # Perturb a_i
                a_plus = a_coeffs.copy()
                a_plus[i] += eps
                h_plus = h_from_fourier(a_plus, b_coeffs)
                h_plus = project_feasible(h_plus)
                c_plus = evaluate_h(h_plus)
                
                a_minus = a_coeffs.copy()
                a_minus[i] -= eps
                h_minus = h_from_fourier(a_minus, b_coeffs)
                h_minus = project_feasible(h_minus)
                c_minus = evaluate_h(h_minus)
                
                grad_a[i] = (c_plus - c_minus) / (2 * eps)
                
                # Perturb b_i
                b_plus = b_coeffs.copy()
                b_plus[i] += eps
                h_plus = h_from_fourier(a_coeffs, b_plus)
                h_plus = project_feasible(h_plus)
                c_plus = evaluate_h(h_plus)
                
                b_minus = b_coeffs.copy()
                b_minus[i] -= eps
                h_minus = h_from_fourier(a_coeffs, b_minus)
                h_minus = project_feasible(h_minus)
                c_minus = evaluate_h(h_minus)
                
                grad_b[i] = (c_plus - c_minus) / (2 * eps)
                
            # Adam updates
            m_a = adam_beta1 * m_a + (1 - adam_beta1) * grad_a
            v_a = adam_beta2 * v_a + (1 - adam_beta2) * grad_a**2
            m_a_hat = m_a / (1 - adam_beta1**(iter_count + 1))
            v_a_hat = v_a / (1 - adam_beta2**(iter_count + 1))
            
            m_b = adam_beta1 * m_b + (1 - adam_beta1) * grad_b
            v_b = adam_beta2 * v_b + (1 - adam_beta2) * grad_b**2
            m_b_hat = m_b / (1 - adam_beta1**(iter_count + 1))
            v_b_hat = v_b / (1 - adam_beta2**(iter_count + 1))
            
            a_coeffs -= learning_rate * m_a_hat / (np.sqrt(v_a_hat) + 1e-8)
            b_coeffs -= learning_rate * m_b_hat / (np.sqrt(v_b_hat) + 1e-8)
            
            # Check improvement
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_a = a_coeffs.copy()
                best_b = b_coeffs.copy()
                last_improvement = 0
            else:
                last_improvement += 1
                
            if last_improvement > 1000:
                break
                
        if early_stop:
            break
            
    # Final refinement
    h_final = h_from_fourier(best_a, best_b)
    h_final = project_feasible(h_final)
    final_c5 = evaluate_h(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
