import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_gradient(x, alpha=100):
    """Gradient of smooth max function"""
    exp_x = np.exp(alpha * x)
    sum_exp = np.sum(exp_x)
    return alpha * exp_x / sum_exp

def compute_overlap_gradient(h, n_points, dx):
    """Compute gradient of the overlap with respect to h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Find the maximum index
    max_idx = np.argmax(corr)
    # Get the gradient of smooth max
    grad_smooth = smooth_max_gradient(corr, alpha=100)
    # We only care about gradient at max correlation value
    # For simplicity, use finite difference approximation for entire gradient
    grad = np.zeros_like(h)
    eps = 1e-6
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    """Optimize using gradient descent with projections"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a simple projected gradient descent approach
    max_iter = 500
    learning_rate = 0.01
    epsilon = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Compute current overlap and gradient
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient using finite differences
        grad = np.zeros_like(best_h)
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - max_val) / epsilon
        
        # Gradient descent step
        new_h = best_h - learning_rate * grad
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        
        # Adaptive learning rate decrease
        if iteration % 20 == 0:
            learning_rate *= 0.95
            
    return best_h, best_c5
