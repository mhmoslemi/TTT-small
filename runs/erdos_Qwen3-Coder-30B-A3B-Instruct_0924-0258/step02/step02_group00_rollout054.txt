import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    
    # Binary search approach for finding tight C5
    c_low = 0.37
    c_high = 0.38085
    tolerance = 1e-5
    max_iterations = 10
    
    # For each iteration in binary search
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Define convex feasibility problem
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,  # integral constraint
            h_var >= 0,  # lower bound
            h_var <= 1   # upper bound
        ]
        
        # Add overlap constraints for all possible shifts
        for k in range(n_points):
            if k == 0:
                continue  # Skip zero shift
            # For shift k, we sum over i from 0 to n-k-1
            # h[i] * (1 - h[i+k]) <= c_mid * dx
            # But we need to handle period properly
            # Let's just compute the full convolution via array operations
            corr_sum = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr_sum += h_var[i] * (1 - h_var[i + k])
                else:
                    break  # if we run off the edge, skip
            if corr_sum > 0:  # Only add constraint if meaningful
                constraints.append(corr_sum <= c_mid)
        
        # Form and solve the problem
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_var.value
                c5_feas = evaluate_erdos_solution(h_feasible, dx)
                if c5_feas < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_feas
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
            
    # Post-processing with gradient descent to refine
    # Choose best h to refine
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Create a smooth version of the solution
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def project_to_feasible(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            return h
        
        # Gradient descent with projections
        h_refined = best_h.copy()
        step_size = 0.01
        for i in range(200):  # few iterations to avoid premature convergence
            if time() - start_time > budget_s * 0.95:
                break
            # Stochastic gradient descent step
            h_new = h_refined - step_size * np.random.normal(0, 0.1, n_points)
            h_new = project_to_feasible(h_new)
            if objective(h_new) < objective(h_refined):
                h_refined = h_new

        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
    return (list(best_h), best_c5, n_points)
