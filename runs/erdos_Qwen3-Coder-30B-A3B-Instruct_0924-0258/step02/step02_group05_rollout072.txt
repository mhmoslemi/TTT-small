import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # Binary search approach
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    # LP solver settings
    try:
        from cvxpy import Variable, Minimize, Problem, sum, multiply, norm
        solver = "ECOS"
    except ImportError:
        solver = None
    
    while high - low > tolerance and (time.time() - start_time) < budget_s - 1:
        if (time.time() - start_time) > budget_s - 1:
            break
            
        C5_candidate = (low + high) / 2.0
        
        # Build constraints for linear program
        A_eq = np.array([np.ones(n_points)])
        b_eq = np.array([target_sum])
        
        # Build inequality constraints (overlap constraints)
        A_ub = []
        b_ub = []
        
        # Generate all the overlap constraint rows
        for k in range(-n_points+1, n_points):
            # Compute the indices involved in the convolution
            row = np.zeros(n_points)
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in [0, n_points-k-1]
                for i in range(n_points - k):
                    row[i] = 1.0
            else:
                # h[i] * (1 - h[i+k]) for i in [0, n_points+k-1] when k < 0
                for i in range(n_points + k):
                    row[i + k] = 1.0 
            # Flip the sign to match inequalities format (A_ub * h <= b_ub)
            A_ub.append(-row)
            b_ub.append(-C5_candidate * dx)
            
        # Bounds for variables
        bounds = [(0, 1) for _ in range(n_points)]
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Solve LP using scipy
        try:
            result = linprog(
                c=np.zeros(n_points),
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method='highs'
            )
            
            if result.success:
                h_candidate = result.x
                # Check if it satisfies our discrete evaluation
                C5_eval = compute_overlap(h_candidate)
                if C5_eval <= C5_candidate + 1e-6:
                    high = C5_candidate
                    best_h = h_candidate.copy()
                    best_c5 = C5_eval
                else:
                    low = C5_candidate
            else:
                low = C5_candidate
        except:
            low = C5_candidate
            
    # Final check with best solution
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
