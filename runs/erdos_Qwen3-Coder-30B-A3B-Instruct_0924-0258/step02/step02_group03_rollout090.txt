import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points/2.0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # MSPF parameters
    n_particles = 100
    temperature = 1.0
    proposal_std = 0.1
    acceptance_target = 0.3
    min_acceptance = 0.1
    max_acceptance = 0.5
    patience_limit = 50
    patience_count = 0
    best_c5 = float('inf')
    best_h = h0.copy()
    
    # Initialize particles
    particles = [h0.copy()]
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, proposal_std, n_points)
        particles.append(project_to_feasible(h, n_points/2.0))
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % 100 == 0:
            # Adaptive temperature
            temperature *= 0.99
        
        # Importance sampling with weights
        weights = []
        new_particles = []
        c5_vals = []
        
        for h in particles:
            if np.random.random() < 0.1:  # Occasionally generate fresh particles
                h_prime = h + np.random.normal(0, proposal_std, n_points)
            else:
                h_prime = h + np.random.normal(0, proposal_std, n_points)
            h_prime = project_to_feasible(h_prime, n_points/2.0)
            c5 = evaluate(h_prime, 2.0/n_points)
            weights.append(np.exp(-c5 / temperature) if temperature > 1e-6 else 0.0)
            new_particles.append(h_prime)
            c5_vals.append(c5)
        
        # Resampling
        if sum(weights) > 0:
            weights = np.array(weights)
            weights /= np.sum(weights)
            indices = np.random.choice(n_particles, size=n_particles//2, p=weights)
            particles = [new_particles[i] for i in indices]
        else:
            particles = new_particles[:n_particles//2]
        
        # Add mutations
        for _ in range(5):
            h = particles[np.random.randint(len(particles))]
            # Global perturbation
            h_new = h + np.random.normal(0, proposal_std, n_points)
            h_new = project_to_feasible(h_new, n_points/2.0)
            particles.append(h_new)
            # Local perturbation
            h_new = h + np.random.normal(0, 0.05, n_points)
            h_new = project_to_feasible(h_new, n_points/2.0)
            particles.append(h_new)
        
        # Keep best particles and re-init bottom ones
        c5_vals = [evaluate(h, 2.0/n_points) for h in particles]
        sorted_indices = np.argsort(c5_vals)
        top_10_percent = sorted_indices[:n_particles//10]
        best_particles = [particles[i] for i in top_10_percent]
        
        # Replace bottom particles
        for i in range(len(particles)):
            if i not in top_10_percent:
                if np.random.random() < 0.5:
                    h = particles[i]
                    # Local mutation
                    h_new = h + np.random.normal(0, 0.05, n_points)
                    h_new = project_to_feasible(h_new, n_points/2.0)
                else:
                    # Random from top 10 percent
                    h_new = best_particles[np.random.randint(len(best_particles))]
                particles[i] = h_new
        
        # Update temperature and proposal std
        # Update best
        current_best_c5 = min(c5_vals)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = particles[np.argmin(c5_vals)].copy()
            patience_count = 0
        else:
            patience_count += 1
        if patience_count > patience_limit:
            break
            
        # Adapt proposal standard deviation based on acceptance rate
        acceptance_rate = np.mean([c5_val < best_c5 * 1.1 for c5_val in c5_vals])
        if acceptance_rate < min_acceptance:
            proposal_std *= 0.9
        elif acceptance_rate > max_acceptance:
            proposal_std *= 1.1
            
    # Final evaluation
    final_c5 = evaluate(best_h, 2.0/n_points)
    return (best_h.tolist(), float(final_c5), n_points)
