import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Optimization loop
    t = 0
    last_improvement = 0
    max_iter = 100000
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation and find best lag
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for j in range(n_points):
            grad[j] = (1 - best_h[(j + k) % n_points]) - best_h[j]
        grad *= dx
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Apply update
        updated_h = best_h - update
        
        # Project to feasible region
        updated_h = project_feasible(updated_h, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(updated_h, dx)
        if c5 < best_c5:
            best_h = updated_h.copy()
            best_c5 = c5
            last_improvement = i
        else:
            # Restart with perturbed solution if no improvement for a while
            if i - last_improvement > 1000 and i > 5000:
                perturbation_strength = 0.05
                h_perturbed = np.clip(best_h + np.random.normal(0, perturbation_strength, n_points), 0, 1)
                h_projected = project_feasible(h_perturbed, n_points)
                c5_perturbed = evaluate_erdos_solution(h_projected, dx)
                if c5_perturbed < best_c5:
                    best_h = h_projected
                    best_c5 = c5_perturbed
                    last_improvement = i
        
        # Adaptive refinement
        if i % 5000 == 0 and n_points < 100 and time() - start_time < budget_s * 0.8:
            # Increase resolution
            n_points_new = min(n_points * 2, 100)
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_interp = np.interp(x_new, x_old, best_h)
            best_h = project_feasible(h_interp, n_points_new)
            n_points = n_points_new
            dx = 2.0 / n_points
            
    return (list(best_h), best_c5, n_points)
