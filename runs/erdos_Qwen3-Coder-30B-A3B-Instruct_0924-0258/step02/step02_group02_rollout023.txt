import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h_init satisfies the constraint
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_individual(h_vec):
        # Returns -C5 (negative for minimization)
        return -compute_c5(h_vec)

    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h_init.copy())
    for i in range(1, population_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_h(h)
        population.append(h)
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    elapsed = 0
    
    # Start evolution loop
    for generation in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate all individuals
        fitness_scores = [evaluate_individual(ind) for ind in population]
        
        # Selection: tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population with crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size] if i + 1 < population_size else selected[0]
            
            # Crossover with probability
            if np.random.random() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation with probability
            for child in [child1, child2]:
                if np.random.random() < mutation_rate:
                    # Randomly mutate 1 or 2 values
                    num_mutations = np.random.choice([1, 2])
                    mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                    for idx in mutation_indices:
                        # Add noise with small magnitude
                        child[idx] += np.random.normal(0, 0.01)
                        # Project back to valid range
                        child[idx] = np.clip(child[idx], 0, 1)
                    # Rescale to maintain constraint
                    child = project_h(child)
                new_population.append(child)
            
            if len(new_population) >= population_size:
                break
        
        # Keep population size consistent
        population = new_population[:population_size]
        
        # Update best solution
        for i, ind in enumerate(population):
            current_c5 = -fitness_scores[i]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = ind.copy()
    
    # Final validation and return best solution
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
