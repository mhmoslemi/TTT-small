import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a piecewise constant pattern
        # Start with a simple symmetric configuration
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def simplex_init(n_points, n_particles):
    """Initialize particles using simplex method"""
    # Start with a basic initial solution
    particles = [initialize_h(n_points)]
    
    # Generate additional particles by making small perturbations
    for i in range(n_particles - 1):
        perturbed = particles[0].copy()
        idx = i % len(perturbed)
        perturbed[idx] = min(1.0, max(0.0, perturbed[idx] + (np.random.rand() - 0.5) * 0.2))
        particles.append(project_to_feasible(perturbed, n_points))
    
    return particles

def local_refinement(h_start, n_points, dx):
    """Refine solution using L-BFGS-B optimization"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 50})
        if res.success:
            return project_to_feasible(res.x, n_points)
    except:
        pass
    return h_start

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # PSO parameters
    n_points = 30
    dx = 2.0 / n_points
    n_particles = 10
    max_iter = 2000
    w = 0.7
    c1 = c2 = 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    
    # Initialize particles
    particles = simplex_init(n_points, n_particles)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Evaluate initial particles
    particle_values = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Track personal bests
    pbest = particles.copy()
    pbest_values = particle_values.copy()
    
    # Find global best
    gbest_idx = np.argmin(particle_values)
    gbest = particles[gbest_idx].copy()
    gbest_value = particle_values[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_value
    best_n_points = n_points
    
    # PSO iteration
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate current particle
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best if needed
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    iter_no_improve = 0
                    
                    # Store the best solution
                    best_h = gbest.copy()
                    best_c5 = gbest_value
                    best_n_points = n_points
                    
            else:
                iter_no_improve += 1
        
        # Periodically perform shaking if no improvement
        if iter_no_improve >= shaking_threshold:
            for i in range(n_particles):
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            iter_no_improve = 0
            
        # Periodically apply L-BFGS-B refinement
        if iteration % lbfgs_interval == 0 and iteration > 0:
            top_candidates = np.argsort(particle_values)[:min(10, n_particles)]
            for i in top_candidates:
                refined = local_refinement(particles[i], n_points, dx)
                refined_value = evaluate_erdos_solution(refined, dx)
                if refined_value < particle_values[i]:
                    particles[i] = refined
                    particle_values[i] = refined_value
                    if refined_value < gbest_value:
                        gbest = refined.copy()
                        gbest_value = refined_value
                        best_h = gbest.copy()
                        best_c5 = gbest_value
        # Early termination if we're close to the target
        if best_c5 < 0.3807:
            break
    
    return (list(best_h), best_c5, best_n_points)
