import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimizations
    population_size = 30
    max_iterations = 50
    
    # Create initial population
    population = [project_to_feasible(h_start.copy(), n_points)]
    
    for i in range(1, population_size):
        h_perturb = project_to_feasible(np.random.normal(h_start, 0.05), n_points)
        population.append(h_perturb)
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current population
        population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Find best in current population
        best_idx = np.argmin(population_c5)
        if population_c5[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = population_c5[best_idx]
        
        # Create new population from current best
        new_population = []
        for h in population:
            # Perturb
            h_perturb = h.copy()
            indices = np.random.choice(n_points, size=max(1, n_points//5), replace=False)
            h_perturb[indices] += np.random.normal(0, 0.05, len(indices))
            
            # Project to feasible set
            h_perturb = project_to_feasible(h_perturb, n_points)
            
            # Optimize with SQP
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(lambda x: evaluate_erdos_solution(x, dx), h_perturb, 
                               method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-8})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    new_population.append(h_opt)
                else:
                    new_population.append(h_perturb)
            except:
                new_population.append(h_perturb)
        
        population = new_population
    
    # Resolution refinement
    resolutions = [70, 100, 150]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
    
        # Interpolate
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_restart = project_to_feasible(h_interp, res)
        
        # Local optimization at higher resolution
        bounds = [(0, 1) for _ in range(res)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - res / 2}]
        
        try:
            res_opt = minimize(lambda x: evaluate_erdos_solution(x, 2.0/res), h_restart,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 50, 'ftol': 1e-8})
            if res_opt.success:
                h_opt = project_to_feasible(res_opt.x, res)
                c5_opt = evaluate_erdos_solution(h_opt, 2.0/res)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    n_points = res
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
