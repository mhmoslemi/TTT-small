import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    # Optimization loop
    max_iter = int(budget_s * 10)  # Rough estimate
    last_improvement = 0
    restart_count = 0
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.99:
            break
            
        t += 1
        
        # Compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for the dominant lag k
        grad = np.zeros(n_points)
        for j in range(n_points):
            # j + k is valid in the convolution
            if j + k < n_points:
                grad[j] = (1 - best_h[j + k]) - best_h[j]
            else:
                # Wrap around
                grad[j] = (1 - best_h[(j + k) % n_points]) - best_h[j]
        
        grad *= dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        best_h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        best_h = project_feasible(best_h, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            last_improvement = i
        else:
            # Restart if no improvement for 1000 iterations
            if i - last_improvement > 1000 and restart_count < 5:
                perturbation = np.random.normal(0, 0.05, n_points)
                best_h = project_feasible(best_h + perturbation, n_points)
                restart_count += 1
                last_improvement = i
    
    # Resolution refinement if time allows
    resolutions = [50, 70, 100]
    for n_res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n_res)
        
        # Re-optimize with higher resolution
        current_h = h_interp
        current_c5 = evaluate_erdos_solution(current_h, 2.0 / n_res)
        
        # Short optimization
        for _ in range(50):
            if time() - start_time > budget_s * 0.98:
                break
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            k = np.argmax(corr)
            
            grad = np.zeros(n_res)
            for j in range(n_res):
                if j + k < n_res:
                    grad[j] = (1 - current_h[j + k]) - current_h[j]
                else:
                    grad[j] = (1 - current_h[(j + k) % n_res]) - current_h[j]
            
            grad *= (2.0 / n_res)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            current_h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
            current_h = project_feasible(current_h, n_res)
            
            new_c5 = evaluate_erdos_solution(current_h, 2.0 / n_res)
            if new_c5 < current_c5:
                current_c5 = new_c5
                best_h = current_h.copy()
                best_c5 = current_c5
                n_points = n_res
                t += 1
    
    return (list(best_h), best_c5, n_points)
