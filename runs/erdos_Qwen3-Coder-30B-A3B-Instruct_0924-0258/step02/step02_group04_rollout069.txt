import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use a binary LABS-like pattern to get low correlation initially
        h = np.zeros(n_points)
        ones_count = n_points // 2
        positions = np.linspace(0, n_points - 1, ones_count, dtype=int)
        h[positions] = 1.0
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use a more structured approach
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Try different resolutions
    for n_points in [15, 30, 45, 75]:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Optimization with better method
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        # Constraints and bounds
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        # Run several optimizations with different restarts
        h_best = h_start.copy()
        c5_best = evaluate_erdos_solution(h_start, dx)
        
        # Multiple restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturb
            noise = np.random.normal(0, 0.05, n_points)
            h_perturbed = project_to_feasible(h_best + noise, n_points)
            
            # L-BFGS-B optimization
            try:
                res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < c5_best:
                        h_best = h_opt.copy()
                        c5_best = c5_opt
            except:
                pass
            
            # Also try a direct perturbation
            h_direct = project_to_feasible(h_best + np.random.normal(0, 0.03, n_points), n_points)
            c5_direct = evaluate_erdos_solution(h_direct, dx)
            if c5_direct < c5_best:
                h_best = h_direct
                c5_best = c5_direct
                
        if c5_best < best_c5:
            best_h = h_best.copy()
            best_c5 = c5_best
            best_n_points = n_points
    
    # Final optimization at highest resolution to get the best
    if best_h is not None and best_n_points < 75:
        n_points = 75
        dx = 2.0 / n_points
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_n_points), best_h)
        h_start = project_to_feasible(h_start, n_points)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
                best_n_points = n_points

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            best_h = project_to_feasible(res.x, n_points)
            best_c5 = evaluate_erdos_solution(best_h, dx)
            best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
