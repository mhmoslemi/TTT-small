import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    scale = target_sum / current_sum if current_sum > 1e-10 else 1.0
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize swarm
    n_points = 70
    swarm_size = 40
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    for i in range(swarm_size):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Best solutions tracking
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate(p, 2.0/n_points) for p in particles]
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO parameters
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.05
    
    iteration = 0
    
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v = omega * velocities[i] + \
                c1 * r1 * (pbest[i] - particles[i]) + \
                c2 * r2 * (gbest - particles[i])
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            new_h = particles[i] + v
            new_h = project_to_feasible(new_h, n_points / 2.0)
            particles[i] = new_h
            
            # Evaluate new solution
            new_value = evaluate(new_h, 2.0/n_points)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = new_h.copy()
                pbest_values[i] = new_value
                
                # Update global best
                if new_value < gbest_value:
                    gbest = new_h.copy()
                    gbest_value = new_value
        
        iteration += 1
    
    return (gbest.tolist(), float(gbest_value), n_points)
