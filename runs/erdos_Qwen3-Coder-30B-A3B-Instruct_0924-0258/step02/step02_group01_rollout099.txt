import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    # GA parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = int(budget_s * 10)
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_init = project_feasible(h_start, n_points)
        population.append(h_init)
        fitness = evaluate_erdos_solution(h_init, dx)
        population_fitness.append(fitness)
    
    # Add random initial solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
        fitness = evaluate_erdos_solution(h, dx)
        population_fitness.append(fitness)
    
    # Track best solution
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # Evolution loop
    gen = 0
    while time() - start_time < budget_s * 0.98 and gen < max_generations:
        gen += 1
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_k)
            tournament_fitness = [population_fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover with probability
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation with probability
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
                
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)

    return (list(best_h), best_c5, n_points)
