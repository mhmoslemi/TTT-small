import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    n_particles = 30
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        swarm.append(project_feasible(h))
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        
    # Fill remaining particles
    for _ in range(n_particles - len(swarm)):
        particle = np.random.uniform(0, 1, n_points)
        particle = project_feasible(particle)
        swarm.append(particle)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Track personal bests
    personal_best = [compute_overlap(p) for p in swarm]
    personal_best_positions = [p.copy() for p in swarm]
    
    # Track global best
    global_best_idx = np.argmin(personal_best)
    global_best = personal_best[global_best_idx]
    global_best_position = swarm[global_best_idx].copy()
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_positions[i] - swarm[i]) +
                           c2 * r2 * (global_best_position - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate new position
            current_overlap = compute_overlap(swarm[i])
            
            # Update personal best
            if current_overlap < personal_best[i]:
                personal_best[i] = current_overlap
                personal_best_positions[i] = swarm[i].copy()
                
                # Update global best
                if current_overlap < global_best:
                    global_best = current_overlap
                    global_best_position = swarm[i].copy()
    
    # Return best solution
    return (global_best_position.tolist(), global_best, n_points)
