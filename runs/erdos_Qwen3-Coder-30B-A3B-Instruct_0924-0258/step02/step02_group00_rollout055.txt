import numpy as np
import cvxpy as cp
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    low = 0.37
    high = 0.38085
    tol = 1e-6
    max_iter = 40
    binary_search_time = 60
    
    # Initialize result tracking
    best_feasible_h = None
    best_feasible_c5 = float('inf')
    
    # Binary search phase
    for iteration in range(max_iter):
        if time() - start_time > binary_search_time:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate convex feasibility problem
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Constraint for each shift k
        for k in range(1 - n_points, n_points):
            # Circular index for shift k
            shifted_indices = np.arange(n_points) - k
            shifted_indices = shifted_indices % n_points
            
            # Sum of h[i] * (1 - h[shifted_indices[i]])
            corr_sum = cp.sum(cp.multiply(h, 1 - h[shifted_indices]))
            constraints.append(corr_sum <= mid * n_points / 2.0)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
                h_solution = h.value
                if h_solution is not None:
                    c5_test = evaluate_erdos_solution(h_solution, dx)
                    if c5_test < best_feasible_c5:
                        best_feasible_h = h_solution.copy()
                        best_feasible_c5 = c5_test
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
    
    # If no feasible solution found, use best from binary search
    if best_feasible_h is not None:
        final_h = best_feasible_h
        final_c5 = best_feasible_c5
    else:
        final_h = h_start
        final_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement phase with gradient descent
    refine_time = budget_s - (time() - start_time)
    if refine_time > 10:
        # Simple gradient descent with projection
        h_refined = final_h.copy()
        learning_rate = 0.01
        max_refine_iter = int(refine_time * 50)
        
        for i in range(max_refine_iter):
            if time() - start_time > budget_s - 1:
                break
                
            # Compute correlations and gradients
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx]
            
            # Gradient for the maximum correlation (simplified)
            # We focus on improving the specific worst-case shift
            k_opt = max_idx - (len(corr) - 1) // 2  # Convert full correlation index to shift
            
            # Compute gradient of correlation for shift k_opt
            grad = np.zeros_like(h_refined)
            if k_opt >= 0:
                for i in range(n_points):
                    grad[i] += 1 - h_refined[(i + k_opt) % n_points]
            else:
                for i in range(n_points):
                    grad[i] += 1 - h_refined[(i + k_opt) % n_points]
            
            # Update with gradient descent
            h_update = h_refined - learning_rate * grad
            h_refined = np.clip(h_update, 0, 1)
            
            # Project to correct sum
            h_sum = np.sum(h_refined)
            h_refined = h_refined * (n_points / 2) / h_sum
            
            # Evaluate and update best
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < final_c5:
                final_c5 = c5_refined
                final_h = h_refined.copy()
    
    return (list(final_h), final_c5, n_points)
