import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern with slight variation
        h = np.ones(n_points) * 0.5
        # Add some variation
        h[::3] = 0.7
        h[1::3] = 0.3
        h[2::3] = 0.6
    return project_to_feasible(h, n_points)

def latin_hypercube_sample(n_points, num_samples=50):
    """Generate Latin hypercube samples"""
    samples = []
    for _ in range(num_samples):
        sample = np.random.uniform(0, 1, n_points)
        # Normalize to sum to n_points/2
        sample = sample * (n_points / 2) / np.sum(sample)
        # Keep within [0,1]
        sample = np.clip(sample, 0, 1)
        samples.append(sample)
    return samples

def optimize_single(h_start, n_points, dx, time_limit):
    """Perform optimization on a single starting point"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Latin hypercube sampling for better diversification
    initial_samples = latin_hypercube_sample(n_points, 10)
    
    # Random restarts with different initializations
    for sample in initial_samples:
        if time() - start_time > time_limit:
            break
            
        # Local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        try:
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
        
            res = minimize(obj, sample, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts with Latin hypercube sampling
        h_restart = h_start.copy()
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = optimize_single(h_restart, n_points, dx, time_per_res / 10)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_restart = np.random.uniform(0, 1, n_points)
            h_restart = project_to_feasible(h_restart, n_points)

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_single(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
