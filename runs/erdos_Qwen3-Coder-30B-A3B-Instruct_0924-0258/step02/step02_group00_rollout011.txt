import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    exp_x = np.exp(alpha * x)
    return exp_x / np.sum(exp_x)

def compute_overlap_gradient(h, n_points, dx):
    # Compute the full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Smooth max gradient
    sm_grad = smooth_max_grad(corr, alpha=100)
    
    # Compute gradient with respect to h
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Compute derivative of the convolution term
        # For each k, we have correlation[i] = sum(h[j] * (1 - h[(j+k)%n]))
        # Gradient w.r.t h[i] that contributes to corr[i+k] when j=i
        # Also, gradient w.r.t (j+k)%n when j+i corresponds to that
        for k in range(n_points):
            idx = (i - k) % n_points
            if idx < n_points:
                grad[i] += sm_grad[k] * (1 - h[idx]) * dx
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use standard GD with restarts, but better termination
    lr = 0.01
    patience = 0
    max_patience = 100
    last_improvement = 0
    
    for i in range(1000):
        if time() - start_time > time_limit:
            break
            
        # Compute gradient
        try:
            grad = compute_overlap_gradient(best_h, n_points, dx)
            # Update with gradient descent
            h_new = best_h - lr * grad
            h_new = project_to_feasible(h_new, n_points)
            
            # Evaluate with new value
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                last_improvement = i
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    lr *= 0.8
                    patience = 0
        except:
            pass
            
    return best_h, best_c5

    np.random.seed(seed)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [25, 30, 35, 40, 45]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Multiple restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb h_start a bit
            h_perturb = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            h_opt, c5_opt = optimize_with_gradient_descent(h_perturb, n_points, dx_try, 
                                                          min(100, budget_s / 5))
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
                
        h_start = best_h.copy()
        
    return (list(best_h), best_c5, best_n_points)
