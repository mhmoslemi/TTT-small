import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    mask = np.random.random(len(individual)) < mutation_rate
    if np.any(mask):
        individual += np.random.normal(0, std, len(individual)) * mask
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        population.append(h_start)
        for _ in range(pop_size - 1):
            h = np.random.random(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.random(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolution loop
    no_improvement_count = 0
    max_no_improvement = 20
    
    while time() - start_time < budget_s * 0.95:
        # Selection
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Projection
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Survival - keep best individuals
        combined_population = population + new_population
        combined_fitnesses = fitnesses + new_fitnesses
        
        # Sort by fitness
        sorted_indices = np.argsort(combined_fitnesses)
        survival_indices = sorted_indices[:pop_size]
        
        population = [combined_population[i] for i in survival_indices]
        fitnesses = [combined_fitnesses[i] for i in survival_indices]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    return (list(best_h), best_c5, n_points)
