import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Best solution found
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic Algorithm
    tournament_k = 5
    mutation_rate = 0.2
    crossover_rate = 0.8
    max_generations = int(budget_s * 20)  # Rough estimate
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_k)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Reproduction
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            fit = evaluate_erdos_solution(h, 2.0 / n_points)
            new_fitness.append(fit)
        
        # Survival: keep best individuals and replace worst
        combined_pop = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_pop[i] for i in sorted_indices[:population_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
    
    return (list(best_h), best_c5, n_points)
