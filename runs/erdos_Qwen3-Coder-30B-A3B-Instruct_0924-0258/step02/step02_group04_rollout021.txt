import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a modified piecewise constant pattern that was empirically effective
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def update_spectrum(H, penalty_weight=0.1):
    # Compute power spectrum
    P = np.abs(H)**2
    
    # Gradient of penalty: sum of |H|^4, derivative with respect to H
    # d/df |H(f)|^4 = 4 * |H(f)|^2 * H(f) * conj(H(f)) * (1/H(f)) if H(f) != 0
    # Simplified as 4 * |H(f)|^2 * conj(H(f)) = 4 * P(f) * conj(H(f))
    # for update step, we take negative gradient
    gamma = 1e-4
    # Smooth gradient-based update to penalize high spectral energy
    grad_penalty = 4 * P * np.conj(H)
    H_new = H - gamma * grad_penalty * penalty_weight
    
    # Ensure real signal properties (Hermitian symmetry)
    # this is a sensible adjustment for real signals in DFT domain
    return H_new

def fourier_optimize(h_start, n_points, dx, time_limit):
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Initialize H from h_start
    H = np.fft.fft(h_start)
    
    # Optimization parameters
    penalty_weight = 0.1
    max_iter = 5000
    early_stop_count = 0
    patience = 200
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Optimization step
        H = update_spectrum(H, penalty_weight)
        
        # Transform back to time domain
        h = np.real(np.fft.ifft(H))
        h = project_to_feasible(h, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < c5_best:
            h_best = h.copy()
            c5_best = c5
            early_stop_count = 0
            penalty_weight = max(0.01, penalty_weight * 0.995)  # Gradually reduce penalty
        else:
            early_stop_count += 1
            if early_stop_count > patience:
                penalty_weight = min(1.0, penalty_weight * 1.1)
                
        # Early stopping if no real improvement
        if iteration > 100 and early_stop_count > 100:
            break
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with small resolution for initial exploration
    resolutions = [15, 30, 45, 60]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    start_time = time()
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), budget_s * 0.2)
        
        # Run Fourier-based optimization
        h_opt, c5_opt = fourier_optimize(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
    
    # Final refinement if needed, and ensure there's a result
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = fourier_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
