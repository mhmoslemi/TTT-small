import numpy as np
import cvxpy as cp
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    best_feasible_h = None
    best_feasible_C = C_high
    iterations = 40
    time_per_iter = (budget_s - 60) / iterations  # Reserve 60s for refinement
    
    # Binary search phase
    current_C = (C_low + C_high) / 2
    for i in range(iterations):
        if time() - start_time > budget_s - 10:
            break
            
        # Create variables and constraints
        h = cp.Variable(n_points)
        
        # Sum constraint
        sum_constraint = cp.sum(h) == n_points / 2
        
        # Convolution constraints (all k values)
        convolution_constraints = []
        for k in range(-n_points + 1, n_points):
            # Create shifted version using circular indexing
            shift = k
            if shift < 0:
                h_shifted = h[-shift:] + h[:-shift]
            elif shift > 0:
                h_shifted = h[:-shift] + h[-shift:]
            else:
                h_shifted = h
            
            # Sum of h[i] * (1 - h_shifted[i])
            conv_sum = cp.sum(cp.multiply(h, 1 - h_shifted))
            convolution_constraints.append(conv_sum <= current_C * n_points / 2)
        
        # Bound constraints
        bound_constraints = [h >= 0, h <= 1]
        
        # Construct the problem
        prob = cp.Problem(cp.Minimize(0), 
                         sum_constraint + convolution_constraints + bound_constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
                h_solution = h.value
                if h_solution is not None and np.all(h_solution >= 0) and np.all(h_solution <= 1):
                    c5_solution = evaluate_erdos_solution(h_solution, dx)
                    if c5_solution < best_feasible_C:
                        best_feasible_C = c5_solution
                        best_feasible_h = h_solution.copy()
                    C_high = current_C
                else:
                    C_low = current_C
            else:
                C_low = current_C
        except Exception as e:
            C_low = current_C
            
        current_C = (C_low + C_high) / 2
    
    # Refinement phase with gradient-based method
    if best_feasible_h is not None:
        best_h = best_feasible_h.copy()
        best_c5 = best_feasible_C
    
    # Gradient-based refinement with projected gradient method
    # Using small perturbations around the feasible solution
    best_refined_h = best_h.copy()
    best_refined_c5 = best_c5
    
    remaining_time = budget_s - (time() - start_time)
    # Simple refinement iterations
    for i in range(1000):
        if time() - start_time > budget_s - 10:
            break
        # Take a random small perturbation
        perturb = np.random.normal(0, 0.01, n_points)
        h_pert = best_refined_h + perturb
        # Project to feasible region
        h_pert = np.clip(h_pert, 0, 1)
        h_sum = np.sum(h_pert)
        if h_sum > 0:
            h_pert = h_pert * (n_points / 2) / h_sum
        # Evaluate
        c5_new = evaluate_erdos_solution(h_pert, dx)
        if c5_new < best_refined_c5:
            best_refined_h = h_pert.copy()
            best_refined_c5 = c5_new
    
    return (list(best_refined_h), best_refined_c5, n_points)
