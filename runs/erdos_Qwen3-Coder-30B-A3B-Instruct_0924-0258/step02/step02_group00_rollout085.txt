import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max approximation"""
    exp_x = np.exp(alpha * x)
    sum_exp = np.sum(exp_x)
    return alpha * exp_x / sum_exp

def compute_overlap_gradient(h, n_points, dx):
    """Compute gradient of overlap with respect to h"""
    # Compute correlations for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Find where max is achieved
    max_idx = np.argmax(corr)
    # For main diagonal elements: differences with 1-h
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Dot product of h with (1-h) shifted by i
        k = max_idx - i
        if 0 <= k < n_points:
            grad[i] += (1 - h[k]) * dx
        # Also account for other shifts that might be max
        for shift in range(n_points):
            k = shift
            if 0 <= k < n_points:
                grad[i] += (1 - h[(i + k) % n_points]) * dx
    return grad

def optimize_one_step(h_start, n_points, dx, time_limit):
    """Optimize one step using BFGS with feasibility projection"""
    start_time = time()
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    def obj_and_grad(h):
        h_proj = project_to_feasible(h, n_points)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        corr_max_idx = np.argmax(corr)
        corr_max = corr[corr_max_idx]
        # Smooth max to avoid non-differentiable max
        smooth_c5 = smooth_max(corr, alpha=100)
        # Gradient calculation using finite differences for objective
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            obj_plus = smooth_max(np.correlate(h_plus, 1 - h_plus, mode='full'), alpha=100)
            obj_base = smooth_c5
            grad[i] = (obj_plus - obj_base) / eps
        # Return function value to minimize and gradient
        return smooth_c5, grad
    
    # BFGS with lots of restarts
    for _ in range(1000):
        if time() - start_time > time_limit:
            break
        try:
            # Use local solver on projected points
            res = minimize(lambda x: smooth_max(np.correlate(project_to_feasible(x, n_points), 1 - project_to_feasible(x, n_points), mode='full'), alpha=100),
                           h_current, method='L-BFGS-B',
                           bounds=[(0, 1)] * n_points, 
                           options={'maxiter': 30, 'ftol': 1e-8, 'gtol': 1e-5})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    h_current = h_opt.copy()
                else:
                    h_current = project_to_feasible(h_current + np.random.normal(0, 0.01, n_points), n_points)
            else:
                h_current = project_to_feasible(h_current + np.random.normal(0, 0.01, n_points), n_points)
        except:
            h_current = project_to_feasible(h_current + np.random.normal(0, 0.01, n_points), n_points)
    
    return best_h, best_c5
