import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h

    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    return h

def compute_gradient(h, dx):
    """Compute approximate gradient for trust region method."""
    n = len(h)
    # Compute correlations for each lag
    corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(corr)
    # The gradient for component i is approximately the derivative
    # of the maximum term in the correlation sum, but we compute it as
    # the actual effect of modifying h[i] on max correlation
    grad = np.zeros_like(h)
    
    # For each index, compute partial derivative
    # By finite differences
    eps = 1e-4
    h_plus = h.copy()
    h_minus = h.copy()
    for i in range(n):
        h_plus[i] += eps
        h_minus[i] -= eps
        h_plus = project_to_feasible(h_plus, n/2.0)
        h_minus = project_to_feasible(h_minus, n/2.0)
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2.0 * eps)
        h_plus[i] = h[i]
        h_minus[i] = h[i]
        
    return grad * dx

def trust_region_step(h, grad, hessian, trust_radius, dx):
    """Solve step within trust region using projected gradient descent."""
    n = len(h)
    # Using simple quadratic approximation: minimize g^T*d + 0.5*d^T*H*d
    # subject to ||d|| <= trust_radius
    # For simplicity, use identity Hessian and solve constraint directly
    if np.linalg.norm(grad) <= 1e-10:
        return np.zeros_like(h)
    
    d = -grad
    d_norm = np.linalg.norm(d)
    if d_norm > trust_radius:
        d = d * trust_radius / d_norm
    
    return d

    no_improvement_count = 0
    max_no_improvement = 100

    # Initial evaluations
    old_c5 = evaluate(h, dx)
    h = project_to_feasible(h, n_points/2.0)
    old_c5 = evaluate(h, dx)
    p = 0

        h_old = h.copy()
