import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_feasible(h_start.copy(), n_points)
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    max_no_improve = 50
    no_improve_count = 0
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Compute dominant lag k for gradient
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr) - (len(h) - 1)
        
        # Compute gradient (approximate)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k) % len(h)
            grad[i] = -(1 - h[j] - h[i])
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        sum_h = np.sum(h)
        penalty_grad += 2 * lambda_penalty * (sum_h - n_points/2)
        penalty_grad += 2 * lambda_penalty * np.clip(h, 0, 1) - np.clip(h, 1, np.inf)
        penalty_grad -= 2 * lambda_penalty * np.clip(-h, 0, np.inf)
        
        # Update h
        h_new = h - eta * (grad + penalty_grad)
        
        # Project to feasible region
        h_new = project_feasible(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        delta_c5 = new_c5 - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_new.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Cooling
        T *= alpha
        
        # Random restart if no improvement
        if no_improve_count > max_no_improve:
            h = project_feasible(np.random.rand(n_points) * 0.1 + 0.45, n_points)
            current_c5 = evaluate_erdos_solution(h, dx)
            no_improve_count = 0
        
        # Resolution refinement if time allows
        if time() - start_time > budget_s * 0.7 and n_points < 100:
            n_points *= 2
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
            h = project_feasible(h, n_points)
            dx = 2.0 / n_points
            current_c5 = evaluate_erdos_solution(h, dx)
            best_h = h.copy()
            best_c5 = current_c5
            T = 0.5
            no_improve_count = 0
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
