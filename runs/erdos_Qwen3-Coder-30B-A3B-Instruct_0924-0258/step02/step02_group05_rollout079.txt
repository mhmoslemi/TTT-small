import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def differential_evolution_step(population, fitness, F=0.5, CR=0.7):
        new_population = []
        for i in range(len(population)):
            # Select three distinct individuals
            indices = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            
            # Crossover
            trial = np.where(np.random.random(n_points) < CR, mutant, population[i])
            
            # Projection
            trial = project_feasible(trial)
            
            new_population.append(trial)
        
        return new_population
    
    # Evolution parameters
    pop_size = 25
    max_generations = 1000
    F = 0.7
    CR = 0.5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        perturbed = h + np.random.normal(0, 0.05, n_points)
        population.append(project_feasible(perturbed))
    
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = [compute_overlap(ind) for ind in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Differential evolution step
        population = differential_evolution_step(population, fitness, F, CR)
        generation += 1
    
    # Final validation
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
