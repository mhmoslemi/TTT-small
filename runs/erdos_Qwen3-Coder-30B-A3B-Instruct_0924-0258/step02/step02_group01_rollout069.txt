import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_feasible(h_start, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    penalty_weight = 1e-3
    max_no_improve = 50
    no_improve_count = 0
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Find dominant lag for gradient
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute approx gradient for this lag
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = -(1 - h[j] - h[i])
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        h_sum = np.sum(h)
        penalty_grad += 2 * penalty_weight * (h_sum - n_points / 2)
        penalty_grad += penalty_weight * (h > 1) - penalty_weight * (h < 0)
        
        # Take gradient step
        h_candidate = h - eta * (grad + penalty_grad)
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - best_c5
        
        # Metropolis acceptance
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cool temperature
        T *= alpha
        
        # Random perturbation if no improvement
        if no_improve_count > max_no_improve:
            h += np.random.randn(n_points) * 0.05
            h = project_feasible(h, n_points)
            no_improve_count = 0
            
    # Multi-resolution refinement
    if time() - start_time < budget_s * 0.9:
        n_points_new = n_points * 2
        dx_new = 2.0 / n_points_new
        h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
        h_new = project_feasible(h_new, n_points_new)
        
        # Fine tune
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            k_max = np.argmax(corr)
            grad = np.zeros_like(h_new)
            for i in range(n_points_new):
                j = (i + k_max) % n_points_new
                grad[i] = -(1 - h_new[j] - h_new[i])
            
            h_new = h_new - 0.005 * grad
            h_new = project_feasible(h_new, n_points_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    return (list(best_h), best_c5, len(best_h))
