import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Function to evaluate C5 from a step function
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    # LP based approach for improving the heuristic
    def solve_feasible_lp(C_candidate):
        # Create equality constraint: sum(h) = target_sum
        A_eq = np.array([np.ones(n_points)])
        b_eq = np.array([target_sum])
        
        # Generate inequality constraints for all k
        A_ub = []
        b_ub = []
        
        for k in range(-n_points+1, n_points):
            # Compute indices for h[i] and h[i+k]
            if k >= 0:
                i_indices = np.arange(n_points - k)
                k_indices = np.arange(k, n_points)
            else:
                i_indices = np.arange(-k, n_points)
                k_indices = np.arange(n_points + k)
            
            # Build constraint vector
            constraint = np.zeros(n_points)
            for i, j in zip(i_indices, k_indices):
                constraint[i] += 1.0  # h[i] contributes to overlap with (1-h[j])
            
            A_ub.append(constraint)
            b_ub.append(C_candidate * dx)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Variable bounds [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Solve the LP
        try:
            result = linprog(
                c=np.zeros(n_points),
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method='highs',
                options={'presolve': True}
            )
            if result.success:
                return result.x
        except:
            pass
        
        return None
    
    # Main search loop
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    best_solution = h.copy()
    best_C5 = compute_overlap(best_solution)
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        if time.time() - start_time > budget_s - 1:
            break
            
        C_candidate = (low + high) / 2.0
        
        # Try to solve the LP for this C_candidate
        solution = solve_feasible_lp(C_candidate)
        
        if solution is not None:
            # Feasible solution found, try to improve the bound
            high = C_candidate
            # Project to ensure constraints
            solution = np.clip(solution, 0, 1)
            solution = solution * target_sum / np.sum(solution)
            current_C5 = compute_overlap(solution)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_solution = solution.copy()
        else:
            # Infeasible solution, increase C_candidate
            low = C_candidate
    
    # Final validation and projection
    best_C5 = compute_overlap(best_solution)
    
    return (best_solution.tolist(), best_C5, n_points)
