import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Fallback: simple symmetric construction
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    # Ensure h_start is feasible
    h_start = project(h_start, n_points)
    
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    population = [project(np.random.normal(h_start, 0.1), n_points) for _ in range(population_size)]
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Genetic Algorithm Parameters
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    stagnation_count = 0
    max_stagnation = 200
    
    # Keep track of best across generations
    best_overall = best_c5
    
    # Time allocation
    gen_time = 0.5  # Estimated time per generation
    max_generations = int((budget_s * 0.9) / gen_time)
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(individual, dx) for individual in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            best_overall = min(best_overall, best_c5)
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Check for stagnation
        if stagnation_count > max_stagnation:
            # Reset population with better solutions
            for i in range(population_size):
                population[i] = project(np.random.normal(best_h, 0.1), n_points)
            stagnation_count = 0
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create offspring through crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            if np.random.rand() < crossover_rate:
                # Two-point crossover
                idx1, idx2 = sorted(np.random.choice(n_points, 2, replace=False))
                offspring1 = np.copy(parent1)
                offspring2 = np.copy(parent2)
                offspring1[idx1:idx2] = parent2[idx1:idx2]
                offspring2[idx1:idx2] = parent1[idx1:idx2]
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
                
            # Mutation
            offspring1 = project(mutate(offspring1, mutation_rate, 0.1), n_points)
            offspring2 = project(mutate(offspring2, mutation_rate, 0.1), n_points)
            
            new_population.extend([offspring1, offspring2])
            
        # Ensure we have the right population size
        population = new_population[:population_size]
        
        # Adapt mutation rate
        if stagnation_count > 100:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        elif stagnation_count < 20:
            mutation_rate = max(0.05, mutation_rate * 0.9)
    
    return (list(best_h), best_c5, n_points)

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate a solution with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
    return h
