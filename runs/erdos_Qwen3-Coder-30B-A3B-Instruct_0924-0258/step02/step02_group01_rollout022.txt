import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 50

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Create initial population
    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand, n_points)
        population.append(h_rand)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    generation = 0
    max_generations = int(budget_s / 0.5) if budget_s > 0 else 1000
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Tournament selection
        selected = []
        tournament_size = 5
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Crossover
            if np.random.rand() < 0.8:
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            offspring.extend([child1, child2])
        
        # Evaluate offspring
        offspring_fitness = []
        for h in offspring[:population_size]:
            offspring_fitness.append(evaluate_erdos_solution(h, dx))
        
        # Survival selection (keep best 25% + replace worst with mutated best)
        all_individuals = population + offspring[:population_size]
        all_fitness = fitness + offspring_fitness
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(all_fitness)
        next_generation = [all_individuals[i] for i in sorted_indices[:population_size]]
        next_fitness = [all_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update population and fitness
        population = next_generation
        fitness = next_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        generation += 1
    
    return (list(best_h), best_c5, n_points)
