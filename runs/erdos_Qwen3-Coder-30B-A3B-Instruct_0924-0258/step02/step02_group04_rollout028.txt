import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def latin_hypercube_sampling(n_points, n_samples=50):
    """Generate Latin Hypercube samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        h = np.random.random(n_points)
        # Normalize so that sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        samples.append(h)
    return samples

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def lbfgsb_optimize(h_start, n_points, dx, maxiter=20):
    """Optimize using L-BFGS-B method"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                      constraints=cons, options={'maxiter': maxiter})
        if res.success:
            return project_to_feasible(res.x, n_points)
    except:
        pass
    return h_start

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use multiple resolutions
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Track best overall candidate
    global_best_h = None
    global_best_c5 = float('inf')
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # For small resolutions, do more thorough search
        n_restarts = 15 if n_points <= 30 else 5
        
        for _ in range(n_restarts):
            if time() - start_time > budget_s * 0.95:
                break
            h = initialize_h(n_points)
            h_opt = lbfgsb_optimize(h, n_points, dx, maxiter=20)
            
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            
            # Also try random perturbations for exploration
            for _ in range(3):
                if time() - start_time > budget_s * 0.95:
                    break
                h_pert = perturb_h(h_opt, 0.1)
                h_opt_pert = lbfgsb_optimize(h_pert, n_points, dx, maxiter=10)
                c5_opt_pert = evaluate_erdos_solution(h_opt_pert, dx)
                if c5_opt_pert < best_c5:
                    best_h = h_opt_pert.copy()
                    best_c5 = c5_opt_pert
                    best_n_points = n_points
                    
            # Update global best
            if best_c5 < global_best_c5:
                global_best_c5 = best_c5
                global_best_h = best_h.copy()
    
    # Final optimization on best found solution
    if global_best_h is not None:
        final_h = lbfgsb_optimize(global_best_h, best_n_points, 2.0/best_n_points, maxiter=50)
        final_c5 = evaluate_erdos_solution(final_h, 2.0/best_n_points)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            best_n_points = best_n_points
            
    return (list(best_h), best_c5, best_n_points)
