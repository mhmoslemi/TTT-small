import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    selected = []
    for _ in range(len(population)):
        idxs = np.random.choice(len(population), k)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(population[best_idx])
    return selected

def blend_crossover(parent1, parent2, alpha=0.5):
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, std, len(individual))
    mutated = individual + mask * noise
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    n_points = 35
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        population.append(initial_h)
        n_points = len(initial_h)
    else:
        initial_h = np.ones(n_points) * 0.5
        population.append(initial_h)
    
    # Add random individuals
    for _ in range(pop_size - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    dx = 2.0 / n_points
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolution loop
    gen = 0
    stagnation_count = 0
    prev_best = best_c5
    
    while time() - start_time < budget_s * 0.98:
        gen += 1
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and Mutation
        offspring = []
        for i in range(0, pop_size, 2):
            p1, p2 = selected[i], selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.random() < crossover_rate:
                o1, o2 = blend_crossover(p1, p2)
            else:
                o1, o2 = p1.copy(), p2.copy()
            
            o1 = mutate(o1, mutation_rate)
            o2 = mutate(o2, mutation_rate)
            
            o1 = project_feasible(o1, n_points)
            o2 = project_feasible(o2, n_points)
            
            offspring.extend([o1, o2])
        
        # Evaluate offspring
        offspring_fitnesses = [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Survival: keep best 25% + random perturbations
        all_candidates = population + offspring
        all_fitnesses = fitnesses + offspring_fitnesses
        
        # Sort and select top individuals
        sorted_indices = np.argsort(all_fitnesses)
        surviving_indices = sorted_indices[:pop_size]
        
        population = [all_candidates[i] for i in surviving_indices]
        fitnesses = [all_fitnesses[i] for i in surviving_indices]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Stagnation check
        if stagnation_count > 20:
            # Introduce diversity
            for i in range(pop_size // 4):
                idx = np.random.randint(0, pop_size)
                population[idx] = project_feasible(np.random.random(n_points), n_points)
        
        # Early exit if no improvement
        if abs(prev_best - best_c5) < 1e-6:
            stagnation_count += 1
        else:
            prev_best = best_c5
            
        if stagnation_count > 50:
            break
    
    return (list(best_h), best_c5, n_points)
