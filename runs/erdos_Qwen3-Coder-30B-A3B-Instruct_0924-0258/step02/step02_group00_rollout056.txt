import numpy as np
from time import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Warm start with initial solution
    best_h = h_start.copy()
    best_c5 = None
    best_n_points = n_points
    
    # Feasibility check function
    def check_feasible(h, C, n_points, dx):
        # Create variables
        h_var = cp.Variable(n_points)
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # Convolution constraint
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h_var[i] * (1 - h_var[j])
            constraints.append(corr_sum * dx <= C)
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return True, h_var.value
        except:
            pass
        return False, None
    
    # Binary search on C
    C_low = 0.37
    C_high = 0.38085
    iterations = 40
    best_solution = None
    
    # Binary search loop
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
        C_mid = (C_low + C_high) / 2
        feasible, h_sol = check_feasible(best_h, C_mid, n_points, dx)
        if feasible and h_sol is not None:
            C_high = C_mid
            best_solution = h_sol.copy()
        else:
            C_low = C_mid
    
    # If we found a feasible solution, refine it
    if best_solution is not None:
        # It's already feasible, just compute its actual C5
        h_final = best_solution.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5 = 0.0
        for i in range(n_points):
            for j in range(n_points):
                k = j - i
                if k < 0:
                    k += n_points
                if k >= n_points:
                    k -= n_points
                c5 = max(c5, h_final[i] * (1 - h_final[j]))
        c5 *= dx
        # Return refined solution
        return (list(h_final), c5, n_points)
    
    # Fallback
    c5 = 0.0
    for i in range(n_points):
        for j in range(n_points):
            k = j - i
            if k < 0:
                k += n_points
            if k >= n_points:
                k -= n_points
            c5 = max(c5, best_h[i] * (1 - best_h[j]))
    c5 *= dx
    return (list(best_h), c5, n_points)
