import numpy as np
from time import time

def project_feasible(candidate_h, n_points):
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def tournament_selection(population, fitness, k=5):
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        individual += noise
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = 60
    
    # Initialize population
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        generation += 1
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 25% and randomly mutate worst to fill
        new_population = sorted_population[:population_size//4]
        while len(new_population) < population_size:
            # Select two parents and perform crossover/mutation
            parent1 = tournament_selection(sorted_population, sorted_fitness, tournament_k)
            parent2 = tournament_selection(sorted_population, sorted_fitness, tournament_k)
            
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
                
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Project to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.append(offspring1)
            if len(new_population) < population_size:
                new_population.append(offspring2)
        
        # Evaluate new population
        new_fitness = []
        for h in new_population:
            fit = evaluate_erdos_solution(h, dx)
            new_fitness.append(fit)
        
        # Update best solution
        best_new_idx = np.argmin(new_fitness)
        if new_fitness[best_new_idx] < best_c5:
            best_h = new_population[best_new_idx].copy()
            best_c5 = new_fitness[best_new_idx]
            last_improvement = generation
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early stopping if no improvement
        if stagnation_count > 20:
            break
        
        population = new_population
        fitness = new_fitness
    
    return (list(best_h), best_c5, n_points)
