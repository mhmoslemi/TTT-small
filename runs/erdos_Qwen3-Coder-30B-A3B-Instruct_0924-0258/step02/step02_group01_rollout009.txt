import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary strategy
    population_size = 50
    max_generations = 100
    perturbation_rate = 0.2
    perturbation_std = 0.1
    
    # Generate initial population
    population = [project_to_feasible(h_start.copy(), n_points)]
    for _ in range(population_size - 1):
        h = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        population.append(project_to_feasible(h, n_points))
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # Evolutionary loop
    no_improvement_count = 0
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new population
        new_population = []
        new_fitness = []
        
        for i, h in enumerate(population):
            # Perturb solution
            h_perturbed = h.copy()
            indices = np.random.choice(n_points, int(n_points * perturbation_rate), replace=False)
            h_perturbed[indices] += np.random.normal(0, perturbation_std, len(indices))
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    new_population.append(h_opt)
                    new_fitness.append(c5_opt)
                else:
                    new_population.append(h_perturbed)
                    new_fitness.append(evaluate_erdos_solution(h_perturbed, dx))
            except:
                new_population.append(h_perturbed)
                new_fitness.append(evaluate_erdos_solution(h_perturbed, dx))
                
            # Add original if it's better
            if fitness[i] < new_fitness[-1]:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        # Update population
        combined_pop = new_population
        combined_fitness = new_fitness
        
        # Keep best half of the population
        sorted_indices = np.argsort(combined_fitness)
        top_indices = sorted_indices[:population_size]
        population = [combined_pop[i] for i in top_indices]
        fitness = [combined_fitness[i] for i in top_indices]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 20:
            break
    
    # Refinement at higher resolution
    if n_points < 1000 and time() - start_time < budget_s * 0.9:
        # Interpolate to higher resolution
        higher_res = min(1000, n_points * 2)
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, higher_res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_to_feasible(h_interp, higher_res)
        
        # Local optimization at higher resolution
        dx_high = 2.0 / higher_res
        
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx_high)
            
        bounds = [(0, 1) for _ in range(higher_res)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - higher_res / 2}]
        
        try:
            res = minimize(obj, h_interp, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, higher_res)
                c5_final = evaluate_erdos_solution(h_final, dx_high)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    n_points = higher_res
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
