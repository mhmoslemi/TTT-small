import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin hypercube samples for initial population"""
    samples = []
    for _ in range(n_samples):
        sample = np.zeros(n_points)
        # Distribute points roughly evenly with some noise
        for i in range(n_points):
            base = i / n_points
            noise = (random.random() - 0.5) * 2 / n_points
            sample[i] = max(0, min(1, base + noise))
        samples.append(project_to_feasible(sample, n_points))
    return samples

def local_optimize(h_start, n_points, dx, maxiter=20):
    """Perform local optimization using L-BFGS-B"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                       constraints=cons, options={'maxiter': maxiter})
        if res.success:
            return project_to_feasible(res.x, n_points)
    except:
        pass
    return h_start

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Resolution schedule from coarse to fine
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_to_feasible(h_init, len(h_init))
    else:
        h_init = np.ones(15) * 0.5
    
    # Try multiple resolutions progressively
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Use latin hypercube sampling for diverse starting points
        samples = latin_hypercube_sample(n_points, n_samples=20)
        
        # Evaluate starting sample
        h_start = project_to_feasible(h_init.copy() if n_points == 15 else np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_init)), h_init), n_points)
        samples.append(h_start)
        
        for h_sample in samples:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Local optimization
            h_opt = local_optimize(h_sample, n_points, dx, maxiter=30)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points

    # Final optimization on best solution
    if best_h is not None:
        dx = 2.0 / best_n_points
        final_h = local_optimize(best_h, best_n_points, dx, 100)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
            
    # Return best result or fallback
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = project_to_feasible(np.ones(n_points) * 0.5, n_points)
        best_h = local_optimize(h_start, n_points, dx, 50)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
