import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    # Use Fourier coefficients optimization
    # Represent h using Fourier series with few terms
    n_freq = 10  # Start with small number of frequency components
    n_coeffs = 2 * n_freq + 1  # a0, a1, b1, a2, b2, ...
    
    # Initialize coefficients

        # Use initial h to initialize coefficients
        coeffs = np.fft.rfft(h)
        coeffs = coeffs[:n_coeffs//2 + 1]
        # Make it real
        coeffs = np.concatenate([np.real(coeffs), np.imag(coeffs)[1:-1]])
    else:
        coeffs = np.random.normal(0, 0.1, n_coeffs)
    
    def coeffs_to_h(coeffs):
        # Reconstruct h from Fourier coefficients
        # a0/2 + sum(a_k * cos(k*x) + b_k * sin(k*x))
        h_vals = np.zeros(n_points)
        a0 = coeffs[0]
        for k in range(1, n_freq + 1):
            ak = coeffs[2*k-1]
            bk = coeffs[2*k]
            x_vals = np.linspace(0, 2, n_points)
            h_vals += ak * np.cos(np.pi * k * x_vals) + bk * np.sin(np.pi * k * x_vals)
        h_vals += a0 / 2
        return h_vals
    
    def optimize_coeffs(coeffs, max_iter=1000):
        # Simple gradient descent on coefficients
        learning_rate = 0.01
        best_val = float('inf')
        best_coeffs = coeffs.copy()
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_current = coeffs_to_h(coeffs)
            h_current = project_feasible(h_current)
            current_val = compute_overlap(h_current)
            
            if current_val < best_val:
                best_val = current_val
                best_coeffs = coeffs.copy()
            
            # Compute gradient using finite differences for each coefficient
            grad = np.zeros_like(coeffs)

                coeffs_plus = coeffs.copy()
                coeffs_plus[j] += eps
                h_plus = coeffs_to_h(coeffs_plus)
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                coeffs_minus = coeffs.copy()
                coeffs_minus[j] -= eps
                h_minus = coeffs_to_h(coeffs_minus)
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                grad[j] = (C_plus - C_minus) / (2 * eps)
                
            # Update using gradient descent
            coeffs = coeffs - learning_rate * grad
            coeffs = np.clip(coeffs, -10, 10)  # Clip to prevent explosion
            
        return best_coeffs, best_val

    resolutions = [n_points]
    for i, res in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break

        if i == 0:
            # If having initial_h_values, transform to Fourier space
            try:

                # Initialize with existing values
                current_coeffs = coeffs
            except:
                current_coeffs = np.random.random(n_coeffs) * 0.1
        else:
            # Upsample and interpolate
            h_upsampled = upsample_h(h, res)
            h_upsampled = project_feasible(h_upsampled)
            # Use the upsampled vector to create coefficients
            current_coeffs = np.random.random(n_coeffs) * 0.1
        
        current_coeffs, current_val = optimize_coeffs(current_coeffs, 5000)
        current_h = coeffs_to_h(current_coeffs)
        current_h = project_feasible(current_h)
        
        # Ensure we have the right number of points
        current_h = current_h[:n_points]
        if len(current_h) < n_points:
            current_h = np.pad(current_h, (0, n_points - len(current_h)), 'constant')
        current_h = project_feasible(current_h)
        
        # Keep the best solution found
        best_h = current_h.copy()
        best_val = current_val
    
    # Final optimization of the best solution
    final_coeffs = np.random.normal(0, 0.01, n_coeffs)
    final_coeffs, final_val = optimize_coeffs(final_coeffs, 1000)
    final_h = coeffs_to_h(final_coeffs)
    final_h = project_feasible(final_h)
    final_h = final_h[:n_points]
    if len(final_h) < n_points:
        final_h = np.pad(final_h, (0, n_points - len(final_h)), 'constant')
    final_h = project_feasible(final_h)
    final_val = compute_overlap(final_h)
    
    return (final_h.tolist(), final_val, n_points)
