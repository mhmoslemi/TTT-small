import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h_initial = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Fourier Basis Approach
    def fourier_basis(x, coeffs):
        """Compute Fourier basis functions"""
        a0, *ab_coeffs = coeffs
        n_freqs = (len(ab_coeffs) // 2)
        result = a0 * 0.5  # DC component
        for i in range(n_freqs):
            k = i + 1
            a_k = ab_coeffs[2*i]
            b_k = ab_coeffs[2*i + 1]
            result += a_k * np.cos(2 * np.pi * k * x) + b_k * np.sin(2 * np.pi * k * x)
        return result
    
    def reconstruct_h_from_coeffs(freq_coeffs, n_points):
        """Reconstruct step function h from frequency coefficients"""
        xs = np.linspace(0, 2, n_points)
        h = fourier_basis(xs, freq_coeffs)
        # Clamp to [0, 1] and normalize
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    def compute_fourier_gradient(freq_coeffs, h):
        """Compute gradient of overlap w.r.t. Fourier coefficients"""
        # Approximate gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(freq_coeffs)
        
        for i in range(len(freq_coeffs)):
            coeffs_plus = freq_coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = reconstruct_h_from_coeffs(coeffs_plus, n_points)
            C_plus = compute_overlap(h_plus)
            
            coeffs_minus = freq_coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = reconstruct_h_from_coeffs(coeffs_minus, n_points)
            C_minus = compute_overlap(h_minus)
            
            grad[i] = (C_plus - C_minus) / (2 * eps)
            
        return grad
    
    # Start with a few frequency components
    n_freqs = 5
    n_coeffs = 1 + 2 * n_freqs  # a0 + 2 * n_freqs coefficients
    freq_coeffs = np.random.randn(n_coeffs) * 0.1
    
    # Initialize with fourier transform of the initial solution
    try:
        h_initial = np.array(initial_h_values)
        # Create fourier approximate of initial h
        x = np.linspace(0, 2, n_points)
        freq_coeffs = np.fft.fft(h_initial)[:n_freqs+1].real
        freq_coeffs = np.concatenate([freq_coeffs[:1], freq_coeffs[1:].real, freq_coeffs[1:].imag])
    except NameError:
        # Initialize with small random values
        freq_coeffs = np.random.randn(n_coeffs) * 0.01
    
    # Adam optimizer variables
    m = np.zeros_like(freq_coeffs)
    v = np.zeros_like(freq_coeffs)
    
    # Optimization parameters
    adam_lr = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    adam_epsilon = 1e-8
    
    best_val = float('inf')
    best_coeffs = freq_coeffs.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Reconstruct h from current freq_coeffs
        h = reconstruct_h_from_coeffs(freq_coeffs, n_points)
        
        # Compute current overlap
        current_val = compute_overlap(h)
        
        # Update best solution if better
        if current_val < best_val:
            best_val = current_val
            best_coeffs = freq_coeffs.copy()
            
        # Compute approximate gradient
        grad = compute_fourier_gradient(freq_coeffs, h)
        
        # Adam update step
        m = adam_beta1 * m + (1 - adam_beta1) * grad
        v = adam_beta2 * v + (1 - adam_beta2) * grad**2
        m_hat = m / (1 - adam_beta1**(iter_count + 1))
        v_hat = v / (1 - adam_beta2**(iter_count + 1))
        
        freq_coeffs = freq_coeffs - adam_lr * m_hat / (np.sqrt(v_hat) + adam_epsilon)
        
        iter_count += 1
        
        # Increase resolution if needed
        if iter_count % 5000 == 0 and n_freqs < 15:
            n_freqs += 1
            n_coeffs = 1 + 2 * n_freqs
            # Extend coefficients with zeros
            if len(freq_coeffs) < n_coeffs:
                freq_coeffs = np.pad(freq_coeffs, (0, n_coeffs - len(freq_coeffs)), 'constant')
    
    # Final reconstruction with best coefficients
    final_h = reconstruct_h_from_coeffs(best_coeffs, n_points)
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
