import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a basic symmetric pattern
        h = np.ones(n_points) * 0.5
        # Add a small peak in the middle
        mid_idx = n_points // 2
        h[mid_idx] = 0.8
        h[(mid_idx-1)%n_points] = 0.3
        h[(mid_idx+1)%n_points] = 0.3
    return project_to_feasible(h, n_points)

def fourier_optimize(h_start, n_points, dx, time_limit):
    """Optimize using Fourier domain approach with penalty on spectrum"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Initial penalty weight and learning rate
    penalty_weight = 0.1
    lr = 1e-4
    
    # Fourier transform of initial h
    H = np.fft.fft(h_start)
    
    # Define frequency indices for penalty (excluding DC and Nyquist)
    freq_indices = np.arange(1, n_points//2)
    if n_points % 2 == 0:
        freq_indices = freq_indices[:-1]  # Remove Nyquist if even
    
    iteration = 0
    stagnant = 0
    
    while time() - start_time < time_limit and stagnant < 500:
        # Recompute h from current spectrum
        h = np.real(np.fft.ifft(H))
        h = project_to_feasible(h, n_points)
        
        # Evaluate objective
        c5_current = evaluate_erdos_solution(h, dx)
        if c5_current < c5_best:
            h_best = h.copy()
            c5_best = c5_current
            stagnant = 0
        else:
            stagnant += 1
        
        # Compute spectrum again
        H = np.fft.fft(h)
        
        # Compute gradient penalty on power spectrum
        power_spectrum = np.abs(H)**2
        penalty = np.sum(power_spectrum**2)  # Squared magnitude penalty
        
        # Compute gradient of penalty w.r.t. H
        grad_penalty = 4 * H * power_spectrum
        
        # Update spectrum
        H_new = H - lr * grad_penalty
        
        # Ensure real signal by symmetry for real h
        H = H_new
        H[0] = np.real(H[0])  # DC component
        if n_points % 2 == 0:
            H[n_points//2] = np.real(H[n_points//2])  # Nyquist
        
        for k in range(1, (n_points+1)//2):
            H[-k] = np.conj(H[k])
        
        H = H * (1.0 - penalty_weight * 1e-2)
        
        iteration += 1
        
        # Adaptive adjustments
        if iteration % 10 == 0:
            lr *= 0.99
            penalty_weight *= 1.01
            
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts with different perturbations
        h_restart = h_start.copy()
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = fourier_optimize(h_restart, n_points, dx, time_per_res / 3)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            # Perturb for next restart
            noise = np.random.normal(0, 0.05, len(h_restart))
            h_restart = project_to_feasible(h_restart + noise, len(h_restart))
    
    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = fourier_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
