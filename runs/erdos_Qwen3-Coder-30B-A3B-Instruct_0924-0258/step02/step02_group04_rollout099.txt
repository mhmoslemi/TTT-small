import numpy as np
from time import time
from scipy.fft import fft, ifft

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a flat spectrum
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def compute_spectrum(h):
    """Compute the DFT of h"""
    return fft(h)

def update_spectrum(H, penalty_weight=0.1):
    """Update spectrum to reduce autocorrelation peaks"""
    # Compute |H|^2 and |H|^4
    H_abs_sq = np.abs(H)**2
    H_abs_four = np.abs(H)**4 
    
    # Gradient of penalty term |H|^4
    grad = 4 * penalty_weight * H_abs_four * np.conj(H) / (H_abs_sq + 1e-12)
    
    # Update spectrum (simple gradient descent)
    step_size = 1e-4
    H_new = H - step_size * grad
    
    # Keep real part for real signal spectrum
    H_new = (H_new + np.conj(H_new[::-1])) / 2
    
    return H_new

def spectrum_to_time_domain(H):
    """Convert spectrum to time domain"""
    h = np.real(ifft(H))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize variables
    n_points = 15
    dx = 2.0 / n_points
    h = initialize_h(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Main optimization loop
    iteration = 0
    last_improvement = 0
    penalty_weight = 0.1
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Periodically increase resolution
        if iteration % 200 == 0 and n_points < 75:
            n_points *= 2
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        
        # Compute spectrum
        H = compute_spectrum(h)
        
        # Update spectrum to penalize high autocorrelation
        H_new = update_spectrum(H, penalty_weight)
        
        # Convert back to time domain
        h_new = spectrum_to_time_domain(H_new)
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iteration
        
        # Update current solution
        h = h_new.copy()
        
        # Adjust penalty weight based on progress
        if iteration > 100 and iteration - last_improvement > 50:
            penalty_weight *= 1.1
        
        # Terminate if no improvement for a while
        if iteration - last_improvement > 500:
            break
    
    # Final evaluation of best solution
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
