import numpy as np
from scipy import signal
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    n_points = 35
    dx = 2.0 / n_points
    max_iter = 200
    n_particles = 50
    omega_start = 0.7
    omega_end = 0.4
    c1, c2 = 2.0, 2.0
    lr = 0.001
    
    # Load initial solution if exists
    if 'initial_h_values' in globals():
        h_global_best = np.array(initial_h_values)
        n_points = len(h_global_best)
        dx = 2.0 / n_points
    else:
        h_global_best = np.ones(n_points) * 0.5
    
    h_global_best = project_to_feasible(h_global_best, n_points)
    c5_global_best = evaluate_erdos_solution(h_global_best, dx)
    
    # Multi-resolution improvement
    resolutions = [25, 30, 35]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution if needed
        if res != n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, h_global_best)
            h_global_best = project_to_feasible(h_interp, res)
            n_points = res
            dx = 2.0 / n_points
        
        # Initialize particles
        particles = []
        velocities = []
        p_best = []
        p_best_scores = []
        
        for _ in range(n_particles):
            h = project_to_feasible(h_global_best + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.normal(0, 0.01, n_points))
            score = evaluate_erdos_solution(h, dx)
            p_best.append(h.copy())
            p_best_scores.append(score)
        
        # PSO Loop
        omega = omega_start
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update each particle
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(), np.random.rand()
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (p_best[i] - particles[i]) +
                                c2 * r2 * (h_global_best - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                score = evaluate_erdos_solution(particles[i], dx)
                
                # Update personal best
                if score < p_best_scores[i]:
                    p_best[i] = particles[i].copy()
                    p_best_scores[i] = score
                
                # Update global best
                if score < c5_global_best:
                    h_global_best = particles[i].copy()
                    c5_global_best = score
            
            # Update inertia weight
            omega = omega_end + (omega_start - omega_end) * (1.0 - iteration / max_iter)
        
        # Refinement with gradient descent
        h_refined = h_global_best.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Estimate gradient with finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx)
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[j] = (score_plus - base_score) / eps
            
            # Step
            h_refined -= lr * grad
            h_refined = project_to_feasible(h_refined, n_points)
            
            # Check for improvement
            new_score = evaluate_erdos_solution(h_refined, dx)
            if new_score < c5_global_best:
                c5_global_best = new_score
                h_global_best = h_refined.copy()
    
    return (list(h_global_best), c5_global_best, n_points)
