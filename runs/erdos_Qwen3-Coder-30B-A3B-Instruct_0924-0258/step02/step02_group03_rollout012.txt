import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, hess, trust_radius, target_sum):
    """Perform trust region step, return updated h"""
    n = len(h)
    # Solve trust region subproblem approximately with projected gradient descent
    step = -np.linalg.solve(hess + 1e-8 * np.eye(n), grad)
    
    # Project to feasible region
    step = np.clip(step, -h, 1-h)
    
    # Normalize step if necessary
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    
    new_h = h + step
    new_h = project_to_feasible(new_h, target_sum)
    
    return new_h

    h = project_to_feasible(h, n_points/2.0)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Hessian approximations
    H = np.eye(n_points)
    trust_radius = 0.2
    min_trust_radius = 0.1
    max_trust_radius = 1.0
    improvement_threshold = 1e-4
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Iteration loop
    iteration = 0
    while time.time() - start_time < budget_s:
        # Compute gradient approximation
        # For a one-dim problem, we can compute it via finite differences
        # But since it's a bit complex for full correlations, we'll use the fact
        # that f(h) = max_k int h(x)(1 - h(x+k)) dx. 
        # By approximation, we evaluate gradients using finite differences
        epsilon = 1e-4
        grad = np.zeros(n_points)
        
        # Approximate gradient via central differences
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = project_to_feasible(h_plus, n_points/2.0)
            h_minus = project_to_feasible(h_minus, n_points/2.0)
            f_plus = evaluate(h_plus, dx)
            f_minus = evaluate(h_minus, dx)
            grad[i] = (f_plus - f_minus) / (2*epsilon)
    
        # Perform trust region step
        new_h = trust_region_step(h, grad, H, trust_radius, n_points/2.0)
        new_c5 = evaluate(new_h, dx)
        
        # Accept step if improvement
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
            
            # Update Hessian via BFGS
            s = new_h - h
            y = grad - np.linalg.solve(H + 1e-8 * np.eye(n_points), grad)
            if np.dot(s, y) > 1e-10:
                rho = 1.0 / np.dot(s, y)
                H = H - rho * np.outer(H @ s, y) - rho * np.outer(y, H @ s) + rho * np.outer(y, y) * (np.dot(s, y))
                # Ensure positive definiteness
                H = (H + H.T) / 2.0
                H += 1e-6 * np.eye(n_points)

        if no_improvement_count < 10:
            trust_radius = min(max_trust_radius, trust_radius * 1.2)
        
        # Early exit
        if no_improvement_count >= max_no_improvement:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
