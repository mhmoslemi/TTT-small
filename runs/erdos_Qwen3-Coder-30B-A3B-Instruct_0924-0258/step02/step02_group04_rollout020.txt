import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def fourier_optimize(h_start, n_points, dx, time_limit):
    """Optimize using Fourier domain approach with penalty on spectrum"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Fourier domain optimization with penalty
    max_iter = 2000
    step_size = 1e-4
    penalty_weight = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
        
        # Current solution
        h = h_best.copy()
        
        # Convert to frequency domain
        H = np.fft.fft(h)
        H_squared = np.abs(H)**2
        
        # Compute gradients
        H_grad = np.zeros_like(H, dtype=complex)
        for i in range(len(H)):
            # Gradient of spectrum penalty: -4 * H[i] * |H[i]|^2
            H_grad[i] = -4 * penalty_weight * H[i] * H_squared[i]
            
        # Update spectrum with gradient descent (real update to ensure real signal)
        H_new = H - step_size * H_grad
        
        # Ensure real signal property
        if len(H) % 2 == 0:
            # For even length, symmetry constraint for real signal
            H_new[0] = np.real(H_new[0])  # DC component
            H_new[-1] = np.real(H_new[-1])  # Nyquist component
            for i in range(1, len(H)//2):
                H_new[i] = (H_new[i] + np.conj(H_new[-i]))/2
                H_new[-i] = np.conj(H_new[i])
        else:
            H_new[0] = np.real(H_new[0])  # DC component
            for i in range(1, (len(H)+1)//2):
                H_new[i] = (H_new[i] + np.conj(H_new[-i]))/2
                H_new[-i] = np.conj(H_new[i])
        
        # Inverse transform
        h_new = np.real(np.fft.ifft(H_new))
        
        # Apply feasibility repair
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update best solution
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            
        # Adaptive step size and penalty weight
        if iteration % 100 == 0:
            penalty_weight *= 1.05
            step_size *= 0.98
        
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Resolution schedule
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Allocate time
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Single run
        h_opt, c5_opt = fourier_optimize(h_start, n_points, dx, time_per_res)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
    
    if best_h is None:
        # Fallback
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = fourier_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
