import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.min(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_gradient(h_vec, shift):
        # Compute gradient with respect to minimizing shift
        n = len(h_vec)
        grad = np.zeros(n)
        if shift >= 0:
            # grad[i] += h[i+shift] for i in range(n-shift)
            grad[:-shift] += h_vec[shift:]
        else:
            # grad[i] += h[i-shift] for i in range(-shift, n)
            grad[-shift:] += h_vec[:-shift]
        return grad

    # Initial evaluation
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Optimization loop
    learning_rate = 0.01
    no_improvement_count = 0
    max_no_improvement = 2000
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute all autocorrelations to find minimal
        corr = np.correlate(h, 1 - h, mode='full')
        min_idx = np.argmin(corr)
        shift = min_idx - (len(h) - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient(h, shift)
        
        # Use gradient ascent to increase minimum correlation
        h_new = h + learning_rate * grad
        h_new = project_h(h_new)
        
        # Evaluate new C5
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Reduce learning rate if no improvement
            if no_improvement_count % 100 == 0:
                learning_rate *= 0.98
                
            # Add noise to escape local minima
            noise = 0.0005 * np.random.normal(0, 1, n_points)
            h_noisy = h + noise
            h_noisy = project_h(h_noisy)
            noisy_c5 = compute_c5(h_noisy)
            
            if noisy_c5 < best_c5:
                best_c5 = noisy_c5
                best_h = h_noisy.copy()
                no_improvement_count = 0
        
        # Always update h with new solution if it's better
        if new_c5 < compute_c5(h):
            h = h_new
        else:
            # Occasionally perform random perturbation
            if np.random.random() < 0.05:
                noise = 0.001 * np.random.normal(0, 1, n_points)
                h = h + noise
                h = project_h(h)
        
        # Early termination if too many iterations without improvement
        if no_improvement_count > max_no_improvement:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
