import numpy as np
from scipy import signal
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with available construction or create starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Hyperparameters for PSO
    n_particles = 50
    max_iter = 200
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    inertia_weight = w_start
    
    # Multi-resolution scheme
    resolutions = [25, 30, 35]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Interpolate from previous resolution if needed
        if n_points != len(h_start):
            h_interp = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points)
            
        # Initialize particles
        particles = []
        velocities = []
        pbest_positions = []
        pbest_values = []
        
        for _ in range(n_particles):
            # Initialize particle around current best solution
            if n_points == len(best_h):
                h_particle = project_to_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
            else:
                h_particle = np.random.rand(n_points)
                h_particle = project_to_feasible(h_particle, n_points)
            particles.append(h_particle.copy())
            velocities.append(np.random.rand(n_points) * 0.1)
            pbest_positions.append(h_particle.copy())
            pbest_values.append(evaluate_erdos_solution(h_particle, dx))
        
        # Global best
        gbest_idx = np.argmin(pbest_values)
        gbest_position = pbest_positions[gbest_idx]
        gbest_value = pbest_values[gbest_idx]
        
        # PSO iterations
        for iter_count in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            inertia_weight = w_start - (w_start - w_end) * iter_count / max_iter
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia_weight * velocities[i] +
                                c1 * r1 * (pbest_positions[i] - particles[i]) +
                                c2 * r2 * (gbest_position - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Ensure feasibility
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                current_value = evaluate_erdos_solution(particles[i], dx)
                
                # Update particle best
                if current_value < pbest_values[i]:
                    pbest_positions[i] = particles[i].copy()
                    pbest_values[i] = current_value
                    
                # Update global best
                if current_value < gbest_value:
                    gbest_position = particles[i].copy()
                    gbest_value = current_value
                    
        # Update global best solution
        if gbest_value < best_c5:
            best_h = gbest_position.copy()
            best_c5 = gbest_value
            best_n_points = n_points
            
        h_start = best_h.copy()
        
    # Refine with gradient descent
    h_refined = best_h.copy()
    learning_rate = 0.001
    
    for i in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Approximate gradient using finite differences
        grad = np.zeros_like(h_refined)
        epsilon = 1e-5
        for j in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[j] += epsilon
            h_plus = project_to_feasible(h_plus, best_n_points)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h_refined.copy()
            h_minus[j] -= epsilon
            h_minus = project_to_feasible(h_minus, best_n_points)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
            
        # Update with gradient ascent (equivalent to descent on the negative overlap)
        h_refined -= learning_rate * grad
        h_refined = project_to_feasible(h_refined, best_n_points)
        new_value = evaluate_erdos_solution(h_refined, dx)
        
        if new_value > best_c5:
            h_refined = project_to_feasible(best_h, best_n_points)
            break
            
        best_c5 = new_value
        best_h = h_refined.copy()
        
    return (list(best_h), best_c5, best_n_points)
