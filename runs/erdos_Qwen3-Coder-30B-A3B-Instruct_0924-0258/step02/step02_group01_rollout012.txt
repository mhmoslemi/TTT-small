import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    h = project_feasible(h_start.copy(), n_points)
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    max_iterations = 5000
    restart_interval = 1000
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the maximum overlap lag
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for this specific lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h[j]) - h[i]
        
        grad *= dx
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h = project_feasible(h, n_points)
        
        # Evaluate and track best
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Restart if no improvement for a while
        if iteration % restart_interval == 0 and iteration > 0:
            h = project_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
    
    return (list(best_h), best_c5, n_points)
