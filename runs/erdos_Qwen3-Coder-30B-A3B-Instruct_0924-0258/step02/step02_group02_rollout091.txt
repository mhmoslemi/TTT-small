import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Precompute shifts for correlation calculation
    shifts = np.arange(n_points)
    shift_indices = np.arange(n_points)
    
    # Start optimization
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Use cutting plane approach to optimize
    constraints_list = []
    
    while time.time() - start_time < budget_s - 1:
        # Identify worst case shifts
        n_shifts = min(200, n_points * 2)
        shift_grid = np.linspace(0, 2, n_shifts, endpoint=False)
        max_c = -1
        worst_shift = 0
        
        for s in shift_grid:
            corr = np.correlate(h, 1 - np.roll(h, int(round(s/dx))), mode='full')
            val = np.max(corr) * dx
            if val > max_c:
                max_c = val
                worst_shift = s
        
        # Add constraint for worst shift
        h_var = Variable(n_points)
        eta = Variable()
        
        # Construct constraint for this shift
        s_idx = int(round(worst_shift / dx))
        if s_idx < n_points:
            conv = convolve(h_var, 1 - np.roll(h_var, s_idx), mode='full')
            conv *= dx
            # Constraint for max value
            constraints_list.append(eta >= max(conv))
        else:
            # For large shifts, compute directly
            corr = np.correlate(h_var, 1 - np.roll(h_var, s_idx), mode='full')
            corr *= dx
            constraints_list.append(eta >= max(corr))
        
        # Add bounds
        constraints_list.extend([
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == target_sum
        ])
        
        # Create and solve optimization problem
        objective = Minimize(eta)
        prob = Problem(objective, constraints_list)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
                h = h_new
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    # Add noise
                    h = project_h(h + 0.001 * np.random.randn(n_points))
        except Exception:
            # Fallback if optimization fails
            h = project_h(h)
            
        # Early termination if no improvement in a while
        if compute_c5(h) >= best_c5 - 1e-6:
            pass  # Continue optimization
        
        # Add additional constraint to the list
        pass  # The constraint is added in the optimization loop
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
