import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    return h

    iter_count = 0

    last_improvement = 0

        if candidate_c5 < current_c5:

                best_c5 = candidate_c5
                last_improvement = iter_count
        else:
            # Accept worse solution with probability
            delta = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = candidate_h
                current_c5 = candidate_c5
                accept_count += 1
                
        # Cool down
        T = max(T * alpha, min_T)
        iter_count += 1
        
        # Adaptive cooling based on acceptance rate
        if iter_count % 100 == 0:
            if accept_count < 10 and T > min_T:
                T = min(T * 1.1, 0.5)  # Increase temp if too strict
            elif accept_count > 50:
                T = max(T * 0.95, min_T)  # Decrease temp if too flexible

        # Reset temperature if no improvement in a while
        if iter_count - last_improvement > 500:
            T = min(T * 1.5, 0.5)
            last_improvement = iter_count
