import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def objective_with_smooth_max(h, n_points, dx):
    """Objective using smooth max to approximate the maximum"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Smooth max of the correlation values
    smooth_max_val = smooth_max(corr, alpha=100)
    return smooth_max_val

def gradient_smooth_max(h, n_points, dx, eps=1e-8):
    """Gradient of the smooth max approximation using finite differences"""
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    grad_corr = alpha * exp_corr / sum_exp
    
    # Gradient of correlation w.r.t h using finite differences
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = np.max(corr_plus) - np.max(corr)
    return grad / eps

    decay_rate = 0.9
    decay_steps = 50
    iteration = 0
    
    while time() - start_time < time_limit:
        iteration += 1

            def obj(h):
                return objective_with_smooth_max(h, n_points, dx)

            obj_val = obj(h_proj)

            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)

                best_h = h_new.copy()
                best_c5 = c5_new
                
            h_start = h_new
            if iteration % decay_steps == 0:
                lr *= decay_rate
        except Exception:
            # If optimization fails, continue with the best solution
            break
            
    return best_h, best_c5
