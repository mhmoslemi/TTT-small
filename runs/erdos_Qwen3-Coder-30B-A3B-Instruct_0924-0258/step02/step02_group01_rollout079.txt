import numpy as np
from time import time

def project_feasible(candidate_h, n_points):
    """Project candidate_h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def tournament_selection(population, fitnesses, k=5):
    """Select an individual via tournament selection"""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, noise_std=0.1):
    """Apply mutation to an individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(individual))
        mutated = individual + noise
        return mutated
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    time_limit = budget_s * 0.95
    
    # Initialize population
    n_points = 35
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        initial_h = np.ones(n_points) * 0.5
    
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    
    # Create initial population
    population = []
    population.append(initial_h)
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        fitness = evaluate_erdos_solution(h, 2.0 / n_points)
        fitnesses.append(fitness)
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    gen = 0
    stagnation_count = 0
    max_stagnation = 20
    
    while time() - start_time < time_limit:
        gen += 1
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top 25% elites and make new population
        new_population = sorted_population[:pop_size // 4]
        new_fitnesses = sorted_fitnesses[:pop_size // 4]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection and crossover
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
                
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Feasibility projection
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            # Add to new population
            new_population.extend([offspring1, offspring2])
            new_fitnesses.extend([evaluate_erdos_solution(offspring1, 2.0 / n_points),
                                 evaluate_erdos_solution(offspring2, 2.0 / n_points)])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        fitnesses = new_fitnesses[:pop_size]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early stopping if stagnation
        if stagnation_count >= max_stagnation:
            break
            
        # Occasionally diversify by adding random individuals
        if gen % 10 == 0:
            for i in range(pop_size // 4):
                if len(population) < pop_size:
                    h = np.random.rand(n_points)
                    h = project_feasible(h, n_points)
                    population.append(h)
                    fitness = evaluate_erdos_solution(h, 2.0 / n_points)
                    fitnesses.append(fitness)
    
    return (list(best_h), best_c5, n_points)
