import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Determine resolution and initial values
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values)
    else:
        h_best = np.ones(n_points) * 0.5
    
    # Since we're seeking a better bound, prefer the best so far
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Binary search for optimal C5
    c_low = 0.37
    c_high = 0.38085
    best_feasible_h = h_best.copy()
    best_feasible_c5 = c5_best
    
    # Binary search iterations (about 40 for good resolution)
    binary_search_iterations = 0
    max_binary_search = 40
    
    while binary_search_iterations < max_binary_search:
        if time() - start_time > budget_s * 0.9:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # CVXPY setup for feasibility check
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add all the overlap constraints
        # For circular indices
        for k in range(-n_points + 1, n_points):
            # Compute sum over all i: h[i] * (1 - h[(i + k) % n])
            # Use CVXPY to construct sum efficiently
            sum_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                sum_expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(sum_expr <= c_mid * n_points / 2)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_feasible = h_vars.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_feasible_c5:
                    best_feasible_h = h_feasible.copy()
                    best_feasible_c5 = c5_feasible
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
        
        binary_search_iterations += 1
        
    # Optionally refine the best point found with a local optimizer
    h_refined = best_feasible_h.copy()
    dx_refined = 2.0 / n_points
    c5_refined = best_feasible_c5
    
    refinements_left = 3
    last_improvement = 0
    while refinements_left > 0 and time() - start_time < budget_s * 0.95:
        # Define objective function for local refinement
        def obj(h):
            # Prevent infeasible solutions from being accepted
            if np.any(h < 0) or np.any(h > 1):
                return 1e10
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-6:
                return 1e10
            corr = np.correlate(h, 1 - h, mode='full')
            return -np.max(corr) * dx_refined  # Negative for maximization via minimization
        
        # Define constraint function for clipping
        def constraint_func(h):
            return np.sum(h) - n_points / 2
        
        # Constraints for minimization
        cons = [{'type': 'eq', 'fun': constraint_func}]
        bounds = [(0, 1) for _ in range(n_points)]

        try:
            # Use SLSQP for robust optimization with constraints
            res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 20, 'ftol': 1e-6, 'gtol': 1e-6})
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_sum = np.sum(h_new)
                h_new = h_new * (n_points / 2) / h_sum  # Re-scale appropriately
                c5_attempt = evaluate_erdos_solution(h_new, dx_refined)
                if c5_attempt < c5_refined:
                    h_refined = h_new
                    c5_refined = c5_attempt
                    last_improvement = 0
                else:
                    last_improvement += 1
            else:
                last_improvement += 1
        except:
            last_improvement += 1
            
        refinements_left -= 1
        if last_improvement > 1:
            break

    # Final validation and return
    c5_result = evaluate_erdos_solution(h_refined, dx_refined)
    if c5_result < best_feasible_c5:
        best_feasible_c5 = c5_result
        best_feasible_h = h_refined.copy()
        
    return (list(best_feasible_h), best_feasible_c5, n_points)
