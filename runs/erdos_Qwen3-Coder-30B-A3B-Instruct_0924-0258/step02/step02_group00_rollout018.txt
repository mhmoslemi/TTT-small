import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, tol=1e-6):
    """Frank-Wolfe optimization for the Erdős problem"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        # Compute all overlaps and find the maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute subgradient at maximum overlap index
        # For overlap at shift k, subgradient dC/dh[i] = (1 - h[(i-k)%n]) - h[i]
        # Here we represent this as a sparse structure for efficiency
        subgradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i - max_k) % n_points
            subgradient[i] = (1 - h[j]) - h[i]
            
        # Minimize g^T s subject to sum(s) = 0, 0 <= s <= 1
        # Linear programming problem
        # We solve this via projected gradient descent over feasible set
        # The optimal solution is at the extremal point of set, so
        # s = argmin_{s} g^T s subject to sum(s) = 0, 0 <= s <= 1
        # Projected steepest descent step to solve
        # s = argmin_s g^T s such that s is feasible
        
        # The solution to this is simple: solve with dual gradient step
        # s = argmax_s -g^T s subject to sum(s) = 0, and 0 <= s <= 1
        # This is equivalent to:
        # min_{s: sum(s) = 0, 0 <= s <= 1} g^T s = min_{s: sum(s) = 0} g^T s + indicator(s in [0,1])
        # This requires projecting g onto the feasible set of sum=0, [0,1]
        # Instead, we compute s directly as a vector that makes g^T s smallest
        
        # As a heuristic, let's take the negative of subgradient and project it
        # to have zero sum and lie in [0,1]
        s = -subgradient
        # Project to zero-sum
        s -= np.mean(s)
        # Project to [0,1]
        s = np.clip(s, 0, 1)
        
        # Line search to find suitable step size
        # Use backtracking line search for gamma in [0.1, 0.9]
        best_gamma = 0.1
        for gamma in np.linspace(0.1, 0.9, 9):
            new_h = h + gamma * (s - h)
            new_h = project_to_feasible(new_h, n_points)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                best_gamma = gamma
        
        h = best_h.copy()
        if iteration % 10 == 0:
            if abs(evaluate_erdos_solution(h, dx) - best_c5) < tol:
                break
                
    return h, best_c5

    best_n_points = n_points
    
    # Try different resolutions
    resolutions = [25, 30, 35]
    
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try),

            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Run Frank-Wolfe optimization with time constraints  
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 120)

        for restart in range(3):
            if time() - start_time > budget_s * 0.95:

            # Small perturbation around current best
            h_restart = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            h_opt, c5_opt = frank_wolfe_step(h_restart, n_points, dx_try, max_iter=30)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Final refinement with larger iterations at finest resolution
    if time() - start_time < budget_s * 0.9:
        n_points_final = max(35, best_n_points)

        h_refined, c5_refined = frank_wolfe_step(best_h, n_points_final, dx_final, max_iter=40)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
    return (list(best_h), best_c5, best_n_points)
