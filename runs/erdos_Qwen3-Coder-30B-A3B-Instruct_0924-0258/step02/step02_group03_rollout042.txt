import numpy as np
import time
from scipy.signal import correlate

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize to correct sum
    h0 = project_to_feasible(h0, n_points / 2.0)
    
    # MSPF parameters
    n_particles = 100
    max_iter = 1000
    temperature = 1.0
    proposal_std = 0.1
    acceptance_rate_target = 0.3
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Initialize particles
    particles = [h0.copy()]
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Importance sampling
        weights = []
        particle_values = []
        
        for particle in particles:
            h_new = particle + np.random.normal(0, proposal_std, n_points)
            h_new = project_to_feasible(h_new, n_points/2.0)
            value = evaluate(h_new, 2.0/n_points)
            # Importance weight: exp(-C5/temperature)
            weight = np.exp(-value / temperature) if temperature > 0 else 0
            particle_values.append(value)
            weights.append(weight)
        
        # Normalize weights
        weights = np.array(weights)
        total_weight = np.sum(weights)
        if total_weight > 0:
            weights = weights / total_weight
        else:
            weights = np.ones(len(weights)) / len(weights)
        
        # Resampling (select top 50%)
        indices = np.random.choice(len(particles), size=len(particles)//2, 
                                  p=weights, replace=True)
        selected_particles = [particles[i] for i in indices]
        selected_values = [particle_values[i] for i in indices]
        
        # Mutation: 2 mutations per particle
        new_particles = []
        new_values = []
        
        for p_idx, particle in enumerate(selected_particles):
            for _ in range(2):
                # Global mutation
                mutation = np.random.normal(0, proposal_std * 2.0, n_points)
                h_mutated = particle + mutation
                h_mutated = project_to_feasible(h_mutated, n_points/2.0)
                val = evaluate(h_mutated, 2.0/n_points)
                new_particles.append(h_mutated)
                new_values.append(val)
        
        # Replace bottom 40% particles with mutations
        # Sort particles by value (best first)
        combined_particles = selected_particles + new_particles
        combined_values = selected_values + new_values
        combined_indices = np.argsort(combined_values)
        top_10 = combined_indices[:len(combined_indices)//10]
        bottom_40 = combined_indices[-len(combined_indices)*4//10:]
        
        # Update particles with top 10 and mutated particles replacing bottom 40
        particles = [combined_particles[i] for i in top_10]
        replacement_indices = [i for i in combined_indices if i not in top_10][:len(bottom_40)]
        new_particles_list = [combined_particles[i] for i in replacement_indices]
        particles.extend(new_particles_list)
        
        # Update best solution
        min_val = min(combined_values)
        if min_val < best_c5:
            best_c5 = min_val
            best_h = combined_particles[np.argmin(combined_values)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt temperature and noise
        temperature *= 0.99
        if no_improvement_count > 10:
            # Increase noise to escape local minima
            proposal_std *= 1.1
        elif no_improvement_count == 0:
            # Decrease noise as we improve
            proposal_std *= 0.95
            
        # Adjust acceptance rate
        if iteration % 20 == 0:
            current_acceptance = 0.3  # Estimate based on history
            if current_acceptance < acceptance_rate_target:
                proposal_std *= 0.95
            elif current_acceptance > acceptance_rate_target:
                proposal_std *= 1.05
                
    # Return final solution
    if best_h is not None:
        final_c5 = evaluate(best_h, 2.0/n_points)
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h0, n_points/2.0)
        final_c5 = evaluate(final_h, 2.0/n_points)
        return (final_h.tolist(), float(final_c5), n_points)
