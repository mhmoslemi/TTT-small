import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation for upsample
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    # Fourier approach with small number of components
    N_freqs = 8  # Number of frequency components to optimize
    # Initial parameters: a0, a1, b1, a2, b2, ...
    init_params = np.random.randn(2 * N_freqs + 1) * 0.1
    # Add small initialization to avoid degenerate cases
    if 'initial_h_values' in globals():
        # Use FFT of initial construction as starting point
        h_fft = np.fft.fft(initial_h_values)
        init_params[0] = np.real(h_fft[0])  # DC component
        for i in range(N_freqs):
            if i < len(h_fft)//2:
                init_params[2*i+1] = np.real(h_fft[i+1])
                init_params[2*i+2] = np.imag(h_fft[i+1])
    
    # Objective function in frequency domain
    def fourier_to_h(freq_params):
        a0 = freq_params[0]
        a_vals = freq_params[1::2]
        b_vals = freq_params[2::2]
        
        x_points = np.linspace(0, 2, n_points)
        h_vals = a0 * 0.5 * np.ones_like(x_points)
        for i in range(N_freqs):
            h_vals += a_vals[i] * np.cos(2 * np.pi * (i+1) * x_points) + \
                      b_vals[i] * np.sin(2 * np.pi * (i+1) * x_points)
        return h_vals
    
    def fourier_objective(freq_params):
        h_vals = fourier_to_h(freq_params)
        h_vals = project_feasible(h_vals)
        return compute_overlap(h_vals)
    
    # Adam optimizer for frequency parameters
    adam_lr = 0.02
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    
    # Initialize Adam variables
    m = np.zeros_like(init_params)
    v = np.zeros_like(init_params)
    
    # Optimization loop
    best_obj = float('inf')
    best_params = init_params.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current h and C5
        h_curr = fourier_to_h(init_params)
        h_curr = project_feasible(h_curr)
        current_obj = compute_overlap(h_curr)
        
        if current_obj < best_obj:
            best_obj = current_obj
            best_params = init_params.copy()
            
        # Gradient estimation with finite differences
        grad = np.zeros_like(init_params)
        eps = 1e-4
        for i in range(len(init_params)):
            params_plus = init_params.copy()
            params_plus[i] += eps
            h_plus = fourier_to_h(params_plus)
            h_plus = project_feasible(h_plus)
            C_plus = compute_overlap(h_plus)
            
            params_minus = init_params.copy()
            params_minus[i] -= eps
            h_minus = fourier_to_h(params_minus)
            h_minus = project_feasible(h_minus)
            C_minus = compute_overlap(h_minus)
            
            grad[i] = (C_plus - C_minus) / (2 * eps)
        
        # Adam update
        m = adam_beta1 * m + (1 - adam_beta1) * grad
        v = adam_beta2 * v + (1 - adam_beta2) * grad**2
        m_hat = m / (1 - adam_beta1**(iter_count + 1))
        v_hat = v / (1 - adam_beta2**(iter_count + 1))
        init_params = init_params - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        iter_count += 1
        
        # Occasionally print progress
        if iter_count % 500 == 0:
            pass  # Debugging info

    # Final conversion from parameters to step function h
    final_h = fourier_to_h(best_params)
    final_h = project_feasible(final_h)
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
