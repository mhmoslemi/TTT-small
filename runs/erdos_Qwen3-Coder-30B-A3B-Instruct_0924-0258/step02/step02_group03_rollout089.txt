import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, trust_radius, target_sum, dx):
    """Perform a trust region step with gradient information."""
    # Simple gradient descent with trust region
    step = -grad * 0.01
    # Project to trust region
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        step = step * trust_radius / step_norm
    
    # Compute new point and project to feasible region
    h_new = h + step
    h_new = project_to_feasible(h_new, target_sum)
    
    return h_new

    corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(corr)
    # It's a bit tricky to compute the exact gradient analytically for this discrete case
    # Instead we use finite difference to estimate the gradient components
    
    # As a proxy, we compute each component's contribution to correlation
    grad = np.zeros_like(h)
    
    # Approximate gradient by finite difference
    epsilon = 1e-4

        corr_plus = correlate(temp_h, 1 - temp_h, mode="full")
        grad[i] = (corr_plus[max_k] - corr[max_k]) / epsilon
    
    return grad * dx
