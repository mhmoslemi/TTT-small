import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin Hypercube samples for diverse initializations"""
    samples = []
    for _ in range(n_samples):
        sample = np.random.rand(n_points)
        # Shuffle each dimension
        for i in range(n_points):
            sample[i] = (sample[i] + i) / n_points
        # Random permutation
        np.random.shuffle(sample)
        samples.append(sample)
    return samples

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def optimize_single_h(h_start, n_points, dx, time_limit):
    """Optimize a single h using L-BFGS-B"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 50
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Perturb and optimize
        h_perturbed = perturb_h(h_best, 0.1)
        
        # Optimization with bounds and sum constraint
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Schedule resolutions
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Early optimization on coarse grids
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Multiple restarts
        for i in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Generate sample using Latin Hypercube
            samples = latin_hypercube_sample(n_points, 1)
            h_sample = samples[0]
            
            # Try with both starting points
            for h_init in [h_start, h_sample]:
                remaining_time = budget_s - (time() - start_time)
                max_time_per_opt = min(remaining_time / 10, 50)
                
                h_opt, c5_opt = optimize_single_h(h_init, n_points, dx, max_time_per_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
    
    # Refine best solution with more iterations
    if best_h is not None and time() - start_time < budget_s * 0.9:
        dx = 2.0 / best_n_points
        remaining_time = budget_s - (time() - start_time)
        h_refined, c5_refined = optimize_single_h(best_h, best_n_points, dx, remaining_time * 0.8)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Final validation
    if best_h is not None:
        dx = 2.0 / best_n_points
        validated_c5 = evaluate_erdos_solution(best_h, dx)
        best_c5 = validated_c5
    
    # Fallback if nothing found
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_single_h(h_start, n_points, dx, budget_s)
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
