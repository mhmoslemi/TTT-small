import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

def trust_region_update(h, grad, hessian, trust_radius, target_sum):
    n = len(h)
    # BFGS update for Hessian approximation
    hessian = np.eye(n)
    # Simple trust region subproblem: minimize grad^T * delta + 0.5 * delta^T * H * delta
    # subject to ||delta|| <= trust_radius
    # Using a simple projected gradient approach
    delta = -np.dot(hessian, grad)
    norm_delta = np.linalg.norm(delta)
    if norm_delta > trust_radius:
        delta = delta * trust_radius / norm_delta
    
    new_h = h + delta
    new_h = project_to_feasible(new_h, target_sum)
    return new_h, delta

    h = h0.copy()

    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    trust_radius = 0.2
    min_trust_radius = 0.1
    max_trust_radius = 1.0
    no_improvement_count = 0
    max_no_improvement = 100
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Gradient approximation (centered difference)
        grad = np.zeros(n_points)
        current_c5 = evaluate(h, dx)
        
        # Estimate gradient with finite differences
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points / 2.0)
            c5_plus = evaluate(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, n_points / 2.0)
            c5_minus = evaluate(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with trust region
        new_h, delta = trust_region_update(h, grad, np.eye(n_points), trust_radius, n_points / 2.0)

    # Final evaluation with best solution
    final_c5 = evaluate(best_h, dx)
