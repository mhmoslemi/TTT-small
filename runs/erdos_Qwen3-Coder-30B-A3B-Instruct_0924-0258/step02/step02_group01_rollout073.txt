import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

            population.append(h)
    
    # Evaluate initial population
    fitness = []

        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic algorithm loop
    while time() - start_time < budget_s * 0.95:
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_k, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover (arithmetic mean)
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                # Clone parents
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child1 += noise
                child1 = project_feasible(child1, n_points)
            
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_strength, n_points)
                child2 += noise
                child2 = project_feasible(child2, n_points)
                
            offspring.extend([child1, child2])
        
        # Evaluate offspring
        offspring_fitness = []
        for h in offspring[:population_size]:
            offspring_fitness.append(evaluate_erdos_solution(h, dx))

        all_fitness = fitness + offspring_fitness[:population_size]

        best_indices = np.argsort(all_fitness)[:population_size]

        fitness = [all_fitness[i] for i in best_indices]

            best_h = population[current_best_idx].copy()
    
    return (list(best_h), best_c5, n_points)
