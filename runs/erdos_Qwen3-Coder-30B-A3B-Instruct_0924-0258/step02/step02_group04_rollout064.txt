import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def generate_simplex(n_points, m):
    """Generate m+1 points forming a simplex"""
    if 'initial_h_values' in globals():
        # Start with the initial construction
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        # Fallback to uniform distribution
        h0 = np.ones(n_points) * 0.5
        
    h0 = project_to_feasible(h0, n_points)
    simplex = [h0]
    
    # Generate additional points by perturbing along random directions
    for _ in range(m - 1):
        # Generate a random direction
        direction = np.random.normal(0, 0.2, n_points)
        # Perturb h0
        h_perturbed = h0 + direction
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        simplex.append(h_perturbed)
        
    return simplex

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    swarm_size = 10
    max_iter = 2000
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    
    # Generate initial simplex
    simplex = generate_simplex(n_points, swarm_size)
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Evaluate all particles
    fitness = [evaluate_erdos_solution(particle, dx) for particle in simplex]
    
    # Find initial best
    best_idx = np.argmin(fitness)
    gbest = simplex[best_idx].copy()
    gbest_fitness = fitness[best_idx]
    
    # Store personal bests
    pbest = [particle.copy() for particle in simplex]
    pbest_fitness = fitness[:]
    
    iter_no_improve = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions for each particle
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                             cognitive_weight * r1 * (pbest[i] - simplex[i]) + 
                             social_weight * r2 * (gbest - simplex[i]))
            
            simplex[i] += velocities[i]
            
            # Project to feasible region
            simplex[i] = project_to_feasible(simplex[i], n_points)
            
            # Evaluate fitness
            new_fitness = evaluate_erdos_solution(simplex[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = simplex[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = simplex[i].copy()
                    gbest_fitness = new_fitness
                    iter_no_improve = 0
            else:
                iter_no_improve += 1
        
        # Apply shaking if no improvement for too long
        if iter_no_improve > shaking_threshold:
            for i in range(swarm_size):
                # Add noise to velocity and position
                noise = np.random.normal(0, 0.05, n_points)
                velocities[i] += noise
                simplex[i] += noise
                simplex[i] = project_to_feasible(simplex[i], n_points)
            iter_no_improve = 0
        
        # Apply local refinement periodically
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Refine top 5 particles with L-BFGS-B
            top_indices = np.argsort(pbest_fitness)[:min(5, len(pbest_fitness))]
            for i in top_indices:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
                
                try:
                    res = minimize(obj, pbest[i], method='L-BFGS-B', 
                                  bounds=bounds, constraints=cons, 
                                  options={'maxiter': 10, 'ftol': 1e-6})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_fitness = evaluate_erdos_solution(refined, dx)
                        if refined_fitness < pbest_fitness[i]:
                            pbest[i] = refined
                            pbest_fitness[i] = refined_fitness
                except:
                    pass
    
    # Final refinement of the global best
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_fitness = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_fitness, n_points)
