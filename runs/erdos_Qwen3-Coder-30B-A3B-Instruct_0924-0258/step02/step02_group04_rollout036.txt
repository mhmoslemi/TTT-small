import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def simplex_init(n_points, n_particles):
    """Initialize particles using simplex method"""
    particles = []
    base = initialize_h(n_points)
    particles.append(base.copy())
    
    # Generate random directions from base
    for i in range(n_particles - 1):
        pert = np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(base + pert, n_points)
        particles.append(h)
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia_w = 0.7
    c1 = c2 = 2.0
    shake_threshold = 50
    lbfgs_interval = 200
    max_shake = 50
    
    # Initialize particles using simplex
    particles = simplex_init(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    
    # Evaluate initial particles
    fitness = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Track best positions
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    no_improvement = 0
    iteration = 0
    
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        iteration += 1
        
        # PSO update
        r1, r2 = np.random.rand(2, swarm_size, n_points)
        velocities = (inertia_w * velocities + 
                     c1 * r1 * (pbest - particles) + 
                     c2 * r2 * (gbest - particles))
        
        particles += velocities
        
        # Project all particles to feasible region
        for i in range(swarm_size):
            particles[i] = project_to_feasible(particles[i], n_points)
        
        # Re-evaluate fitness
        fitness = np.array([evaluate_erdos_solution(p, dx) for p in particles])
        
        # Update personal best
        improved = fitness < pbest_fitness
        pbest[improved] = particles[improved]
        pbest_fitness[improved] = fitness[improved]
        
        # Update global best
        new_gbest = np.argmin(fitness)
        if fitness[new_gbest] < gbest_fitness:
            gbest = particles[new_gbest].copy()
            gbest_fitness = fitness[new_gbest]
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Shaking mechanism
        if no_improvement >= shake_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            no_improvement = 0
            
        # L-BFGS refinement
        if iteration % lbfgs_interval == 0:
            # Refine top 10 particles
            top_indices = np.argsort(fitness)[:10]
            for idx in top_indices:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, particles[idx], method='L-BFGS-B', 
                                 bounds=bounds, options={'maxiter': 10})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        new_fitness = evaluate_erdos_solution(refined, dx)
                        if new_fitness < fitness[idx]:
                            particles[idx] = refined
                            fitness[idx] = new_fitness
                            
                            if new_fitness < pbest_fitness[idx]:
                                pbest[idx] = refined
                                pbest_fitness[idx] = new_fitness
                                
                            if new_fitness < gbest_fitness:
                                gbest = refined.copy()
                                gbest_fitness = new_fitness
                except:
                    pass
    
    # Final refinement
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, options={'maxiter': 20})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_fitness = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_fitness, n_points)
