import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Some exploration parameters
    num_candidates = 50
    num_iterations = 100
    perturb_rate = 0.2
    perturb_std = 0.1
    max_no_improve = 20
    
    # Generate initial population
    population = [h_start.copy()]
    for _ in range(num_candidates - 1):
        h_rand = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        population.append(h_rand)
    
    no_improve_count = 0
    
    for iteration in range(num_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        improved = False
        
        # Evaluate all candidates
        candidate_scores = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Find best in population
        best_idx = np.argmin(candidate_scores)
        if candidate_scores[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = candidate_scores[best_idx]
            improved = True
        
        if not improved:
            no_improve_count += 1
        else:
            no_improve_count = 0
        
        if no_improve_count >= max_no_improve:
            break
            
        # Create new population via perturbation and local optimization
        new_population = []
        for h in population:
            # Perturb
            h_perturbed = h.copy()
            indices = np.random.choice(n_points, int(perturb_rate * n_points), replace=False)
            h_perturbed[indices] += np.random.normal(0, perturb_std, len(indices))
            
            # Project
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    new_population.append(h_opt)
                else:
                    new_population.append(h_perturbed)
            except:
                new_population.append(h_perturbed)
        
        population = new_population
        
    # Refinement step - increase resolution
    current_h = best_h.copy()
    current_n = n_points
    current_dx = dx
    
    while current_n < 1000 and (time() - start_time) < budget_s * 0.98:
        new_n = min(current_n * 2, 1000)
        new_x = np.linspace(0, 2, new_n)
        old_x = np.linspace(0, 2, current_n)
        new_h = np.interp(new_x, old_x, current_h)
        new_h = project_to_feasible(new_h, new_n)
        
        # Local optimization at new resolution
        dx_new = 2.0 / new_n
        def obj(h):
            return evaluate_erdos_solution(h, dx_new)
        
        bounds = [(0, 1) for _ in range(new_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - new_n / 2}]
        
        try:
            res = minimize(obj, new_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                new_h = project_to_feasible(res.x, new_n)
                new_c5 = evaluate_erdos_solution(new_h, dx_new)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    current_h = new_h
                    current_n = new_n
                    current_dx = dx_new
                else:
                    break
            else:
                break
        except:
            break
    
    return (list(best_h), best_c5, current_n)
