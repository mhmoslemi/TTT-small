import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute all correlations with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum and scale by dx
        return np.max(corr) * dx

    def project_h(h_vec):
        # Clip values to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to match target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic algorithm parameters
    pop_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h_init.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    # Store best result
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Main GA loop
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness of entire population
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(-c5)  # Negative for minimization
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Mutate 1 or 2 positions
                n_mutations = np.random.choice([1, 2])
                mutation_indices = np.random.choice(n_points, n_mutations, replace=False)
                for idx in mutation_indices:
                    child1[idx] = np.random.rand()
            if np.random.rand() < mutation_rate:
                n_mutations = np.random.choice([1, 2])
                mutation_indices = np.random.choice(n_points, n_mutations, replace=False)
                for idx in mutation_indices:
                    child2[idx] = np.random.rand()
            
            # Project to valid space
            offspring.append(project_h(child1))
            offspring.append(project_h(child2))
        
        # Replace lowest fitness half with offspring
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        for i in range(pop_size // 2):
            population[sorted_indices[i]] = offspring[i]
        
        # Update best
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
