import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    else:
        h = np.ones_like(h) * (target_sum / len(h))
    
    return h

    swarm_size = 40
    max_iter = 10000
    w = 0.7  # Inertia weight
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    v_max = 0.05  # Max velocity as percentage of range
    
    # Initialize swarm
    particles = []
    velocities = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            # Perturb the initial solution
            h = h0 + np.random.normal(0, 0.02, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.02, 0.02, n_points))
    
    # Track best
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate(p, 2.0/n_points) for p in particles]
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            vel_range = np.max(particles[i]) - np.min(particles[i])
            if vel_range > 0:
                velocities[i] = np.clip(velocities[i], -v_max * vel_range, v_max * vel_range)
            else:
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points / 2.0)
            
            # Evaluate
            current_value = evaluate(particles[i], 2.0/n_points)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
    
    # Return best solution found
    return (gbest.tolist(), float(gbest_value), n_points)
