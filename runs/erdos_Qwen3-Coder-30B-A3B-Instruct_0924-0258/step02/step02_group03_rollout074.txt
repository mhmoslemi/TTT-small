import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    current_c5 = evaluate(current_h, 2.0/n_points)

    best_c5 = current_c5
    
    # Simulated Annealing parameters
    initial_temp = 0.5
    cooling_rate = 0.995
    temp = initial_temp
    iteration = 0
    max_iter = 50000  # Prevent infinite loops
    
    # Track acceptance rate
    acceptance_count = 0
    total_attempts = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        # Generate candidate
        candidate_h = current_h + np.random.normal(0, temp, n_points)

        candidate_c5 = evaluate(candidate_h, 2.0/n_points)
        
        # Accept or reject
        total_attempts += 1
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-(candidate_c5 - current_c5) / temp):
            current_h = candidate_h
            current_c5 = candidate_c5
            acceptance_count += 1
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temp = max(temp * cooling_rate, 1e-6)
        
        # Adapt temperature based on acceptance rate
        if iteration % 50 == 0 and total_attempts > 0:
            acceptance_rate = acceptance_count / total_attempts
            if acceptance_rate < 0.2:
                temp *= 0.9
            elif acceptance_rate > 0.5:
                temp *= 1.1
            acceptance_count = 0
            total_attempts = 0
    
    return (best_h.tolist(), float(best_c5), n_points)
