import numpy as np
from scipy import signal
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    inertia_weight_start = 0.7
    inertia_weight_end = 0.4
    cognitive_coeff = 2.0
    social_coeff = 2.0
    
    # Multi-resolution sweep
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Initialize particles
        particles = []
        velocities = []
        pbest_positions = []
        pbest_values = []
        
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            pbest_positions.append(h.copy())
            pbest_values.append(evaluate_erdos_solution(h, dx_try))
            
        # Global best
        best_pos_idx = np.argmin(pbest_values)
        gbest_position = pbest_positions[best_pos_idx].copy()
        gbest_value = pbest_values[best_pos_idx]
        
        # PSO iterations
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update inertia weight
            inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (iteration / max_iter)
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                cognitive = cognitive_coeff * r1 * (pbest_positions[i] - particles[i])
                social = social_coeff * r2 * (gbest_position - particles[i])
                velocities[i] = inertia_weight * velocities[i] + cognitive + social
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                current_value = evaluate_erdos_solution(particles[i], dx_try)
                
                # Update personal best
                if current_value < pbest_values[i]:
                    pbest_positions[i] = particles[i].copy()
                    pbest_values[i] = current_value
                    
                # Update global best
                if current_value < gbest_value:
                    gbest_position = particles[i].copy()
                    gbest_value = current_value
            
            # Update best solution
            if gbest_value < best_c5:
                best_h = gbest_position.copy()
                best_c5 = gbest_value
                best_n_points = n_points
                
        h_start = best_h.copy()
    
    # Refinement with gradient descent
    refinement_time = budget_s - (time() - start_time)
    if refinement_time > 1.0:
        final_h = best_h.copy()
        for iter_ref in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            corr = np.correlate(final_h, 1 - final_h, mode='full')
            max_idx = np.argmax(corr)
            grad = np.zeros_like(final_h)
            for i in range(n_points):
                h_plus = final_h.copy()
                h_plus[i] += 1e-4
                h_plus = project_to_feasible(h_plus, n_points)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (corr_plus[max_idx] - corr[max_idx]) / 1e-4
                
            final_h -= 0.001 * grad
            final_h = project_to_feasible(final_h, n_points)
            
            c5_refined = evaluate_erdos_solution(final_h, dx)
            if c5_refined < best_c5:
                best_h = final_h.copy()
                best_c5 = c5_refined
    
    return (list(best_h), best_c5, best_n_points)
