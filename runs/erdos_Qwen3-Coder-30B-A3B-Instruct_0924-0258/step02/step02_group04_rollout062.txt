import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m=10):
    """Initialize a simplex of particles"""
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        if len(h_ref) != n_points:
            h_ref = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_ref)), h_ref)
    else:
        h_ref = np.ones(n_points) * 0.5
        
    particles = [project_to_feasible(h_ref, n_points)]
    
    # Generate additional particles by moving along random directions
    for _ in range(m-1):
        direction = np.random.randn(n_points)
        step_size = 0.2
        h_new = particles[0] + step_size * direction
        particles.append(project_to_feasible(h_new, n_points))
        
    return particles

def lbfgsb_refine(h, n_points, dx, maxiter=20):
    """Refine solution using L-BFGS-B"""
    def obj(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2}
    
    try:
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': maxiter})
        if res.success:
            return project_to_feasible(res.x, n_points)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia_w = 0.7
    c1 = c2 = 2.0
    shake_threshold = 50
    refine_interval = 200
    
    # Initialize particles
    particles = initialize_simplex(n_points, swarm_size)
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Evaluate initial particles
    fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new position
            new_fitness = evaluate_erdos_solution(particles[i], dx)
            
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
                
                if new_fitness < gbest_fitness:
                    gbest = particles[i].copy()
                    gbest_fitness = new_fitness
                    last_improvement = iteration
        
        # Check for stagnation and apply shaking
        if iteration - last_improvement > shake_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
                # Re-evaluate after shaking
                new_fitness = evaluate_erdos_solution(particles[i], dx)
                if new_fitness < pbest_fitness[i]:
                    pbest[i] = particles[i].copy()
                    pbest_fitness[i] = new_fitness
                    if new_fitness < gbest_fitness:
                        gbest = particles[i].copy()
                        gbest_fitness = new_fitness
                        last_improvement = iteration
                        
        # Periodically refine top particles
        if iteration % refine_interval == 0 and iteration > 0:
            # Refine top 10 particles based on pbest_fitness
            top_indices = np.argsort(pbest_fitness)[:10]
            for idx in top_indices:
                refined = lbfgsb_refine(pbest[idx], n_points, dx)
                new_fitness = evaluate_erdos_solution(refined, dx)
                if new_fitness < pbest_fitness[idx]:
                    pbest[idx] = refined
                    pbest_fitness[idx] = new_fitness
                    if new_fitness < gbest_fitness:
                        gbest = refined
                        gbest_fitness = new_fitness
                        last_improvement = iteration

    # Final refinement of the best solution
    final_h = lbfgsb_refine(gbest, n_points, dx, maxiter=50)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
