import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with smooth step function that's close to 0.5
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def optimize_single(h_start, n_points, dx, time_limit):
    """Optimize a single starting point"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Use a simple optimization loop with restarts
    for iteration in range(100):
        if time() - start_time > time_limit:
            break
            
        # Random perturbation
        h_perturbed = perturb_h(h_best, 0.05)
        
        # Local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        try:
            # Simple bounds constrained optimization with SLSQP
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, h_perturbed, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 30, 'ftol': 1e-8})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Try different resolution levels
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 150)
        
        # Run multiple random restarts with Latin hypercube sampling approach
        for run_i in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_start = initialize_h(n_points)
            
            # Small randomization
            if run_i > 0:
                factor = 1.0 + (run_i / 10.0)
                h_start = np.clip(h_start * factor, 0, 1)
                h_start = project_to_feasible(h_start, n_points)
            
            h_opt, c5_opt = optimize_single(h_start, n_points, dx, time_per_res / 5)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
    
    # Final refinement if we have a good solution
    if best_h is not None and time() - start_time < budget_s * 0.9:
        dx = 2.0 / best_n_points
        final_time = min(budget_s - (time() - start_time), 100)
        best_h, best_c5 = optimize_single(best_h, best_n_points, dx, final_time)
    
    return (list(best_h), best_c5, best_n_points)
