    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function with projection
    def objective(h_vec):
        h_vec = project_feasible(h_vec)
        return compute_overlap(h_vec)
    
    # Differential evolution optimization
    try:
        result = differential_evolution(
            objective,
            bounds,
            maxiter=500,
            popsize=20,
            mutation=(0.5, 1.0),
            recombination=0.7,
            seed=seed,
            disp=False,
            callback=lambda x, convergence: None,
            strategy='best1bin'
        )

        final_C5 = compute_overlap(best_h)
