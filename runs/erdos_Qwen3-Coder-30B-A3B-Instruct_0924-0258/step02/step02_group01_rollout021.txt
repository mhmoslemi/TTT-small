import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 50

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    # Initialize population
    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = project_feasible(h_random, n_points)
        population.append(h_random)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fitness.append(evaluate_erdos_solution(h, 2.0 / n_points))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = int((budget_s - (time() - start_time)) / 0.5)
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                offspring1 = alpha * parent1 + (1 - alpha) * parent2
                offspring2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring2 += noise
                
            # Project to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
        
        # Evaluate new population
        new_fitness = []
        for h in new_population[:population_size]:
            new_fitness.append(evaluate_erdos_solution(h, 2.0 / n_points))
            
        # Survival: keep best individuals and replace worst with mutated best
        combined_population = population + new_population[:population_size]
        combined_fitness = fitness + new_fitness[:population_size]
        
        # Order by fitness
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_population[i] for i in sorted_indices[:population_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
    
    return (list(best_h), best_c5, n_points)
