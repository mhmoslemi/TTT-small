import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        n_points = len(h0)
    else:

        h0 = np.ones(n_points) * 0.5
    
    h_current = project_to_feasible(h0.copy(), n_points/2.0)
    c5_current = evaluate(h_current, 2.0/n_points)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Simulated Annealing parameters
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    min_T = 1e-6  # Minimum temperature
    
    # Tracks acceptance rate
    acceptance_count = 0
    total_moves = 0
    stale_iterations = 0
    max_stale = 100
    
    while time.time() - start_time < budget_s:
        # Generate candidate
        h_candidate = h_current + np.random.normal(0, T, n_points)
        h_candidate = project_to_feasible(h_candidate, n_points/2.0)
        c5_candidate = evaluate(h_candidate, 2.0/n_points)
        
        # Accept or reject
        delta = c5_candidate - c5_current
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h_current = h_candidate
            c5_current = c5_candidate
            acceptance_count += 1
            stale_iterations = 0
        else:
            stale_iterations += 1
            
        total_moves += 1

        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Temperature adaptation
        if total_moves % 50 == 0:
            acceptance_rate = acceptance_count / max(1, total_moves)
            if acceptance_rate < 0.2:
                alpha = max(0.99, alpha - 0.001)
            elif acceptance_rate > 0.5:
                alpha = min(0.999, alpha + 0.001)
            
        # Cooling schedule
        if T > min_T:
            T = max(min_T, T * alpha)
            
        # Reset temperature if stagnated
        if stale_iterations > max_stale:
            T = 0.5
            stale_iterations = 0
            
    return (best_h.tolist(), float(best_c5), n_points)
