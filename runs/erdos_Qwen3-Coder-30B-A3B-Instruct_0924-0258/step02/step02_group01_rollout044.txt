import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, dx, n_points):
    """Compute gradient for dominant lag k"""
    grad = np.zeros_like(h)
    for i in range(n_points):
        grad[i] = -(1 - h[(i + k) % n_points] - h[i]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_feasible(h_start, n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    max_iterations = 5000
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = current_c5
    no_improvement_count = 0
    
    # Main simulation loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find dominant lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h_current, k, dx, n_points)
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h_current)
        sum_h = np.sum(h_current)
        penalty_grad += 2 * lambda_penalty * (sum_h - n_points / 2)
        penalty_grad += 2 * lambda_penalty * np.clip(h_current, 0, 1) * (h_current > 1) - \
                        2 * lambda_penalty * np.clip(-h_current, 0, 1) * (h_current < 0)
        
        # Generate candidate
        h_candidate = h_current + eta * (-grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.random():
            h_current = h_candidate
            current_c5 = c5_candidate
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling
        T *= alpha
        
        # Random restart if no improvement
        if no_improvement_count > 50:
            h_current = project_feasible(h_current + np.random.randn(n_points) * 0.05, n_points)
            no_improvement_count = 0
            
    # Multi-resolution refinement
    resolutions = [60, 120]
    for n_points_res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Interpolate
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points_res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_start = project_feasible(h_interp, n_points_res)
        
        # Local optimization
        dx_res = 2.0 / n_points_res
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Find dominant lag
            corr = np.correlate(h_start, 1 - h_start, mode='full')
            k = np.argmax(corr)
            grad = compute_gradient(h_start, k, dx_res, n_points_res)
            
            h_candidate = h_start + 0.01 * (-grad)
            h_candidate = project_feasible(h_candidate, n_points_res)
            
            c5_candidate = evaluate_erdos_solution(h_candidate, dx_res)
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                h_start = h_candidate.copy()
            else:
                h_start = h_candidate.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    return (list(best_h), final_c5, len(best_h))
