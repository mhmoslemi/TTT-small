import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    best_C = C_high
    best_h = h_start.copy()
    
    # Binary search loop
    for iteration in range(40):
        if time() - start_time > budget_s * 0.9:
            break
            
        C_mid = (C_low + C_high) / 2
        
        # Set up CVXPY problem
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # Compute indices for circular correlation
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h[i] * (1 - h[j])
            constraints.append(corr_sum <= C_mid * n_points / 2)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, verbose=False)
            if prob.status == 'optimal':
                h_solution = h.value
                if h_solution is not None:
                    C_high = C_mid
                    best_C = C_mid
                    best_h = h_solution.copy()
        except:
            pass
        
        if C_high - C_low < 1e-6:
            break
    
    # Refinement with gradient descent
    refinement_time = budget_s - (time() - start_time)
    if refinement_time > 10:
        h_current = best_h.copy()
        dx_refine = 2.0 / n_points
        
        # Gradient descent steps
        for step in range(500):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute current overlap
            current_C = evaluate_erdos_solution(h_current, dx_refine)
            
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (n_points / 2) / h_plus_sum
                c_plus = evaluate_erdos_solution(h_plus, dx_refine)
                grad[i] = (c_plus - current_C) / eps
            
            # Update step
            step_size = 0.01
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum > 0:
                h_new = h_new * (n_points / 2) / h_new_sum
            
            # Accept if improvement
            c_new = evaluate_erdos_solution(h_new, dx_refine)
            if c_new < current_C:
                h_current = h_new.copy()
                best_C = c_new
    
    # Final evaluation
    final_C = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_C, n_points)
