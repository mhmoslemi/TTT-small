import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    # Initialize Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    # Start optimization
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize temporary h
    h = h_start.copy()
    
    # Optimization loop
    while time() - start_time < budget_s:
        # Compute full correlation to find the most constraining lag
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for this specific lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h[j]) - h[i]
        
        grad *= dx
        
        # Update Adam moments
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update parameters
        h -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
