import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Optimization parameters
    max_iter = int(budget_s * 10)  # Rough estimate of iterations
    restart_interval = 100
    perturb_strength = 0.05
    
    h_current = best_h.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute full correlation
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for specific lag k
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k) % n_points
            grad[j] = (1 - h_current[idx]) - h_current[j]
        grad *= dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        h_current -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_current = project_feasible(h_current, n_points)
        
        # Evaluate
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Restart if no improvement for a while
        if i % restart_interval == 0 and i > 0:
            # Add perturbation to current best
            h_current = project_feasible(
                h_current + np.random.normal(0, perturb_strength, n_points), 
                n_points
            )
            # Reset Adam moments
            m.fill(0)
            v.fill(0)
            
    return (list(best_h), best_c5, n_points)
