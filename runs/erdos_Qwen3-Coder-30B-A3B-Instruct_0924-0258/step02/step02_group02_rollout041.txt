import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_overlaps(h_vec, shift_indices):
        overlaps = []
        for s in shift_indices:
            if s >= 0:
                corr = np.sum(h_vec[:-s] * (1 - h_vec[s:])) * dx
            else:
                corr = np.sum(h_vec[-s:] * (1 - h_vec[:-s])) * dx
            overlaps.append(corr)
        return np.array(overlaps)

    # Initialize optimization variables
    h_var = cp.Variable(n_points)
    eta = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum, eta >= 0]

    # Optimizer with cutting planes
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Sample a coarse set of shifts to get initial worst-case
    coarse_shifts = np.arange(0, n_points, max(1, n_points // 10))
    initial_overlaps = compute_overlaps(h, coarse_shifts)
    worst_shift_idx = np.argmax(initial_overlaps)
    worst_shift = coarse_shifts[worst_shift_idx]
    
    # Refine the search around the worst shift
    fine_shifts = np.arange(max(0, worst_shift - 5), min(n_points, worst_shift + 6))
    
    iteration = 0
    max_iter = 500
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:  # Leave 5s margin
            break
            
        # Compute overlaps involving current worst shifts
        overlaps = compute_overlaps(h, fine_shifts)
        
        if len(overlaps) > 0:
            worst_idx = np.argmax(overlaps)
            worst_shift = fine_shifts[worst_idx]
        
            # Update constraints
            h_var.value = h.copy()
            eta.value = best_c5 + 1  # Initialize to something high
            
            # Add constraint for the most violated shift
            constraints.append(eta >= 0)
            # Do a rough approximation of the constraint based on the current h
            temp_overlap = 0
            if worst_shift >= 0:
                temp_overlap = np.sum(h[:-worst_shift] * (1 - h[worst_shift:])) * dx
            else:
                temp_overlap = np.sum(h[-worst_shift:] * (1 - h[:-worst_shift])) * dx
            
            # Add the effective constraint
            constraints.append(eta >= temp_overlap)
            
            # Solve with CVXPY
            obj = cp.Minimize(eta)
            prob = cp.Problem(obj, constraints)
            
            try:
                prob.solve(solver=ECOS, verbose=False, max_iter=1000)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    new_h = h_var.value
                    new_c5 = eta.value
                    
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = new_h.copy()
                        h = new_h.copy()
                        h = project_h(h)
                    
                    # Refine the set of shifts for next round
                    fine_shifts = np.arange(max(0, worst_shift - 5), min(n_points, worst_shift + 6))
                else:
                    # Fallback to simple gradient projection if optimization fails
                    h = project_h(h)
            except:
                h = project_h(h)
        else:
            h = project_h(h)
            
        iteration += 1
        if iteration > max_iter:
            break

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
