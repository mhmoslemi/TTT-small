import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, hessian, trust_radius, target_sum, dx):
    """Perform a trust region step."""
    n = len(h)
    try:
        # Solve trust region subproblem
        # minimize 1/2 * d^T * H * d + grad^T * d
        # subject to ||d|| <= trust_radius
        
        # Simplified version using identity hessian
        # This is a quadratic subproblem approximation
        # We directly project the gradient step to the trust region
        grad_step = -grad
        
        norm_grad = np.linalg.norm(grad_step)
        if norm_grad > 0:
            scaled_grad = grad_step * trust_radius / norm_grad
        else:
            scaled_grad = np.zeros_like(grad_step)
        
        candidate = h + scaled_grad
        candidate = project_to_feasible(candidate, target_sum)
        
        # Return the step (candidate - h) and the objective value
        return candidate, evaluate(candidate, dx)
    except Exception:
        # Fallback to small perturbation if optimization fails
        return project_to_feasible(h + np.random.normal(0, 0.01, n), target_sum), evaluate(project_to_feasible(h + np.random.normal(0, 0.01, n), target_sum), dx)
