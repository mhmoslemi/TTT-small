import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin hypercube samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        # Create a shuffled list of indices for each dimension
        sample = np.random.permutation(n_points).astype(float)
        # Map to [0,1] interval
        sample = sample / (n_points - 1)
        samples.append(sample)
    return np.array(samples)

def initialize_h(n_points):
    """Initialize h using Latin hypercube sampling or initial_h_values"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Generate initial samples using Latin hypercube sampling
        samples = latin_hypercube_sample(n_points, 20)
        # Use a diversified population center
        h = np.mean(samples, axis=0)
    return project_to_feasible(h, n_points)

def perturb_h(h, variance=0.2):
    """Add controlled Gaussian noise to the function"""
    noise = np.random.normal(0, variance, len(h))
    return project_to_feasible(h + noise, len(h))

def lbfgsb_optimize(h_start, n_points, dx, max_iter=20):
    """Optimize using L-BFGS-B with bounds"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': max_iter, 'ftol': 1e-8})
        if res.success:
            return project_to_feasible(res.x, n_points)
    except:
        pass
    return h_start

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for faster exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Allocate time for different phases
    coarse_time = budget_s * 0.2
    fine_time = budget_s * 0.8
    
    # For coarse, start with 15 and 30 points
    coarse_resolutions = [15, 30]
    fine_resolutions = [45, 60, 75]
    
    # Coarse optimization phase: more exploration
    for n_points in coarse_resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Use more random restarts for coarse
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            h_perturbed = perturb_h(h_start)
            h_opt = lbfgsb_optimize(h_perturbed, n_points, dx, max_iter=15)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points

    # Fine optimization phase with larger resolutions
    for n_points in fine_resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        # Start with the best found so far as initialization
        if best_h is not None and best_n_points == n_points:
            h_start = best_h
        else:
            h_start = initialize_h(n_points)
        
        # Fewer restarts for fine tuning
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Try different starting points
            h_perturbed = perturb_h(h_start, 0.1)
            h_opt = lbfgsb_optimize(h_perturbed, n_points, dx, max_iter=30)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
    
    # Final cleanup pass if necessary
    if best_h is not None:
        final_h = lbfgsb_optimize(best_h, best_n_points, 2.0/best_n_points, max_iter=50)
        final_c5 = evaluate_erdos_solution(final_h, 2.0/best_n_points)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    # If we don't yet have a solution, use the standard fallback
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        h_opt = lbfgsb_optimize(h_start, n_points, dx)
        best_h = h_opt
        best_c5 = evaluate_erdos_solution(h_opt, dx)
        best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
