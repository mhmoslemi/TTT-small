import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def solve_optimization(h_vec, constraints_list):
        # Define variables
        h_vars = cp.Variable(n_points)
        eta = cp.Variable()
        
        # Define constraints
        constr = [h_vars[i] >= 0 for i in range(n_points)]
        constr += [h_vars[i] <= 1 for i in range(n_points)]
        constr += [cp.sum(h_vars) == target_sum]
        constr += constraints_list

        # Objective
        objective = cp.Minimize(eta)
        
        # Solve
        prob = cp.Problem(objective, constr)
        try:
            prob.solve(solver=CPARSER, verbose=False, max_iter=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                new_h = h_vars.value
                new_eta = eta.value
                return new_h, new_eta
        except:
            pass

        return h_vec, compute_c5(h_vec)

    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Initial coarse search for worst shift
    coarse_shifts = np.linspace(0, 2, 10)
    worst_shift_idx = 0
    max_corr = -1

    # Compute overlaps for coarse shifts to find initial worst
    for k in coarse_shifts:
        shift_idx = int(np.round(k / dx))
        if shift_idx < n_points:
            corr = np.correlate(h, 1 - h, mode='full')
            corr *= dx
            if corr[shift_idx] > max_corr:
                max_corr = corr[shift_idx]
                worst_shift_idx = shift_idx

    # Main iterative optimization: cutting plane method
    constraints = []
    max_iter = 500
    iteration = 0

    while time.time() - start_time < budget_s and iteration < max_iter:
        # Refine around the worst shift
        fine_shifts = np.linspace(max(0, worst_shift_idx - 5), min(n_points - 1, worst_shift_idx + 5), 20)
        max_corr = -1
        best_shift_idx = worst_shift_idx
        
        for k_idx in fine_shifts:
            k_idx = int(k_idx)
            if k_idx < n_points:
                corr = np.correlate(h, 1 - h, mode='full')
                corr *= dx
                if corr[k_idx] > max_corr:
                    max_corr = corr[k_idx]
                    best_shift_idx = k_idx

        # Add constraint for this shift
        shift_constraint = cp.sum(cp.multiply(h, cp.roll(1 - h, best_shift_idx))) * dx <= cp.Constant(1e10)
        constraints.append(shift_constraint)
        
        # Solve optimization with new constraint
        new_h, new_eta = solve_optimization(h, constraints)
        
        if new_eta < best_c5:
            best_c5 = new_eta
            best_h = new_h.copy()
            h = new_h.copy()
        else:
            # If solution didn't improve, try with noise perturbation
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = project_h(h + noise)
            
        iteration += 1

    # Final check and projection
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)

    return (final_h.tolist(), final_c5, n_points)
