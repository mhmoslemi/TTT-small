import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize a simplex of m particles"""
    particles = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        particles.append(project_to_feasible(h0, n_points))
    else:
        # Initialize with uniform distribution
        particles.append(project_to_feasible(np.ones(n_points) * 0.5, n_points))
    
    # Generate additional particles by perturbing
    for i in range(m - 1):
        h = particles[0].copy()
        # Randomly perturb some values
        indices = np.random.choice(n_points, size=n_points//3, replace=False)
        for idx in indices:
            h[idx] += np.random.normal(0, 0.1)
        particles.append(project_to_feasible(h, n_points))
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    max_iterations = 2000
    swarm_size = 10
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    shake_threshold = 50
    lbfgs_interval = 200
    max_shake = 100
    
    # Initialize swarm
    particles = initialize_simplex(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    
    # Evaluate initial particles
    costs = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Track best
    best_idx = np.argmin(costs)
    gbest = particles[best_idx].copy()
    gbest_cost = costs[best_idx]
    
    # Personal best
    pbest = particles.copy()
    pbest_cost = costs.copy()
    
    # Track stagnation
    last_improve = 0
    shake_count = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           cognitive_weight * r1 * (pbest[i] - particles[i]) + 
                           social_weight * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            new_cost = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_cost < pbest_cost[i]:
                pbest[i] = particles[i].copy()
                pbest_cost[i] = new_cost
                
                # Check if global best
                if new_cost < gbest_cost:
                    gbest = particles[i].copy()
                    gbest_cost = new_cost
                    last_improve = iteration
        
        # Check for stagnation
        if iteration - last_improve > shake_threshold:
            shake_count += 1
            if shake_count < max_shake:
                # Shake particles
                for i in range(swarm_size):
                    noise = np.random.normal(0, 0.1, n_points)
                    particles[i] = project_to_feasible(particles[i] + noise, n_points)
                
                # Re-evaluate
                costs = np.array([evaluate_erdos_solution(p, dx) for p in particles])
                best_idx = np.argmin(costs)
                if costs[best_idx] < gbest_cost:
                    gbest = particles[best_idx].copy()
                    gbest_cost = costs[best_idx]
                    last_improve = iteration
        
        # Periodically run L-BFGS on best particles
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Sort by cost and improve top 10
            sorted_indices = np.argsort(pbest_cost)[:min(10, len(pbest_cost))]
            for idx in sorted_indices:
                if time() - start_time > budget_s * 0.99:
                    break
                h0 = pbest[idx].copy()
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 100})
                    if res.success:
                        h_star = project_to_feasible(res.x, n_points)
                        new_cost = evaluate_erdos_solution(h_star, dx)
                        if new_cost < pbest_cost[idx]:
                            pbest[idx] = h_star
                            pbest_cost[idx] = new_cost
                            if new_cost < gbest_cost:
                                gbest = h_star
                                gbest_cost = new_cost
                                last_improve = iteration
                except:
                    pass

    # Final local optimization on best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_cost = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_cost, n_points)
