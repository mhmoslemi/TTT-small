def initial_step_function(h_values, n_partitions=5):
    """Convert initial_h_values to step function with n_partitions"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Find partition points by finding where h changes value significantly
    partition_points = [0.0]
    prev_val = h_values[0]
    for i in range(1, n_points):
        if abs(h_values[i] - prev_val) > 1e-3:
            partition_points.append(i * dx)
            prev_val = h_values[i]
    partition_points.append(2.0)
    
    # Ensure we have exactly n_partitions + 1 points
    if len(partition_points) > n_partitions + 1:
        # Average segments to reduce number of partitions
        new_points = [0.0]
        step = len(partition_points) // (n_partitions + 1)
        for i in range(1, n_partitions):
            new_points.append(partition_points[i * step])
        new_points.append(2.0)
        partition_points = new_points
    elif len(partition_points) < n_partitions + 1:
        # Add more points
        needed = n_partitions + 1 - len(partition_points)
        for i in range(needed):
            new_point = (partition_points[i] + partition_points[i+1]) / 2
            partition_points.insert(i+1, new_point)
        partition_points = partition_points[:n_partitions+1]
    
    # Determine values for each segment
    segment_values = []
    for i in range(len(partition_points) - 1):
        # Use average of original points in this segment
        start_idx = int(partition_points[i] / dx)
        end_idx = int(partition_points[i+1] / dx)
        if end_idx > start_idx:
            segment_values.append(np.mean(h_values[start_idx:end_idx]))
        else:
            segment_values.append(h_values[start_idx])
    
    return np.array(partition_points), np.array(segment_values)

def compute_overlap(partition_points, values):
    """Compute maximum overlap for given partition structure"""
    n_points = len(partition_points) - 1
    dx = 2.0 / len(values)
    
    # Compute the integral for all shifts
    max_c = 0
    # For each possible shift (k from 0 to 2 with some sampling)
    shift_samples = np.linspace(0, 2, 100, endpoint=False)
    for k in shift_samples:
        integral = 0
        for i in range(n_points):
            # Found which parts of h intersect with (x+k) in the range of h
            start = partition_points[i]
            end = partition_points[i+1]
            # Check overlap with [start+k, end+k]
            overlap_start = max(start, start + k)
            overlap_end = min(end, end + k)
            if overlap_start < overlap_end:
                # Need to integrate h(x)(1-h(x+k))
                # If h(x) = values[i] and h(x+k) = values[j] (depending on k)
                # We consider k small enough that h(x+k) only takes value from 
                # a single segment.
                # Approximate the contribution using segments
                # We'll compute it for standard corrections
                if overlap_start >= start and overlap_end <= end:
                    integral += values[i] * (1 - values[i]) * (overlap_end - overlap_start)
        max_c = max(max_c, integral)
    
    return max_c

def eval_h(start_points, values):
    """Convert partitions to h array"""
    dx = 2.0 / len(values)
    h = np.zeros_like(values)
    for i, (start, end) in enumerate(zip(start_points[:-1], start_points[1:])):
        h[i] = values[i]
    return h
