import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    
    # Initial state
    h_current = project_feasible(h_start, n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = current_c5
    no_improvement_count = 0
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Compute gradient direction for dominant lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_lag = np.argmax(corr)
        n_corr = len(corr)
        # Center lag for correlation
        lag_offset = n_corr // 2
        
        # Compute gradient for dominant lag
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            j = (i + dominant_lag - lag_offset) % n_points
            grad[i] = -(1 - h_current[j] - h_current[i])
        
        # Add penalty gradient (simplified version)
        penalty_grad = np.zeros_like(h_current)
        sum_h = np.sum(h_current)
        penalty_grad += 2 * lambda_penalty * (sum_h - n_points/2)
        
        # Apply gradient step
        h_candidate = h_current - eta * (grad + penalty_grad)
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate new solution
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - current_c5
        
        # Accept or reject based on simulated annealing criterion
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h_current = h_candidate
            current_c5 = c5_candidate
            no_improvement_count = 0
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
        else:
            no_improvement_count += 1
            
        # Cool temperature
        T *= alpha
        
        # Random perturbation when stuck
        if no_improvement_count > 50:
            noise = np.random.randn(n_points) * 0.05
            h_current = project_feasible(h_current + noise, n_points)
            current_c5 = evaluate_erdos_solution(h_current, dx)
            no_improvement_count = 0
    
    # Multi-resolution refinement  
    resolutions = [60, 120]
    for res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, res)
        h_old = project_feasible(best_h, n_points)
        h_interp = np.interp(x_new, x_old, h_old)
        h_interp = project_feasible(h_interp, res)

        bounds = [(0, 1) for _ in range(res)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - res / 2}]

            return evaluate_erdos_solution(h, 2.0 / res)
        
        try:
            res_opt = minimize(obj, h_interp, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-6})
            if res_opt.success:
                h_check = project_feasible(res_opt.x, res)
                c5_check = evaluate_erdos_solution(h_check, 2.0 / res)
                if c5_check < best_c5:
                    best_h = h_check
                    best_c5 = c5_check
                    n_points = res
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
