import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    start_time = time()
    
    # Initialize population
    pop_size = 50

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h_random = np.random.rand(n_points)
        h_random = project_feasible(h_random, n_points)
        population.append(h_random)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, 2.0/n_points) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    generation = 0
    while time() - start_time < budget_s * 0.95:
        generation += 1
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, 5, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            # Crossover
            if np.random.rand() < 0.8:
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            offspring.extend([child1, child2])
        
        # Evaluate offspring
        offspring = offspring[:pop_size]
        offspring_fitness = [evaluate_erdos_solution(h, 2.0/n_points) for h in offspring]
        
        # Survival (keep best 25% and replace worst with mutated best)
        all_individuals = population + offspring
        all_fitness = fitness + offspring_fitness
        
        # Keep best individuals
        sorted_indices = np.argsort(all_fitness)
        keep_indices = sorted_indices[:pop_size//4]
        survivors = [all_individuals[i] for i in keep_indices]
        
        # Fill with mutated versions of best individuals
        best_individuals = [all_individuals[i] for i in sorted_indices[:pop_size//4]]
        for _ in range(pop_size - len(survivors)):
            parent = best_individuals[np.random.randint(0, len(best_individuals))]
            mutated = parent + np.random.normal(0, 0.05, n_points)
            mutated = project_feasible(mutated, n_points)
            survivors.append(mutated)
        
        population = survivors
        fitness = [evaluate_erdos_solution(h, 2.0/n_points) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
    
    # Final re-evaluation
    best_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    return (list(best_h), best_c5, n_points)
