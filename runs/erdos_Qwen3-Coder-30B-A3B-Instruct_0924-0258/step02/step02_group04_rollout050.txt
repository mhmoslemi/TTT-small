import numpy as np
from scipy.optimize import minimize
from time import time
import itertools

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin Hypercube samples for initial solutions"""
    samples = []
    for _ in range(n_samples):
        h = np.zeros(n_points)
        # Distribute evenly across [0,1] and add some randomness
        for i in range(n_points):
            h[i] = (i + np.random.random()) / n_points
        # Scale to correct sum
        h = project_to_feasible(h, n_points)
        samples.append(h)
    return samples

def initial_h_values_2(n_points):
    """Generate a better starting pattern"""
    h = np.ones(n_points) * 0.5
    # Create a more optimized piecewise constant pattern
    mid = n_points // 2
    h[:mid] = 0.7
    h[mid:] = 0.3
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Sample parameters
    n_samples = 50
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_initial_samples = latin_hypercube_sample(n_points, n_samples)
        
        # Also try the improved initial construction
        h_initial_samples.append(initial_h_values_2(n_points))
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Try each initial solution with optimization
        for i, h_start in enumerate(h_initial_samples):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt = h_start.copy()
            
            # Try with different optimization algorithms
            max_iter = 30
            for iter in range(max_iter):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Random perturbation
                h_perturbed = np.copy(h_opt)
                noise = np.random.normal(0, 0.05, n_points)
                h_perturbed += noise
                h_perturbed = project_to_feasible(h_perturbed, n_points)
                
                # Optimization step
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                # Use L-BFGS-B
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, h_perturbed, method='L-BFGS-B',
                                   bounds=bounds, options={'maxiter': 20})
                    if res.success:
                        h_candidate = project_to_feasible(res.x, n_points)
                        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                        if c5_candidate < best_c5:
                            best_h = h_candidate.copy()
                            best_c5 = c5_candidate
                            best_n_points = n_points
                        if c5_candidate < evaluate_erdos_solution(h_opt, dx):
                            h_opt = h_candidate
                except:
                    pass
            
            # Final check of the starting solution
            c5_start = evaluate_erdos_solution(h_start, dx)
            if c5_start < best_c5:
                best_h = h_start.copy()
                best_c5 = c5_start
                best_n_points = n_points
    
    # Final optimization of the best solution found
    if best_h is not None:
        dx = 2.0 / best_n_points
        # Perform another round of optimization on the best solution
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(best_n_points)]
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
            if res.success:
                best_h = project_to_feasible(res.x, best_n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initial_h_values_2(n_points)
        best_h, best_c5 = h_start, evaluate_erdos_solution(h_start, dx)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
