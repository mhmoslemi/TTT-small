import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    # ES parameters
    pop_size = 30
    elite_ratio = 0.1
    mutate_ratio = 0.4
    beta = 0.2
    max_generations = 10000
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(pop_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Initialize mutation strength
    sigma = np.full(pop_size, 0.05)
    
    # Evaluate initial population
    fitness = [evaluate(h, 2.0/n_points) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    stagnation_count = 0
    generation = 0
    
    while time.time() - start_time < budget_s and stagnation_count < 50:
        # Sort population by fitness
        indices = np.argsort(fitness)
        sorted_population = [population[i] for i in indices]
        sorted_fitness = [fitness[i] for i in indices]
        
        # Elitism - keep best individuals
        elite_count = max(1, int(elite_ratio * pop_size))
        elite = sorted_population[:elite_count]
        elite_fitness = sorted_fitness[:elite_count]
        
        # Create new population
        new_population = elite.copy()
        new_fitness = elite_fitness.copy()
        
        # Generate mutants for rest of population
        mutate_count = pop_size - elite_count
        for _ in range(mutate_count):
            # Select parent based on rank
            ranks = np.arange(len(sorted_fitness), 0, -1)
            probs = ranks / np.sum(ranks)
            parent_idx = np.random.choice(len(sorted_fitness), p=probs)
            parent = sorted_population[parent_idx]
            s = sigma[parent_idx]
            
            # Mutate
            mutant = parent + np.random.normal(0, s, n_points)
            mutant = project_to_feasible(mutant, n_points/2.0)
            
            new_population.append(mutant)
        
        # Evaluate new population
        new_fitness = [evaluate(h, 2.0/n_points) for h in new_population]
        
        # Update mutation strengths
        for i in range(pop_size):
            if np.random.random() < 0.5:
                sigma[i] = sigma[i] * np.exp(beta * np.random.normal(0, 1))
                sigma[i] = np.clip(sigma[i], 0.001, 0.5)
        
        # Update best
        current_best_idx = np.argmin(new_fitness)
        if new_fitness[current_best_idx] < best_c5:
            best_c5 = new_fitness[current_best_idx]
            best_h = new_population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        population = new_population
        fitness = new_fitness
        generation += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
