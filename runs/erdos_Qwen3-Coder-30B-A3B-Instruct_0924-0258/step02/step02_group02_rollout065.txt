import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
        
    def compute_overlaps(h_vec, shifts):
        overlaps = []
        for k in shifts:
            s = int(round(k / dx))
            if s >= 0:
                overlap = sum(h_vec[i] * (1 - h_vec[i + s]) for i in range(len(h_vec) - s))
            else:
                overlap = sum(h_vec[i + (-s)] * (1 - h_vec[i]) for i in range(len(h_vec) + s))
            overlaps.append(overlap * dx)
        return np.array(overlaps)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Efficient initial set of shifts to test
    shift_grid = np.linspace(0, 2, 100)
    
    # Initial solution
    h = project_h(h)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 1000
    iteration = 0
    
    # Initialize with a coarse search
    overlaps = compute_overlaps(h, shift_grid)
    max_overlap_idx = np.argmax(overlaps)
    worst_shift = shift_grid[max_overlap_idx]
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 2:  # Leave time for final check
            break
            
        # Refine around the worst shift with dense grid
        local_grid = np.linspace(max(0, worst_shift - dx), min(2, worst_shift + dx), 50)
        
        # Compute overlaps at local shifts
        overlaps_local = compute_overlaps(h, local_grid)
        
        # Find the new worst shift
        max_local_idx = np.argmax(overlaps_local)
        worst_shift_local = local_grid[max_local_idx]
        current_max_overlap = overlaps_local[max_local_idx]
        
        if current_max_overlap < best_c5 - 1e-6:
            best_c5 = current_max_overlap
            best_h = h.copy()
        
        # Build convex optimization problem with cutting planes
        # Decision variables
        h_vars = Variable(n_points)
        eta = Variable(1)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == target_sum,
            eta >= 0
        ]
        
        # Add constraints for all local shifts
        for i, k in enumerate(local_grid):
            s = int(round(k / dx))
            if s >= 0:
                overlap_expr = sum(h_vars[j] * (1 - h_vars[j + s]) for j in range(len(h_vars) - s))
            else:
                overlap_expr = sum(h_vars[j + (-s)] * (1 - h_vars[j]) for j in range(len(h_vars) + s))
            constraints.append(overlap_expr <= eta)
        
        # Objective
        objective = Minimize(eta)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_vars.value).flatten()
                h = project_h(h_new)
            else:
                # Fallback to projected gradient step
                h = project_h(h)
        except:
            h = project_h(h)
        
        # Update worst shift
        overlaps = compute_overlaps(h, local_grid)
        max_overlap_idx = np.argmax(overlaps)
        worst_shift = local_grid[max_overlap_idx]
        
        iteration += 1
        if iteration > max_iter:
            break
            
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
