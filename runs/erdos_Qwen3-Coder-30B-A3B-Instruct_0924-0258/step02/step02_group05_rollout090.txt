import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Fallback: create symmetric initial guess
        n_points = 45
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Fourier-based representation
    def h_from_fourier(coeffs, n):
        # coeffs[0] = a0, coeffs[1:1+N] = a_k, coeffs[1+N:1+2*N] = b_k
        N = (len(coeffs) - 1) // 2
        a0, a_coeffs, b_coeffs = coeffs[0], coeffs[1:1+N], coeffs[1+N:1+2*N]
        
        x = np.linspace(0, 2, n)
        h = a0 / 2.0
        for k in range(1, N+1):
            h += a_coeffs[k-1] * np.cos(2*np.pi*k*x) + b_coeffs[k-1] * np.sin(2*np.pi*k*x)
        return h
    
    def gradient_h_from_fourier(coeffs, n):
        N = (len(coeffs) - 1) // 2
        a0, a_coeffs, b_coeffs = coeffs[0], coeffs[1:1+N], coeffs[1+N:1+2*N]
        
        x = np.linspace(0, 2, n)
        dh = np.zeros((n, len(coeffs)))
        dh[:, 0] = 0.5  # a0 contribution
        
        for k in range(1, N+1):
            dh[:, k] = np.cos(2*np.pi*k*x)    # a_k contribution
            dh[:, k+N] = np.sin(2*np.pi*k*x)  # b_k contribution
        
        return dh

    # Optimization parameters
    max_fourier_components = 15
    adam_lr = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    max_iter = 5000
    
    # Multi-resolution optimization: 5, 10, 15, 20, final resolution
    resolutions = [5, 10, max_fourier_components]
    best_C5 = float('inf')
    best_h = h_init.copy()
    current_coeffs = None
    
    for i, N in enumerate(resolutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Initialize coefficients
        if i == 0:
            coeffs = np.zeros(1 + 2*N)
        else:
            # Start from previous solution, but repair it
            if current_coeffs is not None:
                coeffs = current_coeffs.copy()
                # Zero out higher-order coefficients
                coeffs[1+2*N:] = 0.0
            else:
                coeffs = np.zeros(1 + 2*N)
                
        # Adam variables
        m = np.zeros_like(coeffs)
        v = np.zeros_like(coeffs)
        
        best_local_C5 = float('inf')
        best_local_coeffs = coeffs.copy()
        
        # Local optimization at this resolution
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute h from coeffs
            h_vals = h_from_fourier(coeffs, n_points)
            h_vals = project_feasible(h_vals)
            
            # Compute current C5
            current_C5 = compute_overlap(h_vals)
            
            if current_C5 < best_local_C5:
                best_local_C5 = current_C5
                best_local_coeffs = coeffs.copy()
                
            # Compute gradient using finite differences
            gradient = np.zeros_like(coeffs)
            eps = 1e-4
            
            for j in range(len(coeffs)):
                coeffs_plus = coeffs.copy()
                coeffs_plus[j] += eps
                h_plus = h_from_fourier(coeffs_plus, n_points)
                h_plus = project_feasible(h_plus)
                C_plus = compute_overlap(h_plus)
                
                coeffs_minus = coeffs.copy()
                coeffs_minus[j] -= eps
                h_minus = h_from_fourier(coeffs_minus, n_points)
                h_minus = project_feasible(h_minus)
                C_minus = compute_overlap(h_minus)
                
                gradient[j] = (C_plus - C_minus) / (2 * eps)
            
            # Adam update
            m = adam_beta1 * m + (1 - adam_beta1) * gradient
            v = adam_beta2 * v + (1 - adam_beta2) * gradient**2
            m_hat = m / (1 - adam_beta1**(iter_count + 1))
            v_hat = v / (1 - adam_beta2**(iter_count + 1))
            coeffs = coeffs - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            # Clamp learning rate if necessary to prevent divergence
            if np.any(np.abs(coeffs) > 1000):
                adam_lr *= 0.5
                if adam_lr < 1e-6:
                    break
        
        if best_local_C5 < best_C5:
            best_C5 = best_local_C5
            best_h = h_from_fourier(best_local_coeffs, n_points)
            best_h = project_feasible(best_h)
            
        # Set next coeffs for higher resolution
        current_coeffs = best_local_coeffs.copy()

    # Final check
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
