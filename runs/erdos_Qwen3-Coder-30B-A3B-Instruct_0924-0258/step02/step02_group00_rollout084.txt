import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Use initial_h_values if available, otherwise create a reasonable start
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    search_iterations = 40
    
    # Store best feasible solution
    best_feasible_h = None
    best_feasible_c5 = float('inf')
    
    # Initial solution from parent
    if 'initial_h_values' in globals():
        try:
            h_refined = project_to_feasible(np.array(initial_h_values), n_points)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_feasible_c5:
                best_feasible_h = h_refined
                best_feasible_c5 = c5_refined
        except:
            pass
    
    # Binary search with CVXPY
    for i in range(search_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        C_mid = (C_low + C_high) / 2.0
        
        # Construct the problem: minimize 0, subject to constraints
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints for all k
        for k in range(-n_points+1, n_points):
            # Compute the correlation term
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h_var[i] * (1 - h_var[j])
            constraints.append(corr_sum <= C_mid * n_points/2)
        
        # Formulate the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Solve with timeout
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_feasible = project_to_feasible(h_sol, n_points)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_feasible_c5:
                        best_feasible_h = h_feasible
                        best_feasible_c5 = c5_feasible
                    C_high = C_mid
                else:
                    C_low = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
    
    # Refinement using gradient descent
    if best_feasible_h is not None:
        h_opt_final = best_feasible_h.copy()
        best_c5_final = best_feasible_c5
        best_h = best_feasible_h
        best_c5 = best_feasible_c5
        remaining_time = budget_s - (time() - start_time)
        refinement_time = min(remaining_time, 500)
        ref_start = time()
        
        # Refinement loop
        for _ in range(1000):
            if time() - ref_start > refinement_time:
                break
                
            # Gradient step on smooth objective with regularization
            corr = np.correlate(h_opt_final, 1 - h_opt_final, mode='full')
            max_corr = np.max(corr)
            # Simple gradient descent step to reduce the maximum correlation
            # However, ensure we stay within bounds and sum constraint
            
            # Compute a per-element derivative
            grad = np.zeros_like(h_opt_final)
            for i in range(n_points):
                # Derivative of the max correlation term with respect to h[i]
                # We take gradient of a specific max k term; this is an approximation
                # Use finite differences for safety
                eps = 1e-6
                h_plus = h_opt_final.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                corr_plus_max = np.max(corr_plus)
                
                h_minus = h_opt_final.copy()
                h_minus[i] -= eps
                h_minus = project_to_feasible(h_minus, n_points)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                corr_minus_max = np.max(corr_minus)
                
                grad[i] = (corr_plus_max - corr_minus_max) / (2 * eps)
            
            # Use projected gradient descent
            step_size = 0.01
            h_new = h_opt_final - step_size * grad
            h_new = project_to_feasible(h_new, n_points)
            
            if evaluate_erdos_solution(h_new, dx) < evaluate_erdos_solution(h_opt_final, dx):
                h_opt_final = h_new
            
            # Check improvement
            new_c5 = evaluate_erdos_solution(h_opt_final, dx)
            if new_c5 < best_c5_final:
                best_h = h_opt_final.copy()
                best_c5_final = new_c5
        
        best_c5 = best_c5_final
        best_h = best_h
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
