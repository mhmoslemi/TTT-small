import numpy as np
from scipy.fftpack import fft, ifft
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set initial resolution
    n_points = 15
    dx = 2.0 / n_points
    
    # Initialize
    h = initialize_h(n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Fourier-based optimization
    max_iter = 1000
    n_iter = 0
    
    last_improvement = 0
    best_c5_history = []
    
    while n_iter < max_iter and (time() - start_time) < budget_s * 0.98:
        n_iter += 1
        
        # Store progress
        best_c5_history.append(best_c5)
        if len(best_c5_history) > 500:
            best_c5_history.pop(0)
            
        # Fourier transform
        H = fft(h)
        
        # Compute current autocorrelation peak
        corr = np.correlate(h, 1 - h, mode='full')
        current_max = np.max(corr) * dx
        
        # Convert to power spectrum
        P = np.abs(H) ** 2
        
        # Add penalty to enforce flat spectrum (minimize concentration)
        penalty = np.sum(P ** 2)  # This encourages flatness
        
        # Compute gradient in frequency domain
        grad_P = 2 * P  # Gradient of sum of squares
        grad_H = grad_P * np.conj(H)  # But we need real output, so we symmetrize
        
        # Ensure the gradient is compatible with real signals
        grad_H = (grad_H + np.conj(np.flip(grad_H[1:-1]))) / 2
        grad_H[0] = np.real(grad_H[0])
        if len(P) % 2 == 0:
            grad_H[-1] = np.real(grad_H[-1])
            
        # Update in frequency domain with small step
        step_size = 1e-4 * (1.0 + 0.01 * n_iter)
        H_new = H - step_size * grad_H
        
        # Convert back to time domain
        h_new = np.real(ifft(H_new))
        
        # Repair feasibility
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate the candidate solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            last_improvement = n_iter
            
        # Refinement for resolution upgrades
        if n_iter == 300:
            # Try a higher resolution
            n_points = 30
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = project_to_feasible(h, n_points)
            best_c5 = evaluate_erdos_solution(h, dx)
            best_h = h.copy()
            last_improvement = n_iter
        elif n_iter == 600:
            n_points = 45
            dx = 2.0 / n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = project_to_feasible(h, n_points)
            best_c5 = evaluate_erdos_solution(h, dx)
            best_h = h.copy()
            last_improvement = n_iter
            
    # Final robust check and return best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
