import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO Parameters
    swarm_size = 40
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_vel = 0.15
    min_vel = -0.15
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        particles.append(h.copy())
        velocities.append(np.random.uniform(min_vel, max_vel, n_points))
    else:
        particles.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.uniform(min_vel, max_vel, n_points))
    
    # Add random particles
    for _ in range(swarm_size - len(particles)):
        h_rand = np.random.rand(n_points) * 0.5 + 0.25
        h_rand = project_feasible(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.uniform(min_vel, max_vel, n_points))
    
    # Particle best and global best
    particle_best = [p.copy() for p in particles]
    particle_best_vals = [compute_overlap(p) for p in particles]
    global_best_idx = np.argmin(particle_best_vals)
    global_best = particle_best[global_best_idx].copy()
    global_best_val = particle_best_vals[global_best_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(len(particles)):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (c1 * r1 * (particle_best[i] - particles[i]) + 
                         c2 * r2 * (global_best - particles[i]))
            velocities[i] = w * velocities[i] + vel_update
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], min_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_feasible(particles[i])
            
            # Evaluate new position
            new_val = compute_overlap(particles[i])
            
            # Update particle best
            if new_val < particle_best_vals[i]:
                particle_best[i] = particles[i].copy()
                particle_best_vals[i] = new_val
                
                # Update global best
                if new_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = new_val
    
    # Final evaluation
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
