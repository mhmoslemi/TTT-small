import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, hessian, trust_radius, target_sum):
    """Solve trust region subproblem."""
    n = len(h)
    # In practice, solving tr-subproblem exactly is complex, so we'll use a simple approach:
    # Take a step in the negative gradient direction, scaled by trust_radius
    step = -grad * trust_radius / np.linalg.norm(grad)
    
    # Project to feasible region
    new_h = h + step
    new_h = project_to_feasible(new_h, target_sum)
    return new_h

def compute_gradient(h, dx, n_points):

    # Here we'll compute an estimate by perturbing h

    return grad
