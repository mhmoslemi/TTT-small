import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize m particles using simplex method"""
    particles = []
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        particles.append(project_to_feasible(h0, n_points))
    else:
        h0 = np.ones(n_points) * 0.5
        particles.append(project_to_feasible(h0, n_points))
    
    # Generate m-1 additional particles by moving along random directions
    for _ in range(m - 1):
        h = particles[0].copy()
        # Add small random perturbation
        pert = np.random.normal(0, 0.1, n_points)
        h = h + pert
        # Project to feasible set
        h = project_to_feasible(h, n_points)
        particles.append(h)
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia_weight = 0.7
    cognitive_weight = 2.0
    social_weight = 2.0
    shake_threshold = 50
    lbfgs_interval = 200
    max_shake = 100
    
    # Initialize particles
    particles = initialize_simplex(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    personal_best = particles.copy()
    pbest_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    global_best_idx = np.argmin(pbest_values)
    global_best = particles[global_best_idx].copy()
    global_best_value = pbest_values[global_best_idx]
    
    iter_no_improve = 0
    best_h = global_best.copy()
    best_c5 = global_best_value
    best_n_points = n_points
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            velocities[i] = (
                inertia_weight * velocities[i] +
                cognitive_weight * r1 * (personal_best[i] - particles[i]) +
                social_weight * r2 * (global_best - particles[i])
            )
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate and update personal best
            current_value = evaluate_erdos_solution(particles[i], dx)
            if current_value < pbest_values[i]:
                personal_best[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
                    iter_no_improve = 0
                    if current_value < best_c5:
                        best_h = global_best.copy()
                        best_c5 = current_value
                        best_n_points = n_points
            else:
                iter_no_improve += 1

        # Shake mechanism
        if iter_no_improve > shake_threshold:
            for i in range(swarm_size):
                # Add noise to escape local minima
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] = particles[i] + noise
                particles[i] = project_to_feasible(particles[i], n_points)
            iter_no_improve = 0

        # Periodic L-BFGS refinement
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Refine top particles
            top_indices = np.argsort(pbest_values)[:min(10, swarm_size)]
            for idx in top_indices:
                h0 = particles[idx].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                
                try:
                    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                                   options={'maxiter': 100})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < pbest_values[idx]:
                            personal_best[idx] = h_refined
                            pbest_values[idx] = c5_refined
                            if c5_refined < global_best_value:
                                global_best = h_refined
                                global_best_value = c5_refined
                                iter_no_improve = 0
                                if c5_refined < best_c5:
                                    best_h = global_best.copy()
                                    best_c5 = c5_refined
                                    best_n_points = n_points
                except:
                    pass

    # Final refinement on the best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    try:
        res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, 
                       options={'maxiter': 200})
        if res.success:
            best_h = project_to_feasible(res.x, n_points)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, best_n_points)
