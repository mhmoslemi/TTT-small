import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin Hypercube samples for initialization"""
    samples = []
    for _ in range(n_samples):
        h = np.zeros(n_points)
        # Create stratified samples
        for i in range(n_points):
            h[i] = (i + random.random()) / n_points
        # Add some randomness while maintaining sum constraint
        h = project_to_feasible(h + np.random.normal(0, 0.1, n_points), n_points)
        samples.append(h)
    return samples

def initialize_h(n_points):
    """Initialize h from initial_h_values if available"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def optimize_local(h_start, n_points, dx, time_limit):
    """Local optimization using scipy minimize"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Use multiple local optimizations with different starting points
    for i in range(20):
        if time() - start_time > time_limit:
            break
        
        h_perturbed = perturb_h(h_start, 0.05)
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        try:
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 10, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use different resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple restarts with different strategies
        samples = latin_hypercube_sample(n_points, 20)
        for i, h in enumerate(samples):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = optimize_local(h, n_points, dx, time_per_res / 20)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
                
        # Try a gradient-based local search
        h_opt, c5_opt = optimize_local(h_start, n_points, dx, time_per_res / 2)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points
            
    # If no progress found, fallback to simple approach
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_local(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
