import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def perturb_solution(h, perturbation_strength):
    """Apply perturbation to the function"""
    h = h.copy()
    mask = np.random.random(len(h)) < 0.2
    noise = np.random.normal(0, perturbation_strength, len(h))
    h[mask] += noise[mask]
    return project_to_feasible(h, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from existing construction or create new
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Main optimization loop
    max_iterations = 1000
    patience = 20
    patience_counter = 0
    
    for iter_num in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Randomize a few candidates
        num_candidates = min(30, n_points)
        candidates = []
        for _ in range(num_candidates):
            h_pert = perturb_solution(best_h, 0.1)
            candidates.append(h_pert)
            
        # Optimization using SLSQP
        for h_candidate in candidates:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        patience_counter = 0
            except:
                pass
                
        patience_counter += 1
        if patience_counter >= patience:
            break
            
    # Refinement at higher resolution
    if n_points < 500:
        n_points_ref = min(500, n_points * 2)
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, n_points_ref)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_start_ref = project_to_feasible(h_interp, n_points_ref)
        
        dx_ref = 2.0 / n_points_ref
        bounds = [(0, 1) for _ in range(n_points_ref)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_ref / 2}]
        
        try:
            res = minimize(lambda h: evaluate_erdos_solution(h, dx_ref), 
                           h_start_ref, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_ref = project_to_feasible(res.x, n_points_ref)
                c5_ref = evaluate_erdos_solution(h_ref, dx_ref)
                if c5_ref < best_c5:
                    best_h = h_ref
                    best_c5 = c5_ref
                    n_points = n_points_ref
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
