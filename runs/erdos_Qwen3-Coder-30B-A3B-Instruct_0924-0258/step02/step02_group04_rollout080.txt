import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population with variations of initial_h_values or random"""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base)), base)
    else:
        base = np.ones(n_points) * 0.5
    
    population = []
    for i in range(pop_size):
        if i == 0:
            h = base.copy()
        else:
            noise = np.random.normal(0, 0.1, n_points)
            h = base + noise
        h = project(h, n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, k=5):
    """Tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Settings
    pop_size = 50
    n_points = 30
    dx = 2.0 / n_points
    max_generations = 1000
    stagnation_limit = 200
    mutation_rate = 0.2
    crossover_rate = 0.7
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    # Evaluate initial fitness
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Termination criteria
    generations_without_improvement = 0
    last_best_c5 = best_c5
    
    # GA loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best
        new_population.append(best_h.copy())
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring = crossover(parent1, parent2)
            else:
                offspring = parent1.copy()
            
            # Mutation
            offspring = mutate(offspring, mutation_rate, 0.1)
            
            new_population.append(offspring)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            generations_without_improvement = 0
            last_best_c5 = best_c5
        else:
            generations_without_improvement += 1
            
        # Adapt mutation rate
        if generations_without_improvement > 50:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
            
        # Reset if stagnating
        if generations_without_improvement > stagnation_limit:
            population = initialize_population(n_points, pop_size)
            fitness = [evaluate_erdos_solution(h, dx) for h in population]
            generations_without_improvement = 0
            mutation_rate = 0.2
    
    return (list(best_h), best_c5, n_points)
