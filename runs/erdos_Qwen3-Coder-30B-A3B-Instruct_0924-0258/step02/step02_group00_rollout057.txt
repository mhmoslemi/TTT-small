import numpy as np
import cvxpy as cp
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set up resolution and parameters
    n_points = 35
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Binary search bounds
    low_c = 0.37
    high_c = 0.38085
    
    def solve_feasibility(h_init, C):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(-n_points + 1, n_points):
            # Compute shift index with circular boundary
            corr_sum = 0
            for i in range(n_points):
                shifted_idx = (i + k) % n_points
                corr_sum += h[i] * (1 - h[shifted_idx])
            constraints.append(corr_sum <= C * n_points / 2)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                return h.value
        except:
            pass
        return None
    
    # Binary search loop
    search_time = 60
    search_end_time = start_time + search_time
    
    for iteration in range(40):
        if time() > search_end_time:
            break
            
        C_mid = (low_c + high_c) / 2
        h_feasible = solve_feasibility(h_start, C_mid)
        
        if h_feasible is not None:
            # Refine this solution
            h_refined = h_feasible.copy()
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
                best_n_points = n_points
                
            high_c = C_mid
        else:
            low_c = C_mid
            
    # Refinement with gradient descent
    refine_time = budget_s - (time() - start_time) - 5
    if refine_time > 10:
        h_refined = best_h.copy()
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def constraint_sum(h):
            return np.sum(h) * dx - 1
        
        def gradient_h(h):
            grad = np.zeros_like(h)
            for i in range(len(h)):
                eps = 1e-8
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                obj_plus = objective(h_plus)
                obj_base = objective(h)
                grad[i] = (obj_plus - obj_base) / eps
            return grad
        
        # Iteratively refine the solution
        for i in range(500):
            if time() - start_time > budget_s - 1:
                break
            try:
                grad = gradient_h(h_refined)
                h_new = h_refined - 0.01 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                obj_new = objective(h_new)
                if obj_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = obj_new
                    best_n_points = n_points
                h_refined = h_new
            except:
                break
    
    return (list(best_h), best_c5, best_n_points)
