import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Quick Fourier-based optimization approach
    n_points = 15
    dx = 2.0 / n_points
    h = initialize_h(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Start with a Fourier-domain optimization approach
    def Fourier_optimize(h_start, n_points, dx, time_limit):
        h_best = h_start.copy()
        c5_best = evaluate_erdos_solution(h_start, dx)
        start_time = time()
        
        # Initial spectrum - flat
        H = np.fft.fft(h_start)
        H_mag = np.abs(H)
        phase = np.angle(H)
        
        # Small identity matrix to help avoid rank deficiency in optimization
        I = np.eye(n_points)
        
        # Spectral penalty
        def spectral_penalty(H):
            H_mag = np.abs(H)
            # More aggressive penalty for peaks
            return np.sum(H_mag**4) * 1e-2
        
        step_size = 1e-4
        momentum = 0.9
        v = np.zeros_like(H)
        
        iter_count = 0
        stagnation_count = 0
        last_improvement = 0
        
        while time() - start_time < time_limit and stagnation_count < 1000:
            iter_count += 1
            
            # Project back to time domain
            h = np.real(np.fft.ifft(H))
            h = project_to_feasible(h, n_points)
            current_c5 = evaluate_erdos_solution(h, dx)
            
            if current_c5 < c5_best:
                c5_best = current_c5
                h_best = h.copy()
                last_improvement = iter_count
                
            # Feedback is that we are More sensitive to really bad values in the spectrum.
            # So compute gradient w.r.t. to H directly by finite differences
            # or better, by observing how the spectral penalty is affected by small gradient steps
            
            # Compute the gradient in the frequency domain towards minimizing the penalty
            H_mag = np.abs(H)
            # High frequency scaling to discourage modulated peaks in frequency
            penalty_grad = 4 * (H_mag**3) / (1 + H_mag**4)  # nicer derivative
            penalty_grad = np.fft.fftshift(penalty_grad, axes=0)  # Optional: shift before inverse
            
            # Compute desired direction (transpose gradient of penalty in frequency domain)
            # Let's proceed with computing an averaged gradient based on a few similar steps
            # Introduce a modeled update step
            
            # Momentum gradient update (this is heuristic)
            v = momentum * v - step_size * penalty_grad * np.exp(1j*phase)
            
            # Update frequency domain representation (complex magnitude)
            H = H * np.exp(1j * np.angle(v))
            H += v
            
            # Keep magnitude consistent with real signal  
            # For real signal, H(k) = H(-k)*, i.e. conjugate symmetry
            H_symm = (H + np.conj(np.flip(H[1:-1]))) / 2  # symmetry assumed for real values
            H = H_symm
            
            # Apply to increase stability
            if np.linalg.norm(v) > 1:
                v = v / np.linalg.norm(v)
        
        return h_best, c5_best
    
    # Main optimization loop
    resolutions = [15, 30, 45, 60]
    time_per_res = budget_s / len(resolutions)
    
    for n_points in resolutions:
        if time() - start_time > budget_s - 10:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), budget_s / 2)
        
        h_opt, c5_opt = Fourier_optimize(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n_points = n_points
            
        # For more resolution
        if n_points >= 15:
            # Interpolate best previous solution to new resolution and improve
            h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            h_interp = project_to_feasible(h_interp, n_points)
            h_opt, c5_opt = Fourier_optimize(h_interp, n_points, dx, time_per_res / 2)
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
                
    if best_c5 >= 0.381:
        # Try another approach using local optimization with re-starts
        # Reinitialize with better starting point
        h_zero = np.zeros(n_points)
        h_zero[n_points//4:3*n_points//4] = 1.0
        h_zero = project_to_feasible(h_zero, n_points)
        h_start = perturb_h(h_zero, 0.1)
        h_opt, c5_opt = augmented_lagrangian_optimize(h_start, n_points, dx, 10)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    return (list(best_h), best_c5, best_n_points)

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def augmented_lagrangian_optimize(h_start, n_points, dx, time_limit):
    """Optimize using augmented Lagrangian method"""
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    start_time = time()

    for iteration in range(500):
        if time() - start_time > time_limit:
            break
            
        # Perturb the current solution
        h_perturbed = perturb_h(h_best, 0.05)
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        try:
            from scipy.optimize import minimize
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj, h_perturbed, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
            
        if iteration % 20 == 0:
            h_global = perturb_h(h_best, 0.1)
            c5_global = evaluate_erdos_solution(h_global, dx)
            if c5_global < c5_best:
                h_best = h_global
                c5_best = c5_global

    return h_best, c5_best
