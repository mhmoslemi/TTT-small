import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Find initial feasible solution
    def solve_feasibility(h0, C):
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1]
        constraints.append(sum(h) == n_points / 2)
        
        # For each possible shift k, add constraint  
        for k in range(n_points):
            if k == 0:
                # Avoid self overlap - we actually don't need this since it's 0 anyway
                continue
            # Compute overlap for shift k
            # This is sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            expr = 0
            for i in range(n_points - k):
                expr += h[i] * (1 - h[i + k])
            constraints.append(expr <= C)
            
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            return h.value is not None and not np.isnan(h.value).any()
        except:
            return False
    
    # Binary search for the best C
    low, high = 0.37, 0.38085
    best_C = high
    best_h = h_start.copy()
    
    # First try the initial solution
    c5 = evaluate_erdos_solution(h_start, dx)
    if c5 < best_C:
        best_C = c5
        best_h = h_start.copy()
    
    iterations = 0
    while time.time() - start_time < budget_s * 0.9 and iterations < 10:
        mid = (low + high) / 2
        if solvable := solve_feasibility(h_start, mid):
            high = mid
            # Now try to refine the solution to minimize C5
            h_opt = Variable(n_points)
            constraints = [h_opt >= 0, h_opt <= 1, sum(h_opt) == n_points / 2]
            
            # Add overlap constraints
            for k in range(n_points):
                if k == 0:
                    continue
                expr = 0
                for i in range(n_points - k):
                    expr += h_opt[i] * (1 - h_opt[i + k])
                constraints.append(expr <= mid)
            
            # Objective is to minimize the maximum overlap integral
            prob = Problem(Minimize(0), constraints)
            try:
                prob.solve(solver=ECOS, verbose=False, max_iters=1000)
                if h_opt.value is not None and not np.isnan(h_opt.value).any():
                    h_new = h_opt.value
                    c5_refined = evaluate_erdos_solution(h_new, dx)
                    if c5_refined < best_C:
                        best_C = c5_refined
                        best_h = h_new.copy()
            except:
                pass
        else:
            low = mid
            
        iterations += 1
        
    # Final refinement with SQP
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Use projected gradient descent if better solution not found via CVX
    if time.time() - start_time < budget_s * 0.9:
        h_init = best_h
        dx_ref = 2.0 / n_points
        
        # Refine with local optimization
        from scipy.optimize import minimize
        def obj(h):
            return np.max(np.correlate(h, 1 - h, mode='full')) * dx_ref
            
        def constraint_eq(h):
            return np.sum(h) * dx_ref - 1
            
        cons = {'type': 'eq', 'fun': constraint_eq}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_init, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                c5_refined = evaluate_erdos_solution(h_refined, dx_ref)
                if c5_refined < best_C:
                    best_C = c5_refined
                    best_h = h_refined
        except:
            pass
            
    return (list(best_h), best_C, n_points)
