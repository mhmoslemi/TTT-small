import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover with alpha parameter"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, noise_std=0.1):
    """Apply Gaussian noise mutation to an individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(individual))
        individual += noise
    return individual

    mutation_rate = 0.2

    max_generations = 1000

    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        fitness = evaluate_erdos_solution(h, 2.0 / n_points)
        fitnesses.append(fitness)
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    generation = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        new_fitnesses = []
        
        # Elitism: keep top 25%
        elite_count = population_size // 4
        for i in range(elite_count):
            new_population.append(sorted_population[i].copy())
            new_fitnesses.append(sorted_fitnesses[i])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Projection to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
            new_fitnesses.extend([evaluate_erdos_solution(offspring1, 2.0 / n_points),
                                  evaluate_erdos_solution(offspring2, 2.0 / n_points)])
        
        # Trim to exact population size
        population = new_population[:population_size]
        fitnesses = new_fitnesses[:population_size]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)

            best_c5 = current_best_fitness
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Stop if no improvement for a while
        if stagnation_count > 20:
            break

    return (list(best_h), best_c5, n_points)
