import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    # Scale to meet exact sum constraint
    if abs(np.sum(h) - target_sum) > 1e-10:
        h *= target_sum / np.sum(h)
    
    return h

    swarm = []
    velocities = []

            particle = h0 + np.random.normal(0, 0.02, n_points)

        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Evaluate initial swarm
    fitness = [evaluate(p, 2.0/n_points) for p in swarm]
    
    # Track personal bests
    pbest = swarm.copy()
    pbest_fitness = fitness.copy()
    
    # Track global best
    gbest_idx = np.argmin(fitness)
    gbest = swarm[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # PSO iterations
    iter_count = 0
    while time.time() - start_time < budget_s and iter_count < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            vel_update = (w * velocities[i] + 
                         c1 * r1 * (pbest[i] - swarm[i]) + 
                         c2 * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            vel_update = np.clip(vel_update, -v_max, v_max)
            
            velocities[i] = vel_update
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Apply constraints
            swarm[i] = project_to_feasible(swarm[i], n_points / 2.0)
            
            # Evaluate fitness
            new_fitness = evaluate(swarm[i], 2.0/n_points)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = swarm[i].copy()
                pbest_fitness[i] = new_fitness
                
                # Update global best
                if new_fitness < gbest_fitness:
                    gbest = swarm[i].copy()
                    gbest_fitness = new_fitness
        
        iter_count += 1
