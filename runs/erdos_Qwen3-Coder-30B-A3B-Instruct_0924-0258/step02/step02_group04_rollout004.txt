```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, samples=50):
    """Generate Latin hypercube samples for initial population"""
    h_samples = []
    for _ in range(samples):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        h_samples.append(h)
    return h_samples

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Resolution schedule
    resolutions = [15, 30, 45, 60, 75]
    
    # Multi-resolution optimization
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Generate multiple initial samples via Latin hypercube
        initial_samples = latin_hypercube_sample(n_points, samples=10)
        
        # Optimization for each sample
        for h_start in initial_samples:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Local optimization with L-BFGS-B
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_fun(h):
                return np.sum(h) - n_points / 2
                
            cons = [{'type': 'eq', 'fun': constraint_fun}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Ensure feasibility
            h_start = project_to_feasible(h_start, n_points)
            
            try:
                res = minimize(obj, h_start, method='L-BFGS-B', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except:
                pass
            
            # Also try additional local search with perturbation
            for _ in range(3):
                if time() - start_time > budget_s * 0.95:
                    break
                h_pert = h_start + np.random.normal(0, 0.05, n_points)
                h_pert = project_to_feasible(h_pert, n_points)
                try:
                    res = minimize(obj, h_pert, method='L-BFGS-B', 
                                  bounds=bounds, constraints=cons, 
                                  options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-8})
                    if res.success:
                        h_opt = project_to_feasible(res.x, n_points)
                        c5_opt = evaluate_erdos_solution(h_opt, dx)
                        if c5_opt < best_c5:
                            best_h = h_opt.copy()
                            best_c5 = c5_opt
                            best_n_points = n_points
                except:
                    pass

    # Fallback to a simple construction if nothing found
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        # Create a piecewise function
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points)
        best_h = h
        best_c5 = evaluate_erdos_solution(h, dx)
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
```
