import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * target_sum / np.sum(h_rand)
        population.append(h_rand)
    
    # Evaluate initial population
    fitnesses = []
    for individual in population:
        fitness = -compute_c5(individual)
        fitnesses.append(fitness)
    
    best_index = np.argmin(fitnesses)
    best_h = population[best_index].copy()
    best_c5 = -fitnesses[best_index]
    
    start_time = time.time()
    
    # Genetic algorithm loop
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(population[winner_index].copy())
        
        # Create new population via crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        
        for i in range(1, population_size, 2):
            parent1 = selected[i-1]
            parent2 = selected[i]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Perturb some elements
                num_mutations = np.random.randint(1, 3)
                indices = np.random.choice(n_points, num_mutations, replace=False)
                for idx in indices:
                    child1[idx] += np.random.normal(0, 0.01)
            
            if np.random.rand() < mutation_rate:
                # Perturb some elements
                num_mutations = np.random.randint(1, 3)
                indices = np.random.choice(n_points, num_mutations, replace=False)
                for idx in indices:
                    child2[idx] += np.random.normal(0, 0.01)
            
            # Ensure constraints
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Replace population
        population = new_population[:population_size]
        
        # Evaluate new population
        fitnesses = []
        for individual in population:
            fitness = -compute_c5(individual)
            fitnesses.append(fitness)
            
            # Update best solution
            if -fitness < best_c5:
                best_c5 = -fitness
                best_h = individual.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
