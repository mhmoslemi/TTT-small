import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 30
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution if available
    if 'initial_h_values' in globals():
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Add random particles
    for i in range(swarm_size - len(swarm)):
        h_i = np.random.uniform(0, 1, n_points)
        h_i = project_feasible(h_i)
        swarm.append(h_i.copy())
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # PSO main loop
    best_global = float('inf')
    best_global_pos = None
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        w = w_start - (w_start - w_end) * (iterations / max_iter)
        
        for i, particle in enumerate(swarm):
            # Compute overlap for this particle
            overlap = compute_overlap(particle)
            
            # Update personal best
            if overlap < best_global:
                best_global = overlap
                best_global_pos = particle.copy()
                
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = c1 * r1 * (particle - particle)
            social = c2 * r2 * (best_global_pos - particle)
            
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particle += velocities[i]
            
            # Project to feasible region
            particle = project_feasible(particle)
            
        iterations += 1
    
    # Final check
    final_h = best_global_pos.copy()
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
