import numpy as np
import time
from scipy.ndimage import gaussian_filter1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies constraint
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h using full convolution
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h_init.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Evolve the population
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            c5 = compute_c5(h)
            fitnesses.append(-c5)  # Negative for minimization
        
        # Selection: tournament selection
        new_population = []
        for _ in range(population_size // 2):
            # Select two parents via tournament
            parent1_idx = np.argmin(np.random.choice(fitnesses, tournament_size))
            parent2_idx = np.argmin(np.random.choice(fitnesses, tournament_size))
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Mutate one or two random elements
                n_mutations = np.random.choice([1, 2])
                for _ in range(n_mutations):
                    idx = np.random.randint(0, n_points)
                    child1[idx] += np.random.normal(0, 0.01)
            
            if np.random.rand() < mutation_rate:
                n_mutations = np.random.choice([1, 2])
                for _ in range(n_mutations):
                    idx = np.random.randint(0, n_points)
                    child2[idx] += np.random.normal(0, 0.01)
            
            # Project to valid region
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            new_population.extend([child1, child2])
        
        # Replace population
        population = new_population[:population_size]
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
