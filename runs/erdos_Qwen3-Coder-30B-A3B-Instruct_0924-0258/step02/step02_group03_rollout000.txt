import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h *= scale
    return h

    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.05  # 5% of range
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        # Small perturbation around initial solution
        if 'initial_h_values' in globals():
            h = h0 + np.random.normal(0, 0.02, n_points)
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate(h, 2.0/n_points))
    
    # Global best
    best_idx = np.argmin(pbest_vals)
    gbest = pbest[best_idx].copy()
    gbest_val = pbest_vals[best_idx]
    
    # PSO iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = project_to_feasible(swarm[i], n_points / 2.0)
            
            # Evaluate
            current_val = evaluate(swarm[i], 2.0/n_points)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
        
        # Periodic logging
        if iter_num % 50 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 2:
                break
    
    return (gbest.tolist(), float(gbest_val), n_points)
