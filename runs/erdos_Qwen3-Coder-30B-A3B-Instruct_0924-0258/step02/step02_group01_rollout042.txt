import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_feasible(h_start, n_points)
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    penalty_weight = 1e-3
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_opt = np.argmax(corr)
        lag = k_opt - (len(h) - 1)
        
        # Compute gradient for the dominant lag
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + lag) % len(h)
            grad[i] = -(1 - h[j] - h[i])
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        penalty_grad += 2 * (np.sum(h) - n_points / 2)
        penalty_grad += 2 * np.clip(h, 0, 1) * (h > 1) - 2 * np.clip(h, 0, 1) * (h < 0)
        
        # Step candidate
        h_candidate = h - eta * (grad + penalty_weight * penalty_grad)
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            current_c5 = c5_candidate
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Annealing
        T *= alpha
        
        # Random perturbation if stuck
        if no_improvement_count >= max_no_improvement:
            h += np.random.randn(len(h)) * 0.05
            h = project_feasible(h, n_points)
            current_c5 = evaluate_erdos_solution(h, dx)
            no_improvement_count = 0
    
    # Multi-resolution refinement
    resolution_sweep = [70, 140]
    for new_n_points in resolution_sweep:
        if time() - start_time > budget_s * 0.9:
            break
        # Interpolate
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, new_n_points)

            h_candidate = project_feasible(h_candidate, new_n_points)

            if c5_candidate < current_c5:

                current_c5 = c5_candidate
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
    
    return (list(best_h), best_c5, len(best_h))
