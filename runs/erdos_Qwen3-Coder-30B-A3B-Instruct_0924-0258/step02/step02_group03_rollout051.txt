import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    scale = target_sum / current_sum
    h *= scale
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Particle Swarm Optimization implementation
    n_points = 70
    swarm_size = 40
    max_iter = 10000
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        # Start near the initial solution with small noise
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate(h, 2.0/n_points))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    # PSO parameters
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.05  # 5% of range

    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            new_velocity = (omega * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            new_velocity = np.clip(new_velocity, -max_velocity, max_velocity)
            
            # Update position
            new_position = particles[i] + new_velocity
            
            # Apply constraints
            new_position = project_to_feasible(new_position, n_points / 2.0)
            
            # Evaluate
            new_value = evaluate(new_position, 2.0/n_points)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = new_position.copy()
                pbest_values[i] = new_value
            
            # Store new particle and velocity
            particles[i] = new_position
            velocities[i] = new_velocity
            
            # Update global best
            if new_value < gbest_value:
                gbest = new_position.copy()
                gbest_value = new_value
        
        # Print progress every 1000 iterations
        if iteration % 1000 == 0:
            if time.time() - start_time > budget_s - 1:
                break

    return (gbest.tolist(), float(gbest_value), n_points)
