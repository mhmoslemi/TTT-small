import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.ones(n_points) * 0.5
    h0 = project(h0, n_points)
    
    population = []
    for i in range(pop_size):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project(h, n_points)
        population.append(h)
    return population

def tournament_selection(population, fitness, tournament_size=5):
    selected = []
    indices = np.random.choice(len(population), size=(len(population), tournament_size))
    for i in range(len(population)):
        best_idx = indices[i][np.argmin([fitness[j] for j in indices[i]])]
        selected.append(population[best_idx])
    return selected

def crossover(h1, h2):
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return offspring

def mutate(h, mutation_rate=0.2, sigma=0.1):
    if np.random.rand() < mutation_rate:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    max_generations = 2000
    stagnation_limit = 200
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Genetic algorithm loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create offspring
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected) and np.random.rand() < crossover_rate:
                child1 = crossover(selected[i], selected[i+1])
                child2 = crossover(selected[i+1], selected[i])
            else:
                child1 = selected[i].copy()
                child2 = selected[i+1].copy()
            offspring.extend([child1, child2])
        
        # Mutation
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i], mutation_rate, 0.1)
            
        # Evaluate offspring
        offspring_fitness = []
        for h in offspring:
            c5 = evaluate_erdos_solution(h, dx)
            offspring_fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Replace worst individuals
        combined_population = population + offspring
        combined_fitness = fitness + offspring_fitness
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_population[i] for i in sorted_indices[:pop_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Adaptive mutation rate
        if generation > 0 and generation % 50 == 0:
            improvement = np.mean(fitness[:pop_size//2]) - np.mean(fitness[pop_size//2:])
            if improvement < 1e-6:
                mutation_rate = min(mutation_rate * 1.2, 0.5)
                stagnation_count += 1
            else:
                stagnation_count = 0
        if stagnation_count > stagnation_limit:
            break
    
    # Ensure best_h is feasible
    best_h = project(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
