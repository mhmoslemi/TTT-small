import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def latin_hypercube_sampling(n_points, n_samples=50):
    """Generate Latin Hypercube samples"""
    samples = []
    for _ in range(n_samples):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        samples.append(h)
    return samples

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def optimize_single_h(h_start, n_points, dx, time_limit):
    """Optimize a single h using L-BFGS-B"""
    start_time = time()
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    best_h = h_start.copy()
    best_c5 = obj(h_start)
    
    # Run optimization with L-BFGS-B
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                      constraints=cons, options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_opt = project_to_feasible(res.x, n_points)
            c5_opt = obj(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
    except:
        pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Strategy: start with coarse grids and refine as we improve
    resolutions = [15, 30, 60, 75]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Sample multiple initial points to find a good start
        samples = latin_hypercube_sampling(n_points, 25)
        
        for h_sample in samples:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Optimize from this sample using L-BFGS-B
            h_opt, c5_opt = optimize_single_h(h_sample, n_points, dx, 
                                            budget_s/len(resolutions))
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
                
        # Occasionally try a larger noise to escape local minimum
        if n_points >= 30:
            for _ in range(10):
                if time() - start_time > budget_s * 0.95:
                    break
                h_perturbed = perturb_h(h_start, 0.2)
                h_opt, c5_opt = optimize_single_h(h_perturbed, n_points, dx, 
                                                budget_s/len(resolutions))
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
    
    if best_h is None:
        # Fall back to a small resolution optimized with L-BFGS-B
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_single_h(h_start, n_points, dx, budget_s)
        best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
