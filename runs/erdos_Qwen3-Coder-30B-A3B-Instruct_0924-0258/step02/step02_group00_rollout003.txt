import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach
    c_low = 0.37
    c_high = 0.38085
    c_tolerance = 1e-5
    max_iterations = 10
    
    # For each iteration of binary search
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
        c_mid = (c_low + c_high) / 2.0
        
        # Convex optimization to check feasibility
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Rolling sum for overlap constraint
            if k == 0:
                overlap_sum = cp.sum(h_var * (1 - h_var)) * dx
            else:
                overlap_sum = cp.sum(cp.multiply(h_var[:-k], 1 - h_var[k:])) * dx
            constraints.append(overlap_sum <= c_mid)
        
        # No objective needed - just feasibility check
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_var.value
                # Check if this is better
                if h_feasible is not None:
                    # Re-evaluate with our evaluator to make sure
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
            
    # Post-feasibility refinement with gradient-based optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        # Use scipy to fine-tune
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        # Optimize with bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
