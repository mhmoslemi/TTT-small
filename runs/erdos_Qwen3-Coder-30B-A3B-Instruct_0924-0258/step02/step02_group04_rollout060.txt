```python
import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m_particles=10):
    """Initialize a simplex of feasible particles"""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base)), base)
    else:
        base = np.ones(n_points) * 0.5
    
    particles = [project_to_feasible(base, n_points)]
    
    # Generate additional random particles in the feasible set
    for _ in range(m_particles - 1):
        pert = np.random.normal(0, 0.1, n_points)
        new_h = project_to_feasible(base + pert, n_points)
        particles.append(new_h)
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Settings
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    min_swarm_size = 10
    max_swarm_size = 50
    noise_level = 0.1
    
    # Initialize swarm
    particles = initialize_simplex(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    last_improvement = 0
    current_swarm_size = swarm_size
    
    # PSO loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(len(particles)):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest[i] - particles[i]) + 
                           social * r2 * (gbest - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate and update personal best
            current_value = evaluate_erdos_solution(particles[i], dx)
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    last_improvement = iteration
        
        # Shaking
        if iteration - last_improvement > shaking_threshold:
            for i in range(len(particles)):
                noise = np.random.normal(0, noise_level, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            
            last_improvement = iteration
        
        # Adaptive swarm size
        if iteration % 100 == 0 and current_swarm_size < max_swarm_size:
            current_swarm_size = min(current_swarm_size + 5, max_swarm_size)
            
        # L-BFGS refinement
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Select top candidates
            top_indices = np.argsort(pbest_values)[:5]
            for idx in top_indices:
                h0 = pbest[idx].copy()
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds, 
                                  options={'maxiter': 20, 'ftol': 1e-8})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        ref_value = evaluate_erdos_solution(h_refined, dx)
                        if ref_value < pbest_values[idx]:
                            pbest[idx] = h_refined
                            pbest_values[idx] = ref_value
                            
                            if ref_value < gbest_value:
                                gbest = h_refined
                                gbest_value = ref_value
                                last_improvement = iteration
                except:
                    pass
    
    # Final refinement of best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 50, 'ftol': 1e-10})
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < gbest_value:
                gbest = final_h
                gbest_value = final_c5
    except:
        pass
    
    return (list(gbest), gbest_value, n_points)
```
