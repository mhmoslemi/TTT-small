import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Get the valid range (shifts from -n+1 to n-1)
        # We want shifts from 0 to n-1 (positive shifts only)
        # The middle of the correlation corresponds to shift 0
        mid = len(corr) // 2
        autocorr_positive = corr[mid:]
        # Now, we want to compute h(x) * (1 - h(x+k))
        # This is similar to correlation but with 1-h
        corr_with_complement = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr_with_complement *= dx
        return np.max(corr_with_complement)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_gradients(h_vec, s_min):
        # Compute gradient of A_s_min w.r.t. h
        grad = np.zeros_like(h_vec)
        # For shift s, d/dh[i] (h[i] * h[i+s]) = h[i+s] if i+s < n, 0 otherwise
        # But since we have h[i] * (1 - h[i+s]), we have: 
        # d/dh[i] = -h[i+s] if i+s < n, 0 otherwise
        # d/dh[i+s] = -h[i] if i >= 0, 0 otherwise
        s = s_min
        for i in range(len(h_vec)):
            if i + s < len(h_vec):
                grad[i] -= h_vec[i + s]
            if i - s >= 0:
                grad[i] -= h_vec[i - s]
        return grad

    # Start optimization
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 50000
    iteration = 0
    last_improvement = 0
    learning_rate = 0.01
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        # Compute autocorrelations
        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        s_min = np.argmax(corr) - (len(h) - 1)  # Convert back to shift index

        # Compute gradient and update
        grad = compute_gradients(h, abs(s_min))
        h = h + learning_rate * grad

        # Project
        h = project_h(h)

        # Check if we have improved
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        if last_improvement > 300:
            learning_rate *= 0.9
            last_improvement = 0

        if iteration % 500 == 0:
            pass  # For debugging if needed

        iteration += 1
        if iteration > max_iter:
            break

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
