import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_feasible(h, n_points)
    
    # Parameters
    T = 0.5  # Initial temperature
    alpha = 0.995  # Cooling rate
    eta = 0.01  # Learning rate
    lambda_penalty = 1e-3  # Penalty weight
    max_iter_no_improve = 50
    iteration_no_improve = 0
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing
    while time() - start_time < budget_s * 0.95:
        # Compute current C5 and dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (len(h) - 1)
        
        # Compute gradient for dominant lag
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k) % len(h)
            grad[i] = -(1 - h[j] - h[i])
        
        # Compute penalty gradient
        penalty_grad = np.zeros_like(h)
        sum_h = np.sum(h)
        penalty_grad += 2 * lambda_penalty * (sum_h - n_points/2) * np.ones_like(h)
        penalty_grad += lambda_penalty * (h > 1) * (h - 1)
        penalty_grad += lambda_penalty * (h < 0) * h
        
        # Update h
        h_candidate = h + eta * (-grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - best_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                iteration_no_improve = 0
            else:
                iteration_no_improve += 1
        else:
            iteration_no_improve += 1
            
        # Temperature cooling
        T *= alpha
        
        # Random perturbation if no improvement
        if iteration_no_improve > max_iter_no_improve:
            noise = np.random.normal(0, 0.05, len(h))
            h = project_feasible(h + noise, n_points)
            iteration_no_improve = 0
    
    # Multi-resolution refinement
    resolutions = [60, 120]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, len(best_h)), best_h)
        h_new = project_feasible(h_new, res)
        dx_new = 2.0 / res
        
        # Local improvement
        for _ in range(50):
            corr = np.correlate(h_new, 1 - h_new, mode='full')
            max_index = np.argmax(corr)
            k = max_index - (len(h_new) - 1)
            
            grad = np.zeros_like(h_new)
            for i in range(len(h_new)):
                j = (i + k) % len(h_new)
                grad[i] = -(1 - h_new[j] - h_new[i])
                
            h_new = project_feasible(h_new + 0.005 * (-grad), res)
            
            new_c5 = evaluate_erdos_solution(h_new, dx_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                n_points = res

    return (list(best_h), best_c5, n_points)
