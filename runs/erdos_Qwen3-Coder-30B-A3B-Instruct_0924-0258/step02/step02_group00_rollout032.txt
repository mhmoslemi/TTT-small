import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    max_iter = 40
    iter_count = 0
    
    # Use CVXPY for convex optimization
    solver = 'ECOS'
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.9:
        C_mid = (C_low + C_high) / 2.0
        
        # Create CVXPY variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2.0,  # Integral constraint
            h >= 0,  # Lower bound
            h <= 1   # Upper bound
        ]
        
        # Correlation constraints
        for k in range(-n_points + 1, n_points):
            # Circular indexing
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h[i] * (1 - h[j])
            constraints.append(corr_sum <= C_mid * n_points / 2.0)
        
        # Problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Solve with timeout protection
        try:
            prob.solve(solver=solver, feastol=1e-6, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                c5_trial = evaluate_erdos_solution(h_sol, dx)
                if c5_trial < best_c5:
                    best_h = h_sol.copy()
                    best_c5 = c5_trial
                    best_n_points = n_points
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
            
        iter_count += 1
        
        if (time() - start_time) > budget_s * 0.8:
            break
    
    # Refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Use local gradient refinement
        def objective(h):
            # Return negative of correlation (since we want to minimize)
            corr = np.correlate(h, 1 - h, mode='full')
            return -np.max(corr) * dx
        
        def constraint_func(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Perform several local optimizations
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Small perturbations
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            try:
                res = minimize(objective, h_pert, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except:
                pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, best_n_points)
