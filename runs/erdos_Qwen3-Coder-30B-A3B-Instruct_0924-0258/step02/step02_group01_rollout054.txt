import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def initialize_population(n_points, population_size, initial_h_values=None):
    """Initialize population with feasible individuals"""
    population = []
    if initial_h_values is not None:
        h_start = np.array(initial_h_values)
        population.append(project_feasible(h_start, n_points))
    else:
        h_start = np.ones(n_points) * 0.5
        population.append(project_feasible(h_start, n_points))
    
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        population.append(project_feasible(h_rand, n_points))
    
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    selected_idx = np.random.choice(len(population), k)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Mutate an individual"""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        mutated = individual + noise
        return project_feasible(mutated, len(individual))
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = 1000
    
    # Get initial values
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, population_size, 
                                      None if 'initial_h_values' not in globals() 
                                      else initial_h_values)
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolve
    last_improvement = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.98 and generation < max_generations:
        generation += 1
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # New population
        new_population = []
        
        # Elitism: keep best 25%
        elite_count = population_size // 4
        new_population.extend(sorted_population[:elite_count])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            new_population.extend([offspring1, offspring2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Evaluate fitness
        new_fitnesses = [evaluate_erdos_solution(ind, dx) for ind in new_population]
        
        # Update best
        min_idx = np.argmin(new_fitnesses)
        if new_fitnesses[min_idx] < best_c5:
            best_c5 = new_fitnesses[min_idx]
            best_h = new_population[min_idx].copy()
            last_improvement = generation
        
        # Replace old population
        population = new_population
        fitnesses = new_fitnesses
    
    return (list(best_h), best_c5, n_points)
