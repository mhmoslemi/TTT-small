import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation for upsample
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    def compute_fourier_coeffs(h_vec):
        # Compute discrete Fourier transform
        f = np.fft.fft(h_vec)
        return f.real, f.imag
    
    def reconstruct_from_fourier(a_real, a_imag):
        # Reconstruct signal from Fourier coeffs
        n = len(a_real)
        # We only store the first half of the coefficients (symmetric)
        f_coeffs = np.zeros(n, dtype=complex)
        f_coeffs[0] = a_real[0] + 0j
        f_coeffs[1:n//2] = (a_real[1:n//2] + 1j * a_imag[1:n//2])
        if n % 2 == 0:
            f_coeffs[n//2] = a_real[n//2] + 0j
        else:
            f_coeffs[n//2] = a_real[n//2] + 1j * a_imag[n//2]
        # IFFT to get signal
        h = np.fft.ifft(f_coeffs).real
        return h
    
    # Multi-resolution optimization with Fourier coefficients
    n_fourier = 15  # Number of frequency components
    max_iterations_per_res = 5000
    
    # Intialize
    def initialize_fourier(res):
        n = res
        a_real = np.zeros(n)
        a_imag = np.zeros(n)
        # Use initial_h_values if available
        try:
            h_init = np.array(initial_h_values)
            if len(h_init) == n:
                a_real, a_imag = compute_fourier_coeffs(h_init)
            else:
                # Interpolate to n points
                h_interp = upsample_h(h_init, n)
                a_real, a_imag = compute_fourier_coeffs(h_interp)
        except NameError:
            # Simple initialization
            pass
        
        # Perturb slightly
        a_real += np.random.normal(0, 0.1, n)
        a_imag += np.random.normal(0, 0.1, n)
        
        return a_real.copy(), a_imag.copy()
    
    # Adam optimizer variables
    def initialize_adam_vars(shape):
        return np.zeros(shape), np.zeros(shape)
    
    def adam_update(m, v, grad, beta1, beta2, lr, t):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        return m, v, m_hat / (np.sqrt(v_hat) + 1e-8)
    
    current_a_real, current_a_imag = initialize_fourier(n_points)
    
    # Main optimization loop
    iteration = 0
    best_val = float('inf')
    best_a_real, best_a_imag = current_a_real.copy(), current_a_imag.copy()
    
    # Optimization parameters
    adam_lr = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    
    # Store history for convergence check
    history = []
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations_per_res * 10:
        # Reconstruct h from current coefficients
        h = reconstruct_from_fourier(current_a_real, current_a_imag)
        h = project_feasible(h)
        
        # Compute C5
        c5 = compute_overlap(h)
        
        # Store best
        if c5 < best_val:
            best_val = c5
            best_a_real, best_a_imag = current_a_real.copy(), current_a_imag.copy()
        
        # Gradient computation using finite differences
        gradient_real = np.zeros_like(current_a_real)
        gradient_imag = np.zeros_like(current_a_imag)
        eps = 1e-4
        
        for k in range(len(current_a_real)):
            # Perturb real part
            a_real_plus = current_a_real.copy()
            a_real_plus[k] += eps
            h_plus = reconstruct_from_fourier(a_real_plus, current_a_imag)
            h_plus = project_feasible(h_plus)
            c_plus = compute_overlap(h_plus)
            
            a_real_minus = current_a_real.copy()
            a_real_minus[k] -= eps
            h_minus = reconstruct_from_fourier(a_real_minus, current_a_imag)
            h_minus = project_feasible(h_minus)
            c_minus = compute_overlap(h_minus)
            
            gradient_real[k] = (c_plus - c_minus) / (2 * eps)
            
            # Perturb imaginary part
            a_imag_plus = current_a_imag.copy()
            a_imag_plus[k] += eps
            h_plus = reconstruct_from_fourier(current_a_real, a_imag_plus)
            h_plus = project_feasible(h_plus)
            c_plus = compute_overlap(h_plus)
            
            a_imag_minus = current_a_imag.copy()
            a_imag_minus[k] -= eps
            h_minus = reconstruct_from_fourier(current_a_real, a_imag_minus)
            h_minus = project_feasible(h_minus)
            c_minus = compute_overlap(h_minus)
            
            gradient_imag[k] = (c_plus - c_minus) / (2 * eps)
        
        # Adam update
        iteration += 1
        m_real, v_real, g_real = adam_update(
            np.zeros_like(current_a_real),
            np.zeros_like(current_a_real),
            gradient_real,
            adam_beta1,
            adam_beta2,
            adam_lr,
            iteration
        )
        m_imag, v_imag, g_imag = adam_update(
            np.zeros_like(current_a_imag),
            np.zeros_like(current_a_imag),
            gradient_imag,
            adam_beta1,
            adam_beta2,
            adam_lr,
            iteration
        )
        
        # Apply updates
        current_a_real -= adam_lr * g_real
        current_a_imag -= adam_lr * g_imag
        
        history.append(c5)
        if len(history) > 10:
            history = history[-10:]
        
        # Check for convergence
        if len(history) >= 10 and abs(history[-1] - history[0]) < 1e-6:
            break
    
    # Final best solution
    h_final = reconstruct_from_fourier(best_a_real, best_a_imag)
    h_final = project_feasible(h_final)
    final_C5 = compute_overlap(h_final)
    
    return (h_final.tolist(), final_C5, n_points)
