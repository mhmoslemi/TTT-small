import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, population_size):
    """Initialize population using initial_h_values if available"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(population_size):
        h = project(base_h + np.random.normal(0, 0.1, n_points), n_points)
        population.append(h)
    return population

def crossover(h1, h2):
    """Perform two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return offspring

def mutate(h, mutation_rate=0.2, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < mutation_rate:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
    return project(h, len(h))

def tournament_selection(population, fitness, tournament_size=5):
    """Select best individual from tournament"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    population_size = 50
    n_points = 30
    dx = 2.0 / n_points
    max_generations = 2000
    stagnation_limit = 200
    mutation_rate = 0.2
    sigma = 0.1
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_h = population[np.argmin(fitness)]
    best_c5 = min(fitness)
    best_n_points = n_points
    
    gen = 0
    stagnation_count = 0
    last_best = best_c5
    
    # Evolution loop
    while gen < max_generations:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Tournament selection
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            offspring = crossover(parent1, parent2)
            
            # Mutation
            offspring = mutate(offspring, mutation_rate, sigma)
            
            new_population.append(offspring)
        
        # Evaluate new population
        new_fitness = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Replace worst individuals
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_population[i] for i in sorted_indices[:population_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:population_size]]
        
        # Track best
        current_best = min(fitness)
        if current_best < best_c5:
            best_h = population[np.argmin(fitness)]
            best_c5 = current_best
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive mutation rate
        if stagnation_count > 50:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
        else:
            mutation_rate = 0.2
            
        gen += 1
        
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            break
    
    return (list(best_h), best_c5, best_n_points)
