import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial setup
    n_points = 35
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        population.append(h_start.copy())
        # Generate random individuals
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    else:
        # Generate initial population
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
        
    dx = 2.0 / n_points
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_idx = np.random.choice(pop_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_idx]
            winner_idx = tournament_idx[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover (arithmetic)
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
                
            # Re-project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            offspring.extend([child1, child2])
        
        # Keep only pop_size individuals
        offspring = offspring[:pop_size]
        
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Survival (elitism + replacement)
        # Keep best 25% from current pop
        elite_indices = np.argsort(fitness)[:pop_size//4]
        new_pop = [population[i].copy() for i in elite_indices]
        new_fitness = [fitness[i] for i in elite_indices]
        
        # Add mutated versions of elite
        for i in range(pop_size//4, pop_size//2):
            elite_idx = np.random.choice(elite_indices)
            mutant = population[elite_idx].copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.05, n_points)
                mutant += noise
            mutant = project_feasible(mutant, n_points)
            new_pop.append(mutant)
            new_fitness.append(evaluate_erdos_solution(mutant, dx))
        
        # Add offspring
        for i in range(len(offspring)):
            if len(new_pop) < pop_size:
                new_pop.append(offspring[i])
                new_fitness.append(offspring_fitness[i])
        
        # Trim to pop_size
        if len(new_pop) > pop_size:
            # Keep best
            indices = np.argsort(new_fitness)[:pop_size]
            new_pop = [new_pop[i] for i in indices]
            new_fitness = [new_fitness[i] for i in indices]
        
        population = new_pop
        fitness = new_fitness
        
        # Update best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitness[min_idx]
    
    return (list(best_h), best_c5, n_points)
