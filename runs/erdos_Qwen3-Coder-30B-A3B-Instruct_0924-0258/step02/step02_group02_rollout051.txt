import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_autocorrelations(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Take only valid shifts for the overlap computation
        n = len(h_vec)
        valid_corr = corr[n-1:]
        return valid_corr

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def objective(h_vec):
        return compute_c5(h_vec)

    def gradient_wrt_shift(h_vec, shift):
        # Compute gradient of A_shift = sum h[i] * h[i+shift]
        grad = np.zeros_like(h_vec)
        if shift >= 0:
            # h[i] * h[i+shift] for i in range(n-shift)
            grad[:-shift] += h_vec[shift:]
        else:
            grad[-shift:] += h_vec[:shift]
        return grad

    # Stochastic gradient-based optimization
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    lr = 0.01
    patience = 0
    max_patience = 200
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Find the shift with minimum autocorrelation
        auto_corr = compute_autocorrelations(h)
        min_shift = np.argmin(auto_corr)
        min_val = auto_corr[min_shift]
        
        # Compute gradient w.r.t. the minimal shift
        grad = gradient_wrt_shift(h, min_shift)
        # Use a learning rate that decreases over time
        h = h + lr * grad
        h = project_h(h)
        
        # Check if we improved
        curr_c5 = compute_c5(h)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
            patience = 0
            lr = 0.01  # Reset learning rate
        else:
            patience += 1
            if patience > max_patience:
                # Reduce learning rate
                lr *= 0.9
                patience = 0
            # Small random perturbation to escape local minima
            h = h + 0.0001 * np.random.normal(0, 1, n_points)
            h = project_h(h)
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
