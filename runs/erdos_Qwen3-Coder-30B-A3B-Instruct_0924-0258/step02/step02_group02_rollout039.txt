import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

        n_points = 45

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_shifted_overlaps(h_vec):
        # Compute all overlaps for shifts 0 to n_points-1
        overlaps = []
        for shift in range(n_points):
            overlap = 0
            for i in range(n_points - shift):
                overlap += h_vec[i] * (1 - h_vec[i + shift]) * dx
            overlaps.append(overlap)
        return np.array(overlaps)

    def update_h_for_shift(h_vec, shift, eta):
        # Compute gradient of overlap w.r.t h for a given shift
        grad = np.zeros_like(h_vec)
        for i in range(n_points - shift):
            grad[i] += (1 - h_vec[i + shift]) * dx
            grad[i + shift] -= h_vec[i] * dx
        # Gradient descent step
        step_size = 0.01
        updated = h_vec - step_size * grad
        # Project back to feasible region
        updated = project_h(updated)
        return updated

    # Start optimization
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    patience = 200
    learning_rate = 0.01

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlaps for all shifts
        overlaps = compute_shifted_overlaps(h)
        current_max = np.max(overlaps)
        
        # Update best solution if we find a better one
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > patience:
                # Try a random perturbation to escape local minima
                noise = 0.001 * np.random.normal(0, 1, n_points)
                h = h + noise
                h = project_h(h)
                no_improvement_count = 0
        
        # Optimize for the shift that currently causes maximum overlap
        max_shift = np.argmax(overlaps)
        # Update h to reduce overlap for this shift
        h = update_h_for_shift(h, max_shift, current_max)
        
        # Reduce learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.1
            
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
