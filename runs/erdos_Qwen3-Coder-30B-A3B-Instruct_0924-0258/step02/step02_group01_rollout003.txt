import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_select(population, fitnesses, k=5):
    """Perform tournament selection"""
    selected_idx = np.random.choice(len(population), k)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, noise_std=0.1):
    """Apply Gaussian mutation to individual"""
    mask = np.random.random(len(individual)) < mutation_rate
    if not np.any(mask):
        mask[np.random.randint(len(individual))] = True
    individual = individual.copy()
    individual[mask] += np.random.normal(0, noise_std, np.sum(mask))
    return individual

def genetic_optimize(h_start, n_points, dx, time_limit):
    """Optimize using genetic algorithm"""
    start_time = time()
    
    # Parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    generations = 0
    stagnation_count = 0
    max_stagnation = 20
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_random = np.random.random(n_points)
        population.append(project_feasible(h_random, n_points))
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Evolve
    while time() - start_time < time_limit:
        generations += 1
        if stagnation_count >= max_stagnation:
            break
            
        # Selection
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Feasibility projection
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(ind, dx) for ind in new_population]
        
        # Survival: combine and keep best
        combined_population = population + new_population
        combined_fitnesses = fitnesses + new_fitnesses
        
        # Keep top 25% and fill remaining slots
        sorted_indices = np.argsort(combined_fitnesses)
        survival_indices = sorted_indices[:population_size]
        
        # Select survivors
        population = [combined_population[i] for i in survival_indices]
        fitnesses = [combined_fitnesses[i] for i in survival_indices]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
    
    return best_h, best_c5

    resolutions = [25, 30, 35]
    best_h_all = h_start.copy()
    best_c5_all = best_c5
    
    for n_points_res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if len(h_start) != n_points_res:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points_res)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = project_feasible(h_interp, n_points_res)
        
        dx_res = 2.0 / n_points_res
        remaining_time = budget_s - (time() - start_time)
        time_per_res = remaining_time / len(resolutions)
        
        # Run genetic optimization
        h_opt, c5_opt = genetic_optimize(h_start, n_points_res, dx_res, time_per_res * 0.8)
        
        if c5_opt < best_c5_all:
            best_c5_all = c5_opt
            best_h_all = h_opt.copy()
            n_points = n_points_res
            
        h_start = h_opt.copy()
        
    # Final validation
    best_h_all = project_feasible(best_h_all, n_points)
    final_c5 = evaluate_erdos_solution(best_h_all, 2.0 / n_points)
    
    return (list(best_h_all), final_c5, n_points)
