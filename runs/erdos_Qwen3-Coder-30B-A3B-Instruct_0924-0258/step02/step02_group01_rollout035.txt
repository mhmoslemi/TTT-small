import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    dx = 2.0 / n_points
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for the maximizing lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h[j]) - h[i]
        
        grad *= dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        t = iteration + 1
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update parameters
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h = project_feasible(h, n_points)
        
        # Check for improvement
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = iteration
            
        # Restart if no improvement in last 100 iterations
        if iteration - last_improvement > 100:
            h = project_feasible(h + np.random.normal(0, 0.05, n_points), n_points)
            last_improvement = iteration
            
        iteration += 1
        
        # Occasionally refine resolution
        if iteration % 500 == 0 and n_points < 100:
            n_points = min(n_points * 2, 100)
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
            h = project_feasible(h, n_points)
            dx = 2.0 / n_points
            # Reset optimizer states
            m = np.zeros(n_points)
            v = np.zeros(n_points)
    
    return (list(best_h), best_c5, n_points)
