import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize with available construction or default
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    
    # Initialize best solution
    h_current = project_feasible(h_start.copy(), n_points)
    c5_current = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Track iterations without improvement
    no_improvement = 0
    max_no_improvement = 50
    
    # Simulated annealing loop
    while time() - start_time < budget_s * 0.95:
        # Find dominant lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_lag_idx = np.argmax(corr)
        dominant_lag = dominant_lag_idx - (n_points - 1)
        
        # Compute gradient for dominant lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + dominant_lag) % n_points
            grad[i] = -(1 - h_current[j] - h_current[i])
        
        # Compute penalty gradient
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(h_current)
        penalty_grad += 2 * lambda_penalty * (h_sum - n_points / 2)
        penalty_grad += 2 * lambda_penalty * np.clip(h_current, 0, 1) * (h_current > 1) 
        penalty_grad -= 2 * lambda_penalty * np.clip(h_current, 0, 1) * (h_current < 0)
        
        # Step
        h_candidate = h_current - eta * (grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta_c5 = c5_candidate - c5_current
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.random():
            h_current = h_candidate
            c5_current = c5_candidate
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1
            
        # Cool down
        T *= alpha
        
        # Random perturbation if stuck
        if no_improvement >= max_no_improvement:
            h_current = project_feasible(h_current + np.random.randn(n_points) * 0.05, n_points)
            no_improvement = 0
            
        # Occasionally increase resolution
        if np.random.random() < 0.01 and n_points < 100:
            # Double the resolution
            n_points_new = min(n_points * 2, 100)
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_interp = np.interp(x_new, x_old, h_current)
            h_current = project_feasible(h_interp, n_points_new)
            n_points = n_points_new
            dx = 2.0 / n_points
            c5_current = evaluate_erdos_solution(h_current, dx)
    
    return (list(best_h), best_c5, n_points)
