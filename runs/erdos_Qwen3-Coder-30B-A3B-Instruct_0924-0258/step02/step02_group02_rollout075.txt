import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_min_autocorr_and_grad(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Compute all correlations with (1-h)
        corr_h_complement = np.correlate(h_vec, 1 - h_vec, mode='full')
        # We just need the correlation values where h and (1-h) overlap
        # The overlap length is n_points + n_points - 1 = 2*n_points - 1
        # The range of k values is -n_points+1 to n_points-1
        # For k >= 0, we compute sum(h[i] * (1-h[i+k]))
        # For k < 0, we compute sum(h[i+k] * (1-h[i]))
        # So we compute total correlation matrix for all shifts 
        # and extract the negative of the correlation with complement
        # The overlap integral C5 = max_k \int h(x)(1-h(x+k)) dx = max_k corr_h_complement[k + (n_points-1)]
        corr_values = corr_h_complement[n_points-1:]  # only positive lags
        min_idx = np.argmin(corr_values)
        min_corr = corr_values[min_idx]
        # Compute gradient of this specific correlation
        grad = np.zeros_like(h_vec)
        # Correlation at shift k (0-based index k+n_points-1 in corr_h_complement)
        # For k=0 (first element), it's correlation with 1-h[0];
        # for k=1, it's correlation with 1-h[1];
        # etc. But we want h[i]*(1-h[i+k])
        k = min_idx
        for i in range(len(h_vec)):
            if i+k < len(h_vec):
                # d/dh_i [ h[i] * (1-h[i+k]) ] = (1-h[i+k]) 
                grad[i] += (1 - h_vec[i+k])
        return min_corr, grad, min_idx

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Use scipy minimize with SLSQP for better optimization
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Early termination counter
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    improvement_streak = 0
    learning_rate = 0.01
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        min_corr, grad, min_idx = compute_min_autocorr_and_grad(h)
        
        # Optimization step: try to increase the minimum correlation by adjusting h
        h_new = h - learning_rate * grad
        
        # Project to feasible values
        h_new = project_h(h_new)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            improvement_streak = 0
            h = h_new
        else:
            improvement_streak += 1
            # Add noise to escape local minima if no progress
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
            
        # Adjust learning rate
        if improvement_streak > 100:
            learning_rate *= 0.5
            improvement_streak = 0
            
        # If no improvement in a while, adjust the approach
        if improvement_streak > 500:
            # Try random restart
            h = np.random.rand(n_points) * 0.5
            h = project_h(h)
            improvement_streak = 0
        
        if improvement_streak > max_iter:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
