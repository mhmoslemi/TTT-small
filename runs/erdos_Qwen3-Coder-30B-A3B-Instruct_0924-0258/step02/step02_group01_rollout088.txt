import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    # Optimization loop
    max_iter = 10000
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        t += 1
        
        # Compute full correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for this lag only
        grad = np.zeros(n_points)
        for j in range(n_points):
            idx = (j + k) % n_points
            grad[j] = (1 - best_h[idx]) - best_h[j]
        grad *= dx
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update h
        best_h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        best_h = project_feasible(best_h, n_points)
        
        # Evaluate and update best
        c5 = evaluate_erdos_solution(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            
        # Restart with perturbation if stuck
        if i % 500 == 0 and i > 0:
            noise = np.random.normal(0, 0.05, n_points)
            h_perturbed = best_h + noise
            h_perturbed = project_feasible(h_perturbed, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
