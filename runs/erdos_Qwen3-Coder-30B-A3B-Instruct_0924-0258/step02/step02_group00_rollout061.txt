import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    exp_x = np.exp(alpha * x)
    sum_exp = np.sum(exp_x)
    return alpha * exp_x / sum_exp

def correlate_grad(h, n_points):
    # Return gradient of correlation with respect to h
    # This is a simplified version; full gradient computation 
    # would involve a more complex convolution gradient
    corr = np.correlate(h, 1 - h, mode='full')
    smooth_max_val = smooth_max(corr, alpha=100)
    grad_corr = smooth_max_grad(corr, alpha=100)
    
    # Simplified gradient, approximate for large dimension  
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Forward difference approximation of gradient
        eps = 1e-6
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        smooth_plus = smooth_max(corr_plus, alpha=100)
        grad[i] = (smooth_plus - smooth_max_val) / eps
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    def obj(h):
        h = project_to_feasible(h, n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return smooth_max(corr, alpha=100)
    
    def grad(h):
        h = project_to_feasible(h, n_points)
        return smooth_max_grad(np.correlate(h, 1 - h, mode='full'), alpha=100)
    
    # Optimization parameters
    learning_rate = 0.01
    # Run gradient steps with adaptive learning rate
    iteration = 0
    while time() - start_time < time_limit:
        h_new = best_h - learning_rate * grad(best_h)
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Reduce learning rate if we have improved
            learning_rate = max(0.001, learning_rate * 0.99)
        else:
            # Increase learning rate if no improvement
            learning_rate = min(0.1, learning_rate * 1.01)
        iteration += 1
    return best_h, best_c5
