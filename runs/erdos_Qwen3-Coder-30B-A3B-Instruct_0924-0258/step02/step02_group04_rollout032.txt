import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population with variations of initial_h_values or random"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for i in range(pop_size):
        if i == 0:
            h = base_h.copy()
        else:
            noise = np.random.normal(0, 0.1, n_points)
            h = base_h + noise
        h = project(h, n_points)
        population.append(h)
    return population

def mutate(h, mutation_rate=0.2, sigma=0.1):
    """Apply mutation with given rate and noise level"""
    if np.random.rand() < mutation_rate:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2, crossover_rate=0.7):
    """Apply crossover between two individuals"""
    if np.random.rand() < crossover_rate:
        idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
        offspring = np.copy(h1)
        offspring[idx1:idx2] = h2[idx1:idx2]
        return project(offspring, len(offspring))
    return h1

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), size=tournament_size, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    n_points = 30
    pop_size = 50
    max_generations = 2000
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    best_h = None
    best_c5 = float('inf')
    
    # Tracking variables
    stagnation_count = 0
    mutation_rate = 0.2
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
        
        # Adaptive mutation rate
        if generation % 50 == 0 and generation > 0:
            improvement = np.mean(fitnesses) - np.min(fitnesses)
            if improvement < 1e-6:
                stagnation_count += 1
                if stagnation_count > 5:
                    mutation_rate = min(mutation_rate * 1.2, 0.5)
            else:
                stagnation_count = 0
        
        # Selection and reproduction
        new_population = []
        for _ in range(pop_size):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            offspring = crossover(parent1, parent2)
            offspring = mutate(offspring, mutation_rate)
            new_population.append(offspring)
        
        population = new_population
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_erdos_solution(best_h, dx)
        best_c5 = min(best_c5, final_c5)
    else:
        # Fallback evaluation
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
