import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, tol=1e-6):
    """Frank-Wolfe optimization for minimizing the maximum overlap"""
    h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    prev_c5 = float('inf')
    
    for iteration in range(max_iter):
        if abs(prev_c5 - best_c5) < tol:
            break
        prev_c5 = best_c5
        
        # Evaluate all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Compute the subgradient for this max overlap
        g = np.zeros_like(h)
        k = max_idx - (n_points - 1)  # shift for correlation
        
        # Derivative of overlap w.r.t. h[i] is (1 - h[(i - k) % n]) - h[i]
        # For all i: dC/dh[i] = sum_k of (1 - h[(i - k) % n]) - h[i] where k is such that 
        # the correlation achieves max
        # Let's directly compute subgradient for max overlapping case
        for i in range(n_points):
            if (i - k) % n_points < n_points:
                j = (i - k) % n_points
                g[i] += (1 - h[j]) - h[i]
        
        # Solve linear optimization subproblem to find descent direction
        # min_s g^T s s.t. sum(s) = 0, 0 <= s <= 1
        # We can solve this with a simple approach: find s such that g^T s is minimized
        # Among all s where sum(s) = 0, 0 <= s <= 1
        
        # Let's use a very simple greedy choice where s[i] = 0 if g[i] <= 0 or s[i] = 1 otherwise
        # But we must satisfy sum(s) = 0  
        # This requires more care: we need to change sign of the gradient or reduce to feasible set
        # The correct way is to solve:
        # s[i] = max(0, min(1, -g[i])) or similar
        # But in fact, the standard Frank-Wolfe step is:
        # s = argmin_s g^T s s.t. s >= 0, sum(s) = 0 use Lagrange multipliers to fix the linear constraint.
        # For constraint sum = 0, this only makes sense if we move from center.
        # In this case, we look at s[i] = -g[i] to minimize the linearized objective

        s = -g 
        s = s - np.mean(s)  # Ensure it's centered
        s = np.clip(s, 0, 1)  # Clip but not sure if this is right...
        # Re-center s 
        s = s - np.mean(s)
        # Take a small positive step towards s
        gamma = 0.5
        h_new = h + gamma * (s - h)
        h_new = project_to_feasible(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            gamma *= 0.5
            h_new = h + gamma * (s - h)
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                
    return h, best_c5
