import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a structured initial guess with low correlation properties
        h = np.zeros(n_points)
        # Place 1s equidistantly to spread out correlations
        ones_positions = np.linspace(0, n_points-1, n_points//2, dtype=int)
        h[ones_positions] = 1.0
        # Normalize to required sum
        h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use a multi-resolution approach
    resolutions = [15, 30, 45, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Time tracking
    remaining_time = budget_s
    
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Allocate time for this resolution
        if res_idx == len(resolutions) - 1:
            time_per_res = remaining_time * 0.8
        else:
            time_per_res = remaining_time / (len(resolutions) - res_idx)
        
        # Start with several random restarts to avoid local optima
        h_best = h_start.copy()
        c5_best = evaluate_erdos_solution(h_best, dx)
        
        # Try 3 different random restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Add controlled noise
            noise = np.random.normal(0, 0.05, n_points)
            h_restart = project_to_feasible(h_best + noise, n_points)
            
            # lbfgs optimization with bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            try:
                res = minimize(obj, h_restart, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < c5_best:
                        c5_best = c5_opt
                        h_best = h_opt.copy()
            except:
                pass
                
            # Early termination check
            if time() - start_time > budget_s * 0.95:
                break
        
        # Keep track of the best solution so far
        if c5_best < best_c5:
            best_h = h_best.copy()
            best_c5 = c5_best
            best_n_points = n_points
            
        # Decrease remaining time
        elapsed = time() - start_time
        remaining_time = max(0, budget_s - elapsed)
    
    # Final optimization on the best result
    if best_h is not None:
        dx = 2.0 / best_n_points
        h_final = best_h.copy()
        c5_final = best_c5
        
        # Final optimization
        bounds = [(0, 1) for _ in range(best_n_points)]
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        try:
            res = minimize(obj, h_final, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_opt = project_to_feasible(res.x, best_n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_final:
                    h_final = h_opt
                    c5_final = c5_opt
        except:
            pass
        
        return (list(h_final), c5_final, best_n_points)
    
    # Fallback if no good start
    n_points = 30
    dx = 2.0 / n_points
    h_start = initialize_h(n_points)
    h_final = h_start.copy()
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        res = minimize(obj, h_final, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
    except:
        pass
    
    return (list(h_final), c5_final, n_points)
