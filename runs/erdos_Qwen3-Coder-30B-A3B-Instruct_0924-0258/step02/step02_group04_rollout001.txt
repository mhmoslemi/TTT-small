import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    samples = []
    for _ in range(n_samples):
        sample = np.random.random(n_points)
        # Sort and use inverse CDF
        sorted_sample = np.sort(sample)
        scaled = sorted_sample * (n_points / 2)
        # Distribute evenly in [0,1] and then scale
        uniform = np.linspace(0, 1, n_points)
        h = np.interp(uniform, sorted_sample, scaled)
        samples.append(h)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Resolution schedule
    resolutions = [15, 30, 45, 60, 75]
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        
        dx = 2.0 / n_points
        n_samples = max(10, min(50, int(2000 / n_points)))
        
        # Generate Latin Hypercube samples
        lhc_samples = latin_hypercube_sample(n_points, n_samples)
        
        # Local search for each sample
        for sample_idx, h_start in enumerate(lhc_samples):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Optimize with L-BFGS-B
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            
            try:
                res = minimize(objective, h_start, method='L-BFGS-B', 
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 50, 'ftol': 1e-8})
                
                if res.success and not np.isnan(res.fun):
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except Exception:
                pass
                
        # Fine-tune the best solution found so far on this resolution
        if best_h is not None and n_points == resolutions[-1]:
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                res = minimize(objective, best_h, method='L-BFGS-B', 
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 100, 'ftol': 1e-10})
                
                if res.success and not np.isnan(res.fun):
                    h_finetuned = project_to_feasible(res.x, n_points)
                    c5_finetuned = evaluate_erdos_solution(h_finetuned, dx)
                    
                    if c5_finetuned < best_c5:
                        best_h = h_finetuned.copy()
                        best_c5 = c5_finetuned
            except Exception:
                pass
                
    # Final fallback optimization
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_fallback = np.ones(n_points) * 0.5
        h_fallback = project_to_feasible(h_fallback, n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        
        try:
            res = minimize(evaluate_erdos_solution, h_fallback, args=(dx,), 
                          method='L-BFGS-B', bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-10})
            if res.success:
                best_h = project_to_feasible(res.x, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
                best_n_points = n_points
        except Exception:
            pass

    return (list(best_h), best_c5, best_n_points)
