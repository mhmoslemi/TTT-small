import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    population_size = 30
    elite_size = population_size // 10
    mutation_rate = 0.4
    beta = 0.2
    max_generations = 10000
    stagnation_threshold = 50
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Initialize mutation strength
    sigma = np.full(population_size, 0.05)
    
    best_h = None
    best_c5 = float('inf')
    stagnant_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate population
        fitness = [evaluate(h, 2.0/n_points) for h in population]
        
        # Update best solution
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            stagnant_count = 0
        else:
            stagnant_count += 1
            
        if stagnant_count >= stagnation_threshold:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        elites = population[:elite_size]
        new_population = elites[:]
        
        # Mutation and selection
        while len(new_population) < population_size:
            # Select parent using tournament selection
            parent_idx = np.random.randint(0, population_size - elite_size) + elite_size
            parent = population[parent_idx]
            
            # Mutate
            if np.random.random() < mutation_rate:
                mutant = parent + np.random.normal(0, sigma[parent_idx], n_points)
                mutant = project_to_feasible(mutant, n_points/2.0)
                new_population.append(mutant)
            else:
                new_population.append(parent.copy())
        
        population = new_population
        
        # Update sigma (self-adaptation)
        sigma = sigma * np.exp(beta * np.random.normal(0, 1, population_size))
