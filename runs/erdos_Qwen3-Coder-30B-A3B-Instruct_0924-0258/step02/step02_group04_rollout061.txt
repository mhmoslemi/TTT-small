import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m_particles=10):
    """Initialize particles using simplex method"""
    particles = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        # Simple starting guess
        h0 = np.ones(n_points) * 0.5
        
    h0 = project_to_feasible(h0, n_points)
    particles.append(h0.copy())
    
    # Generate other particles by taking random directions
    for i in range(m_particles - 1):
        # Small random perturbation
        pert = np.random.normal(0, 0.1, n_points)
        hp = project_to_feasible(h0 + pert, n_points)
        particles.append(hp)
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    w = 0.7  # Inertia weight
    c1 = c2 = 2.0  # Cognitive and social weights
    m_particles = 10
    max_iter = 2000
    shaking_threshold = 50
    lbfgs_interval = 200
    max_shake = 50
    
    # Initialize
    particles = initialize_simplex(n_points, m_particles)
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_vals = [evaluate_erdos_solution(p, dx) for p in particles]
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    stuck_count = 0
    swarm_size = m_particles
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Position update
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest_vals[i] = current_val
                pbest[i] = particles[i].copy()
                
                # Update global best
                if current_val < gbest_val:
                    gbest_val = current_val
                    gbest = particles[i].copy()
                    last_improvement = iteration
                    stuck_count = 0
                else:
                    stuck_count += 1
            else:
                stuck_count += 1
        
        # Apply shaking if stuck
        if stuck_count >= shaking_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            stuck_count = 0
        
        # Apply L-BFGS-B refinement every few iterations
        if iteration % lbfgs_interval == 0 and iteration > 0:
            top_indices = np.argsort(pbest_vals)[:min(10, len(pbest_vals))]
            for idx in top_indices:
                h0 = pbest[idx]
                bounds = [(0, 1) for _ in range(n_points)]
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                try:
                    res = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                                 options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        refined_val = evaluate_erdos_solution(h_refined, dx)
                        if refined_val < pbest_vals[idx]:
                            pbest_vals[idx] = refined_val
                            pbest[idx] = h_refined.copy()
                            if refined_val < gbest_val:
                                gbest_val = refined_val
                                gbest = h_refined.copy()
                                last_improvement = iteration
                except:
                    pass
    
    # Final refinement on the best solution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_val = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_val, n_points)
