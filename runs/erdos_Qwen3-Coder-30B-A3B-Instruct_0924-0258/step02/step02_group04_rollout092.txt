import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a low-correlation binary-like pattern based on LABS principles
        h = np.zeros(n_points)
        ones_positions = set()
        # Place 'ones' in a way that minimizes correlation
        for i in range(n_points // 2):
            pos = (i * 7) % n_points
            ones_positions.add(pos)
        for i in ones_positions:
            h[i] = 1.0
        h = project_to_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Multi-resolution approach: start with low resolution to find good configurations
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # More aggressive optimization for first few resolutions
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple restarts with structured perturbations
        h_current = h_start.copy()
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt = h_current.copy()
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            
            # Use L-BFGS-B for efficient local optimization with box constraints
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            try:
                res = minimize(obj, h_opt, method='L-BFGS-B', bounds=bounds, 
                              options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
            except:
                pass
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            
            # Perturb for next restart
            noise = np.random.normal(0, 0.05, n_points)
            h_current = project_to_feasible(h_opt + noise, n_points)
        
        if best_h is not None and best_c5 <= 0.3807:
            break  # Early stopping if we have a good enough solution
    
    if best_h is None:
        # Fallback to finest resolution
        n_points = 75
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        h_opt, c5_opt = h_start.copy(), evaluate_erdos_solution(h_start, dx)
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(obj, h_opt, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
        except:
            pass
        
        best_h, best_c5, best_n_points = h_opt, c5_opt, n_points
        
    return (list(best_h), best_c5, best_n_points)
