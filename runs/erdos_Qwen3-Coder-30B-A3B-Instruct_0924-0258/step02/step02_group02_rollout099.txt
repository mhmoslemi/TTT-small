import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial vector
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    # GA parameters
    pop_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size]
            
            if np.random.rand() < crossover_rate and i + 1 < pop_size:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(n_points)
                child1[mutate_idx] = np.random.rand()
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(n_points)
                child2[mutate_idx] = np.random.rand()
            
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            offspring.extend([child1, child2])
        
        # Replace bottom half of population with offspring
        sorted_indices = np.argsort(fitness_scores)
        for i in range(pop_size // 2):
            population[sorted_indices[i]] = offspring[i]
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
