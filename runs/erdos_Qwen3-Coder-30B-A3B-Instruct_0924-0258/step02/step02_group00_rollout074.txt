import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def fft_correlation(h):
    """Compute correlation using FFT for efficiency"""
    n = len(h)
    # Use zero-padding to avoid circular convolution effects
    corr = correlate(h, 1 - h, mode='full')
    # Return the relevant part (non-negative lags)
    return corr[n-1:]

def pso_optimize(initial_h, n_points, dx, time_limit):
    """Particle Swarm Optimization for minimizing C5"""
    # PSO parameters
    n_particles = 50
    max_iter = 200
    w_start = 0.7
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize particles
    particles = np.tile(initial_h, (n_particles, 1))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Add some noise to initial particles
    for i in range(n_particles):
        noise = np.random.normal(0, 0.05, n_points)
        particles[i] = project_to_feasible(particles[i] + noise, n_points)
    
    # Evaluate initial particles
    particle_costs = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    particle_best = particles.copy()
    particle_best_costs = particle_costs.copy()
    
    # Global best
    global_best_idx = np.argmin(particle_costs)
    global_best = particles[global_best_idx].copy()
    global_best_cost = particle_costs[global_best_idx]
    
    # Time tracking
    start_time = time()
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (particle_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            cost = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if cost < particle_best_costs[i]:
                particle_best[i] = particles[i].copy()
                particle_best_costs[i] = cost
            
            # Update global best
            if cost < global_best_cost:
                global_best = particles[i].copy()
                global_best_cost = cost
                
    return global_best, global_best_cost

def gradient_descent_refine(h, n_points, dx, max_iter=100, lr=0.001):
    """Refine solution using gradient descent on a smoothed version"""
    h = h.copy()
    best_h = h.copy()
    best_cost = evaluate_erdos_solution(h, dx)
    
    for i in range(max_iter):
        # Simple gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            cost_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_to_feasible(h_minus, n_points)
            cost_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (cost_plus - cost_minus) / (2 * eps)
        
        # Update with gradient
        h = h - lr * grad
        h = project_to_feasible(h, n_points)
        
        cost = evaluate_erdos_solution(h, dx)
        if cost < best_cost:
            best_h = h.copy()
            best_cost = cost
    
    return best_h, best_cost

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    time_left = budget_s
    
    # Use initial solution if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Try different resolutions in order of increasing detail
    resolutions = [25, 30, 35]
    
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > time_left * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        
        # Interpolate from previous if needed
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # PSO with time constraint
        pso_time = min(100, time_left / len(resolutions))
        h_pso, c5_pso = pso_optimize(h_start, n_points, dx_try, pso_time)
        
        # Refine with GD
        h_refined, c5_refined = gradient_descent_refine(h_pso, n_points, dx_try, 
                                                       max_iter=100, lr=0.001)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            best_n_points = n_points
            
        # Update time tracking
        time_left = budget_s - (time() - start_time)
        h_start = best_h.copy()
    
    # Final check with full evaluation
    final_c5 = evaluate_erdos_solution(best_h, 2.0/best_n_points)
    
    return (list(best_h), final_c5, best_n_points)
