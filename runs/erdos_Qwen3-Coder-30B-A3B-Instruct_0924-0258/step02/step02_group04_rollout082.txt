import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population using initial_h_values if available"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for i in range(pop_size):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = project(h, n_points)
        population.append(h)
    return population

def mutate(h, mutation_rate, sigma=0.1):
    """Apply mutation to a single individual"""
    if np.random.rand() < mutation_rate:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2, crossover_rate=0.7):
    """Perform crossover between two individuals"""
    if np.random.rand() < crossover_rate:
        idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
        offspring = np.copy(h1)
        offspring[idx1:idx2] = h2[idx1:idx2]
        return project(offspring, len(offspring))
    else:
        return h1.copy()

def tournament_selection(population, fitness, tournament_size=5):
    """Select parent using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    pop_size = 50
    mutation_rate = 0.2
    max_generations = 1000
    dx = 2.0 / n_points
    
    # Initialize
    population = initialize_population(n_points, pop_size)
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    last_improvement = 0
    
    # GA loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = generation
                
        # Check for stagnation
        if generation - last_improvement > 200:
            # Reset population if no improvement
            population = initialize_population(n_points, pop_size)
            stagnation_count += 1
            if stagnation_count > 5:
                break
        else:
            stagnation_count = 0
            
        # Selection
        selected_parents = tournament_selection(population, fitness)
        
        # Create offspring
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected_parents[i]
            parent2 = selected_parents[i+1] if i+1 < pop_size else selected_parents[0]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
        # Increase mutation rate if stagnating
        if generation - last_improvement > 100:
            mutation_rate = min(0.5, mutation_rate + 0.01)
    
    # Ensure we return at least a valid solution
    if best_h is None:
        # Fallback
        h = initialize_population(n_points, 1)[0]
        best_c5 = evaluate_erdos_solution(h, dx)
        best_h = h
    
    return (list(best_h), best_c5, n_points)
