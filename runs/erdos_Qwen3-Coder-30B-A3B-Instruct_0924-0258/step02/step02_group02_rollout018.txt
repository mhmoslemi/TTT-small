import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_overlaps(h_vec):
        # Compute overlap for all locations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return corr

    def solve_convex_optimization(h_vec, worst_shifts):
        # Formulate the convex optimization problem
        h_vars = Variable(n_points)
        eta = Variable()
        
        # Constraints
        constraints = [
            h_vars <= 1,
            h_vars >= 0,
            sum(h_vars) == target_sum,
            eta >= 0
        ]
        
        # Add overlap constraints
        overlaps = compute_overlaps(h_vec)
        for s in worst_shifts:
            if s < len(overlaps):
                constraints.append(overlaps[s] <= eta)
        
        # Objective
        objective = Minimize(eta)
        
        # Solve
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=CPLEX, verbose=False)
            new_h = h_vars.value
            new_eta = eta.value
            return new_h, new_eta
        except:
            return h_vec, compute_c5(h_vec)
    
    # Start optimization
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Initial coarse search for worst case
    shift_indices = np.arange(0, 2 * n_points, max(1, n_points // 20))
    
    max_iter = 500
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Find the current worst shift
        overlaps = compute_overlaps(h)
        worst_idx = np.argmax(overlaps)
        worst_shifts = [worst_idx]
        
        # Refine around worst shift
        start_s = max(0, worst_idx - 5)
        end_s = min(len(overlaps), worst_idx + 5)
        fine_shifts = np.arange(start_s, end_s)
        
        # Add some random shifts for exploration
        if len(fine_shifts) < 20:
            random_shifts = np.random.randint(0, len(overlaps), size=20 - len(fine_shifts))
            fine_shifts = np.concatenate([fine_shifts, random_shifts])
        
        worst_shifts.extend(fine_shifts)
        
        # Solve optimization problem
        new_h, new_c5 = solve_convex_optimization(h, worst_shifts)
        
        #Projected 
        new_h = project_h(new_h)
        
        # Check improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
        
        # Update h for next iteration
        h = new_h
        
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
