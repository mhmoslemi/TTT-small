import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def differential_evolution_step(pop, F=0.8, CR=0.7):
        new_pop = []
        n_pop = len(pop)
        for i in range(n_pop):
            # Select three distinct random individuals
            idxs = np.random.choice(n_pop, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            
            # Mutation
            mutant = a + F * (b - c)
            
            # Crossover
            trial = []
            j_rand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial.append(mutant[j])
                else:
                    trial.append(pop[i][j])
            
            # Project to feasible region
            trial = project_feasible(trial)
            new_pop.append(trial)
            
        return new_pop
    
    # DE parameters
    population_size = 25
    max_generations = 500
    F = 0.7
    CR = 0.5
    
    # Initialize population
    population = [project_feasible(h + np.random.normal(0, 0.05, n_points)) for _ in range(population_size)]
    
    best_h = h.copy()
    best_val = compute_overlap(best_h)
    
    progression = 0
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        fitness = [compute_overlap(ind) for ind in population]
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_val:
            best_val = min_fitness
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            progression = 0
        else:
            progression += 1
            
        # Evolution step
        population = differential_evolution_step(population, F, CR)
        
        # Occasionally decrease F (adaptive control)
        if generation % 50 == 0 and F > 0.5:
            F -= 0.05
            
        if progression > 50:
            # Reset with new random individuals
            for i in range(population_size // 2):
                population[i] = project_feasible(h + np.random.normal(0, 0.1, n_points))
            progression = 0

    # Final check
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
