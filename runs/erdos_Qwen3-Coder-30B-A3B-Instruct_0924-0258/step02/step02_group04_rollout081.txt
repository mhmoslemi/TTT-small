import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population using initial_h_values or a simple pattern"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        h = project(base_h + np.random.normal(0, 0.1, n_points), n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate an individual with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=max(1, int(len(h)*0.2)), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select parent using tournament selection"""
    indices = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    pop_size = 50
    max_generations = 1000
    mutation_prob = 0.2
    crossover_prob = 0.7
    stagnation_limit = 200
    
    # Initialize
    dx = 2.0 / n_points
    population = initialize_population(n_points, pop_size)
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    best_n_points = n_points
    
    # GA loop
    generation = 0
    stagnation_count = 0
    last_best_c5 = best_c5
    
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate current population
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Check for improvement
        current_best = np.min(fitnesses)
        if current_best < best_c5:
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_c5 = current_best
            stagnation_count = 0
            last_best_c5 = best_c5
        else:
            stagnation_count += 1
            
        # Termination condition
        if stagnation_count > stagnation_limit:
            break
            
        # Selection, crossover, mutation
        new_population = []
        for _ in range(pop_size):
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < crossover_prob:
                offspring = crossover(parent1, parent2)
            else:
                offspring = parent1.copy()
                
            # Mutation
            offspring = mutate(offspring, mutation_prob, 0.1)
            new_population.append(offspring)
            
        population = new_population
        generation += 1
        
    return (list(best_h), best_c5, best_n_points)
