import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Parameters
    max_iter = 5000
    learning_rate = 0.01
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    patience_counter = 0
    
    start_time = time.time()
    
    # Precompute shifts
    shifts = list(range(n_points))  # Shifts from 0 to n_points-1
    
    for iter_num in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Randomize shift order to avoid local minima
        np.random.shuffle(shifts)
        
        # For each shift, compute gradient and update h
        for shift in shifts:
            # Compute partial derivative for current shift
            # C_shift = sum h[i] * (1 - h[i+shift]) for i = 0..n_points-shift-1
            # dC/dh[i] = (1 - h[i+shift]) if i < n_points-shift, else 0
            # dC/dh[i+shift] = -h[i] if i >= 0 and i+shift < n_points, else 0
            
            grad = np.zeros_like(h)
            # If using a full convolution, just build gradient from specific terms:
            for i in range(n_points - shift):
                grad[i] += (1 - h[i + shift])
                if i + shift < n_points:
                    grad[i + shift] -= h[i]
            
            # Update h
            h = h - learning_rate * grad
            h = project_h(h)
            
            # If gradient descent failed or got worse, try a different strategy
            if iter_num % 100 == 0 and iter_num > 0:
                # Try one-step optimization
                try:
                    result = minimize(evaluate_objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
                    if result.success:
                        h = result.x
                    h = project_h(h)
                except:
                    pass
                    
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter >= patience:
            learning_rate *= 0.95
            patience_counter = 0
            
        # Occasionally add noise to escape local minima
        if iter_num % 100 == 0 and iter_num > 0:
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
