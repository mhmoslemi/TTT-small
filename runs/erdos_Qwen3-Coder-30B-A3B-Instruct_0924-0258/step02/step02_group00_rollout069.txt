import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_subgradient(h, n_points, max_shifts):
    """Compute subgradient based on maximum overlap shifts"""
    corr = np.correlate(h, 1 - h, mode='full')
    subgrad = np.zeros_like(h)
    
    for k in max_shifts:
        # Compute gradient contribution for shift k
        for i in range(n_points):
            j = (i + k) % n_points
            subgrad[i] += (1 - h[j]) - h[i]
    
    return subgrad

def frank_wolfe_step(h, subgrad, n_points):
    """Frank-Wolfe step to find descent direction"""
    # Linear optimization: minimize subgrad * s subject to constraints
    # s in simplex: sum(s) = n_points/2, 0 <= s <= 1
    # We want the s that minimizes g^T s
    s = np.zeros_like(h)
    
    # Sort indices by subgrad values to find the descent direction
    idx_sorted = np.argsort(subgrad)
    
    # Greedily select indices with negative subgradients
    total = 0
    for idx in idx_sorted:
        if total < n_points / 2:
            s[idx] = min(1.0, n_points / 2 - total)
            total += s[idx]
        else:
            break
    
    # Project to sum = n_points/2
    if total > 0:
        s = s * (n_points / 2) / total
    
    return s

def backtracking_line_search(h, s, subgrad, n_points, dx, max_iter=10):
    """Backtracking line search for step size"""
    alpha = 1.0
    gamma = 0.5  # step size reduction factor
    
    initial_c5 = evaluate_erdos_solution(h, dx)
    
    for _ in range(max_iter):
        h_new = h + alpha * (s - h)
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < initial_c5:
            return h_new, alpha
        alpha *= gamma
    return h, alpha

def frank_wolfe_optimize(h_start, n_points, dx, max_iter=40, tol=1e-6):
    """Frank-Wolfe optimization with line search"""
    h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr)
        max_shifts = np.where(corr >= max_overlap - tol)[0]
        if len(max_shifts) == 0:
            # fallback, pick one max
            max_shifts = [np.argmax(corr)]
            
        subgrad = compute_subgradient(h, n_points, max_shifts)
        s = frank_wolfe_step(h, subgrad, n_points)
        h, alpha = backtracking_line_search(h, s, subgrad, n_points, dx)
        h = project_to_feasible(h, n_points)
        
        c5 = evaluate_erdos_solution(h, dx)
        # For improved stability, use a smoother view
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Early stopping
        if iteration > 0 and abs(c5 - best_c5) < 1e-6:
            break
        
    return best_h, best_c5
