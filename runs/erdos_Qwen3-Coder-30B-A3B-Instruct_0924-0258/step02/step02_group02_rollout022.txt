import numpy as np
import time
from copy import deepcopy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h_init satisfies the constraint
    h_init = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic Algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h_init.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative for maximization
        
        # Select top 50% using tournament selection
        selected = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size)
            best_idx = tournament[np.argmax([fitness_scores[i] for i in tournament])]
            selected.append(population[best_idx].copy())
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, len(selected)-1)]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, 
                                                   size=np.random.randint(1, 3), 
                                                   replace=False)
                for idx in mutation_indices:
                    child1[idx] += np.random.normal(0, 0.05)
            
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, 
                                                   size=np.random.randint(1, 3), 
                                                   replace=False)
                for idx in mutation_indices:
                    child2[idx] += np.random.normal(0, 0.05)
            
            # Project to valid range and normalize
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            offspring.extend([child1, child2])
        
        # Create next generation (replace bottom 50% with offspring)
        sorted_indices = np.argsort(fitness_scores)[::-1]
        new_population = []
        for i in range(population_size // 2):
            new_population.append(population[sorted_indices[i]])
        for i in range(min(len(offspring), population_size // 2)):
            new_population.append(offspring[i])
        
        population = new_population
        
        # Update best solution
        for h in population:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
