import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin hypercube samples"""
    samples = []
    for _ in range(n_samples):
        sample = []
        for i in range(n_points):
            # Generate a random value in the i-th segment
            segment_start = i / n_points
            segment_end = (i + 1) / n_points
            sample.append(np.random.uniform(segment_start, segment_end))
        # Shuffle
        random.shuffle(sample)
        samples.append(sample)
    return samples

def optimize_single_h(h_start, n_points, dx, max_iter=20):
    """Optimize a single h using L-BFGS-B"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': max_iter, 'ftol': 1e-6})
        if res.success:
            h_result = project_to_feasible(res.x, n_points)
            c5_result = evaluate_erdos_solution(h_result, dx)
            return h_result, c5_result
    except:
        pass
        
    return h_start, evaluate_erdos_solution(h_start, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        # Ensure compatibility with 15-point resolution

    else:
        initial_h = np.ones(15) * 0.5
    
    # Use a consistent approach with different grid sizes
    grid_sizes = [15, 30]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in grid_sizes:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Use Latin hypercube starting points
        lh_samples = latin_hypercube_sample(n_points, 5)
        best_in_resolution = (None, float('inf'))
        
        for sample in lh_samples:
            h_init = np.array(sample)
            h_proj = project_to_feasible(h_init, n_points)
            
            # Optimize with L-BFGS-B
            h_opt, c5_opt = optimize_single_h(h_proj, n_points, dx, max_iter=30)
            
            if c5_opt < best_in_resolution[1]:
                best_in_resolution = (h_opt, c5_opt)
                
        # Check if this resolution gave better result
        if best_in_resolution[1] < best_c5:
            best_h = best_in_resolution[0]
            best_c5 = best_in_resolution[1]
            best_n_points = n_points
    
    # Do a final refinement with a higher resolution if time allows
    if best_h is not None and time() - start_time < budget_s * 0.9:
        dx = 2.0 / 60
        n_points = 60
        
        # Use previous best as starting point
        h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_proj = project_to_feasible(h_init, n_points)
        
        h_opt, c5_opt = optimize_single_h(h_proj, n_points, dx, max_iter=50)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n_points = n_points
            
    # Final verification
    if best_h is not None:
        dx = 2.0 / best_n_points
        final_c5 = evaluate_erdos_solution(best_h, dx)
        return (list(best_h), final_c5, best_n_points)
    else:
        # Fallback to initial construction with finest resolution
        dx = 2.0 / 30
        n_points = 30
        h_init = project_to_feasible(initial_h, n_points)
        h_opt, c5_opt = optimize_single_h(h_init, n_points, dx, max_iter=50)
        return (list(h_opt), c5_opt, n_points)
