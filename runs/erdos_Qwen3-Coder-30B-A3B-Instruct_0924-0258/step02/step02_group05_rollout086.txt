import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 30
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    for i in range(swarm_size):
        if i == 0:
            # Use initial solution
            particle = h.copy()
        else:
            # Random perturbation around initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = project_feasible(particle)
        swarm.append(particle)
    
    # Initialize personal bests
    pbest = [swarm[i].copy() for i in range(swarm_size)]
    pbest_vals = [compute_overlap(swarm[i]) for i in range(swarm_size)]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # Velocity initialization
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Compute current particle's overlap
            current_val = compute_overlap(swarm[i])
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                
                # Check if it's a new global best
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
        
        # Update velocities and positions
        for i in range(swarm_size):
            # Random numbers
            r1, r2 = np.random.random(2)
            
            # Update velocity
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - swarm[i]) +
                c2 * r2 * (gbest - swarm[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
    
    # Final evaluation
    final_C5 = compute_overlap(gbest)
    
    return (gbest.tolist(), final_C5, n_points)
