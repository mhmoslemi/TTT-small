import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max"""
    exp_x = np.exp(alpha * x)
    return exp_x / (np.sum(exp_x))

def objective_with_smooth_max(h, n_points, dx):
    """Objective using smooth max to approximate the maximum"""
    corr = np.correlate(h, 1 - h, mode='full')
    smooth_max_val = smooth_max(corr, alpha=100)
    penalty = 0.1 * (smooth_max_val - 0.3)**2
    return smooth_max_val + penalty

def gradient_smooth_max(h, n_points, dx):
    """Gradient of the smooth max approximation"""
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    grad_corr = smooth_max_grad(corr, alpha=alpha)
    
    # Compute gradient of convolution with respect to h
    grad = np.zeros_like(h)
    
    # For kernel 1-h, convolve with grad_corr
    # We simulate the derivative of correlation: 
    # d/dh_i sum_j h[j] * (1-h[i+j]) = (1-h[i+j]) - h[i+j] 
    # let us compute directly the gradient using finite difference for numerical stability
    
    # Instead of complex symbolic gradient, use finite difference for robustness  
    eps = 1e-6
    grad_fd = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        obj_plus = objective_with_smooth_max(h_plus, n_points, dx)
        obj_base = objective_with_smooth_max(h, n_points, dx)
        grad_fd[i] = (obj_plus - obj_base) / eps
    
    return grad_fd

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    """Optimize using gradient descent with adaptive learning rate"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iter = 500
    learning_rate = 0.01
    decay_rate = 0.95
    decay_interval = 50
    
    for i in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Compute gradient using finite difference
        grad = gradient_smooth_max(best_h, n_points, dx)
        
        # Gradient descent step
        h_new = best_h - learning_rate * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Reduce learning rate
            learning_rate *= decay_rate
        
        # Adaptive restarts
        if i % decay_interval == 0 and i > 0:
            # Small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_restart = project_to_feasible(best_h + perturbation, n_points)
            c5_restart = evaluate_erdos_solution(h_restart, dx)
            if c5_restart < best_c5:
                best_h = h_restart.copy()
                best_c5 = c5_restart
        
        # Decrease learning rate
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.95
            
    return best_h, best_c5
