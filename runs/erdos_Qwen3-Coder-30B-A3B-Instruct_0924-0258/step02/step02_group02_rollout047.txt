import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Genetic Algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        individual = np.random.rand(n_points)
        individual = project_h(individual)
        population.append(individual)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for individual in population:
            c5 = compute_c5(individual)
            fitness_scores.append(-c5)  # Negative for minimization
        
        # Selection using tournament
        new_population = []
        for _ in range(population_size // 2):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(new_population), 2):
            parent1 = new_population[i]
            parent2 = new_population[i+1] if i+1 < len(new_population) else new_population[0]
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, np.random.randint(1, 3), replace=False)
                for idx in mutation_indices:
                    child1[idx] = np.random.rand()
            
            if np.random.rand() < mutation_rate:
                mutation_indices = np.random.choice(n_points, np.random.randint(1, 3), replace=False)
                for idx in mutation_indices:
                    child2[idx] = np.random.rand()
            
            # Project to constraints
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            offspring.extend([child1, child2])
        
        # Replace worst individuals with offspring
        # Sort population by fitness (descending, so negative values are better)
        sorted_indices = np.argsort(fitness_scores)[::-1] 
        for i in range(len(offspring)):
            population[sorted_indices[-(i+1)]] = offspring[i]
        
        # Update best solution
        for individual in population:
            c5 = compute_c5(individual)
            if c5 < best_c5:
                best_c5 = c5
                best_h = individual.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
