import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute the overlap integral using convolution
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()

    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate multiple random initial points
    population_size = 30
    population = []
    
    # Add the initial point
    population.append(h_start.copy())
    
    # Add random points
    for _ in range(population_size - 1):
        # Generate random point and project to feasible region
        h_rand = np.random.rand(n_points)
        h_rand = project_to_feasible(h_rand, n_points)
        population.append(h_rand)
    
    # Optimization parameters
    max_iter = 100
    patience = 20
    improvement_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        improved = False
        
        for i in range(len(population)):
            h_current = population[i]
            
            # Apply small random perturbation
            h_perturbed = h_current.copy()
            indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            for idx in indices:
                noise = np.random.normal(0, 0.08)
                h_perturbed[idx] += noise
            
            # Project to feasible region
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization using SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        improved = True
                        improvement_count = 0  # reset patience counter
            except:
                pass
                
            # Update the population with the optimized version if it improved
            if not improved:
                # no improvement this pass, we might look for more diversity
                pass  
                
        improvement_count += 1
        if improvement_count > patience:
            # reduce diversity
            for i in range(len(population)):

                population[i] = h_rand

        except:
