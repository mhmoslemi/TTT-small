import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = len(h)

        # Fallback: create a simple initial guess

        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Optimizer parameters
    adam_lr = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    
    current_h = h.copy()
    best_val = compute_overlap(current_h)
    best_h = current_h.copy()
    
    # Adam optimizer variables
    m = np.zeros_like(current_h)
    v = np.zeros_like(current_h)
    
    last_improvement = 0
    max_improvements = 1000
    
    # Local optimization
    for iter_count in range(50000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps to identify which k contributes most
        conv = np.correlate(current_h, 1 - current_h, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        # Top contribution indices
        top_k = np.argsort(overlaps)[-3:][::-1]
        
        # Compute approximate gradient using finite differences
        gradient = np.zeros_like(current_h)
        eps = 1e-4
        for j in range(len(current_h)):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project_feasible(h_plus)
            C_plus = compute_overlap(h_plus)
            
            h_minus = current_h.copy()
            h_minus[j] -= eps
            h_minus = project_feasible(h_minus)
            C_minus = compute_overlap(h_minus)
            
            gradient[j] = (C_plus - C_minus) / (2 * eps)
        
        # Clip gradient
        gradient = np.clip(gradient, -10, 10)
        
        # Adam update
        m = adam_beta1 * m + (1 - adam_beta1) * gradient
        v = adam_beta2 * v + (1 - adam_beta2) * gradient**2
        m_hat = m / (1 - adam_beta1**(iter_count + 1))
        v_hat = v / (1 - adam_beta2**(iter_count + 1))
        current_h = current_h - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project to feasible region
        current_h = project_feasible(current_h)
        
        # Check if improvement
        current_val = compute_overlap(current_h)
        if current_val < best_val:
            best_val = current_val
            best_h = current_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if last_improvement > max_improvements:
            # Early stopping
            break
    
    # Final check with full evaluation
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
