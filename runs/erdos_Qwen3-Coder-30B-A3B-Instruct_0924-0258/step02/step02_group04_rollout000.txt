import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def generate_latin_hypercube(n_points, n_samples=50):
    """Generate Latin hypercube samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        # Generate a Latin hypercube sample
        h = np.random.rand(n_points)
        # Sort each dimension and assign values
        perm = np.random.permutation(n_points)
        h = np.sort(h)
        h = np.array([h[i] for i in perm])
        # Scale to [0,1] and project to sum constraint
        samples.append(project_to_feasible(h, n_points))
    return samples

def initialize_h(n_points):
    """Initialize h using either initial_h_values or a pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def optimize_single(h_start, n_points, dx, max_iter=20, time_limit=None):
    """Optimize a single starting point using L-BFGS-B"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Simple local optimization step with L-BFGS-B
    try:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        res = minimize(obj, h_start, method='L-BFGS-B', 
                      bounds=bounds, 
                      constraints=cons, 
                      options={'maxiter': max_iter, 'ftol': 1e-8})
        if res.success:
            h_opt = project_to_feasible(res.x, n_points)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < c5_best:
                h_best = h_opt.copy()
                c5_best = c5_opt
    except Exception:
        pass

    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Schedules for different resolutions
    resolutions = [15, 30, 45, 60]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Generate diversified Latin hypercube samples
    lh_samples = generate_latin_hypercube(15, 10)
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        current_time = time()
        time_per_res = (budget_s - (current_time - start_time)) / len(resolutions)
        
        # Use several restarts starting from Latin hypercube samples
        restarts = min(10, max(1, int(50 * (time_per_res / 100))))
        
        for _ in range(restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Start with Latin hypercube sample or initialization
            if len(lh_samples) > 0:
                h_start = lh_samples.pop()
            else:
                h_start = initialize_h(n_points)
                
            # Try optimization
            h_opt, c5_opt = optimize_single(h_start, n_points, dx, max_iter=30)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_single(h_start, n_points, dx)
        best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
