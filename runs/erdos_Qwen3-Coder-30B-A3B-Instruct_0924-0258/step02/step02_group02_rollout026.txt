import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Extract the valid range
        mid = len(corr) // 2
        valid_corr = corr[mid:]
        return np.min(valid_corr)

    def compute_autocorr_grad(h_vec, s_min):
        # Compute gradient of autocorrelation at shift s_min
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            if i + s_min < len(h_vec):
                grad[i] += h_vec[i + s_min]
            if i >= s_min:
                grad[i] += h_vec[i - s_min]
        return grad
    
    # Gradient-based approach
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    learning_rate = 0.01
    no_improvement_count = 0
    max_no_improvement = 500
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current autocorrelations and find minimum
        corr_full = np.correlate(h, h, mode='full')
        mid = len(corr_full) // 2
        corr_valid = corr_full[mid:]
        s_min = np.argmin(corr_valid)
        
        # Compute gradient of the minimal autocorrelation
        grad = compute_autocorr_grad(h, s_min)
        grad = grad * (1.0 / np.linalg.norm(grad))  # Normalize
        
        # Try to increase the minimal autocorrelation
        h_new = h + learning_rate * grad
        h_new = project_h(h_new)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            # Add noise for escape
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h_noisy = h + noise
            h_noisy = project_h(h_noisy)
            new_c5_noisy = compute_c5(h_noisy)
            
            if new_c5_noisy < best_c5:
                h = h_noisy
                best_c5 = new_c5_noisy
                best_h = h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                if no_improvement_count % 100 == 0:
                    learning_rate *= 0.95  # Decrease learning rate

        if no_improvement_count > max_no_improvement:
            # Finish optimization if no improvement over many iterations
            break

    # Final validation and return
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
