import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use maximum entropy approach in frequency domain with iterative refinement
    n_points = 15
    dx = 2.0 / n_points
    h = initialize_h(n_points)
    c5_best = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Fourier-based optimization loop
    max_iter = 1000
    iter_count = 0
    improvement_count = 0
    last_improvement_iter = 0
    
    # Frequency domain optimization helper
    def update_spectrum(h, weight=0.1):
        # Fourier transform
        H = np.fft.fft(h)
        # Compute magnitude squared
        mag_sq = np.abs(H) ** 2
        # Penalize peaks - encourage flat spectrum
        penalty_grad = 4 * mag_sq ** 3  # derivative of sum |H|^4
        # Apply gradient update
        H_new = H - weight * H * penalty_grad / (mag_sq + 1e-8)
        # Maintain real signal property (conjugate symmetry)
        H_new[0] = np.real(H_new[0])  # DC component
        if n_points % 2 == 0:
            H_new[n_points // 2] = np.real(H_new[n_points // 2])  # Nyquist
        # Inverse transform
        h_new = np.real(np.fft.ifft(H_new))
        return h_new
    
    while iter_count < max_iter and (time() - start_time) < budget_s:
        # Try to update the spectrum in frequency domain
        h_candidate = update_spectrum(h, 0.001 * (1 + 0.1 * (iter_count // 100)))
        # Ensure it's feasible
        h_candidate = project_to_feasible(h_candidate, n_points)
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        if c5_candidate < c5_best:
            h = h_candidate
            c5_best = c5_candidate
            improvement_count += 1
            last_improvement_iter = iter_count
            best_h = h.copy()
        else:
            # Occasionally perturb to escape local optima
            if iter_count % 20 == 0:
                h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
        iter_count += 1
        # Check for stagnation
        if iter_count - last_improvement_iter > 500:
            break
    
    # Try higher resolution if time allows
    if (time() - start_time) < budget_s * 0.8 and n_points < 60:
        n_points = 30
        dx = 2.0 / n_points
        h = initialize_h(n_points)
        c5_best = evaluate_erdos_solution(h, dx)
        best_h = h.copy()
        max_iter = 500
        iter_count = 0
        improvement_count = 0
        last_improvement_iter = 0
        
        while iter_count < max_iter and (time() - start_time) < budget_s:
            h_candidate = update_spectrum(h, 0.0005 * (1 + 0.1 * (iter_count // 100)))
            h_candidate = project_to_feasible(h_candidate, n_points)
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < c5_best:
                h = h_candidate
                c5_best = c5_candidate
                improvement_count += 1
                last_improvement_iter = iter_count
                best_h = h.copy()
            else:
                if iter_count % 20 == 0:
                    h = project_to_feasible(h + np.random.normal(0, 0.005, n_points), n_points)
            iter_count += 1
            if iter_count - last_improvement_iter > 500:
                break
    
    # Final cleanup
    final_h = project_to_feasible(best_h, len(best_h))
    final_c5 = evaluate_erdos_solution(final_h, 2.0 / len(final_h))
    
    return (list(final_h), final_c5, len(final_h))
