import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            h_base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_base)), h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        h = project(h_base + np.random.normal(0, 0.1, n_points), n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitnesses, tournament_size=5):
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.7
    stagnation_limit = 200
    stagnation_count = 0
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    best_n_points = n_points
    
    generation = 0
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate fitness of current population
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination if no improvement
        if stagnation_count > stagnation_limit:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % pop_size] if i + 1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                offspring1 = crossover(parent1, parent2)
                offspring2 = crossover(parent2, parent1)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            offspring1 = mutate(offspring1, mutation_rate, 0.1)
            offspring2 = mutate(offspring2, mutation_rate, 0.1)
            
            new_population.extend([offspring1, offspring2])
        
        population = new_population[:pop_size]
        
        # Adaptive mutation rate
        if stagnation_count > 50:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
            
        generation += 1
    
    return (list(best_h), best_c5, best_n_points)
