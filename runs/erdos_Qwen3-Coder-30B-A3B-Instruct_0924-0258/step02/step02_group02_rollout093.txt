import numpy as np
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies sum constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    # Initial set of constraints
    constraints = []
    eta = Variable()
    
    # Helper to compute overlap for shift k
    def compute_overlap(h_vec, k):
        # k is normalized shift value
        s = round(k / dx)
        if s >= 0:
            return np.sum(h_vec[:-s] * (1 - h_vec[s:])) * dx
        else:
            return np.sum(h_vec[-s:] * (1 - h_vec[:-s])) * dx

    # Start with initial overlaps
    initial_c5 = compute_c5(h)
    current_h = h.copy()
    best_c5 = initial_c5
    best_h = h.copy()
    
    start_time = time.time()
    max_iters = 500
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iters:
        if len(constraints) == 0:
            # Initialize with coarse grid
            shifts = np.linspace(0, 2, 10)
            max_shift = 0
            max_overlap = -1
            for s in shifts:
                overlap = compute_overlap(current_h, s)
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_shift = s
            constraints.append(eta >= compute_overlap(current_h, max_shift))
        else:
            # Use refined search in vicinity of worst-case shift
            # Evaluate a finer grid around max_shift
            neighbors = np.linspace(max_shift - 0.1, max_shift + 0.1, 20)
            max_shift = 0
            max_overlap = -1
            for s in neighbors:
                overlap = compute_overlap(current_h, s)
                if overlap > max_overlap:
                    max_overlap = overlap
                    max_shift = s
            # Add constraint for this shift
            constraints.append(eta >= compute_overlap(current_h, max_shift))
        
        # Create and solve optimization problem
        h_var = Variable(n_points)
        obj = Minimize(eta)
        
        # Bounds and constraints
        bounds = [h_var >= 0, h_var <= 1]
        sum_constraint = [sum(h_var) == target_sum]
        prob = Problem(obj, bounds + sum_constraint + constraints)
        
        # Try to solve with multiple solvers
        try:
            prob.solve(solver=MOSEK, verbose=False)
        except:
            try:
                prob.solve(solver=SCS, verbose=False)
            except:
                prob.solve(solver=ECOS, verbose=False)
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            new_h = h_var.value
            if new_h is not None:
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * target_sum / np.sum(new_h)
                current_h = new_h
                new_c5 = compute_c5(current_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()
        else:
            # If optimization fails, fall back to gradient descent or
            # just project current solution
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * target_sum / np.sum(current_h)
            new_c5 = compute_c5(current_h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        
        iter_count += 1
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
