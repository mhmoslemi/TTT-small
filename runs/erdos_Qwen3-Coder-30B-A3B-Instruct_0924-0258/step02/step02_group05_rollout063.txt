import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Particle Swarm Optimization
    P = 30  # swarm size
    max_iter = 2000
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 2.0
    v_max = 0.1
    
    # Initialize swarm
    swarm = []
    for i in range(P):
        if i == 0:
            # Use initial solution as first particle
            swarm.append(h.copy())
        else:
            # Random perturbations
            pert = np.random.normal(0, 0.1, n_points)
            particle = h + pert
            swarm.append(project_feasible(particle))
    
    # Personal best positions and values
    pbest = [swarm[i].copy() for i in range(P)]
    pbest_vals = [compute_overlap(swarm[i]) for i in range(P)]
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # Initialize velocities
    vel = [np.random.uniform(-v_max, v_max, n_points) for _ in range(P)]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(P):
            # Evaluate current particle
            current = swarm[i]
            current_val = compute_overlap(current)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = current.copy()
                pbest_vals[i] = current_val
                
                if current_val < gbest_val:
                    gbest = current.copy()
                    gbest_val = current_val
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            vel[i] = (w * vel[i] + 
                     c1 * r1 * (pbest[i] - current) + 
                     c2 * r2 * (gbest - current))
            
            # Clip velocities
            vel[i] = np.clip(vel[i], -v_max, v_max)
            
            # Update position
            swarm[i] = current + vel[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
    
    # Final validation
    final_C5 = compute_overlap(gbest)
    
    return (gbest.tolist(), final_C5, n_points)
