import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, backtracking=True):
    """Frank-Wolfe optimization for the Erdős problem"""
    # Start with the best h so far
    h_current = h.copy()
    
    for iteration in range(max_iter):
        # Compute correlations (overlaps) for all shifts
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        
        # Find maximum overlap and its corresponding shift(s)
        max_val = np.max(corr_full)
        max_indices = np.where(np.isclose(corr_full, max_val))[0]
        
        # Compute subgradient at maximum point (simplified)
        # We take the indices where the max occurs and average the contributions
        subgradient = np.zeros_like(h_current)
        for i in range(len(h_current)):
            for idx in max_indices:
                # Compute contribution from correlation at shift idx
                shift = idx - (len(h_current) - 1)
                if shift > 0:
                    j = (i - shift) % len(h_current)
                    subgradient[i] += (1 - h_current[j])
                else:
                    j = (i + (-shift)) % len(h_current)
                    subgradient[i] += (1 - h_current[j]) - 1
        
        # Normalize the subgradient
        subgradient = subgradient / len(max_indices)
        
        # Solve linearized problem: min s <g, s> subject to constraints
        # Leading term in Frank-Wolfe is -g per coordinate
        # s = argmin_s <g, s> + (1/2)*||s||^2  s.t. sum_s = n/2, 0 <= s <= 1
        # solution is s = max(0, 1 - g) if sum is > n/2, etc.
        # Empirical solution: move toward component that maximizes -g[i]
        min_g_index = np.argmin(subgradient)
        s = np.zeros_like(h_current)
        s[min_g_index] = 1  # Find a direction that reduces the subgradient
        
        # Project s back to feasible domain
        # This is simplified for here: target to increase value at min_g_index
        # We approximate by setting that component to 1 and projecting
        s = project_to_feasible(s, n_points)

        # Backtracking line search (simplified)
        if backtracking:
            gamma = 0.9
            for _ in range(20):  # few iterations to find good step size   
                h_new = h_current + gamma * (s - h_current)
                h_new = project_to_feasible(h_new, n_points)
                if evaluate_erdos_solution(h_new, dx) < evaluate_erdos_solution(h_current, dx):
                    h_current = h_new
                    break
                else:
                    gamma *= 0.7
        else:
            gamma = 0.5
            h_current = h_current + gamma * (s - h_current)
            h_current = project_to_feasible(h_current, n_points)
    
    return h_current
