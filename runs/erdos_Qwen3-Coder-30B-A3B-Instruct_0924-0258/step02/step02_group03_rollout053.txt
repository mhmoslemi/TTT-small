import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    
    # Clamp to [0,1] again after scaling
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 70
        initial_h = np.ones(n_points) * 0.5
    
    # PSO parameters
    swarm_size = 40
    max_iter = 10000
    w = 0.7   # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        # Start from initial_h with small perturbations
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate(h, 2.0 / n_points))
    
    # Initialize global best
    best_idx = np.argmin(pbest_values)
    gbest = pbest[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            for j in range(n_points):
                velocities[i][j] = (w * velocities[i][j] +
                                   c1 * r1 * (pbest[i][j] - particles[i][j]) +
                                   c2 * r2 * (gbest[j] - particles[i][j]))
                
                # Clip velocity to prevent overshooting
                max_vel = 0.05 * (1.0 - 0.0)  # 5% of range
                velocities[i][j] = np.clip(velocities[i][j], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points / 2.0)
            
            # Evaluate new position
            new_value = evaluate(particles[i], 2.0 / n_points)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = new_value
                
                # Update global best if needed
                if new_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = new_value
        
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate(gbest, 2.0 / n_points)
    return (gbest.tolist(), float(final_c5), n_points)
