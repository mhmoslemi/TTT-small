import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population using initial_h_values if available"""
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            h_base = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h_base)), h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        h = project(h_base + np.random.normal(0, 0.1, n_points), n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Apply mutation to an individual"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Perform two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Settings
    pop_size = 50
    n_points = 30
    max_generations = 2000
    dx = 2.0 / n_points
    
    # Initialize
    population = initialize_population(n_points, pop_size)
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    best_n_points = n_points
    
    stagnation_count = 0
    mutation_rate = 0.2
    
    # GA loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        offspring = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1 = crossover(selected[i], selected[i+1])
                child2 = crossover(selected[i+1], selected[i])
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                offspring.extend([child1, child2])
            else:
                child = mutate(selected[i], mutation_rate)
                offspring.append(child)
        
        # Replace worst individuals
        offspring_fitness = [evaluate_erdos_solution(h, dx) for h in offspring]
        combined_population = population + offspring
        combined_fitness = fitness + offspring_fitness
        
        # Keep best individuals
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_population[i] for i in sorted_indices[:pop_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:pop_size]]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive mutation rate
        if stagnation_count > 50:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
            stagnation_count = 0
    
    return (list(best_h), best_c5, best_n_points)
