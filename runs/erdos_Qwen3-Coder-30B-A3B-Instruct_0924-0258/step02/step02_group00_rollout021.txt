import numpy as np
from scipy import signal
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available

        n_points = len(h_start)

        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    omega = 0.7
    omega_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Multi-resolution sweep
    resolutions = [25, 30, 35]
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_vals = []
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            pbest.append(h.copy())
            pbest_vals.append(evaluate_erdos_solution(h, dx_try))
            
        # Global best
        gbest_idx = np.argmin(pbest_vals)
        gbest = pbest[gbest_idx].copy()
        gbest_val = pbest_vals[gbest_idx]
        
        # PSO iteration
        for iter_idx in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update inertia weight
            omega_t = omega - (omega - omega_end) * (iter_idx / max_iter)
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (omega_t * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply feasibility
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                curr_val = evaluate_erdos_solution(particles[i], dx_try)
                
                # Update personal best
                if curr_val < pbest_vals[i]:
                    pbest[i] = particles[i].copy()
                    pbest_vals[i] = curr_val
                    
                    # Update global best
                    if curr_val < gbest_val:
                        gbest = particles[i].copy()
                        gbest_val = curr_val
                        
            # Update best solution
            if gbest_val < best_c5:
                best_h = gbest.copy()
                best_c5 = gbest_val
                
        h_start = best_h.copy()

        h = h_start.copy()
        for _ in range(n_iter):
            if time() - start_time > budget_s * 0.95:
                break
            # Evaluate current value and gradient
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            max_val = corr[max_idx]
            # Approximate gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, len(h))
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
            h -= 0.001 * grad
            h = project_to_feasible(h, len(h))
        return h
    
    # Attempt refinement if time allows
    if time() - start_time < budget_s * 0.9:
        refined_h = gradient_descent_refine(best_h, dx, 100)
        refined_c5 = evaluate_erdos_solution(refined_h, dx)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    return (list(best_h), best_c5, n_points)
