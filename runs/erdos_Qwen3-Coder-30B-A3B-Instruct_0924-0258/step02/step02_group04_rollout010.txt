```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize simplex with m points"""
    simplex = []
    
    # Start with initial_h_values or a simple pattern
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h0 = np.array(initial_h_values)
    else:
        # Simple uniform distribution as fallback
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_to_feasible(h0, n_points)
    simplex.append(h0.copy())
    
    # Generate additional points by moving along random directions
    np.random.seed(42)
    for i in range(m - 1):
        h = h0 + np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, n_points)
        simplex.append(h.copy())
    
    return simplex

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # PSO parameters
    n_points = 30
    dx = 2.0 / n_points
    swarm_size = 10
    max_iter = 2000
    w = 0.7
    c1 = c2 = 2.0
    shaking_threshold = 50
    grad_refine_interval = 200
    max_shake = 0.1
    
    # Initialize swarm
    simplex = initialize_simplex(n_points, swarm_size)
    swarm = [h.copy() for h in simplex]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Initialize personal and global bests
    pbest = [h.copy() for h in swarm]
    pbest_c5 = [evaluate_erdos_solution(h, dx) for h in swarm]
    
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # Track stagnation
    stagnation_count = 0
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if c5_new < pbest_c5[i]:
                pbest[i] = swarm[i].copy()
                pbest_c5[i] = c5_new
                
                # Check if global best improved
                if c5_new < gbest_c5:
                    gbest = swarm[i].copy()
                    gbest_c5 = c5_new
                    stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Apply shaking if no improvement
        if stagnation_count > shaking_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, max_shake, n_points)
                swarm[i] = project_to_feasible(swarm[i] + noise, n_points)
            
            stagnation_count = 0
        
        # Periodically apply gradient refinement
        if iteration % grad_refine_interval == 0 and iteration > 0:
            # Refine top 10 particles
            top_indices = np.argsort(pbest_c5)[:min(10, len(pbest_c5))]
            for idx in top_indices:
                # L-BFGS-B optimization
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, pbest[idx], method='L-BFGS-B', 
                                 bounds=bounds, options={'maxiter': 100})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_c5 = evaluate_erdos_solution(refined, dx)
                        if refined_c5 < pbest_c5[idx]:
                            pbest[idx] = refined
                            pbest_c5[idx] = refined_c5
                            
                            # Update global best if necessary
                            if refined_c5 < gbest_c5:
                                gbest = refined.copy()
                                gbest_c5 = refined_c5
                except:
                    pass
    
    # Final refinement of global best
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, 
                     options={'maxiter': 200})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_c5 = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_c5, n_points)
```
