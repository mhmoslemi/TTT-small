import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    # Set up optimization parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 500
    
    # dx for discretization
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute all correlations (full mode)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def initialize_population():
        population = []
        # Seed with current h
        population.append(h.copy())
        # Add random variations
        for _ in range(population_size - 1):
            pert = np.random.normal(0, 0.05, n_points)
            new_h = h + pert
            population.append(project_feasible(new_h))
        return population
    
    def differential_evolution_step(population, generation):
        new_population = []
        for i in range(population_size):
            # Select three distinct parents
            parents_indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[parents_indices[0]], population[parents_indices[1]], population[parents_indices[2]]
            
            # Mutation
            trial = a + mutation_factor * (b - c)
            
            # Crossover
            for j in range(n_points):
                if np.random.rand() > crossover_prob:
                    trial[j] = population[i][j]
            
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Selection
            if compute_overlap(trial) < compute_overlap(population[i]):
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        return new_population
    
    # Initialize population
    population = initialize_population()
    
    # Track the best solution
    best_h = None
    best_C5 = float('inf')
    last_improvement = 0
    
    # Evolution loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate population
        fitnesses = [compute_overlap(ind) for ind in population]
        current_best_idx = np.argmin(fitnesses)
        current_best_C5 = fitnesses[current_best_idx]
        
        if current_best_C5 < best_C5:
            best_C5 = current_best_C5
            best_h = population[current_best_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Stop if no improvement for a while
        if last_improvement > 30:
            break
            
        # Generate next population
        population = differential_evolution_step(population, gen)
    
    # Final validation
    if best_h is None:
        best_h = h.copy()
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
