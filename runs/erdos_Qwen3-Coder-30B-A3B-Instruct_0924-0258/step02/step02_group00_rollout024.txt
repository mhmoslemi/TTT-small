import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    omega = 0.7
    omega_decay = 0.99
    c1, c2 = 2.0, 2.0
    inertia_weight = omega
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / res
        if res != n_points:
            # Interpolate from previous resolution
            old_h = best_h
            x_old = np.linspace(0, 2, len(old_h))
            x_new = np.linspace(0, 2, res)
            h_interp = np.interp(x_new, x_old, old_h)
            h_start = project_to_feasible(h_interp, res)
            n_points = res
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_scores = []
        
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.1, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            score = evaluate_erdos_solution(h, dx)
            personal_best.append(h.copy())
            personal_best_scores.append(score)
            
        # Global best
        global_best_idx = np.argmin(personal_best_scores)
        global_best = personal_best[global_best_idx].copy()
        global_best_score = personal_best_scores[global_best_idx]
        
        # PSO iterations
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            for p in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[p] = (inertia_weight * velocities[p] + 
                                c1 * r1 * (personal_best[p] - particles[p]) +
                                c2 * r2 * (global_best - particles[p]))
                
                # Update position
                particles[p] += velocities[p]
                particles[p] = project_to_feasible(particles[p], n_points)
                
                # Evaluate new score
                score = evaluate_erdos_solution(particles[p], dx)
                
                # Update personal best
                if score < personal_best_scores[p]:
                    personal_best[p] = particles[p].copy()
                    personal_best_scores[p] = score
                    
                    # Update global best
                    if score < global_best_score:
                        global_best = particles[p].copy()
                        global_best_score = score
            
            inertia_weight *= omega_decay
            
            # Early stopping
            if i > 10 and abs(global_best_score - best_c5) < 1e-5:
                break
    
        # Refinement
        h_refined = global_best.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros_like(h_refined)
            for i in range(len(h_refined)):
                eps = 1e-5
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                c5_base = evaluate_erdos_solution(h_refined, dx)
                grad[i] = (c5_plus - c5_base) / eps
            h_refined -= 0.001 * grad
            h_refined = project_to_feasible(h_refined, n_points)
        
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
