import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Particle Swarm Optimization
    swarm_size = 30
    max_iter = 1000
    inertia = 0.9
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution if available
    if 'h_init' in locals():
        swarm.append(h_init.copy())
        velocities.append(np.zeros_like(h_init))
    
    # Fill the rest with random particles
    for i in range(swarm_size - len(swarm)):
        particle = np.random.uniform(0, 1, n_points)
        particle = project_feasible(particle)
        swarm.append(particle)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # PSO parameters
    best_positions = [p.copy() for p in swarm]
    best_values = [compute_overlap(p) for p in swarm]
    global_best_idx = np.argmin(best_values)
    global_best = best_positions[global_best_idx].copy()
    global_best_value = best_values[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight (linearly decreasing)
        inertia = 0.9 - (0.5 * iteration / max_iter)
        
        for i in range(len(swarm)):
            particle = swarm[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (best_positions[i] - particle)
            social = c2 * r2 * (global_best - particle)
            velocity = inertia * velocity + cognitive + social
            
            # Clip velocity
            velocity = np.clip(velocity, -velocity_max, velocity_max)
            
            # Update position
            particle += velocity
            
            # Project to feasible region
            particle = project_feasible(particle)
            
            # Evaluate fitness
            current_value = compute_overlap(particle)
            
            # Update personal best
            if current_value < best_values[i]:
                best_values[i] = current_value
                best_positions[i] = particle.copy()
                
                # Update global best
                if current_value < global_best_value:
                    global_best_value = current_value
                    global_best = particle.copy()
            
            # Update swarm
            swarm[i] = particle
            velocities[i] = velocity
    
    # Final validation
    final_h = global_best
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
