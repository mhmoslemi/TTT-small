import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    pop_size = 50
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        population = [h_start.copy()]
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    else:
        n_points = 35
        population = []
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    
    dx = 2.0 / n_points
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic algorithm
    tournament_k = 5
    crossover_rate = 0.8
    mutation_rate = 0.2
    mutation_strength = 0.1
    
    gen = 0
    while time() - start_time < budget_s * 0.98:
        gen += 1
        # Selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_k)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            p1, p2 = selected[i], selected[i+1]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * p1 + (1 - alpha) * p2
                child2 = (1 - alpha) * p1 + alpha * p2
            else:
                child1, child2 = p1.copy(), p2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child1 += np.random.normal(0, mutation_strength, n_points)
            if np.random.rand() < mutation_rate:
                child2 += np.random.normal(0, mutation_strength, n_points)
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Keep population size
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        new_fitness = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Survival (elitism + replacement)
        combined_pop = list(zip(population, fitness))
        combined_new = list(zip(new_population, new_fitness))
        combined = sorted(combined_pop + combined_new, key=lambda x: x[1])
        
        population = [h for h, _ in combined[:pop_size]]
        fitness = [f for _, f in combined[:pop_size]]
        
        # Update best
        new_best_idx = np.argmin(fitness)
        if fitness[new_best_idx] < best_c5:
            best_h = population[new_best_idx].copy()
            best_c5 = fitness[new_best_idx]
    
    return (list(best_h), best_c5, n_points)
