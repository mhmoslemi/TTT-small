import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    P = 40  # swarm size
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.15
    
    # Initialize swarm
    swarm = []
    for i in range(P):
        if i == 0:
            # Use initial solution if available
            particle = h.copy()
        else:
            # Random perturbation around the initial solution
            particle = h + np.random.normal(0, 0.1, n_points)
        particle = project_feasible(particle)
        swarm.append(particle)
    
    # Personal best positions and values
    personal_best = [p.copy() for p in swarm]
    personal_best_values = [compute_overlap(p) for p in swarm]
    
    # Global best
    best_idx = np.argmin(personal_best_values)
    global_best = personal_best[best_idx].copy()
    global_best_value = personal_best_values[best_idx]
    
    # Velocity
    velocities = [np.zeros(n_points) for _ in range(P)]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(P):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate new position
            new_value = compute_overlap(swarm[i])
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = new_value
                
                # Update global best
                if new_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = new_value
    
    # Final evaluation
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
