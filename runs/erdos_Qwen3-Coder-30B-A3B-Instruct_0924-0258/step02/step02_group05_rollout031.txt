import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def solve_convex_subproblem(h_fixed):
        # Create variables for the unconstrained h
        h_var = cp.Variable(n_points)
        
        # Build the constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Build the objective: minimize max overlap
        # We use a big-M formulation with auxiliary variables
        M = 1000.0
        max_overlap = cp.Variable()
        # Add constraints for each overlap
        for k in range(2 * n_points - 1):
            # This constructs the convolution properly
            idx = k - (n_points - 1)
            if idx >= 0:
                # When k >= n_points-1, the overlap is from h[i] to h[i+idx]
                # Sum over valid indices (i, i+idx)
                if idx < n_points:
                    # The actual calculation of the overlap
                    # h[i] * (1 - h[i+idx]) for all valid i
                    expr = 0
                    for i in range(n_points - idx):
                        expr += h_var[i] * (1 - h_fixed[i+idx])
                    # Add constraint: overlap <= max_overlap
                    constraints.append(expr <= max_overlap)
            else:
                # When k < n_points-1, the overlap is from h[i+idx] to h[i]
                # This means i+idx >= 0, so i >= -idx  
                # And i < n_points, so i+idx < n_points+idx
                if idx >= -n_points + 1:
                    expr = 0
                    for i in range(-idx, n_points):
                        expr += h_var[i] * (1 - h_fixed[i+idx])
                    constraints.append(expr <= max_overlap)
        
        # Minimize max_overlap
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CBC, verbose=False)
            return h_var.value
        except:
            # If optimization fails, return original
            return h_fixed
        
    # Optimization loop
    current_h = h.copy()
    best_h = current_h.copy()
    best_C5 = compute_overlap(current_h)
    
    max_iter = 100
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex subproblem
        new_h = solve_convex_subproblem(current_h)
        
        # Project to feasible region
        new_h = project_feasible(new_h)
        
        # Check if it improved
        new_C5 = compute_overlap(new_h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h.copy()
            
        current_h = new_h.copy()
        
        # Early stopping if no significant improvement
        if abs(new_C5 - best_C5) < 1e-6:
            break
    
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
