import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population with variations of initial_h_values or random"""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base)), base)
    else:
        base = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        h = base + np.random.normal(0, 0.1, n_points)
        h = project(h, n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, tournament_size=5):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in indices]
        winner_idx = indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 2000
    mutation_rate = 0.2
    crossover_rate = 0.7
    stagnation_limit = 200
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    best_n_points = n_points
    
    stagnation = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create offspring
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            if np.random.rand() < crossover_rate:
                child1 = crossover(parent1, parent2)
                child2 = crossover(parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            offspring.extend([child1, child2])
        
        # Replace worst individuals
        population = offspring[:pop_size]
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            stagnation = 0
        else:
            stagnation += 1
        
        # Adapt mutation rate
        if stagnation > 50:
            mutation_rate = min(mutation_rate * 1.1, 0.5)
        
        if stagnation > stagnation_limit:
            # Reset population with new random individuals
            population = initialize_population(n_points, pop_size)
            fitness = [evaluate_erdos_solution(h, dx) for h in population]
            stagnation = 0
            mutation_rate = 0.2
            
        generation += 1
    
    return (list(best_h), best_c5, best_n_points)
