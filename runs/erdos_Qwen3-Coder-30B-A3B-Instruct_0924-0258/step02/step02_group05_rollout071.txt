import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def solve_feasible_lp(c5_candidate):
        # Define the LP problem for fixed C₅
        # Variables: h[0], ..., h[n_points-1]
        # Objective: minimize 0
        # Equality constraint: sum(h) = target_sum
        # Inequality constraints: for all valid k values, overlap constraint
        
        # Consistent with problem setup
        # For each k from -n_points+1 to n_points-1:
        #   sum_{i} h[i] * (1 - h[i + k]) * dx <= c5_candidate
        
        A_eq = np.array([np.ones(n_points)])
        b_eq = np.array([target_sum])
        
        # Build inequality constraints
        A_ub_list = []
        b_ub_list = []
        
        # Generate overlap constraints for each k
        for k in range(-n_points+1, n_points):
            # Compute the indices i and i+k that appear in the overlap sum
            # For each i, we may have i+k in bounds
            
            # For i in range(n_points), compute the contribution of h[i] to the sum
            row = np.zeros(n_points)
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in [0, n_points-k-1]
                for i in range(n_points-k):
                    row[i] = 1
            else:
                # h[i] * (1 - h[i+k]) for i in [-k, n_points-1]
                # i+k >= 0, so i >= -k = |k|
                for i in range(-k, n_points):
                    row[i] = 1
                    
            # But we also need to subtract h[i]*h[i+k] terms as well, causing them
            # to be indicated by the full convolution approach
            
            # Better formulation:
            # Sum_{i=0}^{n_points+k-1} h[i] * (1 - h[i-k]) dx <= c5
            # When i-k < 0, we ignore (no overlapping)
            # When i-k >= n_points, we ignore (out of bounds)
            
            # Let's rebuild this differently:
            # We want: sum_{i} h[i] * (1 - h[i+k]) * dx <= c5
            # i goes from 0 to total_len-1
            # i+k also goes from k to k+total_len-1
            # So, we eat from both ends. 
            
            # Better idea: First build convolution matrix
            # Actually write direct indexing:
            c = np.zeros(n_points)
            for i in range(n_points):
                j = i + k
                if 0 <= j < n_points:
                    c[i] = 1.0   # correlation = (1 - h[i+k]) contribution of h[i]
            c = c * dx
            A_ub_list.append(c)
            b_ub_list.append(c5_candidate)
            
        # Add bounds on h[i]
        A_ub_list.append(np.eye(n_points))
        A_ub_list.append(-np.eye(n_points))
        b_ub_list.extend([1.0] * n_points)  # h[i] <= 1
        b_ub_list.extend([0.0] * n_points)  # 0 <= h[i] (implies -h[i] <= 0)
        
        A_ub = np.vstack(A_ub_list)
        b_ub = np.array(b_ub_list)
        
        # Solve as a linear feasibility problem (objective = 0)
        # Note: linprog minimizes c @ x, so we minimize 0 @ x to just check feasibility
        # But we don't actually minimize anything, just want to find a feasible solution
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = linprog(np.zeros(n_points), A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        
        if result.success:
            return result.x
        else:
            return None
    
    # Binary search over C₅
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    while (high - low) > tolerance and time.time() - start_time < budget_s - 1:
        C5_candidate = (low + high) / 2
        
        # Solve LP for current candidate
        attempt_h = solve_feasible_lp(C5_candidate)
        
        if attempt_h is not None:
            # Check that the solution is valid
            attempt_h = project_feasible(attempt_h)
            attempt_c5 = evaluate_c5(attempt_h)
            
            if attempt_c5 <= C5_candidate + 1e-12:  # Allow numerical tolerance
                # Feasible and within bound, try smaller C5
                high = C5_candidate
                if attempt_c5 < best_c5:
                    best_c5 = attempt_c5
                    best_h = attempt_h.copy()
            else:
                # There might have been an issue with solver tolerance - go larger
                low = C5_candidate
        else:
            # Not feasible, go larger
            low = C5_candidate
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
