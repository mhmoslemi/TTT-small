import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Only consider positive shifts
        mid = len(corr) // 2
        autocorr = corr[mid:]
        # Compute the cross-correlation with 1-h
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum of positive shifts
        return np.max(cross_corr[mid:])

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_gradient(h_vec, shift_idx):
        # Compute gradient for correlation at shift_idx
        grad = np.zeros_like(h_vec)
        # This gradient computation assumes the correlation is from left to right
        # Since cross_corr[i] = sum_{j} h[j] * (1-h[j+shift_idx])
        # Gradient w.r.t. h[i] = (1-h[i+shift_idx]) (when i+shift_idx < n_points)
        # and 1-h[i-shift_idx] (when i >= shift_idx)
        if shift_idx < n_points:
            # For h[i] we look at the contribution to A_{shift_idx}
            if shift_idx > 0:
                grad[shift_idx:] += (1 - h_vec[:-shift_idx])
            if shift_idx < n_points:
                grad[:-shift_idx] += (1 - h_vec[shift_idx:])
        return grad

    def compute_shift_grads(h_vec):
        # Compute all the gradients for all shifts
        # Returns the shift with min autocorrelation and its gradient
        cross_corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(cross_corr) // 2
        shifts = cross_corr[mid:]
        min_shift = np.argmin(shifts)
        min_val = shifts[min_shift]
        grad = compute_gradient(h_vec, min_shift)
        return min_shift, grad, min_val

    # Optimization parameters
    learning_rate = 0.01
    patience = 50
    patience_counter = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute the shift with smallest correlation
        min_shift, grad, min_corr = compute_shift_grads(h)
        
        # Update h using gradient
        h = h + learning_rate * grad
        h = project_h(h)
        
        # Reevaluate
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            # Adjust learning rate if no improvement for a while
            if patience_counter >= patience:
                learning_rate *= 0.5
                patience_counter = 0
                
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
