import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, target_sum, dx, trust_radius, max_iter=100):
    """Perform trust-region step for minimizing C5"""
    # Function to compute gradient(-C5) and objective
    def objective_and_grad(h_vec):
        h = h_vec.copy()
        h = project_to_feasible(h, target_sum)
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        # Gradient of -C5 with respect to h[i]
        # Use finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (corr_plus[max_k] - corr_minus[max_k]) / (2 * eps)
        # Since corr[max_k] increases as h[i] increases, we want negative gradient
        return np.max(corr) * dx, -grad
    
    x = h.copy()
    radius = trust_radius
    last_improvement = 0
    for iteration in range(max_iter):
        curr_val, curr_grad = objective_and_grad(x)
        if curr_val < 1e-10:
            return x.copy(), curr_val
        
        # Update with gradient step (projected to trust region if needed)
        x_new = x + 0.1 * curr_grad
        x_new = project_to_feasible(x_new, target_sum)
        
        try:
            new_val, _ = objective_and_grad(x_new)
        except:
            x_new = project_to_feasible(x + 0.1 * curr_grad, target_sum)
            try:
                new_val, _ = objective_and_grad(x_new)
            except:
                new_val = np.inf
            
        if new_val < curr_val:
            x = x_new.copy()
            last_improvement = iteration
            radius = min(1.0, radius * 1.2)
        else:
            radius = max(0.1, radius * 0.5)
            
        if iteration - last_improvement > 50:
            break
            
    return x, evaluate(x, dx)
