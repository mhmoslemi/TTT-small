import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Since correlation is symmetric, we're interested in the valid range
        # The first and last n_points-1 values are not valid as they're from different ranges
        valid_corr = corr[n_points-1:2*n_points-1]
        return np.min(valid_corr)

    def compute_gradient_min_autocorr(h_vec):
        # Find the minimal autocorrelation shift
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        valid_corr = corr[n_points-1:2*n_points-1]
        min_shift = np.argmin(valid_corr)
        # The shift that gives min autocorrelation
        shift = min_shift # corresponds to correlation index from 0 to n_points-1
        
        # Compute gradient for this shift
        grad = np.zeros_like(h_vec)
        # For shift s, the autocorrelation is sum h[i]*h[i+s]
        # Gradient w.r.t. h[i] is h[i+s] if i+s < n_points
        # This needs careful handling - indices
        for i in range(n_points):
            if i + shift < n_points:
                grad[i] += h_vec[i + shift]
            if i >= shift:
                grad[i] += h_vec[i - shift]
        return grad

    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    learning_rate = 0.01
    patience = 0
    patience_limit = 1000
    
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute gradient for minimal autocorrelation
        grad = compute_gradient_min_autocorr(h)
        
        # Update h in direction of increasing minimal autocorrelation
        h_new = h + learning_rate * grad
        h_new = project_h(h_new)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                # Reduce learning rate if no progress
                learning_rate *= 0.5
                patience = 0
            # Small random perturbation to escape local minima
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
        h = h_new
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
