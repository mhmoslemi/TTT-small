import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    population = [project(base_h + np.random.normal(0, 0.05, n_points), n_points) for _ in range(pop_size)]
    return population

def mutate(h, prob=0.2, sigma=0.1):
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitness, tournament_size=5):
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def genetic_algorithm(n_points, pop_size=50, max_generations=1000, time_limit=600):
    start_time = time()
    
    # Initialize
    population = initialize_population(n_points, pop_size)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    mutation_rate = 0.2
    stagnant_generations = 0
    
    for generation in range(max_generations):
        if time() - start_time > time_limit * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(individual, dx) for individual in population]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnant_generations = 0
        else:
            stagnant_generations += 1
            
        # Early stopping
        if stagnant_generations > 200 or mutation_rate > 0.5:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Create offspring
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            if np.random.rand() < 0.7:
                child1 = crossover(parent1, parent2)
                child2 = crossover(parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate, 0.1)
            child2 = mutate(child2, mutation_rate, 0.1)
            
            offspring.extend([child1, child2])
        
        # Replace worst individuals
        population = sorted(population, key=lambda x: evaluate_erdos_solution(x, dx))
        for i in range(min(len(offspring), len(population) // 2)):
            population[-(i+1)] = offspring[i]
            
        # Adapt mutation rate
        if stagnant_generations > 50:
            mutation_rate = min(0.5, mutation_rate + 0.02)
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    
    # Initial population
    population = initialize_population(n_points, 50)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    # Optimize at resolution 30  
    h_opt, c5_opt = genetic_algorithm(n_points, pop_size=50, time_limit=budget_s)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt
    
    # If we have more time, try to refine further  
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and best_h is not None:
        # Try one more optimization pass
        h_refined, c5_refined = genetic_algorithm(n_points, pop_size=100, time_limit=remaining_time)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    return (list(best_h), best_c5, n_points)
