import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_points = len(h_init)
    else:
        # Default initial construction if none provided
        n_points = 45
        h_init = np.full(n_points, 1.0 / n_points)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h_init * target_sum / np.sum(h_init)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use scipy minimize with SLSQP
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Track best solution
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    iteration = 0
    
    # Gradient-based optimization with projection
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Optimize using scipy minimize
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 50, 'ftol': 1e-9})
            if result.success:
                h = result.x
        except:
            # Fallback to simple gradient descent with projection
            pass
            
        h = project_h(h)
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Add small random perturbation to escape local minima
        noise = 0.002 * np.random.normal(0, 1, n_points)
        h = h + noise
        h = project_h(h)
        
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
