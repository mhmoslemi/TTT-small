import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    # Start with a coarse search to identify worst cases
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    # Priority queue approach using cutting planes to refine
    # Keep adding the axis that gives the maximal overlap
    current_h = h.copy()
    max_iter = 10000
    iteration = 0
    
    # Initialize constraints set - start with a coarse search
    coarse_shifts = np.linspace(0, 2, 21)  # 21 points from 0 to 2
    constraints_list = []  # list of (shift, constraint_value)
    
    # Initial evaluation
    def compute_overlap_for_shift(h_vec, shift):
        # Compute overlap integral for specific shift value
        shift_idx = int(shift / dx)  # convert to index
        if shift_idx >= len(h_vec):
            return 0.0
        # Shifted version of 1-h at that shift
        shifted_one_minus_h = np.concatenate([np.zeros(shift_idx), 1 - h_vec[:len(h_vec)-shift_idx]])
        overlap = np.sum(h_vec * shifted_one_minus_h) * dx
        return overlap
    
    # Simple optimization with adaptive shift search
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Get current C5 value
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Identify worst performing shift by sampling in fine grid
        fine_shifts = np.linspace(0, 2, 201)
        best_shift = 0.0
        max_overlap = -1.0
        for shift in fine_shifts:
            overlap_val = compute_overlap_for_shift(current_h, shift)
            if overlap_val > max_overlap:
                max_overlap = overlap_val
                best_shift = shift
                
        # If we've explored similar shifts, check if we're converging
        if any(abs(shift - best_shift) < 0.01 for shift, _ in constraints_list):
            # Add a different small perturbation
            perturbed_shift = best_shift + 0.02 * (np.random.rand() - 0.5)
            perturbed_shift = max(0, min(2, perturbed_shift))
            best_shift = perturbed_shift
        
        # Add to constraints
        constraints_list.append((best_shift, max_overlap))
        
        # Apply projected gradient descent with sparse constraints
        # Formulate convex problem to improve h:
        # min h1 + h2 + ... + hn subject to overlap constraints
        # This is a convex feasibility problem approach
        h_vars = cp.Variable(n_points)
        
        # Constraint: sum = target_sum
        sum_constraint = cp.Equality(cp.sum(h_vars), target_sum)
        
        # Constraint: bounds
        bounds_constraints = [0 <= h_vars[i], h_vars[i] <= 1 for i in range(n_points)]
        
        # Constraint: overlaps
        constraints = [sum_constraint] + bounds_constraints
        slack_var = cp.Variable(1)
        for shift, value in constraints_list:
            shift_idx = int(shift / dx)
            # Formulate correlation inequality for this shift  
            if shift_idx < len(h_vars):
                shifted_one_minus_h = cp.hstack([cp.Constant(0.0)] * shift_idx + 
                                                    [1 - h_vars[i] for i in range(len(h_vars) - shift_idx)])  
                corr_val = cp.sum(cp.multiply(h_vars[:len(h_vars)-shift_idx], shifted_one_minus_h[:len(h_vars)-shift_idx])) * dx
                constraints.append(corr_val <= slack_var)
            else:
                # vertical shift beyond array bounds - skip
                pass
        
        objective = cp.Minimize(slack_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                # Project back to valid range
                h_new = np.clip(h_new, 0, 1)
                # Ensure sum constraint
                h_new = h_new * target_sum / np.sum(h_new)
                current_h = h_new
        except:
            # Fall back to simple projection if CVX doesn't work
            pass
        
        iteration += 1
        if iteration >= max_iter:
            break

    # Final projection and validation
    current_h = project_h(current_h)
    final_c5 = compute_c5(current_h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = current_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
