import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_objective(h_vec):
        return compute_c5(h_vec)
    
    # Use a more structured optimization approach with cutting planes
    start_time = time.time()
    
    # Compute initial overlaps
    def get_overlaps(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return corr
    
    # Get initial worst case
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Set up constraints
    h_vars = Variable(n_points)
    eta = Variable()
    
    constraints = [h_vars >= 0, h_vars <= 1, sum(h_vars) == target_sum, eta >= 0]
    
    # Arbitrary but reasonable initial overlap calculation
    overlap_indices = np.arange(n_points)
    
    # Evaluate a few initial shifts to seed the solver
    num_shifts = min(50, n_points)
    shift_indices = np.linspace(0, n_points - 1, num_shifts, dtype=int)
    
    # Initial: use a slightly better construction than basic uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)
    
    # Starting point - slightly modified to avoid trivial solution
    h = project_h(h)
    
    # Iterative refinement with cutting plane
    log_idx = 0
    max_iter = 10000
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Compute all overlaps
        overlaps = get_overlaps(h)
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        else:
            # Add small randomization to escape local minima
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = project_h(h)
        
        # Better approach: use an optimization with cutting plane
        # Re-estimate the maximum overlap with more precision
        # Sample more densely if improvement isn't seen in many iterations
        if iteration % 100 == 0:
            # Do a small BS with varying shift grids to avoid missing better overlaps
            shift_step = max(1, n_points // 20)
            dense_shifts = np.arange(0, n_points, shift_step)
            dense_shifts = np.clip(dense_shifts, 0, n_points - 1)
            
            # Values to cut
            temp_overlaps = []
            for s in dense_shifts:
                # A fast way to compute overlap for shift s
                if s == 0:
                    o = np.sum(h * (1 - h))
                else:
                    o = np.sum(h[:-s] * (1 - h[s:]))
                temp_overlaps.append(o)
            
            # Refine around max of dense grid
            max_dense_idx = np.argmax(temp_overlaps)
            best_shift_in_dense = dense_shifts[max_dense_idx]
            
            # Possibly smart enumeration in window
            sample_shifts = np.arange(max(best_shift_in_dense - 3, 0), min(best_shift_in_dense + 4, n_points))
            sample_overlaps = []
            for s in sample_shifts:
                if s == 0:
                    o = np.sum(h * (1 - h))
                else:
                    o = np.sum(h[:-s] * (1 - h[s:]))
                sample_overlaps.append(o)
            try:
                best_window_shift = sample_shifts[np.argmax(sample_overlaps)]
            except:
                best_window_shift = best_shift_in_dense
                
            # Try to improve with that position
            if best_window_shift < 10 or best_window_shift > (n_points - 10):
                # Too near edges, try more uniform
                h = project_h(h + np.random.normal(0, 0.001, n_points))
            else:
                # Focus on local improvement
                mag_factor = 0.01
                # Simple gradient ascent idea (would be like using central diff)
                h = h + np.random.normal(0, mag_factor, n_points)
                h = project_h(h)
        
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final projection and final evaluation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
