import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a simple initial pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def generate_simplex(n_points, m):
    """Generate m points forming a simplex in the feasible region"""
    if m > n_points:
        m = n_points
    simplex = []
    
    # Start with the initial feasible point
    h_base = initialize_h(n_points)
    simplex.append(h_base.copy())
    
    # Generate m-1 additional points by perturbing base
    for i in range(m-1):
        # Create a random direction
        direction = np.random.randn(n_points)
        # Scale to get a reasonable perturbation
        perturbation = 0.1 * direction / np.linalg.norm(direction)
        h_perturbed = project_to_feasible(h_base + perturbation, n_points)
        simplex.append(h_perturbed)
    
    return np.array(simplex)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    
    # PSO parameters
    swarm_size = 10
    w = 0.7
    c1 = c2 = 2.0
    max_iter = 2000
    stagnation_threshold = 50
    refinement_freq = 200
    shaking_threshold = 50
    shaking_noise = 0.1
    
    # Initialize swarm
    swarm = generate_simplex(n_points, swarm_size)
    velocities = np.zeros_like(swarm)
    
    # Evaluate initial swarm
    fitness = np.array([evaluate_erdos_solution(h, dx) for h in swarm])
    
    # Track personal bests
    pbest_positions = swarm.copy()
    pbest_fitness = fitness.copy()
    
    # Global best
    gbest_idx = np.argmin(fitness)
    gbest_position = swarm[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # Track stagnation
    last_improvement = 0
    best_fitness_history = []
    
    iteration = 0
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - swarm[i]) + 
                           c2 * r2 * (gbest_position - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_to_feasible(swarm[i], n_points)
        
        # Evaluate fitness
        new_fitness = np.array([evaluate_erdos_solution(h, dx) for h in swarm])
        
        # Update personal bests
        improved_mask = new_fitness < pbest_fitness
        pbest_positions[improved_mask] = swarm[improved_mask]
        pbest_fitness[improved_mask] = new_fitness[improved_mask]
        
        # Update global best
        new_gbest_idx = np.argmin(new_fitness)
        if new_fitness[new_gbest_idx] < gbest_fitness:
            gbest_position = swarm[new_gbest_idx].copy()
            gbest_fitness = new_fitness[new_gbest_idx]
            last_improvement = iteration
            best_fitness_history = []
        
        best_fitness_history.append(gbest_fitness)
        
        # Stagnation check
        if iteration - last_improvement > shaking_threshold:
            # Apply shaking
            for i in range(swarm_size):
                noise = np.random.normal(0, shaking_noise, n_points)
                swarm[i] = project_to_feasible(swarm[i] + noise, n_points)
            last_improvement = iteration
        
        # Gradient refinement of top candidates periodically
        if iteration % refinement_freq == 0 and iteration > 0:
            # Sort by fitness
            sorted_indices = np.argsort(new_fitness)
            top_indices = sorted_indices[:min(5, swarm_size)]
            
            for idx in top_indices:
                h_refined = swarm[idx].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_refined, method='L-BFGS-B', 
                                 bounds=bounds, constraints=cons, 
                                 options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        refined_fitness = evaluate_erdos_solution(h_refined, dx)
                        if refined_fitness < new_fitness[idx]:
                            swarm[idx] = h_refined
                            new_fitness[idx] = refined_fitness
                            if refined_fitness < gbest_fitness:
                                gbest_position = h_refined
                                gbest_fitness = refined_fitness
                                last_improvement = iteration
                except:
                    pass
        
        iteration += 1
    
    # Final gradient refinement on the best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, gbest_position, method='L-BFGS-B', 
                     bounds=bounds, constraints=cons, 
                     options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < gbest_fitness:
                gbest_position = final_h
                gbest_fitness = final_c5
    except:
        pass
    
    return (list(gbest_position), gbest_fitness, n_points)
