import numpy as np
import cvxpy as cp
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize bounds for binary search
    C_low = 0.37
    C_high = 0.38085
    best_feasible_h = None
    best_feasible_c5 = float('inf')
    
    # Run binary search for 60 seconds
    search_start = time()
    while time() - search_start < 60 and (C_high - C_low) > 1e-6:
        C_mid = (C_low + C_high) / 2
        
        # Create CVXPY variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # Compute correlation sum
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h[i] * (1 - h[j])
            constraints.append(corr_sum <= C_mid * n_points / 2)
        
        # Create and solve problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_result = h.value
                c5_result = evaluate_erdos_solution(h_result, dx)
                if c5_result < best_feasible_c5:
                    best_feasible_h = h_result.copy()
                    best_feasible_c5 = c5_result
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
    
    # Refinement with gradient descent
    if best_feasible_h is not None:
        best_h = best_feasible_h.copy()
        best_c5 = best_feasible_c5
    
    # Refinement loop
    refine_start = time()
    max_refine_iters = 100
    for i in range(max_refine_iters):
        if time() - refine_start > budget_s - 60:
            break
            
        # Gradient descent step
        # Compute gradient of overlap with respect to h
        def gradient(h):
            grad = np.zeros_like(h)
            for i in range(n_points):
                grad_i = 0
                for k in range(-n_points+1, n_points):
                    j = (i + k) % n_points
                    grad_i += (1 - 2*h[i]) * (1 - h[j]) if k >= 0 else (1 - 2*h[i]) * (1 - h[(i-k)%n_points])
                grad[i] = grad_i
            return grad
        
        # Objective function
        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Optimization step with projected gradient
        grad = gradient(best_h)
        learning_rate = 0.01
        h_new = best_h - learning_rate * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Update if better
        if obj(h_new) < obj(best_h):
            best_h = h_new.copy()
            best_c5 = obj(best_h)
    
    return (list(best_h), best_c5, n_points)
