import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use scipy minimize with SLSQP for better optimization
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Early termination counter
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    iteration = 0
    
    # Multi-start approach with different initializations
    initial_configs = []
    for _ in range(5):
        # Random perturbation
        h_perturbed = h + 0.1 * np.random.normal(0, 1, n_points)
        h_perturbed = project_h(h_perturbed)
        initial_configs.append(h_perturbed.copy())
        
        # Greedy step function construction
        h_greedy = np.zeros(n_points)
        # Create a pattern that is likely to reduce overlap
        for i in range(n_points):
            if i < n_points // 2:
                h_greedy[i] = 1.0
            else:
                h_greedy[i] = 0.0
        h_greedy = project_h(h_greedy)
        initial_configs.append(h_greedy.copy())
    
    # Add original h
    initial_configs.append(h.copy())
    
    for i, init_h in enumerate(initial_configs):
        h_test = init_h.copy()
        for _ in range(500):  # Few iterations per start
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break
                
            # Optimize using scipy minimize
            result = minimize(evaluate_objective, h_test, method='SLSQP', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 50, 'ftol': 1e-8})
            
            if result.success:
                h_test = result.x
            else:
                # If optimization failed, do a simple projection descent
                h_test = project_h(h_test)
                
            h_test = project_h(h_test)
            new_c5 = compute_c5(h_test)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_test.copy()
                
            # Add noise to escape local minima
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h_test = h_test + noise
            h_test = project_h(h_test)
            
        if elapsed > budget_s:
            break

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    # Return the best solution within time limit
    return (best_h.tolist(), final_c5, n_points)
