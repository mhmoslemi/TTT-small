import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Start with a simple uniform distribution
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    # Ensure the integral constraint
    dx = 2.0 / len(h)
    h = h * (0.5 / np.sum(h))  # Rescale to make integral = 1
    
    def compute_c5(h_vec):
        # Correlation computation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to match integral constraint
        h_vec = h_vec * (0.5 / np.sum(h_vec))
        return h_vec

    def gradient_descent_step(h_vec, learning_rate=0.01):
        # Get current function value
        current_c5 = compute_c5(h_vec)
        # Compute gradient by finite difference
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update
        h_new = h_vec - learning_rate * grad
        h_new = project_h(h_new)
        return h_new

    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.01
    iteration = 0
    max_iter = 10000

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Perform several gradient steps
        for step in range(5):
            h = gradient_descent_step(h, learning_rate)
        
        # Compute new c5 value
        new_c5 = compute_c5(h)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Decay learning rate
        learning_rate *= 0.999
        
        iteration += 1
        if iteration > max_iter:
            break

    # Final validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
