import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize a simplex of particles"""
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    particles = [project_to_feasible(h0, n_points)]
    
    # Generate additional points by perturbing along random directions
    for _ in range(m - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = project_to_feasible(particles[0] + noise, n_points)
        particles.append(h_perturbed)
    
    return particles

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    w = 0.7
    c1 = c2 = 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    
    # Initialize swarm
    particles = initialize_simplex(n_points, swarm_size)
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Evaluate initial fitness
    fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Track best
    best_idx = np.argmin(fitness)
    gbest = particles[best_idx]
    gbest_fit = fitness[best_idx]
    
    # Track stagnation
    stagnation_count = 0
    
    # PSO loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (particles[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            new_fit = evaluate_erdos_solution(particles[i], dx)
            if new_fit < fitness[i]:
                fitness[i] = new_fit
                
            # Update personal best
            if new_fit < fitness[i]:
                particles[i] = particles[i]  # Already updated
            
        # Update global best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < gbest_fit:
            gbest = particles[best_idx].copy()
            gbest_fit = fitness[best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Shaking mechanism
        if stagnation_count > shaking_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            stagnation_count = 0
        
        # Periodic LBFGS refinement
        if iteration % lbfgs_interval == 0 and iteration > 0:
            # Sort particles by fitness and refine top 10
            indices = np.argsort(fitness)[:min(10, swarm_size)]
            for i in indices:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, particles[i], method='L-BFGS-B', 
                                  bounds=bounds, options={'maxiter': 100})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_fit = evaluate_erdos_solution(refined, dx)
                        if refined_fit < fitness[i]:
                            particles[i] = refined
                            fitness[i] = refined_fit
                            
                            # Update global best if needed
                            if refined_fit < gbest_fit:
                                gbest = refined.copy()
                                gbest_fit = refined_fit
                except:
                    pass
    
    # Final refinement
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 200})
        if res.success:
            gbest_final = project_to_feasible(res.x, n_points)
            final_fit = evaluate_erdos_solution(gbest_final, dx)
            if final_fit < gbest_fit:
                gbest = gbest_final
                gbest_fit = final_fit
    except:
        pass
    
    return (list(gbest), gbest_fit, n_points)
