import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def solve_convex_subproblem(h_fixed, n_points, dx):
        # Build constraint matrix for convex relaxation
        h = cp.Variable(n_points)
        
        # Objective: minimize max overlap
        obj = cp.Minimize(cp.max(h @ (1 - h_fixed) * dx))
        
        # Constraints: h in [0,1], sum = target_sum
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
        
        # Solve problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-6)
        except:
            return h_fixed.copy()
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return np.array(h.value).flatten()
        else:
            return h_fixed.copy()
    
    # Initialize
    current_h = project_feasible(h.copy())
    best_c5 = compute_overlap(current_h)
    best_h = current_h.copy()
    last_improvement = 0
    max_iter = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex subproblem
        new_h = solve_convex_subproblem(current_h, n_points, dx)
        
        # Project to feasible set
        new_h = project_feasible(new_h)
        
        # Evaluate
        new_c5 = compute_overlap(new_h)
        
        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Update current
        current_h = new_h
        
        # Early termination
        if last_improvement > 10:
            break
    
    # Final validation
    final_h = project_feasible(best_h)
    final_c5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
