import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, population_size):
    """Initialize population using initial_h_values or create variations"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    population.append(project(base_h, n_points))
    
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h = project(base_h + noise, n_points)
        population.append(h)
    
    return population

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx]

def crossover(h1, h2):
    """Perform two-point crossover"""
    n_points = len(h1)
    idx1, idx2 = sorted(np.random.choice(n_points, 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return offspring

def mutate(h, mutation_rate, sigma=0.1):
    """Apply Gaussian mutation to a single individual"""
    if np.random.rand() < mutation_rate:
        n_points = len(h)
        num_mutations = max(1, int(0.3 * n_points))
        indexes = np.random.choice(n_points, num_mutations, replace=False)
        h[indexes] += np.random.normal(0, sigma, num_mutations)
        h = project(h, len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    dx = 2.0 / n_points
    max_generations = 1000
    stagnation_limit = 200
    no_improvement_count = 0
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    best_n_points = n_points
    
    generation = 0
    while (time() - start_time < budget_s * 0.95) and (generation < max_generations):
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Stagnation check
        if no_improvement_count > stagnation_limit:
            # Reset population
            population = initialize_population(n_points, population_size)
            fitness = [evaluate_erdos_solution(h, dx) for h in population]
            no_improvement_count = 0
            
        # Selection, crossover and mutation
        new_population = []
        
        # Elitism: keep the best
        elite_idx = np.argmin(fitness)
        new_population.append(population[elite_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring = crossover(parent1, parent2)
            else:
                offspring = parent1.copy()
                
            # Mutation
            offspring = mutate(offspring, mutation_rate)
            
            new_population.append(offspring)
        
        # Ensure we have exact population size
        if len(new_population) > population_size:
            new_population = new_population[:population_size]
        
        population = new_population
        
        generation += 1
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, best_n_points)
