import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def optimize_step(h_vec):
        # Optimize using scipy minimize
        def evaluate_objective(h_vec):
            return compute_c5(h_vec)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        try:
            result = minimize(evaluate_objective, h_vec, method='SLSQP', 
                             bounds=bounds, constraints=constraints, 
                             options={'maxiter': 50, 'ftol': 1e-8})
            if result.success:
                return project_h(result.x)
        except:
            pass
        return project_h(h_vec)
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    population.append(h.copy())
    
    for i in range(1, population_size):
        # Start with a perturbed version of the initial solution
        perturbation = np.random.normal(0, 0.01, n_points)
        h_perturbed = h + perturbation
        population.append(project_h(h_perturbed))
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for individual in population:
            c5 = compute_c5(individual)
            fitness_scores.append(-c5)  # Negative for minimization
        
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Keep top 50% as parents
        parents = population[:population_size//2]
        
        # Generate offspring
        offspring = []
        for i in range(population_size - len(parents)):
            if np.random.random() < crossover_rate and len(parents) >= 2:
                # Crossover
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Clone with mutation
                parent = np.random.choice(parents)
                child = parent.copy()
                
            # Mutation
            if np.random.random() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child[mutate_idx] += np.random.normal(0, 0.02)
                
                # Mutation for 1-2 additional elements
                num_mutations = np.random.randint(1, 3)
                for _ in range(num_mutations):
                    idx = np.random.randint(0, n_points)
                    child[idx] += np.random.normal(0, 0.01)
                
            child = project_h(child)
            offspring.append(child)
        
        # Replace bottom 50% of population with offspring
        population = parents + offspring
        
        # Track best solution
        current_best = population[0]
        current_c5 = compute_c5(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
    
    # Final optimization of the best solution
    best_h = optimize_step(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
