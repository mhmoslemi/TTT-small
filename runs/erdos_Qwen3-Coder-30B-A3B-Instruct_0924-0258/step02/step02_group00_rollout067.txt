import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, backtracking_steps=10):
    # Find maximum overlap and which shifts contribute
    corr = np.correlate(h, 1 - h, mode='full')
    max_k = np.argmax(corr)
    max_val = corr[max_k]
    
    # Compute subgradient for the active shift
    # For k-th shift: g_i = (1 - h[(i-k)%n]) - h[i] for active indices
    g = np.zeros(n_points)
    for i in range(n_points):
        g[i] = (1 - h[(i - max_k) % n_points]) - h[i]
    
    # Frank-Wolfe linear problem: min s: g^T s subject to sum(s) = sum(h), 0 <= s <= 1
    # We solve by projected gradient descent approach
    
    # Simple version: lowest coordinate for descent direction (or the one with max negative g)
    # Adaptively select descent direction
    s = np.zeros(n_points)
    s_flags = np.zeros(n_points, dtype=bool)
    
    # Try to find a helpful descent direction
    argmin_g = np.argmin(g)
    if g[argmin_g] < 0:
        s[argmin_g] = 1.0
    else:
        # All g non-negative; choose greedily
        for i in range(n_points):
            if g[i] < 0:
                s[i] = 1.0
                break
        else:
            # Guarantee descent even if no g < 0
            # Use the one that cancels most (i.e., non-zero components of h)
            # but prioritize low values to fix balance
            s = np.ones_like(h) * 1.0 / n_points  # smallmove
            
    # Scale to match sum(h)
    s_sum = np.sum(s)
    if s_sum > 0:
        s = s * np.sum(h) / s_sum
        
    # Try line search for step size
    best_h_new = h.copy()
    best_val = evaluate_erdos_solution(h, dx)
    
    step_sizes = np.linspace(0.1, 0.9, backtracking_steps)
    for gamma in step_sizes:
        h_new = h + gamma * (s - h)
        h_new = project_to_feasible(h_new, n_points)
        val_new = evaluate_erdos_solution(h_new, dx)
        if val_new < best_val:
            best_h_new = h_new
            best_val = val_new
    
    return best_h_new, best_val
