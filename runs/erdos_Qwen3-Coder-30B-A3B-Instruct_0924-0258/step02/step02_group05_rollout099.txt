import numpy as np
from scipy.optimize import linprog
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # Binary search over C5 values
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        C5_candidate = (low + high) / 2.0
        
        # Formulate LP problem
        # Variables: h[0], h[1], ..., h[n_points-1]
        # Objective: minimize 0
        # Subject to:
        # 1. sum(h) = target_sum
        # 2. for each k: sum_{i} h[i] * (1 - h[i+k]) * dx <= C5_candidate
        
        # Prepare constraints
        # Equality constraint
        A_eq = np.ones((1, n_points))
        b_eq = np.array([target_sum])
        
        # Inequality constraints
        A_ub = []
        b_ub = []
        
        # For all possible overlaps, create constraint
        for k in range(-n_points+1, n_points):
            # Create overlap constraint for this k
            row = np.zeros(n_points)
            for i in range(n_points):
                if 0 <= i - k < n_points:
                    row[i] = 1.0
            A_ub.append(row)
            b_ub.append(C5_candidate * dx)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        # Bounds for variables
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Solve the LP
        try:
            # Use a fast LP solver
            result = linprog(
                c=np.zeros(n_points),
                A_eq=A_eq,
                b_eq=b_eq,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=bounds,
                method='highs'
            )
            
            if result.success:
                h_candidate = result.x
                # Project to feasible domain
                h_candidate = project_feasible(h_candidate)
                
                # Check if this is better than current best
                current_C5 = compute_overlap(h_candidate)
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = h_candidate.copy()
                
                # This C5 is feasible
                high = C5_candidate
            else:
                # This C5 is not feasible
                low = C5_candidate
        except Exception:
            # If LP fails, treat as infeasible
            low = C5_candidate
        
        # Small random perturbation to avoid local minima
        if time.time() - start_time < budget_s - 5:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_feasible(h_perturbed)
            current_C5 = compute_overlap(h_perturbed)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_perturbed.copy()

    # Final verification
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
