import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    omega = 0.7
    omega_decay = 0.99
    c1, c2 = 2.0, 2.0
    
    # Multi-resolution loop
    resolutions = [25, 30, 35]
    
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_vals = []
        
        for i in range(n_particles):
            if res_idx == 0:  # first resolution
                h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            else:  # interpolate from previous solution
                h = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.normal(0, 0.01, n_points))
            val = evaluate_erdos_solution(h, dx_try)
            personal_best.append(h.copy())
            personal_best_vals.append(val)
        
        # Global best tracking
        global_best_idx = np.argmin(personal_best_vals)
        global_best = personal_best[global_best_idx].copy()
        global_best_val = personal_best_vals[global_best_idx]
        
        # PSO iterations
        for iter_count in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (omega * velocities[i] 
                               + c1 * r1 * (personal_best[i] - particles[i])
                               + c2 * r2 * (global_best - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply feasibility constraints
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate current particle
                current_val = evaluate_erdos_solution(particles[i], dx_try)
                
                # Update personal best
                if current_val < personal_best_vals[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_val
            
            # Update inertia weight
            omega *= omega_decay
            
        # Refine best solution with local optimization
        refine_h = global_best.copy()
        refine_c5 = global_best_val
        
        # Use simple gradient-based refinement (gradient descent)
        learning_rate = 0.001
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            corr = np.correlate(refine_h, 1 - refine_h, mode='full')
            # Gradient approximation via finite differences
            grad = np.zeros_like(refine_h)
            eps = 1e-5
            for j in range(len(refine_h)):
                h_plus = refine_h.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                val_plus = evaluate_erdos_solution(h_plus, dx_try)
                grad[j] = (val_plus - refine_c5) / eps
            refine_h = refine_h - learning_rate * grad
            refine_h = project_to_feasible(refine_h, n_points)
            refine_c5 = evaluate_erdos_solution(refine_h, dx_try)
            
            if refine_c5 < best_c5:
                best_h = refine_h.copy()
                best_c5 = refine_c5
        
        h_start = global_best.copy()
            
    return (list(best_h), best_c5, n_points)
