import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization loop
    t = 0
    last_improvement = 0

        corr = np.correlate(best_h, 1 - best_h, mode='full')

        k = np.argmax(corr)

        grad = np.zeros(n_points)

        # Scale by dx
        grad *= dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2

            noise = np.random.normal(0, 0.05, n_points)
