import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a simple piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def create_simplex(n_points, m):
    """Create a simplex of m points in n_points-dimensional space"""
    simplex = []
    
    # Start with the initial point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_to_feasible(h0, n_points)
    simplex.append(h0.copy())
    
    # Add m-1 additional points
    for i in range(1, m):
        h = h0.copy()
        # Add small random noise, then project
        noise = np.random.normal(0, 0.1, n_points)
        h = h + noise
        h = project_to_feasible(h, n_points)
        simplex.append(h.copy())
    
    return simplex

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    n_points = 30
    dx = 2.0 / n_points
    
    # Time allocation for optimization
    max_iterations = 2000
    iteration_time = (budget_s - 5) / max_iterations if max_iterations > 0 else 0.001
    
    # Initialize swarm
    swarm_size = 10
    simplex = create_simplex(n_points, swarm_size)
    
    # Initialize personal bests and global best
    pbest = [s.copy() for s in simplex]
    pbest_values = [evaluate_erdos_solution(s, dx) for s in simplex]
    
    global_best_idx = np.argmin(pbest_values)
    gbest = pbest[global_best_idx].copy()
    gbest_value = pbest_values[global_best_idx]
    
    # PSO parameters
    w = 0.7
    c1, c2 = 2.0, 2.0
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    stagnation_counter = 0
    last_improvement = 0
    
    # Main PSO loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Update velocities and positions
        for i in range(swarm_size):
            if time() - start_time > budget_s * 0.98:
                break
                
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - simplex[i]) + 
                           c2 * r2 * (gbest - simplex[i]))
            
            simplex[i] = simplex[i] + velocities[i]
            
            # Project to feasible region
            simplex[i] = project_to_feasible(simplex[i], n_points)
            
            # Evaluate and update personal best
            current_value = evaluate_erdos_solution(simplex[i], dx)
            if current_value < pbest_values[i]:
                pbest[i] = simplex[i].copy()
                pbest_values[i] = current_value
                
                if current_value < gbest_value:
                    gbest = simplex[i].copy()
                    gbest_value = current_value
                    last_improvement = iteration
                    
        # Check for stagnation and apply shaking if no improvement
        if iteration - last_improvement > 50:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.1, n_points)
                simplex[i] = project_to_feasible(simplex[i] + noise, n_points)
            
            # Also try Local search on the current best
            try:
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx),
                    gbest,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 5, 'ftol': 1e-6}
                )
                if res.success:
                    refined = project_to_feasible(res.x, n_points)
                    refined_value = evaluate_erdos_solution(refined, dx)
                    if refined_value < gbest_value:
                        gbest = refined.copy()
                        gbest_value = refined_value
            except:
                pass
                
            last_improvement = iteration
        
        # Occasionally do local refinement on top solutions
        if iteration % 200 == 0 and iteration > 0:
            # Sort by personal best values and do local search on best few
            sorted_indices = np.argsort(pbest_values)
            for i in range(min(5, len(sorted_indices))):
                idx = sorted_indices[i]
                try:
                    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(
                        lambda h: evaluate_erdos_solution(h, dx),
                        pbest[idx],
                        method='L-BFGS-B',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 3, 'ftol': 1e-6}
                    )
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_value = evaluate_erdos_solution(refined, dx)
                        if refined_value < pbest_values[idx]:
                            pbest[idx] = refined.copy()
                            pbest_values[idx] = refined_value
                            
                            if refined_value < gbest_value:
                                gbest = refined.copy()
                                gbest_value = refined_value
                                last_improvement = iteration
                except:
                    pass
    
    # Final refinement of best solution with L-BFGS-B
    try:
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            lambda h: evaluate_erdos_solution(h, dx),
            gbest,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 10, 'ftol': 1e-8}
        )
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < gbest_value:
                gbest = final_h.copy()
                gbest_value = final_c5
    except:
        pass
        
    return (list(gbest), gbest_value, n_points)
