import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure constraint is satisfied
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_shifted_overlaps(h_vec):
        overlaps = []
        for s in range(n_points):
            if s == 0:
                overlap = np.sum(h_vec * (1 - h_vec)) * dx
            else:
                overlap = np.sum(h_vec[:-s] * (1 - h_vec[s:])) * dx
            overlaps.append(overlap)
        return np.array(overlaps)

    def update_h_for_shift(h_vec, shift_idx, eta):
        # Compute gradient w.r.t. h for given shift
        grad = np.zeros_like(h_vec)
        s = shift_idx
        if s == 0:
            grad = 1 - 2 * h_vec
        else:
            # Handle shift s
            # d/dh_i [h_i * (1 - h_{i+s})] = 1 - h_{i+s} - h_i * delta_{i+s}
            # But it's tricky due to indexing...
            # Let's directly compute for the specific term.
            # We compute the gradient of overlap w.r.t. h:
            # For shift s:
            # overlap = sum_{i=0}^{n-s-1} h[i] * (1 - h[i+s])
            # d/dh_k overlap = 1 - h[k+s] if k < n-s-1 else 0
            # d/dh_k overlap = -h[k-s] if k >= s and k < n
            # So 1 - h[k+s] if 0 <= k <= n-s-1
            #    -h[k-s] if s <= k <= n-1
            # Matching these, for different cases:
            # grad[k] = 1 - h[k+s] if k+s < n => k < n-s
            #         = 0 if k >= n-s
            
            # But let's approach it correctly:
            # The full overlap is sum_{i=0}^{n-s-1} h[i] * (1 - h[i+s])
            # For j-th element, differentiate the sum over i including j:
            # d/dh_j sum_{i: i == j} h[i] * (1 - h[i+s]) = 
            # 1 - h[j+s], if j < n-s
            # = -h[j-s] if j >= s and j < n
            # = 0 otherwise
            
            # Collect F[i] = 1 - h[i+s] for i < n-s
            # Collect G[i] = -h[i-s] for i >= s and i < n
            
            grad = np.zeros_like(h_vec)
            for i in range(n_points - s):
                grad[i] += (1.0 - h_vec[i + s])
            for i in range(s, n_points):
                grad[i] -= h_vec[i - s]
        return grad

    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Alternating optimization with shift ordering
    max_iter = 20000
    iteration = 0
    learning_rate = 0.01
    patience = 0
    patience_limit = 100
    
    shift_order = list(range(n_points))
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current overlaps
        overlaps = compute_shifted_overlaps(h)
        eta = np.max(overlaps)
        
        # Shuffle shift indices for better exploration
        np.random.shuffle(shift_order)
        
        # Update h using gradient descent for each shift using the current eta
        # We'll compute the accumulated gradient
        grad_total = np.zeros_like(h)
        for s in shift_order:
            grad_s = update_h_for_shift(h, s, eta)
            grad_total += grad_s
        
        # Normalize gradient
        grad_total = grad_total / (np.linalg.norm(grad_total) + 1e-10)
        
        # Update h
        h = h - learning_rate * grad_total
        
        # Project h to satisfy constraints
        h = project_h(h)
        
        # Evaluate
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
            # Adapt learning rate
            learning_rate *= 0.99
        else:
            patience += 1
            # Adapt learning rate (faster)
            if patience > 20:
                learning_rate *= 0.9

        iteration += 1
        if iteration > max_iter:
            break

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
