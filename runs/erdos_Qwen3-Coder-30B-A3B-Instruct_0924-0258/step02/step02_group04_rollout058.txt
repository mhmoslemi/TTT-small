import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population using initial_h_values if available"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for _ in range(pop_size):
        perturbed = base_h + np.random.normal(0, 0.05, n_points)
        population.append(project(perturbed, n_points))
    
    return population

def crossover(h1, h2):
    """Two-point crossover"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def mutate(h, mutation_rate=0.2, sigma=0.1):
    """Gaussian mutation"""
    if np.random.rand() < mutation_rate:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select individual with best fitness from tournament"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitness[i] for i in selected_idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.7
    max_generations = 1000
    stagnation_limit = 200
    
    # Initialize
    population = initialize_population(n_points, pop_size)
    dx = 2.0 / n_points
    
    best_fitness = float('inf')
    best_individual = None
    
    last_improvement = 0
    generation = 0
    prev_best_fitness = float('inf')
    
    while time() - start_time < budget_s * 0.98 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for individual in population:
            f = evaluate_erdos_solution(individual, dx)
            fitness.append(f)
            if f < best_fitness:
                best_fitness = f
                best_individual = individual.copy()
                
        # Check stagnation
        if best_fitness < prev_best_fitness:
            last_improvement = generation
            prev_best_fitness = best_fitness
            
        if generation - last_improvement > stagnation_limit:
            # Reset population if no improvement
            population = initialize_population(n_points, pop_size)
            last_improvement = generation
            
        # Selection, crossover, mutation
        new_population = []
        
        # Keep best individual
        best_idx = np.argmin(fitness)
        new_population.append(population[best_idx])
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                offspring1 = crossover(parent1, parent2)
                offspring2 = crossover(parent2, parent1)
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
                
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            new_population.extend([offspring1, offspring2])
        
        population = new_population[:pop_size]
        generation += 1
        
        # Adaptive mutation rate
        if generation % 50 == 0:
            mutation_rate = min(0.5, mutation_rate * 1.1)
    
    if best_individual is None:
        # Fallback
        h_start = np.ones(n_points) * 0.5
        if 'initial_h_values' in globals():
            h_start = np.array(initial_h_values)
            if len(h_start) != n_points:
                h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
        best_individual = project(h_start, n_points)
        best_fitness = evaluate_erdos_solution(best_individual, dx)
    
    return (list(best_individual), best_fitness, n_points)
