import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient for dominant lag k"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = -(1 - h[j] - h[i]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    
    iter_without_improvement = 0
    max_iter_no_improvement = 50
    
    # Optimization loop
    while time() - start_time < budget_s * 0.95:
        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for dominant lag
        grad = compute_gradient(h, k, n_points, dx)
        
        # Simple penalty for constraints
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(h)
        penalty_grad += 2 * lambda_penalty * (h_sum - n_points/2) * np.ones(n_points)
        penalty_grad += lambda_penalty * (h > 1) * (-1)  # gradient of max(0, h-1) penalty
        penalty_grad += lambda_penalty * (h < 0) * 1      # gradient of max(0, -h) penalty
        
        # Update h
        h_candidate = h - eta * (grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Accept or reject
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = candidate_c5 - current_c5
        
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_candidate.copy()
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                iter_without_improvement = 0
            else:
                iter_without_improvement += 1
        else:
            iter_without_improvement += 1
        
        # Cool down
        T *= alpha
        
        # Random perturb if no improvement
        if iter_without_improvement >= max_iter_no_improvement:
            h += np.random.randn(n_points) * 0.05
            h = project_feasible(h, n_points)
            iter_without_improvement = 0
            current_c5 = evaluate_erdos_solution(h, dx)
    
    # Multi-resolution refinement
    resolutions = [60, 120]
    for new_n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, new_n_points)
        h_interp = np.interp(x_new, x_orig, best_h)
        h = project_feasible(h_interp, new_n_points)
        n_points = new_n_points
        dx = 2.0 / n_points
        
        # Local optimization
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            k = np.argmax(corr)
            grad = compute_gradient(h, k, n_points, dx)
            h_candidate = h - eta * grad
            h_candidate = project_feasible(h_candidate, n_points)
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            if candidate_c5 < best_c5:
                h = h_candidate.copy()
                best_c5 = candidate_c5
                best_h = h.copy()
    
    return (list(best_h), best_c5, n_points)
