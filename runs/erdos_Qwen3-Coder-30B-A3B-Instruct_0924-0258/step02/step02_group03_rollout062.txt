import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    return h

def optimize_trust_region(h0, n_points, dx, budget_s, seed=42):
    np.random.seed(seed)
    start_time = time.time()
    
    h = h0.copy()
    h = project_to_feasible(h, n_points / 2.0)
    C5 = evaluate(h, dx)
    
    best_h = h.copy()
    best_C5 = C5
    
    # Initialize trust region
    delta = 0.2
    max_delta = 1.0
    min_delta = 0.1
    
    # Hessian approximation (initially identity)
    H = np.eye(n_points)
    
    # For simplicity, we'll compute numerical gradient
    def compute_gradient(h, dx):
        # Direct numerical gradient computation
        grad = np.zeros(n_points)
        h_eval = h.copy()
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_eval.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points / 2.0)
            C5_plus = evaluate(h_plus, dx)
            
            h_minus = h_eval.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, n_points / 2.0)
            C5_minus = evaluate(h_minus, dx)
            
            grad[i] = (C5_plus - C5_minus) / (2 * eps)
        return grad
    
    # Trust region subproblem (solve approximately using projected gradient)
    def solve_trust_region_subproblem(g, H, delta):
        # Projected gradient descent inside trust region
        d = -g
        # Check if direction is feasible
        if np.linalg.norm(d) > delta:
            d *= delta / np.linalg.norm(d)
        return d
    
    # BFGS update for Hessian
    def update_Hessian(H, s, y):
        if np.dot(y, s) < 1e-10:
            return H
        rho = 1.0 / np.dot(y, s)
        I = np.eye(len(H))
        H_new = (I - rho * np.outer(s, y)) @ H @ (I - rho * np.outer(y, s)) + rho * np.outer(s, s)
        return H_new

    iter_count = 0

    while time.time() - start_time < budget_s:
        iter_count += 1
        grad = compute_gradient(h, dx)
        
        # Step
        d = solve_trust_region_subproblem(grad, H, delta)
        
        # Project h + d to feasible region
        h_new = h + d
        h_new = project_to_feasible(h_new, n_points / 2.0)
        
        C5_new = evaluate(h_new, dx)
        
        # Accept or reject
        if C5_new < C5:
            # Accept
            h = h_new
            C5 = C5_new
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
            # Update Hessian (BFGS)
            if iter_count > 1:
                s = h_new - h_prev  
                y = grad_new - grad_prev
                H = update_Hessian(H, s, y)
            delta = min(delta * 1.2, max_delta)
        else:
            # Reject, shrink trust region
            delta = max(delta * 0.5, min_delta)
        
        # Store previous values
        h_prev = h.copy()
        grad_prev = grad.copy()
        grad_new = grad.copy()
    
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Use initial construction or build one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)

        # Create a good starting point like a step function
        h0 = np.ones(n_points)
        h0[:n_points//2] = 0.5
        h0[n_points//2:] = 0.5
        
    dx = 2.0 / n_points
    
    # Optimize
    h_opt, C5_bound = optimize_trust_region(h0, n_points, dx, budget_s, seed)
    
    return (h_opt.tolist(), float(C5_bound), n_points)
