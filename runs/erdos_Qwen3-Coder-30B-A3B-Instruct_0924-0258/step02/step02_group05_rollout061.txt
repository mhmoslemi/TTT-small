import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    # PSO implementation
    swarm_size = 40
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    max_vel = 0.15
    
    # Initialize positions and velocities
    swarm = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best = None
    global_best_val = float('inf')
    
    # Create initial swarm
    for i in range(swarm_size):
        if i == 0:
            pos = h.copy()
        else:
            # perturb around the initial solution
            pos = h + np.random.normal(0, 0.05, n_points)
            pos = np.clip(pos, 0, 1)
            pos = pos * target_sum / np.sum(pos)
        
        swarm.append(pos.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        val = compute_overlap(pos)
        personal_best.append(pos.copy())
        personal_best_vals.append(val)
        
        if val < global_best_val:
            global_best = pos.copy()
            global_best_val = val

    # PSO main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            current_val = compute_overlap(swarm[i])
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < global_best_val:
                    global_best = swarm[i].copy()
                    global_best_val = current_val

    final_C5 = global_best_val
    return (global_best.tolist(), final_C5, n_points)
