import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction or default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    
    # Best solution tracking
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Perturbation parameters
    perturbation_rate = 0.20
    perturbation_std = 0.1
    max_iter = 200
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidates
        candidates = []
        for _ in range(10):  # Create 10 variations
            # Select elements to perturb
            indices = np.random.choice(n_points, int(n_points * perturbation_rate), replace=False)
            
            # Apply perturbation
            h_pert = h_start.copy()
            h_pert[indices] += np.random.normal(0, perturbation_std, len(indices))
            h_pert = project_to_feasible(h_pert, n_points)
            
            # Optimize with SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    candidates.append((h_opt, c5_opt))
            except:
                pass
        
        # Evaluate candidates
        if candidates:
            best_new = min(candidates, key=lambda x: x[1])
            if best_new[1] < best_c5:
                best_h = best_new[0].copy()
                best_c5 = best_new[1]
                h_start = best_h.copy()
    
    # Resolution refinement
    current_h = best_h.copy()
    current_n = n_points
    current_dx = dx
    
    while current_n < 1000 and time() - start_time < budget_s * 0.95:
        # Double resolution
        current_n *= 2
        current_dx = 2.0 / current_n
        
        # Interpolate
        old_x = np.linspace(0, 2, len(current_h))
        new_x = np.linspace(0, 2, current_n)
        current_h = np.interp(new_x, old_x, current_h)
        current_h = project_to_feasible(current_h, current_n)
        
        # Optimize at this resolution
        def obj(h):
            return evaluate_erdos_solution(h, current_dx)
            
        bounds = [(0, 1) for _ in range(current_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - current_n / 2}]
        
        try:
            res = minimize(obj, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                current_h = project_to_feasible(res.x, current_n)
                new_c5 = evaluate_erdos_solution(current_h, current_dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = current_h.copy()
        except:
            pass
    
    return (list(best_h), best_c5, len(best_h))
