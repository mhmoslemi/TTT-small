import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a low-correlation binary-like pattern, and then smooth it
        h = np.zeros(n_points)
        # Place ones approximately evenly and in a way that avoids strong overlaps
        ones_positions = np.linspace(0, n_points-1, n_points//2, dtype=int)
        h[ones_positions] = 1.0
        
        # Apply a smoothing to get intermediate values
        # Use a window of size 3, extended by 1 in each direction
        h_smooth = np.zeros(n_points)
        for i in range(n_points):
            start = max(0, i-1)
            end = min(n_points, i+2)
            h_smooth[i] = np.mean(h[start:end])
        h = h_smooth
        
        # Ensure the sum constraint is met
        h = project_to_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # For tracking best solution
    best_h_global = None
    best_c5_global = float('inf')
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts with different perturbations
        h_restart = h_start.copy()
        for restart in range(3):  # 3 restarts per resolution
            if time() - start_time > budget_s * 0.95:
                break
            start_h = h_restart.copy()
            
            # Use L-BFGS-B optimization on this restart
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, start_h, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5_global:
                        best_h_global = h_opt.copy()
                        best_c5_global = c5_opt
                        best_n_points = n_points
            except:
                pass
                
            # Perturb the current solution
            noise_level = 0.1
            perturbed = np.random.normal(0, noise_level, n_points)
            h_restart = project_to_feasible(start_h + perturbed, n_points)
    
    if best_h_global is not None:
        best_h = best_h_global
        best_c5 = best_c5_global
    else:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(obj, h_start, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                best_h = project_to_feasible(res.x, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
                best_n_points = n_points
        except:
            # Final fallback: use the initial function
            best_h = h_start
            best_c5 = evaluate_erdos_solution(best_h, dx)
            best_n_points = n_points
            
    return (list(best_h), best_c5, best_n_points)
