import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def initialize_swarm(n_particles, n_points):
        swarm = []
        # Add the initial solution if available
        if 'initial_h_values' in globals():
            swarm.append(project_feasible(h.copy()))
        else:
            swarm.append(project_feasible(np.random.rand(n_points) * 0.5))
        
        # Generate additional random particles
        while len(swarm) < n_particles:
            particle = project_feasible(np.random.rand(n_points) * 0.5)
            swarm.append(particle)
        return swarm
    
    # PSO Parameters
    n_particles = 40
    max_iter = 10000
    inertia_start = 0.9
    inertia_end = 0.4
    c1 = 2.0
    c2 = 2.0
    vel_max = 0.1
    
    # Initialize swarm
    swarm = initialize_swarm(n_particles, n_points)
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track personal bests
    personal_best = [p.copy() for p in swarm]
    personal_best_scores = [compute_overlap(p) for p in swarm]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Update inertia weight
        inertia = inertia_start - (inertia_start - inertia_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate
            score = compute_overlap(swarm[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = swarm[i].copy()
                
                # Update global best if needed
                if score < global_best_score:
                    global_best_score = score
                    global_best = swarm[i].copy()
                    last_improvement = 0
                else:
                    last_improvement += 1
            else:
                last_improvement += 1
        
        iter_count += 1
        
        # Early stopping
        if last_improvement > 1000:
            break
    
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
