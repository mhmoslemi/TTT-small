import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h

    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update

    return h

        # Accept or reject based on Metropolis criterion
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):

            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                
            acceptance_count += 1
        
        # Cooling schedule
        T = max(min_T, T * alpha)
        
        # Adaptive temperature adjustment
        if iteration % acceptance_threshold == 0 and iteration > 0:
            if acceptance_count < 0.2 * acceptance_threshold:
