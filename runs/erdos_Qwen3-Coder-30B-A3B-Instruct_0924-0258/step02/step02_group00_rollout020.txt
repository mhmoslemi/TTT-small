import numpy as np
from scipy.fft import convolve
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = convolve(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values)
        n_points = len(h_best)
    else:
        n_points = 30
        h_best = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    w_start = 0.7
    w_end = 0.4
    c1 = c2 = 2.0
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            # Interpolate from previous best
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_best)), 
                                h_best)
            h_best = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            dx = dx_try
            
        # Initialize particles
        particles = []
        velocities = []
        p_best = []
        p_best_scores = []
        
        # Generate initial particles around current best solution
        for _ in range(n_particles):
            h = project_to_feasible(h_best + np.random.normal(0, 0.1, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            p_best.append(h.copy())
            p_best_scores.append(evaluate_erdos_solution(h, dx))
            
        # Swarm best tracking
        s_best_idx = np.argmin(p_best_scores)
        s_best = p_best[s_best_idx].copy()
        s_best_score = p_best_scores[s_best_idx]
        
        # PSO iterations
        for iter in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            w = w_start - (w_start - w_end) * iter / max_iter
            
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (p_best[i] - particles[i]) + 
                                c2 * r2 * (s_best - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Ensure feasibility
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate and update personal best
                score = evaluate_erdos_solution(particles[i], dx)
                if score < p_best_scores[i]:
                    p_best[i] = particles[i].copy()
                    p_best_scores[i] = score
                    
                    # Update swarm best
                    if score < s_best_score:
                        s_best = particles[i].copy()
                        s_best_score = score
                        if s_best_score < c5_best:
                            h_best = s_best.copy()
                            c5_best = s_best_score
            
            # Early stopping
            if s_best_score < c5_best or iter % 50 == 0:
                if s_best_score < c5_best:
                    h_best = s_best.copy()
                    c5_best = s_best_score
                if iter >= 50 and abs(s_best_score - c5_best) < 1e-5:
                    break
        
        # Refinement with gradient descent
        if time() - start_time < budget_s * 0.9:
            h_refined = h_best.copy()
            learning_rate = 0.001
            for _ in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                corr = convolve(h_refined, 1 - h_refined, mode='full')
                max_idx = np.argmax(corr)
                # Compute a simple gradient approximation
                h_plus = h_refined.copy()
                h_plus[max_idx % n_points] += 0.01
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                if c5_plus < evaluate_erdos_solution(h_refined, dx):
                    h_refined = h_plus.copy()
                else:
                    h_refined = project_to_feasible(h_refined + learning_rate * (1 - 2 * h_refined), n_points)
                    
            final_score = evaluate_erdos_solution(h_refined, dx)
            if final_score < c5_best:
                h_best = h_refined.copy()
                c5_best = final_score
    
    return (list(h_best), c5_best, n_points)
