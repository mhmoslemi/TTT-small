import numpy as np
from scipy.signal import fftconvolve
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize with best known solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # PSO parameters
    n_particles = 50
    max_iter = 200
    inertia_w = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_scores = []
        
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.03, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            score = evaluate_erdos_solution(h, dx_try)
            personal_best.append(h.copy())
            personal_best_scores.append(score)
        
        # Global best
        gbest_idx = np.argmin(personal_best_scores)
        gbest = personal_best[gbest_idx].copy()
        gbest_score = personal_best_scores[gbest_idx]
        
        # PSO optimization
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update particles
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                velocities[i] = (inertia_w * velocities[i] +
                               c1 * r1 * (personal_best[i] - particles[i]) +
                               c2 * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                
                # Apply feasibility constraints
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                score = evaluate_erdos_solution(particles[i], dx_try)
                
                # Update personal best
                if score < personal_best_scores[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
        
        inertia_w = max(0.4, inertia_w * 0.99)  # Decrease inertia
            
        # Refinement using gradient descent
        h_refined = gbest.copy()
        learning_rate = 0.001
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            h_prev = h_refined.copy()
            # Simple gradient descent using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            for j in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = evaluate_erdos_solution(h_plus, dx_try)
                grad[j] = (c5_plus - gbest_score) / eps
            
            h_refined = h_refined - learning_rate * grad
            h_refined = project_to_feasible(h_refined, n_points)
            new_score = evaluate_erdos_solution(h_refined, dx_try)
            if new_score >= gbest_score - 1e-8:  # No improvement
                break
            else:
                gbest_score = new_score
                gbest = h_refined.copy()
        
        if gbest_score < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_score
            best_n_points = n_points
            
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
