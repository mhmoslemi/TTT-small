import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a binary-like pattern with equidistant 1s to minimize correlations
        h = np.zeros(n_points)
        ones_count = n_points // 2
        spacing = n_points / (ones_count + 1)
        for i in range(ones_count):
            idx = int((i + 1) * spacing)
            if idx < n_points:
                h[idx] = 1.0
        # Scale to desired sum
        h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Try different resolutions
    resolutions = [15, 30, 45, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 200)
        
        # Optimize multiple times with different random starts
        h_best_local = h_start.copy()
        c5_best_local = evaluate_erdos_solution(h_start, dx)
        
        # Random restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = h_best_local.copy()
            
            # Add noise to break symmetry
            noise = np.random.normal(0, 0.05, n_points)
            h_restart = project_to_feasible(h_restart + noise, n_points)
            
            # L-BFGS optimization
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_restart, method='L-BFGS-B',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < c5_best_local:
                        h_best_local = h_opt.copy()
                        c5_best_local = c5_opt
            except:
                pass
            
            # Perturb and try again
            if restart < 2:
                noise = np.random.normal(0, 0.1, n_points)
                h_restart = project_to_feasible(h_best_local + noise, n_points)
                h_best_local = h_restart
        
        # Update global best
        if c5_best_local < best_c5:
            best_h = h_best_local.copy()
            best_c5 = c5_best_local
            best_n_points = n_points
    
    # Final refinement at highest resolution if needed
    if best_h is not None and best_n_points < 75:
        n_points = 75
        dx = 2.0 / n_points
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_n_points), best_h)
        h_start = project_to_feasible(h_start, n_points)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(objective, h_start, method='L-BFGS-B',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
                    best_n_points = n_points
        except:
            pass
    
    # Fallback to simplest approach if needed
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        h_best, c5_best = h_start, evaluate_erdos_solution(h_start, dx)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(objective, h_best, method='L-BFGS-B',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_best = project_to_feasible(res.x, n_points)
                c5_best = evaluate_erdos_solution(h_best, dx)
        except:
            pass
            
        best_h = h_best
        best_c5 = c5_best
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
