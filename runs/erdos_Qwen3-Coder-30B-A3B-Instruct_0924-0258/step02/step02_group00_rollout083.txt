import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Binary search parameters
    C_low = 0.37
    C_high = 0.38085
    tolerance = 1e-5
    max_iterations = 40
    best_h = None
    best_C = float('inf')
    
    # Use initial construction as warm start if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Binary search for minimum C
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        C_mid = (C_low + C_high) / 2
        
        # Setup CVXPY problem for feasibility check
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(-n_points + 1, n_points):
            # Compute indices with circular wraparound
            shifted_indices = (np.arange(n_points) + k) % n_points
            overlap_constraint = cp.sum(cp.multiply(h_var, 1 - h_var[shifted_indices])) <= C_mid * n_points / 2
            constraints.append(overlap_constraint)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h_var.value
                h_feasible = project_to_feasible(h_solution, n_points)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                
                if c5_feasible < best_C:
                    best_h = h_feasible.copy()
                    best_C = c5_feasible
                    
                C_high = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
    
    # Refinement phase with gradient-based optimization
    if best_h is not None:
        refined_h = best_h.copy()
        refined_c5 = best_C
        remaining_time = budget_s - (time() - start_time)
        
        # Run gradient-based refinement for remaining time
        while time() - start_time < budget_s * 0.95:
            try:
                # Perturb solution and optimize
                h_perturbed = project_to_feasible(refined_h + np.random.normal(0, 0.02, n_points), n_points)
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def cons_func(h):
                    return np.sum(h) - n_points / 2
                
                # Optimization using SLSQP
                res = minimize(
                    obj, 
                    h_perturbed, 
                    method='SLSQP', 
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints={'type': 'eq', 'fun': cons_func},
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < refined_c5:
                        refined_h = h_opt.copy()
                        refined_c5 = c5_opt
                else:
                    break
                    
            except:
                break
                
        best_h = refined_h
        best_C = refined_c5
    
    # Fallback to initial if nothing was found
    if best_h is None:
        h_start = project_to_feasible(h_start, n_points)
        best_C = evaluate_erdos_solution(h_start, dx)
        best_h = h_start
        
    return (list(best_h), best_C, n_points)
