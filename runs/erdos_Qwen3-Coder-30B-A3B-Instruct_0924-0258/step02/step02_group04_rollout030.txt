import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, pop_size):
    """Initialize population with feasible individuals"""
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        if len(base_h) != n_points:
            base_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base_h)), base_h)
    else:
        base_h = np.ones(n_points) * 0.5
    
    population = []
    for i in range(pop_size):
        if i == 0:
            h = base_h.copy()
        else:
            # Add some random perturbation to the base
            h = base_h + np.random.normal(0, 0.05, n_points)
        h = project(h, n_points)
        population.append(h)
    return population

def mutate(h, prob=0.2, sigma=0.1):
    """Apply mutation to individual"""
    if np.random.rand() < prob:
        indexes = np.random.choice(len(h), size=int(len(h)*0.3), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2):
    """Apply crossover between two individuals"""
    idx1, idx2 = np.sort(np.random.choice(len(h1), 2, replace=False))
    offspring = np.copy(h1)
    offspring[idx1:idx2] = h2[idx1:idx2]
    return project(offspring, len(offspring))

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select parent using tournament selection"""
    selected_indices = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_index)
    return [population[i] for i in selected_indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 30
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 1000
    stagnation_limit = 200
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    best_n_points = n_points
    
    generation = 0
    stagnation_count = 0
    mutation_prob = 0.2
    
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adapt mutation rate if stagnation
        if stagnation_count > 100:
            mutation_prob = min(0.5, mutation_prob * 1.1)
        
        # Selection
        selected_population = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected_population[i]
            parent2 = selected_population[(i+1) % pop_size]
            
            if np.random.rand() < 0.7:
                offspring1 = crossover(parent1, parent2)
                offspring2 = crossover(parent2, parent1)
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
                
            offspring1 = mutate(offspring1, mutation_prob)
            offspring2 = mutate(offspring2, mutation_prob)
            
            new_population.extend([offspring1, offspring2])
            
        # Keep only pop_size individuals
        population = new_population[:pop_size]
        
        generation += 1
        
        # Periodic reset if stagnation
        if stagnation_count > stagnation_limit:
            population = initialize_population(n_points, pop_size)
            stagnation_count = 0
            mutation_prob = 0.2
            
    return (list(best_h), best_c5, best_n_points)
