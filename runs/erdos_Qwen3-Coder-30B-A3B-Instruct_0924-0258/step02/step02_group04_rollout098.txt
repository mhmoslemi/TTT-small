import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def fft_optimize(h_start, n_points, dx, time_limit):
    """Optimize using FFT based spectral method"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 1000
    step_size = 1e-4
    penalty_weight = 0.1
    
    # For frequency-domain optimization
    freqs = np.fft.fftfreq(n_points, 1.0 / n_points)
    freqs = freqs[:n_points // 2 + 1]  # Only positive frequencies, including DC
    
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Current solution evaluation
        c5_current = evaluate_erdos_solution(h_best, dx)
        if c5_current < c5_best:
            h_best = h_best.copy()
            c5_best = c5_current
            
        # FFT transform
        H = np.fft.rfft(h_best)
        H_abs = np.abs(H)

        # Add penalty to flatten spectrum (reduce autocorrelation peaks)
        penalty = np.sum(H_abs**4)
        
        # Gradient of penalty
        grad = 4 * H_abs**3
        grad = np.fft.irfft(grad, n_points) 
        # Only update toward reducing peaks in spectrum
        H_new = H - step_size * (np.conj(H) * grad) * penalty_weight
    
        # Reconstruct h
        h_new = np.fft.irfft(H_new, n_points).real
        h_new = project_to_feasible(h_new, n_points)
        
        # Accept new solution if it's better
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            
        # Reduce learning rate if stagnated
        if iteration % 100 == 0 and iteration > 0:
            step_size *= 0.95
    
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple random restarts
        h_restart = h_start.copy()
        for i in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_opt, c5_opt = fft_optimize(h_restart, n_points, dx, time_per_res / 3)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_restart = h_opt + np.random.normal(0, 0.02, n_points)
            h_restart = project_to_feasible(h_restart, n_points)

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = fft_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
