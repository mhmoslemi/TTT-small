import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Particle Swarm Optimization Parameters
    swarm_size = 30
    max_iter = 1000
    inertia_weight = 0.9
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution
    swarm.append(h.copy())
    velocities.append(np.zeros_like(h))
    
    # Add random particles
    for i in range(swarm_size - 1):
        h_init = np.random.rand(n_points) * 0.5 + 0.25  # Random between 0.25 and 0.75
        h_init = project_feasible(h_init)
        swarm.append(h_init)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)  # Random velocity
    
    # Track best positions
    swarm_best = [h.copy() for _ in range(swarm_size)]
    swarm_best_vals = [compute_overlap(h) for h in swarm]
    
    # Global best
    global_best_idx = np.argmin(swarm_best_vals)
    global_best = swarm_best[global_best_idx].copy()
    global_best_val = swarm_best_vals[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight (linear decrease)
        w = inertia_weight - (inertia_weight - 0.4) * (iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (swarm_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity to max
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate new solution
            new_val = compute_overlap(swarm[i])
            
            # Update particle best if improvement
            if new_val < swarm_best_vals[i]:
                swarm_best[i] = swarm[i].copy()
                swarm_best_vals[i] = new_val
                
                # Update global best if improvement
                if new_val < global_best_val:
                    global_best = swarm[i].copy()
                    global_best_val = new_val
        
        # Early stopping if improvement is negligible
        if iteration > 50 and np.std(swarm_best_vals) < 1e-8:
            break
    
    # Return final result
    return (global_best.tolist(), global_best_val, n_points)
