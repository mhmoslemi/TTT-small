import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess with 45 points
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # DE parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 500
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_candidate = h + np.random.normal(0, 0.05, n_points)
        h_candidate = project_feasible(h_candidate)
        population.append(h_candidate)
    
    # Track best solution
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Evolutionary loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h_ind in population:
            fit = compute_overlap(h_ind)
            fitness.append(fit)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        fitness_sorted = [fitness[i] for i in sorted_indices]
        population_sorted = [population[i] for i in sorted_indices]
        
        # Update best solution if needed
        if fitness_sorted[0] < best_C5:
            best_C5 = fitness_sorted[0]
            best_h = population_sorted[0].copy()
            
        # Create new population
        new_population = [population_sorted[0]]  # Keep best
        
        # Differential evolution
        for i in range(1, population_size):
            # Select three different parents
            idxs = [j for j in range(population_size) if j != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population_sorted[a] + mutation_factor * (population_sorted[b] - population_sorted[c])
            mutant = project_feasible(mutant)
            
            # Crossover
            trial = np.copy(population_sorted[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
                    
            # Selection
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitness_sorted[i]:
                new_population.append(trial)
            else:
                new_population.append(population_sorted[i])
                
        population = new_population
    
    # Final validation
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
