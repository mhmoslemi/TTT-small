import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Scale to meet sum constraint
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    else:
        h = np.full_like(h, target_sum / len(h))
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    swarm_size = 40
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max_ratio = 0.05
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        # Start near initial solution with noise
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate(h, 2.0/n_points))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    iteration = 0
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            vel_cognitive = c1 * r1 * (pbest[i] - particles[i])
            vel_social = c2 * r2 * (gbest - particles[i])
            velocities[i] = omega * velocities[i] + vel_cognitive + vel_social
            
            # Limit velocity
            v_max = v_max_ratio * (np.max(particles[i]) - np.min(particles[i]))
            if v_max == 0:
                v_max = 0.1
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points/2.0)
            
            # Evaluate
            current_value = evaluate(particles[i], 2.0/n_points)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
        
        iteration += 1
    
    return (gbest.tolist(), float(gbest_value), n_points)
