import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    # Start timing
    start_time = time.time()
    
    # Initialize variables
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Use a grid for shift evaluation to efficiently find constraint shifts
    shift_indices = np.linspace(0, n_points - 1, min(100, n_points), dtype=int)
    
    # Simple gradient-based descent with constraints
    max_iter = 2000
    iteration = 0
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Evaluate current overlaps
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute finite difference gradient approximation
        grad = np.zeros_like(h)
        epsilon = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            grad[i] = (c5_plus - current_c5) / epsilon
        
        # Optimize with projected gradient step
        step_size = 1e-3
        h_new = h - step_size * grad
        h_new = project_h(h_new)
        
        h = h_new
        iteration += 1
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
