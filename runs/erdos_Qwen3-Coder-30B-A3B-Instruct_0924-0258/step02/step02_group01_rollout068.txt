import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    # Simulated annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    max_no_improvement = 50
    no_improvement_counter = 0
    
    # Initialize
    h_current = project_feasible(h_start.copy(), n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    
    best_h = h_current.copy()
    best_c5 = current_c5
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Compute gradient (approximate)
        # We use the dominant lag for gradient direction
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_lag = np.argmax(corr)
        n_corr = len(corr)
        lag = dominant_lag - (n_corr // 2)
        
        # Approximate gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + lag) % n_points
            gradient[i] = -(1 - h_current[j] - h_current[i])
        
        # Add penalty gradient for feasibility
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(h_current)
        penalty_grad += 2 * lambda_penalty * (h_sum - n_points / 2)
        penalty_grad += 2 * lambda_penalty * (h_current > 1) * (h_current - 1)

        h_candidate = h_current - eta * (gradient + penalty_grad)

            current_c5 = c5_candidate
            no_improvement_counter = 0

                best_h = h_current.copy()
                best_c5 = current_c5
        else:
            no_improvement_counter += 1
            
        # Cool down
        T *= alpha

    return (list(best_h), best_c5, n_points)
