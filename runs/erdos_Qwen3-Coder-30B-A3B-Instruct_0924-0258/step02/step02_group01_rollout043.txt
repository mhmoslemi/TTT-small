import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, dx):
    """Compute gradient for overlap at lag k"""
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = (i + k) % n_points
        grad[i] = -(1 - h[j] - h[i]) * dx
    return grad

    cooling_rate = 0.995
