import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate(h, 2.0/n_points) for h in population]

        # Sort by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Check for best solution
        current_best = fitness[0]
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= 50:
            # Early termination if no improvement in 50 generations
            break
        
        # Create new population
        new_population = [population[i].copy() for i in range(num_elite)]
        
        # Add mutated individuals
        for _ in range(num_mutants):
            # Select parent
            parent_idx = np.random.randint(0, population_size)
            parent = population[parent_idx]
            
            # Mutate
            mut_strength = mutation_strength * np.exp(np.random.normal(0, beta))
