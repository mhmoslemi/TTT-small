import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    # Ensure h satisfies the constraint
    h = h_initial * target_sum / np.sum(h_initial)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use scipy minimize with SLSQP for better optimization
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Early termination counter
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    iteration = 0
    
    # Add a small random perturbation to escape initial local minima
    h = h + 1e-4 * np.random.normal(0, 1, n_points)

    for restart in range(5):

            break

        h_start = project_h(h_start)
        
        # Optimize using scipy minimize
        result = minimize(evaluate_objective, h_start, method='SLSQP', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            h_current = result.x
        else:
            h_current = h_start
            
        h_current = project_h(h_current)
        new_c5 = compute_c5(h_current)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_current.copy()

    while True:
        elapsed = time.time() - start_time

            h = result.x
        else:
            # If optimization failed, do a simple projection descent
            h = project_h(h)

        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Add small noise to escape local minima
            noise = 0.0005 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
        
        iteration += 1
        if iteration > max_iter:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
