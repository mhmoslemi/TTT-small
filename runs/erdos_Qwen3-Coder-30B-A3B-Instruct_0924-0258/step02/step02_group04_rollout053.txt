import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin Hypercube samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        h = np.random.random(n_points)
        # Ensure sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        samples.append(h)
    return samples

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise use Latin Hypercube"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use Latin Hypercube sampling for diversity
        lh_samples = latin_hypercube_sample(n_points, 10)
        h = lh_samples[0]  # Take first sample
    return project_to_feasible(h, n_points)

def perturb_h(h, noise_level=0.1):
    """Add controlled noise to the function"""
    noise = np.random.normal(0, noise_level, len(h))
    return project_to_feasible(h + noise, len(h))

def optimize_single(h_start, n_points, dx, time_limit):
    """Optimize a single candidate using L-BFGS-B"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Objective function for optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    # Optimization constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform several optimization steps
    for _ in range(20):
        if time() - start_time > time_limit:
            break
        try:
            res = minimize(obj, h_best, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 10})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
            
        # Perturb and restart locally
        h_perturbed = perturb_h(h_best, 0.05)
        try:
            res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 10})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass
            
    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Start with resolutions
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Use more samples at coarse resolution and fewer at fine
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple restarts
        best_in_res = float('inf')
        best_h_in_res = None
        
        # Use more restarts for coarser grids
        n_restarts = max(5, 20 - n_points // 3)
        
        for _ in range(n_restarts):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = optimize_single(h_start, n_points, dx, time_per_res / n_restarts)
            if c5_opt < best_in_res:
                best_in_res = c5_opt
                best_h_in_res = h_opt.copy()
            h_start = perturb_h(h_opt, 0.1)
            
        if best_in_res < best_c5:
            best_h = best_h_in_res.copy()
            best_c5 = best_in_res
            best_n_points = n_points
    
    if best_h is None:
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_single(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
