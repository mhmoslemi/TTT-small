import numpy as np
from scipy.optimize import linprog
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Initial guess is already feasible
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = compute_overlap(current_h)
    
    # Binary search over C5
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    # Helper to build LP constraints properly
    def build_constraints(C5_val):
        # Compute overlap constraints
        # The constraint: sum h[i] * (1 - h[i + k]) * dx <= C5_val for all k
        # For a fixed k, we need to compute all valid pairs
        
        # Precompute data structures for constraining overlap
        A_eq = np.zeros(n_points)
        A_eq[0] = 1.0  # sum h[i] = target_sum
        b_eq = target_sum
        
        # Build constraints for all k
        A_ineq = []
        b_ineq = []
        
        # For each possible k, compute the constraint
        for k in range(-n_points+1, n_points):
            # Compute coefficients for this constraint
            row = np.zeros(n_points)
            # For each i in [0, n_points-1], we consider h[i] * (1 - h[i+k])
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    # This is a valid pairing for k
                    row[i] = 1.0
            A_ineq.append(row)
            b_ineq.append(C5_val * dx)

        return np.array(A_eq).reshape(1, -1), np.array(b_eq), np.array(A_ineq), np.array(b_ineq)
    
    def solve_lp(C5_candidate):
        A_eq, b_eq, A_ineq, b_ineq = build_constraints(C5_candidate)
        
        # Lack of bounds in linprog constraints lowers h[i] to 0, so we add bounds directly
        bounds = [(0, 1) for _ in range(n_points)]
        
        # We can set up our LP as minimize 0 subject to A*x <= b, A_eq*x = b_eq, bounds
        # But since we are a feasibility check, we just need to check if there's a solution.
        # Using linprog with an arbitrary objective (0) should return a feasible solution if exists.
        try:
            result = linprog(
                c=np.zeros(n_points),
                A_ub=A_ineq,
                b_ub=b_ineq,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method='highs'
            )
            return result.success
        except:
            return False
    
    # Binary search loop
    while (high - low) > tolerance and time.time() - start_time < budget_s - 1:
        C5_candidate = (low + high) / 2.0
        if solve_lp(C5_candidate):
            high = C5_candidate
            # Check current best C5, update if needed
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        else:
            low = C5_candidate
            
        # Check time and break if need be
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final refinement with optimization if time allows
    best_c5_final = compute_overlap(best_h)
    
    # Fine-tune with local optimization if time remains
    if time.time() - start_time < budget_s - 5:
        # Objective: minimize max overlap
        def objective(h_vec):
            return compute_overlap(h_vec)
        
        def cons_func(h_vec):
            return np.sum(h_vec) - target_sum  # equality constraint
        
        # Use scipy's minimize with SLSQP for fine-tuning
        from scipy.optimize import minimize
        res = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': cons_func},
            options={'maxiter': 100, 'ftol': 1e-6},
            tol=1e-6
        )
        
        if res.success:
            fine_h = project_feasible(res.x)
            fine_c5 = compute_overlap(fine_h)
            if fine_c5 < best_c5_final:
                best_c5_final = fine_c5
                best_h = fine_h
        
    return (best_h.tolist(), best_c5_final, n_points)
