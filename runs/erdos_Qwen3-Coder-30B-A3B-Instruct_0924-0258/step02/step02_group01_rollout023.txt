import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    n_points = 35
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Initialize population
    pop_size = 50
    population = []
    population_fitness = []
    
    # Add the initial solution
    h_feasible = project_feasible(h_start, n_points)
    fitness = evaluate_erdos_solution(h_feasible, 2.0 / n_points)
    population.append(h_feasible)
    population_fitness.append(fitness)
    
    # Add random individuals
    for _ in range(pop_size - 1):
        h_random = np.random.rand(n_points)
        h_random = project_feasible(h_random, n_points)
        fitness = evaluate_erdos_solution(h_random, 2.0 / n_points)
        population.append(h_random)
        population_fitness.append(fitness)
    
    # Best solution tracking
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # Evolution parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Main evolution loop
    while time() - start_time < budget_s * 0.95:
        # Tournament selection
        selected_indices = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_k)
            tournament_fitness = [population_fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_indices.append(winner_idx)
        
        # Generate offspring
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = population[selected_indices[i]]
            parent2 = population[selected_indices[i + 1]] if i + 1 < pop_size else population[selected_indices[0]]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            offspring.extend([child1, child2])
        
        # Evaluate offspring
        offspring_fitness = []
        for child in offspring[:pop_size]:
            fitness = evaluate_erdos_solution(child, 2.0 / n_points)
            offspring_fitness.append(fitness)
        
        # Combine and select best
        all_population = population + offspring[:pop_size]
        all_fitness = population_fitness + offspring_fitness[:pop_size]
        
        # Sort indices by fitness
        sorted_indices = np.argsort(all_fitness)
        top_indices = sorted_indices[:pop_size]
        
        population = [all_population[i] for i in top_indices]
        population_fitness = [all_fitness[i] for i in top_indices]
        
        # Update best solution
        current_best_idx = np.argmin(population_fitness)
        if population_fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = population_fitness[current_best_idx]
    
    return (list(best_h), best_c5, n_points)
