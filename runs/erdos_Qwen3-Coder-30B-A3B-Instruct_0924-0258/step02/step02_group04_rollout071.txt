import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def optimize_fourier_spectrum(h_start, n_points, dx, time_limit):
    """Optimize using Fourier-based approach with penalty in frequency domain"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Fourier domain optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Make a copy of current solution
        h_current = h_best.copy()
        
        # Compute DFT of h
        H = np.fft.fft(h_current)
        
        # Create penalty based on power spectrum (encourage flat spectrum)
        power_spectrum = np.abs(H)**2
        penalty = np.sum(power_spectrum**2)  # Fourth power penalty to discourage peaks
        
        # Compute gradient w.r.t. H
        grad_H = 4 * np.conj(H) * power_spectrum
        # Update H by steepest descent
        step_size = 1e-4
        H_new = H - step_size * grad_H
        
        # Ensure real signal property (Hermitian symmetry for real signal)
        H_symmetric = (H_new + np.conj(H_new[::-1])) / 2
        
        # Inverse FFT to get new h
        h_new = np.real(np.fft.ifft(H_symmetric))
        
        # Apply constraints in time domain
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate objective
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept better solution
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            
        # occasional global perturbation
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = project_to_feasible(h_best + perturbation, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < c5_best:
                h_best = h_perturbed.copy()
                c5_best = c5_perturbed

    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Multiple restarts with different initializations
        h_restart = h_start.copy()
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_opt, c5_opt = optimize_fourier_spectrum(h_restart, n_points, dx, time_per_res / 3)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = n_points
            h_restart = np.random.rand(n_points) * 0.2 + 0.4  # Perturb

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = optimize_fourier_spectrum(h_start, n_points, dx, budget_s)
        best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
