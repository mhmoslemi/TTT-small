import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    velocity_max = 0.15
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros(n_points))
    
    # Fill the rest with random perturbations
    while len(swarm) < swarm_size:
        h_new = np.random.rand(n_points) * 0.5 + 0.25
        h_new = project_feasible(h_new)
        swarm.append(h_new)
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
    
    # Track best solutions
    particle_best = [swarm[i].copy() for i in range(swarm_size)]
    particle_best_scores = [compute_overlap(swarm[i]) for i in range(swarm_size)]
    swarm_best = particle_best[np.argmin(particle_best_scores)].copy()
    swarm_best_score = min(particle_best_scores)
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(swarm_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = (w * velocities[i] + 
                 c1 * r1 * (particle_best[i] - swarm[i]) + 
                 c2 * r2 * (swarm_best - swarm[i]))
            
            # Clip velocity
            v = np.clip(v, -velocity_max, velocity_max)
            velocities[i] = v
            
            # Update position
            swarm[i] = swarm[i] + v
            
            # Project to feasible space
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate and update personal best
            score = compute_overlap(swarm[i])
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best[i] = swarm[i].copy()
                
                if score < swarm_best_score:
                    swarm_best_score = score
                    swarm_best = swarm[i].copy()
                    last_improvement = 0
            else:
                last_improvement += 1
                
        iter_count += 1
        
        if last_improvement > 200:
            # Occasionally perturb particles to escape local minima
            for i in range(swarm_size):
                if np.random.rand() < 0.1:
                    swarm[i] = project_feasible(swarm[i] + np.random.normal(0, 0.01, n_points))
                    particle_best_scores[i] = compute_overlap(swarm[i])
                    particle_best[i] = swarm[i].copy()
                    if particle_best_scores[i] < swarm_best_score:
                        swarm_best_score = particle_best_scores[i]
                        swarm_best = swarm[i].copy()
            last_improvement = 0
    
    # Final validation
    final_C5 = compute_overlap(swarm_best)
    
    return (swarm_best.tolist(), final_C5, n_points)
