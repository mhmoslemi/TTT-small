import numpy as np
import cvxpy as cp
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search bounds
    C_low = 0.37
    C_high = 0.38085
    best_feasible_C = C_high
    best_feasible_h = best_h.copy()
    
    # Binary search for minimal C
    search_time = 60  # seconds allocated for binary search
    remaining_time = budget_s - (time() - start_time)
    
    # Use cvxpy for robust feasibility checks
    for iteration in range(40):
        if time() - start_time > budget_s * 0.9:
            break
            
        C_mid = (C_low + C_high) / 2
        
        # Define CVXPY optimization problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # Compute correlation for shift k
            corr_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(corr_sum <= C_mid * n_points / 2)
        
        # Problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h_vars.value
                if h_sol is not None:
                    h_feasible = project_to_feasible(h_sol, n_points)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_feasible_C:
                        best_feasible_C = c5_feasible
                        best_feasible_h = h_feasible.copy()
                    C_high = C_mid
                else:
                    C_low = C_mid
            else:
                C_low = C_mid
        except:
            C_low = C_mid
            
    # Refinement with gradient-based optimization
    if time() - start_time < budget_s * 0.9:
        def obj_func(h, C_target):
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr) * dx
            # Penalty if exceeds target
            penalty = max(0, max_corr - C_target) ** 2
            return max_corr + penalty
        
        def grad_func(h, C_target):
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            g = np.zeros_like(h)
            # Gradient of correlation at max position
            for i in range(n_points):
                j = (i - max_idx) % n_points
                g[i] += (1 - h[j]) * dx
                g[j] -= h[i] * dx
            # Gradient penalty
            max_corr = np.max(corr) * dx
            if max_corr > C_target:
                g += 2 * (max_corr - C_target) * g
            return g
        
        h_refined = best_feasible_h.copy()
        C_target = best_feasible_C
        
        # Try multiple restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.9:
                break
            h_restart = project_to_feasible(h_refined + np.random.normal(0, 0.01, n_points), n_points)
            
            # Minimize using scipy
            def fun(h):
                return obj_func(h, C_target)
                
            def jac(h):
                return grad_func(h, C_target)
            
            try:
                res = minimize(fun, h_restart, method='L-BFGS-B', bounds=[(0, 1)]*n_points, jac=jac, options={'maxiter': 100})
                if res.success:
                    h_result = project_to_feasible(res.x, n_points)
                    c5_result = evaluate_erdos_solution(h_result, dx)
                    if c5_result < best_c5:
                        best_h = h_result.copy()
                        best_c5 = c5_result
            except:
                pass
                
            h_refined = h_result.copy()
    
    # Final check
    if best_c5 > best_feasible_C:
        best_h = best_feasible_h
        best_c5 = best_feasible_C
    
    return (list(best_h), best_c5, n_points)
