import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover for two parents"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Apply Gaussian mutation to individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        mutated = individual + noise
        return mutated
    return individual

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Create initial population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Main genetic algorithm loop
    generation = 0
    stagnation_count = 0
    min_fitness = best_c5
    
    while time() - start_time < budget_s * 0.98:
        generation += 1
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top 25% (elitism)
        elite_size = population_size // 4
        new_population = sorted_population[:elite_size]
        new_fitnesses = sorted_fitnesses[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Project to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
            new_fitnesses.extend([evaluate_erdos_solution(offspring1, dx), 
                                evaluate_erdos_solution(offspring2, dx)])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        new_fitnesses = new_fitnesses[:population_size]
        
        # Survival: keep best individuals
        combined_population = new_population + sorted_population[:elite_size]
        combined_fitnesses = new_fitnesses + sorted_fitnesses[:elite_size]
        
        # Sort combined population
        combined_indices = np.argsort(combined_fitnesses)
        population = [combined_population[i] for i in combined_indices[:population_size]]
        fitnesses = [combined_fitnesses[i] for i in combined_indices[:population_size]]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early stopping if no improvement in 20 generations
        if stagnation_count > 20:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
