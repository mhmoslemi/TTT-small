import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, backtracking_steps=10):
    """Frank-Wolfe optimization step for minimizing the maximum overlap"""
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        # Find shift indices that give the maximum overlap
        max_indices = np.where(corr == np.max(corr))[0]
        # Use top 5% of indices or all if less than 20
        num_top = max(1, len(max_indices) // 20)
        selected_indices = max_indices[-num_top:] if len(max_indices) > num_top else max_indices

        # Compute subgradient based on these indices
        subgradient = np.zeros_like(h)
        for k in selected_indices:
            # For each shift k, the derivative is (1-h[i-k]) - h[i] 
            # mod indexing for circular convolution
            for i in range(n_points):
                j = (i - k) % n_points
                subgradient[i] += (1 - h[j]) - h[i]
        
        # Take average subgradient
        subgradient /= len(selected_indices)
        
        # Frank-Wolfe direction - solve linear program
        # min s: <subgradient, s> subject to sum(s) = 0, 0 <= s[i] <= 1
        # This means we want s to be such that s - h is a descent direction
        
        # Let's choose s that minimizes inner product with subgradient
        s = np.zeros_like(h)
        # Set s[j] = 1 if subgradient[j] < 0, else 0
        # But we need sum(s) = 0, so we adjust to project to feasible set
        # Set s to minimize <subgradient, s> subject to constraints
        # The direction s that minimizes <g, s> subject to sum(s) = 0, 0 <= s[i] <= 1
        # is to set s[i] = max(0, min(1, -g[i]/2)) + adjust so sum = 0
        # But simpler: set s[i] = 0 if subgradient[i] >= 0, else 1 if possible
        # Then adjust for the sum constraint
        
        # Made a simplification for linear optimization approach:
        # Just pick s that is a descent direction with small step size
        s = np.zeros_like(h)
        for i in range(n_points):
            # For robust Frank-Wolfe, choose simple descent direction
            s[i] = 1.0 if subgradient[i] < 0 else 0.0

        # Ensure sum(s) = 0 by adjusting
        s_sum = np.sum(s)
        # We want sum(s) = 0, we adjust s = s - s_sum/n_points
        # then make sure it doesn't go negative
        s = s - s_sum / n_points
        s = np.clip(s, 0, 1)
        s = project_to_feasible(s, n_points)
        
        # Line search
        best_gamma = 0.0
        best_val = best_c5
        for gamma in np.linspace(0.1, 0.9, backtracking_steps):
            h_trial = h + gamma * (s - h)
            h_trial = project_to_feasible(h_trial, n_points)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_val:
                best_val = c5_trial
                best_gamma = gamma
        
        if best_val < best_c5:
            h = h + best_gamma * (s - h)
            h = project_to_feasible(h, n_points)
            best_c5 = best_val
            
            if best_c5 < 0.3807 and iteration > 5:
                break
    
    return h, best_c5
