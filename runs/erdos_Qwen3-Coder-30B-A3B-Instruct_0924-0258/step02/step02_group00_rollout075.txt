import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_convex_feasibility(h0, n_points, dx, C):
    """Solve the convex feasibility problem for given C"""
    h = Variable(n_points)
    
    # Constraints: h in [0,1], sum(h) * dx = 1
    constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
    
    # Add overlap constraints
    for k in range(n_points):
        # Compute sum of h[i] * (1 - h[i + k]) for valid i
        if k == 0:
            overlap_sum = 0
            for i in range(n_points):
                overlap_sum += h[i] * (1 - h[i])
        else:
            overlap_sum = 0
            for i in range(n_points - k):
                overlap_sum += h[i] * (1 - h[i + k])
    
        # Include the dx factor in the constraint
        constraints.append(overlap_sum * dx <= C)
    
    # Create and solve problem
    objective = Minimize(0)  # dummy objective
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            return h.value
    except:
        pass
    return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # If available, start with the precomputed initial_h_values
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Binary search bounds
    C_low = 0.37
    C_high = 0.38085
    best_h = h_start.copy()
    best_C = evaluate_erdos_solution(h_start, dx)
    n_iter = 0
    
    # Binary search loop
    while n_iter < 20 and (time.time() - start_time) < budget_s * 0.9:
        C_mid = (C_low + C_high) / 2.0
        
        # Try solving the feasibility problem
        h_feasible = solve_convex_feasibility(best_h, n_points, dx, C_mid)
        
        if h_feasible is not None:
            # Feasible - try to improve by optimization
            h_opt = project_to_feasible(h_feasible, n_points)
            C_val = evaluate_erdos_solution(h_opt, dx)
            
            if C_val < best_C:
                best_h = h_opt.copy()
                best_C = C_val
                
            C_high = C_mid
        else:
            C_low = C_mid
        
        n_iter += 1
        
        # Break early if we're very close
        if abs(C_high - C_low) < 1e-6:
            break
    
    # Post-processing refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Perform gradient ascent to refine the solution
    h_refined = project_to_feasible(best_h.copy(), n_points)
    
    # Use a simple gradient-like method
    for i in range(50):
        if (time.time() - start_time) > budget_s * 0.95:
            break
            
        h_trial = h_refined.copy()
        step_size = 0.01
        for j in range(n_points):
            v_new = h_trial[j] + step_size
            h_test = h_trial.copy()
            h_test[j] = v_new
            h_test = project_to_feasible(h_test, n_points)
            val_new = evaluate_erdos_solution(h_test, dx)
            
            if val_new < objective(h_trial):
                h_refined = h_test.copy()
        
        # Small random perturbations
        h_perturb = project_to_feasible(h_refined + np.random.normal(0, 0.01, n_points), n_points)
        if evaluate_erdos_solution(h_perturb, dx) < objective(h_refined):
            h_refined = h_perturb
        
        # Quit if improvement is too small
        if abs(objective(h_refined) - objective(h_refined)) < 1e-6:
            break
            
    final_C = evaluate_erdos_solution(h_refined, dx)
    
    return (list(h_refined), final_C, n_points)
