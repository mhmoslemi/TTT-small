import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Use a LABS-inspired approach with equidistant 1s
        h = np.zeros(n_points)
        ones_positions = np.linspace(0, n_points-1, n_points//2, dtype=int)
        h[ones_positions] = 1.0
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use a multi-resolution approach with focus on lower resolution for initial search
    resolutions = [15, 30, 45, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for res_idx, n_points in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Allocate time for this resolution
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / (len(resolutions) - res_idx), 150)
        
        # Multiple restarts
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_current = h_start.copy()
            
            # Inner optimization loop
            for iteration in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Optimization step
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                try:
                    # L-BFGS-B optimization with bounds
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(obj, h_current, method='L-BFGS-B',
                                  bounds=bounds, options={'maxiter': 30})
                    if res.success:
                        h_opt = project_to_feasible(res.x, n_points)
                        c5_opt = evaluate_erdos_solution(h_opt, dx)
                        if c5_opt < best_c5:
                            best_h = h_opt.copy()
                            best_c5 = c5_opt
                            best_n_points = n_points
                        h_current = h_opt
                    else:
                        h_current = project_to_feasible(h_current, n_points)
                except:
                    h_current = project_to_feasible(h_current, n_points)
                
                # Occasionally perturb to escape local minima
                if iteration % 30 == 0:
                    noise = np.random.normal(0, 0.05, n_points)
                    h_perturbed = project_to_feasible(h_current + noise, n_points)
                    c5_pert = evaluate_erdos_solution(h_perturbed, dx)
                    if c5_pert < best_c5:
                        best_h = h_perturbed.copy()
                        best_c5 = c5_pert
                        best_n_points = n_points
                    h_current = h_perturbed
                    
        # Set new start for next resolution
        if best_h is not None and res_idx < len(resolutions) - 1:
            h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_n_points), best_h)
            h_start = project_to_feasible(h_start, n_points)
    
    # Final optimization at best resolution if not already done
    if best_h is not None:
        dx = 2.0 / best_n_points
        h_current = best_h.copy()
        for iteration in range(50):
            if time() - start_time > budget_s * 0.95:
                break
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            try:
                bounds = [(0, 1) for _ in range(best_n_points)]
                res = minimize(obj, h_current, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 20})
                if res.success:
                    h_opt = project_to_feasible(res.x, best_n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
            except:
                pass
            h_current = project_to_feasible(h_current, best_n_points)
    
    if best_h is None:
        # Fallback
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        h_opt, c5_opt = augmented_lagrangian_optimize(h_start, n_points, dx, budget_s)
        best_h = h_opt
        best_c5 = c5_opt
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)

def augmented_lagrangian_optimize(h_start, n_points, dx, time_limit):
    """Fallback optimization if main strategy fails"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    for iteration in range(200):
        if time() - start_time > time_limit:
            break
        # Perturb
        noise = np.random.normal(0, 0.05, n_points)
        h_perturbed = project_to_feasible(h_best + noise, n_points)
        
        # Simple local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_perturbed, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 20})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt
                    c5_best = c5_opt
        except:
            pass
            
    return h_best, c5_best
