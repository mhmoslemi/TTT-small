import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization
    population_size = 50
    population = []
    
    # Generate initial population
    population.append(best_h.copy())
    for _ in range(population_size - 1):
        h = project_to_feasible(np.random.normal(0.5, 0.1, n_points), n_points)
        population.append(h)
    
    # Evolve population
    convergence_patience = 20
    patience = 0
    iteration = 0
    max_iterations = 100
    
    while iteration < max_iterations and time() - start_time < budget_s * 0.9:
        improved = False
        for i in range(len(population)):
            h_current = population[i].copy()
            
            # Apply perturbation
            h_perturbed = h_current.copy()
            indices = np.random.choice(n_points, int(n_points * 0.2), replace=False)
            h_perturbed[indices] += np.random.normal(0, 0.1, len(indices))
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            # Local optimization with SQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        improved = True
                        patience = 0
                    if c5_opt < evaluate_erdos_solution(population[i], dx):
                        population[i] = h_opt
                else:
                    population[i] = h_perturbed
            except:
                population[i] = h_perturbed
                
        if not improved:
            patience += 1
            if patience > convergence_patience:
                break
        iteration += 1
    
    # Resolution refinement
    current_resolution = n_points
    while time() - start_time < budget_s * 0.95 and current_resolution < 1000:
        new_resolution = min(current_resolution * 2, 1000)
        x_orig = np.linspace(0, 2, current_resolution)
        x_new = np.linspace(0, 2, new_resolution)
        h_interp = np.interp(x_new, x_orig, best_h)
        best_h = project_to_feasible(h_interp, new_resolution)
        dx_refined = 2.0 / new_resolution
        
        # Refine at new resolution
        def obj(h):
            return evaluate_erdos_solution(h, dx_refined)
            
        bounds = [(0, 1) for _ in range(new_resolution)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - new_resolution / 2}]
        
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                best_h = project_to_feasible(res.x, new_resolution)
                best_c5 = evaluate_erdos_solution(best_h, dx_refined)
        except:
            pass
            
        current_resolution = new_resolution
    
    return (list(best_h), best_c5, len(best_h))
