import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        h_init = np.full(45, 1.0 / 45)
    
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(project_h(h_init))
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    # Genetic algorithm main loop
    for generation in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for h in population:
            c5 = compute_c5(h)
            fitness_scores.append(-c5)  # Negative for minimization
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]  # Descending order
        sorted_population = [population[i] for i in sorted_indices]
        
        # Check for better solution
        current_best = compute_c5(sorted_population[0])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = sorted_population[0].copy()
        
        # Select top 50% (tournament selection)
        selected_indices = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size // 2, tournament_size)
            winner = tournament[np.argmax([fitness_scores[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, len(selected_indices), 2):
            parent1 = sorted_population[selected_indices[i]]
            parent2 = sorted_population[selected_indices[min(i+1, len(selected_indices)-1)]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Mutate 1-2 indices
                    n_mutate = np.random.randint(1, 3)
                    indices = np.random.choice(n_points, n_mutate, replace=False)
                    for idx in indices:
                        child[idx] += np.random.normal(0, 0.02)
                
                # Ensure constraints
                offspring.append(project_h(child))
        
        # Replace bottom half with offspring
        for i in range(len(offspring)):
            population[population_size // 2 + i] = offspring[i]
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
