import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085
    max_iterations = 40
    best_feasible_h = None
    best_feasible_c = float('inf')
    
    # Use CVXPY solver for feasibility checks
    solver = cp.ECOS
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Create CVXPY variables and constraints
        h = cp.Variable(n_points)
        
        # Constraint: sum of h = n_points / 2
        sum_constraint = cp.sum(h) == n_points / 2
        
        # Constraint: each correlation <= c_mid * n_points / 2
        constraints = [h >= 0, h <= 1, sum_constraint]
        
        # Add correlation constraints
        n_corr = 2 * n_points - 1
        for k in range(-n_points + 1, n_points):
            # Circular indexing
            shifted = np.roll(h_start, k)
            if k >= 0:
                corr_expr = cp.sum(cp.multiply(h[:n_points-k], 1 - shifted[k:]))
            else:
                corr_expr = cp.sum(cp.multiply(h[-k:], 1 - shifted[:n_points+k]))
            constraints.append(corr_expr <= c_mid * n_points / 2)
        
        # Feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=solver, feastol=1e-6, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Extract solution and validate
                h_sol = h.value
                if h_sol is not None:
                    h_feasible = np.clip(h_sol, 0, 1)
                    c5_value = evaluate_erdos_solution(h_feasible, dx)
                    if c5_value < best_feasible_c:
                        best_feasible_c = c5_value
                        best_feasible_h = h_feasible.copy()
        except Exception:
            pass
        
        # Update binary search bounds
        if best_feasible_c < float('inf') and best_feasible_c <= c_mid:
            c_high = c_mid
        else:
            c_low = c_mid
    
    # Refinement of the best feasible solution found
    if best_feasible_h is not None:
        h_refined = best_feasible_h.copy()
        c5_refined = best_feasible_c
        step_size = 0.001
        
        # Simple gradient descent refinement
        last_improve_time = time()
        while time() - start_time < budget_s * 0.95:
            # Calculate correlations with current h_refined
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            max_corr_idx = np.argmax(corr)
            max_corr_val = corr[max_corr_idx]
            
            # Gradient direction for current maximum correlation
            # This is a simplified gradient approximation
            h_new = h_refined.copy()
            h_new[max_corr_idx % n_points] -= step_size * 0.5
            h_new[(max_corr_idx + 1) % n_points] += step_size * 0.5
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum > 0:
                h_new = h_new * (n_points / 2) / h_new_sum
                
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_refined:
                h_refined = h_new.copy()
                c5_refined = c5_new
                last_improve_time = time()
            elif time() - last_improve_time > 30:  # No improvement in last 30s
                break
                
            step_size *= 0.99  # Decreasing step size
            
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
