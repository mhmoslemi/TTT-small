import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    max_iter = 10000
    no_improvement_limit = 50
    no_improvement_count = 0
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find dominant lag for gradient computation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        dominant_lag = max_idx - (len(h) - 1)
        
        # Compute gradient for dominant lag
        gradient = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + dominant_lag) % len(h)
            gradient[i] = -(1 - h[j] - h[i]) * dx
        
        # Add penalty gradient (project to feasible region)
        penalty_grad = np.zeros_like(h)
        h_sum = np.sum(h)
        penalty_grad += 2 * lambda_penalty * (h_sum - n_points / 2) / n_points
        
        # Apply gradient update
        h_candidate = h + eta * (-gradient + penalty_grad)
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate the change
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - best_c5
        
        # Accept or reject based on simulated annealing criteria
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Temperature cooling
        T *= alpha
        
        # Random perturbation if no improvement
        if no_improvement_count > no_improvement_limit:
            h += np.random.normal(0, 0.05, len(h))
            h = project_feasible(h, n_points)
            no_improvement_count = 0
    
    # Multi-resolution refinement
    resolutions = [40, 45, 50, 55, 60]
    for n_points_res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points_res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n_points_res)
        
        # Local optimization at new resolution
        dx_res = 2.0 / n_points_res
        for _ in range(5):
            # Perturb
            h_perturbed = h_interp + np.random.normal(0, 0.02, len(h_interp))
            h_perturbed = project_feasible(h_perturbed, n_points_res)
            
            # Gradient update
            corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            max_idx = np.argmax(corr)
            dominant_lag = max_idx - (len(h_perturbed) - 1)
            
            gradient = np.zeros_like(h_perturbed)
            for i in range(len(h_perturbed)):
                j = (i + dominant_lag) % len(h_perturbed)
                gradient[i] = -(1 - h_perturbed[j] - h_perturbed[i]) * dx_res
                
            h_candidate = h_perturbed + eta * (-gradient)
            h_candidate = project_feasible(h_candidate, n_points_res)
            
            c5_candidate = evaluate_erdos_solution(h_candidate, dx_res)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                
        h_interp = best_h.copy()
    
    return (list(best_h), best_c5, len(best_h))
