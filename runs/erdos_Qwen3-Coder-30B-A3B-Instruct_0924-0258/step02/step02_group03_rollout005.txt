import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

    population_size = 30
    elite_size = population_size // 10
    mutate_fraction = 0.4
    beta = 0.2
    mutation_strength = 0.05
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate(h, 2.0/n_points) for h in population]
    
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    while time.time() - start_time < budget_s:
        # Update best
        min_val = min(fitness)
        if min_val < best_c5:
            best_c5 = min_val
            best_h = population[np.argmin(fitness)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 50:
                break
        
        # Sort population by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]

        elites = population[:elite_size]
        
        # Generate new population
        new_population = elites[:]
        
        # Mutate individuals
        for i in range(int(populate_size * mutate_fraction)):
            parent_idx = np.random.randint(0, population_size)
            parent = population[parent_idx]
            mutate_strength = mutation_strength * np.exp(beta * np.random.normal(0, 1))
            mutant = parent + np.random.normal(0, mutate_strength, n_points)
            mutant = project_to_feasible(mutant, n_points/2.0)
            new_population.append(mutant)
        
        # Fill remaining slots with random individuals
        while len(new_population) < population_size:
            h = h0 + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points/2.0)
            new_population.append(h)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate(h, 2.0/n_points) for h in population]
