import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Ensure we have enough points for resolution
    n_points = max(n_points, 30)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        # Compute all correlations (full mode)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Differential Evolution algorithm parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        # Start with a slightly perturbed initial solution
        perturbed = h + np.random.normal(0, 0.05, n_points)
        population.append(project_feasible(perturbed))
    
    # Track best solution
    best_fitness = float('inf')
    best_h = h.copy()
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = []
        for individual in population:
            fitness = compute_overlap(individual)
            fitnesses.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_h = individual.copy()
        
        # Create new population
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals for mutation
            indices = list(range(population_size))
            indices.remove(i)
            a, b, c = np.random.choice(indices, 3, replace=False)
            
            # Differential mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.random() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            
            # Ensure feasibility
            trial = project_feasible(trial)
            
            # Selection: keep trial if it's better
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Final validation
    final_fitness = compute_overlap(best_h)
    
    return (best_h.tolist(), final_fitness, n_points)
