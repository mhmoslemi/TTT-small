import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def solve_convex_subproblem(h_old):
        # Construct optimization variables
        h_new = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == target_sum
        ]
        
        # Objective: minimize maximum overlap
        # We'll build this via constraint relaxation
        # Find the maximum component in the overlap constraint
        max_overlap = cp.Variable()
        constraints.append(max_overlap >= 0)
        
        # Recompute current correlations
        conv_old = np.correlate(h_old, 1 - h_old, mode='full')
        mid = len(conv_old) // 2
        overlaps_old = conv_old[mid-n_points+1:mid+n_points-1] * dx
        
        # Now, for each shifted overlap term
        # sum h[i] * (1 - h[j]) * dx <= max_overlap
        # where i = j + k, for k from 0 to 2*n_points-2
        # - For k in range(n_points + 1, 2*n_points-1):
        #   sum_{i} h_old[i] * (1 - h_new[i-k]) * dx <= max_overlap
        #   sum_{i} h_new[i-k] * dx + sum_{i} h_old[i] * dx <= max_overlap + sum_{i} h_old[i] * dx
        # Therefore, due to symmetry in over/underlap constraints:
        # max_overlap with h_old fixed
        for k in range(n_points-1):
            if k < n_points-1:
                # For overlap k
                # sum_{i, j s.t. j=i+k} h[i] * (1 - h[j]) * dx  
                # Edges are tricky to compute; use direct sum with shifted indices
                lhs = 0.0
                for i in range(n_points):
                    j = i + k
                    if j < n_points:
                        lhs += h_old[i] * (1 - h_new[j])
                    else:
                        break
                constraints.append(lhs * dx <= max_overlap)
                
        # Add inequality constraints for corner shifts
        for k in range(n_points, 2*n_points-1):
            # This is the opposite tail of overlap so flip indices
            # sum_{i s.t. i+k < n_points} h_old[i] * (1 - h_new[i+k]) * dx
            lhs = 0.0
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    lhs += h_old[i] * (1 - h_new[j])
                else:
                    break
            constraints.append(lhs * dx <= max_overlap)
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=5000)
            if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                h_new_val = h_new.value
                return h_new_val
        except:
            pass
        
        # Fallback: return previous h
        return h_old.copy()
    
    # Optimization 
    current_h = h.copy()
    best_val = compute_overlap(current_h)
    best_h = current_h.copy()
    patience = 0
    last_improvement = 0
    
    max_iter = 50
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use convex relaxation at current point (CCCP style)
        h_updated = solve_convex_subproblem(current_h)
        h_projected = project_feasible(h_updated)
            
        current_val = compute_overlap(h_projected)
        
        if current_val < best_val:
            best_val = current_val
            best_h = h_projected.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Update current_h
        current_h = h_projected
        
        if last_improvement > 20:
            break
        
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
