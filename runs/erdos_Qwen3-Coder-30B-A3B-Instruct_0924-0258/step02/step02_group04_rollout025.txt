import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a mixed pattern
        h = np.ones(n_points) * 0.5
        # Add some variation to encourage exploration
        for i in range(0, n_points, 4):
            if i < n_points:
                h[i] = 0.3
        for i in range(1, n_points, 4):
            if i < n_points:
                h[i] = 0.7
    return project_to_feasible(h, n_points)

def lhc_sampling(n_points, n_samples):
    """Latin Hypercube Sampling for diverse initial conditions"""
    samples = []
    for _ in range(n_samples):
        h = np.zeros(n_points)
        # Generate stratified samples for each variable
        for i in range(n_points):
            h[i] = np.random.uniform(i / n_points, (i + 1) / n_points)
        # Add some randomness
        np.random.shuffle(h)
        samples.append(project_to_feasible(h, n_points))
    return samples

def improved_optimize(h_start, n_points, dx, time_limit):
    """Improved optimization using multiple restarts and L-BFGS-B"""
    start_time = time()
    h_best = h_start.copy()
    c5_best = evaluate_erdos_solution(h_start, dx)
    
    # Multiple restarts
    restarts = min(50, max(5, int(1000 * time_limit / 600.0)))
    lhc_samples = lhc_sampling(n_points, min(20, restarts))
    
    for i in range(restarts):
        if time() - start_time > time_limit:
            break
            
        # Choose starting point
        if i < len(lhc_samples):
            h_current = lhc_samples[i]
        else:
            # Random perturbation
            h_current = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            
        # Local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_current, method='L-BFGS-B', bounds=bounds, 
                          options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < c5_best:
                    h_best = h_opt.copy()
                    c5_best = c5_opt
        except:
            pass

    return h_best, c5_best

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with smaller resolutions for exploration
    resolutions = [15, 30, 45, 60, 75]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        # Use a fixed benchmark time for each resolution
        time_per_res = min(remaining_time / len(resolutions), 100)
        
        # Run optimization with multiple restarts
        h_opt, c5_opt = improved_optimize(h_start, n_points, dx, time_per_res)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points

    if best_h is None:
        # Fallback to simplest approach
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        best_h, best_c5 = improved_optimize(h_start, n_points, dx, budget_s)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
