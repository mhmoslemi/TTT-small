import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    # Final scaling to match sum target
    if abs(current_sum) > 1e-10:
        h *= target_sum / current_sum
    else:
        h.fill(target_sum / len(h))
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    swarm_size = 40
    max_iter = 10000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max_ratio = 0.05
    
    # Initialize particles
    if 'initial_h_values' in globals():

    else:

    # Create swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate(h, 2.0 / n_points))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            vel_update = (omega * velocities[i] +
                        c1 * r1 * (pbest[i] - particles[i]) +
                        c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            v_max = v_max_ratio * (np.max(particles[i]) - np.min(particles[i]))
            if v_max == 0:
                v_max = 0.05
            vel_update = np.clip(vel_update, -v_max, v_max)
            
            velocities[i] = vel_update
            
            # Update position
            new_h = particles[i] + velocities[i]
            new_h = project_to_feasible(new_h, n_points / 2.0)
            particles[i] = new_h
            
            # Evaluate new position
            new_val = evaluate(new_h, 2.0 / n_points)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = new_h.copy()
                pbest_vals[i] = new_val
                
                # Update global best
                if new_val < gbest_val:
                    gbest = new_h.copy()
                    gbest_val = new_val
                    
        iteration += 1
    
    return (gbest.tolist(), float(gbest_val), n_points)
