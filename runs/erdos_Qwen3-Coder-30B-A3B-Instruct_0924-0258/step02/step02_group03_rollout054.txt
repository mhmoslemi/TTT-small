import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    swarm_size = 40
    max_iter = 1000
    w = 0.7  # Inertia weight
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    v_max_factor = 0.05  # Max velocity as fraction of range
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Initialize particles around the initial solution
    for i in range(swarm_size):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.1, n_points)
        h = project_to_feasible(h, n_points/2.0)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Best positions and values
    pbest = [p.copy() for p in particles]
    pbest_values = [evaluate(p, 2.0/n_points) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        iteration += 1
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = c1 * r1 * (pbest[i] - particles[i])
            social = c2 * r2 * (gbest - particles[i])
            
            # Velocity update
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            h_range = np.max(particles[i]) - np.min(particles[i])
            if h_range < 1e-10:
                h_range = 1.0
            v_max = v_max_factor * h_range
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points/2.0)
            
            # Evaluate new position
            current_value = evaluate(particles[i], 2.0/n_points)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best if needed
                if current_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_value
                    
    # Return best solution so far
    return (gbest.tolist(), float(gbest_value), n_points)
