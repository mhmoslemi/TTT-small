import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    # Use evolutionary strategy with adaptive parameters
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Parameters
    population_size = 20
    max_generations = 10000
    mutation_strength = 0.05
    elite_ratio = 0.2
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_pert = h + np.random.normal(0, mutation_strength, n_points) * 0.1
        h_pert = project_h(h_pert)
        population.append(h_pert)
    
    # Keep track of best solution
    elite_size = int(population_size * elite_ratio)
    
    generation = 0
    no_improvement = 0
    patience = 50
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness of all individuals
        fitness = [compute_c5(ind) for ind in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            # Inject diversity with random individuals
            for i in range(elite_size, population_size):
                h_new = np.random.rand(n_points)
                h_new = project_h(h_new)
                population[i] = h_new
            no_improvement = 0
            
        # Create new population
        new_population = population[:elite_size]  # Elites
        
        # Generate offspring through mutation of elites
        for i in range(elite_size, population_size):
            parent_idx = np.random.randint(0, elite_size)
            parent = population[parent_idx]
            
            # Mutation
            child = parent + np.random.normal(0, mutation_strength, n_points)
            child = project_h(child)
            
            new_population.append(child)
            
        # Replace old population
        population = new_population
        
        generation += 1
        if generation > max_generations:
            break
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), final_c5, n_points)
