import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max"""
    exp_x = np.exp(alpha * x)
    sum_exp = np.sum(exp_x)
    return alpha * exp_x / sum_exp

def compute_corr_grad(h, n_points):
    """Compute gradient of correlation with respect to h"""
    # Correlation with 1-h, compute gradients
    # This is a simplified approach for gradient
    # For better accuracy, this would need_symbolic gradients
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Simplified gradient computation using finite differences
        eps = 1e-6
        h_plus = h.copy()
        h_plus[i] += eps
        corr_plus = np.correlate(h_plus, 1 - h, mode='full')
        corr_base = np.correlate(h, 1 - h, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr_base)) / eps
    return grad

def optimize_step(h_start, n_points, dx, time_limit):
    """Optimize using L-BFGS with custom objective"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def obj(h):
        h = project_to_feasible(h, n_points)
        c5_val = evaluate_erdos_solution(h, dx)
        return c5_val

    def grad(h):
        h = project_to_feasible(h, n_points)
        grad_val = compute_corr_grad(h, n_points)
        # Add gradient for constraint violation
        constraint_grad = np.ones_like(h) * (np.sum(h) - n_points / 2) / n_points
        return grad_val + 1e2 * constraint_grad
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try with different local optimizers
    initial_h = h_start.copy()
    
    for _ in range(20):  # Multiple restarts with different initializations
        if time() - start_time > time_limit:
            break
            
        # Local search
        try:
            res = minimize(obj, initial_h, method='L-BFGS-B',
                           bounds=bounds, jac=grad,
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_res = project_to_feasible(res.x, n_points)
                c5 = evaluate_erdos_solution(h_res, dx)
                if c5 < best_c5:
                    best_h = h_res.copy()
                    best_c5 = c5
        except:
            pass
        
        # Random perturbation
        initial_h = project_to_feasible(h_start + np.random.normal(0, 0.02, n_points), n_points)
        h_start = best_h.copy()
    
    return best_h, best_c5
