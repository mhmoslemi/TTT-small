import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, hess, trust_radius, target_sum):
    """Perform a trust region step."""
    n = len(h)
    # Simple projection method: take a small step in gradient direction
    delta_h = -0.01 * grad
    new_h = h + delta_h
    new_h = project_to_feasible(new_h, target_sum)
    
    return new_h

def compute_gradient(h, dx):
    """Compute approximate gradient using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute correlation of h with 1-h
    corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(corr)
    # Get the indices where correlation is maximized
    corr_indices = np.arange(len(corr))
    max_corr_idx = np.argmax(corr)
    
    # Compute gradient as difference of h values
    for i in range(n):
        grad[i] = 0
        for k in range(n):
            if i + k < n:
                grad[i] += (1 - h[i+k]) * dx
            if i - k >= 0:
                grad[i] += (1 - h[i-k]) * dx
    
    return grad
