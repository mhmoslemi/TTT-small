import numpy as np
import time

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h = h * scale
    return h

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    swarm_size = 40
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity_ratio = 0.05
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    particles = []
    velocities = []
    
    # Initialize swarm around h0 with small perturbations
    for _ in range(swarm_size):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points / 2.0)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Initialize personal bests and global best
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate(p, 2.0/n_points) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO iteration
    iteration = 0
    while time.time() - start_time < budget_s:
        if iteration % 50 == 0:
            # Occasionally reinitialize to avoid local minima
            for i in range(swarm_size):
                if np.random.rand() < 0.2:
                    h = h0 + np.random.normal(0, 0.05, n_points)
                    h = project_to_feasible(h, n_points / 2.0)
                    particles[i] = h.copy()
                    velocities[i] = np.random.uniform(-0.05, 0.05, n_points)
                    pbest[i] = h.copy()
                    pbest_scores[i] = evaluate(h, 2.0/n_points)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (pbest[i] - particles[i])
            social = c2 * r2 * (gbest - particles[i])
            velocities[i] = omega * velocities[i] + cognitive + social
            
            # Clip velocity
            max_vel = max_velocity_ratio * (np.max(particles[i]) - np.min(particles[i]))
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points / 2.0)
            
            # Evaluate and update personal best
            score = evaluate(particles[i], 2.0/n_points)
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
        
        iteration += 1
    
    return (gbest.tolist(), float(gbest_score), n_points)
