import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback to simple symmetric construction
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    velocity_max = 0.15
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add the initial solution
    if 'initial_h_values' in globals():
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros(n_points))
        
    # Fill the rest with random solutions around initial
    for _ in range(swarm_size - len(swarm)):
        perturbed = h + np.random.normal(0, 0.1, n_points)
        swarm.append(project_feasible(perturbed))
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Track best solutions
    personal_best = [compute_overlap(p) for p in swarm]
    personal_best_positions = [p.copy() for p in swarm]
    global_best_idx = np.argmin(personal_best)
    global_best = personal_best[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive inertia
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v = (w * velocities[i] + 
                 c1 * r1 * (personal_best_positions[i] - swarm[i]) +
                 c2 * r2 * (swarm[global_best_idx] - swarm[i]))
            
            # Clip velocity
            v = np.clip(v, -velocity_max, velocity_max)
            velocities[i] = v
            
            # Update position
            swarm[i] = swarm[i] + v
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate new position
            fitness = compute_overlap(swarm[i])
            
            # Update personal best
            if fitness < personal_best[i]:
                personal_best[i] = fitness
                personal_best_positions[i] = swarm[i].copy()
                
                # Update global best
                if fitness < global_best:
                    global_best = fitness
                    global_best_idx = i
    
    # Final best solution
    final_solution = swarm[global_best_idx].copy()
    final_C5 = compute_overlap(final_solution)
    
    return (final_solution.tolist(), final_C5, n_points)
