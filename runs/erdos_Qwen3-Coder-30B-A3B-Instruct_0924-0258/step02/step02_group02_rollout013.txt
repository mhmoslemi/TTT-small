import time

    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    best_c5 = compute_c5(h)
    best_h = h.copy()

    max_patience = 50
    lr = 0.01

    while True:

        if elapsed > budget_s:

        old_c5 = compute_c5(h)

            h_plus = h.copy()
            h_plus[i] += eps

        h = h - lr * grad
