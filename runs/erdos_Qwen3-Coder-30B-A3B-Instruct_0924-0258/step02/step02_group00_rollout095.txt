import numpy as np
from scipy import signal
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values)
        n_points = len(h_best)
    else:
        n_points = 35
        h_best = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    
    for n_points_new in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if n_points_new != n_points:
            # Interpolate to new resolution
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_new = np.interp(x_new, x_old, h_best)
            h_best = project_to_feasible(h_new, n_points_new)
            n_points = n_points_new
            dx = 2.0 / n_points

        # PSO parameters
        n_particles = 50
        max_iter = 200
        inertia_weight_start = 0.7
        inertia_weight_end = 0.4
        c1 = 2.0
        c2 = 2.0
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_scores = []
        
        # Warm start from current h_best
        for _ in range(n_particles):
            if time() - start_time > budget_s * 0.95:
                break
            # Small random perturbation
            h = project_to_feasible(h_best + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            score = evaluate_erdos_solution(h, dx)
            pbest.append(h.copy())
            pbest_scores.append(score)
            if score < c5_best:
                c5_best = score
                h_best = h.copy()
        
        gbest_index = np.argmin(pbest_scores)
        gbest = pbest[gbest_index].copy()
        gbest_score = pbest_scores[gbest_index]
        
        # PSO iterations
        for iter_idx in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * iter_idx / max_iter
            
            for i in range(n_particles):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (inertia_weight * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Update position
                particles[i] = particles[i] + velocities[i]
                
                # Project to feasible set
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                score = evaluate_erdos_solution(particles[i], dx)
                
                # Update personal best
                if score < pbest_scores[i]:
                    pbest[i] = particles[i].copy()
                    pbest_scores[i] = score
                    
                    if score < c5_best:
                        c5_best = score
                        h_best = particles[i].copy()
                        
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    if score < c5_best:
                        c5_best = score
                        h_best = particles[i].copy()
        
        # Refinement with gradient descent for best solution
        if time() - start_time < budget_s * 0.9:
            # Simple gradient descent refinement
            refine_h = h_best.copy()
            learning_rate = 0.001
            for _ in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                # compute approximate gradient using finite differences
                grad = np.zeros_like(refine_h)
                epsilon = 1e-5
                base_score = evaluate_erdos_solution(refine_h, dx)
                for j in range(n_points):
                    h_plus = refine_h.copy()
                    h_plus[j] += epsilon
                    h_plus = project_to_feasible(h_plus, n_points)
                    score_plus = evaluate_erdos_solution(h_plus, dx)
                    grad[j] = (score_plus - base_score) / epsilon
                
                refine_h = refine_h - learning_rate * grad
                refine_h = project_to_feasible(refine_h, n_points)
                new_score = evaluate_erdos_solution(refine_h, dx)
                if new_score < c5_best:
                    c5_best = new_score
                    h_best = refine_h.copy()
    
    return (list(h_best), c5_best, n_points)
