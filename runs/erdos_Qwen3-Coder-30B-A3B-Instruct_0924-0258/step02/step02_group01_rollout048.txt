import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate fitness of initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic Algorithm
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = int(budget_s * 10)
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection with tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_k)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Crossover
            if np.random.random() < crossover_rate:
                alpha = np.random.random()
                offspring1 = alpha * parent1 + (1 - alpha) * parent2
                offspring2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring1 += noise
                offspring1 = project_feasible(offspring1, n_points)
            
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                offspring2 += noise
                offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
        
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, 2.0 / n_points)
            fitness.append(fit)
            
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
    
    return (list(best_h), best_c5, n_points)
