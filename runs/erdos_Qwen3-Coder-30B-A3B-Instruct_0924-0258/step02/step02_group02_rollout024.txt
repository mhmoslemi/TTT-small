import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_template = np.array(initial_h_values, dtype=float)
    else:
        h_template = np.full(45, 1.0 / 45)
    
    n_points = len(h_template)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    # Genetic Algorithm Parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(project_h(h_template))
    for i in range(1, population_size):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    elapsed = 0
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(-c5)  # Negative for minimization
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmax([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=np.random.randint(1, 3), replace=False)
                for idx in mut_indices:
                    child1[idx] += np.random.normal(0, 0.01)
            if np.random.rand() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=np.random.randint(1, 3), replace=False)
                for idx in mut_indices:
                    child2[idx] += np.random.normal(0, 0.01)
            
            offspring.append(project_h(child1))
            offspring.append(project_h(child2))
        
        # Replace population
        # Keep top 50% from previous generation
        sorted_indices = np.argsort(fitness)[::-1][:population_size//2]
        new_population = [population[i].copy() for i in sorted_indices]
        # Add offspring
        new_population.extend(offspring[:population_size - len(new_population)])
        population = new_population
        
        # Update best
        for ind in population:
            c5 = compute_c5(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
