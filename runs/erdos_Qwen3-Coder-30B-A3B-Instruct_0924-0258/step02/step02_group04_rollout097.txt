import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize optimization parameters
    n_points = 15
    dx = 2.0 / n_points
    h_current = initialize_h(n_points)
    
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Hyperparameters for optimization
    lr = 1e-3
    max_iter = 1000
    patience = 500
    improvement_threshold = 1e-6
    
    # Define penalty function for Fourier domain optimization
    def fourier_penalty(H):
        # Penalize high power concentration
        return np.sum(np.abs(H)**4)
    
    # Metric to optimize
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Gradient computation
    def compute_gradient(h, dx):
        # This is a simplified gradient using finite differences for demonstration
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Optimizer with alternating descent steps
    def alternating_optimize(h_start, n_points, dx, time_limit):
        h_best = h_start.copy()
        c5_best = objective(h_best)
        
        start_opt = time()
        for iteration in range(max_iter):
            if time() - start_opt > time_limit:
                break
                
            # Stochastic gradient descent step
            grad = compute_gradient(h_best, dx)
            h_new = h_best - lr * grad
            h_new = project_to_feasible(h_new, n_points)
            c5_new = objective(h_new)
            
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
            
            # Occasionally use global perturbation
            if iteration % 100 == 0:
                h_perturb = project_to_feasible(h_best + np.random.normal(0, 0.01, n_points), n_points)
                c5_perturb = objective(h_perturb)
                if c5_perturb < c5_best:
                    h_best = h_perturb
                    c5_best = c5_perturb
                    
        return h_best, c5_best
    
    # Perform initial optimization
    h_opt, c5_opt = alternating_optimize(h_current, n_points, dx, budget_s * 0.4)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt
    
    # Try increasing resolution
    for new_points in [30, 45]:
        if time() - start_time > budget_s * 0.9:
            break
        dx_new = 2.0 / new_points
        h_new = np.interp(np.linspace(0, 2, new_points), np.linspace(0, 2, n_points), best_h)
        h_new = project_to_feasible(h_new, new_points)
        h_opt, c5_opt = alternating_optimize(h_new, new_points, dx_new, budget_s * 0.3)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            n_points = new_points
    
    return (list(best_h), best_c5, n_points)
