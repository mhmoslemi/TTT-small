import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), k)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_std=0.1):
    """Apply Gaussian mutation to individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, noise_std, len(individual))
        individual = individual + noise
    return individual

def genetic_algorithm_optimize(h_start, n_points, dx, time_limit):
    """Optimize using a genetic algorithm approach"""
    start_time = time()
    
    # GA parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    generations = 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Add initial solution if available
    if len(h_start) == n_points:
        population.append(h_start.copy())
        fitness.append(evaluate_erdos_solution(h_start, dx))
    else:
        population.append(np.ones(n_points) * 0.5)
        fitness.append(float('inf'))
    
    # Fill the rest of the population
    for _ in range(pop_size - 1):
        h = np.random.random(n_points)
        h = project_feasible(h, n_points)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    stagnation_count = 0
    for gen in range(generations):
        if time() - start_time > time_limit:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep top 25%
        sorted_indices = np.argsort(fitness)
        elite_count = pop_size // 4
        for i in range(elite_count):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Ensure feasibility
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        new_fitness = []
        for individual in new_population:
            new_fitness.append(evaluate_erdos_solution(individual, dx))
        
        # Update best solution
        min_fitness_idx = np.argmin(new_fitness)
        if new_fitness[min_fitness_idx] < best_c5:
            best_h = new_population[min_fitness_idx].copy()
            best_c5 = new_fitness[min_fitness_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early stopping if no improvement
        if stagnation_count > 20:
            break
            
        # Update population and fitness
        population = new_population
        fitness = new_fitness
    
    return best_h, best_c5

    resolutions = [15, 20, 25, 30, 35]
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    for i, n_points_res in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if len(h_start) != n_points_res:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points_res)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_start = project_feasible(h_interp, n_points_res)
        
        dx_res = 2.0 / n_points_res
        time_per_res = (budget_s - (time() - start_time)) / len(resolutions)
        
        # Run GA optimization at this resolution
        h_opt, c5_opt = genetic_algorithm_optimize(h_start, n_points_res, dx_res, time_per_res * 0.8)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            n_points = n_points_res
            
        h_start = h_opt.copy()
    
    return (list(best_h), best_c5, n_points)
