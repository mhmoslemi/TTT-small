import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C
    c_low = 0.37
    c_high = 0.38085
    c_tolerance = 1e-5
    max_iterations = 10
    iteration = 0
    
    while iteration < max_iterations and (c_high - c_low) > c_tolerance:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Formulate CVX problem
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1]
        constraints.append(sum(h) * dx == 1)
        
        # Add constraints for each shift
        for k in range(n_points):
            if k == 0:
                continue
            # Correlation for shift k
            corr_sum = 0
            for i in range(n_points - k):
                corr_sum += h[i] * (1 - h[i + k])
            constraints.append(corr_sum * dx <= c_mid)
            
        # Try to solve feasibility problem
        prob = Problem(Maximize(0), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_opt = h.value
                c5_check = evaluate_erdos_solution(h_opt, dx)
                if c5_check < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_check
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
            
        iteration += 1
    
    # Refinement using gradient descent
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        from scipy.optimize import minimize
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_func(h):
            return np.sum(h) * dx - 1
            
        cons = [{'type': 'eq', 'fun': constraint_func}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple restarts with different noise
        for i in range(5):
            if time.time() - start_time > budget_s * 0.95:
                break
            noise = np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(best_h + noise, 0, 1)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            try:
                res = minimize(obj, h_restart, method='SLSQP', bounds=bounds, 
                               constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_result = np.clip(res.x, 0, 1)
                    h_result = h_result * (n_points / 2) / np.sum(h_result)
                    c5_result = evaluate_erdos_solution(h_result, dx)
                    if c5_result < best_c5:
                        best_h = h_result
                        best_c5 = c5_result
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
