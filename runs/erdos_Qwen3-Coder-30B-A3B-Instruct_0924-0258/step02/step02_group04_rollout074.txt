import numpy as np
from time import time
from scipy.fft import fft, ifft

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with step function maximizing entropy in frequency domain
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def optimize_spectrum(h, n_points, dx, max_iter=1000, step_size=1e-4):
    # Convert to frequency domain
    H = fft(h)
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for it in range(max_iter):
        # Compute current autocorrelation
        C5_current = evaluate_erdos_solution(h, dx)
        if C5_current < best_c5:
            best_h = h.copy()
            best_c5 = C5_current

        # Focus on penalty in frequency domain
        # Encourage flat spectrum for minimal autocorrelation
        power_spectrum = np.abs(H)**2
        penalty = np.sum(power_spectrum**2)
        grad_H = 4 * np.conj(H) * power_spectrum

        # Update spectrum
        H_new = H - step_size * grad_H
        # Ensure real symmetry for real signal
        H_new[0] = np.real(H_new[0])
        if n_points % 2 == 0:
            H_new[n_points//2] = np.real(H_new[n_points//2])
        # Mirror for real signal
        for i in range(1, n_points//2):
            H_new[-i] = np.conj(H_new[i])
        H = H_new

        # Inverse FFT to get h
        h = np.real(ifft(H))
        # Fix area constraint and bounds
        h = project_to_feasible(h, n_points)

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 15
    dx = 2.0 / n_points
    h_start = initialize_h(n_points)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    curr_time = time()
    while curr_time - start_time < budget_s * 0.95:
        h_opt, c5_opt = optimize_spectrum(h_start, n_points, dx, max_iter=100)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            last_improvement = iter_count
        
        # Try different starting points
        h_start = np.random.uniform(0, 1, n_points)
        h_start = project_to_feasible(h_start, n_points)
        iter_count += 1
        curr_time = time()
        
        if iter_count - last_improvement > 500:
            break

    # Refine at higher resolution if time permits
    if curr_time - start_time < budget_s * 0.8 and n_points < 75:
        for points in [30, 45, 60, 75]:
            if curr_time - start_time > budget_s * 0.95:
                break
            dx_new = 2.0 / points
            h_start_new = np.interp(np.linspace(0, 2, points), np.linspace(0, 2, n_points), best_h)
            h_start_new = project_to_feasible(h_start_new, points)
            h_opt_new, c5_opt_new = optimize_spectrum(h_start_new, points, dx_new, max_iter=200)
            if c5_opt_new < best_c5:
                best_h = h_opt_new.copy()
                best_c5 = c5_opt_new
                n_points = points
            curr_time = time()
            
    return (list(best_h), best_c5, n_points)
