import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO parameters
    swarm_size = 40
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.15
    
    # Initialize swarm
    swarm = []
    for i in range(swarm_size):
        if i == 0:
            # Use initial solution as first particle
            particle = h.copy()
        else:
            # Initialize particles around initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = project_feasible(particle)
        swarm.append(particle)
    
    # Initialize personal bests and global best
    personal_bests = [p.copy() for p in swarm]
    personal_best_values = [compute_overlap(p) for p in swarm]
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_bests[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iteration
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(swarm_size):
            particle = swarm[i]
            pbest = personal_bests[i]
            
            # Generate random numbers for velocity update
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            v = w * (particle - pbest) + c1 * r1 * (global_best - particle) + c2 * r2 * (pbest - particle)
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update particle position
            particle = particle + v
            # Project to feasible region
            particle = project_feasible(particle)
            
            # Evaluate new solution
            new_value = compute_overlap(particle)
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_bests[i] = particle.copy()
                personal_best_values[i] = new_value
                
                # Update global best if needed
                if new_value < global_best_value:
                    global_best = particle.copy()
                    global_best_value = new_value
                    
            # Update swarm
            swarm[i] = particle
        
    # Final evaluation
    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
