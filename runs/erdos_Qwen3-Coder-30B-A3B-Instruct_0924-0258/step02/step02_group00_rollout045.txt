import numpy as np
from scipy.fft import fft, ifft
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    # Compute correlation using FFT for efficiency
    H = fft(h, 2 * len(h) - 1)
    H_conj = np.conj(H)
    corr = ifft(H * H_conj).real[:len(h)]
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Multi-resolution PSO parameters
    n_particles = 50
    max_iter = 200
    omega_start = 0.7
    omega_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    dx = 2.0 / n_points
    
    # Resolution levels
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed
        if n_points_try != n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            dx = 2.0 / n_points
            
        # Initialize particles
        particles = []
        velocities = []
        pbest_h = []
        pbest_c5 = []
        
        for _ in range(n_particles):
            h_p = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            c5_p = evaluate_erdos_solution(h_p, dx)
            particles.append(h_p.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            pbest_h.append(h_p.copy())
            pbest_c5.append(c5_p)
        
        # Global best
        gbest_idx = np.argmin(pbest_c5)
        gbest_h = pbest_h[gbest_idx].copy()
        gbest_c5 = pbest_c5[gbest_idx]
        
        # PSO iterations
        for iter_count in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            omega = omega_start - (omega_start - omega_end) * (iter_count / max_iter)
            
            for i in range(n_particles):
                h_i = particles[i]
                v_i = velocities[i]
                
                # Update velocity
                r1, r2 = np.random.random(2)
                v_i = (omega * v_i + 
                       c1 * r1 * (pbest_h[i] - h_i) + 
                       c2 * r2 * (gbest_h - h_i))
                
                # Update position
                h_i_new = h_i + v_i
                h_i_new = project_to_feasible(h_i_new, n_points)
                
                # Evaluate
                c5_new = evaluate_erdos_solution(h_i_new, dx)
                
                # Update personal best
                if c5_new < pbest_c5[i]:
                    pbest_h[i] = h_i_new.copy()
                    pbest_c5[i] = c5_new
                    
                    # Update global best
                    if c5_new < gbest_c5:
                        gbest_h = h_i_new.copy()
                        gbest_c5 = c5_new
                
                particles[i] = h_i_new.copy()
                velocities[i] = v_i.copy()
        
        # Update global best
        if gbest_c5 < best_c5:
            best_h = gbest_h.copy()
            best_c5 = gbest_c5
            
    # Final refinement with gradient descent
    h_refine = best_h.copy()
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient approximation using finite differences
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        for i in range(len(h_refine)):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps
            
        # Update step
        h_refine = h_refine - 0.001 * grad
        h_refine = project_to_feasible(h_refine, n_points)
        best_c5 = evaluate_erdos_solution(h_refine, dx)
        
        if time() - start_time > budget_s * 0.95:
            break
            
    return (list(h_refine), best_c5, n_points)
