import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def perturb_solution(h, n_points):
    """Apply perturbation to the function"""
    h_pert = h.copy()
    indices = np.random.choice(len(h), size=int(0.2 * len(h)), replace=False)
    h_pert[indices] += np.random.normal(0, 0.1, len(indices))
    return project_to_feasible(h_pert, n_points)

def optimize_candidate(h, n_points, dx):
    """Optimize a single candidate using SLSQP"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            return project_to_feasible(res.x, n_points)
    except:
        pass
    return h

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial population
    population = [h_start.copy()]
    for _ in range(49):
        h_new = perturb_solution(h_start, n_points)
        population.append(h_new)
    
    # Evaluate initial population
    for i, h in enumerate(population):
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Evolutionary optimization
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 20
    max_iterations = 100
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iterations and no_improvement_count < max_no_improvement:
        iter_count += 1
        improved = False
        
        # Mutate and optimize each individual
        new_population = []
        for h in population:
            h_mut = perturb_solution(h, n_points)
            h_opt = optimize_candidate(h_mut, n_points, dx)
            new_population.append(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                improved = True
                
        population = new_population
        if not improved:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
    
    # Resolution refinement
    resolutions = [50, 75, 100, 150, 200, 300, 500, 1000]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, res)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_projected = project_to_feasible(h_interp, res)
        
        # Local optimization at finer resolution
        h_fine = optimize_candidate(h_projected, res, 2.0/res)
        c5_fine = evaluate_erdos_solution(h_fine, 2.0/res)
        
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()
            n_points = res
            
    return (list(best_h), best_c5, n_points)
