import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Apply Gaussian mutation to an individual"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        mutated = individual + noise
        return mutated
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        population.append(h_start.copy())
        for _ in range(pop_size - 1):
            h = np.random.random(n_points) * 0.5
            h = project_feasible(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.random(n_points) * 0.5
            h = project_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Genetic algorithm
    generation = 0
    stagnation_counter = 0
    prev_best = best_c5
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        new_population = []
        new_fitnesses = []
        
        # Keep top performers
        top_indices = np.argsort(fitnesses)[:pop_size // 4]
        for idx in top_indices:
            new_population.append(population[idx].copy())
            new_fitnesses.append(fitnesses[idx])
        
        # Generate new offspring
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Feasibility projection
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
            new_fitnesses.extend([evaluate_erdos_solution(child1, dx),
                                  evaluate_erdos_solution(child2, dx)])
        
        # Trim to population size
        idxs = np.argsort(new_fitnesses)[:pop_size]
        population = [new_population[i] for i in idxs]
        fitnesses = [new_fitnesses[i] for i in idxs]
        
        # Update best
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_c5 = current_best
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Early termination if no improvement
        if stagnation_counter > 20:
            break
    
    return (list(best_h), best_c5, n_points)
