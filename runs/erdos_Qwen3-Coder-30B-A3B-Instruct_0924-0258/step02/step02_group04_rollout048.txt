import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use Fourier domain optimization approach
    n_points = 15
    dx = 2.0 / n_points
    h = initialize_h(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    last_improvement = 0
    max_iterations = 2000
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        iteration += 1
        
        # Create a new candidate by perturbing current solution
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        
        # Evaluate
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Occasionally do a more global search
        if iteration % 100 == 0 or no_improvement_count > 50:
            h_global = np.random.rand(n_points) * 0.5 + 0.25
            h_global = project_to_feasible(h_global, n_points)
            c5_global = evaluate_erdos_solution(h_global, dx)
            if c5_global < best_c5:
                best_h = h_global.copy()
                best_c5 = c5_global
                last_improvement = iteration
                no_improvement_count = 0
                
        h = h_perturbed
    
    # Refine the best solution with local optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Use SLSQP to refine
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            best_h = project_to_feasible(res.x, n_points)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
