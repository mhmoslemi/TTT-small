def run(seed=42, budget_s=600):

        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def upsample_h(h_vec, target_len):
        # Linear interpolation for upsample
        xs = np.linspace(0, 1, len(h_vec))
        xs_new = np.linspace(0, 1, target_len)
        return np.interp(xs_new, xs, h_vec)
    
    # Start with lower resolution
    n_res = 20
    h_vec = np.ones(n_res) * 0.5
    dx = 2.0 / n_res

        grad = np.zeros_like(h_vec)
        base_val = compute_overlap(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_feasible(h_plus)
            val_plus = compute_overlap(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_feasible(h_minus)
            val_minus = compute_overlap(h_minus)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
        return grad

    adam_lr = 0.01
    adam_beta1 = 0.9
    adam_beta2 = 0.999
    m = np.zeros_like(h_vec)
    v = np.zeros_like(h_vec)

    best_h = h_vec.copy()

        gradient = compute_gradient(h_vec)

        h_vec = h_vec - adam_lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            h_vec = project_feasible(h_vec)
            
            current_val = compute_overlap(h_vec)
            if current_val < best_val:
                best_val = current_val
                best_h = h_vec.copy()
    
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
