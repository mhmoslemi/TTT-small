import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_gradient(x, alpha=100):
    """Gradient of smooth max function"""
    exp_x = np.exp(alpha * x)
    return alpha * exp_x / np.sum(exp_x)

def objective_with_smooth_max(h, n_points, dx):
    """Objective using smooth max to approximate the maximum"""
    corr = np.correlate(h, 1 - h, mode='full')
    smooth_max_val = smooth_max(corr, alpha=100)
    return smooth_max_val

def compute_gradient(h, n_points, dx):
    """Compute gradient of smooth max with respect to h"""
    corr = np.correlate(h, 1 - h, mode='full')
    grad_corr = smooth_max_gradient(corr, alpha=100)
    
    # Build the full gradient matrix
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Contribution to gradient from correlation terms
        for k in range(2 * n_points - 1):
            # For each correlation term in the full convolution
            j = (i - k) % n_points
            if 0 <= j < n_points:
                grad[i] += grad_corr[k] * (1 - h[j]) * dx
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    """Optimize using gradient descent with projections"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Learning rate and decay
    learning_rate = 0.01
    decay_rate = 0.9
    iteration = 0
    max_iter = 1000
    patience = 0
    patience_limit = 50
    
    current_h = h_start.copy()
    
    for iter in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Compute gradient
        grad = compute_gradient(current_h, n_points, dx)
        
        # Update step
        new_h = current_h - learning_rate * grad
        
        # Project to feasible set
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(new_h, dx)
        
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            patience = 0
        else:
            patience += 1
        
        current_h = new_h.copy()
        iteration += 1
        
        if patience > patience_limit:
            learning_rate *= decay_rate
            patience = 0
            
    return best_h, best_c5

    np.random.seed(seed)

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Integer resolutions to try
    resolutions = [25, 30, 35]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res == n_points:
            current_h = h_start
        else:
            # Interpolate from existing
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, res)
            current_h = np.interp(x_new, x_old, h_start)
            current_h = project_to_feasible(current_h, res)
            n_points = res
        
        # Several restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb initial
            h_perturbed = project_to_feasible(current_h + np.random.normal(0, 0.05, res), res)
            
            # Optimize
            remaining_time = budget_s - (time() - start_time)
            opt_time = min(remaining_time / 5, 100)
            
            h_opt, c5_opt = optimize_with_gradient_descent(h_perturbed, res, 2.0/res, opt_time)
            
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                best_n_points = res
                
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
