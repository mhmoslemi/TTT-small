import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_C5 = compute_C5(h, n_points, dx)
    
    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def compute_C5(h_vec, n, dx):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n+1:mid+n-1] * dx
        return np.max(overlaps)
    
    # Convex-concave procedure (CCCP)
    max_cccp_iter = 50
    cc_iter = 0
    
    while cc_iter < max_cccp_iter and (time.time() - start_time) < budget_s - 1:
        if (time.time() - start_time) > budget_s - 1:
            break
            
        # Create variables for convex optimization
        h_vars = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints for convex relaxation
        constraints = [h_vars >= 0, h_vars <= 1]
        constraints.append(cp.sum(h_vars) == target_sum)
        
        # Build quadratic constraints for each shift
        # This is a convex relaxation - we treat constants as fixed
        # For each shift k, apply lower bound of the bilinear term
        
        # Get current values to build fixed part
        current_h = best_h
        
        # Build objective
        # We want to minimize the maximum overlap
        # Approximate by linearizing at current point
        obj = cp.Minimize(C5_var)
        prob = cp.Problem(obj, constraints)
        
        # Manually perform one step of CCCP
        # Fix current h values to linearize
        # In practice, solve a simpler QP for updating h
        
        # Create a working subproblem using scipy
        def obj_to_minimize(h_vec):
            h_vec = np.array(h_vec)
            return compute_C5(h_vec, n_points, dx)
            
        # Optimize using L-BFGS-B (simpler than full CVXPY)
        res = optimize.minimize(
            obj_to_minimize, 
            best_h.copy(), 
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        if res.success:
            updated_h = project_feasible(res.x)
            new_C5 = compute_C5(updated_h, n_points, dx)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = updated_h.copy()
        else:
            # If optimization fails, use previous best
            break
            
        cc_iter += 1
        
    # Final retuning steps
    # Try a few more rounds with a different initial.seed
    for trial in range(3):
        if (time.time() - start_time) > budget_s - 1:
            break
        h_trial = np.random.rand(n_points) * 0.5 + 0.25
        h_trial = project_feasible(h_trial)
        
        res = optimize.minimize(
            compute_C5,
            h_trial,
            args=(n_points, dx),
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        if res.success:
            updated_h = project_feasible(res.x)
            new_C5 = compute_C5(updated_h, n_points, dx)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = updated_h.copy()
    
    # Final validation 
    final_C5 = compute_C5(best_h, n_points, dx)
    
    return (best_h.tolist(), final_C5, n_points)
