import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    best_c5 = float('inf')
    best_h = None
    n_points = 35
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Systematic search with local optimizations
    max_iter = 300
    iteration = 0
    
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Create multiple random perturbations
        candidates = []
        for _ in range(20):
            # Random perturbation
            h_pert = h_start.copy()
            # Randomly select indices to perturb
            num_pert = max(1, int(0.1 * len(h_pert)))
            indices = np.random.choice(len(h_pert), num_pert, replace=False)
            h_pert[indices] += np.random.normal(0, 0.05, num_pert)
            
            # Project to feasible region
            h_pert = project_to_feasible(h_pert, n_points)
            candidates.append(h_pert)
        
        # Local optimization for each candidate
        for h_candidate in candidates:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        # Reset iteration counter for more exploration after improvement
                        iteration = 0
            except:
                pass
        
        iteration += 1
        # Gradually decrease perturbation strength
        if iteration % 10 == 0:
            # Adjust the h_start for next iteration
            pass # Keep current best solution
    
    # Refinement at higher resolution
    if (time() - start_time) < budget_s * 0.9:
        n_points_high = min(1000, n_points * 2)
        if n_points_high > n_points:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_high)
            h_interp = np.interp(x_new, x_orig, best_h)
            h_start_high = project_to_feasible(h_interp, n_points_high)
            
            dx_high = 2.0 / n_points_high
            best_c5 = evaluate_erdos_solution(h_start_high, dx_high)
            best_h = h_start_high
            n_points = n_points_high
            
            # Single optimization step at high resolution
            def obj(h):
                return evaluate_erdos_solution(h, dx_high)
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    best_h = project_to_feasible(res.x, n_points)
                    best_c5 = evaluate_erdos_solution(best_h, dx_high)
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
