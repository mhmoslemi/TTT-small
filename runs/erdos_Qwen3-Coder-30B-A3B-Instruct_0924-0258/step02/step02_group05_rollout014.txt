import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 50
        h = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)

    def project_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # PSO Parameters
    swarm_size = 40
    max_iter = 1000
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    velocity_max = 0.15
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        swarm.append(h.copy())
        velocities.append(np.zeros_like(h))
    else:
        swarm.append(np.ones(n_points) * 0.5)
        velocities.append(np.zeros_like(h))
    
    # Add random particles
    for i in range(swarm_size - len(swarm)):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = project_feasible(h_rand)
        swarm.append(h_rand)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, n_points))
    
    # Track best solutions
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_vals = [compute_overlap(swarm[i]) for i in range(swarm_size)]
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (personal_best[i] - swarm[i])
            social = c2 * r2 * (global_best - swarm[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_feasible(swarm[i])
            
            # Evaluate and update personal best
            current_val = compute_overlap(swarm[i])
            if current_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < global_best_val:
                    global_best = swarm[i].copy()
                    global_best_val = current_val

    final_C5 = compute_overlap(global_best)
    
    return (global_best.tolist(), final_C5, n_points)
