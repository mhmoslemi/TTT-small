import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sampling(n_points, n_samples=50):
    """Generate Latin Hypercube samples for diverse initializations"""
    samples = []
    for _ in range(n_samples):
        sample = np.zeros(n_points)
        # Generate a random permutation for each dimension
        for i in range(n_points):
            sample[i] = (i + random.random()) / n_points
        # Apply random scaling and shifting
        sample = np.random.permutation(sample) * 0.8 + 0.1
        samples.append(sample)
    return samples

def optimize_single(h_start, n_points, dx, max_iter=50):
    """Optimize a single starting point using L-BFGS-B"""
    def obj(h):
        return evaluate_erdos_solution(h, dx)

    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8})
        if res.success:
            h_opt = project_to_feasible(res.x, n_points)
            return h_opt, evaluate_erdos_solution(h_opt, dx)
    except:
        pass
    return h_start, evaluate_erdos_solution(h_start, dx)

    for n_points in coarse_resolutions:

        lhs_samples = latin_hypercube_sampling(n_points, 20)

            h_test = project_to_feasible(sample, n_points)

                best_h = h_opt.copy()
