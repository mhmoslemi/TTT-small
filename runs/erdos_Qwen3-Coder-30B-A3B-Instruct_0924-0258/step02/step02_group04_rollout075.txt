import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sampling(n_points, n_samples=50):
    """Generate Latin Hypercube samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        sample = np.zeros(n_points)
        # Generate uniform random points within each interval
        for i in range(n_points):
            sample[i] = (i + random.random()) / n_points
        # Shuffle
        random.shuffle(sample)
        samples.append(sample)
    return samples

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial resolution and adaptation
    resolutions = [15, 30, 45, 60, 75]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Use Latin hypercube sampling for diverse initializations
        lh_samples = latin_hypercube_sampling(n_points, 20)
        best_in_resolution = float('inf')
        h_best_in_resolution = None
        
        for i, h_sample in enumerate(lh_samples):
            # Scale to match constraint
            h_start = project_to_feasible(h_sample, n_points)
            
            # Optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            
            try:
                res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 30})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_in_resolution:
                        best_in_resolution = c5_opt
                        h_best_in_resolution = h_opt.copy()
            except Exception:
                continue
                
            if time() - start_time > budget_s * 0.95:
                break
                
        if best_in_resolution < best_c5:
            best_c5 = best_in_resolution
            best_h = h_best_in_resolution.copy()
            best_n_points = n_points
            
        # Early termination based on time spent
        if time() - start_time > budget_s * 0.95:
            break
    
    if best_h is None:
        # Fallback solution
        n_points = 30
        dx = 2.0 / n_points
        h_sample = np.random.rand(n_points)
        h_start = project_to_feasible(h_sample, n_points)
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 50})
            if res.success:
                best_h = project_to_feasible(res.x, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
                best_n_points = n_points
        except Exception:
            pass
            
    if best_h is None:
        # Another fallback to simple initialization
        n_points = 30
        dx = 2.0 / n_points
        h_start = np.ones(n_points) * 0.5
        h_start = project_to_feasible(h_start, n_points)
        best_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
