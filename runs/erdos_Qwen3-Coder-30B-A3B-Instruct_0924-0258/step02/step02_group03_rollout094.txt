import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points/2.0)
    else:
        h0 = np.ones(n_points) * 0.5

    # Parameters for MSPF
    n_particles = 100
    temperature = 1.0
    proposal_std = 0.1
    acceptance_rate_target = 0.3
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Initialize particles
    particles = [h0.copy()]
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        particles.append(project_to_feasible(h, n_points/2.0))

    best_h = h0.copy()
    best_c5 = evaluate(best_h, 2.0/n_points)
    
    iteration = 0
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Importance sampling
        weights = []
        new_particles = []
        acceptance_count = 0
        
        for h in particles:
            # Sample new particle
            h_prime = h + np.random.normal(0, proposal_std, n_points)
            h_prime = project_to_feasible(h_prime, n_points/2.0)
            c5_val = evaluate(h_prime, 2.0/n_points)
            
            # Importance weight
            weight = np.exp(-c5_val / temperature)
            weights.append(weight)
            new_particles.append(h_prime)
            
            # Accept if acceptable
            if np.random.random() < min(1.0, weight / (np.exp(-evaluate(h, 2.0/n_points) / temperature) + 1e-10)):
                acceptance_count += 1
        
        # Normalize weights
        total_weight = sum(weights)
        if total_weight > 0:
            weights = [w / total_weight for w in weights]
        else:
            weights = [1.0 / len(weights)] * len(weights)
        
        # Resampling
        selected_particles = []
        num_selected = max(1, len(particles) // 2)
        idxs = np.random.choice(len(particles), size=num_selected, p=weights)
        selected_particles = [new_particles[i] for i in idxs]
        
        # Mutation (standard idea: one global and one local)
        mutated_particles = []
        for h in selected_particles:
            # Global mutation
            h1 = h + np.random.normal(0, proposal_std, n_points)
            h1 = project_to_feasible(h1, n_points/2.0)
            mutated_particles.append(h1)
            
            # Local mutation
            h2 = h + np.random.normal(0, 0.05, n_points)
            h2 = project_to_feasible(h2, n_points/2.0)
            mutated_particles.append(h2)
        
        # Replace bottom 40% with mutations
        num_replacements = max(1, len(particles) // 2)
        replacement_indices = np.random.choice(len(particles), size=num_replacements, replace=False)
        for i, idx in enumerate(replacement_indices):
            if i < len(mutated_particles):
                particles[idx] = mutated_particles[i]
        
        # Keep top 10%
        particle_scores = [evaluate(p, 2.0/n_points) for p in particles]
        top_indices = np.argsort(particle_scores)[:len(particles)//10]
        best_particles = [particles[i] for i in top_indices]
        
        # New particles set
        particles = best_particles + mutated_particles
        
        # Update best
        current_best = min(particle_scores)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = particles[np.argmin(particle_scores)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adapt temperature and proposal std
        temperature *= 0.99
        if acceptance_count > 0:
            actual_acceptance = acceptance_count / n_particles
            if actual_acceptance < acceptance_rate_target:
                proposal_std *= 0.95
            elif actual_acceptance > acceptance_rate_target:
                proposal_std *= 1.05
        
        # Ensure proposal_std is not too small or large
        proposal_std = np.clip(proposal_std, 0.01, 0.5)
        
    # Final evaluation of best solution
    final_c5 = evaluate(best_h, 2.0/n_points)
    return (best_h.tolist(), float(final_c5), n_points)
