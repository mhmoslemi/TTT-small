import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction if exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    h = project_feasible(h, n_points)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.02
    lambda_penalty = 1e-3
    
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    def penalty(h):
        penalty_sum = (np.sum(h) - n_points/2)**2
        penalty_sum += np.sum(np.maximum(0, h - 1)) + np.sum(np.maximum(0, -h))
        return penalty_sum
    
    while time() - start_time < budget_s * 0.95:
        # Compute dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        if k_max > n_points:
            k_max = k_max - n_points
        
        # Compute gradient (approximated)
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient approximation for the dominant lag k
            j = (i + k_max) % n_points
            grad[i] = -1 + 2 * h[j]  # This is derivative of h[j] * (1 - h[i]) 
        grad = grad * dx
        
        # Compute penalty gradient
        penalty_grad = np.zeros_like(h)
        penalty_grad += 2 * (np.sum(h) - n_points/2)
        penalty_grad += np.maximum(0, h - 1)  # Fixed values outside bounds
        penalty_grad -= np.maximum(0, -h)
        
        # Candidate step
        h_candidate = h - eta * (grad + lambda_penalty * penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate change
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - best_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Cool temperature
        T *= alpha
        
        # Random perturbation if no improvement
        if no_improvement_count >= max_no_improvement:
            h += np.random.randn(n_points) * 0.05
            h = project_feasible(h, n_points)
            no_improvement_count = 0
            
        iterations += 1
        
        # Update resolution if time to do so and we're not at max resolution
        if iterations % 100 == 0 and n_points < 100:
            if time() - start_time > budget_s * 0.8:
                break
            # Double resolution
            n_points_new = min(n_points * 2, 100)
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_new)
            h_interp = np.interp(x_new, x_old, h)
            h = project_feasible(h_interp, n_points_new)
            n_points = n_points_new
            dx = 2.0 / n_points
            if best_c5 > 0.3805:
                T = 0.5
            else:
                T = 0.1
            
    return (list(best_h), best_c5, n_points)
