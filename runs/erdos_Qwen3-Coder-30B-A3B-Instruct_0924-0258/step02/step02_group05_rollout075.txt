import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    # Differential Evolution parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        # Add small noise to initial solution
        h_perturbed = h + np.random.normal(0, 0.05, n_points)
        h_perturbed = project_feasible(h_perturbed)
        population.append(h_perturbed)
    
    # Store best solution
    best_h = h.copy()
    best_C5 = compute_overlap(best_h)
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            fit = compute_overlap(individual)
            fitness.append(fit)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_C5:
            best_C5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Create new population
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Differential mutation
            trial = population[a] + mutation_factor * (population[b] - population[c])
            
            # Binomial crossover
            for j in range(n_points):
                if np.random.rand() > crossover_prob:
                    trial[j] = population[i][j]
            
            # Project to feasible region
            trial = project_feasible(trial)
            
            # Selection
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Final check with full evaluation
    final_C5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
