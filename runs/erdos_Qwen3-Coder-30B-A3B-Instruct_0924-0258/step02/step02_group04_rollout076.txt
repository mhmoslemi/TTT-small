import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def latin_hypercube_sample(n_points, n_samples=50):
    """Generate Latin Hypercube samples for diverse initialization"""
    samples = []
    for _ in range(n_samples):
        sample = np.random.random(n_points)
        # Shuffle each dimension
        for i in range(n_points):
            sample[i] = (sample[i] + i) / n_points
        samples.append(sample)
    return samples

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a more structured initial pattern
        h = np.ones(n_points) * 0.5
        # Add some structure for better starting point
        mid = n_points // 2
        h[:mid] = 0.7
        h[mid:] = 0.3
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Try multiple configurations
    resolutions = [15, 30, 45, 60, 75]
    
    # Start with 30 points as a good balance
    n_points = 30
    dx = 2.0 / n_points
    
    # Use multiple random initializations
    initial_samples = latin_hypercube_sample(n_points, 20)
    
    # Add some deterministic modifications to explore diverse regions
    for i, sample in enumerate(initial_samples):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create initial proposal
        h_start = sample.copy()
        h_start = project_to_feasible(h_start, n_points)
        
        # Create a local optimization worker
        trial_h = h_start.copy()
        trial_h = project_to_feasible(trial_h, n_points)
        
        # Use SLSQP for constrained optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(obj, trial_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
        except:
            pass
        
        # Add a few random perturbations around current best
        if best_h is not None and time() - start_time < budget_s * 0.9:
            for _ in range(5):
                h_perturb = best_h + np.random.normal(0, 0.05, n_points)
                h_perturb = project_to_feasible(h_perturb, n_points)
                c5_pert = evaluate_erdos_solution(h_perturb, dx)
                if c5_pert < best_c5:
                    best_h = h_perturb.copy()
                    best_c5 = c5_pert
                    best_n_points = n_points
    
    # Final fine-tune if time allows
    if best_h is not None and time() - start_time < budget_s * 0.8:
        h_finetune = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        try:
            res = minimize(lambda h: evaluate_erdos_solution(h, dx), 
                          h_finetune, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
                    best_n_points = n_points
        except:
            pass
    
    if best_h is None:
        # Fallback
        n_points = 30
        h_start = initialize_h(n_points)
        dx = 2.0 / n_points
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        try:
            res = minimize(lambda h: evaluate_erdos_solution(h, dx), 
                          h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                best_h = project_to_feasible(res.x, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
                best_n_points = n_points
        except:
            best_h = h_start
            best_c5 = evaluate_erdos_solution(best_h, dx)
            best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
