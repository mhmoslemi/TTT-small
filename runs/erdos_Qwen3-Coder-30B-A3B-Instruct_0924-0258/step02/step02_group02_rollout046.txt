import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0 / 45)
        
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial solution satisfies constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_c5(h_vec):
        # Compute all correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    # Evolution loop
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h)
            fitnesses.append(-c5)  # Negative for minimization
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(population[winner_idx])
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size] if i + 1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Apply mutation
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=np.random.choice([1, 2]), replace=False)
                for idx in indices:
                    child1[idx] += np.random.normal(0, 0.01)
            
            if np.random.rand() < mutation_rate:
                indices = np.random.choice(n_points, size=np.random.choice([1, 2]), replace=False)
                for idx in indices:
                    child2[idx] += np.random.normal(0, 0.01)
            
            offspring.append(project_h(child1))
            if len(offspring) < population_size:
                offspring.append(project_h(child2))
        
        # Replace worst individuals with offspring
        # Sort population by fitness (ascending due to negative)
        sorted_indices = np.argsort(fitnesses)
        for i in range(population_size // 2):
            population[sorted_indices[i]] = offspring[i]
            
        # Update best solution
        current_best_idx = np.argmax(fitnesses)
        current_best_c5 = -fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
