import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Start with a piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    # Try multiple resolutions with different optimizations
    resolutions = [15, 30, 45, 60, 75]
    time_per_res = budget_s / len(resolutions)
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Multi-start optimization with gradient-based approach
        h_best = h_start.copy()
        c5_best = evaluate_erdos_solution(h_start, dx)
        
        for restart in range(10):  # Multiple restarts
            if time() - start_time > budget_s * 0.95:
                break
            h_perturbed = project_to_feasible(h_best + np.random.normal(0, 0.05, n_points), n_points)
            
            # Local optimization with bounds
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            try:
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                
                res = minimize(obj, h_perturbed, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < c5_best:
                        h_best = h_opt.copy()
                        c5_best = c5_opt
            except:
                pass
            
            # Random search refinement
            for _ in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                h_candidate = project_to_feasible(h_best + np.random.normal(0, 0.03, n_points), n_points)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < c5_best:
                    h_best = h_candidate
                    c5_best = c5_candidate
        
        if c5_best < best_c5:
            best_h = h_best.copy()
            best_c5 = c5_best
            best_n_points = n_points
    
    # Final refinement with the best found solution
    if best_h is not None:
        dx = 2.0 / best_n_points
        # Final optimization of the best solution
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(best_n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - best_n_points / 2}]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                best_h = project_to_feasible(res.x, best_n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass
    
    # Return best solution found
    if best_h is None:
        n_points = 15
        dx = 2.0 / n_points
        best_h = initialize_h(n_points)
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = n_points
        
    return (list(best_h), best_c5, best_n_points)
