import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():

        n_points = len(h0)

    # Initialize
    h = h0.copy()
    h = project_to_feasible(h, n_points/2.0)
    current_c5 = evaluate(h, 2.0/n_points)
    
    # Annealing parameters
    T = 0.5
    alpha = 0.995
    min_T = 1e-6
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Track acceptance rate
    accept_count = 0
    total_count = 0
    acceptance_window = 50
    
    iteration = 0
    while time.time() - start_time < budget_s:
        total_count += 1
        
        # Generate candidate
        h_prime = h + np.random.normal(0, T, n_points)
        h_prime = project_to_feasible(h_prime, n_points/2.0)
        candidate_c5 = evaluate(h_prime, 2.0/n_points)
        
        # Accept or reject
        if candidate_c5 < current_c5:
            h = h_prime
            current_c5 = candidate_c5
            accept_count += 1
        else:
            # Probability of acceptance for worse solution
            p_accept = np.exp(-(candidate_c5 - current_c5) / T)
            if np.random.random() < p_accept:
                h = h_prime
                current_c5 = candidate_c5
                accept_count += 1
        
        # Update best solution found
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adaptive cooling
        if total_count % acceptance_window == 0:
            acceptance_rate = accept_count / acceptance_window
            if acceptance_rate < 0.2:
                alpha = max(0.99, alpha * 0.99)
            elif acceptance_rate > 0.5:
                alpha = min(0.999, alpha * 1.01)
            
            accept_count = 0
        
        # Cool temperature
        T = max(min_T, T * alpha)
        
        iteration += 1
    
    final_c5 = evaluate(best_h, 2.0/n_points)
    return (best_h.tolist(), float(final_c5), n_points)
