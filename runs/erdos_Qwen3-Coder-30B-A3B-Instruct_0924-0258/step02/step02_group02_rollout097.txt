import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_fitness(h_vec):
        return -compute_c5(h_vec)
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = [evaluate_fitness(population[i]) for i in range(population_size)]
        
        # Select top 50% using tournament selection
        selected_indices = []
        for _ in range(population_size // 2):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmax([fitness_scores[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create offspring
        offspring = []
        for i in range(0, len(selected_indices), 2):
            if i + 1 < len(selected_indices) and np.random.rand() < crossover_rate:
                parent1 = population[selected_indices[i]]
                parent2 = population[selected_indices[i+1]]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    mut_idx = np.random.randint(n_points)
                    child1[mut_idx] = np.random.rand()
                if np.random.rand() < mutation_rate:
                    mut_idx = np.random.randint(n_points)
                    child2[mut_idx] = np.random.rand()
                
                offspring.append(project_h(child1))
                offspring.append(project_h(child2))
            else:
                offspring.append(population[selected_indices[i]])
                if i + 1 < len(selected_indices):
                    offspring.append(population[selected_indices[i+1]])
        
        # Fill the rest of the population if necessary
        while len(offspring) < population_size:
            offspring.append(population[np.random.randint(population_size)])
            
        # Replace bottom 50% with offspring
        sorted_indices = np.argsort(fitness_scores)
        for i in range(population_size // 2):
            population[sorted_indices[i]] = offspring[i]
        
        # Update best solution
        current_best_idx = np.argmax(fitness_scores)
        current_best_c5 = -fitness_scores[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
