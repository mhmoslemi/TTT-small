import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def get_max_shifts(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    max_indices = np.where(np.abs(corr - max_val) < 1e-10)[0]
    # Adjust indices to be in [0, n_points-1]
    max_shifts = [idx - n_points + 1 if idx >= n_points else idx for idx in max_indices]
    return max_shifts

    # Simple line search: try setting w_i = 1 - h_i or 0 - h_i to minimize grad·w

        else:
            s[i] = max(0.0 - h[i], -1.0)  # Amount to decrease

    # The Frank-Wolfe descent direction is putting all weight on the most negative gradient

    # Let's compute the best (linearly constrained) descent direction:
    # argmin s : (grad·(-s)) subject to sum(s) = 0, s in [0,1]^n, h+s in [0,1]^n

    # Assume: s proportional to -grad, but normalized to sum = 0, and clipped to [0,1]
    # Let's compute convex combination smoothly:
    
    # Given s = -grad, make s feasible: sum(s)=0 and 0<=s<=1
    # This is similar to fingerprinting method in experiments
    
    # Adjust so sum = 0, but stay in [0,1] band on average
    # Gibbs-like approach without normalization:
    s = s - np.mean(s)  # Make it zero-mean

    # Don't directly solve project; instead sample domain to get better descent

    new_h = project_to_feasible(new_h, n_points)
