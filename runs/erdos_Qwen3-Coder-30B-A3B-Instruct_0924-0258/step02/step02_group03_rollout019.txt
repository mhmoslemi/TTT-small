import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    n_particles = 100
    particles = [h0 + np.random.normal(0, 0.02, n_points) for _ in range(n_particles)]
    particles = [project_to_feasible(p, n_points/2.0) for p in particles]
    
    # Temperature and noise adaptation
    temp = 1.0
    sigma = 0.1
    acceptance_rate = 0.3
    target_acceptance = 0.3
    adaptation_interval = 10
    
    best_h = h0.copy()
    best_c5 = evaluate(best_h, 2.0/n_points)
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s:
        # Propose new particles
        new_particles = []
        weights = []
        
        for h in particles:
            # Random walk proposal
            h_new = h + np.random.normal(0, sigma, n_points)
            h_new = project_to_feasible(h_new, n_points/2.0)
            c5 = evaluate(h_new, 2.0/n_points)
            
            # Importance weight based on temperature
            weight = np.exp(-c5 / temp)
            weights.append(weight)
            new_particles.append(h_new)
        
        # Normalize weights
        weights = np.array(weights)
        weights = weights / (np.sum(weights) + 1e-10)
        
        # Resampling
        indices = np.random.choice(n_particles, size=n_particles//2, p=weights)
        particles = [new_particles[i] for i in indices]
        
        # Generate mutations for selected particles
        mutations = []
        for h in particles:
            # Global exploration
            h1 = h + np.random.normal(0, sigma, n_points)
            h1 = project_to_feasible(h1, n_points/2.0)
            mutations.append(h1)
            # Local exploitation
            h2 = h + np.random.normal(0, 0.05, n_points)
            h2 = project_to_feasible(h2, n_points/2.0)
            mutations.append(h2)
        
        # Replace with new particles
        particles = particles + mutations
        
        # Update best
        for h in particles:
            c5 = evaluate(h, 2.0/n_points)
            if c5 < best_c5:

                best_h = h.copy()
                no_improvement_count = 0
