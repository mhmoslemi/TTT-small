import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

def trust_region_step(h, grad, hessian, trust_radius, target_sum):
    n = len(h)
    # Simplified trust region subproblem: minimize g^T d + 0.5 d^T H d subject to ||d|| <= trust_radius
    # Since we don't need full precision, use a simple projected gradient step with clipping.
    # First compute a naive step (this is a simplified version)
    try:
        # Simple Newton step
        if len(hessian.shape) == 1:
            hessian = np.diag(hessian)
        
        step = -np.linalg.solve(hessian + 1e-8 * np.eye(n), grad)
        
        # Project step to trust region
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        new_h = h + step
    except:
        # Fallback to gradient descent
        step = -grad * 0.001
        new_h = h + step
    
    # Project to feasible set
    new_h = project_to_feasible(new_h, target_sum)
    
    return new_h

    best_c5 = evaluate(h, dx)
    
    trust_radius = 0.2
    min_radius = 0.1
    max_radius = 1.0
    improvement_threshold = 1e-4
    no_improve_count = 0
    max_no_improve = 100
    
    # Initialize gradient and Hessian approximations
    # We approximate using finite differences and do not compute full Hessian due to complexity.
    # For simplicity, we'll compute the gradient using finite differences and update using BFGS-like steps.
    
    # Function to compute gradient using finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_val = evaluate(h, dx)
        for i in range(len(h)):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = project_to_feasible(h_eps, target_sum)
            val = evaluate(h_eps, dx)
            grad[i] = (val - base_val) / eps
