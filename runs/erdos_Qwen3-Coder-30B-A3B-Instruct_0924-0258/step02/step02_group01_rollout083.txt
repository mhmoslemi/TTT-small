import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial setup - use provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iter = 100
    patience = 20
    improvement_count = 0
    
    # Generate population of candidates
    population_size = 50
    candidates = [h_start.copy()]
    for _ in range(population_size - 1):
        h = np.random.dirichlet(np.ones(n_points)) * (n_points / 2)
        candidates.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in candidates]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = candidates[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb each candidate
        new_candidates = []
        for h in candidates:
            # Create perturbed version
            h_pert = h.copy()
            indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            h_pert[indices] += np.random.normal(0, 0.1, len(indices))
            
            # Project to feasible region
            h_pert = project_to_feasible(h_pert, n_points)
            
            # Optimize with SQP
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            try:
                res = minimize(obj, h_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    new_candidates.append(h_opt)
                else:
                    new_candidates.append(h_pert)
            except:
                new_candidates.append(h_pert)
        
        # Evaluate new candidates
        new_fitness = [evaluate_erdos_solution(h, dx) for h in new_candidates]
        
        # Update candidates
        for i, (h, f) in enumerate(zip(new_candidates, new_fitness)):
            if f < fitness[i]:  # Improvement
                candidates[i] = h
                fitness[i] = f
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = candidates[best_idx].copy()
            best_c5 = fitness[best_idx]
            improvement_count = 0
        else:
            improvement_count += 1
        
        if improvement_count >= patience:
            break
    
    # Resolution refinement
    current_h = best_h.copy()
    current_n = n_points
    current_c5 = best_c5
    
    while current_n < 1000 and time() - start_time < budget_s * 0.9:
        current_n *= 2
        x_orig = np.linspace(0, 2, len(current_h))
        x_new = np.linspace(0, 2, current_n)
        h_interp = np.interp(x_new, x_orig, current_h)
        current_h = project_to_feasible(h_interp, current_n)
        dx_new = 2.0 / current_n
        
        # Optimize at new resolution
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx_new)
        
        bounds = [(0, 1) for _ in range(current_n)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - current_n / 2}]
        
        try:
            res = minimize(obj, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                current_h = project_to_feasible(res.x, current_n)
                current_c5 = evaluate_erdos_solution(current_h, dx_new)
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
        except:
            pass
    
    return (list(best_h), best_c5, len(best_h))
