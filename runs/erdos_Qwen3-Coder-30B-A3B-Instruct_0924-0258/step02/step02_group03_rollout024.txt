import numpy as np
import time

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    h = h0.copy()
    h = project_to_feasible(h, n_points/2.0)
    current_c5 = evaluate(h, 2.0/n_points)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing
    T0 = 0.5
    T = T0
    alpha = 0.995
    max_iter = 10000
    iteration = 0
    accept_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        # Generate candidate
        candidate_h = h + np.random.normal(0, T, n_points)
        candidate_h = project_to_feasible(candidate_h, n_points/2.0)
        candidate_c5 = evaluate(candidate_h, 2.0/n_points)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.random() < np.exp(-(candidate_c5 - current_c5) / T):
            h = candidate_h
            current_c5 = candidate_c5
            accept_count += 1
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = iteration
        
        # Cooling
        T = max(T * alpha, 1e-6)
        
        # Reset temperature if no improvement in a while
        if iteration - last_improvement > 200:
            T = T0
            last_improvement = iteration
            
        iteration += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
