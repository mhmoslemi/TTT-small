import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_fitness(h_vec):
        return -compute_c5(h_vec)

    # Genetic Algorithm Parameters
    population_size = 100
    generations = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    population.append(h.copy())
    for i in range(1, population_size):
        h_mut = h + np.random.normal(0, 0.05, n_points)
        population.append(project_h(h_mut))
    
    # Evaluate initial fitness
    fitness_scores = [evaluate_fitness(ind) for ind in population]
    best_idx = np.argmax(fitness_scores)
    best_h = population[best_idx].copy()
    best_c5 = -fitness_scores[best_idx]

    start_time = time.time()
    
    for gen in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:  # Leave some buffer
            break
            
        # Tournament selection
        selected_indices = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmax([fitness_scores[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = population[selected_indices[i]]
            parent2 = population[selected_indices[i+1]] if i+1 < population_size else population[selected_indices[0]]
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child1[mutate_idx] = np.clip(child1[mutate_idx] + np.random.normal(0, 0.05), 0, 1)
            
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child2[mutate_idx] = np.clip(child2[mutate_idx] + np.random.normal(0, 0.05), 0, 1)
            
            # Project to valid domain
            child1 = project_h(child1)
            child2 = project_h(child2)
            
            offspring.extend([child1, child2])
        
        # Replace bottom half of population with offspring
        sorted_indices = np.argsort(fitness_scores)[::-1]  # Descending order
        for i in range(population_size // 2):
            population[sorted_indices[-(i+1)]] = offspring[i]
        
        # Update fitness scores
        fitness_scores = [evaluate_fitness(ind) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmax(fitness_scores)
        if -fitness_scores[current_best_idx] < best_c5:
            best_c5 = -fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
