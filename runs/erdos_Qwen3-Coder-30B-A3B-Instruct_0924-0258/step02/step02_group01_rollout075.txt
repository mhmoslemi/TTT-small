import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped  # Edge case
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover between two parents"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Apply Gaussian mutation to individual"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, std, len(individual))
    individual += mask * noise
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = 200
    patience = 20
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    population = []
    fitnesses = []
    
    # Add the initial solution if available
    h_feasible = project_feasible(initial_h, n_points)
    population.append(h_feasible)
    c5 = evaluate_erdos_solution(h_feasible, dx)
    fitnesses.append(c5)
    
    # Fill the rest with random feasible solutions
    for _ in range(pop_size - 1):
        h_rand = np.random.rand(n_points)
        h_feasible = project_feasible(h_rand, n_points)
        population.append(h_feasible)
        c5 = evaluate_erdos_solution(h_feasible, dx)
        fitnesses.append(c5)
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Genetic Algorithm
    generation = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness for all individuals (already computed)
        # Order population by fitness (ascending - minimize)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Check for improvement
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > patience:
                break
        
        # Create new population
        new_population = [population[0].copy()]
        new_fitnesses = [fitnesses[0]]
        
        # Keep top 25% (best individuals)
        keep_count = pop_size // 4
        for i in range(1, keep_count):
            new_population.append(population[i].copy())
            new_fitnesses.append(fitnesses[i])
            
        # Generate remainder via selection, crossover, and mutation
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_k)
            parent2 = tournament_selection(population, fitnesses, tournament_k)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            # Evaluate
            c5_1 = evaluate_erdos_solution(child1, dx)
            c5_2 = evaluate_erdos_solution(child2, dx)
            
            new_population.extend([child1, child2])
            new_fitnesses.extend([c5_1, c5_2])
        
        # Trim to exact population size
        indices = np.argsort(new_fitnesses)[:pop_size]
        population = [new_population[i] for i in indices]
        fitnesses = [new_fitnesses[i] for i in indices]
        
        generation += 1
        
    return (list(best_h), best_c5, n_points)
