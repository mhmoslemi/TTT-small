import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    h_start = project_feasible(h_start, n_points)
    dx = 2.0 / n_points
    
    # Simulated annealing parameters
    T = 0.5
    alpha = 0.995
    eta = 0.01
    lambda_penalty = 1e-3
    
    # Initialize
    h_current = h_start.copy()
    c5_current = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    iterations_no_improve = 0
    max_no_improve = 100
    
    # Annealing loop
    while time() - start_time < budget_s * 0.95:
        # Find dominant lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        dominant_lag = np.argmax(corr)
        
        # Compute gradient at dominant lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + dominant_lag) % n_points
            grad[i] = -(1 - h_current[j] - h_current[i])
        
        # Compute penalty term gradient
        penalty_grad = np.zeros(n_points)
        h_sum = np.sum(h_current)
        penalty_grad += 2 * lambda_penalty * (h_sum - n_points / 2)
        
        # Update step
        h_candidate = h_current + eta * (-grad + penalty_grad)
        
        # Project to feasible region
        h_candidate = project_feasible(h_candidate, n_points)
        
        # Evaluate new solution
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta_c5 = c5_candidate - c5_current
        
        # Accept or reject
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.random():
            h_current = h_candidate
            c5_current = c5_candidate
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                iterations_no_improve = 0
            else:
                iterations_no_improve += 1
        else:
            iterations_no_improve += 1
            
        # Cooling schedule
        T *= alpha
        
        # Reset if no improvement for too long
        if iterations_no_improve > max_no_improve:
            h_current = project_feasible(h_current + np.random.randn(n_points) * 0.05, n_points)
            c5_current = evaluate_erdos_solution(h_current, dx)
            iterations_no_improve = 0
    
    # Multi-resolution refinement
    resolutions = [60, 120]
    for new_n in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate
        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, new_n)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, new_n)
        
        dx_new = 2.0 / new_n
        c5_new = evaluate_erdos_solution(h_interp, dx_new)
        
        if c5_new < best_c5:
            best_h = h_interp
            best_c5 = c5_new
            n_points = new_n
            dx = dx_new
    
    return (list(best_h), best_c5, n_points)
