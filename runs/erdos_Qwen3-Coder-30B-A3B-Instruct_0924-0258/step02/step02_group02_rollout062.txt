import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_overlaps(h_vec):
        # Compute all overlaps for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return corr

    # Initialize
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    learning_rate = 0.01
    patience = 0
    max_patience = 200
    
    # We'll do a more systematic optimization approach
    # Iterate through all possible shifts and update h accordingly
    shift_order = list(range(n_points * 2 - 1))  # full correlation length
    # We only consider significant shifts (those that make sense for the problem)
    shift_order = list(range(1, n_points))  # more realistic shifts
    
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Do multiple passes of optimization
        for s in shift_order:
            # Sample the current overlap for this shift
            overlaps = compute_overlaps(h)
            current_c5 = overlaps[s + n_points - 1]  # This is now C(s)
            
            # Try to decrease this overlap by changing h[i] and h[i+s]
            # Might be expensive, instead do projected gradient descent
            # on this specific shift
            if current_c5 > best_c5 * 0.9999:  # Don't waste time on small changes
                continue
                
            # Simple gradient descent idea for this one shift
            # More precise but expensive: compute gradient with respect to h
            
            # Instead, let's do a local search:
            # Try small perturbations to reduce overlap for this shift
            h_backup = h.copy()
            
            # Try small step in direction of decrease for this shift
            
            # Perturb h at a few points to reduce impact
            for i in range(0, n_points, max(1, n_points // 10)):
                if i + s >= n_points:
                    continue
                
                # Try one small change to reduce the overlap
                step = 0.0001
                h_plus = h.copy()
                h_minus = h.copy()
                
                # Try increase h[i] and decrease h[i+s]
                if h[i] < 1 - 0.0001:
                    h_plus[i] += step
                if h[i+s] > 0.0001:
                    h_plus[i+s] -= step
                
                h_plus = project_h(h_plus)
                overlap_plus = compute_overlaps(h_plus)[s + n_points - 1]
                
                h_minus[i] -= step
                h_minus[i+s] += step
                h_minus = project_h(h_minus)
                overlap_minus = compute_overlaps(h_minus)[s + n_points - 1]
                
                # Choose better (or just use one to improve diversity)
                if overlap_minus < overlap_plus and overlap_minus < current_c5:
                    h = h_minus.copy()
                elif overlap_plus < current_c5:
                    h = h_plus.copy()
                
                # Safety check that we're still valid after change
                h = project_h(h)
                
                # If we actually improved something significant, take it
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    patience = 0
                else:
                    patience += 1
                    if patience > max_patience:
                        break
                        
            # Restore original for next shift
            h = h_backup.copy()
            
        # Periodic check for improvement
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience * 2:
                # Maybe need to restart with a little noise
                h = best_h.copy()
                noise = np.random.normal(0, 0.001, n_points)
                h = h + noise
                h = project_h(h)
            if patience > max_patience * 3:
                break
                
        # Cool down learning rate occasionally
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.1
            
        iteration += 1
        
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
