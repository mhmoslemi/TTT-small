import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use a different approach: iterative projection with adaptive learning rate
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 50000
    iteration = 0
    learning_rate = 0.02
    patience = 0
    patience_limit = 100
    improvement_threshold = 1e-6
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        new_h = h.copy()
        # For each position, update based on gradient approximation
        # This is a simple gradient descent variant
        for i in range(n_points):
            # Compute the influence of changing h[i] on each correlation term
            grad = np.zeros(n_points)
            h_temp = h.copy()
            h_temp[i] += 1e-6
            
            # Compute overlaps with perturbed h
            corr_orig = np.correlate(h, 1-h, mode='full') * dx
            corr_pert = np.correlate(h_temp, 1-h_temp, mode='full') * dx
            
            # Approximate gradient for this point
            delta_cors = corr_pert - corr_orig
            grad = delta_cors[:n_points]  # Only relevant part
            
            new_h[i] -= learning_rate * grad[i]
        
        new_h = project_h(new_h)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                learning_rate *= 0.8
                patience = 0
                if learning_rate < 1e-6:
                    break
        
        h = new_h
        iteration += 1
        if iteration % 1000 == 0:
            # Periodically check if time is running out
            if time.time() - start_time > budget_s * 0.95:
                break
        if iteration >= max_iter:
            break
    
    # Final validation with more precise optimization
    final_h = project_h(best_h.copy())
    final_c5 = compute_c5(final_h)
    
    # Try a bit more optimization
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try scipy optimization to fine tune
    result = minimize(evaluate_objective, final_h, method='SLSQP', 
                      bounds=bounds, constraints=constraints, 
                      options={'maxiter': 50, 'ftol': 1e-10})
    
    if result.success:
        final_h = project_h(result.x)
        final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
