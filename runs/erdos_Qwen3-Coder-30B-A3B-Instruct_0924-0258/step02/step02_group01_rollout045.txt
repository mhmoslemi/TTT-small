import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    population = [project_feasible(h_start.copy(), n_points)]
    for _ in range(pop_size - 1):
        new_h = np.random.rand(n_points)
        population.append(project_feasible(new_h, n_points))
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(ind, 2.0 / n_points) for ind in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # GA parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    max_generations = int(budget_s * 20)  # Rough estimate
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_k)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create new population through crossover and mutation
        new_population = [population[best_idx].copy()]  # Elitism
        
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate:
                parent1, parent2 = np.random.choice(pop_size, 2, replace=False)
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, selected[parent1], selected[parent2])
            else:
                parent = np.random.choice(pop_size)
                child = selected[parent].copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                child = project_feasible(child, n_points)
            
            new_population.append(child)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_erdos_solution(ind, 2.0 / n_points) for ind in population]
        
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
    
    return (list(best_h), best_c5, n_points)
