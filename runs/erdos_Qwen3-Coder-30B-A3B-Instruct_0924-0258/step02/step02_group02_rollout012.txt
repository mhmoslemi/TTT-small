import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use a hybrid optimization approach
    # Start with the provided initial solution
    h = project_h(h)
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    max_iter = 10000
    iteration = 0
    patience = 0
    patience_limit = 100
    lr = 0.01
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Simple gradient descent with projection
        # 1. Compute gradients via finite difference
        grad = np.zeros_like(h)
        eps = 1e-6
        current_c5 = compute_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update h using gradient descent
        h = h - lr * grad
        
        # Project back to feasible region
        h = project_h(h)
        
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
            lr = max(1e-6, lr * 1.05)  # Increase learning rate
        else:
            patience += 1
            if patience > patience_limit:
                lr = max(1e-6, lr * 0.5)  # Decrease learning rate
                patience = 0
        
        iteration += 1
        if iteration > max_iter:
            break
        
        # Occasionally add noise to escape local minima
        if iteration % 50 == 0:
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
