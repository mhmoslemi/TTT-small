import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    np.random.seed(seed)
    start_time = time()

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    num_generations = int(budget_s / 0.5) if budget_s > 1 else 1000
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = project_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
            
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            offspring.extend([child1, child2])
        
        # Keep only pop_size individuals
        offspring = offspring[:pop_size]
        
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(h, dx) for h in offspring]
        
        # Survival: Keep best individuals
        all_individuals = population + offspring
        all_fitness = fitness + offspring_fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(all_fitness)
        top_indices = sorted_indices[:pop_size]
        
        # Update population
        population = [all_individuals[i] for i in top_indices]
        fitness = [all_fitness[i] for i in top_indices]
        
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    return (list(best_h), best_c5, n_points)
