import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n_points = n_points
    
    # Binary search parameters
    low = 0.37
    high = 0.38085
    tolerance = 1e-5
    max_iterations = 40
    
    # Precompute shift indices for correlation computation
    shift_indices = []
    for k in range(-n_points+1, n_points):
        shift_indices.append(np.arange(n_points) - k)
    
    # Binary search loop
    for _ in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        mid = (low + high) / 2.0
        
        # Define convex feasibility problem
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add correlation constraints
        for k in range(-n_points+1, n_points):
            # Circular shift indices
            indices = np.arange(n_points) - k
            indices = indices % n_points
            # Compute correlation sum
            corr_sum = cp.sum(cp.multiply(h, 1 - h[indices]))
            constraints.append(corr_sum <= mid * n_points / 2)
        
        # Problem setup
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iter=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
                h_solution = h.value
                c5 = evaluate_erdos_solution(h_solution, dx)
                if c5 < best_c5:
                    best_h = h_solution.copy()
                    best_c5 = c5
                    best_n_points = n_points
                high = mid
            else:
                low = mid
        except:
            low = mid
    
    # Refinement with gradient-based method if we found a good solution
    if time() - start_time < budget_s * 0.95:
        ref_time = budget_s - (time() - start_time)
        
        def objective(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        def gradient(h):
            # Approximate gradient using finite differences
            eps = 1e-6
            grad = np.zeros_like(h)
            base_val = objective(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                grad[i] = (objective(h_plus) - base_val) / eps
            return grad
        
        # Refine best solution found
        h_refine = best_h.copy()
        # Multiple restarts
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = project_to_feasible(h_refine + np.random.normal(0, 0.01, n_points), n_points)
            try:
                res = minimize(objective, h_restart, method='L-BFGS-B', 
                               bounds=[(0,1) for _ in range(n_points)],
                               jac=gradient, options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_res = project_to_feasible(res.x, n_points)
                    c5_res = evaluate_erdos_solution(h_res, dx)
                    if c5_res < best_c5:
                        best_h = h_res.copy()
                        best_c5 = c5_res
                        best_n_points = n_points
            except:
                pass
            # Perturb further for exploration
            h_refine = project_to_feasible(h_refine + np.random.normal(0, 0.01, n_points), n_points)
    
    return (list(best_h), best_c5, best_n_points)
