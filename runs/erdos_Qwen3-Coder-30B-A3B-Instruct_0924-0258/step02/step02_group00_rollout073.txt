import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from precomputed construction or default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution strategy with particle swarm optimization
    resolutions = [25, 30, 35]
    
    # PSO parameters
    num_particles = 50
    max_iters = 200
    omega_start, omega_end = 0.7, 0.4
    c1, c2 = 2.0, 2.0
    inertia_decay = (omega_start - omega_end) / max_iters
    
    for res_idx, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
            
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            
        # Initialize particles
        particles = []
        velocities = []
        personal_best = []
        personal_best_values = []

        for _ in range(num_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
            val = evaluate_erdos_solution(h, dx_try)
            personal_best.append(h.copy())
            personal_best_values.append(val)
        
        global_best_idx = np.argmin(personal_best_values)
        global_best = personal_best[global_best_idx].copy()
        global_best_value = personal_best_values[global_best_idx]
        
        # PSO iteration
        omega = omega_start
        for i in range(max_iters):
            if time() - start_time > budget_s * 0.95:
                break
                
            for p in range(num_particles):
                r1, r2 = np.random.rand(), np.random.rand()
                velocities[p] = (omega * velocities[p] + 
                                c1 * r1 * (personal_best[p] - particles[p]) + 
                                c2 * r2 * (global_best - particles[p]))
                
                particles[p] += velocities[p]
                particles[p] = project_to_feasible(particles[p], n_points)
                
                val = evaluate_erdos_solution(particles[p], dx_try)
                
                if val < personal_best_values[p]:
                    personal_best[p] = particles[p].copy()
                    personal_best_values[p] = val
                    
                if val < global_best_value:
                    global_best = particles[p].copy()
                    global_best_value = val
                    
            omega -= inertia_decay
            
        # Update the best solution found so far
        if global_best_value < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_value
            best_n_points = n_points
            
        h_start = best_h.copy()

    # Refinement with gradient descent
    refined_h = best_h.copy()
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        opt_result = minimize(
            lambda x: evaluate_erdos_solution(project_to_feasible(x, best_n_points), dx),
            refined_h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(best_n_points)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if opt_result.success:
            refined_h = project_to_feasible(opt_result.x, best_n_points)
            c5_refined = evaluate_erdos_solution(refined_h, dx)
            if c5_refined < best_c5:
                best_h = refined_h
                best_c5 = c5_refined

    return (list(best_h), best_c5, best_n_points)
