import numpy as np
import cvxpy as cp
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available, otherwise create a reasonable starting point
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for the minimum C5
    C_low = 0.37
    C_high = 0.38085
    best_feasible_h = None
    best_feasible_C = float('inf')
    
    # Setup CVXPY problem for feasibility check
    h_vars = cp.Variable(n_points)
    constraints = []
    
    # Add initial constraints
    constraints.append(cp.sum(h_vars) == n_points / 2)
    constraints.append(h_vars >= 0)
    constraints.append(h_vars <= 1)
    
    # Binary search loop
    binary_search_time = 60
    search_start = time()
    
    while time() - search_start < binary_search_time and C_high - C_low > 1e-6:
        C_mid = (C_low + C_high) / 2
        
        # Build the constraint matrix for all shifts
        # We'll add constraints for all possible shifts (k)
        for k in range(-n_points + 1, n_points):
            # Calculate the correlation constraint
            # sum_{i} h[i] * (1 - h[(i+k) % n]) <= C_mid * n/2 * dx 
            expr = 0
            for i in range(n_points):
                shifted_i = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[shifted_i])
            expr *= dx
            constraints.append(expr <= C_mid)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL or prob.status == cp.LOCALLY_OPTIMAL:
                feasible_h = h_vars.value
                best_feasible_h = feasible_h
                best_feasible_C = C_mid
                C_high = C_mid
            else:
                C_low = C_mid
        except Exception as e:
            C_low = C_mid
            
        # Clean up constraints for next iteration
        constraints = constraints[:-2 * n_points + 1]  # Remove the last set of shift constraints
    
    # Refinement using gradient-based optimization
    if best_feasible_h is not None:
        h_best = best_feasible_h
        # Normalize (ensure the area constraint is met)
        h_best = h_best * (n_points / 2) / np.sum(h_best)
        current_c5 = evaluate_erdos_solution(h_best, dx)
        best_c5 = current_c5
        best_h = h_best.copy()
    else:
        # If binary search failed, return the initial solution
        current_c5 = best_c5
        best_h = h_start.copy()
    
    # Refinement step with gradient descent
    refine_start = time()
    max_refinement_time = budget_s - (time() - start_time) - 10
    
    def obj_func(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def proj_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def gradient_descent_refinement(h_init, max_time):
        h_curr = h_init.copy()
        iter_count = 0
        t_start = time()
        while time() - t_start < max_time:
            iter_count += 1
            # Small perturbation
            h_pert = h_curr + np.random.normal(0, 0.02, n_points)
            h_pert = proj_h(h_pert)
            
            c5_curr = obj_func(h_curr)
            c5_pert = obj_func(h_pert)
            
            if c5_pert < c5_curr:
                h_curr = h_pert.copy()
            else:
                # Occasionally accept worse solutions to escape local minima
                if np.random.rand() < 0.1:
                    h_curr = h_pert.copy()
        return h_curr
    
    if time() - refine_start < max_refinement_time:
        try:
            refined_h = gradient_descent_refinement(best_h, max_refinement_time)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
