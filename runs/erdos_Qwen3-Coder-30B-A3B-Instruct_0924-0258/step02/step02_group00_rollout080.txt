import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_n = n_points

    # Binary search settings
    c_low = 0.37
    c_high = 0.38085
    tol = 1e-5
    max_iter = 40
    
    # Solver settings
    solver_options = {'feastol': 1e-6}
    
    # Keep track of best feasible solution
    best_feasible_h = None
    best_feasible_c = float('inf')
    
    # Binary search to find minimal C
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        c_mid = (c_low + c_high) / 2
        
        # Formulate convex feasibility problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add correlation constraints
        for k in range(-(n_points-1), n_points):
            # Compute correlation for shift k
            corr_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                corr_expr += h_var[i] * (1 - h_var[j])
            
            constraints.append(corr_expr <= c_mid * n_points / 2)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, **solver_options)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                if h_sol is not None:
                    c5_sol = evaluate_erdos_solution(h_sol, dx)

                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except Exception:
            c_low = c_mid

    # If we found a feasible solution, try to refine it with gradient descent

    c5_refined = best_feasible_c if best_feasible_c < float('inf') else best_c5

            corr = np.correlate(h, 1 - h, mode='full')
            return np.log(np.sum(np.exp(alpha * corr))) / alpha
        
        def project_h(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            return h

        h_current = h_refined.copy()

            eps = 1e-6
            corr_orig = np.correlate(h_current, 1 - h_current, mode='full')

            h_grad = np.zeros_like(h_current)

                h_plus = h_current.copy()

                h_grad[i] = (max_plus - max_orig) / eps

            h_new = h_current - step_size * h_grad

            c5_new = evaluate_erdos_solution(h_current, dx)

    # Final return
