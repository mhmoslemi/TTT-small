import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    h_start = project_feasible(h_start, n_points)
    dx = 2.0 / n_points
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    iteration = 0
    stagnation_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95:
        # Compute correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for the max lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - best_h[j]) - best_h[i]
        grad *= dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        t = iteration + 1
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update h
        best_h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        best_h = project_feasible(best_h, n_points)
        
        # Evaluate and track best
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            last_improvement = iteration
        else:
            stagnation_count = iteration - last_improvement
            
        # Restart if stagnated
        if stagnation_count > 100:
            h_restart = project_feasible(best_h + np.random.normal(0, 0.05, n_points), n_points)
            best_h = h_restart.copy()
            stagnation_count = 0
        
        iteration += 1
        
        # Optional refinement at the end
        if iteration % 500 == 0 and n_points < 100:
            # Refine resolution
            new_n_points = min(n_points * 2, 100)
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, new_n_points)
            h_interp = np.interp(x_new, x_orig, best_h)
            best_h = project_feasible(h_interp, new_n_points)
            n_points = new_n_points
            dx = 2.0 / n_points
            
    return (list(best_h), best_c5, n_points)
