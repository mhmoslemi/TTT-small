import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 50
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # DE parameters
    pop_size = 25
    F = 0.7
    CR = 0.5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        individual = h.copy() + np.random.normal(0, 0.05, n_points)
        individual = project_feasible(individual)
        population.append(individual)
    
    best_fitness = float('inf')
    best_individual = h.copy()
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = []
        for individual in population:
            fitness = compute_overlap(individual)
            fitnesses.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = individual.copy()
        
        # Create new population
        new_population = []
        for i in range(pop_size):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Select three distinct individuals
            indices = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Differential mutation
            mutant = a + F * (b - c)
            
            # Ensure feasibility
            mutant = project_feasible(mutant)
            
            # Crossover
            trial = np.zeros_like(mutant)
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Selection
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Final validation
    final_C5 = compute_overlap(best_individual)
    return (best_individual.tolist(), final_C5, n_points)
