import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in indices]
    winner_idx = indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.5):
    """Blend crossover for real-valued individuals"""
    offspring1 = alpha * parent1 + (1 - alpha) * parent2
    offspring2 = (1 - alpha) * parent1 + alpha * parent2
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.2, std=0.1):
    """Mutate an individual with Gaussian noise"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        individual += noise
    return individual

def genetic_optimize(h_start, n_points, dx, time_limit):
    """Optimize using genetic algorithm"""
    start_time = time()
    
    # Parameters
    pop_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    generations = 0
    stagnation_count = 0
    max_stagnation = 20
    
    # Initialize population
    population = []
    if h_start is not None:
        population.append(h_start.copy())
        for _ in range(pop_size - 1):
            h_rand = np.random.rand(n_points)
            h_rand = project_feasible(h_rand, n_points)
            population.append(h_rand)
    else:
        for _ in range(pop_size):
            h_rand = np.random.rand(n_points)
            h_rand = project_feasible(h_rand, n_points)
            population.append(h_rand)
    
    # Evaluate fitness
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Main GA loop
    while time() - start_time < time_limit:
        generations += 1
        if stagnation_count >= max_stagnation:
            break
            
        # Create new population
        new_population = []
        
        # Elitism - keep best individuals
        sorted_indices = np.argsort(fitnesses)
        for i in range(pop_size // 4):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                offspring1, offspring2 = blend_crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Project to feasible region
            offspring1 = project_feasible(offspring1, n_points)
            offspring2 = project_feasible(offspring2, n_points)
            
            new_population.extend([offspring1, offspring2])
            if len(new_population) > pop_size:
                new_population.pop()
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Survival selection (keep top 75%)
        all_individuals = new_population + population
        all_fitnesses = new_fitnesses + fitnesses
        sorted_indices = np.argsort(all_fitnesses)
        survivors = [all_individuals[i] for i in sorted_indices[:pop_size]]
        survivor_fitnesses = [all_fitnesses[i] for i in sorted_indices[:pop_size]]
        
        # Update population and best solution
        population = survivors
        fitnesses = survivor_fitnesses
        
        # Update global best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
    
    return best_h, best_c5
