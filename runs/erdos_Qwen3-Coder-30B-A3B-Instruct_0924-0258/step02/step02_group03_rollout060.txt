import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Distribute difference evenly
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        # Find indices to adjust
        if diff > 0:
            # Need more sum
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            # Need less sum
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

def trust_region_step(h, grad, hessian, tr_radius, target_sum):
    """Solve trust-region subproblem and return step."""
    n = len(h)
    # Use simple approximation for Hessian as identity
    H = np.eye(n)
    
    # Solve: min_{s} grad^T s + 0.5 s^T H s such that ||s|| <= tr_radius
    # This is a quadratic program, we can solve via projection
    try:
        # Simple gradient descent step within trust region
        step = -grad
        step_norm = np.linalg.norm(step)
        if step_norm > tr_radius:
            step = step * tr_radius / step_norm
        
        # Project to feasible set
        candidate = h + step
        candidate = project_to_feasible(candidate, target_sum)
        return candidate - h
    except:
        # Fall back to small random step
        return np.random.normal(0, tr_radius/10, n)

    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    tr_radius = 0.2
    tr_radius_min = 0.1
    tr_radius_max = 1.0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Compute initial gradient - for now we'll compute this numerically
    # This is the gradient of the max correlation term
    def compute_gradient(h, dx):
        # This is a heuristic approximation of the gradient
        # We estimate dC5/dh[i] ≈ (C5(h+δ) - C5(h-δ)) / (2δ)
        eps = 1e-4
        grad = np.zeros_like(h)
        base_c5 = evaluate(h, dx)
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            h_plus = project_to_feasible(h_plus, n_points/2.0)
            h_minus = project_to_feasible(h_minus, n_points/2.0)

            c_minus = evaluate(h_minus, dx)
            grad[i] = (c_plus - c_minus) / (2 * eps)

        iteration += 1
        
        # Compute gradient

        try:

            h_trial = project_to_feasible(h_trial, n_points / 2.0)
            
            # Evaluate trial point
            c5_trial = evaluate(h_trial, dx)
            
            # Accept the update if it improves
            if c5_trial < best_c5:
