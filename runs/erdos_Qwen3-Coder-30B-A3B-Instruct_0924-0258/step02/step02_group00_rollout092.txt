import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx):
    """Perform one Frank-Wolfe step"""
    # Compute all correlation values
    corr = np.correlate(h, 1 - h, mode='full')
    # Find the indices where maximum overlap occurs
    max_indices = np.where(corr == np.max(corr))[0]
    
    # Use the first max index for simplicity, but could use average
    max_shift = max_indices[0] - (n_points - 1)
    
    # Compute the subgradient for the maximum overlap
    # For shift k, the gradient for h[i] is (1 - h[i-k]) - h[i]
    # But we're considering the full correlation
    # So we consider the gradient as it would affect the max
    subgradient = np.zeros(n_points)
    for i in range(n_points):
        k = (i - (i - max_shift) % n_points) % n_points
        this_shift = k - (n_points - 1)
        if abs(this_shift) == abs(max_shift):
            # This is the active shift
            # Analytical subgradient: derivative of overlap w.r.t. h[i]
            # This is (1 - h[(i - k) % n]) - h[i], but with original form
            # Let's just compute a finite difference for stability
            # For now, we simplify and take this as contribution from overlap
            # For maximizing the correlation symmetrically:
            # Gradient w.r.t h[i] of h[i](1-h[i+k]) is (1 - h[i+k]) - h[i]
            #  h[i+k] has same role as our current value in correlation
            # Since that is fixed, and we vary h, we get:
            # For a fixed shift k: (1 - h[i-k]) - h[i]  but extend both ways
            offset = (i - max_shift) % n_points
            subgradient[i] = (1 - h[offset]) - h[i]
    
    # Linear optimization over feasible region
    # Minimize g ⋅ s subject to sum(s) = 0, 0 <= s <= 1
    # where g is the subgradient
    # solve: min sum(g_i * s_i) s.t. sum(s_i) = 0, 0 <= s_i <= 1
    # A simplified approach: find s such that g.s is minimal
    # The most straightforward is taking s with negative g (0 if g > 0) after normalization
    s = np.zeros(n_points)
    
    # Simple direction that aims to reduce the maximum overlap
    # Let's take s = sign(g) * from UNIFIED approach
    # For Frank-Wolfe, find s s.t. s = argmin s^T g subject to constraints
    # Take feasible descent direction
    # The standard Frank-Wolfe direction is simple:
    # Take a vertex of the feasible region (a feasible direction) that minimizes g^T s
    # For this case, s = argmin g^T s subject to sum s = 0, 0<=s<=1
    # Just find the direction of negative gradient from feasible vertex
    # By counting, find the minimum inner prod with gradient
    
    # Compute the s that corresponds to minimising g.s
    # Put s = -sign(g) + shifted to satisfy constraint sum(s) = 0
    # Flip negative g to positive for initial direction
    direction = np.sign(subgradient)
    s = -direction  # We want reduce negative so reverse sign
    
    # Best descent direction in feasible polyhedron with sum=0
    # Use simple iterative method:
    # 1) Pick s where negative g[i] to increase h[i] (but still feasible)
    # 2) Pick s where positive g[i] to decrease h[i] 
    # But we use to project onto feasible set, not full search
    
    # Average gradient method:
    # For Frank-Wolfe: find s such that g^T s is minimized, inside subset
    # S = {x | sum(x) = 0, 0 <= x <= 1}
    # Point is to find the vertex achieving the lowest inner product
    # since s in the constraint polyhedron with sum=0 is like:
    # The gradient is g, we want min g^T s subject to 0 ≤ s_i ≤ 1, sum(s) = 0
    # This is not standard, most of usual is rank-1 constraints.
    # However, setting s[i] = (g[i] < 0 ? 1 : 0) is too approximate
    # We scan all negative components and set s_i = 1, then subtract
    # so that sum_s = 0, then clip & normalize?
    
    # Very simplified: just solve for descent  with small but feasible next step  
    # This is frank-wolfe descent using unit direction:  
    dir_neg = np.where(subgradient < 0)[0]
    dir_pos = np.where(subgradient > 0)[0]
    s = np.zeros(n_points)
    if len(dir_neg) > 0:
        # Try increasing those that decrease g
        s[dir_neg] = 1.0
        s[dir_pos] = -1.0 * len(dir_neg) / len(dir_pos)  # approximate sum=0
        s = s / np.sum(s) * (n_points / 2)  # now sum =n/2, fit scaling approx
        
        # Alternative: set fixed number to 1, 0 elsewhere
    
    # set vertices:
    # and abs(sum) = 0, so to get back to feasible subpoint
    # Or simply, argmin g^T s s.t. sum=s, 0<=s_i<=1
    # We could brute-force or solve this in a simple way:
    # Solve it using feasible version (like l1 ball) but specialized one.
    # For now, a simple direct approximation
    
    # Compute best likely descent: 
    # Let s[i] = -sign(g[i])
    # then it's not feasible (sum not 0), so move towards vertices
    # start from zero trajectory update s = vertex that minimizes g^T s
    
    # Good approximation: Find vertex of feasible region
    # trick: reduce to l1-ball flavor
    # S = { s | sum(s)=0, 0 <= s_i <= 1 }
    # But there's a simpler version: choose a subgradient direction and find feasible s that moves
    # We will directly try to define s_i = 1 or -1
    # that is practical for gradient descent on this specific relaxed problem
    # we try building one such "slack vector"
    
    # Better heuristic Frank-Wolfe update:
    # Initialize s = -sign(g) - found such that sum = 0, 0 <= s <= 1
    # By trial and error approach
    # tries mixing one large drop with some additions to satisfy sum constraint
    s = np.zeros_like(h)
    s_weights = np.copy(subgradient)
    
    # Use a simple greedy doing first try with proper signs from gradient
    # and reduce the total to be 0, do i.e. effective subgradient-wolfe update:
    
    # projection into { g^T s | sum(s)=0, 0 <= s <=1 } of s=0 vector
    # Take some bit of vs -= sign(g_i) to make sum constraint match
    
    # Find direction most likely to reduce g(s) on shifting one point
    # approximation: assume we bring positive where g negative and vice versa, with weights
    
    # Direct: set s to correspond to direct gradient(descend direction) with accounting of sum=0:
    
    # Choose s to minimize g.T s:
    # subject to sum(s) = 0, 0 <= s_i <= 1
    
    # Have good idea of specific boundary vector
    # Assign values that shut down many g components, i.e. try to go toward negative gradient
    # So let's take s such that components where g < 0 are incremented and 
    # where g > 0 are decremented, so that each one happens maximally
    
    # On sum computation
    # Instead of perfect adjustment, just use scaled deviation adjustment 
    # Scaled imperfection
    # Update 1 by gradually shifting among indices
    
    sum0 = 0
    
    # More detailed box-constrained approach:
    # Find s where g.T s is minimal and sum(s) = 0
    # Apply simplification:
    
    # For constraint set: 0 <= s_i <= 1, sum(s) = 0
    # What we want is minimizing g^T s: this becomes finding a
    # vertex of the convex hull that will reduce the linear form from g
    
    # Solution idea: take s = -sign(g) and adjust to satisfy sum = 0
    # It's really about scaling to match.
    # Or assume: find a feasible s minimizing g^Ts
    # Since s has zero sum, positive values for negative g, negative for positive
    # This heuristics is Arnold's
    s = -np.sign(subgradient)
    
    # Normalize it to sum = 0
    pos_sum = np.sum(s[s > 0])
    neg_sum = -np.sum(s[s < 0])
    # We want equal sum but with flat sums
    # Apply sum_fix so that sum equals 0 now
    if pos_sum > neg_sum:
        # scale down positives
        s[s > 0] = s[s > 0] / pos_sum * neg_sum
    elif neg_sum > pos_sum:
        s[s < 0] = s[s < 0] / neg_sum * pos_sum
    
    return s

def optimize_with_frank_wolfe(h_start, n_points, dx, time_limit):
    start_time = time()
    
    h_current = h_start.copy()
    current_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_c5 = current_c5
    
    max_iter = 100  # Try 100 iterations max
    for i in range(max_iter):
        if time() - start_time > time_limit * 0.98:
            break
            
        # Step size based on iteration
        gamma = 0.1 # initgamma * (1 + i / max_iter)
        
        # Frank Wolfe step
        s = frank_wolfe_step(h_current, n_points, dx)
        
        # Update h
        h_new = h_current + gamma * (s - h_current)
        
        # Project to feasible set
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    
    return best_h, best_c5
