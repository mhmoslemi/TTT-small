import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def get_gradients(h, n_points, dx):
    """Get gradient of smooth max approximation w.r.t. h"""
    corr = np.correlate(h, 1 - h, mode='full')
    alpha = 100
    exp_corr = np.exp(alpha * corr)
    sum_exp = np.sum(exp_corr)
    grad_corr = alpha * exp_corr / sum_exp
    
    # Compute gradient w.r.t. h
    grad = np.zeros_like(h)
    for i in range(n_points):
        # Jacobian of correlation entry w.r.t. h[i]
        # For k=0 to 2*n_points-2, correlation[i] = sum_j h[j] * (1 - h[(j+k)%n_points])
        # Partial derivative w.r.t. h[i] is what j=i contributes
        # Actually compute for all shifts
        for k in range(2*n_points - 1):
            j = (i - k) % n_points
            if j < n_points:
                grad[i] += grad_corr[k] * (1 - h[j]) * dx
    
    return grad

def optimize_with_gradient_descent(h_start, n_points, dx, time_limit):
    """Optimize using gradient descent with projection"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    learning_rate = 0.01
    decay_rate = 0.99
    max_iter = 500
    
    for i in range(max_iter):
        if time() - start_time > time_limit:
            break
            
        # Compute current gradient
        try:
            grad = get_gradients(best_h, n_points, dx)
            # Gradient step
            h_new = best_h - learning_rate * grad
            # Project to feasible region
            h_new = project_to_feasible(h_new, n_points)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        except Exception as e:
            pass  # Skip this iteration
            
        # Adjust learning rate
        if i % 50 == 0 and i > 0:
            learning_rate *= decay_rate
            
    return best_h, best_c5

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [30, 35, 40, 45]
    
    for i, n_points_try in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            # Interpolate to new resolution
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            
        # Multiple restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Small random perturbation
            noise = np.random.normal(0, 0.05, n_points)
            h_restart = project_to_feasible(h_start + noise, n_points)
            
            # Run gradient optimization
            remaining_time = budget_s - (time() - start_time)
