import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40, gamma_range=(0.1, 0.9)):
    """Frank-Wolfe optimization step for the Erdős problem"""
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_indices = np.where(corr == np.max(corr))[0]
        max_k = max_indices[0]  # Take first one if multiple
        
        # Compute subgradient at maximum overlap
        subgradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            subgradient[i] = (1 - h[j]) - h[i]  # dC/dh_i for shift k
        
        # Solve linear optimization: min_s <g, s> s.t. sum(s) = 0, 0 <= s <= 1
        # This is a linear program with simple constraints.
        # We take the direction that decreases the objective most.
        # s = argmin_g^T s s.t. sum(s) = 0, 0 <= s <= 1
        # Projected gradient descent-like step
        s = np.zeros(n_points)
        for i in range(n_points):
            if subgradient[i] < 0:
                s[i] = 1
            elif subgradient[i] > 0:
                s[i] = 0
            else:
                s[i] = h[i]
        
        # Ensure sum(s) = 0 by shifting
        s_sum = np.sum(s)
        if abs(s_sum) > 1e-10:
            s = s - s_sum / n_points
            
        # Ensure s is feasible
        s = np.clip(s, 0, 1)
        
        # Line search to select step size
        best_gamma = gamma_range[0]
        best_h_trial = h.copy()
        best_c5_trial = best_c5
        
        for gamma in np.linspace(gamma_range[0], gamma_range[1], 10):
            h_trial = h + gamma * (s - h)
            h_trial = project_to_feasible(h_trial, n_points)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5_trial:
                best_c5_trial = c5_trial
                best_h_trial = h_trial.copy()
                best_gamma = gamma
            
        h = best_h_trial.copy()
        if best_c5_trial < best_c5:
            best_c5 = best_c5_trial
            best_h = h.copy()
            
        if iteration % 10 == 0:
            # Check for improvement
            pass
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        # Default starting point
        n_points = 25
        h_start = np.ones(n_points) * 0.5
        # Interleave 0s and 1s
        for i in range(n_points):
            if i % 2 == 0:
                h_start[i] = 0.0
            else:
                h_start[i] = 1.0
        # Normalize to sum to n_points/2
        h_start = project_to_feasible(h_start, n_points)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    resolutions = [25, 30, 35]
    
    for res_idx, n_res in enumerate(resolutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_res
        if n_res != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_res), np.linspace(0, 2, len(h_start)), h_start)
            h_start = project_to_feasible(h_interp, n_res)
            n_points = n_res
            
        # Run FW optimization with restarts
        num_restarts = 3 if res_idx == 0 else 1
        
        for restart in range(num_restarts):
            current_h = h_start.copy()
            if restart > 0:
                # Add slight noise to restart
                noise = np.random.normal(0, 0.01, n_points)
                current_h = project_to_feasible(current_h + noise, n_points)
                
            h_opt, c5_opt = frank_wolfe_step(current_h, n_points, dx, max_iter=30)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                if res_idx == len(resolutions) - 1:
                    # Final refinement with more iterations
                    h_opt_refined, c5_opt_refined = frank_wolfe_step(h_opt, n_points, dx, max_iter=50)
                    if c5_opt_refined < best_c5:
                        best_h = h_opt_refined.copy()
                        best_c5 = c5_opt_refined
                        
        h_start = best_h.copy()
        
    return (list(best_h), best_c5, n_points)
