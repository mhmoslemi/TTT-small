import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if abs(diff) > 1e-10:
        if diff > 0:
            idxs = np.argsort(h)
            for i in range(len(idxs)-1, -1, -1):
                if diff <= 0:
                    break
                j = idxs[i]
                if h[j] < 1:
                    update = min(diff, 1 - h[j])
                    h[j] += update
                    diff -= update
        else:
            idxs = np.argsort(h)
            for i in range(len(idxs)):
                if diff >= 0:
                    break
                j = idxs[i]
                if h[j] > 0:
                    update = min(-diff, h[j])
                    h[j] -= update
                    diff += update
    
    return h

    pop_size = 30
    elite_size = pop_size // 10  # 10% elitism
    mutation_strength = 0.05 * np.ones(n_points)
    beta = 0.2
    max_stagnation = 50
    stagnation_count = 0
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = project_to_feasible(h, n_points/2.0)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate(h, 2.0/n_points) for h in population]
    
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    generation = 0
    while time.time() - start_time < budget_s and stagnation_count < max_stagnation:
        # Sort population by fitness
        indices = np.argsort(fitness)
        sorted_population = [population[i] for i in indices]
        sorted_fitness = [fitness[i] for i in indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Selection: keep elites
        elites = sorted_population[:elite_size]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate mutants for rest of population
        while len(new_population) < pop_size:
            # Tournament selection
            parent_idx = np.random.randint(0, pop_size)
            # Compare with random individual
            other_idx = np.random.randint(0, pop_size)
            if fitness[parent_idx] > fitness[other_idx]:
                parent_idx = other_idx
            
            parent = sorted_population[parent_idx]
            # Mutate
            mutant = parent + np.random.normal(0, mutation_strength, n_points)
            mutant = project_to_feasible(mutant, n_points/2.0)
            new_population.append(mutant)
        
        # Replace population
        population = new_population
        
        # Recompute fitness
        fitness = [evaluate(h, 2.0/n_points) for h in population]
        
        # Adapt mutation strength (self-adaptation)
        mutation_strength *= np.exp(beta * np.random.normal(0, 1, n_points))
        mutation_strength = np.clip(mutation_strength, 0.001, 0.2)
        
        generation += 1
    
    return (best_h.tolist(), float(best_c5), n_points)
