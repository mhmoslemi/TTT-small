import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = project_to_feasible(h_start, n_points)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    # Moments for Adam
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    t = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Optimization loop
    while time() - start_time < budget_s * 0.95:
        t += 1
        
        # Compute full correlation to find max lag
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k = np.argmax(corr)
        
        # Compute gradient for lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h_current[j]) - h_current[i]
        grad *= dx
        
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        # Update step
        h_current -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible set
        h_current = project_to_feasible(h_current, n_points)
        
        # Evaluate and track best
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Restart if no improvement for too long
        if no_improvement_count > max_no_improvement:
            h_current = project_to_feasible(
                h_current + np.random.normal(0, 0.05, n_points), 
                n_points
            )
            no_improvement_count = 0
            
        # Try higher resolution if needed
        if time() - start_time > budget_s * 0.7 and n_points < 100:
            n_points_new = min(n_points * 2, 100)
            if n_points_new > n_points:
                x_orig = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, n_points_new)
                h_interp = np.interp(x_new, x_orig, best_h)
                h_current = project_to_feasible(h_interp, n_points_new)
                n_points = n_points_new
                dx = 2.0 / n_points
                m.fill(0)
                v.fill(0)
                t = 0
    
    return (list(best_h), best_c5, n_points)
