import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    """Gradient of smooth max"""
    exp_x = np.exp(alpha * x)
    return exp_x / (np.sum(exp_x))

def compute_overlap_gradient(h, n_points, dx):
    """Compute gradient of the overlap integral with respect to h"""
    # Correlation values for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    
    # Gradient for the max correlation term
    grad = np.zeros_like(h)
    # The full correlation is of length 2*n-1, but we only care about indices 0 to n-1
    # because h is of length n and corr[i] is h convolved with (1-h) at shift i
    # Taking derivative of correlation with respect to h:
    for i in range(n_points):
        # The i-th element of the correlation consists of contributions from 
        # h[j] * (1 - h[(i+j)%n]) for all j.
        # We only need to iterate the valid shifts
        for j in range(n_points):
            idx = (i + j) % n_points
            if idx == max_idx:
                grad[i] += 1 - h[idx]
    return grad * dx

def optimize_step(h_start, n_points, dx, max_iter=300):
    """Optimize a single step with gradient descent"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Adaptive learning rate
    lr = 0.01
    decay = 0.99
    
    for i in range(max_iter):
        # Compute gradient for the overlap term
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        
        # Gradient of the correlation that achieves the max
        grad = np.zeros_like(h)
        for k in range(n_points):
            j = (max_idx - k) % n_points
            if 0 <= j < n_points:
                grad[k] += 1 - h[j]
        
        # Update
        h = h - lr * grad
        h = project_to_feasible(h, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        lr *= decay
    
    return best_h, best_c5

    # Optimization loop
    iter_count = 0
    while time() - start_time < budget_s * 0.95:
        # Try various resolutions around current
        resolutions = [25, 30, 35, 40]
        for i in range(len(resolutions)):
            if time() - start_time > budget_s * 0.95:
                break
            n_try = resolutions[i]
            if n_try == n_points:

            # Interpolate
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_orig, h_start)
