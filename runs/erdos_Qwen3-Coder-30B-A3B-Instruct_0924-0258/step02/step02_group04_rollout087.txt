import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m):
    """Initialize a simplex with m particles"""
    particles = []
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        # Otherwise create a basic pattern
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_to_feasible(h0, n_points)
    particles.append(h0)
    
    # Generate m-1 additional points by adding small random perturbations
    for _ in range(m - 1):
        h_pert = h0 + np.random.normal(0, 0.05, n_points)
        h_pert = project_to_feasible(h_pert, n_points)
        particles.append(h_pert)
    
    return np.array(particles)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia_w = 0.7
    c1 = c2 = 2.0
    shake_threshold = 50
    lbfgs_interval = 200
    min_swarm_size = 10
    max_swarm_size = 50
    
    # Initialize particles
    particles = initialize_simplex(n_points, swarm_size)
    velocities = np.zeros_like(particles)
    
    # Evaluate initial particles
    fitness = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Track personal bests
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    
    # Track global best
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # Track stagnation
    last_improvement = 0
    current_swarm_size = swarm_size
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update velocities and positions
        for i in range(len(particles)):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
        
        # Evaluate new fitness
        fitness = np.array([evaluate_erdos_solution(p, dx) for p in particles])
        
        # Update personal bests
        for i in range(len(particles)):
            if fitness[i] < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = fitness[i]
        
        # Update global best
        new_gbest_idx = np.argmin(fitness)
        if fitness[new_gbest_idx] < gbest_fitness:
            gbest = particles[new_gbest_idx].copy()
            gbest_fitness = fitness[new_gbest_idx]
            last_improvement = iteration
        
        # Shaking mechanism
        if iteration - last_improvement > shake_threshold:
            for i in range(len(particles)):
                # Add noise to break local optima
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] += noise
                particles[i] = project_to_feasible(particles[i], n_points)
            last_improvement = iteration
            
            # Expand swarm if needed
            if current_swarm_size < max_swarm_size:
                extra_particles = current_swarm_size // 2
                for _ in range(extra_particles):
                    h_new = gbest + np.random.normal(0, 0.1, n_points)
                    h_new = project_to_feasible(h_new, n_points)
                    particles = np.vstack([particles, h_new])
                    velocities = np.vstack([velocities, np.zeros(n_points)])
                    pbest = np.vstack([pbest, h_new])
                    pbest_fitness = np.append(pbest_fitness, evaluate_erdos_solution(h_new, dx))
                current_swarm_size += extra_particles
                
        # Periodic L-BFGS refinement
        if iteration % lbfgs_interval == 0 and iteration > 0:
            top_indices = np.argsort(fitness)[:10]  # Top 10 particles
            for idx in top_indices:
                # Local refinement
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                
                try:
                    res = minimize(obj, particles[idx], method='L-BFGS-B', 
                                bounds=bounds, constraints=cons, options={'maxiter': 10})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_fitness = evaluate_erdos_solution(refined, dx)
                        if refined_fitness < fitness[idx]:
                            particles[idx] = refined
                            fitness[idx] = refined_fitness
                            if refined_fitness < pbest_fitness[idx]:
                                pbest[idx] = refined.copy()
                                pbest_fitness[idx] = refined_fitness
                            if refined_fitness < gbest_fitness:
                                gbest = refined.copy()
                                gbest_fitness = refined_fitness
                except:
                    pass
        
        # Ensure the best solution stays valid throughout
        gbest = project_to_feasible(gbest, n_points)
        gbest_fitness = evaluate_erdos_solution(gbest, dx)
    
    # Final refinement of the best solution
    def final_obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    try:
        res = minimize(final_obj, gbest, method='L-BFGS-B', 
                    bounds=bounds, constraints=cons, options={'maxiter': 20})
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < gbest_fitness:
                gbest = final_h
                gbest_fitness = final_c5
    except:
        pass
    
    return (list(gbest), gbest_fitness, n_points)
