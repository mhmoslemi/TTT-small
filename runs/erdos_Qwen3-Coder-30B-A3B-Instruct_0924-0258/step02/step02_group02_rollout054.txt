import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_min_correlation_and_grad(h_vec):
        # Compute all correlations with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        min_idx = np.argmin(corr)
        min_corr = corr[min_idx]
        
        # Find the corresponding shift for the minimum correlation
        # Full correlation has length 2*n_points - 1
        shift = min_idx - (n_points - 1)
        
        # Compute gradient of this correlation term w.r.t. h
        grad = np.zeros_like(h_vec)
        # In convolution, the gradient w.r.t. h[i] of corr with shift k
        # is 1-h[i+k] when i+k < n_points and 1-h[i-k] when i >= k
        for i in range(n_points):
            if i + shift >= 0 and i + shift < n_points:
                grad[i] += 1 - h_vec[i + shift]
            if shift >= 0 and i - shift >= 0:
                grad[i] += 1 - h_vec[i - shift]
        return min_corr, grad

    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    iteration = 0
    learning_rate = 0.01
    no_improve_count = 0
    max_no_improve = 500
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current minimum correlation and its gradient
        min_corr, grad = compute_min_correlation_and_grad(h)
        
        # Update h using gradient ascent to increase min correlation
        h = h + learning_rate * grad
        
        # Project h to maintain constraints
        h = project_h(h)
        
        # Check for improvement
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Adjust learning rate if no improvement for a while
        if no_improve_count > 100:
            learning_rate *= 0.5
            no_improve_count = 0
            
        iteration += 1
        if no_improve_count > max_no_improve:
            break
            
    return (best_h.tolist(), best_c5, n_points)
