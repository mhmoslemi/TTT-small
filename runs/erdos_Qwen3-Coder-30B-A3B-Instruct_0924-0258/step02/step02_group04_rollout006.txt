import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = (n_points / 2) / current_sum
        h = h * scale
    return h

def initialize_population(n_points, population_size):
    """Initialize population using initial_h_values if available"""
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            h_base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_base)), h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    population = [project(h_base.copy(), n_points)]
    
    # Add variations
    for _ in range(population_size - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h = project(h_base + noise, n_points)
        population.append(h)
    
    return population

def mutate(h, mutation_rate, sigma=0.1):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < mutation_rate:
        indexes = np.random.choice(len(h), size=max(1, int(len(h) * 0.2)), replace=False)
        h[indexes] += np.random.normal(0, sigma, len(indexes))
        h = project(h, len(h))
    return h

def crossover(h1, h2, crossover_rate=0.7):
    """Perform two-point crossover"""
    if np.random.rand() < crossover_rate:
        idx1, idx2 = sorted(np.random.choice(len(h1), 2, replace=False))
        offspring = h1.copy()
        offspring[idx1:idx2] = h2[idx1:idx2]
        return project(offspring, len(offspring))
    return h1.copy()

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use smaller n_points for faster optimization
    n_points = 30
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    stagnation_limit = 200
    stagnation_count = 0
    best_fitness = float('inf')
    best_individual = None
    
    # Initialize population
    population = initialize_population(n_points, population_size)
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Selection, crossover, mutation
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            offspring = crossover(parent1, parent2)
            offspring = mutate(offspring, mutation_rate)
            new_population.append(offspring)
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Replace worst individuals
        # Combine old and new populations and select top half
        combined_population = population + new_population
        combined_fitnesses = fitnesses + new_fitnesses
        indices = np.argsort(combined_fitnesses)
        population = [combined_population[i] for i in indices[:population_size]]
        fitnesses = [combined_fitnesses[i] for i in indices[:population_size]]
        
        # Update best
        current_best_fitness = min(fitnesses)
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive mutation rate
        if stagnation_count > 50:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        if stagnation_count > stagnation_limit:
            # Reset population
            population = initialize_population(n_points, population_size)
            fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
            stagnation_count = 0
            mutation_rate = 0.2
            
        generation += 1
    
    if best_individual is None:
        # Fallback to simplest approach
        best_individual = initialize_population(n_points, 1)[0]
        best_fitness = evaluate_erdos_solution(best_individual, dx)
    
    return (list(best_individual), best_fitness, n_points)
