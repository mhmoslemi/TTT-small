import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_overlaps(h_vec):
        # Compute all overlaps for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return corr

    def evaluate_objective(h_vec):
        return compute_c5(h_vec)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    # Use alternating optimization approach
    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 500
    patience_counter = 0
    prev_best_c5 = float('inf')
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute overlaps for current h
        overlaps = compute_overlaps(h)
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        # Create objective that penalizes exceeding max_overlap
        def penalty_objective(h_vec):
            # Compute current overlaps
            curr_overlaps = compute_overlaps(h_vec)
            # Penalize violations
            penalty = np.sum(np.maximum(0, curr_overlaps - max_overlap))
            return compute_c5(h_vec) + 1e6 * penalty
        
        # Try to reduce max overlap by optimizing for smaller values
        # Randomly perturb h and recompute
        noise = 0.0005 * np.random.normal(0, 1, n_points)
        h_candidate = h + noise
        h_candidate = project_h(h_candidate)
        
        candidate_c5 = compute_c5(h_candidate)
        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = h_candidate.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Slightly perturb to escape local minima
        noise = 0.001 * np.random.normal(0, 1, n_points)
        h = h + noise
        h = project_h(h)
        
        # Adapt learning rate if no improvement for a while
        if patience_counter > patience:
            learning_rate *= 0.9
            patience_counter = 0

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
