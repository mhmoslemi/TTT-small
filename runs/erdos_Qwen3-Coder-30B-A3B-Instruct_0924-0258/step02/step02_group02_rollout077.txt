import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute all autocorrelations
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Only consider valid shifts
        valid_corr = corr[len(h_vec)-1:]
        return np.min(valid_corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_gradient(h_vec, shift_idx):
        # Gradient of correlation at shift_idx w.r.t. h
        grad = np.zeros_like(h_vec)
        # Only h[i] contributes to correlation at shift idx for terms h[i]*h[i+shift_idx]
        # So we add 1 to grad[i+shift_idx] each time h[i] is used
        
        for i in range(len(h_vec)):
            if i + shift_idx < len(h_vec):
                grad[i + shift_idx] += h_vec[i]
            # Also for backwards terms in case of symmetric computation
            if i - shift_idx >= 0:
                grad[i - shift_idx] += h_vec[i]
        
        return grad

    def optimize_step(h_vec):
        # Find minimum autocorrelation and its shift
        corr = np.correlate(h_vec, h_vec, mode='full')
        valid_corr = corr[len(h_vec)-1:]
        min_shift = np.argmin(valid_corr)
        min_corr = valid_corr[min_shift]
        
        # Compute gradient for this shift
        grad = compute_gradient(h_vec, min_shift)
        
        # Perform gradient ascent (maximize the minimum autocorrelation)
        learning_rate = 0.01
        h_new = h_vec + learning_rate * grad
        
        # Project back into feasible region
        h_new = project_h(h_new)
        
        return h_new

    # Initial computation
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Try multiple gradient steps
        h_temp = h.copy()
        for _ in range(5):
            h_temp = optimize_step(h_temp)
        
        # Evaluate the new solution
        new_c5 = compute_c5(h_temp)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_temp.copy()
            last_improvement = iteration
            
        # Update h for next round
        h = h_temp.copy()
        
        # Add noise occasionally to escape local optima 
        if iteration % 50 == 0:
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
            
        # Early termination if no improvement in a while
        if iteration - last_improvement > 1000:
            break
            
        iteration += 1
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
