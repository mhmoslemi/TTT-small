import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    h = project_feasible(h, n_points)
    dx = 2.0 / n_points
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.98:
        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Find the lag with maximum overlap
        k = np.argmax(corr)
        
        # Compute gradient for this specific lag k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h[j]) - h[i]
        
        grad *= dx
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        # Bias correction
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        # Update h
        h -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h = project_feasible(h, n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        
        # Update best if improved
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            last_improvement = iteration
        
        iteration += 1
        
        # Restart if no improvement for a while
        if iteration - last_improvement > 1000:
            h = project_feasible(np.random.normal(best_h, 0.05), n_points)
            last_improvement = iteration
            # Reset Adam moments
            m.fill(0)
            v.fill(0)
    
    return (list(best_h), best_c5, n_points)
