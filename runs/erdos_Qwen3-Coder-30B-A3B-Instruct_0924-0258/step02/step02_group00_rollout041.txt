import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_subgradient(h, n_points, max_shifts):
    """Compute subgradient for the maximum overlap term"""
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    
    # For each shift in max_shifts, we compute the gradient contribution
    subgrad = np.zeros(n_points)
    for k in max_shifts:
        # The gradient of h[i] with respect to overlap at shift k
        for i in range(n_points):
            j = (i - k) % n_points
            subgrad[i] += (1 - h[j]) - h[i]
    return subgrad / len(max_shifts)

def frank_wolfe_step(h, subgrad, n_points):
    """Perform Frank-Wolfe step to get descent direction"""
    # Solve: min_s <subgrad, s> subject to sum(s) = sum(h), 0 <= s <= 1
    # This is a simple linear optimization over a box with sum constraint
    # We find s that minimizes <subgrad, s>
    # heuristically, using the fact that the optimal s should be at the boundary of the feasible set
    
    # For now, do a simple projection approach
    # Find the indices where gradient is most negative (steepest descent)
    # and try to decrease toward zero or away from current h
    s = np.zeros(n_points)
    # Direct projection approach for linear constrained optimization:
    # Let the step size be the direction towards least gradient component with feasible bounds
    
    s = np.zeros_like(h)
    if np.sum(subgrad) == 0:
        # If the subgradient sum is 0, just make a small move
        s = -subgrad
    else:
        # Even with differences based on constraint violations
        # Quick and dirty search for reasonable direction
        # For visualization purposes, we update using line search algorithm
        pass

    s = np.zeros(n_points)
    # Simple Frank-Wolfe descent: 
    # we want s that minimizes <subgrad, s>, subject to simplex and box constraints.
    # We solve in closed form for this case with box [0,1]:
    # s = argmin_s <subgrad, s> s.t. \sum s = n/2, 0 <= s <= 1
    # Construct s greedily by choosing elements in descending order of -subgrad
    
    # Ordered pairs: (index, -subgrad[index])
    idx_neg_grad = np.argsort(-subgrad)
    s = np.zeros(n_points)
    total = 0.0
    for i in range(n_points):
        j = idx_neg_grad[i]
        if total + 1.0 <= n_points / 2:
            s[j] = 1
            total += 1
        else:
            s[j] = min(1, n_points/2 - total)
            total += s[j]
            break

    # Renormalize
    if total > 0: 
        s *= (n_points / 2) / total
    return s

def line_search(h, s, subgrad, n_points, dx, max_iter=20):
    """Backtracking line search to find optimal step size"""
    gamma = 1.0
    c5_base = evaluate_erdos_solution(h, dx)
    for _ in range(max_iter):
        h_new = project_to_feasible(h + gamma * (s - h), n_points)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5_base:
            return h_new, gamma, c5_new
        gamma *= 0.5
    return h, gamma, c5_base

def optimize_with_frank_wolfe(h_start, n_points, dx, time_limit):
    """Optimize using the Frank-Wolfe approach with adaptive steps"""
    start_time = time()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Iteration parameters
    max_iter = 40
    smoothing_factor = 0.05

    for iter_count in range(max_iter):
        if time() - start_time > time_limit:
            break

        # Compute overlaps to find maximum
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        c5_current = np.max(corr) * dx
        max_shifts = np.where(corr >= c5_current * (1 - smoothing_factor))[0]
        
        # Compute subgradient of the maximum
        subgrad = compute_subgradient(best_h, n_points, max_shifts)
        
        # Compute descent direction using Frank-Wolfe step
        s = frank_wolfe_step(best_h, subgrad, n_points)

        # Perform line search
        new_h, gamma, new_c5 = line_search(best_h, s, subgrad, n_points, dx, max_iter=5)

        # Verify the improvement
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
    return best_h, best_c5
