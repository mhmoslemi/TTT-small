import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_autocorr(h_vec):
        # Compute all autocorrelations for all shifts
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Only keep the valid shifts (from -n+1 to n-1)
        mid = len(corr) // 2
        autocorr = corr[mid:mid+n_points]
        # Simulate the actual overlap as described in C5 definition
        # This is a bit tricky because it uses (1-h) and should compute the 
        # overlap of h with 1-h, but correlations are based on h
        # So we compute correlations of h with 1-h, which is a bit different semantic.
        # Let's modify to compute actual integral formulation.
        # The actual problem is about computing: int h(x) * (1 - h(x+k)) dx 
        # So we correlate h with 1-h, for all shifts k positions.
        # So we should compute: correlation of h with 1-h 
        # autoco = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Then we just pick max value, but we compute it from the inflated 
        # arrays and then map back, so we don't double-count.
        # Simpler is to compute manually.
        autocorr_all = []
        for k in range(-n_points+1, n_points):
            # Compute integral of h(x) * (1 - h(x+k)) for shift k
            if k >= 0:
                # Sample at [0, n_points-k-1] with h[i]*h[i+k] contribution
                integral = np.sum(h_vec[:n_points-k] * (1 - h_vec[k:]))
            else:
                # k < 0: sample at [0, n_points+k-1] with h[i]*h[i-k] contribution  
                integral = np.sum(h_vec[-k:] * (1 - h_vec[:-k]))
            integral *= dx
            autocorr_all.append(integral)
        return np.array(autocorr_all)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def gradient_ascent_step(h_vec, steps=100):
        for _ in range(steps):
            # Compute all autocorrelation integrals
            corr = compute_autocorr(h_vec)
            # Get the minimum value and its index
            min_idx = np.argmin(corr)
            min_val = corr[min_idx]
            
            # Now we want to increase the minimum C_k value
            # We want to compute the gradient of min_val with respect to h
   
            # Let's simplify approach. Instead of gradient of correlation, 
            # we will perform a direct update that tries to increase the leaking 
            # overlapping. Let's try the algorithm that directly targets the worst case.

            # This simplification helps us go faster.
            
            # k corresponds to index min_idx - (n_points-1) 
            # i.e. shift k = min_idx - (n_points-1)
            shift = min_idx - (n_points-1)
            
            # We want to increase overlap at shift shift
            # We're mentally doing: minimize int h(x) * (1-h(x+k)) dx = maximize int h(x) * h(x+k) dx
            # But we're not maximizing, we're adjusting h to decrease the minimum nonoverlap
            
            # It's easier to "flip" shift
            # For shift k, we're computing h[i]*h[i+k] for all valid i.  
            # So we can compute gradient of this in terms of the influences
            
            # If k is nonnegative and we take h[i] * h[i+k], then for h[i] the contribution is h[i+k], 
            # and for h[i+k] the contribution is h[i], all for 1 to n-k-1.
            # Use this for updating 
            # we want to compute update direction that increases the corresponding correlation value
            # This is effect of changing h[i]. The influence of h[i] on the sum 
            # Sum_i h[i] * h[i+k] (k fixed)
            # gradient wrt h[i]: h[i+k] (if i+k < n and i<n)  and h[i-k]  (if i-k >= 0)
            
            # So for each sample point, we compute influence:
            grad = np.zeros_like(h_vec)
            if shift >= 0:
                # h[i] * h[i+shift] - for this shift, for h[i] when i+shift < n we add h[i+shift]
                # Problem: this is not computing the right gradient
                # Directly from definition:
                # A = sum_i h[i] * h[i+shift] 
                # gradient_wrt_h[i] = h[i+shift] (provided i+shift < n and i<n)
                # Also when i >= shift: it can also be h[i-shift] for h[i-shift]
                for i in range(n_points):  # i is now loop variable in original paper
                    if i + shift < n_points:
                        grad[i] += h_vec[i+shift]  # added contribution of h[i+shift] to loss at h[i]
                    if i >= shift:              # we also consider from mirrored j->j-shift when j>=shift
                        grad[i] += h_vec[i-shift]
            else:
                # shift < 0, likely shift = -k for k positive
                # So we think of h[i+k]*h[i]. For i=0 we have h[0]*h[k].
                # gradient_wrt_h[i] = h[i-shift] 
                # but shift < 0 so -shift = k. So we look at h[i+k] 
                # i+shift >= 0  and i+shift < n  (current check to ensure)
                for i in range(n_points):
                    if i - shift < n_points:
                        # else we just don't update these indices
                        grad[i] += h_vec[i-shift] 
                    if i >= -shift:  # i >= k, so we consider the term h[i] * h[i+k]  
                        grad[i] += h_vec[i+shift]
            
            # Normalize the gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                grad = grad / grad_norm
            else:
                # If gradient is zero, skip
                continue
            
            # Perform update with learning rate
            lr = 0.01
            h_vec_new = h_vec + lr * grad * (1 - 2 * int(min_idx < n_points))  # binary direction for sign adjustment
            
            h_vec = project_h(h_vec_new)

        return h_vec

    # Initialize optimization parameters
    c5_value = compute_c5(h)
    best_c5 = c5_value
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    lr = 0.01
    
    # First, do better initial optimization
    # Avoid full optimization, go with fast batches
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Run gradient ascent a few steps at a time
        h = gradient_ascent_step(h, steps=100)
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
        
        if iteration - last_improvement > 1000:
            # Optionally add momentum or restart strategy
            h = project_h(h + 0.001 * np.random.normal(0, 1, n_points))
        
        iteration += 1
    
    # Final projection
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
