import numpy as np
import cvxpy as cp
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085
    c_mid = (c_low + c_high) / 2
    iterations = 40
    
    best_feasible_h = h_start.copy()
    best_feasible_c = best_c5
    
    # Binary search 
    for i in range(iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Formulate feasibility problem
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints for all shifts
        # For shift k = -(n-1) to n-1, but we compute for k = 0 to 2*n-2        
        for k in range(2*n_points - 1):
            # Compute the circular correlation constraint for shift k
            corr_sum = 0
            for i in range(n_points):
                j = (i + k - (n_points - 1)) % n_points
                corr_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(corr_sum <= c_mid * n_points / 2)
        
        # Convex feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, feastol=1e-6, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_solution = h_vars.value
                c5_sol = evaluate_erdos_solution(h_solution, dx)
                if c5_sol < best_feasible_c:
                    best_feasible_c = c5_sol
                    best_feasible_h = h_solution.copy()
                
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
            
        c_mid = (c_low + c_high) / 2

    # Refinement: gradient descent with projected updates
    h_refine = best_feasible_h.copy()
    dx = 2.0 / n_points
    
    # Gradient descent parameters:
    learning_rate = 1e-3
    max_iter_refine = 1000
    tol = 1e-6
    
    for i in range(max_iter_refine):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        max_k = np.argmax(corr)
        max_value = corr[max_k]
        
        # Simple gradient: perturbation that decreases max correlation
        # If we can reduce the max correlation by changing h at a few points
        # For simplicity, just project a small random perturbation  
        pert = np.random.normal(0, 0.01, n_points)
        h_new = h_refine + learning_rate * pert
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)  # Recompute norm
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < evaluate_erdos_solution(h_refine, dx):
            h_refine = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_refine.copy()
    
    # Final validation and safeguard
    if best_c5 >= 0.38085:
        # fallback to initial if needed
        best_h = best_feasible_h.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    return (list(best_h), best_c5, n_points)
