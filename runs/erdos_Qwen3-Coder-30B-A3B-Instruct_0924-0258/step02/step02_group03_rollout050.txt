import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    swarm_size = 40
    max_iter = 10000
    omega = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max_ratio = 0.05
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    for _ in range(swarm_size):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points/2.0)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        val = evaluate(h, 2.0/n_points)
        pbest.append(h.copy())
        pbest_vals.append(val)
        if val < gbest_val:
            gbest_val = val
            gbest = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            v = (omega * velocities[i] + 
                 c1 * r1 * (pbest[i] - swarm[i]) + 
                 c2 * r2 * (gbest - swarm[i]))
            
            # Apply velocity limit
            v_max = v_max_ratio * (np.max(swarm[i]) - np.min(swarm[i]))
            if v_max <= 0:
                v_max = 0.05
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_h = swarm[i] + v
            new_h = project_to_feasible(new_h, n_points/2.0)
            
            # Evaluate
            new_val = evaluate(new_h, 2.0/n_points)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = new_h.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest_val = new_val
                    gbest = new_h.copy()
            
            swarm[i] = new_h
            velocities[i] = v
        
        iteration += 1
    
    if gbest is not None:
        final_c5 = evaluate(gbest, 2.0/n_points)
        return (gbest.tolist(), float(final_c5), n_points)
    else:
        final_h = project_to_feasible(h0, n_points/2.0)
        final_c5 = evaluate(final_h, 2.0/n_points)
        return (final_h.tolist(), float(final_c5), n_points)
