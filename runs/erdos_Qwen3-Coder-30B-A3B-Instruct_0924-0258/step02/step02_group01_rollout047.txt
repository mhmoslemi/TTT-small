import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

def genetic_algorithm_optimize(h_start, n_points, dx, time_limit):
    """Optimize using a genetic algorithm approach"""
    start_time = time()
    
    # Parameters
    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_k = 5
    num_generations = int((time_limit - (time() - start_time)) / 0.5) if (time_limit - (time() - start_time)) > 0.5 else 100
    
    # Initialize population
    population = []
    fitness = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        population.append(h_init)
        fitness.append(evaluate_erdos_solution(h_init, dx))
    else:
        h_init = h_start
        population.append(h_init)
        fitness.append(evaluate_erdos_solution(h_init, dx))
    
    # Fill remaining population with random feasible solutions
    for i in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = project_feasible(h_rand, n_points)
        population.append(h_rand)
        fitness.append(evaluate_erdos_solution(h_rand, dx))
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    for generation in range(num_generations):
        if time() - start_time > time_limit * 0.95:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_idx = np.random.choice(population_size, tournament_k)
            tournament_fitness = [fitness[i] for i in tournament_idx]
            winner_idx = tournament_idx[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
            
            # Project to feasible region
            child1 = project_feasible(child1, n_points)
            child2 = project_feasible(child2, n_points)
            
            offspring.extend([child1, child2])
        
        # Evaluate offspring
        offspring_fitness = [evaluate_erdos_solution(h, dx) for h in offspring[:population_size]]
        
        # Survival: Combine current population and offspring
        all_individuals = population + offspring[:population_size]
        all_fitness = fitness + offspring_fitness[:population_size]
        
        # Keep best individuals
        sorted_indices = np.argsort(all_fitness)
        population = [all_individuals[i] for i in sorted_indices[:population_size]]
        fitness = [all_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    return best_h, best_c5

    # dx is fixed for given n_points
    dx = 2.0 / n_points
    
    # Optimizer
    best_h, best_c5 = genetic_algorithm_optimize(h_start, n_points, dx, budget_s)
    
    # Final verification with original function
    final_c5 = evaluate_erdos_solution(best_h, dx)
