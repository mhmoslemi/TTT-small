import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Simple piecewise constant pattern
        h = np.ones(n_points) * 0.5
    return project_to_feasible(h, n_points)

def simplex_init(n_points, n_particles):
    """Initialize particles in a simplex configuration"""
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values)
        if len(base) != n_points:
            base = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(base)), base)
    else:
        base = np.ones(n_points) * 0.5
        
    particles = [base.copy()]
    for i in range(min(n_particles - 1, n_points)):
        pert = np.zeros(n_points)
        pert[i] = 0.1
        particles.append(project_to_feasible(base + pert, n_points))
    
    return particles

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 30
    dx = 2.0 / n_points
    max_iter = 2000
    swarm_size = 10
    inertia = 0.7
    c1 = c2 = 2.0
    shake_threshold = 50
    lbfgs_interval = 200
    
    # Initialize particles
    particles = simplex_init(n_points, swarm_size)
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    iter_no_improve = 0
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest[i] = particles[i].copy()
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    iter_no_improve = 0
            else:
                iter_no_improve += 1
        
        # Shake if no improvement
        if iter_no_improve > shake_threshold:
            for i in range(swarm_size):
                noise = np.random.normal(0, 0.1, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
            iter_no_improve = 0
            
        # Periodic LBFGS refinement
        if iteration % lbfgs_interval == 0:
            top_indices = np.argsort(pbest_scores)[:min(10, len(pbest_scores))]
            for idx in top_indices:
                try:
                    bounds = [(0, 1) for _ in range(n_points)]
                    def obj(h):
                        return evaluate_erdos_solution(h, dx)
                    res = minimize(obj, pbest[idx], method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 10, 'ftol': 1e-8})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        score = evaluate_erdos_solution(refined, dx)
                        if score < pbest_scores[idx]:
                            pbest_scores[idx] = score
                            pbest[idx] = refined
                            if score < gbest_score:
                                gbest = refined
                                gbest_score = score
                except:
                    pass
    
    # Final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 50, 'ftol': 1e-10})
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_score = evaluate_erdos_solution(final_h, dx)
            if final_score < gbest_score:
                gbest = final_h
                gbest_score = final_score
    except:
        pass
        
    return (list(gbest), gbest_score, n_points)
