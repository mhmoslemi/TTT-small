import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradient(h, k, n_points, dx):
    """Compute gradient for dominant lag k"""
    grad = np.zeros_like(h)
    for i in range(n_points):
        grad[i] = -(1 - h[(i + k) % n_points] - h[i]) * dx
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_feasible(h, n_points)
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated annealing parameters
    T = 0.5
    alpha = 0.995
    learning_rate = 0.02
    penalty_weight = 1e-3
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Initialize the main optimization loop
    while time() - start_time < budget_s:
        # Calculate current dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr)
        k = k - (len(corr) - 1) // 2  # Convert to lag
        if k < 0:
            k += len(corr) - 1
        
        # Compute gradient
        grad = compute_gradient(h, k, n_points, dx)
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        h_sum = np.sum(h)
        penalty_grad += 2 * penalty_weight * (h_sum - n_points / 2)
        
        # Take gradient step
        h_candidate = h - learning_rate * (grad + penalty_grad)
        h_candidate = project_feasible(h_candidate, n_points)
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        delta_c5 = candidate_c5 - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
        
        # Cooling schedule
        T *= alpha
        
        # Random perturbation if no improvement
        if no_improvement_count > max_no_improvement:
            perturbation = np.random.normal(0, 0.05, n_points)
            h = project_feasible(h + perturbation, n_points)
            current_c5 = evaluate_erdos_solution(h, dx)
            no_improvement_count = 0
        
        # Multi-resolution refinement
        if time() - start_time > budget_s * 0.75 and n_points < 100:
            n_points *= 2
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
            h = project_feasible(h, n_points)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return (list(best_h), best_c5, len(best_h))
