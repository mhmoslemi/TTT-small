import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=40):
    """Perform Frank-Wolfe optimization step for minimum overlap."""
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        # Compute all correlation values to find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Get the shift index that gives the max
        shift = max_idx - (n_points - 1)
        # Compute gradient (subgradient) at max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - shift) % n_points
            grad[i] = 1 - h[j] - h[i]
        
        # Frank-Wolfe linear minimization step:
        # solve min_s  <grad, s>  subject to sum(s) = n/2, 0 <= s <= 1
        # Simply choose s = argmin <grad, s> which is s[i] = 1 if grad[i] < 0 else 0,
        # but we need sum(s) = n/2. After clipping, we scale to sum = n/2.
        s = np.zeros(n_points)
        # To maximize <grad, s>, we put s[i] = 1 where grad[i] < 0
        s[grad < 0] = 1
        # Now we need sum(s) = n/2. Scale it.
        if np.sum(s) > 0:
            s = s * (n_points / 2) / np.sum(s)
        # Clip to [0,1]
        s = np.clip(s, 0, 1)
        
        # Line search for good step size gamma
        gamma = 0.5
        # Try different gamma if necessary
        for test_gamma in [0.9, 0.7, 0.5, 0.3, 0.1]:
            h_new = h + test_gamma * (s - h)
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                gamma = test_gamma
                break
        # Perform the update
        h = h + gamma * (s - h)
        h = project_to_feasible(h, n_points)
        c5_new = evaluate_erdos_solution(h, dx)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
    
    return best_h, best_c5
