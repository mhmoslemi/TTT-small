import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, alpha=100):
    return np.log(np.sum(np.exp(alpha * x))) / alpha

def smooth_max_grad(x, alpha=100):
    exp_x = np.exp(alpha * x)
    sum_exp = np.sum(exp_x)
    return exp_x / (sum_exp * alpha)

def compute_overlap_grad(h, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    
    # Gradient of the correlation with respect to h
    grad = np.zeros_like(h)
    # We need to compute the gradient of the maximum correlation
    # With respect to each component of h
    # For k = max_idx - (n_points-1) to max_idx + (n_points-1)
    # We have that corr[k] = sum_i h[i] * (1 - h[(i+k)%n_points])
    # So grad_{h[i]} = 1 - h[(i + max_idx - (n_points-1)) % n_points]
    # for all valid indices
    
    # Simplified: compute gradient using finite differences for robustness
    eps = 1e-6
    grad = np.zeros_like(h)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
    return grad

def optimize_step(h_start, n_points, dx, max_iter=300, lr=0.01, decay=0.99):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for i in range(max_iter):
        # Compute overlap and gradient
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Use smooth max instead for gradient
        smooth_max_val = smooth_max(corr, alpha=100)
        grad = compute_overlap_grad(h, n_points)
        
        # Gradient step
        h = h - lr * grad
        h = project_to_feasible(h, n_points)
        
        # Adaptive learning rate
        if i % 50 == 0:
            lr *= decay
            
        # Check if we improved
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    return best_h, best_c5
