import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_feasible(candidate_h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h_clipped = np.clip(candidate_h, 0, 1)
    h_sum = np.sum(h_clipped)
    if h_sum == 0:
        return h_clipped
    scaling = (n_points / 2) / h_sum
    return h_clipped * scaling

    start_time = time()

    # Create initial population
    population = []
    population.append(h_start)
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = project_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Genetic Algorithm loop
    generation = 0
    max_generations = int(budget_s / 0.5) if budget_s > 1 else 1000
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        generation += 1
        
        # Tournament selection
        selected = []
        tournament_size = 5
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Crossover (80% probability)
            if np.random.random() < 0.8:
                alpha = np.random.random()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation (20% probability)
            if np.random.random() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child1 += noise
                child1 = project_feasible(child1, n_points)
            
            if np.random.random() < 0.2:
                noise = np.random.normal(0, 0.1, n_points)
                child2 += noise
                child2 = project_feasible(child2, n_points)
            
            offspring.append(child1)
            if len(offspring) < population_size:
                offspring.append(child2)
        
        # Evaluate offspring
        offspring_fitness = []
        for h in offspring:
            fit = evaluate_erdos_solution(h, 2.0 / n_points)
            offspring_fitness.append(fit)
        
        # Survival: keep best 25% and replace worst with mutated best
        all_individuals = population + offspring
        all_fitness = fitness + offspring_fitness
        
        # Sort by fitness
        sorted_indices = np.argsort(all_fitness)
        next_generation_indices = sorted_indices[:population_size]
        
        # Ensure we're always keeping the best
        if fitness[best_idx] > all_fitness[sorted_indices[0]]:
            best_idx = sorted_indices[0]
            best_h = all_individuals[best_idx].copy()
            best_c5 = all_fitness[best_idx]
        
        # Build next generation
        population = [all_individuals[i] for i in next_generation_indices]
        fitness = [all_fitness[i] for i in next_generation_indices]
    
    return (list(best_h), best_c5, n_points)
