import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate(h, dx):
    """Compute the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = target_sum - current_sum
    if diff > 0:
        # Need more sum
        idxs = np.argsort(h)
        for i in range(len(idxs)-1, -1, -1):
            if diff <= 0:
                break
            j = idxs[i]
            if h[j] < 1:
                update = min(diff, 1 - h[j])
                h[j] += update
                diff -= update
    else:
        # Need less sum
        idxs = np.argsort(h)
        for i in range(len(idxs)):
            if diff >= 0:
                break
            j = idxs[i]
            if h[j] > 0:
                update = min(-diff, h[j])
                h[j] -= update
                diff += update
    
    return h

def trust_region_step(h, grad, trust_radius, target_sum, dx):
    """Compute a step with trust region constraints."""
    # Simple gradient descent with trust region
    step = -grad * dx
    norm_step = np.linalg.norm(step)
    
    if norm_step > trust_radius and norm_step > 1e-10:
        step = step * trust_radius / norm_step
    
    new_h = h + step
    new_h = project_to_feasible(new_h, target_sum)
