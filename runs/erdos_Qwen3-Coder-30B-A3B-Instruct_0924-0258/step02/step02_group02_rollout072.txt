import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        initial_h = np.full(45, 1.0 / 45)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial_h satisfies the constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_fitness(h_vec):
        return -compute_c5(h_vec)
    
    # Genetic algorithm parameters
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    generations = 50
    
    # Initialize population
    population = []
    population.append(initial_h.copy())
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
    
    best_solution = initial_h.copy()
    best_fitness = evaluate_fitness(best_solution)
    
    start_time = time.time()
    
    # Genetic algorithm main loop
    for generation in range(generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Evaluate fitness for entire population
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Select parents using tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmax(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Create offspring through crossover and mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                # Mutate 1 or 2 random indices
                num_mutations = np.random.choice([1, 2])
                mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                for idx in mutation_indices:
                    child1[idx] += np.random.normal(0, 0.01)
            
            if np.random.rand() < mutation_rate:
                # Mutate 1 or 2 random indices
                num_mutations = np.random.choice([1, 2])
                mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                for idx in mutation_indices:
                    child2[idx] += np.random.normal(0, 0.01)
            
            offspring.append(child1)
            offspring.append(child2)
        
        # Replace population
        for i in range(population_size):
            population[i] = project_h(offspring[i] if i < len(offspring) else population[i])
        
        # Track best solution
        current_best_idx = np.argmax(fitnesses)
        if fitnesses[current_best_idx] > best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_solution = population[current_best_idx].copy()
    
    # Final validation
    best_solution = project_h(best_solution)
    final_c5 = compute_c5(best_solution)
    
    return (best_solution.tolist(), final_c5, n_points)
