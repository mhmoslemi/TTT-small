import numpy as np
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_overlap(h_vec):
        h_vec = np.array(h_vec)
        # Compute all correlations (full mode)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take only the relevant part (max k values)
        mid = len(conv) // 2
        overlaps = conv[mid-n_points+1:mid+n_points-1] * dx
        return np.max(overlaps)
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to target sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec
    
    def evaluate_population(population):
        fitness = []
        for individual in population:
            ind = project_feasible(individual)
            val = compute_overlap(ind)
            fitness.append(val)
        return np.array(fitness)
    
    # Differential Evolution parameters
    population_size = 25
    mutation_factor = 0.7
    crossover_prob = 0.5
    max_generations = 300
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.random() < 0.5:
            # Perturb initial solution
            perturbed = h + np.random.normal(0, 0.05, n_points)
        else:
            # Random initialization
            perturbed = np.random.uniform(0, 1, n_points)
        population.append(perturbed)
    
    # Evaluate initial population
    fitness = evaluate_population(population)
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_val = fitness[best_idx]
    
    # Evolution loop
    generation = 0
    stagnation = 0
    max_stagnation = 50
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            indices = list(range(population_size))
            indices.remove(i)
            a, b, c = np.random.choice(indices, 3, replace=False)
            
            # Differential mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.random() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            
            trial = np.array(trial)
            
            # Ensure feasibility
            trial = project_feasible(trial)
            
            # Selection
            trial_fitness = compute_overlap(trial)
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                if trial_fitness < best_val:
                    best_val = trial_fitness
                    best_h = trial.copy()
                    stagnation = 0
            else:
                new_population.append(population[i])
                stagnation += 1
                
        population = new_population
        fitness = evaluate_population(population)
        
        # Check for stagnation
        if stagnation >= max_stagnation:
            break
            
        generation += 1
    
    # Final validation
    final_h = project_feasible(best_h)
    final_C5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
