import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time()
    
    # Initialize with precomputed values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            
        # PSO parameters
        n_particles = 50
        max_iter = 200
        omega = 0.7
        c1, c2 = 2.0, 2.0
        v_max = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        pbest = []
        pbest_values = []
        
        for _ in range(n_particles):
            h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
            pbest.append(h.copy())
            pbest_values.append(evaluate_erdos_solution(h, dx_try))
        
        # Global best
        gbest_idx = np.argmin(pbest_values)
        gbest = pbest[gbest_idx].copy()
        gbest_value = pbest_values[gbest_idx]
        
        # PSO iterations
        for iter_idx in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update particles
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Apply velocity limits
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                
                # Project to feasible region
                particles[i] = project_to_feasible(particles[i], n_points)
                
                # Evaluate
                curr_value = evaluate_erdos_solution(particles[i], dx_try)
                
                # Update personal best
                if curr_value < pbest_values[i]:
                    pbest[i] = particles[i].copy()
                    pbest_values[i] = curr_value
                
                # Update global best
                if curr_value < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = curr_value
            
            # Update inertia weight
            omega = 0.7 - (0.3 * iter_idx / max_iter)
            
        # Check if current solution is better
        if gbest_value < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_value
            best_n_points = n_points
            
        h_start = best_h.copy()
    
    # Final refinement with gradient descent
    def refined_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def refined_gradient(h):
        # Simple numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = refined_objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            grad[i] = (refined_objective(h_plus) - base_val) / eps
        return grad
    
    # Gradient descent with small steps
    h_refined = best_h.copy()
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        grad = refined_gradient(h_refined)
        h_refined -= 0.001 * grad
        h_refined = project_to_feasible(h_refined, n_points)
    
    final_c5 = evaluate_erdos_solution(h_refined, dx)
    if final_c5 < best_c5:
        best_h = h_refined
        best_c5 = final_c5
        best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
