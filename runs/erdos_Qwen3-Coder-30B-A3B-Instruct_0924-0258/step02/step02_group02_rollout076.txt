import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def compute_autocorr(h_vec):
        # Compute all autocorrelations
        conv = np.correlate(h_vec, h_vec, mode='full')
        # Only keep the valid correlations where both sequences overlap
        n = len(h_vec)
        start_idx = n - 1
        autocorr_vals = conv[start_idx:]
        return autocorr_vals

    def compute_shifted_autocorr(h_vec, shift):
        # Compute h[i] * h[i+shift] for valid indices
        if shift >= len(h_vec):
            return 0.0
        # Multiply corresponding elements
        prod = h_vec[:-shift] * h_vec[shift:]
        return np.sum(prod)

    def compute_min_autocorr(h_vec):
        min_val = float('inf')
        n = len(h_vec)
        # For shift s in [0, n-1], compute h[i] * h[i+s]
        for s in range(n):
            val = compute_shifted_autocorr(h_vec, s)
            if val < min_val:
                min_val = val
        return min_val

    def compute_gradient(h_vec, shift):
        # Compute gradient for shift
        grad = np.zeros_like(h_vec)
        n = len(h_vec)
        # Compute gradient with respect to A_s = sum(h[i] * h[i+s]) 
        # Gradient w.r.t. h[i] = h[i+s] for i+s < n
        if shift < n:
            grad[:-shift] += h_vec[shift:]
            # Gradient w.r.t. h[i+s] = h[i] for i < n-shift
            grad[shift:] += h_vec[:-shift]
        return grad

    def manage_constraint(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        sum_h = np.sum(h_vec)
        if sum_h > 0:
            h_vec = h_vec * target_sum / sum_h
        return h_vec

    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    learning_rate = 0.01
    patience_counter = 0
    max_patience = 1000
    min_improvement = 1e-8

    for iteration in range(100000):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Compute current autocorrelations
        corr = np.correlate(h, 1 - h, mode='full')
        corr *= dx
        c5 = np.max(corr)
        
        # Find the min correlations in 1-h correlations
        corr_1_h = np.correlate(h, 1 - h, mode='full')
        corr_1_h *= dx
        min_corr_idx = np.argmin(corr_1_h)
        min_corr_val = corr_1_h[min_corr_idx]
        
        # Adjust h to increase this minimal correlation
        if min_corr_idx > len(h):
            shift = min_corr_idx - len(h)
        else:
            shift = 0
            
        # The shift we want to increase the correlation for
        # is from the autocorrelation interpretation
        actual_shift = min_corr_idx - (len(h) - 1)
        if actual_shift < 0:
            actual_shift = 0
            
        # Compute gradient of correlation for this shift
        grad = np.zeros_like(h)
        if actual_shift < len(h):
            # h[i] * h[i+shift] terms contribute
            grad[:-actual_shift] += h[actual_shift:]
            grad[actual_shift:] += h[:-actual_shift]
        
        # Adjust step size 
        h_new = h + learning_rate * grad
        h_new = manage_constraint(h_new)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience_counter = 0
            if learning_rate > 1e-5:
                learning_rate *= 0.999
        else:
            patience_counter += 1
            if patience_counter > max_patience:
                learning_rate *= 0.5
                patience_counter = 0

        h = h_new.copy()
        if patience_counter > 2000:
            # Add noise to escape local minima
            noise = 0.0001 * np.random.normal(0, 1, len(h))
            h = h + noise
            h = manage_constraint(h)
            patience_counter = 0
        

    return (best_h.tolist(), best_c5, n_points)
