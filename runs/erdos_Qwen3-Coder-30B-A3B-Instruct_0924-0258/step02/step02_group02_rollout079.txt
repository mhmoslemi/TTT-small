import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a default initial vector if none provided
        n_points = 45
        h = np.full(n_points, 1.0 / n_points)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h satisfies the constraint
    h = h * target_sum / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute max correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        corr *= dx
        return np.max(corr)

    def compute_min_autocorr(h_vec):
        # Compute autocorrelations for all shifts
        corr = np.correlate(h_vec, h_vec, mode='full')
        # Extract valid shifts [0, n_points-1]
        valid_corr = corr[len(h_vec)-1:]
        return np.min(valid_corr)

    def compute_min_shift_and_grad(h_vec):
        # Compute correlation with 1-h
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Extract valid shifts in [0, n_points-1] with actual length
        full_corr = corr
        shifts = np.arange(len(full_corr))
        # get shifts from 0 to n_points-1 that are valid
        # full correlation has length 2*n_points - 1, centered at n_points-1
        start_idx = n_points - 1
        valid_corr = full_corr[start_idx:]
        min_shift = np.argmin(valid_corr)
        min_val = valid_corr[min_shift]
        # Compute gradient for correlation at min_shift
        grad = np.zeros_like(h_vec)
        # For shift k, derivative w.r.t. h[i] is h[i+k]
        # when i+k < n_points
        if min_shift > 0:
            for i in range(n_points - min_shift):
                grad[i] += (1 - h_vec[i + min_shift])
        return min_shift, min_val, grad
    
    def project_h(h_vec):
        # Clip to valid range
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to target_sum
        h_vec = h_vec * target_sum / np.sum(h_vec)
        return h_vec

    def update_h(h_vec, grad, lr):
        h_new = h_vec + lr * grad
        return project_h(h_new)

    start_time = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Adaptive learning rate
    lr = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        min_shift, min_val, grad = compute_min_shift_and_grad(h)
        
        # Update with learning rate
        h_candidate = update_h(h, grad, lr)
        
        new_c5 = compute_c5(h_candidate)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_candidate.copy()
            no_improvement_count = 0
            lr = min(0.05, lr * 1.05)  # Increase learning rate slightly
        else:
            no_improvement_count += 1
            lr = max(1e-6, lr * 0.95)  # Decrease learning rate
            
            # Add noise to escape local minima
            noise = 0.001 * np.random.normal(0, 1, n_points)
            h = h + noise
            h = project_h(h)
            
            # If no improvement for some time, also try a small SGD step
            if no_improvement_count > max_no_improvement:
                # Try a small randomized step
                random_grad = np.random.rand(n_points) - 0.5
                h = update_h(h, random_grad, 0.5 * lr)
                no_improvement_count = 0

        h = h_candidate
        iteration += 1

    # Final validation
    best_h = project_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
