import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from initial_h_values if available, otherwise create pattern"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    else:
        # Create a low-correlation binary-like pattern
        h = np.zeros(n_points)
        ones_count = n_points // 2
        positions = np.linspace(0, n_points-1, ones_count, dtype=int)
        h[positions] = 1.0
        # Slight perturbation to avoid regular patterns
        for i in range(ones_count):
            if i > 0 and i < ones_count - 1:
                if np.random.random() < 0.3:
                    idx = positions[i]
                    h[idx] = 0.0
                    new_idx = (idx + np.random.randint(1, n_points-1)) % n_points
                    h[new_idx] = 1.0
    return project_to_feasible(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use a simple but effective multi-resolution approach
    resolutions = [15, 30, 45, 75]
    best_h = None
    best_c5 = float('inf')
    best_n_points = 0
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        
        # Time allocation
        remaining_time = budget_s - (time() - start_time)
        time_per_res = min(remaining_time / len(resolutions), 150)
        
        # Multiple restarts with different noise patterns
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_current = h_start.copy()
            # Perturb with controlled noise
            noise_scale = 0.1 / (restart + 1)
            noise = np.random.normal(0, noise_scale, n_points)
            h_current = project_to_feasible(h_current + noise, n_points)
            
            # Optimization with L-BFGS-B
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            
            try:
                res = minimize(objective, h_current, method='L-BFGS-B', 
                              bounds=bounds, constraints=constraints, 
                              options={'maxiter': 30, 'ftol': 1e-6, 'gtol': 1e-6})
                if res.success:
                    h_opt = project_to_feasible(res.x, n_points)
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
                        best_n_points = n_points
            except Exception:
                pass
                
            # Periodic refinement with perturbations
            for iter in range(20):
                if time() - start_time > budget_s * 0.95:
                    break
                try:
                    # Generate new random configuration in feasible region
                    h_new = np.random.rand(n_points)
                    h_new = project_to_feasible(h_new, n_points)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        best_n_points = n_points
                except:
                    pass

    if best_h is None:
        # Fallback to 30-point configuration
        n_points = 30
        dx = 2.0 / n_points
        h_start = initialize_h(n_points)
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            res = minimize(objective, h_start, method='L-BFGS-B', 
                          bounds=bounds, constraints=constraints, 
                          options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
            if res.success:
                best_h = project_to_feasible(res.x, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
                best_n_points = n_points
        except:
            best_h = h_start
            best_c5 = evaluate_erdos_solution(best_h, dx)
            best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
