import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_simplex(n_points, m=10):
    """Initialize m particles using simplex method"""
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    particles = [project_to_feasible(h0.copy(), n_points)]
    
    # Generate additional particles by moving along random directions
    for i in range(1, min(m, n_points + 1)):
        h = h0.copy()
        idx = i % n_points
        h[idx] = np.clip(h[idx] + np.random.normal(0, 0.1), 0, 1)
        particles.append(project_to_feasible(h, n_points))
    
    return particles

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters for PSO
    n_points = 30
    dx = 2.0 / n_points
    swarm_size = 10
    max_iter = 2000
    w = 0.7
    c1, c2 = 2.0, 2.0
    shaking_threshold = 50
    lbfgs_interval = 200
    
    # Initialize particles
    particles = initialize_simplex(n_points, swarm_size)
    velocities = [np.zeros(n_points) for _ in range(len(particles))]
    
    # Evaluate initial particles
    fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Track personal bests
    pbest = particles.copy()
    pbest_fitness = fitness.copy()
    
    # Track global best
    gbest_idx = np.argmin(fitness)
    gbest = particles[gbest_idx].copy()
    gbest_fitness = fitness[gbest_idx]
    
    # PSO main loop
    iter_no_improve = 0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(len(particles)):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new fitness
            new_fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_fitness < pbest_fitness[i]:
                pbest[i] = particles[i].copy()
                pbest_fitness[i] = new_fitness
            
            # Update global best
            if new_fitness < gbest_fitness:
                gbest = particles[i].copy()
                gbest_fitness = new_fitness
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        
        # Shaking mechanism to escape local minima
        if iter_no_improve >= shaking_threshold:
            for i in range(len(particles)):
                noise = np.random.normal(0, 0.05, n_points)
                particles[i] = project_to_feasible(particles[i] + noise, n_points)
                particles[i] = project_to_feasible(particles[i], n_points)
            iter_no_improve = 0
        
        # Periodically apply L-BFGS-B to top candidates for precision
        if iter_num % lbfgs_interval == 0 and iter_num > 0:
            # Identify top 10 particles
            top_indices = np.argsort(pbest_fitness)[:min(10, len(pbest_fitness))]
            for idx in top_indices:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, pbest[idx], method='L-BFGS-B', bounds=bounds, 
                                 options={'maxiter': 10})
                    if res.success:
                        refined = project_to_feasible(res.x, n_points)
                        refined_fitness = evaluate_erdos_solution(refined, dx)
                        if refined_fitness < pbest_fitness[idx]:
                            pbest[idx] = refined
                            pbest_fitness[idx] = refined_fitness
                            if refined_fitness < gbest_fitness:
                                gbest = refined
                                gbest_fitness = refined_fitness
                except:
                    pass
    
    # Final L-BFGS refinement on the best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        res = minimize(obj, gbest, method='L-BFGS-B', bounds=bounds, 
                     options={'maxiter': 50})
        if res.success:
            gbest = project_to_feasible(res.x, n_points)
            gbest_fitness = evaluate_erdos_solution(gbest, dx)
    except:
        pass
    
    return (list(gbest), gbest_fitness, n_points)
