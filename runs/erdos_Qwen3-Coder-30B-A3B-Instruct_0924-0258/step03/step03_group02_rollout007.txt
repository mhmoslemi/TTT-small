import numpy as np
from scipy.optimize import minimize
from time import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    popsize = 50
    F = 0.5
    CR = 0.9
    maxiter = 1000
    
    # Initialize population
    population = []
    for _ in range(popsize):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add initial solution if it's better than random
    if evaluate_erdos_solution(h_start, dx) < best_c5:
        population[0] = h_start
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # DE loop
    iteration = 0
    stagnation_count = 0
    last_best_c5 = best_c5 + 1  # Ensure no initial stagnation
    
    while iteration < maxiter and (time() - start_time) < budget_s * 0.85:
        if stagnation_count > 100:
            break
            
        new_population = []
        for i, h in enumerate(population):
            # Mutation: select three distinct individuals
            idxs = [idx for idx in range(popsize) if idx != i]
            a, b, c = [population[j] for j in np.random.choice(idxs, 3, replace=False)]
            
            # Mutation
            mutant = a + F * (b - c)
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, h)
            
            # Projection
            trial = project_to_feasible(trial, n_points)
            
            # Selection
            if evaluate_erdos_solution(trial, dx) < evaluate_erdos_solution(h, dx):
                new_population.append(trial)
            else:
                new_population.append(h)
        
        population = new_population
        
        # Check for improvement
        current_best = min([evaluate_erdos_solution(h, dx) for h in population])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin([evaluate_erdos_solution(h, dx) for h in population])].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        iteration += 1
    
    # SLSQP refinement of the best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
