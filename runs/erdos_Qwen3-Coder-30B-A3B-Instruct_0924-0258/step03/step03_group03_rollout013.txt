import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(h, beta):
    """Smooth approximation to the maximum for differentiability"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.log(np.sum(np.exp(beta * corr))) / beta

def smooth_gradient(h, beta, eps=1e-5):
    """Approximate gradient via finite differences of smooth max"""
    grad = np.zeros_like(h)
    base_val = smooth_max(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_max(h_plus, beta) - base_val) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_to_feasible(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-5
    max_iter = 1000

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = smooth_gradient(h, beta)
        
        # Try a step
        h_try = h - step_size * grad
        h_try = project_to_feasible(h_try, n_points)
        c5_try = evaluate_erdos_solution(h_try, dx)
        
        # Accept the step if it improves
        if c5_try < best_c5:
            h = h_try.copy()
            best_c5 = c5_try
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Adjust smoothing parameter
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.5
            
        # Early termination if no improvement
        if np.abs(best_c5 - smooth_max(h, beta)) < tolerance:
            break

    # Final pass with exact computation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return (list(best_h), best_c5, n_points)
