import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple construction meeting integral constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    max_no_improve = 50
    no_improve_count = 0
    
    # Local search parameters
    local_iterations = 10
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.7:
        # Generate neighbors
        neighbors = []
        num_neighbors = 100
        for _ in range(num_neighbors):
            h_candidate = best_h.copy()
            # Perturb 20% of points
            indices_to_perturb = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            for i in indices_to_perturb:
                # Add small random perturbation
                delta = np.random.normal(0, 0.05)
                h_candidate[i] = np.clip(h_candidate[i] + delta, 0, 1)
            
            # Project to satisfy integral constraint
            h_sum = np.sum(h_candidate)
            if h_sum > 0:
                h_candidate = h_candidate * (n_points / 2) / h_sum
            
            neighbors.append(h_candidate)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Convert to tuple for tabu checking
            h_tuple = tuple(h_candidate)
            if h_tuple in tabu_list:
                # Check for escape
                if np.random.rand() < 0.1:  # 10% chance to escape
                    pass
                else:
                    continue
            
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = c5_candidate
        
        if best_neighbor is None:
            # No valid neighbor, break
            break
            
        # Move to best neighbor
        best_h = best_neighbor.copy()
        best_c5 = best_neighbor_c5
        
        # Update tabu list
        tabu_list.append(tuple(best_h))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Update no improvement counter
        if best_neighbor_c5 < best_c5:
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        if no_improve_count >= max_no_improve:
            break
    
    # Local refinement with gradient-based optimization
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Run several local optimizations with different starting points
        best_local = best_h.copy()
        best_local_c5 = best_c5
        
        # Start from the current best
        try:
            for _ in range(local_iterations):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                # Local optimization step using SLSQP
                cons = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Perturb current best to avoid local minima
                h_perturb = best_local + np.random.normal(0, 0.01, n_points)
                # Ensure feasibility after perturbation
                h_perturb = np.clip(h_perturb, 0, 1)
                
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_local_c5:
                        best_local = res.x.copy()
                        best_local_c5 = c5_test
                else:
                    # If optimization fails, just try a few more random steps
                    pass
        except:
            # If anything goes wrong, continue with existing solution
            pass
        
        # Final validation
        if best_local_c5 < best_c5:
            best_h = best_local
            best_c5 = best_local_c5
    
    return (list(best_h), best_c5, n_points)
