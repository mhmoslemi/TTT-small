import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create a structured initialization with clustered regions
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based Nelder-Mead approach
    population_size = 15
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    # Nelder-Mead iterations
    nelder_mead_iter = 0
    max_iter = 30  # Allow up to 30 simplex iterations
    
    while nelder_mead_iter < max_iter and time() - start_time < budget_s * 0.9:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate all candidates in population
        costs = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(costs)
        population = [population[i] for i in sorted_indices]
        costs = [costs[i] for i in sorted_indices]
        
        if costs[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = costs[0]
        
        # Create new simplex using reflection, expansion, etc.
        simplexes = []
        for i in range(population_size - 1):
            # Simplex: all but worst point
            simplex = [population[j] for j in range(i, population_size - 1)]
            # Compute centroid (excluding worst member)
            centroid = np.mean(simplex[:-1], axis=0)
            
            # Reflection
            r = 2 * centroid - simplex[-1]
            r = np.clip(r, 0, 1)
            r = r * (n_points/2) / np.sum(r)
            
            # Try expansion if beneficial
            cost_r = evaluate_erdos_solution(r, dx)
            if cost_r < costs[i]:
                # Try expansion
                e = centroid + 2 * (r - centroid)
                e = np.clip(e, 0, 1)
                e = e * (n_points/2) / np.sum(e)
                cost_e = evaluate_erdos_solution(e, dx)
                if cost_e < cost_r:
                    simplex[-1] = e
                else:
                    simplex[-1] = r
            else:
                simplex[-1] = r
                
            # Store valid simplex for next iteration
            simplexes.append(simplex)
        
        # Update population with best of each simplex
        new_population = []
        for simplex in simplexes:
            costs_simplex = [evaluate_erdos_solution(h, dx) for h in simplex]
            best_idx = np.argmin(costs_simplex)
            new_population.append(simplex[best_idx])
        
        # Add some diversity
        for _ in range(population_size - len(new_population)):
            h_new = best_h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_population.append(h_new)
            
        population = new_population
        nelder_mead_iter += 1
    
    # Final local refinement with gradient descent
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine using SLSQP with restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.99:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
