import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive steps
    max_iter = 10000
    patience = 100
    patience_counter = 0
    
    # Initialize variables
    h_current = best_h.copy()
    current_c5 = best_c5
    prev_c5 = current_c5 + 1.0
    
    # Add some entropy to make initial solution more interesting
    h_current += np.random.normal(0, 0.01, n_points)
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient using finite differences
        h_grad = np.zeros_like(h_current)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # Project to feasible set
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute directional derivative approximating gradient
            c_plus = evaluate_erdos_solution(h_plus, dx)
            c_minus = evaluate_erdos_solution(h_minus, dx)
            h_grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Adaptive learning rate
        lr = 1e-3 / (1.0 + iteration / 1000)
        
        # Update step
        h_new = h_current - lr * h_grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if patience_counter >= patience:
            break
            
        # Occasionally add noise to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            current_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Final check
    final_c5 = evaluate_erdos_solution(h_current, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_current.copy()
    
    return (list(best_h), best_c5, n_points)
