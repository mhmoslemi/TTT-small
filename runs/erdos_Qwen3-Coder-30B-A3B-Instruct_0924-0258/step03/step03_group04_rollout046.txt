import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_update(h, k_shift, n_points, dx):
    # Compute shifted version of h
    h_shifted = np.concatenate([h[k_shift:], np.zeros(k_shift)])
    # Linearized objective (dot product with (1 - h_shifted))
    linear_obj = 1 - h_shifted
    
    # Find extreme point (binary vector with n_points/2 ones) 
    # minimizing the linear objective
    # Sorting indices to get largest values of (1 - h_shifted)
    indices_sorted = np.argsort(-linear_obj)
    h_extreme = np.zeros(n_points)
    h_extreme[indices_sorted[:n_points//2]] = 1.0
    
    return h_extreme

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a periodic binary pattern
        h_start = np.zeros(n_points)
        ones_positions = np.arange(0, n_points, 2)
        h_start[ones_positions[:n_points//2]] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    max_iter = 10000
    step_size = 0.5
    num_improvements = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute max overlap index k
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        # Adjust index to be within valid range [0, n_points-1]
        actual_k = max_k - (n_points - 1)  # The zero shift is at the center
        
        # Frank-Wolfe step
        h_extreme = frank_wolfe_update(best_h, actual_k, n_points, dx)
        new_h = (1 - step_size) * best_h + step_size * h_extreme
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            num_improvements += 1
            last_improvement = iteration
        else:
            # Perturbation to escape local optima
            if np.random.random() < 0.1:
                eps = 0.01
                noise = np.random.normal(0, eps, n_points)
                perturbed_h = best_h + noise
                perturbed_h = project_to_feasible(perturbed_h, n_points)
                perturbed_c5 = evaluate_erdos_solution(perturbed_h, dx)
                if perturbed_c5 < best_c5:
                    best_h = perturbed_h.copy()
                    best_c5 = perturbed_c5
        
        # Adapt step size based on recent improvements
        if iteration - last_improvement > 100:
            step_size = min(0.9, step_size * 1.05)
        else:
            step_size = max(0.1, step_size * 0.95)
    
    # Final check with L-BFGS
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
