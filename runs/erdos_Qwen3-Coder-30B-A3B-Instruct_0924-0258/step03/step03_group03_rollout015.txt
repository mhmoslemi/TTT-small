import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points
    else:
        # Start with 30 points
        n_points = 30
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points

    # Optimization parameters
    max_iter = 300
    step_size = 0.01
    tolerance = 1e-6
    n_particles = 30

    # PSO-like swarm optimization with local refinement
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []

    # Initialize with multiple starting points
    for _ in range(n_particles):
        h = project_to_feasible(h_start + np.random.normal(0, 0.05, n_points), n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        score = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_scores.append(score)

        if score < best_c5:
            best_h = h.copy()
            best_c5 = score
            best_n_points = n_points

    # Global best
    gb_idx = np.argmin(personal_best_scores)
    global_best = personal_best[gb_idx].copy()
    global_best_score = personal_best_scores[gb_idx]

    # Main PSO loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for p in range(n_particles):
            # Update velocity with better inertia and parameters
            r1, r2 = np.random.rand(2)
            inertia = 0.7 + 0.3 * (1 - it / max_iter)
            velocities[p] = (inertia * velocities[p] + 
                           2.0 * r1 * (personal_best[p] - particles[p]) + 
                           2.0 * r2 * (global_best - particles[p]))
            
            # Update position
            particles[p] = particles[p] + velocities[p]
            
            # Project to feasible region
            particles[p] = project_to_feasible(particles[p], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(particles[p], dx)
            
            # Update personal best
            if score < personal_best_scores[p]:
                personal_best[p] = particles[p].copy()
                personal_best_scores[p] = score
                if score < best_c5:
                    best_h = particles[p].copy()
                    best_c5 = score
                    best_n_points = n_points
            
            # Update global best
            if score < global_best_score:
                global_best = particles[p].copy()
                global_best_score = score

    # Local refinement with gradient descent
    if time() - start_time < budget_s * 0.9:
        h_refined = best_h.copy()
        score_refined = best_c5
        dx_refined = 2.0 / best_n_points
        
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx_refined)
            
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx_refined)
                
                if np.isfinite(score_plus) and np.isfinite(base_score):
                    grad[i] = (score_plus - base_score) / eps

            # Update with projected step
            h_new = h_refined - step_size * grad
            h_new = project_to_feasible(h_new, best_n_points)
            
            new_score = evaluate_erdos_solution(h_new, dx_refined)
            
            if new_score < score_refined:
                h_refined = h_new
                score_refined = new_score
                if score_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = score_refined
                    best_n_points = best_n_points

    return (list(best_h), best_c5, best_n_points)
