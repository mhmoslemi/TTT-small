import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based simulated annealing with SQP refinement
    population_size = 20
    population = []
    for i in range(population_size):
        h = h_start + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Parameters for simulated annealing
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    
    iterations = 30
    for _ in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # SQP refinement
        for i in range(len(population)):
            h = population[i].copy()
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < fitness[i]:
                        population[i] = h_refined
                        fitness[i] = c5_refined
                        if c5_refined < best_c5:
                            best_c5 = c5_refined
                            best_h = h_refined.copy()
            except:
                pass
        
        # Simulated Annealing
        for i in range(len(population)):
            # Generate neighbor
            neighbor = population[i] + np.random.normal(0, 0.01, n_points)
            neighbor = np.clip(neighbor, 0, 1)
            neighbor = neighbor * (n_points/2) / np.sum(neighbor)
            
            c5_neighbor = evaluate_erdos_solution(neighbor, dx)
            
            # Accept or reject based on temperature
            if c5_neighbor < fitness[i] or np.random.rand() < np.exp((fitness[i] - c5_neighbor) / temp):
                population[i] = neighbor
                fitness[i] = c5_neighbor
                if c5_neighbor < best_c5:
                    best_c5 = c5_neighbor
                    best_h = neighbor.copy()
        
        # Cool down temperature
        temp *= cooling_rate
    
    return (list(best_h), best_c5, n_points)
