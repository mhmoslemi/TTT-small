import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    # Genetic Algorithm parameters
    population_size = 50
    num_generations = 100
    tournament_size = 5
    mutation_rate = 0.2
    crossover_rate = 0.8
    
    # Initialize population
    def initialize_population(size, n_points):
        population = []
        if 'initial_h_values' in globals():
            population.append(h_start)
            size -= 1
        
        for _ in range(size):
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)  # Normalize
            population.append(h)
        return population
    
    # Selection using tournament
    def tournament_selection(population, fitnesses, k=tournament_size):
        selected_idx = random.randint(0, len(population)-1)
        for _ in range(k-1):
            idx = random.randint(0, len(population)-1)
            if fitnesses[idx] < fitnesses[selected_idx]:
                selected_idx = idx
        return population[selected_idx]
    
    # Crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = random.randint(1, len(parent1)-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Renormalize
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    # Mutation
    def mutate(h, rate=mutation_rate):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < rate:
                noise = np.random.normal(0, 0.1)
                h[i] += noise
        # Clip and renormalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main GA loop
    best_fitness_history = []
    no_improvement_count = 0
    
    # Ensure enough time for refinement
    time_left = budget_s * 0.95
    
    population = initialize_population(population_size, n_points)
    
    for generation in range(num_generations):
        if time() - start_time > time_left:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]

        min_fitness = min(fitnesses)
        best_idx = fitnesses.index(min_fitness)

            child1 = mutate(child1)
            child2 = mutate(child2)

        # Trim to exact population size
        population = new_population[:population_size]

        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

                global_best_h = h_final
                global_best_c5 = final_c5
    except:
        pass
    
    # Final verification and return
    final_c5 = evaluate_erdos_solution(global_best_h, dx)
    return (list(global_best_h), final_c5, n_points)
