import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with random perturbations
    max_iter = 50000
    learning_rate = 1e-3
    p_perturb = 0.1
    perturb_magnitude = 0.05
    convergence_patience = 100
    
    h_current = h_start.copy()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            last_improvement = iteration
        
        # Compute gradient (subgradient) w.r.t. max overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)  # Convert to actual shift
        
        gradient = np.zeros(n_points)
        if max_shift > 0:
            # Positive shift case
            for i in range(n_points - max_shift):
                gradient[i] += (1 - h_current[i + max_shift])
        elif max_shift < 0:
            # Negative shift case
            for i in range(-max_shift, n_points):
                gradient[i] += -(1 - h_current[i + max_shift])
        else:
            # Zero shift: no gradient contribution
            pass
            
        # Update with gradient and project
        h_new = h_current - learning_rate * gradient
        h_new = project_to_feasible(h_new, n_points)
        
        # Apply random perturbation occasionally
        if np.random.rand() < p_perturb:
            perturbation = np.random.normal(0, perturb_magnitude, n_points)
            h_new += perturbation
            h_new = project_to_feasible(h_new, n_points)
            
        h_current = h_new
        
        # Early stopping if not improving
        if iteration - last_improvement > convergence_patience:
            break
    
    return (list(best_h), best_c5, n_points)
