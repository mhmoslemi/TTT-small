import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal-like initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe with perturbation
    h_current = h_start.copy()
    max_iter = 10000
    step_size = 0.5
    perturb_rate = 0.1
    patience = 200
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        # Finding the maximum overlap shift
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)
        
        # Compute shift for maximum overlap
        shifted_ones = np.roll(1 - h_current, k_max)
        # Linearization: we want to minimize sum(h_diff * (1 - h_shifted_k))
        # So we use h_diff = shifted_ones to create gradient
        grad = shifted_ones
        
        # Heuristic to find extreme point: top k indices where gradient is most negative
        # Ergo, best approximate solution for binary case
        # We want to place ones at positions where (1 - h_shifted_k) is largest
        # So position-wise, we want to select ones where h_current[i] -> 0 
        # Given that we're approximating with binary solutions:
        indices = np.argsort(grad)[::-1]  # Sort descending
        h_extreme = np.zeros_like(h_current)
        ones_count = n_points // 2
        h_extreme[indices[:ones_count]] = 1.0
        
        # Frank-Wolfe update
        h_new = (1 - step_size) * h_current + step_size * h_extreme
        
        # Project back to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate and update
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 200  # reset patience
        else:
            patience -= 1
            if patience <= 0:
                # Reduce step size to prevent cycling
                step_size *= 0.8
                if step_size < 1e-5:
                    break
        
        h_current = h_new
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.random() < perturb_rate:
            perturb_indices = np.random.choice(n_points, size=n_points//4, replace=False)
            h_current[perturb_indices] = np.random.random(n_points//4)
            h_current = project_to_feasible(h_current, n_points)
    
    # Final refinement with gradient descent 
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
