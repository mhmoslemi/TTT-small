import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based parametrization
    def fourier_to_h(coeffs):
        # coeffs: array of length d+1
        d = len(coeffs) - 1
        x = np.linspace(0, 2, n_points)
        g = np.zeros_like(x)
        for k in range(d + 1):
            g += coeffs[k] * np.cos(k * np.pi * x)
        h = 1.0 / (1.0 + np.exp(-g))
        return h
    
    def objective(coeffs):
        h = fourier_to_h(coeffs)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h, dx)
    
    # Constraints for sum
    def sum_constraint(coeffs):
        h = fourier_to_h(coeffs)
        return np.sum(h) - n_points / 2
    
    # Optimization bounds
    bounds = [(-10, 10) for _ in range(6)]  # 6 coefficients (0 to 5)
    constraints = {'type': 'eq', 'fun': sum_constraint}
    
    # Try multiple random restarts
    best_coeffs = None
    min_c5 = best_c5
    
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initial guess
        coeffs0 = np.random.uniform(-2, 2, 6)
        
        try:
            res = minimize(objective, coeffs0, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-6})
            
            if res.success:
                c5_test = objective(res.x)
                if c5_test < min_c5:
                    min_c5 = c5_test
                    best_coeffs = res.x.copy()
        except:
            continue
    
    # Final refinement
    if best_coeffs is not None:
        h_final = fourier_to_h(best_coeffs)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
