import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different initialization strategies
    init_strategies = [
        lambda: np.ones(n_points) * 0.5,
        lambda: np.random.rand(n_points) * 0.5,
        lambda: np.concatenate([np.ones(n_points//2), np.zeros(n_points//2)]) * 2,
        lambda: np.concatenate([np.ones(n_points//3), np.zeros(2*n_points//3)]) * 1.5,
        lambda: np.maximum(0, 1 - np.abs(np.linspace(-1, 1, n_points))),
    ]
    
    for i, init_func in enumerate(init_strategies):
        if time() - start_time > budget_s * 0.9:
            break
        
        h_init = init_func()
        h_init = np.clip(h_init, 0, 1)
        h_sum = np.sum(h_init)
        if h_sum > 0:
            h_init = h_init * (n_points / 2) / h_sum
            
        # Local optimization
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # If we didn't improve much by now, do a more focused search
    if best_c5 > 0.38:
        # Try wavelet-inspired approach with more structure
        # Create piecewise constant basis
        n_basis = min(20, n_points)
        
        # Generate piecewise constant basis
        basis = np.zeros((n_basis, n_points))
        for i in range(n_basis):
            start = i * n_points // n_basis
            end = (i + 1) * n_points // n_basis
            if i == n_basis - 1:
                end = n_points  # make sure the last one covers to end
            basis[i, start:end] = 1.0
        
        # Normalize basis
        for i in range(n_basis):
            norm = np.sqrt(np.sum(basis[i]**2) * dx)
            if norm > 0:
                basis[i] /= norm
        
        # Try to optimize linear combination of basis functions
        def solve_basis_optimization(h_basis, c0=0.5):
            # Initialize with a simple coefficient
            c = np.ones(n_basis) * c0
            c[0] = c0
            best_c = c.copy()
            
            # Simple gradient descent on coefficients to reduce overlap
            for _ in range(300):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                h = np.dot(c, basis) * dx
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_points / 2) / h_sum
                
                c5 = evaluate_erdos_solution(h, dx)
                
                # Estimate gradient using finite differences
                epsilon = 1e-4
                grad = np.zeros(n_basis)
                for i in range(n_basis):
                    c_plus = c.copy()
                    c_plus[i] += epsilon
                    h_plus = np.dot(c_plus, basis) * dx
                    h_plus = np.clip(h_plus, 0, 1)
                    h_sum_plus = np.sum(h_plus)
                    if h_sum_plus > 0:
                        h_plus = h_plus * (n_points / 2) / h_sum_plus
                    c5_plus = evaluate_erdos_solution(h_plus, dx)
                    grad[i] = (c5_plus - c5) / epsilon
                
                # Move in descending direction with small step
                lr = 0.1
                c_new = c - lr * grad
                c_new = np.clip(c_new, 0, 2)
                c = c_new
                
                # Check if this improves the solution
                h_new = np.dot(c, basis) * dx
                h_new = np.clip(h_new, 0, 1)
                h_sum_new = np.sum(h_new)
                if h_sum_new > 0:
                    h_new = h_new * (n_points / 2) / h_sum_new
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c = c.copy()
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
            return best_h, best_c5
        
        # Try with different initial coefficient values
        h_gen, c5_gen = solve_basis_optimization(basis, 0.5)
        if c5_gen < best_c5:
            best_h = h_gen
            best_c5 = c5_gen
            
        # Try harder optimization
        # Direct optimization on coarse grid first
        h_coarse = np.ones(n_points) * 0.5
        coarse_indices = np.linspace(0, n_points-1, 10, dtype=int)
        for i in range(len(coarse_indices) - 1):
            h_coarse[coarse_indices[i]:coarse_indices[i+1]] = 0.5
            
        # Refine with gradient-based method
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_coarse, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final local refinement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
