import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a more targeted construction
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization with adaptive step size
    max_iter = int(1e5)
    eta = 0.5  # Initial step size
    tol = 1e-6
    
    h_current = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5 - tol:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Find the shift k that maximizes overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        # The maximum overlap occurs at shift k_max (relative to center)
        # For full correlation, k_max is indexed at n_points-1 to 2*n_points-2
        # But actual shift is k = k_max - (n_points - 1)  
        shift_k = k_max - (n_points - 1)
        
        # Compute the gradient at current h for that shift:
        # gradient = 1 - h_shifted_k where the shift is k
        if shift_k >= 0:
            h_shifted = np.concatenate([h_current[shift_k:], np.zeros(shift_k)])
        else:
            h_shifted = np.concatenate([np.zeros(-shift_k), h_current[:shift_k]])
        gradient = 1 - h_shifted
        
        # Find the extreme point that minimizes linearization
        # This is equivalent to finding the top n_points/2 indices
        # where gradient is largest
        top_indices = np.argsort(gradient)[::-1][:n_points//2]
        
        # Create the extreme point h_extreme
        h_extreme = np.zeros(n_points)
        h_extreme[top_indices] = 1.0
        
        # Update with Frank-Wolfe step
        h_new = (1 - eta) * h_current + eta * h_extreme
        
        # Project and normalize
        h_new = project_to_feasible(h_new, n_points)
        h_current = h_new
        
        # Adaptive step size
        if iteration % 10 == 0:
            eta = max(0.01, eta * 0.99)
            
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
