import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create binary pattern with exactly n_points/2 ones
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        np.random.shuffle(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    max_iter = 10000
    step_size = 0.5
    perturb_rate = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current maximum overlap shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        k_shift = k_max - (n_points - 1)
        # Get the part that gets correlated for shift k_max
        if k_shift > 0:
            corr_part = best_h[k_shift:] * (1 - best_h[:n_points-k_shift])
        else:
            corr_part = best_h[:n_points+k_shift] * (1 - best_h[-k_shift:])
        
        # Create linear approximation around the current h
        linear_obj = 1 - best_h
        
        # Find the extreme point that minimizes linear object
        
        # Get the indices to set to 1 (top n/2 largest values in linear_obj)
        indices = np.argsort(linear_obj)[::-1][:n_points//2]
        h_extreme = np.zeros(n_points)
        h_extreme[indices] = 1.0
        
        # Update using Frank-Wolfe rule with perturbation
        h_new = (1 - step_size) * best_h + step_size * h_extreme
        if np.random.random() < perturb_rate:
            # Add small random noise
            h_new += np.random.normal(0, 0.01, n_points)
        h_new = project_to_feasible(h_new, n_points)
        
        # Check if this improves C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            if iteration > 0 and iteration % 100 == 0:
                step_size = max(0.01, step_size * 0.99)
    
    # Final refinement with gradient descent 
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
