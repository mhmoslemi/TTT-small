import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with randomized order
    max_iterations = 500
    tolerance = 1e-6
    current_h = h_start.copy()
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Shuffle indices for randomized update order
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for i in indices:
            if time() - start_time > budget_s * 0.8:
                break
            
            # Store current value
            old_h_i = current_h[i]
            
            # One-dimensional search for optimal h[i]
            best_h_i = old_h_i
            best_overlap = float('inf')
            
            # Simple grid search
            search_points = 100
            h_range = [max(0, old_h_i - 0.1), min(1, old_h_i + 0.1)]
            if h_range[1] - h_range[0] < 1e-6:
                h_range[1] = min(1, h_range[0] + 0.1)
            
            search_h = np.linspace(h_range[0], h_range[1], search_points)
            for candidate_h_i in search_h:
                # Temporarily set current value
                current_h[i] = candidate_h_i
                
                # Project to ensure feasibility
                current_h = project_to_feasible(current_h, n_points)
                
                # Evaluate the overlap
                overlap = evaluate_erdos_solution(current_h, dx)
                if overlap < best_overlap:
                    best_overlap = overlap
                    best_h_i = candidate_h_i
            
            # Apply the best value
            current_h[i] = best_h_i
            current_h = project_to_feasible(current_h, n_points)
            
            if best_overlap < best_c5 - tolerance:
                improved = True
                best_c5 = best_overlap
                best_h = current_h.copy()
        
        # If no improvement, break
        if not improved and iteration > 10:
            break
    
    # SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h.copy(), method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
