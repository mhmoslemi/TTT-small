import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize tabu list
    tabu_list = []
    tabu_length = 10
    
    # Simulated annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    max_iterations = 10000
    
    # Larger cycle for more diverse exploration
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb one element
        i = np.random.randint(0, n_points)
        h_trial = best_h.copy()
        
        # Randomly choose from {0, 0.5, 1} to encourage binary-like solutions
        new_val = np.random.choice([0.0, 0.5, 1.0])
        h_trial[i] = new_val
        
        # Ensure integral constraint is maintained
        h_trial_sum = np.sum(h_trial)
        if h_trial_sum > 0:
            h_trial = h_trial * (n_points / 2) / h_trial_sum
            
        # Clip values to be within [0,1]
        h_trial = np.clip(h_trial, 0, 1)
        
        # Evaluate
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        
        # Accept or reject
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial
        else:
            # Accept with probability based on simulated annealing
            delta = c5_trial - best_c5
            temp = initial_temp * (cooling_rate ** iter_num)
            if temp > min_temp and np.random.rand() < np.exp(-delta / temp):
                # Accept worse solution
                best_h = h_trial.copy()
                best_c5 = c5_trial
        
        # Update tabu list
        if len(tabu_list) >= tabu_length:
            tabu_list.pop(0)
        tabu_list.append(tuple(h_trial))
    
    # Local refinement with Nelder-Mead
    def obj_func(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return evaluate_erdos_solution(h, dx)
    
    # Alternate refinement method (add small noise and optimize)
    h_refined = best_h.copy()
    for _ in range(20):
        if time.time() - start_time > budget_s * 0.98:
            break
        try:
            # Optimization with small perturbations
            h_perturb = h_refined + np.random.normal(0, 0.01, n_points)
            res = minimize(obj_func, h_perturb, method='Nelder-Mead', 
                          options={'maxiter': 50, 'adaptive': True})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    h_refined = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
            
    # Final check
    final_h = np.clip(h_refined, 0, 1)
    final_h_sum = np.sum(final_h)
    if final_h_sum > 0:
        final_h = final_h * (n_points / 2) / final_h_sum
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h.copy()
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
