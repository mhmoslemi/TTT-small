import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth symmetric initialization
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x))  # Smooth symmetric
        h_start = project_to_feasible(h_start, n_points)
    
    best_c5 = float('inf')
    best_h = h_start.copy()
    best_n_points = n_points
    dx = 2.0 / n_points
    
    # Try different configurations
    configs = [
        (30, 1000),
        (35, 1000),
        (40, 1000),
    ]
    
    for n_points_try, max_iter in configs:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        h_current = np.interp(np.linspace(0, 2, n_points_try), 
                             np.linspace(0, 2, len(h_start)), 
                             h_start)
        h_current = project_to_feasible(h_current, n_points_try)
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient approximation
            grad = np.zeros_like(h_current)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_current, dx_try)
            
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points_try)
                score_plus = evaluate_erdos_solution(h_plus, dx_try)
                grad[i] = (score_plus - base_score) / eps
            
            # Update
            step_size = 0.002
            h_current = h_current - step_size * grad
            h_current = project_to_feasible(h_current, n_points_try)
            
            score_current = evaluate_erdos_solution(h_current, dx_try)
            if score_current < best_c5:
                best_h = h_current.copy()
                best_c5 = score_current
                best_n_points = n_points_try
                
    # Final verification with refinement
    if best_h is not None and time() - start_time < budget_s * 0.9:
        h_refined = best_h.copy()
        dx_refined = 2.0 / best_n_points
        for _ in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx_refined)
            
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx_refined)
                grad[i] = (score_plus - base_score) / eps
            
            # Update with small step
            step_size = 0.0005
            h_refined = h_refined - step_size * grad
            h_refined = project_to_feasible(h_refined, best_n_points)

            score_refined = evaluate_erdos_solution(h_refined, dx_refined)
            if score_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = score_refined
                
    return (list(best_h), best_c5, best_n_points)
