import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize solution with structured approach
    h_current = h_start.copy()
    
    # Iterative coordinate descent with constraints
    max_iterations = 1000
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        # Gradient-like update using small perturbations
        h_new = h_current.copy()
        step_size = 0.01
        
        # Try different small perturbations to each component
        for i in range(n_points):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_trial = h_new.copy()
            h_trial[i] = min(1.0, max(0.0, h_trial[i] + step_size))
            
            # Normalize
            h_trial_sum = np.sum(h_trial)
            if h_trial_sum > 0:
                h_trial = h_trial * (n_points / 2) / h_trial_sum
            
            new_c5 = evaluate_erdos_solution(h_trial, dx)
            if new_c5 < current_c5:
                h_new = h_trial
                current_c5 = new_c5
                
        h_current = h_new
        
    # Final optimization with scipy
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
