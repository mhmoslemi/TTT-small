import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple initial solution
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = []
    max_no_improve = 50
    no_improve_count = 0
    
    # Neighborhood generation parameters
    perturbation_rate = 0.2
    perturbation_magnitude = 0.1
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.7:
        # Generate neighbors
        neighbors = []
        for _ in range(20):  # Generate 20 neighbor solutions
            if time.time() - start_time > budget_s * 0.7:
                break
                
            h_new = best_h.copy()
            indices_to_perturb = np.random.choice(n_points, 
                                                size=int(n_points * perturbation_rate), 
                                                replace=False)
            
            for i in indices_to_perturb:
                # Perturb h[i] by adding random amount
                perturbation = np.random.normal(0, perturbation_magnitude)
                h_new[i] += perturbation
                h_new[i] = np.clip(h_new[i], 0, 1)
            
            # Project back to satisfy sum constraint
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
                
            neighbors.append(h_new)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # Check if candidate is in tabu list
            is_tabu = False
            candidate_tuple = tuple(h_candidate.round(6))
            for tabu_h in tabu_list:
                if np.allclose(tabu_h, h_candidate, rtol=1e-5):
                    is_tabu = True
                    break
            
            # Accept neighbor if it's better or not tabu
            if not is_tabu and candidate_c5 < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = candidate_c5
        
        # If no suitable neighbor found, accept one anyway (with escape)
        if best_neighbor is None:
            # Set escape velocity for tabu
            escape_threshold = np.random.uniform(0.01, 0.1)
            h_candidate = neighbors[0]
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            # With some probability, accept even tabu solution
            if np.random.random() < escape_threshold:
                best_neighbor = h_candidate
                best_neighbor_c5 = candidate_c5
        
        if best_neighbor is not None:
            if best_neighbor_c5 < best_c5:
                best_h = best_neighbor
                best_c5 = best_neighbor_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
                
            # Add to tabu list
            tabu_list.append(best_h.copy())
            if len(tabu_list) > tabu_list_size:
                tabu_list.pop(0)
        else:
            no_improve_count += 1
            
        if no_improve_count > max_no_improve:
            break
    
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s * 0.9:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        num_refinements = 10
        for _ in range(num_refinements):
            if time.time() - start_time > budget_s * 0.9:
                break
            try:
                res = minimize(obj_func, best_h.copy(), method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_sum = np.sum(h_final)
    if h_sum > 0:
        h_final = h_final * (n_points / 2) / h_sum
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
