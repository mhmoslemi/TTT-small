import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with a simple gradient-based refinement to see if we can improve
    h_current = best_h.copy()
    
    # Gradient descent refinement with bounds
    def obj_func(h):
        h_proj = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h_proj, dx)
    
    def grad_func(h):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            grad[i] = (obj_func(h_plus) - obj_func(h_minus)) / (2 * eps)
        return grad
    
    # Use a smooth approximation for the gradient
    learning_rate = 1e-3
    max_iter = 500
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        try:
            grad = grad_func(h_current)
            # Update with gradient descent
            h_new = h_current - learning_rate * grad

            perturbation = np.random.normal(0, 0.01, n_points)
