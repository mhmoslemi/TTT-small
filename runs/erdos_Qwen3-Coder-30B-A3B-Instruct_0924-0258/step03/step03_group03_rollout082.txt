import numpy as np
from scipy.signal import correlate
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(values, alpha=10.0):
    """Smooth approximation of max function"""
    return np.log(np.sum(np.exp(alpha * values))) / alpha

def smooth_overlap(h, dx, alpha=10.0):
    """Compute smooth approximation of C5"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Convert to vector form (excluding padding)
    corr_vec = corr[len(h)-1:]
    return smooth_max(corr_vec, alpha) * dx

def trust_region_optimize(h0, n_points, dx, max_iter=1000, trust_radius=0.1):
    """Optimize using trust region method with projections"""
    h_best = h0.copy()
    c5_best = smooth_overlap(h_best, dx)
    
    for iteration in range(max_iter):
        # Evaluate current point
        c5_current = smooth_overlap(h_best, dx)
        
        # Simple gradient approximation via finite differences
        grad = np.zeros_like(h_best)
        eps = 1e-5
        for i in range(len(h_best)):
            h_plus = h_best.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            c5_plus = smooth_overlap(h_plus, dx)
            grad[i] = (c5_plus - c5_current) / eps

        # Trust region update
        delta_h = -grad * 0.01
        h_test = h_best + delta_h
        h_test = project_to_feasible(h_test, n_points)
        
        c5_test = smooth_overlap(h_test, dx)
        if c5_test < c5_best:
            h_best = h_test
            c5_best = c5_test

        # Check convergence
        if np.linalg.norm(delta_h) < 1e-6:
            break

    return h_best, c5_best
