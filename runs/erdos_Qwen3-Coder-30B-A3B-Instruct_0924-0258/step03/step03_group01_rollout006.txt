import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population with diverse candidates
    population_size = 20
    population = [h_start.copy()]
    
    for _ in range(population_size - 1):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Training iterations
    max_iterations = 60
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current population
        population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Find best in current population
        best_idx = np.argmin(population_c5)
        if population_c5[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = population_c5[best_idx]
            
        # Select elite candidates
        elite_indices = np.argsort(population_c5)[:5]
        elite_candidates = [population[i] for i in elite_indices]
        
        # Generate new candidates through perturbations
        new_population = elite_candidates[:]
        for _ in range(population_size - 5):
            parent = elite_candidates[np.random.randint(0, 5)]
            h_new = parent + np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_population.append(h_new)
        
        population = new_population
    
    # Final convex refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Try convex optimization on best current solution
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Constraint for overlap - use an upper bound to avoid
            # overly complex constraints
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            max_overlap = np.max(corr) * dx
            
            # Set an upper bound that's reasonable
            upper_bound = max_overlap * 1.1
            for k in range(2 * n_points - 1):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= upper_bound * dx)
            
            try:
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
