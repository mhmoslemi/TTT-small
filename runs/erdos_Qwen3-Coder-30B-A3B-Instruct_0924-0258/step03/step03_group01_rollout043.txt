import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population size and parameters
    pop_size = 20
    num_generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1
    
    # Initialize population with diverse feasible candidates
    def create_feasible_candidate(base_h):
        # Add small random perturbations
        h_pert = base_h + np.random.normal(0, 0.05, n_points)
        # Clamp to [0, 1]
        h_pert = np.clip(h_pert, 0, 1)
        # Scale to correct sum
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        return h_pert
    
    population = [create_feasible_candidate(h_start) for _ in range(pop_size)]
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_individual = population[np.argmin(fitness)]
    best_c5 = min(fitness)
    
    # Genetic Algorithm loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection: top 50%
        sorted_indices = np.argsort(fitness)
        selected_indices = sorted_indices[:pop_size // 2]
        selected = [population[i] for i in selected_indices]
        selected_fitness = [fitness[i] for i in selected_indices]
        
        # Create new population
        new_population = selected[:]
        
        # Crossover: create offspring
        num_crossover = int(pop_size * crossover_rate)
        for _ in range(num_crossover):
            if len(selected) < 2:
                break
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Repair and maintain feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
            
        # Mutation
        num_mutate = int(pop_size * mutation_rate)
        for _ in range(num_mutate):
            if len(new_population) == 0:
                break
            idx = random.randint(0, len(new_population) - 1)
            h_mut = new_population[idx].copy()
            mu = np.random.normal(0, 0.03, n_points)
            h_mut += mu
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            new_population[idx] = h_mut
            
        # Ensure population size
        while len(new_population) < pop_size:
            new_population.append(create_feasible_candidate(h_start))
            
        # Evaluate new population
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
            
        # Local SQP refinement every few generations
        if generation % sqp_interval == 0:
            refined_pop = []
            for i, h in enumerate(population):
                if i < 3:  # Refine top 3
                    def objective(h_refine):
                        return evaluate_erdos_solution(h_refine, dx)
                    
                    def constraint_sum(h_refine):
                        return np.sum(h_refine) - n_points / 2.0
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    
                    try:
                        res = minimize(objective, h, method='SLSQP',
                                       bounds=bounds, constraints=cons_sum,
                                       options={'maxiter': 20, 'ftol': 1e-6})
                        if res.success:
                            h_refined = res.x
                            refined_h = h_refined
                        else:
                            refined_h = h  # Keep original if optimization fails
                    except:
                        refined_h = h  # Keep original if optimization fails
                    refined_pop.append(refined_h)
                else:
                    refined_pop.append(h)
            population = refined_pop
            fitness = [evaluate_erdos_solution(h, dx) for h in population]
            # Update best solution if needed
            min_fit_idx = np.argmin(fitness)
            if fitness[min_fit_idx] < best_c5:
                best_c5 = fitness[min_fit_idx]
                best_individual = population[min_fit_idx].copy()
    
    # Final check of best solution
    final_c5 = evaluate_erdos_solution(best_individual, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_individual), best_c5, n_points)
