import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization approach
    # Initialize swarm
    swarm_size = 50
    max_iter = 1000
    inertia_w = 0.729
    cognitive_c = 1.49445
    social_c = 1.49445
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(swarm_size):
        h = project_to_feasible(np.random.rand(n_points) * 0.5 + 0.25, n_points)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(pbest_values)
    global_best = pbest[global_best_idx].copy()
    global_best_value = pbest_values[global_best_idx]
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                           cognitive_c * r1 * (pbest[i] - particles[i]) + 
                           social_c * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update pbest
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = particles[i].copy()
                global_best_value = current_value
        
        # Early stopping if no improvement
        if iteration > 50 and global_best_value == best_c5:
            break
    
    # Refine with gradient based method
    def refine_objective(h):
        h_proj = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h_proj, dx)
    
    # Try to improve the global best with L-BFGS
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        # Run one more refinement if enough time
        try:
            res = minimize(refine_objective, global_best, method='L-BFGS-B',
                          bounds=[(0,1) for _ in range(n_points)], 
                          options={'maxiter': 100})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < global_best_value:
                    global_best = h_refined.copy()
                    global_best_value = c5_refined
        except:
            pass
            
    return (list(global_best), global_best_value, n_points)
