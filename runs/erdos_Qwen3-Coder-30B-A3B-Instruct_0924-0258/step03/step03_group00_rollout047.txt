import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple spike to satisfy sum constraint
    
    dx = 2.0 / n_points
    
    # Ensure sum constraint is satisfied
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    max_tabu_size = 20
    perturbation_rate = 0.2
    local_refinement_steps = 10
    max_no_improvement = 50
    
    no_improvement_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        if no_improvement_count >= max_no_improvement:
            break
            
        # Generate neighbors by perturbing 20% of indices
        indices_to_perturb = np.random.choice(n_points, 
                                            size=int(n_points * perturbation_rate), 
                                            replace=False)
        
        neighbors = []
        for _ in range(10):  # Create 10 neighbors
            h_neighbor = best_h.copy()
            for idx in indices_to_perturb:
                # Perturb by ±0.1 with clamping
                perturbation = np.random.uniform(-0.1, 0.1)
                h_neighbor[idx] = np.clip(h_neighbor[idx] + perturbation, 0, 1)
            
            # Project to maintain sum constraint
            h_sum = np.sum(h_neighbor)
            if h_sum > 0:
                h_neighbor = h_neighbor * (n_points / 2) / h_sum
            
            neighbors.append(h_neighbor)
        
        # Evaluate neighbors and find best feasible one
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Check if in tabu list (simplified hash-based check)
            h_tuple = tuple(np.round(h_candidate, decimals=6))
            if h_tuple in tabu_list:
                # With low probability, escape tabu
                if np.random.random() > 0.7:
                    escape_cost = np.random.uniform(0.01, 0.1)
                    candidate_c5 = evaluate_erdos_solution(h_candidate, dx) + escape_cost
                else:
                    continue
            else:
                candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            if candidate_c5 < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = candidate_c5
        
        # If no improvement found, continue
        if best_neighbor is None:
            no_improvement_count += 1
            continue
            
        # Update tabu list
        h_tuple = tuple(np.round(best_h, decimals=6))
        if len(tabu_list) >= max_tabu_size:
            tabu_list.pop(0)
        tabu_list.append(h_tuple)
        
        # Move to best neighbor
        best_h = best_neighbor
        if best_neighbor_c5 < best_c5:
            best_c5 = best_neighbor_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Local refinement step
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for _ in range(local_refinement_steps):
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                res = minimize(obj_func, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 10, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x.copy()
                    h_refined = np.clip(h_refined, 0, 1)
                    h_sum = np.sum(h_refined)
                    if h_sum > 0:
                        h_refined = h_refined * (n_points / 2) / h_sum
                    refined_c5 = evaluate_erdos_solution(h_refined, dx)
                    if refined_c5 < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = refined_c5
                        no_improvement_count = 0
                else:
                    break
            except:
                break
                
    return (list(best_h), best_c5, n_points)
