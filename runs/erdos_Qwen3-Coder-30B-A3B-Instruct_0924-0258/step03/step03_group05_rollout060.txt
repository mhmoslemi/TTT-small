import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with a structured approach: try to make a piecewise constant function
    # that is as uniform as possible while satisfying the constraints.
    
    # We will do a grid search over some key candidate functions
    candidates = []
    
    # Try different fraction of the interval that's 1 vs 0
    for frac in np.linspace(0.1, 0.9, 9):
        # Create a function that is 1 on a fraction of points, 0 elsewhere
        # But ensure the integral constraint ∫h=1 is satisfied
        h_candidate = np.zeros(n_points)
        num_ones = int(frac * n_points)
        h_candidate[:num_ones] = 1.0
        # If we don't have enough, fill up with smaller values to make integral 1
        total = np.sum(h_candidate)
        if total > 0:
            h_candidate = h_candidate * (n_points/2) / total
        candidates.append(h_candidate)
        
        # Try a variation with more concentrated values
        h_candidate2 = np.ones(n_points) * (1.0 / n_points) * 2.0
        h_candidate2[:num_ones] = 1.5
        h_candidate2[num_ones:] = 0.5
        h_candidate2 = np.clip(h_candidate2, 0, 1)
        total = np.sum(h_candidate2)
        if total > 0:
            h_candidate2 = h_candidate2 * (n_points/2) / total
        candidates.append(h_candidate2)
    
    # Also try some simple smooth-like constructions
    x_vals = np.linspace(0, 2, n_points)
    for a in [0.1, 0.3, 0.5, 0.7, 0.9]:
        h_smooth = np.exp(-((x_vals - 1) ** 2) / (2 * a ** 2))
        total = np.sum(h_smooth)
        if total > 0:
            h_smooth = h_smooth * (n_points/2) / total
        candidates.append(h_smooth)

    for h_cand in candidates:
        c5_cand = evaluate_erdos_solution(h_cand, dx)
        if c5_cand < best_c5:
            best_h = h_cand
            best_c5 = c5_cand

    # Now attempt local refinement with a different heuristic
    # If time permits, try a convex constraint approach to get better bounds
    if time() - start_time < budget_s * 0.6:
        # Try to do a more structured optimization
        # Using a larger set of basis functions and optimizing for overlap
        try:
            # Start with our best solution so far
            h_ref = best_h.copy()
            
            # Least squares optimization over shifted overlaps with a linear model
            # All constraints stay the same
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
                
            # Simple bounds
            bounds = [(0, 1) for _ in range(n_points)]

            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

                new_h[[i1, i2]] = new_h[[i2, i1]]
                # Renormalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        except Exception:
            pass

    # Final validation and return
    final_c5 = evaluate_erdos_solution(best_h, dx)

    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
