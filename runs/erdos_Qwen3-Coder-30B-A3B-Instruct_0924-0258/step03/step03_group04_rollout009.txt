import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more informed initial pattern
        # Try a periodic-like shape that might help avoid local minima
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Inner optimization: gradient descent with projections
    max_iter = 100000
    learning_rate = 1e-3
    decay_rate = 0.99995
    p_perturb = 0.05
    perturb_magnitude = 0.05
    
    # Iterate with time control
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for current h
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        # The shift k that gives maximum overlap
        k = max_idx - (n_points - 1)
        
        # Compute subgradient of that maximum overlap
        grad = np.zeros(n_points)
        # For shift k, we only consider valid correlation indices
        if k >= 0:
            # h[i] contributes to overlap with (1-h[i+k]) when i+k < n_points
            valid_indices = np.arange(n_points - k)
            grad[valid_indices] = 1 - best_h[valid_indices + k]
        else:
            # k < 0: h[i+k] contributes to overlap with (1-h[i]) when i+k >= 0
            valid_indices = np.arange(-k, n_points)
            grad[valid_indices] = -(1 - best_h[valid_indices - k])
            
        # Gradient step
        new_h = best_h - learning_rate * grad
        # Project to feasible region
        new_h = project_to_feasible(new_h, n_points)
        
        # Perturbation with some probability
        if np.random.random() < p_perturb:
            perturbation = np.random.normal(0, perturb_magnitude, n_points)
            new_h = new_h + perturbation
            new_h = project_to_feasible(new_h, n_points)
            
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        # Update learning rate
        learning_rate *= decay_rate
        
    return (list(best_h), best_c5, n_points)
