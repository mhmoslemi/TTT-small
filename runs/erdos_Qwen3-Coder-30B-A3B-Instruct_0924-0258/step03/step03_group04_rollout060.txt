import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time < 10:
        return (list(best_h), best_c5, n_points)
    
    # Start with small perturbations based on initial solution
    h_current = h_start.copy()
    
    # Gradient-based refinement with bounds
    def obj(h):
        h_proj = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h_proj, dx)
    
    # Select a small subset of iterations for high-quality descent
    max_iter = 150
    patience = 0
    best_val = best_c5
    patience_limit = 30
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Construct bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B for gradient-based optimization
        res = minimize(obj, h_current, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': 30, 'ftol': 1e-8, 'gtol': 1e-6})
        
        if res.success:
            h_new = project_to_feasible(res.x, n_points)
            new_val = evaluate_erdos_solution(h_new, dx)
            
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                patience = 0
            else:
                patience += 1
                
            h_current = h_new
            
            if patience >= patience_limit:
                # Random perturbation
                perturbation_magnitude = 0.05
                noise = np.random.normal(0, perturbation_magnitude, n_points)
                h_current += noise
                h_current = project_to_feasible(h_current, n_points)
                patience = 0
        else:
            # Fallback: random perturbation if optimization fails
            noise = np.random.normal(0, 0.05, n_points)
            h_current += noise
            h_current = project_to_feasible(h_current, n_points)
    
    return (list(best_h), best_val, n_points)
