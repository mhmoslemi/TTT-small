import numpy as np
from scipy.optimize import minimize
import time
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple construction to satisfy integral constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = deque(maxlen=tabu_list_size)
    memory = set()
    
    # Local search parameters
    max_local_iter = 10
    perturbation_rate = 0.2
    perturbation_magnitude = 0.1
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        if no_improvement_count >= max_no_improvement:
            break
            
        # Generate neighbors by perturbing random indices
        neighbor_h = best_h.copy()
        indices_to_perturb = np.random.choice(n_points, int(n_points * perturbation_rate), replace=False)
        
        for i in indices_to_perturb:
            perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude)
            neighbor_h[i] += perturbation
            neighbor_h[i] = np.clip(neighbor_h[i], 0, 1)
        
        # Project to satisfy integral constraint
        h_sum = np.sum(neighbor_h)
        if h_sum > 0:
            neighbor_h = neighbor_h * (n_points / 2) / h_sum
        
        # Check if neighbor is in tabu list
        neighbor_tuple = tuple(neighbor_h.round(6))
        if neighbor_tuple in memory:
            # Apply escape mechanism with low probability
            if np.random.random() < 0.1:
                # Accept the tabu solution with small probability
                pass
            else:
                continue
        
        # Evaluate the neighbor
        neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
        
        # Update best solution if improved
        if neighbor_c5 < best_c5:
            best_h = neighbor_h.copy()
            best_c5 = neighbor_c5
            no_improvement_count = 0
            
            # Add to tabu list
            tabu_list.append(neighbor_tuple)
            memory.add(neighbor_tuple)
        else:
            no_improvement_count += 1
            
        # Local refinement step (SLSQP)
        if iteration % 5 == 0 and time.time() - start_time < budget_s * 0.9:
            def obj_func(h):
                h_clipped = np.clip(h, 0, 1)
                h_sum = np.sum(h_clipped)
                if h_sum > 0:
                    h_normalized = h_clipped * (n_points / 2) / h_sum
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Perturb current best
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    # Final cleaning
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
