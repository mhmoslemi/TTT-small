import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradients(h, dx):
    """Compute gradients of C5 with respect to h values using finite differences"""
    eps = 1e-5
    base_score = evaluate_erdos_solution(h, dx)
    grad = np.zeros_like(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - base_score) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameter settings
    max_iter = 300
    step_size = 0.01
    max_segments = 40
    min_segments = 10
    
    # Initialize
    if 'initial_h_values' in globals() and len(initial_h_values) > 0:
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 35
        h_current = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_current, dx)
    best_c5 = current_c5
    best_h = h_current.copy()
    best_n_points = n_points
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradients(h_current, dx)
        
        # Update with gradient descent
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5
                best_n_points = n_points
    
    # Final refinement with smaller step size
    step_size *= 0.1
    for iteration in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        grad = compute_gradients(h_current, dx)
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5
                best_n_points = n_points
    
    return (list(best_h), best_c5, best_n_points)
