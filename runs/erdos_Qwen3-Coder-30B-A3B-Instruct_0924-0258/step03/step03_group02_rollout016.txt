import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Cluster-based optimization
    max_iter = 200
    cluster_count = 10
    recluster_freq = 50
    stagnation_limit = 100
    stagnation_count = 0
    
    # Initialize clusters
    h_current = h_start.copy()
    prev_best_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Re-cluster every recluster_freq iterations
        if iteration % recluster_freq == 0:
            # Compute overlap vectors for different shifts
            overlap_vectors = []
            for k in range(n_points):
                if k == 0:
                    vec = h_current * (1 - h_current)
                else:
                    vec = h_current[:-k] * (1 - h_current[k:])
                overlap_vectors.append(np.sum(vec))
            
            # Cluster shifts based on their overlap characteristics
            overlap_array = np.array(overlap_vectors).reshape(-1, 1)
            kmeans = KMeans(n_clusters=cluster_count, random_state=seed).fit(overlap_array)
            cluster_labels = kmeans.labels_
            
        # Optimize each cluster
        for cluster_id in range(cluster_count):
            # Select indices belonging to the current cluster
            indices = np.where(cluster_labels == cluster_id)[0]
            if len(indices) == 0:
                continue
                
            # Objective function for this cluster
            def cluster_obj(h_sub):
                total = 0.0
                for k in indices:
                    if k == 0:
                        total += np.sum(h_sub * (1 - h_sub))
                    else:
                        total += np.sum(h_sub[:-k] * (1 - h_sub[k:]))
                return -total * dx  # Negate for maximization
            
            # Optimize using trust-constr
            bounds = [(0, 1) for _ in range(n_points)]
            try:
                # Only optimize the sub-vector indices
                h_sub = h_current.copy()
                res = minimize(cluster_obj, h_sub, method='trust-constr',
                              bounds=bounds, options={'maxiter': 20})
                if res.success:
                    h_current = project_to_feasible(res.x, n_points)
                    c5_new = evaluate_erdos_solution(h_current, dx)
                    if c5_new < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_new
                        stagnation_count = 0
                    else:
                        stagnation_count += 1
            except Exception:
                pass
        
        # Re-evaluate global best
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            break
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            def refine_obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_con(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_con}
            
            res = minimize(refine_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final projection to ensure feasibility
    best_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
