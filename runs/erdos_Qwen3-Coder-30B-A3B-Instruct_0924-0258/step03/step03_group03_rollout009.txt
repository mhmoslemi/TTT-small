import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_c5 = evaluate_erdos_solution(h_start, dx)

    resolutions = [25, 30, 35, 40, 45]
    
    # Trust region optimization parameters
    trust_radius = 0.1
    max_trust_steps = 50
    min_trust_radius = 1e-6
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate if needed

            x_new = np.linspace(0, 2, n_points_try)
            h_interp = np.interp(x_new, x_orig, h_start)
            h_try = project_to_feasible(h_interp, n_points_try)

            h_try = h_start.copy()

        for tr_step in range(max_trust_steps):

            grad = np.zeros_like(h_try)
