import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different n_points if we have time
    if budget_s > 400:
        n_points = 70
        dx = 2.0 / n_points
        # Initialize to the best solution from lower resolution
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Try to improve initial configuration with local optimization
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Use trust-constr for better results
        res = minimize(evaluate_erdos_solution, h_start, 
                       args=(dx,), method='trust-constr',
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'gtol': 1e-7, 'xtol': 1e-7})
        if res.success:
            h_start = np.clip(res.x, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            current_c5 = evaluate_erdos_solution(h_start, dx)
            if current_c5 < best_c5:
                best_h = h_start
                best_c5 = current_c5
    except:
        pass
    
    # Perturbation-based search
    if time() - start_time > budget_s * 0.7:
        # Early exit if time is running tight
        return (list(best_h), best_c5, n_points)
    
    # Local search with multi-start strategy
    n_starts = 5
    for _ in range(n_starts):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Slightly perturb the current best
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local optimization
        try:
            res = minimize(evaluate_erdos_solution, h_perturbed, 
                           args=(dx,), method='SLSQP',
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 30, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final local optimization
    try:
        res = minimize(evaluate_erdos_solution, best_h, 
                       args=(dx,), method='L-BFGS-B',
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Ensure constraints
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
