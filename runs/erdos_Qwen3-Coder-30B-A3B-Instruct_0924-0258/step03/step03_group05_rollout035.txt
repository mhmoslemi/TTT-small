import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with constraint handling
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_integral}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different optimization strategies
    max_iter = 50
    step_sizes = [0.01, 0.005, 0.001]
    
    for step_idx, step_size in enumerate(step_sizes):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start with current best
        h_current = best_h.copy()
        
        # Simple gradient-like steps to reduce overlap
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient approximation
            grad = np.zeros(n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)  # renormalize
            
            current_c5 = evaluate_erdos_solution(h_current, dx)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] = min(1.0, h_current[i] + eps)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # renormalize
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                # Approximate gradient
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < current_c5:
                h_current = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_current.copy()
    
    # Final optimization with scipy minimize
    if time() - start_time < budget_s * 0.9:
        try:
            # Add bounds and constraints
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 20, 'ftol': 1e-7})
            
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                final_c5 = evaluate_erdos_solution(h_final, dx)
                if final_c5 < best_c5:
                    best_h = h_final
                    best_c5 = final_c5
        except:
            pass
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
