import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use synthetic periodic function as starting point
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with random perturbations
    max_iter = 50000
    learning_rate = 1e-3
    decay_rate = 0.9999
    p_perturb = 0.1
    perturb_magnitude = 0.05
    
    h_current = h_start.copy()
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_overlap_idx = np.argmax(corr)
        max_overlap_val = corr[max_overlap_idx]
        
        # Get the shift that causes maximum overlap
        k = max_overlap_idx - (n_points - 1)
        
        # Compute subgradient of maximum overlap w.r.t h
        grad = np.zeros(n_points)
        
        if k >= 0:
            # Valid indices: h[i] contributes to overlap with h[i+k]
            valid_indices = np.arange(n_points - k)
            grad[valid_indices] = 1 - h_current[valid_indices + k]
        else:
            # k < 0
            valid_indices = np.arange(-k, n_points)
            grad[valid_indices] = 1 - h_current[valid_indices + k]
            
        # Update using gradient descent
        h_new = h_current - learning_rate * grad
        
        # Apply random perturbations occasionally
        if np.random.random() < p_perturb:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_new = h_new + perturb
            
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h_current = h_new.copy()
        learning_rate *= decay_rate
        
        if no_improvement_count > max_no_improvement:
            # Sometimes re-randomize to escape local minima
            if np.random.random() < 0.1:
                h_current = project_to_feasible(
                    np.random.rand(n_points) * 0.5, n_points)
            
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
