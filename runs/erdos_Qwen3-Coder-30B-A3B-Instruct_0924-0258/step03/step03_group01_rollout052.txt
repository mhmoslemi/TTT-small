import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Initialize with structured pattern
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Nelder-Mead parameters
    population_size = 15
    max_iter = 20
    
    # Create initial population
    population = [h_start]
    for _ in range(population_size - 1):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    simplex_size = n_points + 1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Sort by fitness (ascending)
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new simplex
        simplex = population[:simplex_size]
        simplex_fitness = fitness[:simplex_size]
        
        # Nelder-Mead steps
        # Calculate centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = centroid + 2 * (centroid - simplex[-1])
        xr = np.clip(xr, 0, 1)
        xr = xr * (n_points/2) / np.sum(xr)
        
        fr = evaluate_erdos_solution(xr, dx)
        
        # Expansion
        if fr < simplex_fitness[0]:
            xe = centroid + 2 * (xr - centroid)
            xe = np.clip(xe, 0, 1)
            xe = xe * (n_points/2) / np.sum(xe)
            fe = evaluate_erdos_solution(xe, dx)
            if fe < fr:
                simplex[-1] = xe
                simplex_fitness[-1] = fe
            else:
                simplex[-1] = xr
                simplex_fitness[-1] = fr
        elif fr < simplex_fitness[-2]:
            simplex[-1] = xr
            simplex_fitness[-1] = fr
        else:
            # Contraction
            if fr < simplex_fitness[-1]:
                xc = centroid + 0.5 * (xr - centroid)
                xc = np.clip(xc, 0, 1)
                xc = xc * (n_points/2) / np.sum(xc)
                fc = evaluate_erdos_solution(xc, dx)
                if fc < simplex_fitness[-1]:
                    simplex[-1] = xc
                    simplex_fitness[-1] = fc
                else:
                    # Shrink
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                        simplex[i] = np.clip(simplex[i], 0, 1)
                        simplex[i] = simplex[i] * (n_points/2) / np.sum(simplex[i])
                        simplex_fitness[i] = evaluate_erdos_solution(simplex[i], dx)
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (n_points/2) / np.sum(simplex[i])
                    simplex_fitness[i] = evaluate_erdos_solution(simplex[i], dx)
        
        # Update population with new simplex
        for i in range(simplex_size):
            for j in range(len(population)):
                if np.allclose(population[j], simplex[i]):
                    break
                if j == len(population) - 1:
                    population[j] = simplex[i]
    
    # Gradient descent refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine the best solution with several restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
