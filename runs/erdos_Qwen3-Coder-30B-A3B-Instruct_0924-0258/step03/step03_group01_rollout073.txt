import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_max
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best_idx = 0
    global_best_val = best_c5
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        val = evaluate_erdos_solution(h, dx)
        personal_best_vals.append(val)
        if val < global_best_val:
            global_best_val = val
            global_best_idx = i

    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        inertia_weight = w_min + (w_max - w_min) * (1 - iteration / max_iter)
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia_weight * velocities[i] + 
                             c1 * r1 * (personal_best[i] - swarm[i]) + 
                             c2 * r2 * (personal_best[global_best_idx] - swarm[i]))
            swarm[i] += velocities[i]
            
            # Repair constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = val
                if val < global_best_val:
                    global_best_val = val
                    global_best_idx = i
        
        # Local refinement every 5 iterations
        if iteration % 5 == 0:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time > 20:
                for i in range(min(3, swarm_size)):
                    top_h = swarm[i]
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
                    bounds = [(0, 1) for _ in range(n_points)]
                    try:
                        res = minimize(objective, top_h, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 30, 'ftol': 1e-6})
                        if res.success:
                            h_refined = res.x
                            c5_refined = objective(h_refined)
                            if c5_refined < best_c5:
                                best_h = h_refined.copy()
                                best_c5 = c5_refined
                    except:
                        pass
    
    # Final refinement of global best
    if global_best_val < best_c5:
        best_h = swarm[global_best_idx].copy()
        best_c5 = global_best_val
    
    return (list(best_h), best_c5, n_points)
