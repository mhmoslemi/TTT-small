import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def one_dimensional_optimize(h, idx, dx, n_points):
    """Optimize single variable h[idx] using grid search"""
    original = h[idx]
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Grid search
    search_range = np.linspace(0, 1, 50)
    for val in search_range:
        h[idx] = val
        h_proj = project_to_feasible(h, n_points)
        c5 = evaluate_erdos_solution(h_proj, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_proj.copy()
    
    h[idx] = original
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with random order
    max_iterations = 500
    tolerance = 1e-6
    last_improvement = 0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Random order update
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for idx in indices:
            if time() - start_time > budget_s * 0.9:
                break
            h_after, c5_after = one_dimensional_optimize(best_h, idx, dx, n_points)
            if c5_after < best_c5 - tolerance:
                best_h = h_after
                best_c5 = c5_after
                improved = True
                last_improvement = iteration
                
        # Early stopping if no improvement in a while
        if not improved and iteration - last_improvement > 20:
            break
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
