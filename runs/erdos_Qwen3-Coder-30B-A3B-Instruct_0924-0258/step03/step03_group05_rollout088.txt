import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frequency domain approximation with wavelet basis
    # Using Haar wavelets for localization
    def haar_wavelets(n):
        """Generate Haar wavelet basis functions"""
        basis = []
        # Scaling function (constant)
        phi = np.ones(n)
        phi = phi / np.sqrt(np.dot(phi, phi))
        basis.append(phi)
        
        # Wavelet functions
        for j in range(1, int(np.log2(n)) + 1):
            for k in range(2**(j-1)):
                psi = np.zeros(n)
                start = 2**(j-1) * k
                end1 = start + 2**(j-2)
                end2 = start + 2**j
                psi[start:end1] = 1
                psi[end1:end2] = -1
                psi = psi / np.sqrt(np.dot(psi, psi))
                basis.append(psi)
        
        # Ensure we have exactly n_points basis functions
        while len(basis) < n_points:
            basis.append(np.random.randn(n_points))
            basis[-1] = basis[-1] / np.sqrt(np.dot(basis[-1], basis[-1]))
            
        return np.array(basis)
    
    # Optimization with basis functions
    max_iter = 300
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Construct basis
        if iteration == 0:
            basis = haar_wavelets(n_points)
        else:
            # Sample points from current best solution for wavelet construction
            # Use a simple interpolation approach to get better basis
            try:
                # Use current h as guide to create local bases  
                new_basis = []
                # Add current function as one basis (rescaled)
                h_rescaled = np.copy(best_h)
                h_rescaled = h_rescaled / np.sqrt(np.sum(h_rescaled**2))
                new_basis.append(h_rescaled)
                
                # Add some variations
                for i in range(n_points - 1):
                    v = np.random.randn(n_points)
                    v = v / np.sqrt(np.sum(v**2))
                    new_basis.append(v)
                
                basis = np.array(new_basis)
            except:
                basis = haar_wavelets(n_points)
        
        # Optimization variables: coefficients in basis
        coeffs = cp.Variable(n_points)
        
        # Problem: find coeffs such that max overlap is minimized
        # But directly optimizing would be complex
        # So we'll use a simpler SNR-like optimization approach  
        
        # 1. Find a feasible solution using a local search
        def local_search(local_h, n_points, dx):
            # Use a gradient as an approximation to the objective gradient
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            # Simple gradient descent-like step (not real gradient)
            trials = 100
            best_h_local = local_h.copy()
            best_obj = obj(best_h_local)
            
            for _ in range(trials):
                # Simple random perturbation
                h_trial = np.clip(best_h_local + np.random.randn(n_points) * 0.01, 0, 1)

                if obj_trial < best_obj:

                    best_h_local = h_trial

            return best_h_local, best_obj
            
        # Do a local search from current best

            best_c5 = obj_local

        # Try known constraint method: flat wavelet decomposition
        
        # This is approximation - just do one final refine step
        try:
            # Minimize with constraints using SciPy
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
                
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 10, 'ftol': 1e-6})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Early stopping condition
        if iteration > 10 and abs(best_c5 - evaluate_erdos_solution(best_h, dx)) < 1e-6:
            break
    
    # Final check 
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Make sure normalization is correct
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
