import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a periodic pattern that has sum = n_points/2
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temp = 0.1
    temp_decay = 0.99
    min_temp = 1e-6
    max_iter = 10000
    accept_count = 0
    total_iter = 0
    
    # Main SA loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        total_iter += 1
        
        # Create candidate by perturbing a random subset of points
        h_candidate = best_h.copy()
        
        # Choose number of points to perturb (5-10%)
        num_perturb = random.randint(max(1, n_points // 20), max(1, n_points // 5))
        indices = random.sample(range(n_points), num_perturb)
        
        # Apply perturbation
        for i in indices:
            # Add small random change
            delta = random.uniform(-0.1, 0.1)
            h_candidate[i] += delta
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept with simulated annealing probability
        delta = c5_candidate - best_c5
        if delta < 0 or random.random() < np.exp(-delta / temp):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            accept_count += 1
        
        # Update temperature
        if total_iter % 100 == 0:
            temp = max(min_temp, temp * temp_decay)
    
    # Final optimization with L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj_func(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(obj_func, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
