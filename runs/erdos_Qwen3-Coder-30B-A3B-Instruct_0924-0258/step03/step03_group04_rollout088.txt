import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy import signal

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia_weight = 0.7
    cognitive_param = 1.5
    social_param = 1.5
    
    # Initialize swarm
    particles = np.random.rand(swarm_size, n_points) * 0.5
    velocities = np.zeros((swarm_size, n_points))
    personal_best = particles.copy()
    personal_best_scores = np.full(swarm_size, np.inf)
    global_best_idx = 0
    global_best_score = np.inf
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            cognitive_param * r1 * (personal_best[i] - particles[i]) + 
                            social_param * r2 * (personal_best[global_best_idx] - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Keep in bounds [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Make sum equal to n_points/2
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
                if score < global_best_score:
                    global_best_score = score
                    global_best_idx = i
        
        # Early stopping
        if global_best_score < best_c5 * 0.999:
            # Refine with gradient descent
            def refined_obj(h):
                h_proj = project_to_feasible(h, n_points)
                return evaluate_erdos_solution(h_proj, dx)
            
            res = minimize(refined_obj, particles[global_best_idx], method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 10, 'ftol': 1e-9})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
    
    # Final cleanup and refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Local refinement of the best solution
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        if best_c5 < 0.4:  # Only try refinement if not too high
            res = minimize(final_obj, best_h, method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 50, 'ftol': 1e-12})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
