import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple spike configuration
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = []
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Local refinement parameters
    local_refinement_steps = 10
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.7:
        if no_improvement_count >= max_no_improvement:
            break
            
        # Generate neighbors
        neighbors = []
        for _ in range(20):  # Generate 20 neighbor solutions
            h_new = best_h.copy()
            # Perturb about 20% of indices
            indices_to_perturb = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            for i in indices_to_perturb:
                # Add small random perturbation
                perturbation = np.random.normal(0, 0.05)
                h_new[i] += perturbation
                # Clamp to [0,1]
                h_new[i] = np.clip(h_new[i], 0, 1)
            
            # Ensure sum constraint
            current_sum = np.sum(h_new)
            if current_sum > 0:
                h_new = h_new * (n_points / 2) / current_sum
            
            neighbors.append(h_new)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Check if this neighbor is in tabu list (simplified check)
            h_tuple = tuple(h_candidate)
            if h_tuple in tabu_list:
                continue
                
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = c5_candidate
        
        # Update tabu list and solution
        if best_neighbor is not None:
            # Add current best to tabu list
            current_tuple = tuple(best_h)
            if len(tabu_list) >= tabu_list_size:
                tabu_list.pop(0)
            tabu_list.append(current_tuple)
            
            best_h = best_neighbor
            best_c5 = best_neighbor_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    
    # Local refinement phase
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run local optimizers
    for _ in range(local_refinement_steps):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the current solution
        h_perturb = best_h + np.random.normal(0, 0.02, n_points)
        
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (n_points / 2) / final_sum
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
