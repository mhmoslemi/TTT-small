import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 500
    generations = 100
    elite_size = 10
    mutation_rate = 0.05
    crossover_rate = 0.8
    num_restarts = 10
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if time.time() - start_time > budget_s * 0.9:
            break
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        population.append(h)
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = population[:elite_size]
        best_in_gen = fitness[0]
        if best_in_gen < best_c5:
            best_h = population[0].copy()
            best_c5 = best_in_gen
        
        # Generate offspring
        while len(new_pop) < pop_size:
            if np.random.random() < crossover_rate and len(new_pop) + 2 <= pop_size:
                # Selection: tournament
                parent1_idx = np.random.choice(elite_size)
                parent2_idx = np.random.choice(elite_size)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # SBX crossover
                alpha = 0.5
                beta = 1 - alpha
                child1 = alpha * parent1 + beta * parent2
                child2 = beta * parent1 + alpha * parent2
                
                # Mutation
                for i in range(n_points):
                    if np.random.random() < mutation_rate:
                        child1[i] += np.random.normal(0, 0.05)
                        child2[i] += np.random.normal(0, 0.05)
                
                child1 = np.clip(child1, 0, 1)
                child2 = np.clip(child2, 0, 1)
                
                # Normalize
                h_sum1 = np.sum(child1)
                h_sum2 = np.sum(child2)
                if h_sum1 > 0:
                    child1 = child1 * (n_points / 2) / h_sum1
                if h_sum2 > 0:
                    child2 = child2 * (n_points / 2) / h_sum2
                    
                new_pop.append(child1)
                if len(new_pop) < pop_size:
                    new_pop.append(child2)
            else:
                # Random initialization
                h = h_start + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_points / 2) / h_sum
                new_pop.append(h)
        
        population = new_pop
    
    # Local refinement
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Try to improve the best solution found
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
            
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (n_points / 2) / final_sum
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
