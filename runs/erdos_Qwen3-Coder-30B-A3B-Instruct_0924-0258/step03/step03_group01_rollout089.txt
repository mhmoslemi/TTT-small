import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create initial guess with two clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to sum = n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based search with simulated annealing
    population_size = 20
    population = []
    for _ in range(population_size):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        # Renormalize
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    max_iterations = 30
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evolve population
        new_population = []
        for h in population:
            # Local refinement with SQP
            def obj(h_local):
                return evaluate_erdos_solution(h_local, dx)
            
            def con_sum(h_local):
                return np.sum(h_local) - n_points/2
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                else:
                    h_refined = h
            except:
                h_refined = h
            
            # Ensure feasibility
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            new_population.append(h_refined)
            
            # Update best
            c5_val = evaluate_erdos_solution(h_refined, dx)
            if c5_val < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_val
        
        # Perturb best and add to population
        h_pert = best_h + np.random.normal(0, temp, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Accept with probability based on temperature and difference
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5 or np.random.rand() < math.exp(-(c5_pert - best_c5) / temp):
            new_population.append(h_pert)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
        
        # Keep best solutions
        population = sorted(new_population, key=lambda x: evaluate_erdos_solution(x, dx))[:population_size]
        
        # Cool down
        temp *= cooling_rate
        
    # Final refinement
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def con_sum(h):
        return np.sum(h) - n_points/2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        h_final = best_h.copy()
        try:
            res = minimize(obj, h_final, method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
        # Ensure final normalization
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
