import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Interior point optimization
    def compute_gradients(h):
        """Compute gradients of overlap integrals with respect to h"""
        # Get all correlations for different shifts
        corr = np.correlate(h, 1 - h, mode='full')
        # Find k that maximizes correlation
        max_k = np.argmax(corr)
        # Compute gradients using finite differences around max_k
        # This is simplified gradient; exact gradient would require more work
        # For time efficiency, we use a simple approach
        
        # For each i compute how much changing h[i] affects max overlap
        grad = np.zeros_like(h)
        # We'll approximate by looking at local changes
        for i in range(len(h)):
            for k in range(1, len(h)):  # skip k=0
                if i + k < len(h):
                    grad[i] += 1 - 2 * h[i + k]
                # Also include the term where h[i] affects h[i-k] in other correlation
                if i >= k:
                    grad[i] += 1 - 2 * h[i - k]
        
        return grad
    
    # Optimization loop
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Use better starting solution
    h_current = best_h.copy()
    
    # Gradient adjustment constants
    learning_rate = 0.001
    momentum = 0.9
    
    prev_grad = np.zeros_like(h_current)
    
    while time() - start_time < budget_s * 0.95:
        iterations += 1
        if no_improvement_count >= max_no_improvement:
            break
            
        # Compute current gradient
        try:
            grad = compute_gradients(h_current)
        except:
            grad = np.random.normal(0, 0.01, len(h_current))
        
        # Add momentum
        grad = momentum * prev_grad + (1 - momentum) * grad
        prev_grad = grad.copy()
        
        # Update h
        h_new = h_current - learning_rate * grad
        
        # Project to [0,1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to maintain ∫h dx = 1
        h_sum = np.sum(h_new) * dx
        if h_sum > 0:
            h_new = h_new * (1.0 / h_sum)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h_current = h_new
        # Adaptive learning rate
        if time() - start_time > budget_s * 0.8:
            learning_rate *= 0.99
        else:
            learning_rate *= 0.999
    
    # Final refinement with scipy
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
        
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Ensure final result adheres to constraints exactly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
