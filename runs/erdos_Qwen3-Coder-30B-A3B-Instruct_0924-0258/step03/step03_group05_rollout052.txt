import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try larger discretization
    if budget_s > 300:
        n_points = 70
        dx = 2.0 / n_points
        # Interpolate the initial solution
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
    
    # Refinement function
    def refine_solution(h_guess, c_val):
        # Feasibility check with CVX
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1
        ]
        
        # Add overlap constraints for all shifts with proper indexing
        for k in range(1, n_points):
            valid_len = n_points - k
            if valid_len <= 0:
                break
            # For shift k, we compare h[i] with 1-h[i+k]
            # Valid range for i: [0, valid_len-1]
            # So we get h[0:valid_len] with 1-h[k:k+valid_len]
            overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_val
            constraints.append(overlap_constraint)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        return None
    
    # Local improvement with SLSQP
    def local_improve(h_guess):
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_guess, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                return h_refined
        except:
            pass
        return h_guess
    
    # Binary search for optimal C5
    c_low = 0.37
    c_high = 0.38085
    tolerance = 1e-5
    max_iterations = 20
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        h_feasible = refine_solution(best_h, c_mid)
        
        if h_feasible is not None:
            # Good solution found
            c_low = c_mid
            improved_h = local_improve(h_feasible)
            improved_c5 = evaluate_erdos_solution(improved_h, dx)
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5
        else:
            c_high = c_mid
    
    # Final local optimization
    final_h = local_improve(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
