import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

        for i in range(n_segments):
            start_idx = int(i * len(initial_h) / n_segments)
            end_idx = int((i + 1) * len(initial_h) / n_segments)
            if end_idx > start_idx:
                segment_heights[i] = np.mean(initial_h[start_idx:end_idx])
            else:
                segment_heights[i] = 0.5
    else:
        segment_widths = np.ones(n_segments) * (2.0 / n_segments)
        segment_heights = np.ones(n_segments) * 0.5
    
    return segment_widths, segment_heights

def segments_to_h(segment_widths, segment_heights, n_points):
    h = np.zeros(n_points)
    dx_seg = 2.0 / n_points
    cumulative_width = 0.0
    for i, (width, height) in enumerate(zip(segment_widths, segment_heights)):
        start_idx = int(cumulative_width / dx_seg)
        end_idx = int((cumulative_width + width) / dx_seg)
        if end_idx > start_idx:
            h[start_idx:end_idx] = height
        cumulative_width += width
    return h

def compute_segment_c5(segment_widths, segment_heights, dx):
    # Reconstruct h
    n_points = int(2.0 / dx)
    h = segments_to_h(segment_widths, segment_heights, n_points)
    return evaluate_erdos_solution(h, dx)

        segment_heights = np.clip(segment_heights, 0, 1)
    return segment_heights
