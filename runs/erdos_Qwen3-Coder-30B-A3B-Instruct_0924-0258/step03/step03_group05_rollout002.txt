import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try to improve using different discretizations
    for points in [35, 70]:
        if time() - start_time > budget_s * 0.95:
            break
            
        n_points = points
        dx = 2.0 / n_points
        
        # Initialize with best solution so far
        h_init = best_h.copy()
        if len(h_init) != n_points:
            # Interpolate to new resolution
            h_init = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, len(h_init)), h_init)
        
        # Ensure initial sum matches constraint
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Define the optimization problem
        def solve_feasibility(h_guess, c_val):
            h = cp.Variable(n_points)
            constraints = [
                h >= 0,
                h <= 1,
                cp.sum(h) * dx == 1
            ]
            
            # Add overlap constraints
            for k in range(1, n_points):  # Skip k=0
                valid_len = n_points - k
                if valid_len <= 0:
                    break
                # Sum of (h[i] * (1 - h[i+k])) for valid indices
                overlap_term = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx
                constraints.append(overlap_term <= c_val)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    return h.value
            except:
                pass
            return None

        # Binary search
        c_low = 0.37
        c_high = best_c5
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            c_mid = (c_low + c_high) / 2.0
            h_feasible = solve_feasibility(h_init, c_mid)
            
            if h_feasible is not None:
                c_low = c_mid
                h_init = h_feasible
                # Local refinement
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_feasible, method='SLSQP', 
                                   bounds=bounds, constraints=cons, 
                                   options={'maxiter': 50, 'ftol': 1e-7})
                    if res.success:
                        h_refined = np.clip(res.x, 0, 1)
                        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
            else:
                c_high = c_mid

    # Final postprocessing
    final_c5 = evaluate_erdos_solution(best_h, dx)
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    # Final refinement using SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
