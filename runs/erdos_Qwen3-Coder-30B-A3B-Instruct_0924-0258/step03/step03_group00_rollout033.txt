import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to avoid revisiting recent configurations
    tabu_list = []
    tabu_size = 10
    
    # Main optimization loop
    max_iter = 50000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Simulated annealing parameters
        temp = max(0.01, 1.0 * np.exp(-iter_count / 5000))
        
        # Create a new candidate by perturbing a random segment
        h_candidate = best_h.copy()
        idx = np.random.randint(0, n_points)
        
        # Try different perturbations
        perturbations = [0.0, 1.0, np.clip(best_h[idx] + np.random.normal(0, 0.1), 0, 1)]
        best_cand_value = best_c5
        best_cand_h = h_candidate.copy()
        
        for pert in perturbations:
            h_candidate[idx] = pert
            c5_val = evaluate_erdos_solution(h_candidate, dx)
            
            if c5_val < best_cand_value:
                best_cand_value = c5_val
                best_cand_h = h_candidate.copy()
            elif np.random.rand() < np.exp(-(c5_val - best_cand_value) / temp):
                # Accept worse solution with some probability
                best_cand_value = c5_val
                best_cand_h = h_candidate.copy()
                
        # Apply the best candidate
        h_candidate = best_cand_h.copy()
        
        # Ensure integral constraint is maintained
        h_sum = np.sum(h_candidate) * dx
        if abs(h_sum - 1.0) > 1e-6:
            h_candidate = h_candidate * (1.0 / h_sum) if h_sum > 0 else h_candidate
            
        # Check if this is a new configuration to avoid tabu
        h_tuple = tuple(np.round(h_candidate, 5))
        if h_tuple not in tabu_list:
            if best_cand_value < best_c5:
                best_h = h_candidate.copy()
                best_c5 = best_cand_value
            tabu_list.append(h_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Instead of rejecting, try to improve directions
            # Try a random neighbor perturbation for different segment
            new_idx = np.random.randint(0, n_points)
            if new_idx != idx:
                h_candidate[new_idx] = np.clip(h_candidate[new_idx] + np.random.normal(0, 0.05), 0, 1)
                h_sum = np.sum(h_candidate) * dx
                if abs(h_sum - 1.0) > 1e-6:
                    h_candidate = h_candidate * (1.0 / h_sum) if h_sum > 0 else h_candidate
                c5_val = evaluate_erdos_solution(h_candidate, dx)
                if c5_val < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_val
        
        # Occasionally refine using local optimization
        if iter_count % 200 == 0 and (time.time() - start_time) < budget_s * 0.95:
            def obj_func(h_input):
                h_clipped = np.clip(h_input, 0, 1)
                h_sum = np.sum(h_clipped) * dx
                if h_sum > 0:
                    h_normalized = h_clipped * (1.0 / h_sum) if h_sum > 0 else h_clipped
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)

            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}

            try:
                res = minimize(obj_func, best_h.copy(), method='SLSQP',
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 10})
                if res.success:
                    c5_refined = obj_func(res.x)
                    if c5_refined < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_refined
            except:
                pass
    
    # Final refinement using gradient-based method
    if (time.time() - start_time) < budget_s * 0.95:
        def obj_func(h_input):
            h_clipped = np.clip(h_input, 0, 1)
            h_sum = np.sum(h_clipped) * dx
            if h_sum > 0:
                h_normalized = h_clipped * (1.0 / h_sum) if h_sum > 0 else h_clipped
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)

        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}

        try:
            # Random restarts
            for _ in range(5):
                h_start = np.random.rand(n_points) * 0.5 + 0.25
                res = minimize(obj_func, h_start, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50})
                if res.success:
                    c5_try = obj_func(res.x)
                    if c5_try < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_try
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
