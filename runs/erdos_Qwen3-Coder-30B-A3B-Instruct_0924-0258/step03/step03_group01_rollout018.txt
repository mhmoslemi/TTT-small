import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default initialization
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 20
    num_parents = pop_size // 2
    num_offspring = 10
    mutation_rate = 0.3
    mutation_std = 0.03
    sqp_interval = 1
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        # Normalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    generation = 0
    while time() - start_time < budget_s * 0.95:
        generation += 1
        if generation % sqp_interval == 0:
            # SQP refinement for top candidates
            fitness_scores = [evaluate_erdos_solution(h, dx) for h in population]
            top_indices = np.argsort(fitness_scores)[:3]
            
            for idx in top_indices:
                h_refined = population[idx].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
        
        # Evaluate current population
        fitness_scores = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Selection (keep top half)
        selected_indices = np.argsort(fitness_scores)[:num_parents]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(num_offspring):
            if len(selected_population) < 2:
                break
            parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
            child = (parent1 + parent2) / 2.0
            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_std, n_points)
                child += noise
                child = np.clip(child, 0, 1)
            # Renormalize
            child = child * (n_points / 2) / np.sum(child)
            offspring.append(child)
        
        # Replace worst with offspring
        worst_indices = np.argsort(fitness_scores)[num_parents:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Update best
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness_scores[current_best_idx]
    
    # Final refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
