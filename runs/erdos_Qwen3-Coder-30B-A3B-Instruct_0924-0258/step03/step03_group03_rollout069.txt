import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_n_points = n_points
    
    # Try several starting points
    starts = [h_start]
    
    # Add random starts with same constraints
    for _ in range(2):
        h_random = np.random.uniform(0, 1, n_points)
        h_random = project_to_feasible(h_random, n_points)
        starts.append(h_random)
    
    # Genetic algorithm with local search
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    elite_size = 5
    max_evaluations = 5000
    
    evaluations = 0
    current_best_c5 = best_c5
    current_best_h = best_h.copy()
    
    # Initialize population
    population = []
    fitness = []
    
    for i, h_init in enumerate(starts):
        h = project_to_feasible(h_init, n_points)
        score = evaluate_erdos_solution(h, dx)
        population.append(h)
        fitness.append(score)
        evaluations += 1
        if score < current_best_c5:
            current_best_c5 = score
            current_best_h = h.copy()
    
    # Fill up population to size
    while len(population) < population_size:
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        score = evaluate_erdos_solution(h, dx)
        population.append(h)
        fitness.append(score)
        evaluations += 1
        if score < current_best_c5:
            current_best_c5 = score
            current_best_h = h.copy()
    
    # Evolutionary optimization
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        new_fitness = fitness[:elite_size]
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Uniform crossover
            mask = np.random.random(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutate_idx = np.random.randint(n_points)
                child[mutate_idx] = np.random.uniform(0, 1)
            
            child = project_to_feasible(child, n_points)
            score = evaluate_erdos_solution(child, dx)
            
            new_population.append(child)
            new_fitness.append(score)
            
            evaluations += 1
            if score < current_best_c5:
                current_best_c5 = score
                current_best_h = child.copy()
                
            if evaluations >= max_evaluations:
                break
        
        population = new_population
        fitness = new_fitness
        
        # Local refinement with gradient descent
        if gen % 5 == 0 and gen > 0:
            refined_population = []
            for i, h in enumerate(population):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Gradient descent refinement
                h_refined = h.copy()
                step = 0.001
                for _ in range(50):
                    if time() - start_time > budget_s * 0.95:
                        break
                    grad = np.zeros_like(h_refined)
                    eps = 1e-6
                    base_score = evaluate_erdos_solution(h_refined, dx)
                    
                    for j in range(len(h_refined)):
                        h_plus = h_refined.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        score_plus = evaluate_erdos_solution(h_plus, dx)
                        grad[j] = (score_plus - base_score) / eps
                    
                    h_refined = h_refined - step * grad
                    h_refined = project_to_feasible(h_refined, n_points)
                    
                    new_score = evaluate_erdos_solution(h_refined, dx)
                    if new_score < base_score:
                        h_refined = h_refined
                    else:
                        break
                
                refined_population.append(h_refined)
                refined_score = evaluate_erdos_solution(h_refined, dx)
                if refined_score < current_best_c5:
                    current_best_c5 = refined_score
                    current_best_h = h_refined.copy()
                    
            population = refined_population
    
    # Final gradient descent for best solution
    if current_best_c5 < best_c5:
        best_h = current_best_h.copy()
        best_c5 = current_best_c5
        best_n_points = n_points
    
    # Final refinement
    h_final = best_h.copy()
    dx_final = 2.0 / best_n_points
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        grad = np.zeros_like(h_final)
        eps = 1e-5
        base_score = evaluate_erdos_solution(h_final, dx_final)
        
        for i in range(len(h_final)):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, best_n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx_final)
            grad[i] = (score_plus - base_score) / eps

        step_size = 0.001
        h_final = h_final - step_size * grad
        h_final = project_to_feasible(h_final, best_n_points)

        score_final = evaluate_erdos_solution(h_final, dx_final)
        if score_final < best_c5:
            best_h = h_final.copy()
            best_c5 = score_final

    return (list(best_h), best_c5, best_n_points)
