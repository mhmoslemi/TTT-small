import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 20
    generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # GA Loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection (top 50%)
        selected = population[:population_size//2]
        selected_fitness = fitness[:population_size//2]
        
        # Create new generation
        new_population = selected[:]
        
        # Crossover
        for i in range(int(population_size * crossover_rate)):
            if len(selected) < 2:
                break
            parent1, parent2 = random.sample(selected, 2)
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Mutation
        for i in range(int(population_size * mutation_rate)):
            idx = random.randint(0, len(new_population) - 1)
            h = new_population[idx].copy()
            mut = np.random.normal(0, 0.03, n_points)
            h += mut
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            new_population[idx] = h
            
        # Truncate to population size
        population = new_population[:population_size]
        
        # Local refinement
        if gen % sqp_interval == 0 and gen > 0:
            # Refine top 3 with SLSQP
            for i in range(min(3, len(population))):
                h_init = population[i]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                        # Replace in population if better
                        for j, pop_h in enumerate(population):
                            if np.allclose(pop_h, population[i], atol=1e-8):
                                population[j] = h_refined
                                break
                except:
                    pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
