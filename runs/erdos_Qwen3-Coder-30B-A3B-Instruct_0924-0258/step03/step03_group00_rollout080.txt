import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to prevent cycling
    tabu_list = []
    tabu_size = 10
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    
    # Parameter to adjust perturbation strength
    perturbation_strength = 0.1
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Create a new candidate solution by perturbing current best
        h_new = best_h.copy()
        
        # Select a random point to perturb
        idx = np.random.randint(0, n_points)
        # Perturb the value, keeping it within [0,1]
        delta = np.random.uniform(-perturbation_strength, perturbation_strength)
        h_new[idx] += delta
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Adjust the sum to maintain the constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (n_points / 2) / current_sum
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        else:
            # Accept with probability based on temperature
            delta_e = c5_new - best_c5
            if np.random.rand() < np.exp(-delta_e / temp):
                best_h = h_new.copy()
                best_c5 = c5_new
                
        # Update temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Update tabu list
        h_tuple = tuple(h_new.round(6))  # Round for comparison
        if h_tuple in tabu_list:
            # If this state is tabu, we might still accept it occasionally
            if np.random.rand() < 0.1:  # 10% chance to accept anyway
                pass
            else:
                # Revert to best known solution
                h_new = best_h.copy()
                continue
        
        # Add current state to tabu list
        tabu_list.append(h_tuple)
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Occasionally try a different strategy
        if iteration % 100 == 0 and perturbation_strength > 0.01:
            perturbation_strength *= 0.95  # Reduce perturbation strength over time
    
    # Refinement with local optimization
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run several local optimizations with different starting points
    best_local = best_h.copy()
    best_local_c5 = best_c5
    num_restarts = 10
    
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Perturb current best
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_sum = np.sum(h_perturb)
        if h_sum > 0:
            h_perturb = h_perturb * (n_points / 2) / h_sum
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_local_c5:
                    best_local = res.x.copy()
                    best_local_c5 = c5_test
        except:
            continue
    
    # Final validation
    if best_local_c5 < best_c5:
        best_h = best_local
        best_c5 = best_local_c5
    
    return (list(best_h), best_c5, n_points)
