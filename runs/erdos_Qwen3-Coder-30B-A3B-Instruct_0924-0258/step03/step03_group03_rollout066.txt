import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
    
    best_h = h_start.copy()
    best_c5 = current_c5
    best_n_points = n_points
    
    # Direct optimization: begin with projected gradient descent
    max_iter = 1000
    step_size = 0.01
    max_no_improve = 50
    no_improve_count = 0
    last_best = best_c5
    
    h = h_start.copy()
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Estimate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_score = evaluate_erdos_solution(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
        
        # Gradient descent step
        h_new = h - step_size * grad
        # Project to feasible set
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_score = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject step
        if new_score < best_c5:
            h = h_new.copy()
            best_c5 = new_score
            best_h = h.copy()
            no_improve_count = 0
            last_best = new_score
        else:
            no_improve_count += 1
        
        if no_improve_count > max_no_improve:
            step_size *= 0.9
            no_improve_count = 0

    # Try to improve further by trying multiple random restarts
    n_restarts = min(20, int(budget_s / 60))
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
        # Random restart
        h_restart = np.random.uniform(0, 1, n_points)
        h_restart = project_to_feasible(h_restart, n_points)
        restart_score = evaluate_erdos_solution(h_restart, dx)
        if restart_score < best_c5:
            best_h = h_restart.copy()
            best_c5 = restart_score
            best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
