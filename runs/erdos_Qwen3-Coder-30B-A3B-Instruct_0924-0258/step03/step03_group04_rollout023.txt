import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization with binary extreme points
    h_current = h_start.copy()
    max_iter = 10000
    step_size = 0.5
    perturb_rate = 0.1
    improved_count = 0
    stagnation_limit = 100
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_best = np.argmax(corr)
        max_overlap = corr[k_best]
        
        # Linear approximation around current point for shift k_best
        # We want to minimize sum(h_diff * (1 - h_shifted_k))
        h_shifted_k = np.roll(1 - h_current, k_best)
        # For better convergence, use left convolution
        h_shifted_k = h_shifted_k[:len(h_current)]
        linear_grad = h_shifted_k
        
        # Binary optimization to minimize linear objective
        # Find binary h with exactly n_points/2 ones that minimizes linear_grad
        # Sort indices based on negative linear_grad to find best candidates
        sorted_indices = np.argsort(-linear_grad)
        h_extreme = np.zeros(n_points)
        h_extreme[sorted_indices[:n_points//2]] = 1.0
        
        # Frank-Wolfe update
        h_new = (1 - step_size) * h_current + step_size * h_extreme
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Check for improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            improved_count = 0
        else:
            improved_count += 1
            
        if improved_count >= stagnation_limit:
            step_size *= 0.5
            improved_count = 0
            
        h_current = h_new
        
        # Occasionally perturb to escape local optima
        if np.random.random() < perturb_rate:
            h_current += np.random.normal(0, 0.01, n_points)
            h_current = project_to_feasible(h_current, n_points)
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
