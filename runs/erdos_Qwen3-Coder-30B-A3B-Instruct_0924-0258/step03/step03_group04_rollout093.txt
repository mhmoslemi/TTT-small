import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal initial guess for better exploration
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    acceptance_rate = 0.5
    
    # Perturbation settings
    perturb_fraction = 0.1
    max_perturbations = max(1, int(n_points * perturb_fraction))
    
    iteration = 0
    last_improvement = 0
    unchanged_count = 0
    
    # SA main loop
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        if iteration % 100 == 0:
            # Adaptive temperature update
            if acceptance_rate < 0.3:
                T *= 0.9
            elif acceptance_rate > 0.7:
                T *= 1.1
            if T < min_T:
                T = min_T
                
            acceptance_rate = 0.5  # reset for next period
            
        # Generate candidate
        h_candidate = best_h.copy()
        
        # Randomly select indices to perturb
        perturbed_indices = random.sample(range(n_points), 
                                         min(max_perturbations, n_points))
        
        for i in perturbed_indices:
            # Add some random change
            delta = np.random.normal(0, 0.1)
            h_candidate[i] += delta
            
        # Project to feasible set
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_c5 = c5_candidate - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h = h_candidate
            best_c5 = c5_candidate
            acceptance_rate = (acceptance_rate * (iteration - 1) + 1) / iteration
            last_improvement = iteration
        else:
            acceptance_rate = acceptance_rate * (iteration - 1) / iteration
            
        # Track convergence
        if iteration - last_improvement > 500:
            unchanged_count += 1
            if unchanged_count > 10:
                # Cool down faster if no improvement
                T *= 0.95
                unchanged_count = 0
                if T < min_T:
                    T = min_T
        else:
            unchanged_count = 0
            
        # Adapt perturbation size based on performance
        if iteration > 1000 and (iteration % 100 == 0):
            if best_c5 < 0.38:
                perturb_fraction = max(0.05, perturb_fraction * 0.95)
            elif best_c5 > 0.385:
                perturb_fraction = min(0.2, perturb_fraction * 1.05)
            max_perturbations = max(1, int(n_points * perturb_fraction))
            
        # Update temperature
        if iteration % 100 == 0:
            T *= cooling_rate
            
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Try a few local search improvements
        h_refined = best_h.copy()
        for _ in range(5):
            res = minimize(final_obj, h_refined, method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 100})
            if res.success:
                h_trial = project_to_feasible(res.x, n_points)
                c5_trial = evaluate_erdos_solution(h_trial, dx)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial
                    h_refined = h_trial.copy()
            else:
                break  # No improvement from this point
    
    return (list(best_h), best_c5, n_points)
