import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with feasibility projection
    max_iterations = 500
    tolerance = 1e-6
    current_h = h_start.copy()
    last_c5 = float('inf')
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random order of variable updates
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        improved = False
        for idx in indices:
            # For this index, find the value that minimizes the maximum overlap
            # Create a candidate range
            h_candidate = current_h.copy()
            
            # Simple grid search approach for one variable
            best_local_c5 = float('inf')
            best_local_value = current_h[idx]
            
            grid_points = 50
            values = np.linspace(0, 1, grid_points)
            
            for val in values:
                h_candidate[idx] = val
                # Check constraints
                h_candidate = project_to_feasible(h_candidate, n_points)
                
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_local_c5:
                    best_local_c5 = c5_candidate
                    best_local_value = val
            
            # Update the variable
            current_h[idx] = best_local_value
            
            # Apply projection to enforce constraints
            current_h = project_to_feasible(current_h, n_points)
            
            # Check if improvement
            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = current_h.copy()
                improved = True
        
        # Convergence check
        if abs(last_c5 - best_c5) < tolerance:
            break
        last_c5 = best_c5
    
    # SLSQP refinement to exploit local optima
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
