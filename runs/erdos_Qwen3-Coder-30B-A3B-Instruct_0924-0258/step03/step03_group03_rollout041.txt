import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        h_current = h_start.copy()
    else:
        # Start with simple construction
        n_points = 35
        h_current = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        
    h_current = project_to_feasible(h_current, n_points)
    best_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_n_points = n_points
    
    # Determine resolution schedule based on time budget
    resolution_schedule = [35, 30, 25]
    if time() - start_time > budget_s * 0.8:
        resolution_schedule = [35]

    for n_points_try in resolution_schedule:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Interpolate to new resolution if needed
        if n_points_try != n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, n_points_try)
            h_interp = np.interp(x_new, x_old, h_current)
            h_current = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            dx = 2.0 / n_points
            
        # Simulated Annealing approach
        max_iter = 500
        temperature = 1.0
        cooling_rate = 0.995
        best_temp_h = h_current.copy()
        best_temp_c5 = evaluate_erdos_solution(h_current, dx)
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Propose new solution
            h_proposed = h_current + np.random.normal(0, 0.02, n_points)
            h_proposed = project_to_feasible(h_proposed, n_points)
            c5_proposed = evaluate_erdos_solution(h_proposed, dx)
            
            # Accept with simulated annealing probability
            delta = c5_proposed - best_temp_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                h_current = h_proposed
                best_temp_c5 = c5_proposed
                if c5_proposed < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5_proposed
                    best_n_points = n_points
            
            temperature *= cooling_rate
            
            # Occasionally refine
            if i % 50 == 0:
                h_current = project_to_feasible(h_current, n_points)
                
        h_current = best_temp_h.copy()

    # Final gradient descent refinement using finite differences
    if time() - start_time < budget_s * 0.95:
        h_refined = best_h.copy()
        dx_refined = 2.0 / best_n_points

        for _ in range(200):
            if time() - start_time > budget_s * 0.95:
                break
            # Gradient approximation via finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx_refined)
            
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx_refined)
                grad[i] = (score_plus - base_score) / eps

            # Update with small step
            step_size = 0.005
            h_refined = h_refined - step_size * grad
            h_refined = project_to_feasible(h_refined, best_n_points)

            score_refined = evaluate_erdos_solution(h_refined, dx_refined)
            if score_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = score_refined

    return (list(best_h), best_c5, best_n_points)
