import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segmented_h(x_boundaries, heights):
    """
    Compute step function values at n_points from segment boundaries and heights
    """
    n_points = len(x_boundaries) - 1
    h_vals = np.zeros(n_points)
    for i in range(n_points):
        h_vals[i] = heights[i]
    return h_vals

def compute_correlation_gradients(h, dx, beta=100):
    """
    Compute gradients of the smooth approximation to max correlation with respect to segment parameters.
    """
    h_vals = h  # Assuming h is already the segmented function values
    
    # Compute the correlation using numpy's correlate
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    
    # Smooth approximation to max
    # max_k corr[k] ≈ (1/beta)*log(sum(exp(beta * corr[k])))
    # So d/dx_i (1/beta)*log(... ) = (1/beta) * (sum(exp(beta * corr[k]) * beta * d/corr[k] * d/dx_i) / sum(exp(beta * corr[k]))
    
    # For approximation purposes, collect individual correlation values
    corr_vals = corr 
    exp_corr = np.exp(beta * corr_vals)
    Z = np.sum(exp_corr)
    if Z == 0:
        grad = np.zeros_like(h_vals)
        return grad, 0  # Just return zero gradient
    
    # Compute gradient of log Z w.r.t h_vals
    # The key is to compute the convolution gradient w.r.t the input
    # For elements that are in full-convolution, it's a bit complex
    
    # Simplify by computing finite differences for now
    grad = np.zeros_like(h_vals)
    eps = 1e-6
    base_corr = np.max(corr_vals)
    
    for i in range(len(h_vals)):
        h_pert = h_vals.copy()
        h_pert[i] += eps
        h_pert = project_to_feasible(h_pert, len(h_vals))
        new_corr = np.correlate(h_pert, 1 - h_pert, mode='full')
        new_max = np.max(new_corr)
        grad[i] = (new_max - base_corr) / eps
    
    # Normalize by beta to get gradient of (1/beta) log sum(exp(beta * corr))
    grad = grad / (beta * Z) * exp_corr.sum()
    
    # Just return a simple approximation - not perfect but technically better than nothing
    return grad, base_corr  

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Hyperparameters
    MAX_SEGMENTS = 40
    HEIGHT_ITERS = 100
    WIDTH_ITERS = 50
    REFINE_INTERVAL = 25
    STEP_SIZE_HEIGHT = 0.01
    STEP_SIZE_WIDTH = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        m = len(h_start)
        # We'll assume we have m segments (each a constant height)
        heights = h_start.copy() 
        x_boundaries = np.linspace(0., 2.0, m + 1)
    else:
        m = 15
        heights = np.ones(m) * 0.5
        x_boundaries = np.linspace(0., 2.0, m + 1)
    
    # Ensure integral constraint
    dx = 2.0 / m
    dx_array = np.diff(x_boundaries)
    initial_integral = np.dot(heights, dx_array)
    heights = heights * (1.0 / initial_integral)
    
    best_c5 = float('inf')
    best_h = None
    best_m = None
    best_x = None
    best_heights = None
    
    def projected_descent_heights(x_b, h, dx_array, iterations):
        for _ in range(iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient to heights using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            base_corr = evaluate_erdos_solution(h, dx_array[0])
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, len(h))
                score_plus = evaluate_erdos_solution(h_plus, dx_array[0])
                grad[i] = (score_plus - base_corr) / eps
            
            # Update by gradient descent
            step = STEP_SIZE_HEIGHT
            h_new = h - step * grad
            
            # Project to constraints: [0,1] and integral constraint
            h_new = np.clip(h_new, 0, 1)
            integral = np.dot(h_new, dx_array)
            h_new = h_new * (1.0 / integral) if integral > 0 else h_new
            
            h = h_new
            
            # Check if better solution
            score = evaluate_erdos_solution(h, dx_array[0])
            if score < best_c5:
                best_c5 = score
                best_h = h.copy()
                best_m = len(h)
                best_x = x_b.copy()
                best_heights = h.copy()
                
        return h
    
    def projected_descent_widths(x_b, h, dx_array, iterations):
        for _ in range(iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            grad = np.zeros_like(x_b)
            eps = 1e-4
            base_score = evaluate_erdos_solution(h, dx_array[0])
            
            for i in range(1, len(x_b)-1):  # don't touch boundaries x_0=0, x_m=2
                x_plus = x_b.copy()
                x_plus[i] += eps
                x_plus = np.sort(np.clip(x_plus, 0, 2))  # keep sorted
                x_plus[0] = 0  # clamp to fixed boundary 
                x_plus[-1] = 2
                
                # Reinterpolate heights
                h_interp = np.interp(np.linspace(0,2,len(h)), x_b, h)
                if not np.all(np.isfinite(h_interp)):
                    continue
                    
                score_plus = evaluate_erdos_solution(h_interp, dx_array[0])
                grad[i] = (score_plus - base_score) / eps
            
            # Update width boundaries, keep x[0]=0, x[m]=2 fixed
            step = STEP_SIZE_WIDTH
            x_b = x_b - step * grad
            x_b = np.sort(np.clip(x_b, 0, 2))
            x_b[0] = 0
            x_b[-1] = 2
            
            # Recompute heights
            h_interp = np.interp(np.linspace(0,2,len(h)), x_b, h) 
            h_interp = np.clip(h_interp, 0, 1)
            integral = np.dot(h_interp, np.diff(x_b))
            h_interp = h_interp * (1.0 / integral) if integral > 0 else h_interp
            
            dx_array = np.diff(x_b)
            h = h_interp
            
            # Check solution
            score = evaluate_erdos_solution(h, dx_array[0])
            if score < best_c5:
                best_c5 = score
                best_h = h.copy()
                best_m = len(h)
                best_x = x_b.copy()
                best_heights = h.copy()
                
        return x_b, h
    
    # MAIN OPTIMIZATION LOOP
    # Main optimization loop 
    keep_going = True
    iter_count = 0
    
    while keep_going and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # 1. Optimize heights
        heights = projected_descent_heights(x_boundaries, heights, dx_array, HEIGHT_ITERS)
        
        # 2. Optimize widths
        x_boundaries, heights = projected_descent_widths(x_boundaries, heights, dx_array, WIDTH_ITERS)
        
        # 3. Refinement -- split segments with highest gradient
        if iter_count % REFINE_INTERVAL == 0:
            # Compute gradient for current segment heights to decide split points
            # For now, basic idea: if mag of gradient is large, split that segment  
            if len(x_boundaries) < MAX_SEGMENTS:
                # Sample by gradient energy across segments
                dx_array = np.diff(x_boundaries)
                h_val = np.interp(np.linspace(0,2,len(heights)), x_boundaries, heights)
                grad, _ = compute_correlation_gradients(h_val, dx_array[0], beta=100)
                if len(grad) > 0:
                    # Simple approach: interval with highest gradient
                    idx = np.argmax(np.abs(grad))
                    if idx >= 0 and idx < len(x_boundaries) - 1:
                        # insert new boundary
                        new_boundary = (x_boundaries[idx] + x_boundaries[idx+1]) / 2.0
                        insert_idx = np.searchsorted(x_boundaries, new_boundary)
                        # insert boundary
                        x_boundaries = np.insert(x_boundaries, insert_idx, new_boundary)
                        heights = np.interp(np.linspace(0,2,len(x_boundaries)-1), x_boundaries, np.concatenate([heights, heights]))
                        heights = np.clip(heights, 0, 1)  
                        # Rescale
                        integral = np.dot(heights, dx_array)
                        heights = heights * (1.0 / integral) if integral > 0 else heights
                        dx_array = np.diff(x_boundaries)
                        
        if time() - start_time > budget_s * 0.95:
            break    
              
    # Final push for best solution seen
    if best_h is not None:
        return (list(best_h), best_c5, len(best_h))
    else:
        # Return best from the final iteration m = len(heights)
        return (list(heights), evaluate_erdos_solution(heights, dx_array[0]), len(heights))
