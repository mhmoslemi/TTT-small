import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_probability = 0.9
    max_iterations = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # DE main loop
    last_improvement = 0
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Check for improvement
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitness.index(current_best)].copy()
            last_improvement = iteration
        
        # Early termination if stagnation
        if iteration - last_improvement > stagnation_limit:
            break
        
        # Create new population
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals
            indices = random.sample(range(population_size), 3)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            trial = []
            for j in range(n_points):
                if random.random() < crossover_probability:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            
            trial = np.array(trial)
            trial = project_to_feasible(trial, n_points)
            new_population.append(trial)
        
        population = new_population
    
    # Post-optimization refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
