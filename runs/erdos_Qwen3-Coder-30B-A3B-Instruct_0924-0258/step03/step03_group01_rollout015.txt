import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_population(n_points, initial_h, pop_size=20):
    population = []
    for i in range(pop_size):
        if i == 0 and initial_h is not None:
            h = initial_h.copy()
        else:
            h = initial_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def mutate(h, mutation_rate=0.3):
    if random.random() < mutation_rate:
        h = h + np.random.normal(0, 0.03, len(h))
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
    return h

def crossover(h1, h2):
    alpha = random.random()
    h = alpha * h1 + (1 - alpha) * h2
    h = h * (len(h) / 2) / np.sum(h)
    return h

def repair(h):
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else None
    n_points = len(initial_h) if initial_h is not None else 35
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 20
    population = initialize_population(n_points, initial_h, pop_size)
    best_h = None
    best_c5 = float('inf')
    
    # Genetic algorithm with SQP refinement
    generations = 30
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection (top 50%)
        selected_indices = np.argsort(fitness)[:pop_size//2]
        selected = [population[i] for i in selected_indices]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        while len(new_population) < pop_size:
            if len(selected) >= 2:
                i1, i2 = random.sample(range(len(selected)), 2)
                child = crossover(selected[i1], selected[i2])
                child = mutate(child)
                child = repair(child)
                new_population.append(child)
            else:
                # If not enough selected, create random individuals
                h = initial_h + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                new_population.append(h)
                
        # Local refinement with SQP
        if gen % 1 == 0:  # Refine every generation
            for i in range(min(3, len(selected))):
                h_start = selected[i].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                def con_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                try:
                    res = minimize(
                        obj, 
                        h_start, 
                        method='SLSQP',
                        bounds=[(0, 1) for _ in range(n_points)],
                        constraints={'type': 'eq', 'fun': con_sum},
                        options={'maxiter': 30, 'ftol': 1e-6}
                    )
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < best_c5:
                            best_c5 = c5_refined
                            best_h = h_refined.copy()
                except:
                    pass
        
        population = new_population
    
    return (list(best_h), best_c5, n_points)
