import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimizer parameters
    max_iter = 1000
    adaptive_lr = 1e-3
    patience = 100
    patience_counter = 0
    best_c5_saved = best_c5
    
    # Initialize
    h_current = best_h.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Gradient-based optimization
    def gradient_descent_step(h, lr):
        # Compute gradient numerically
        grad = np.zeros_like(h)
        c5_current = evaluate_erdos_solution(h, dx)
        
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update using gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Adaptive learning rate
        current_lr = adaptive_lr / (1 + i / 100)
        
        # Optimization step
        h_new = gradient_descent_step(h_current, current_lr)
        
        # Check for improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_counter = 0
            adaptive_lr = min(adaptive_lr * 1.05, 1e-1)
        else:
            patience_counter += 1
            adaptive_lr = max(adaptive_lr * 0.99, 1e-6)
            
        h_current = h_new.copy()
        
        # Random perturbations to avoid local optima
        if i % 50 == 0 and i > 0:

            h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Final refinement with scipy
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
        
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',

    except:
        pass
    
    return (list(best_h), best_c5, n_points)
