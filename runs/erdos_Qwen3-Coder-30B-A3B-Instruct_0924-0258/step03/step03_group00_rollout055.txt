import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 10
    max_iterations = 10000
    annealing_rate = 0.99
    temperature = 1.0
    
    # Search modifications
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Create a neighbor solution by perturbing one element
        h_trial = best_h.copy()
        idx = np.random.randint(0, n_points)
        
        # Try small perturbations with different values
        candidates = [0, 0.25, 0.5, 0.75, 1.0]
        candidate_vals = [v for v in candidates if abs(v - h_trial[idx]) > 1e-6]
        if candidate_vals:
            new_val = np.random.choice(candidate_vals)
            h_trial[idx] = new_val
        
        # Ensure sum constraint
        h_trial_sum = np.sum(h_trial)
        if h_trial_sum > 0:
            h_trial = h_trial * (n_points / 2) / h_trial_sum
        
        # Evaluate
        c5_trial = evaluate_erdos_solution(h_trial, dx)
        
        # Apply simulated annealing acceptance criterion
        if c5_trial < best_c5 or np.random.random() < np.exp(-(c5_trial - best_c5) / temperature):
            best_h = h_trial.copy()
            best_c5 = c5_trial
            
            # Add to tabu list
            tabu_list.append(h_trial.copy())
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        
        # Update temperature
        temperature *= annealing_rate
    
    # Final refinement with local optimization
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations with different starting points
        best_local = best_h.copy()
        best_local_c5 = best_c5
        num_restarts = 10
        
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Perturb current best
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_local_c5:
                        best_local = res.x.copy()
                        best_local_c5 = c5_test
            except:
                continue
        
        # Final validation
        if best_local_c5 < best_c5:
            best_h = best_local
            best_c5 = best_local_c5
    
    return (list(best_h), best_c5, n_points)
