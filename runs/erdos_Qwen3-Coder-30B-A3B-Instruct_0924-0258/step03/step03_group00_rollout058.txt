import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to avoid revisiting solutions
    tabu_list = []
    tabu_length = 10
    
    # Main optimization loop
    max_iterations = 10000
    iteration = 0
    
    while iteration < max_iterations and time.time() - start_time < budget_s * 0.95:
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Create candidate by perturbing one element
        h_candidate = best_h.copy()
        idx = np.random.randint(0, n_points)
        
        # Try different perturbations
        perturbations = [0.0, 1.0, np.clip(best_h[idx] + np.random.normal(0, 0.1), 0, 1)]
        best_candidate = h_candidate.copy()
        best_candidate_c5 = best_c5
        
        for pert in perturbations:
            h_candidate[idx] = pert
            # Apply constraints
            h_candidate = np.clip(h_candidate, 0, 1)
            h_sum = np.sum(h_candidate)
            if h_sum > 0:
                h_candidate = h_candidate * (n_points / 2) / h_sum
                
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_candidate_c5:
                best_candidate = h_candidate.copy()
                best_candidate_c5 = c5_candidate
                
        # Update best if improved
        if best_candidate_c5 < best_c5:
            best_h = best_candidate.copy()
            best_c5 = best_candidate_c5
            
        # Update tabu list
        solution_tuple = tuple(np.round(best_h, decimals=5))
        if solution_tuple in tabu_list:
            # Remove oldest entry if needed
            if len(tabu_list) >= tabu_length:
                tabu_list.pop(0)
            tabu_list.append(solution_tuple)
        else:
            if len(tabu_list) >= tabu_length:
                tabu_list.pop(0)
            tabu_list.append(solution_tuple)
            
        iteration += 1
    
    # Local refinement using scipy minimize
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple restarts
        num_restarts = 20
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Random perturbation of best solution
            h_perturb = best_h + np.random.normal(0, 0.02, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
