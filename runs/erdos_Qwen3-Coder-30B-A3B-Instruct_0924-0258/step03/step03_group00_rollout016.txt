import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier basis approach
    d = 5  # number of cosine terms
    max_restarts = 20
    trial_time = budget_s / max_restarts
    
    # Define bounds for coefficients
    bounds = [(-10, 10) for _ in range(d + 1)]
    
    def fourier_to_h(coeffs):
        # coeffs: [a0, a1, ..., ad]
        x = np.linspace(0, 2, n_points)
        # Construct g(x) as sum of cosines
        g = coeffs[0] + sum(coeffs[k+1] * np.cos(k * np.pi * x) for k in range(1, d+1))
        # Logistic transformation
        h = 1.0 / (1.0 + np.exp(-g))
        return h
    
    def obj_func(coeffs):
        h = fourier_to_h(coeffs)
        # Normalize
        h_norm = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h_norm, dx)
    
    def constraint_sum(coeffs):
        h = fourier_to_h(coeffs)
        h_norm = h * (n_points / 2) / np.sum(h)
        return np.sum(h_norm) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Random restart
        initial_coeffs = np.random.uniform(-1, 1, d + 1)
        
        try:
            opt_result = minimize(
                obj_func, 
                initial_coeffs, 
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            
            if opt_result.success:
                h_new = fourier_to_h(opt_result.x)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except:
            continue
    
    # Final optimization with local descent
    if time.time() - start_time < budget_s - 5:
        def obj_func_local(h):
            h_clipped = np.clip(h, 0, 1)
            if np.sum(h_clipped) > 0:
                h_scaled = h_clipped * (n_points / 2) / np.sum(h_clipped)
            else:
                h_scaled = h_clipped
            return evaluate_erdos_solution(h_scaled, dx)
        
        def constraint_local(h):
            return np.sum(h) - n_points / 2
        
        cons_local = {'type': 'eq', 'fun': constraint_local}
        bounds_local = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts for better local search
        for _ in range(5):
            if time.time() - start_time > budget_s - 5:
                break
            h_perturb = best_h + np.random.normal(0, 0.05, n_points)
            try:
                res = minimize(obj_func_local, h_perturb, method='SLSQP', 
                              bounds=bounds_local, constraints=cons_local,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func_local(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
