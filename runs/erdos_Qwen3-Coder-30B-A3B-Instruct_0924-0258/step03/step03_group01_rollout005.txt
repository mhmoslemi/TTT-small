import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate initial population for exploration
    pop_size = 20
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
        
    # Policy gradient parameters
    learning_rate = 0.01
    discount_factor = 0.99
    entropy_coef = 0.1

    max_iterations = 60

            break

        fitness = []

            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(-c5)  # negative for maximization (minimize -C5)

        sorted_indices = np.argsort(fitness)[::-1]
        elite_indices = sorted_indices[:5]
        
        # Train on elite members
        for idx in elite_indices:
            h_elite = population[idx].copy()
            # Apply small perturbation
            delta = np.random.normal(0, 0.02, n_points)
            h_new = h_elite + delta
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Convex refinement
            try:
                h_refined = h_new.copy()
                # Set up convex problem
                h_var = cp.Variable(n_points)

                    cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1
                ]

                corr = np.correlate(h_refined, 1 - h_refined, mode='full')

                max_corr = np.max(corr) * dx

                # Skip full constraint set for now to speed up

                prob = cp.Problem(cp.Minimize(0), constraints)

                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)

    remaining_time = budget_s - (time() - start_time)

            return np.sum(h) - n_points / 2.0

                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
