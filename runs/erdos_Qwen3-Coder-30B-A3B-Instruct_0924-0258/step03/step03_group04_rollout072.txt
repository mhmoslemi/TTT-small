import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

        ones_count = n_points // 2

        np.random.shuffle(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    max_iter = 10000  # Enough iterations for 600s budget

        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Linearization at current point
        # computing the gradient of overlap w.r.t. h at the max_k shift
        n = len(h)
        # Construct the vector (1 - h[max_k:])
        gradient = np.zeros(n)
        # build h_shifted_k that aligns with the measured correlation
        if max_k < n:
            # k is shift of h[0] to h[k]
            # we're looking at inner product h[i] * (1 - h[i + k])
            # which is valid for i from 0 to (n-k-1) 
            # linear gradient would be 1-h[max_k:] on range 0 to (n-k-1)
            gradient[:n-max_k] = 1 - h[max_k:]
        else:
            # k >= n => all overlaps are measured with shift
            # i from 0 to (n-1) against (i + k - n) shift
            gradient[max_k-n: max_k] = 1 - h[:n-(max_k-n)]
        
        # Frank-Wolfe subproblem: solve argmin_{h_extreme} <gradient, h_extreme>
        # where h_extreme is binary vector with exactly n/2 ones
        # The greedy choice is to take the n/2 largest gradient entries
        # Indices where gradient > 0 (we want negative gradient contribution)
        indices = np.argsort(-gradient)  # descending order

        # total number = n_points / 2, so we take indices of lowest gradient (or
        # temporarily convert to a problem where we take bottom entries for the binary vector)
        # A more standard way in Frank-Wolfe is to find the extreme point minimizing:
        # - sum_i (gradient_i * h_extreme_i) = sum_i gradient_i * h_extreme_i
        # But since h_extreme is binary, minimizing such scalar product
        # means picking {i : gradient_i > 0} for 0s, rest for 1s.
        # Or to make sure it's positive, we set 1 where gradient_i < 0.
        # Let's compute what vector h_extreme is optimal by ensuring  
        # we can take those with most positive gradient_i in the 'gradient' vector - 
        # that is, when gradient_i > 0, we want h_extreme[i] = 0 to minimize -grad * h_extreme
        # when gradient_i < 0, we want h_extreme[i] = 1

        # To minimize <gradient, binary_h> where sum(binary_h) = n_points/2

        # This is approximated by:

        h_extreme = np.zeros(n_points)

        # For minimizing dot product of gradient with binary h

        # but to properly pick exactly n/2 ones, first track current number of 1s

                    num_ones += 1
                i += 1

        # Update h: h = (1 - eta) * h + eta * h_extreme
        h_new = (1 - eta) * h + eta * h_extreme

        h_new = project_to_feasible(h_new, n_points)
        h = h_new

        c5 = evaluate_erdos_solution(h, dx)

        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and iteration > 0:
            h = h + np.random.normal(0, 0.01, n_points)
            h = project_to_feasible(h, n_points)
    
    # Final refinement with gradient descent using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:

            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 200})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
