import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Wavelet basis approach
    def get_wavelet_basis(n):
        # Haar wavelets on [0, 2]
        basis = []
        # Scaling function (constant)
        basis.append(np.ones(n))
        # Wavelet functions
        for j in range(1, n):
            if 2**j > n:
                break
            wavelet = np.zeros(n)
            wavelet[0::2**j] = 1
            wavelet[1::2**j] = -1
            basis.append(wavelet)
        # Normalize
        for i in range(len(basis)):
            norm = np.linalg.norm(basis[i])
            if norm > 0:
                basis[i] = basis[i] / norm
        return np.array(basis)
    
    # Initialize with wavelet decomposition
    wavelet_basis = get_wavelet_basis(n_points)
    
    # Optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Optimize using wavelet coefficients
        # Use our own CVXPy approach for this
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) * dx == 1
        ]
        
        # Overlap constraints
        # Just constraint all valid shifts, should be about n_points
        for k in range(1, n_points):
            valid_len = n_points - k
            if valid_len <= 0:
                break
            # Constrain: sum(h[i] * (1 - h[i+k])) <= c_val
            # Use a proxy quadratic constraint to encourage small overlaps
            overlap = cp.sum(cp.multiply(h_vars[:valid_len], 1 - h_vars[k:k+valid_len])) * dx
            constraints.append(overlap <= best_c5 * 1.01)  # Provide some slack to improve solution
        
        # Objective: minimize max overlap (this is a proxy)
        # We'll just let feasibility help us
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, feastol=1e-5)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_candidate = h_vars.value
                if h_candidate is not None:
                    h_candidate = np.clip(h_candidate, 0, 1)
                    # Re-normalize
                    h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                    
                    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
        except:
            pass
            
        # Local descent on the best found vector
        def obj_to_minimize(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        # Try L-BFGS-B first as it handles bounds
        try:
            res = minimize(obj_to_minimize, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 20, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Occasionally also try SLSQP
        if iteration % 50 == 0:
            try:
                res = minimize(obj_to_minimize, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 10, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Final rigorous check with local refinement
    def obj_to_minimize(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    try:
        res = minimize(obj_to_minimize, best_h, method='L-BFGS-B', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 30, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Final sanity check
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
