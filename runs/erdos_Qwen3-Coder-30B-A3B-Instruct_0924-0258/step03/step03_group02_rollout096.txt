import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iterations = 500
    tolerance = 1e-6
    min_improvement = 1e-6
    
    h_current = best_h.copy()
    last_c5 = best_c5
    
    # Coordinate descent with random order
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Randomly shuffle indices
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for i in indices:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Save current value
            old_h_i = h_current[i]
            
            # One-dimensional search for optimal value of h[i]
            best_h_i = old_h_i
            best_c5_local = last_c5
            
            # Grid search with 100 steps
            search_range = np.linspace(max(0, old_h_i - 0.2), min(1, old_h_i + 0.2), 50)
            
            for candidate_h_i in search_range:
                if time() - start_time > budget_s * 0.95:
                    break
                # Temporarily update
                h_current[i] = candidate_h_i
                # Project to feasibility (this is done for the whole vector internally)
                h_projected = project_to_feasible(h_current, n_points)
                
                # Evaluate
                c5_temp = evaluate_erdos_solution(h_projected, dx)
                
                if c5_temp < best_c5_local:
                    best_c5_local = c5_temp
                    best_h_i = candidate_h_i
                    
            # Update
            h_current[i] = best_h_i
            # Project after each update to maintain feasibility
            h_current = project_to_feasible(h_current, n_points)
            
            # Check if this is an improvement
            if best_c5_local < last_c5 - tolerance:
                improved = True
                
        # Re-evaluate after one full pass
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        # Update last_c5
        last_c5 = current_c5
        
        # Break if no significant improvement
        if not improved and iteration > 5:
            if abs(last_c5 - best_c5) < min_improvement:
                break
    
    # Post-refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
