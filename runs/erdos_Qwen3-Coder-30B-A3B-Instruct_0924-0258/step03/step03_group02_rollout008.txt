import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_iter = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = np.random.rand(population_size, n_points)
    for i in range(population_size):
        population[i] = project_to_feasible(population[i], n_points)
    
    # Evaluate initial population
    fitness = np.array([evaluate_erdos_solution(ind, dx) for ind in population])
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # DE iteration
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_prob, mutant, population[i])
            
            # Projection
            trial = project_to_feasible(trial, n_points)
            
            # Evaluation
            trial_fitness = evaluate_erdos_solution(trial, dx)
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        population = np.array(new_population)
        fitness = np.array(new_fitness)
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= stagnation_limit:
            # Increase mutation factor to escape local optima
            mutation_factor = min(1.0, mutation_factor * 1.1)
            stagnation_count = 0
            
    # Post-processing with SLSQP optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
