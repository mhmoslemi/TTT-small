import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles and velocities
    particles = np.array([h_start.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Project initial particles
    for i in range(n_particles):
        particles[i] = project_to_feasible(particles[i], n_points)
    
    # Personal bests
    pbest = particles.copy()
    pbest_scores = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Global best
    gb_idx = np.argmin(pbest_scores)
    gbest = pbest[gb_idx].copy()
    gbest_score = pbest_scores[gb_idx]
    if gbest_score < best_c5:
        best_h = gb_idx
        best_c5 = gbest_score
    
    # PSO loop
    no_improvement_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                
                # Update global best
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    no_improvement_count = 0
                    if gbest_score < best_c5:
                        best_h = gbest.copy()
                        best_c5 = gbest_score
            else:
                no_improvement_count += 1
                
        # Early stopping
        if no_improvement_count > 100:
            break
            
        # Decay inertia weight
        w = 0.9 - 0.5 * (iteration / max_iter)
    
    # Refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
