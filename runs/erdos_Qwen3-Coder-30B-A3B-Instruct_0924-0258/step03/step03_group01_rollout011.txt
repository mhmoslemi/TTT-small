import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available, otherwise create one
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization parameters
    population_size = 20
    max_iterations = 30
    temperature = 0.1
    cooling_rate = 0.99
    perturbation_magnitude = 0.01
    
    def enforce_constraints(h):
        # Clamp to [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to make sum = n_points/2
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def refine_solution(h):
        # SQP refinement function
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        res = minimize(objective, h, method='SLSQP', 
                       bounds=[(0, 1) for _ in range(n_points)], 
                       constraints={'type': 'eq', 'fun': constraint_sum},
                       options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            refined = res.x
            return refine_solution(enforce_constraints(refined))
        return h
    
    # Generate initial population
    population = [enforce_constraints(h_start + np.random.normal(0, 0.01, n_points)) 
                  for _ in range(population_size)]
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Process each candidate in the population
        for i in range(population_size):
            # Refine the candidate
            refined = refine_solution(population[i])
            c5_refined = evaluate_erdos_solution(refined, dx)
            
            if c5_refined < best_c5:
                best_h = refined
                best_c5 = c5_refined
                
            # Add random perturbation
            perturbed = refined + np.random.normal(0, perturbation_magnitude, n_points)
            perturbed = enforce_constraints(perturbed)
            
            c5_perturbed = evaluate_erdos_solution(perturbed, dx)
            
            # Accept with probability based on temperature
            if c5_perturbed < best_c5 or random.random() < np.exp(-(c5_perturbed - best_c5) / temperature):
                population[i] = perturbed
                if c5_perturbed < best_c5:
                    best_h = perturbed
                    best_c5 = c5_perturbed
        
        # Cool down the temperature
        temperature *= cooling_rate
    
    # Final refinement
    final_h = refine_solution(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
