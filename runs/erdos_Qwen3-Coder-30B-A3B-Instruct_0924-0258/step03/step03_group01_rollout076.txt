import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Nelder-Mead with restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_to_feasible(h):
        # Ensure h is in [0,1] and sum is n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Create initial population
    population_size = 15
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.01, n_points)
        h = project_to_feasible(h)
        population.append(h)
    
    # Nelder-Mead optimization parameters
    max_iter = 500
    total_iter = 0
    
    # Simplex operations
    while time() - start_time < budget_s * 0.9:
        if total_iter >= max_iter:
            break
            
        # Evaluate all candidates
        fitness = [objective(h) for h in population]
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        # Check if we found a better solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Generate new points using simplex operations
        # We'll use a simplified approach for speed
        new_population = [population[0].copy()]
        for i in range(1, population_size):
            # Reflect
            reflected = population[0] + 1.5 * (population[0] - population[i])
            reflected = project_to_feasible(reflected)
            new_population.append(reflected)
            
        population = new_population
        total_iter += 1
    
    # Final gradient-based refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Refine top candidates
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Select top candidates
            candidate_indices = [0, 1, 3]  # Top 3 candidates
            for idx in candidate_indices:
                h_restart = population[idx].copy()
                
                # Add small perturbation
                h_restart = h_restart + np.random.normal(0, 0.005, n_points)
                h_restart = project_to_feasible(h_restart)
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        h_refined = project_to_feasible(h_refined)
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
