import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with existing solution if available
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Try local optimization on existing construction
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Bounded optimization with SLSQP
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run several local optimizations with different starting points
    for i in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try slightly perturbed version of current best
        h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < c5_current:
                    h_current = h_refined
                    c5_current = c5_refined
        except:
            pass
    
    # Try a phase of gradient-based optimization to refine
    try:
        res = minimize(objective, h_current, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = np.clip(res.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < c5_current:
                h_current = h_refined
                c5_current = c5_refined
    except:
        pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h_current, dx)
    
    return (list(h_current), final_c5, n_points)
