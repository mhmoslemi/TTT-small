import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with randomized variable updates
    max_iterations = 500
    tolerance = 1e-6
    current_h = best_h.copy()
    last_c5 = best_c5
    random.seed(seed)
    
    # Precompute overlap indices for speed
    overlap_indices = []

        overlap_indices.append((k, slice(None, -k if k > 0 else None), slice(k, None) if k > 0 else slice(None)))

            break
            
        # Randomize order of variable updates
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        new_h = current_h.copy()
        
        for i in indices:
            # Save original value
            original_h_i = new_h[i]
            
            # Grid search for optimal h[i]
            best_h_i = original_h_i
            best_overlap = float('inf')
            
            # Try several values for h[i]
            for h_i_trial in np.linspace(max(0, original_h_i - 0.1), min(1, original_h_i + 0.1), 30):

        new_h = project_to_feasible(new_h, n_points)

        # Track best solution
        if last_c5 < best_c5:
            best_c5 = last_c5
            best_h = current_h.copy()
            
        # Early stopping if no improvement
        if not improved and iteration > 10:
            break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2
