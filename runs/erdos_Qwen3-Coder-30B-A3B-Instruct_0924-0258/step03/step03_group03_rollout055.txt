import numpy as np
from scipy.signal import correlate
from time import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(arr, epsilon=1e-5):
    # Smooth approximation of max using softmax
    return np.log(np.sum(np.exp(arr / epsilon))) * epsilon

def build_piecewise_representation(h_vals, n_segments=20):
    """Build piecewise constant representation from h values"""
    # Create uniform segments over [0, 2] 
    segment_boundaries = np.linspace(0, 2, n_segments + 1)
    # assign h_vals to segments
    segment_values = h_vals
    if len(segment_values) != n_segments:
        # interpolate to have right number of segments
        xp = np.linspace(0, 2, len(h_vals))
        segment_values = np.interp(segment_boundaries[:-1], xp, h_vals)
    return segment_boundaries, segment_values

def compute_overlap_with_segments(segment_boundaries, segment_values):
    """Fast computation of C5 using piecewise segments"""
    # use discrete correlation approximation
    n = len(segment_boundaries) - 1
    dx = 2.0 / n
    max_overlap = 0.0
    
    # For each possible shift
    for k in range(-n, n+1):
        if k >= 0:
            shifted = segment_values[k:]
            original = segment_values[:len(shifted)]
        else:
            shifted = segment_values[:k]
            original = segment_values[-k:]
        # Compute overlap for this shift
        if len(shifted) > 0 and len(original) > 0:
            overlap = np.sum(original * (1 - shifted))
            max_overlap = max(max_overlap, overlap)
            
    return max_overlap * dx

def trust_region_optimization(h_start, n_points, budget_s):
    """Optimize using trust region approach"""
    # Use scipy trust region to minimize C5
    start_time = time()
    
    # Convert to variable format
    def objective(h_vars):
        # h_vars contains [h0, h1, ..., hn-1]
        # Need to enforce bounds and sum constraint
        h = np.array(h_vars)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return evaluate_erdos_solution(h, 2.0 / n_points)
    
    def constraint_sum(h_vars):
        h = np.array(h_vars)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        return h_sum - n_points / 2
        
    def constraint_bounds(h_vars, i):
        h = np.array(h_vars)
        return h[i] - 1  # lower bound: -inf, upper bound: 1
    
    # bounds for each h_i  
    bounds = Bounds([0.0] * n_points, [1.0] * n_points)
    
    # Define constraint 
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        # Use trust-constr with bounds
        res = minimize(objective, 
                       h_start, 
                       method='trust-constr',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 500, 'disp': False})
        if res.success:
            return res.x, res.fun
    except:
        pass
    
    return h_start, evaluate_erdos_solution(h_start, 2.0 / n_points)
