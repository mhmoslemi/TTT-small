import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Set middle point to 1
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    max_iterations = 10000
    perturbation_rate = 0.2
    local_refinement_steps = 10
    escape_cost_range = [0.01, 0.1]
    
    # Local search function
    def local_search(h_start, max_iter=10):
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_current = h_start.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                res = minimize(obj_func, h_current, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_current = res.x.copy()
                else:
                    break
            except:
                break
        return h_current
    
    # Main tabu search loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.7:
            break
            
        # Generate neighbors by perturbing 20% of points
        num_perturb = max(1, int(n_points * perturbation_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        neighbors = []
        
        for _ in range(10):  # Generate 10 neighbor candidates
            h_candidate = best_h.copy()
            for i in indices:
                # Perturb
                delta = np.random.normal(0, 0.1)
                h_candidate[i] += delta
                # Clamp
                h_candidate[i] = np.clip(h_candidate[i], 0, 1)
            
            # Normalize to meet integral constraint
            h_sum = np.sum(h_candidate)
            if h_sum > 0:
                h_candidate = h_candidate * (n_points / 2) / h_sum
            
            neighbors.append(h_candidate)
        
        # Evaluate neighbors and find best valid one
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_neighbor in neighbors:
            # Check if it's not in tabu list
            neighbor_tuple = tuple(np.round(h_neighbor, 6))
            if neighbor_tuple in tabu_list:
                # Accept with small probability if it's better
                escape_cost = np.random.uniform(*escape_cost_range)
                if escape_cost < 0.1:
                    pass  # Break into acceptance
                else:
                    continue
            
            c5_val = evaluate_erdos_solution(h_neighbor, dx)
            if c5_val < best_neighbor_c5:
                best_neighbor = h_neighbor.copy()
                best_neighbor_c5 = c5_val
        
        # Update best solution
        if best_neighbor is not None:
            # Update tabu list
            tabu_tuple = tuple(np.round(best_h, 6))
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(tabu_tuple)
            
            # Accept new solution
            best_h = best_neighbor
            if best_neighbor_c5 < best_c5:
                best_c5 = best_neighbor_c5
            
            # Local refinement around current solution
            for _ in range(local_refinement_steps):
                if time.time() - start_time > budget_s * 0.95:
                    break
                refined_h = local_search(best_h)
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
                else:
                    break
    
    # Final local refinement
    if time.time() - start_time < budget_s * 0.95:
        final_h = local_search(best_h, max_iter=20)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
