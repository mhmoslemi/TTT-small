import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach with careful handling
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Constraint function for integral constraint
    def integral_constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': integral_constraint}]
    
    # Use a more robust optimization strategy
    max_iter = 500
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Use L-BFGS-B which handles bounds well
            res = minimize(objective, best_h, method='L-BFGS-B', 
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 20, 'ftol': 1e-8})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                # Renormalize to satisfy integral constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except:
            pass
            
    # Final fine-tune with SLSQP
    try:
        res = minimize(objective, best_h, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 50, 'ftol': 1e-9})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
