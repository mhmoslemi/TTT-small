import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different optimization approaches
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    for i in range(50):  # Limit iterations to prevent overfitting
        if time() - start_time > budget_s * 0.95:
            break
            
        # Randomly select a method and perturb the current solution
        method = np.random.choice(methods)
        h_trial = best_h + np.random.normal(0, 0.01, n_points)
        h_trial = project_to_feasible(h_trial, n_points)
        
        # Gradient-based optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_func(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_func}
        
        try:
            res = minimize(objective, h_trial, method=method,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 20, 'ftol': 1e-8})
            if res.success:
                h_new = project_to_feasible(res.x, n_points)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new

    # Final check on the best solution

                return evaluate_erdos_solution(h, dx)

                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint_func}
            
            res = minimize(objective, best_h, method='trust-constr',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 30, 'ftol': 1e-9})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)

                    best_h = h_final
                    best_c5 = c5_final
        except:

    final_h = project_to_feasible(best_h, n_points)
