import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Try to use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Base symmetric construction
        h_start = np.ones(n_points) * 0.5
        # Adjust for better initialization
        h_start[10:25] = 0.8
        
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_c5 = current_c5
    best_n_points = n_points

    # Convex relaxation helper for structured approach
    def solve_convex_relaxation(h_guess, n_points, dx):
        h = cp.Variable(n_points)
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2]
        
        # Try to build a surrogate that helps with optimization
        # Not directly solving the full optimization, but guiding with structure
        obj = cp.Minimize(cp.sum_squares(h - h_guess))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_guess.copy()
    
    # Refinement via gradient descent with projection
    def refine_solution(h_init, n_points, dx, max_iter=1000):
        h = h_init.copy()
        step_size = 0.001
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            # Approximate gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h, dx)
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[j] = (score_plus - base_score) / eps
            
            # Update
            h = h - step_size * grad
            h = project_to_feasible(h, n_points)
            
            # Evaluate
            new_score = evaluate_erdos_solution(h, dx)
            if new_score < best_c5:
                best_c5 = new_score
                best_h = h.copy()
                best_n_points = n_points
                
        return h
    
    # Try different resolutions systematically
    resolutions = [25, 30, 35, 40]
    for r in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if r != n_points:
            # Interpolate
            h_interp = np.interp(np.linspace(0, 2, r), 
                                np.linspace(0, 2, n_points), 
                                best_h)
            h_start = project_to_feasible(h_interp, r)
            n_points = r
            dx = 2.0 / n_points

        # Convex relaxation approach for structure
        h_relaxed = solve_convex_relaxation(h_start, n_points, dx)
        
        # Refinement
        h_refined = refine_solution(h_relaxed, n_points, dx, 500)
        
        new_score = evaluate_erdos_solution(h_refined, dx)
        if new_score < best_c5:
            best_c5 = new_score
            best_h = h_refined.copy()
            best_n_points = n_points

    # Final refinement
    final_h = refine_solution(best_h, best_n_points, 2.0/best_n_points, 500)
    final_c5 = evaluate_erdos_solution(final_h, 2.0/best_n_points)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
        best_n_points = len(final_h)

    return (list(best_h), best_c5, best_n_points)
