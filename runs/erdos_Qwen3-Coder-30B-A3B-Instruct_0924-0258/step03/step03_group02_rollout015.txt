```python
import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    n_clusters = 10
    max_iter = 200
    cluster_freq = 50
    max_local_iter = 20
    
    # Initialize h
    h_current = project_to_feasible(h_start, n_points)
    
    # Compute overlap vectors for all shifts
    shift_vectors = []
    for k in range(n_points):
        # Compute overlap vector for shift k
        if k == 0:
            overlap_vec = h_current * (1 - h_current)
        else:
            overlap_vec = h_current[:-k] * (1 - h_current[k:])
        shift_vectors.append(overlap_vec)
    
    # Cluster shifts based on overlap characteristics (first few samples)
    shift_data = np.array(shift_vectors).T
    kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
    cluster_labels = kmeans.fit_predict(shift_data[:50]) if shift_data.shape[0] > 50 else kmeans.fit_predict(shift_data)
    
    iteration = 0
    no_improvement_count = 0
    
    # Optimization loop
    while time() - start_time < budget_s * 0.95 and iteration < max_iter:
        iteration += 1
        
        # Periodic re-clustering
        if iteration % cluster_freq == 0:
            # Recompute overlap vectors
            shift_vectors = []
            for k in range(n_points):
                if k == 0:
                    overlap_vec = h_current * (1 - h_current)
                else:
                    overlap_vec = h_current[:-k] * (1 - h_current[k:])
                shift_vectors.append(overlap_vec)
            
            shift_data = np.array(shift_vectors).T
            try:
                kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
                kmeans.fit(shift_data[:50] if shift_data.shape[0] > 50 else shift_data)
                cluster_labels = kmeans.labels_
            except:
                pass
        
        # Optimize each cluster
        for cluster_id in range(n_clusters):
            # Get indices for this cluster
            cluster_indices = np.where(cluster_labels == cluster_id)[0]
            if len(cluster_indices) == 0:
                continue
            
            # Store current h
            h_old = h_current.copy()
            
            # Local optimization
            for _ in range(max_local_iter):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Compute current overlap
                current_overlap = np.zeros(n_points)
                for k_idx in cluster_indices:
                    k = k_idx  # align with 0-based indexing
                    if k == 0:
                        overlap_k = h_current * (1 - h_current)
                    else:
                        overlap_k = h_current[:-k] * (1 - h_current[k:])
                    # Accumulate overlap (applying to all elements where k overlaps)
                    if k == 0:
                        current_overlap += overlap_k
                    else:
                        current_overlap[:len(overlap_k)] += overlap_k
                # Normalize/normally distributed to get gradient direction
                gradient = np.zeros_like(h_current)
                for k_idx in cluster_indices:
                    k = k_idx
                    if k > 0:
                        gradient[k:] += h_current[:-k] - h_current[k:]  # Gradient of the overlap
                    else:
                        # For k=0, not directly part of convolution
                        pass
                
                # Simple gradient descent step
                step_size = 0.01
                h_new = h_current - step_size * gradient
                h_new = project_to_feasible(h_new, n_points)
                h_current = h_new
                
            # Convex relaxation (simplified)
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Approximate coefficient matrix from cluster indices
                C = cp.Constant(np.zeros((len(cluster_indices), n_points)))
                for i, k_idx in enumerate(cluster_indices):
                    k = k_idx
                    if k == 0:
                        # Full convolution correlation
                        C[i, :] = 1 - 2 * h_current  # approximated gradient components
                    else:
                        # Approximate overlap constraints from cluster shifts
                        # This is a heuristic, not exact but scalable
                        # Use a simple way to get gradients for each cluster shift
                        C[i, :len(h_current)-k] = h_current[k:] - h_current[:-k]
                
                obj = cp.Minimize(cp.sum_squares(C @ h_var))
                prob = cp.Problem(cp.Minimize(obj), constraints)
                prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_var.value
                    h_projected = project_to_feasible(h_feasible, n_points)
                    c5_test = evaluate_erdos_solution(h_projected, dx)
                    if c5_test < best_c5:
                        best_h = h_projected.copy()
                        best_c5 = c5_test
                    h_current = h_projected
            except:
                pass
                    
        # Evaluate and check improvement
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 100:
            break
            
        # Occasionally refine with SLSQP
        if iteration % 20 == 0 and time() - start_time < budget_s * 0.95:
            try:
                def refine_obj(h):
                    return evaluate_erdos_solution(h, dx)
                def refine_const(h):
                    return np.sum(h) - n_points / 2
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraints = {'type': 'eq', 'fun': refine_const}
                res = minimize(refine_obj, h_current, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_refined = project_to_feasible(res.x, n_points)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                    h_current = h_refined
            except:
                pass
    
    # Final refinement
    if time() - start_time < budget_s - 10:
        try:
            def refine_obj(h):
                return evaluate_erdos_solution(h, dx)
            def refine_const(h):
                return np.sum(h) - n_points / 2
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_const}
            res = minimize(refine_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
```
