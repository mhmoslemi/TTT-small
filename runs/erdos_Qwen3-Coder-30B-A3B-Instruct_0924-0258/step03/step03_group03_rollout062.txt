import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(h, beta):
    """Smooth approximation of the maximum using log-sum-exp"""
    corr = np.correlate(h, 1 - h, mode='full')
    return (1/beta) * np.log(np.sum(np.exp(beta * corr)))

def gradient(h, beta, eps=1e-5):
    """Numerical gradient of the smooth max objective"""
    grad = np.zeros_like(h)
    base_score = smooth_max(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = smooth_max(h_plus, beta)
        grad[i] = (score_plus - base_score) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Project to feasible region
    h = project_to_feasible(h, n_points)
    
    # Parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    # Start with current solution
    current_c5 = smooth_max(h, beta)
    best_h = h.copy()
    best_c5 = current_c5
    
    iter = 0
    while iter < max_iter and time() - start_time < budget_s * 0.95:
        # Compute gradient
        grad = gradient(h, beta)
        
        # Try step
        new_h = h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)
        new_c5 = smooth_max(new_h, beta)
        
        # Accept step if it improves
        if new_c5 < current_c5:
            h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Update beta
        if iter % 100 == 0 and iter > 0:
            beta *= 1.5
            
        # Check convergence
        if np.abs(current_c5 - smooth_max(h, beta)) < tolerance:
            break
            
        iter += 1
    
    # Final evaluation using the true objective
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
