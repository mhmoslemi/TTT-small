import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Cluster-based initialization
    h_init = np.zeros(n_points)
    cluster_size = 4
    cluster_centers = [0.5, 1.5]  # Positions to place clusters
    
    for c in cluster_centers:
        start = max(0, int((c - cluster_size/2)/dx))
        end = min(n_points, int((c + cluster_size/2)/dx))
        h_init[start:end] = 1.0 / (2*cluster_size)
    
    # Normalize to ensure sum is n/2
    scale = (n_points/2) / np.sum(h_init)
    h_init *= scale
    h_best = h_init.copy()
    c5_best = evaluate_erdos_solution(h_best, dx)
    
    # Population-based optimization with restarts
    population_size = 15
    population = [h_best.copy()]
    for i in range(1, population_size):
        h_pert = h_best + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Nelder-Mead iterations
    max_iterations = 20
    iter_count = 0
    
    while iter_count < max_iterations and (time() - start_time) < budget_s * 0.9:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        costs = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(costs)
        population = [population[i] for i in sorted_indices]
        costs = [costs[i] for i in sorted_indices]
        
        # Check if we improved
        if costs[0] < c5_best:
            h_best = population[0].copy()
            c5_best = costs[0]
        
        # Construct new simplex: take best, worst, and centroid
        best = population[0]
        worst = population[-1]
        centroid = np.mean(population[:-1], axis=0)
        
        # Reflection
        r = 1.0 + 1.0 * (centroid - worst)
        r = np.clip(r, 0, 1)
        r = r * (n_points/2) / np.sum(r)
        
        # Evaluate reflection point
        cr = evaluate_erdos_solution(r, dx)
        if costs[0] <= cr < costs[-2]:
            population[-1] = r
        elif cr < costs[0]:
            # Expansion
            e = 1.0 + 2.0 * (r - centroid)
            e = np.clip(e, 0, 1)
            e = e * (n_points/2) / np.sum(e)
            ce = evaluate_erdos_solution(e, dx)
            if ce < cr:
                population[-1] = e
            else:
                population[-1] = r
        else:
            # Contraction
            c = 0.5 * (worst + centroid)
            c = np.clip(c, 0, 1)
            c = c * (n_points/2) / np.sum(c)
            cc = evaluate_erdos_solution(c, dx)
            if cc < costs[-1]:
                population[-1] = c
            else:
                # Shrink
                for i in range(1, len(population)):
                    population[i] = population[0] + 0.5 * (population[i] - population[0])
                    population[i] = np.clip(population[i], 0, 1)
                    population[i] = population[i] * (n_points/2) / np.sum(population[i])
        
        iter_count += 1
        
    # Final refinement using gradient descent
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    for i in range(3):  # Refine top candidates
        if time() - start_time > budget_s * 0.95:
            break
            
        h_restart = population[i].copy()
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_restart, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < c5_best:
                    h_best = h_refined
                    c5_best = c5_refined
        except:
            pass
    
    return (list(h_best), c5_best, n_points)
