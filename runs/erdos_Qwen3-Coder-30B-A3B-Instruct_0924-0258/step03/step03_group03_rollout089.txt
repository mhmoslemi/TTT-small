import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(corr, beta):
    """Smooth approximation to max function"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def smooth_objective(h, beta):
    """Compute smooth approximation to C5"""
    corr = np.correlate(h, 1 - h, mode='full')
    return smooth_max(corr, beta)

def compute_gradient(h, beta, eps=1e-5):
    """Compute gradient of smooth objective"""
    grad = np.zeros_like(h)
    base_obj = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_obj) / eps
    return grad

    # Project to feasible region
    h = project_to_feasible(h, n_points)
    
    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    best_n_points = n_points

    # Parameters for optimization
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient and update
        grad = compute_gradient(h, beta)
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new objective
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept the step if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Update smoothing parameter
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.5
            
        # Check for convergence
        if iteration > 50 and abs(best_c5 - smooth_objective(h, beta)) < tolerance:
            break

    # Resolution refinement
    if time() - start_time < budget_s * 0.8 and n_points < 100:
        # Try higher resolution
        n_points_refined = min(n_points * 2, 100)
        h_refined = np.interp(np.linspace(0, 2, n_points_refined),

        h_refined = project_to_feasible(h_refined, n_points_refined)

        dx_refined = 2.0 / n_points_refined

            h_refined_new = h_refined - step_size * grad
            h_refined_new = project_to_feasible(h_refined_new, n_points_refined)

                if step_size < 1e-3:

            else:
                step_size *= 0.5
                
    return (list(best_h), best_c5, best_n_points)
