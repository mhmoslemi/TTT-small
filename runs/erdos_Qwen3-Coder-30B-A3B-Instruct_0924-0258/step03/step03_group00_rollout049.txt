import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Set middle point to 1 to satisfy integral constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.7:
        # Generate neighbors by perturbing 20% of points
        neighbor_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        neighbors = []
        
        for _ in range(10):  # Generate 10 neighbor candidates
            h_neighbor = best_h.copy()
            for i in neighbor_indices:
                # Random perturbation
                perturb = np.random.uniform(-0.1, 0.1)
                h_neighbor[i] += perturb
                # Clamp to [0,1]
                h_neighbor[i] = np.clip(h_neighbor[i], 0, 1)
            
            # Project to satisfy integral constraint
            h_sum = np.sum(h_neighbor)
            if h_sum > 0:
                h_neighbor = h_neighbor * (n_points / 2) / h_sum
            
            neighbors.append(h_neighbor)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Check if in tabu list
            candidate_tuple = tuple(h_candidate.round(6))
            if candidate_tuple in tabu_list:
                # With low probability, accept tabu move (escape mechanism)
                if np.random.random() > 0.8:
                    pass  # Allow tabu move
                else:
                    continue  # Skip tabu move
            
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = c5_candidate
        
        # Update best solution
        if best_neighbor is not None:
            # Add previous best to tabu list
            previous_tuple = tuple(best_h.round(6))
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(previous_tuple)
            
            best_h = best_neighbor
            if best_neighbor_c5 < best_c5:
                best_c5 = best_neighbor_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        best_local = best_h.copy()
        best_local_c5 = best_c5
        
        # Try multiple restarts
        for _ in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Random perturbation
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_local_c5:
                        best_local = res.x.copy()
                        best_local_c5 = c5_test
            except:
                continue
        
        if best_local_c5 < best_c5:
            best_h = best_local
            best_c5 = best_local_c5
    
    return (list(best_h), best_c5, n_points)
