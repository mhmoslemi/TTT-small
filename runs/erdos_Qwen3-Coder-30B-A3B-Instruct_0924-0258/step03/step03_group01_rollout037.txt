import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based local optimization with simulated annealing
    population_size = 20
    max_iterations = 30
    initial_temp = 0.1
    decay_rate = 0.99
    
    # Create initial population
    population = []
    for _ in range(population_size):
        candidate = h_start + np.random.normal(0, 0.01, n_points)
        candidate = np.clip(candidate, 0, 1)
        # Normalize to correct sum
        candidate *= (n_points/2) / np.sum(candidate)
        population.append(candidate)
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        temp = initial_temp * (decay_rate ** iteration)
        
        # Evaluate and sort population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = [population[0]]  # Keep best
        
        # Generate offspring with mutations
        for i in range(1, population_size):
            # Select parent
            parent = population[random.randint(0, population_size//2)]
            
            # Create offspring by adding noise
            offspring = parent + np.random.normal(0, 0.01, n_points)
            offspring = np.clip(offspring, 0, 1)
            
            # Normalize sum
            offspring *= (n_points/2) / np.sum(offspring)
            
            # Apply simulated annealing acceptance criterion
            if i < population_size // 2:
                # Try local optimization first
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, offspring, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        offspring = res.x
                except:
                    pass
            
            # Accept or reject with simulated annealing
            current_fitness = evaluate_erdos_solution(offspring, dx)
            if current_fitness < fitness[0]:
                new_population.append(offspring)
            else:
                # Accept with probability based on temperature
                delta = current_fitness - fitness[0]
                if random.random() < np.exp(-delta / temp):
                    new_population.append(offspring)
                else:
                    new_population.append(population[i])
        
        # Update population
        population = new_population
    
    # Final optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
    bounds = [(0, 1) for _ in range(n_points)]
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Try multiple restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.005, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
