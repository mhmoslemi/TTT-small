import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    
    # Initialize PSO
    particles = np.random.rand(swarm_size, n_points) * 0.5 + 0.25
    velocities = np.zeros((swarm_size, n_points))
    particle_best = particles.copy()
    particle_best_values = np.full(swarm_size, np.inf)
    
    # Global best
    g_best = best_h.copy()
    g_best_value = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            h = project_to_feasible(particles[i], n_points)
            c5 = evaluate_erdos_solution(h, dx)
            
            if c5 < particle_best_values[i]:
                particle_best_values[i] = c5
                particle_best[i] = h.copy()
            
            if c5 < g_best_value:
                g_best_value = c5
                g_best = h.copy()
        
        # Update particles
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (inertia * velocities[i] 
                           + cognitive * r1 * (particle_best[i] - particles[i])
                           + social * r2 * (g_best - particles[i]))
            particles[i] += velocities[i]
        
        # Maintain feasibility
        for i in range(swarm_size):
            particles[i] = project_to_feasible(particles[i], n_points)
    
    # Refinement with L-BFGS
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def ref_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(ref_obj, g_best, method='L-BFGS-B', 
                      bounds=[(0,1) for _ in range(n_points)], 
                      options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < g_best_value:
                g_best = h_final.copy()
                g_best_value = c5_final
    
    return (list(g_best), g_best_value, n_points)
