import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a gradient-based approach with local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Try multiple random initializations with different strategies
    strategies = [
        # Original starting point
        lambda: h_start.copy(),
        # Random perturbation
        lambda: np.clip(h_start + np.random.normal(0, 0.1, n_points), 0, 1),
        # Alternating pattern
        lambda: np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]),
        # Single peak
        lambda: np.array([0.5 * (1 + np.cos(np.pi * i / (n_points - 1))) if 0 <= i < n_points else 0 for i in range(n_points)])
    ]
    
    # Global optimization using multiple restarts
    for strategy in strategies:
        remaining_time = budget_s - (time() - start_time)
        if remaining_time < 10:
            break
            
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            
            h_restart = strategy()
            # Ensure sum constraint
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            # Bounds
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue

    # Final convex optimization validation
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add overlap constraints using a sparse grid of shifts
            for k in range(1, n_points):
                # Check a subset of possible overlaps to reduce problem size
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                constraints.append(corr_sum <= best_c5 * dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_convex = h.value
                c5_convex = evaluate_erdos_solution(h_convex, dx)
                if c5_convex < best_c5:
                    best_h = h_convex
                    best_c5 = c5_convex
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
