import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Improved local search
    def local_optimization(h_guess, max_iter=100):
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
        
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_guess, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': max_iter, 'ftol': 1e-8})
        
        if res.success:
            h_opt = np.clip(res.x, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            return h_opt
        else:
            h_opt = np.clip(h_guess, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            return h_opt

    # Refinement with increased resolution
    h_best = best_h.copy()
    c5_best = best_c5

    # Try with higher resolution if time allows
    if budget_s > 400:
        n_points_high = 70
        dx_high = 2.0 / n_points_high
        # Interpolate initial solution to higher resolution
        h_interp = np.interp(np.linspace(0, 2, n_points_high), 
                           np.linspace(0, 2, n_points), h_start)
        h_interp = np.clip(h_interp, 0, 1)
        h_interp = h_interp * (n_points_high / 2) / np.sum(h_interp)
        
        # Local optimization
        h_refined = local_optimization(h_interp, max_iter=50)
        c5_refined = evaluate_erdos_solution(h_refined, dx_high)
        if c5_refined < c5_best:
            c5_best = c5_refined
            h_best = h_refined
            n_points = n_points_high
            dx = dx_high

    # Fine-tune current best
    h_fine = local_optimization(h_best, max_iter=50)
    c5_fine = evaluate_erdos_solution(h_fine, dx)
    if c5_fine < c5_best:
        c5_best = c5_fine
        h_best = h_fine

    # Additional improvement by sampling different strategies
    if time() - start_time < budget_s * 0.7:
        n_attempts = 5
        for attempt in range(n_attempts):
            if time() - start_time > budget_s * 0.95:
                break
            # Random perturbation to escape local minima
            h_random = h_best + np.random.normal(0, 0.05, n_points)
            h_random = np.clip(h_random, 0, 1)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            
            # Local optimization of random guess
            h_opt = local_optimization(h_random, max_iter=30)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < c5_best:
                c5_best = c5_opt
                h_best = h_opt

    return (list(h_best), c5_best, n_points)
