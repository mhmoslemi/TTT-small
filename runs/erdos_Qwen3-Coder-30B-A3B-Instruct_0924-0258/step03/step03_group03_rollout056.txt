import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def build_segmented_initial(n_segments):
    """Build initial segmented representation"""
    h = np.ones(n_segments) * 0.5
    # Create piecewise constant with structure
    # More symmetric than uniform, trying to minimize peak interactions
    if n_segments >= 15:
        # Insert some structure
        for i in range(n_segments):
            if i < n_segments // 3:
                h[i] = 0.3
            elif i >= 2 * n_segments // 3:
                h[i] = 0.7
            else:
                h[i] = 0.5
    elif n_segments >= 5:
        # Structure for medium
        h[0] = 0.2
        h[-1] = 0.8
        h[1] = h[-2] = 0.5
    return h

def smooth_max(x, radius=0.05):
    """Smooth approximation of max function, to enable gradient computation"""
    return np.log(np.sum(np.exp(x / radius))) * radius

def gradient_estimate(h, dx, eps=1e-6):
    """Estimate gradient via finite differences"""
    grad = np.zeros_like(h)
    base_score = evaluate_erdos_solution(h, dx)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - base_score) / eps
    return grad

def trust_region_step(h, dx, trust_radius=0.1, max_iter=100, step_size=0.01):
    """Perform a trust-region step with projected gradient descent"""
    h_current = h.copy()
    
    for _ in range(max_iter):
        grad = gradient_estimate(h_current, dx)
        # Compute update direction (projected gradient descent)
        delta = -step_size * grad
        # Project to trust region
        delta_norm = np.linalg.norm(delta)
        if delta_norm > trust_radius:
            delta = delta * trust_radius / delta_norm
        h_new = h_current + delta
        # Project to constraints
        h_new = project_to_feasible(h_new, len(h_new))
        new_score = evaluate_erdos_solution(h_new, dx)
        old_score = evaluate_erdos_solution(h_current, dx)
        if new_score < old_score:
            h_current = h_new
        else:
            # If no improvement, shrink trust region
            trust_radius *= 0.5
            if trust_radius < 1e-6:
                break
    return h_current

    max_trust_radius = 0.1
    current_h = best_h
    current_c5 = best_c5

    test_points = [15, 20, 25, 30, 35]

    for n_points_try in test_points:

        # Convert existing solution to new resolution
        if n_points_try != n_points:
            # Interpolate with linear interpolation
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, n_points_try)
            current_h = np.interp(new_x, old_x, current_h)
