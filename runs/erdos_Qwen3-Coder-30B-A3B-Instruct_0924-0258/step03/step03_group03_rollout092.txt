import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def line_search(f, grad, x, dx, max_iter=10):
    alpha = 1.0
    beta = 0.5
    c = 1e-4
    fx = f(x)
    for _ in range(max_iter):
        x_new = x - alpha * grad
        x_new = project_to_feasible(x_new, len(x))
        fx_new = f(x_new)
        if fx_new <= fx - c * alpha * np.dot(grad, grad):
            return alpha, x_new
        alpha *= beta
    return alpha, x_new

    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_c5 = current_c5
    
    # Number of iterations to run
    max_iter = 500
    step_size = 0.01
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation via finite differences
        grad = np.zeros_like(best_h)

            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[j] = (score_plus - current_c5) / eps
        
        # Update with gradient descent
        best_h = best_h - step_size * grad
        best_h = project_to_feasible(best_h, n_points)
        current_c5 = evaluate_erdos_solution(best_h, dx)

    for i in range(100):
        if time() - start_time > budget_s * 0.95:

        grad = np.zeros_like(best_h)
        eps = 1e-5
        
        for j in range(len(best_h)):

            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)

            step = 0.01
            
        best_h = best_h - step * grad
        best_h = project_to_feasible(best_h, n_points)
        current_c5 = evaluate_erdos_solution(best_h, dx)
