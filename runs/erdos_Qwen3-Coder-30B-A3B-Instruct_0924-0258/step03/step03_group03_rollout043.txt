import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
    
    # initial evaluation
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_n_points = n_points

    # Try a few different sized samples to get a good baseline
    for n_try in [25, 30, 35, 40]:
        if time() - start_time > budget_s * 0.2:
            break
        dx_try = 2.0 / n_try
        h_interp = np.interp(np.linspace(0, 2, n_try), 
                            np.linspace(0, 2, len(h_start)), 
                            h_start)
        h_feasible = project_to_feasible(h_interp, n_try)
        score = evaluate_erdos_solution(h_feasible, dx_try)
        if score < best_c5:
            best_c5 = score
            best_h = h_feasible.copy()
            best_n_points = n_try

    # Use simulated annealing approach with local search refinements
    # Parameters
    max_iter = 5000
    temp_start = 0.1
    temp_decay = 0.999
    min_temp = 1e-8
    
    current_h = best_h.copy()
    current_c5 = best_c5
    temp = temp_start

    # Mutate function for SA
    def mutate_h(h, dx, temp_val):
        h_new = h.copy()
        # Randomly change a few values
        n_changes = max(1, int(len(h) * 0.1))
        indices = np.random.choice(len(h), n_changes, replace=False)
        for i in indices:
            # Perturb with Gaussian noise scaled by temperature
            pert = np.random.normal(0, temp_val * 0.5)
            h_new[i] = h_new[i] + pert
        return project_to_feasible(h_new, len(h_new))

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor
        h_new = mutate_h(current_h, dx, temp)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temp):
            current_h = h_new.copy()
            current_c5 = c5_new
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                best_n_points = len(current_h)
        
        # Cooling
        temp = max(min_temp, temp * temp_decay)

    # Final gradient descent refinement
    if time() - start_time < budget_s * 0.95:
        h_refined = best_h.copy()
        dx_refined = 2.0 / best_n_points
        
        for _ in range(300):  # Limited gradient descent
            if time() - start_time > budget_s * 0.95:
                break
                
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx_refined)
            
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx_refined)
                grad[i] = (score_plus - base_score) / eps

            # Simple gradient step with small learning rate
            step_size = 0.005
            h_refined = h_refined - step_size * grad
            h_refined = project_to_feasible(h_refined, best_n_points)

            score_refined = evaluate_erdos_solution(h_refined, dx_refined)
            if score_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = score_refined

    return (list(best_h), best_c5, best_n_points)
