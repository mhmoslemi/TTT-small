import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple local search with perturbations
    max_iter = 5000
    iter_count = 0
    
    # Tabu list to avoid revisiting solutions
    tabu_list = []
    tabu_size = 10
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-6
    
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
        # Create neighbor solution by perturbing one element
        h_new = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new[idx] += delta
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Normalize to maintain integral constraint
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        # Check if solution is in tabu list (very basic check)
        h_key = tuple(np.round(h_new, 3))
        if h_key in tabu_list:
            temp *= cooling_rate
            iter_count += 1
            continue
        else:
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(h_key)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            best_h = h_new.copy()
            best_c5 = c5_new
            
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        iter_count += 1
        
        # Occasionally refine with local optimization
        if iter_count % 50 == 0 and time.time() - start_time < budget_s * 0.9:
            def obj_func(h):
                h_clipped = np.clip(h, 0, 1)
                h_sum = np.sum(h_clipped)
                if h_sum > 0:
                    h_normalized = h_clipped * (n_points / 2) / h_sum
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(obj_func, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
    
    return (list(best_h), best_c5, n_points)
