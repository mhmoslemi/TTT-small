import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segment_gradient(h, dx, x_bounds, w_values,
                            beta=100, smooth=True):
    """Compute gradients with respect to segment heights and boundaries"""
    n_segments = len(w_values)
    
    # Build piecewise function from segments
    h_func = np.zeros(int(2/dx) + 1)
    for i in range(n_segments):
        start_idx = int(x_bounds[i] / dx)
        end_idx = int(x_bounds[i+1] / dx)
        h_func[start_idx:end_idx] = w_values[i]
    
    # Evaluate correlation
    corr = np.correlate(h_func, 1 - h_func, mode='full')
    max_corr = np.max(corr)
    
    if not smooth or max_corr <= 0:
        # Gradient for uniform segments
        grad_w = np.zeros(n_segments)
        grad_x = np.zeros(n_segments + 1)
        return grad_w, grad_x
    
    # Smooth approximation: log-sum-exp
    # This is a simplified approximation, works better in practice 
    # than the full log-sum-exp with correct formulations
    sigma_corr = max_corr
    
    # Simplified gradient computation via finite differences
    eps = 1e-5
    grad_w = np.zeros(n_segments)
    grad_x = np.zeros(n_segments + 1)
    
    # Gradient w.r.t. segment heights
    for i in range(n_segments):
        w_plus = w_values.copy()
        w_plus[i] += eps
        # Need to reconstruct h_func
        h_plus = np.zeros_like(h_func)
        for j in range(n_segments):
            start_idx = int(x_bounds[j] / dx)
            end_idx = int(x_bounds[j+1] / dx)
            h_plus[start_idx:end_idx] = w_plus[j] if j == i else w_values[j]
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        max_plus = np.max(corr_plus)
        grad_w[i] = (max_plus - max_corr) / eps
    
    # Gradient w.r.t. segment boundaries
    for i in range(n_segments + 1):
        x_plus = x_bounds.copy()
        if i > 0 and i < len(x_bounds) - 1:
            x_plus[i] += eps
        else:
            continue
        # Constraint: x_bounds must be sorted, so exact update is trickier
        # Simplify for now: skip gradient for boundary updates since they may 
        # not improve if derivatives are low; focus on height optimization.
        continue  # Simplified: don't compute boundary gradient

    return grad_w, grad_x

def segment_refinement(h, dx, x_bounds, w_values, beta=100):

    # Add roughly 1-3 segments

    if new_segs <= 0:
        return x_bounds, w_values
    
    # Add new segments in already large regions
    # Try insert in intervals with high variation or high value
    insert_points = []
    
    # Insert where w values are high/low: signal variation points
    for i in range(n_segs - 1):  # skip last
        if abs(w_values[i] - 0.5) > 0.3:
            # Midpoint approximation
            midpt = (x_bounds[i] + x_bounds[i+1]) / 2
            insert_points.append(midpt)
    
    # Add new segment points
    new_bounds = x_bounds.copy()
    new_w = w_values.copy()
    
    # Place evenly in intervals where values deviate most
    sorted_points = sorted(insert_points[:new_segs])
    for i, x_insert in enumerate(sorted_points):
        pos = np.searchsorted(new_bounds, x_insert)
        new_bounds = np.insert(new_bounds, pos, x_insert)
        # Simple interpolation for weights
        new_w = np.insert(new_w, pos, np.interp(x_insert, new_bounds[:-1], new_w))
    
    return new_bounds, new_w
