import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a different optimization approach - gradient-based refinement
    def smooth_function(h):
        """Apply smoothing to make the function more likely to be optimal"""
        # Smooth by averaging with neighbors
        h_smooth = h.copy()
        for i in range(1, len(h) - 1):
            h_smooth[i] = 0.7 * h[i] + 0.15 * h[i-1] + 0.15 * h[i+1]
        return np.clip(h_smooth, 0, 1)
    
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Helper to maintain constraints
    def normalize_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec

    # Multi-start local search with smart initialization
    num_starts = 10
    max_iter = 50
    
    for i in range(num_starts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation of best solution or fresh start
        if i == 0:
            h_current = best_h.copy()
        else:
            # Add noise to get a new start
            h_current = best_h + np.random.normal(0, 0.05, n_points)
        
        h_current = normalize_h(h_current)
        h_current = smooth_function(h_current)

        # Local optimization using SLSQP
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': max_iter, 'ftol': 1e-7})
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
    
    # Try to increase resolution if time permits
    if time() - start_time < budget_s * 0.8 and n_points == 35:
        n_points = 70
        dx = 2.0 / n_points
        h_resampled = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(best_h)), 
                                best_h)
        h_resampled = normalize_h(h_resampled)
        c5_resampled = evaluate_erdos_solution(h_resampled, dx)
        if c5_resampled < best_c5:
            best_h = h_resampled
            best_c5 = c5_resampled

    # Final refinement
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
