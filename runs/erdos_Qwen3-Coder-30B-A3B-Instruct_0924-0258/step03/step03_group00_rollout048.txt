import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        # Set middle point to satisfy integral constraint
        h_start[n_points//2] = 2.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    perturb_rate = 0.2
    max_no_improvement = 50
    no_improvement_count = 0
    
    def project_to_feasible(h):
        """Project h to satisfy integral constraint"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def get_neighbors(h, num_neighbors=10):
        """Generate neighbor solutions by perturbing h"""
        neighbors = []
        for _ in range(num_neighbors):
            h_new = h.copy()
            indices = np.random.choice(n_points, int(n_points * perturb_rate), replace=False)
            for i in indices:
                # Perturb by random amount
                delta = np.random.uniform(-0.1, 0.1)
                h_new[i] += delta
            h_new = project_to_feasible(h_new)
            neighbors.append(h_new)
        return neighbors
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.7:
        neighbors = get_neighbors(best_h, 10)
        
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_test in neighbors:
            # Check if this solution is in tabu list
            h_tuple = tuple(h_test.round(8))
            if h_tuple in tabu_list:
                continue
            
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_neighbor_c5:
                best_neighbor = h_test
                best_neighbor_c5 = c5_test
        
        if best_neighbor is None:
            # No valid neighbor found, escape by picking a random neighbor
            random_h = np.random.rand(n_points)
            random_h = project_to_feasible(random_h)
            best_neighbor = random_h
            best_neighbor_c5 = evaluate_erdos_solution(random_h, dx)
        
        # Update tabu list
        tabu_list.append(tuple(best_h.round(8)))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Move to better neighbor
        best_h = best_neighbor
        if best_neighbor_c5 < best_c5:
            best_c5 = best_neighbor_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
    
    # Local refinement with gradient-based optimization
    def obj_func(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h_norm = h * (n_points / 2) / h_sum
        else:
            h_norm = h
        return evaluate_erdos_solution(h_norm, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    if time.time() - start_time < budget_s * 0.95:
        # Run local refinement
        num_refinements = 10
        current_h = best_h.copy()
        
        for i in range(num_refinements):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            def obj_func_local(h):
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h_norm = h * (n_points / 2) / h_sum
                else:
                    h_norm = h
                return evaluate_erdos_solution(h_norm, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            res = minimize(obj_func_local, current_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 20, 'ftol': 1e-6})
            
            if res.success:
                current_h = res.x.copy()
            else:
                # Fallback to random perturbation if optimization fails
                current_h = project_to_feasible(current_h + np.random.normal(0, 0.01, n_points))
            
            c5_test = obj_func_local(current_h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = current_h.copy()
    
    return (list(best_h), best_c5, n_points)
