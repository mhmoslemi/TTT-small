import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    start_points = [h_start]
    # Random perturbations
    for _ in range(3):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        start_points.append(h_pert)

    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)

                h_refined = np.clip(res.x, 0, 1)
