import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm with local refinement
    population_size = 500
    generations = 100
    elite_size = 50
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        # Random piecewise constant with 0/1 values and breakpoints
        h = np.random.choice([0, 1], size=n_points)
        # Ensure sum constraint (adjust the first segment)
        current_sum = np.sum(h) * dx
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h
    
    def mutate(individual):
        new_individual = individual.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                # Flip value with 50% probability
                new_individual[i] = 1 - new_individual[i]
        return new_individual
    
    def crossover(a, b):
        if np.random.random() > crossover_rate:
            return a.copy(), b.copy()
        # Single point crossover
        point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([a[:point], b[point:]])
        child2 = np.concatenate([b[:point], a[point:]])
        return child1, child2
    
    def select_parents(population, fitnesses):
        # Tournament selection
        selected = []
        for _ in range(population_size):
            candidates = np.random.choice(len(population), size=5, replace=False)
            best = candidates[np.argmin([fitnesses[i] for i in candidates])]
            selected.append(population[best])
        return selected
    
    # Create initial population
    population = [create_individual() for _ in range(population_size)]
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitnesses = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, dx)
            fitnesses.append(c5)
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]
        
        # Elitism
        new_population = population[:elite_size]
        
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Create offspring
        while len(new_population) < population_size:
            parent_a = random.choice(parents)
            parent_b = random.choice(parents)
            child_a, child_b = crossover(parent_a, parent_b)
            child_a = mutate(child_a)
            child_b = mutate(child_b)
            new_population.extend([child_a, child_b])
        
        # Trim to exact size
        population = new_population[:population_size]
    
    # Local refinement on best individuals
    refined_candidates = []
    for i in range(min(10, len(population))):
        h_refine = population[i].copy()
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_func, h_refine, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                refined_candidates.append((res.x.copy(), c5_test))
        except:
            continue
            
    if refined_candidates:
        refined_candidates.sort(key=lambda x: x[1])
        best_refined = refined_candidates[0]
        if best_refined[1] < best_c5:
            best_h = best_refined[0]
            best_c5 = best_refined[1]
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
