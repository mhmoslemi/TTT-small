import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def get_gradient(h, dx, n_points):
    """Approximate gradient of C5 with respect to h values"""
    eps = 1e-6
    grad = np.zeros_like(h)
    base_score = evaluate_erdos_solution(h, dx)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n_points)
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - base_score) / eps
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    initial_segments = 15

    # Use the initial solution as a starting point
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_c5 = current_c5

    # Main optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = get_gradient(best_h, dx, n_points)
        
        # Gradient descent step
        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        # Adaptive step size
        step_size *= 0.99
    
    # Refinement with more precise gradient descent
    h_refined = best_h.copy()
    dx_refined = 2.0 / n_points
