import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing approach with Powell's method refinement
    # Generate initial candidates
    candidates = [h_start]
    for _ in range(10):
        h_perturb = h_start + np.random.normal(0, 0.05, n_points)
        candidates.append(project_to_feasible(h_perturb, n_points))
    
    # Try each candidate with refinement
    for h_init in candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization with SLSQP
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_init, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 30, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            continue
    
    # Additional exploration with gradient-based optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Random restarts
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_random = np.random.uniform(0, 1, n_points)
            h_random = project_to_feasible(h_random, n_points)
            try:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                def cons(h):
                    return np.sum(h) - n_points / 2
                bounds = [(0, 1) for _ in range(n_points)]
                con = {'type': 'eq', 'fun': cons}
                res = minimize(obj, h_random, method='SLSQP',
                              bounds=bounds, constraints=con,
                              options={'maxiter': 20, 'ftol': 1e-8})
                if res.success:
                    h_final = project_to_feasible(res.x, n_points)
                    c5_final = evaluate_erdos_solution(h_final, dx)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except Exception:
                continue
    
    # Final check to ensure best is feasible
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
