import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        return h
    
    def mutate(h, rate=0.3, std=0.03):
        mask = np.random.random(n_points) < rate
        h = h + np.random.normal(0, std, n_points) * mask
        return repair(h)
    
    def crossover(h1, h2):
        alpha = np.random.random()
        h = alpha * h1 + (1 - alpha) * h2
        return repair(h)
    
    # Initialize population
    population_size = 20
    population = [repair(h_start + np.random.normal(0, 0.05, n_points)) for _ in range(population_size)]
    
    # Keep track of best solution
    current_best_h = best_h.copy()
    current_best_c5 = best_c5
    
    # Genetic algorithm loop
    generation = 0
    while time() - start_time < budget_s * 0.95:
        generation += 1
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Selection (top 50%)
        selected_indices = np.argsort(fitness)[:population_size//2]
        selected = [population[i] for i in selected_indices]
        fitness = [fitness[i] for i in selected_indices]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover
        while len(new_population) < population_size:
            i, j = np.random.choice(len(selected), 2, replace=False)
            child = crossover(selected[i], selected[j])
            new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < 0.3:
                new_population[i] = mutate(new_population[i])
        
        # Update population
        population = new_population
        
        # Local refinement with SQP
        if generation % 1 == 0:
            for i in range(min(3, len(population))):
                h_ref = population[i].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_ref, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < current_best_c5:
                            current_best_h = h_refined
                            current_best_c5 = c5_refined
                except:
                    pass
        
        # Update global best
        for i in range(len(population)):
            c5 = evaluate_erdos_solution(population[i], dx)
            if c5 < current_best_c5:
                current_best_h = population[i]
                current_best_c5 = c5
    
    # Final refinement of best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(objective, current_best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            current_best_h = res.x
            current_best_c5 = objective(current_best_h)
    except:
        pass
    
    return (list(current_best_h), current_best_c5, n_points)
