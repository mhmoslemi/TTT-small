import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm for optimization
    population_size = 500
    generations = 100
    elite_size = 10
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        population.append(h)
    
    # Add initial construction to population
    population.append(h_start)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create offspring
        new_population = population[:elite_size]
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(population) >= 2:
                # Tournament selection
                parent1 = population[np.random.randint(0, len(population))]
                parent2 = population[np.random.randint(0, len(population))]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[i] = np.random.rand()
                    if np.random.rand() < mutation_rate:
                        child2[i] = np.random.rand()
                
                # Normalize
                child1_sum = np.sum(child1)
                if child1_sum > 0:
                    child1 = child1 * (n_points / 2) / child1_sum
                new_population.append(child1)
                
                if len(new_population) < population_size:
                    child2_sum = np.sum(child2)
                    if child2_sum > 0:
                        child2 = child2 * (n_points / 2) / child2_sum
                    new_population.append(child2)
            else:
                # Random individual
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_points / 2) / h_sum
                new_population.append(h)
        
        population = new_population[:population_size]
    
    # Local refinement
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations with different starting points
        num_restarts = 50
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Perturb current best
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
