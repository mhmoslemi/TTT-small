import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with constraints
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def gradient(h_vec):
        # Compute approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        base_val = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Optimization using L-BFGS-B
    max_iter = 1000
    iter_count = 0
    tol = 1e-7
    last_improvement = 0
    
    # Normalize initial solution to meet constraint
    h_norm = h_start * (n_points / 2) / np.sum(h_start)
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Try to improve with local optimization
        try:
            cons = [{'type': 'eq', 'fun': constraint_func}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_norm, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': tol, 'gtol': tol})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    last_improvement = iter_count
        except:
            pass
        
        # Now use gradient-based descent at current best
        try:
            # Small gradient descent step
            grad = gradient(best_h)
            learning_rate = 0.001 / (1.0 + iter_count / 1000.0)
            h_new = best_h - learning_rate * grad
            
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Re-normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                last_improvement = iter_count
        except:
            pass
            
        # Occasionally add some randomness to escape local minima
        if iter_count % 100 == 0 and iter_count > 0:
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.005, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_pert = evaluate_erdos_solution(h_perturbed, dx)
            if c5_pert < best_c5:
                best_h = h_perturbed
                best_c5 = c5_pert
        
        iter_count += 1
        
        # Early stopping if no improvement for too long
        if iter_count - last_improvement > 1000:
            break
    
    # Final refinement with scipy minimize using final solution as guess
    try:
        cons = [{'type': 'eq', 'fun': constraint_func}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Return best solution
    return (list(best_h), best_c5, n_points)
