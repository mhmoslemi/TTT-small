import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization with simulated annealing
    population_size = 20
    max_iterations = 30
    temperature = 0.1
    decay_rate = 0.99
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Create diversified initial solutions
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all population members
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Find best individual in population
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitness_scores[min_idx]
        
        # Refine each individual with SQP
        for i in range(population_size):
            h_old = population[i].copy()
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_old, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    # Ensure feasibility
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    population[i] = h_refined
            except:
                pass
        
        # Simulated Annealing step
        for _ in range(population_size):
            if time() - start_time > budget_s * 0.95:
                break
            # Pick a random individual
            idx = np.random.randint(0, population_size)
            h_current = population[idx].copy()
            
            # Generate neighbor by small random perturbation
            h_neighbor = h_current + np.random.normal(0, 0.01, n_points)
            h_neighbor = np.clip(h_neighbor, 0, 1)
            h_neighbor = h_neighbor * (n_points / 2) / np.sum(h_neighbor)
            
            c5_current = evaluate_erdos_solution(h_current, dx)
            c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
            
            # Accept with simulated annealing probability
            if c5_neighbor < c5_current:
                population[idx] = h_neighbor
            else:
                delta = c5_neighbor - c5_current
                if np.random.rand() < math.exp(-delta / temperature):
                    population[idx] = h_neighbor
        
        # Cool down temperature
        temperature *= decay_rate
    
    return (list(best_h), best_c5, n_points)
