import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic binary pattern
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        h_start[:ones_count] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    max_iter = 10000
    eta = 0.5  # step size
    perturb_rate = 0.1
    
    # Precompute all possible shifts for correlation
    shifts = range(1, n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap
        overlap_vals = np.correlate(best_h, 1 - best_h, mode='full')
        best_k = np.argmax(overlap_vals)
        # Adjust for zero shift
        if best_k == 0:
            overlap_vals[0] = -np.inf
            best_k = np.argmax(overlap_vals)
        
        # Compute shift vector
        if best_k < n_points:
            shifted = np.pad(best_h, (0, best_k), mode='constant')
            shifted = shifted[:n_points]
        else:
            shifted = np.pad(best_h, (best_k, 0), mode='constant')
            shifted = shifted[n_points:]
            
        # Linear approximation:
        # ∫ h(x)(1 - h(x+k)) dx ≈ ∑ h[i] * (1 - h[i+k])
        linear_obj = 1 - best_h
        
        # Find extreme point by selecting top indices
        # This mimics solving the linear subproblem over binary points
        top_indices = np.argsort(linear_obj)[::-1][:n_points//2]
        h_extreme = np.zeros(n_points)
        h_extreme[top_indices] = 1.0
        
        # Update with Frank-Wolfe step
        new_h = (1 - eta) * best_h + eta * h_extreme
        
        # Project to feasible region
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(new_h, dx)
        
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            # Adaptive step size
            eta = min(0.95 * eta, 0.9)
        else:
            # Reset step size if no progress
            eta = 0.5
            # Perturb occasionally to escape local optima
            if np.random.random() < perturb_rate:
                perturb_indices = np.random.choice(n_points, size=n_points//4, replace=False)
                best_h[perturb_indices] = np.random.rand(n_points//4)
                best_h = project_to_feasible(best_h, n_points)
                best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final refinement 
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
