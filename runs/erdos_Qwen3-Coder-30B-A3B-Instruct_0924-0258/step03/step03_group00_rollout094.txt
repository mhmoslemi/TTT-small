import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based parametrization
    # Use 5 cosine terms for smoothness and structure
    d = 5
    
    def h_from_coeffs(a):
        # a: [a0, a1, ..., ad]
        x = np.linspace(0, 2, n_points)
        g = a[0] + np.sum([a[k+1] * np.cos((k+1) * np.pi * x) for k in range(d)], axis=0)
        h_val = 1.0 / (1.0 + np.exp(-g))
        # Ensure sum constraint
        h_val = h_val * (n_points / 2) / np.sum(h_val)
        return np.clip(h_val, 0, 1)
    
    def obj_func(a):
        if time.time() - start_time > budget_s * 0.95:
            return np.inf
        h_val = h_from_coeffs(a)
        return evaluate_erdos_solution(h_val, dx)
    
    # Constraints for valid h(x)
    def sum_constraint(a):
        h_val = h_from_coeffs(a)
        return np.sum(h_val) - n_points / 2
    
    # Initial point
    a_start = np.zeros(d + 1)
    
    # Random restarts
    num_restarts = min(20, int(budget_s / 30))
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random start
        a0 = a_start + np.random.normal(0, 0.5, d + 1)
        
        # Optimization bounds
        bounds = [(-10, 10) for _ in range(d + 1)]
        constraints = {'type': 'eq', 'fun': sum_constraint}
        
        try:
            res = minimize(obj_func, a0, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 100})
            if res.success:
                h_candidate = h_from_coeffs(res.x)
                c5_test = evaluate_erdos_solution(h_candidate, dx)
                if c5_test < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_test
        except:
            continue
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
