import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try refined approaches
    # First, try a different starting strategy
    if 'initial_h_values' not in globals():
        # Construct better initial guess using a two-peak structure
        h_start = np.zeros(n_points)
        peak1 = n_points // 3
        peak2 = 2 * n_points // 3
        width = n_points // 6
        for i in range(n_points):
            if abs(i - peak1) < width:
                h_start[i] += 0.4 * (1 - abs(i - peak1) / width)
            if abs(i - peak2) < width:
                h_start[i] += 0.4 * (1 - abs(i - peak2) / width)
        h_start = h_start * (n_points / 2) / np.sum(h_start)

    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def objective_with_penalty(h_vec, penalty=1e6):
        # Add penalty for constraint violations
        c5_val = evaluate_erdos_solution(h_vec, dx)
        penalty_term = 0
        if np.sum(h_vec) < n_points/2 - 1e-6:
            penalty_term += penalty * (n_points/2 - np.sum(h_vec))
        if np.sum(h_vec) > n_points/2 + 1e-6:
            penalty_term += penalty * (np.sum(h_vec) - n_points/2)

        h_trial = h_start + np.random.normal(0, 0.01, n_points)

                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)

        h_new = np.clip(h_new, 0, 1)

                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
            else:
                break
        except:
            break
    
    return (list(best_h), best_c5, n_points)
