import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define Haar wavelet basis functions
    def haar_basis(x, n_functions):
        basis = np.zeros((len(x), n_functions))
        for i in range(n_functions):
            if i == 0:
                basis[:, i] = np.ones(len(x))
            else:
                # Wavelet basis functions
                j = i - 1
                k = 0
                while 2**k <= j:
                    k += 1
                if k == 0:
                    continue
                else:
                    l = j - 2**(k-1)
                    # Haar wavelet function: scaled and shifted
                    p = 2**(k-1)
                    q = l * 2**(k-1)
                    basis[:, i] = np.where(
                        (x >= q/2**k) & (x < (q+p)/2**k), 
                        1, 
                        np.where(
                            (x >= (q+p)/2**k) & (x < (q+2*p)/2**k), 
                            -1, 
                            0
                        )
                    )
        return basis
    
    # Optimization approach:
    # 1. Regular grid-based optimization
    # 2. Apply local refinement to find better solutions
    
    # Function to compute objective
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Constraints for optimization
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Main optimization loop
    max_iterations = 500
    refinement_steps = 100
    
    for iter in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local optimization using SLSQP
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 10, 'ftol': 1e-7})
            
            if res.success:
                h_new = np.clip(res.x, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    
        except:
            pass
        
        # Every few iterations, slightly perturb the solution to escape local minima
        if iter % 10 == 0 and iter > 0:
            # Add small random perturbation
            eps = 0.01
            perturbation = np.random.uniform(-eps, eps, n_points)
            perturbed = best_h + perturbation
            perturbed = np.clip(perturbed, 0, 1)
            # Renormalize
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            c5_perturbed = evaluate_erdos_solution(perturbed, dx)
            
            if c5_perturbed < best_c5:
                best_h = perturbed
                best_c5 = c5_perturbed
    
    # Final refinement
    for _ in range(refinement_steps):
        if time() - start_time > budget_s * 0.98:
            break
            
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': integral_constraint}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5, 'ftol': 1e-8})
            
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
