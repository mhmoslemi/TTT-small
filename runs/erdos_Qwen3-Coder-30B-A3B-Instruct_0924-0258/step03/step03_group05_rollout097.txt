import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with projected updates
    max_iter = 10000
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    # Initial setup
    h_current = best_h.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Learning rate schedule
    base_lr = 0.01
    decay_factor = 0.995
    
    # Optimization loop
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        if no_improve_count >= max_no_improve:
            # Perturb solution to escape local minimum
            perturb_strength = 0.05
            h_current += np.random.normal(0, perturb_strength, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            no_improve_count = 0
        
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Sample a few directions to approximate gradient
        grad_est = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            # Finite difference estimation
            h_plus = h_current.copy()
            h_plus[i] = min(1.0, h_current[i] + eps)
            h_minus = h_current.copy()
            h_minus[i] = max(0.0, h_current[i] - eps)
            
            c_plus = evaluate_erdos_solution(h_plus, dx)
            c_minus = evaluate_erdos_solution(h_minus, dx)
            grad_est[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient descent
        lr = base_lr * (decay_factor ** iter_count)
        h_new = h_current - lr * grad_est
        
        # Project onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            h_current = h_new
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        iter_count += 1
        
        # Occasionally recompute C5 to ensure no drift
        if iter_count % 100 == 0:
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
    
    # Final refinement with local optimizer
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
