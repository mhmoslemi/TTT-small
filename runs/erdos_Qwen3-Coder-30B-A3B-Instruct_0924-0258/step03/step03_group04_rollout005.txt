import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with a periodic pattern for exploration
    h_current = h_start.copy()
    
    # Parameters for gradient descent
    learning_rate = 1e-3
    decay_rate = 0.999
    perturb_prob = 0.1
    perturb_magnitude = 0.05
    max_iter = 100000
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    for iteration in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        iter_count += 1
        
        # Compute all correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Identify shift k that gives maximum overlap
        k = max_k - (n_points - 1)
        
        # Compute subgradient: d/dh[i] of integral
        # We need the derivative of the maximum term
        subgradient = np.zeros(n_points)
        if k >= 0:
            # We're considering h[i] * (1 - h[i+k]) for valid i
            valid_indices = np.arange(n_points - k)
            # For the k-th shift, the gradient is (1 - h[i+k]) for i in valid_indices
            for i in valid_indices:
                j = i + k
                if j < n_points:
                    subgradient[i] += (1 - h_current[j]) * dx
        else:
            # Consider negative shift
            valid_indices = np.arange(-k, n_points)
            k_abs = -k
            for i in valid_indices:
                j = i - k_abs
                if j >= 0:
                    subgradient[i] += (1 - h_current[j]) * dx
                    
        # Perform gradient step
        h_new = h_current - learning_rate * subgradient
            
        # Apply random perturbation occasionally
        if np.random.random() < perturb_prob:
            perturbation = np.random.normal(0, perturb_magnitude, n_points)
            h_new += perturbation
            h_new = np.clip(h_new, 0, 1)
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Check new objective
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if we improved
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current approximation
        h_current = h_new.copy()
        
        # Update learning rate
        learning_rate *= decay_rate
        
        # Early stopping if no improvement
        if no_improvement_count >= max_no_improvement:
            break
            
    # Final refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
