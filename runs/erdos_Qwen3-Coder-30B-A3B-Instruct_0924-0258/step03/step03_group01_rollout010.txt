import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create structured initial guess
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    max_iter = 500
    temperature = 0.1
    cooling_rate = 0.99
    
    # Population-based refinement
    pop_size = 20
    population = []
    
    # Generate initial population
    for i in range(pop_size):
        h = h_start + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    def refine_with_sqp(h):
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_vec, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                return h_refined
        except:
            pass
        return h_vec
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        for i, h in enumerate(population):
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Refine each population member
        for i in range(pop_size):
            population[i] = refine_with_sqp(population[i])
        
        # Generate new candidates via perturbation
        new_pop = []
        for h in population:
            # Small random perturbation
            h_perturbed = h + np.random.normal(0, 0.005, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            new_pop.append(h_perturbed)
        
        # Accept new candidates with simulated annealing probability
        for i in range(pop_size):
            h_current = population[i]
            h_new = new_pop[i]
            
            c5_current = evaluate_erdos_solution(h_current, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_current:
                population[i] = h_new
            else:
                # Accept with probability based on temperature
                delta = c5_new - c5_current
                if random.random() < np.exp(-delta / temperature):
                    population[i] = h_new
        
        # Cool down temperature
        temperature *= cooling_rate
        
        # Occasionally, add new good candidates
        if iteration % 20 == 0:
            for _ in range(5):
                h = best_h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
                population.pop(0)  # Remove worst
    
    # Final refinement of best solution
    final_h = best_h.copy()
    final_h = refine_with_sqp(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
