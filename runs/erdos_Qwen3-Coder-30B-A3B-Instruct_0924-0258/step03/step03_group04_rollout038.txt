import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Mixed optimization approach: PSO + gradient descent refinement
    max_pso_iter = 500
    max_grad_iter = 50
    
    # PSO parameters
    swarm_size = 50
    omega = 0.7  # inertia weight
    phi_p = 1.5  # cognitive parameter
    phi_g = 1.5  # social parameter
    
    # Initialize swarm
    particles = []
    velocities = []
    for _ in range(swarm_size):
        h = project_to_feasible(np.random.rand(n_points) * 0.5 + 0.25, n_points)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
    
    # Track best particle
    best_particle = particles[0].copy()
    best_particle_c5 = evaluate_erdos_solution(best_particle, dx)
    
    # PSO iterations
    for iter_count in range(max_pso_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(swarm_size):
            h = particles[i]
            
            # Update particle velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (omega * velocities[i] + 
                           phi_p * r1 * (best_particle - h) + 
                           phi_g * r2 * (best_particle - h))
            particles[i] = h + velocities[i]
            
            # Maintain constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            c5 = evaluate_erdos_solution(particles[i], dx)
            if c5 < best_particle_c5:
                best_particle = particles[i].copy()
                best_particle_c5 = c5
                
        # Update global best
        if best_particle_c5 < best_c5:
            best_h = best_particle.copy()
            best_c5 = best_particle_c5
    
    # Gradient descent refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
