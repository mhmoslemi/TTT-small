import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a symmetric initialization with some structure
        h_start = np.ones(n_points)
        # Add some structure to avoid local minima
        mid = n_points // 2
        h_start[:mid] = 0.7
        h_start[mid:] = 0.3
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_n_points = n_points
    
    # CVXPY-based optimization approach for better results
    # First, try a convex relaxation if number of points is small
    if n_points <= 200:
        try:
            # Formulate with CVXPY
            h_var = cp.Variable(n_points, nonneg=True)
            
            # Constraints
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var <= 1
            ]
            
            # Objective: minimize max overlap (approximated using a larger problem)
            # This part won't directly solve C5, so we'll locally refine
            # Use only for a better initialization, then refine with gradient
            prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_init = np.array(h_var.value)
                h_init = project_to_feasible(h_init, n_points)
                c5_init = evaluate_erdos_solution(h_init, dx)
                if c5_init < best_c5:
                    best_c5 = c5_init
                    best_h = h_init.copy()
        except:
            pass

    # Try accelerated gradient descent from the best initialization
    if time() - start_time > budget_s * 0.9:
        return (list(best_h), best_c5, best_n_points)

    h_current = best_h.copy()
    dx = 2.0 / n_points
    
    # Use a simple gradient-based refinement
    max_iter = 5000
    step_size = 0.0001
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation
        grad = np.zeros_like(h_current)
        eps = 1e-6
        base_score = evaluate_erdos_solution(h_current, dx)
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[j] = (score_plus - base_score) / eps
        
        # Update with gradient descent
        h_current = h_current - step_size * grad
        h_current = project_to_feasible(h_current, n_points)
        
        # Evaluate and update best
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()

    # Final check
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, best_n_points)
