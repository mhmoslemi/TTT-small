import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a symmetric pattern inspired by Fourier modes
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(np.pi * x)
        h_start = project_to_feasible(h_start, n_points)

    # Optimization parameters
    max_iter = 200
    step_size = 0.01
    tol = 1e-6
    best_c5 = float('inf')
    best_h = h_start.copy()
    best_n_points = n_points
    dx = 2.0 / n_points

    # Gradient descent loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            best_n_points = n_points

        # Compute gradient via finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-5
        base_score = evaluate_erdos_solution(best_h, dx)

        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps

        # Update with gradient
        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)
        
        # Check for improvement
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            best_n_points = n_points

        # Adaptive step size
        if iteration > 0 and abs(new_c5 - current_c5) < tol:
            step_size *= 0.9
        elif abs(new_c5 - current_c5) > 0.01:
            step_size = min(step_size * 1.1, 0.1)

    # Final refinement with slightly different initialization
    for attempt in range(3):
        if time() - start_time > budget_s * 0.95:
            break
        h_try = h_start + np.random.normal(0, 0.05, n_points)
        h_try = project_to_feasible(h_try, n_points)
        c5_try = evaluate_erdos_solution(h_try, dx)
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try

    # Final validation
    best_h = project_to_feasible(best_h, best_n_points)
    final_c5 = evaluate_erdos_solution(best_h, 2.0 / best_n_points)

    return (list(best_h), final_c5, best_n_points)
