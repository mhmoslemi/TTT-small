import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting points to escape local optima
    start_points = [h_start]
    # Add a few random perturbations
    for _ in range(3):
        pert = np.random.normal(0, 0.05, n_points)
        h_pert = h_start + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        start_points.append(h_pert)
    
    # Try all start points
    for h_init in start_points:
        h_current = h_init.copy()
        c_current = evaluate_erdos_solution(h_current, dx)
        if c_current < best_c5:
            best_h = h_current
            best_c5 = c_current
    
    # Local optimization with SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run multiple local optimizations from different seeds
    for i in range(5):
        if time() - start_time > budget_s * 0.9:
            break
            
        np.random.seed(seed + i)
        h_seed = best_h + np.random.normal(0, 0.01, n_points)
        h_seed = np.clip(h_seed, 0, 1)
        h_seed = h_seed * (n_points / 2) / np.sum(h_seed)
        
        try:
            res = minimize(objective, h_seed, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Try simplex method with lower resolution if time allows
    if n_points == 35 and time() - start_time < budget_s * 0.7:
        n_points = 70
        dx = 2.0 / n_points
        h_coarse = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 35), best_h)
        h_coarse = h_coarse * (n_points / 2) / np.sum(h_coarse)
        
        # Refine on coarse grid
        try:
            bounds_coarse = [(0, 1) for _ in range(n_points)]
            res_coarse = minimize(objective, h_coarse, method='SLSQP',
                                  bounds=bounds_coarse, constraints=cons,
                                  options={'maxiter': 100, 'ftol': 1e-8})
            if res_coarse.success:
                h_fine = np.clip(res_coarse.x, 0, 1)
                h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
                c5_fine = evaluate_erdos_solution(h_fine, dx)
                if c5_fine < best_c5:
                    best_h = h_fine
                    best_c5 = c5_fine
        except:
            pass
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Post-process to ensure constraints
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum

    # Final local refinement
    try:
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
