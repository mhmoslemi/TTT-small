import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create sinusoidal initialization 
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        # Project to feasible region
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.99
    min_temperature = 1e-6
    acceptance_threshold = 0.3
    iterations = 0
    accepted = 0
    total_iterations = 0
    
    # SA main loop
    while time() - start_time < budget_s * 0.95:
        total_iterations += 1
        if total_iterations % 100 == 0:
            temperature *= cooling_rate
            if temperature < min_temperature:
                temperature = min_temperature
            
        # Get current C5 value
        current_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Generate candidate solution
        h_candidate = best_h.copy()
        
        # Select random subset of indices to modify
        num_changes = max(1, int(n_points * 0.05))  # ~5% change
        change_indices = np.random.choice(n_points, num_changes, replace=False)
        
        # Modify selected indices
        for i in change_indices:
            # Random perturbation in [0, 0.2]
            perturbation = np.random.uniform(-0.2, 0.2)
            h_candidate[i] += perturbation
            
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept criteria
        delta = candidate_c5 - current_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / (temperature + 1e-10)):
            best_h = h_candidate.copy()
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
            accepted += 1
            
        # Adaptive temperature update
        if total_iterations % 100 == 0:
            acceptance_rate = accepted / 100.0
            if acceptance_rate < acceptance_threshold:
                temperature *= 1.1  # Increase temp if too low
            else:
                temperature *= 0.9  # Decrease temp if too high
            accepted = 0
        
        # Occasionally refine with local optimization
        if total_iterations % 1000 == 0:
            def refinement_obj(h):
                h_proj = project_to_feasible(h, n_points)
                return evaluate_erdos_solution(h_proj, dx)
            
            res = minimize(refinement_obj, best_h, method='L-BFGS-B',
                                bounds=[(0,1) for _ in range(n_points)],
                                options={'maxiter': 10})
            if res.success:
                h_local = project_to_feasible(res.x, n_points)
                c5_local = evaluate_erdos_solution(h_local, dx)
                if c5_local < best_c5:
                    best_h = h_local.copy()
                    best_c5 = c5_local

    # Final refinement for remaining time
    if time() - start_time < budget_s * 0.05:
        try:
            def final_obj(h):
                return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
            
            res = minimize(final_obj, best_h, method='L-BFGS-B',
                           bounds=[(0, 1) for _ in range(n_points)],
                           options={'maxiter': 100})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
