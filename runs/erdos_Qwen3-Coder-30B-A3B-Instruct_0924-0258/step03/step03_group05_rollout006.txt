import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    num_generations = 200
    mutation_rate = 0.2
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to sum = n_points/2
        h = np.clip(h, 0, 1)
        return h
    
    def create_population():
        population = []
        if 'initial_h_values' in globals():
            population.append(h_start.copy())
        else:
            population.append(create_individual())
        for _ in range(population_size - len(population)):
            population.append(create_individual())
        return population
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitnesses)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        return selected
    
    def crossover(parent1, parent2):
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += np.random.normal(0, 0.1)
        # Renormalize
        individual = individual / np.sum(individual) * (n_points / 2)
        individual = np.clip(individual, 0, 1)
        return individual
    
    def evolve(population):
        fitnesses = [fitness(ind) for ind in population]
        # Sort by fitness
        sorted_pairs = sorted(zip(population, fitnesses), key=lambda x: x[1])
        elite = [pair[0] for pair in sorted_pairs[:elite_size]]
        
        # Selection
        selected = selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = elite[:]
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        return new_population[:population_size]
    
    population = create_population()
    
    # Main GA loop
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while generation < num_generations and time() - start_time < budget_s * 0.95:
        population = evolve(population)
        fitnesses = [fitness(ind) for ind in population]
        best_in_gen = min(fitnesses)
        
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[fitnesses.index(best_in_gen)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        generation += 1
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final local optimization refinement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
