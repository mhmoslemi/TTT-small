import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_objective(h, beta):
    """Smooth approximation of the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    exp_vals = np.exp(beta * corr)
    return (1.0 / beta) * np.log(exp_vals.sum())

def compute_gradient(h, beta, eps=1e-5):
    """Numerical gradient of the smooth objective"""
    grad = np.zeros_like(h)
    base_val = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_val) / eps
    return grad

    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    best_h = h.copy()
    best_c5 = smooth_objective(h, beta)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        grad = compute_gradient(h, beta)
        
        # Line search step
        step = h - step_size * grad
        step = project_to_feasible(step, n_points)
        new_c5 = smooth_objective(step, beta)

            best_c5 = new_c5

        # Update smoothing parameter
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.5
            
        # Early stopping if improvement is small
        if abs(best_c5 - smooth_objective(h, beta)) < tolerance:
            break

    # Final evaluation with exact computation
    final_c5 = evaluate_erdos_solution(h, dx)
    if final_c5 < best_c5:
