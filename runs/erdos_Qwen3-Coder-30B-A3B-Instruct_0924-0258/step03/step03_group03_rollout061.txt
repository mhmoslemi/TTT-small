import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_maximum(corr, beta):
    """Smooth approximation to maximum"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def gradient_smooth_objective(h, beta, eps=1e-5):
    """Compute gradient via finite differences of smooth objective"""
    n = len(h)
    corr_base = np.correlate(h, 1 - h, mode='full')
    base_value = smooth_maximum(corr_base, beta)
    
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        plus_value = smooth_maximum(corr_plus, beta)
        grad[i] = (plus_value - base_value) / eps
    return grad

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    best_n_points = n_points
    
    # Optimization parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iterations = 1000
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute gradient
        grad = gradient_smooth_objective(h, beta)
        
        # Perform update with line search
        step = h - step_size * grad
        step = project_to_feasible(step, n_points)
        
        # Evaluate new objective
        new_c5 = evaluate_erdos_solution(step, dx)
        
        if new_c5 < best_c5:
            h = step.copy()
            best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Update smoothing parameter
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.5
            
        # Check for stagnation
        if abs(current_c5 - new_c5) < tolerance:
            break
            
        current_c5 = new_c5
        
        # Occasionally increase resolution when improvement plateaus
        if iteration > 500 and n_points < 100:
            n_points_try = min(n_points * 2, 100)
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h)), 
                                h)
            h = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
            dx = 2.0 / n_points

            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:

                best_h = h.copy()
