import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segment_gradient(h, boundaries, dx):
    """Compute gradient of overlap with respect to segment heights"""
    m = len(boundaries) - 1
    h_segmented = np.zeros(m)
    for i in range(m):
        h_segmented[i] = np.mean(h[int(boundaries[i]/dx):int(boundaries[i+1]/dx)])
    return h_segmented

def segment_to_function(segments, boundaries, dx):
    """Convert segment representation to function values"""
    n_points = int(2/dx)
    h = np.zeros(n_points)
    for i in range(len(boundaries)-1):
        start_idx = int(boundaries[i]/dx)
        end_idx = int(boundaries[i+1]/dx)
        h[start_idx:end_idx] = segments[i]
    return h

def compute_overlap_approximation(segments, boundaries, dx, beta=100):
    """Smooth approximation of maximum overlap"""
    # Convert segments to function
    h_func = segment_to_function(segments, boundaries, dx)
    # Compute correlation
    corr = np.correlate(h_func, 1 - h_func, mode='full')
    # Smooth max using log-sum-exp
    corr_max = np.max(corr)
    corr_exp = np.exp(beta * (corr - corr_max)) 
    smooth_max = corr_max + np.log(np.sum(corr_exp)) / beta
    return smooth_max

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    max_segments = 40
    initial_segments = 15
    max_iter_height = 100
    max_iter_width = 50
    useRefinement = 25
    final_iter = 100
    step_size = 0.01
    backtrack_factor = 0.5
    max_backtrack = 20
    
    # Initialize segments
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
        dx = 2.0 / n_points
        # Convert to segments
        m = initial_segments
        boundaries = np.linspace(0, 2, m+1)
        segments = np.array([np.mean(h_current[int(boundaries[i]/dx):int(boundaries[i+1]/dx)]) 
                           for i in range(m)])
        # Project to feasible
        segments = np.clip(segments, 0, 1)
        segments = segments * (n_points / 2) / (np.sum(segments * np.diff(boundaries)))
        best_c5 = evaluate_erdos_solution(h_current, dx)
        best_segments = segments.copy()
        best_boundaries = boundaries.copy()
    else:
        m = initial_segments
        boundaries = np.linspace(0, 2, m+1)
        segments = np.ones(m) * 0.5
        dx = 2.0 / n_points
        # Project to feasible
        segments = np.clip(segments, 0, 1)
        segments = segments * (n_points / 2) / (np.sum(segments * np.diff(boundaries)))
        h_current = segment_to_function(segments, boundaries, dx)
        best_c5 = evaluate_erdos_solution(h_current, dx)
        best_segments = segments.copy()
        best_boundaries = boundaries.copy()
    
    # Optimization loop
    iteration = 0
    converged = False
    
    # Main loop
    while time() - start_time < budget_s * 0.95:
        # Height optimization
        for _ in range(max_iter_height):
            if time() - start_time > budget_s * 0.95:
                break
            # Compute gradient
            grad_w = np.zeros_like(segments)
            eps = 1e-5
            base_overlap = compute_overlap_approximation(segments, boundaries, dx)
            
            for i in range(len(segments)):
                segments_plus = segments.copy()
                segments_plus[i] += eps
                segments_plus = np.clip(segments_plus, 0, 1)
                segments_plus = segments_plus * (np.sum(segments * np.diff(boundaries)) / 
                                                 np.sum(segments_plus * np.diff(boundaries))) 
                try:
                    overlap_plus = compute_overlap_approximation(segments_plus, boundaries, dx)
                except:
                    overlap_plus = base_overlap
                grad_w[i] = (overlap_plus - base_overlap) / eps
            
            # Line search
            alpha = step_size
            for _ in range(max_backtrack):
                segments_new = segments - alpha * grad_w
                segments_new = np.clip(segments_new, 0, 1)
                segments_new = segments_new * (np.sum(segments * np.diff(boundaries)) / 
                                               np.sum(segments_new * np.diff(boundaries)))
                new_overlap = compute_overlap_approximation(segments_new, boundaries, dx)
                if new_overlap < base_overlap:
                    segments = segments_new
                    break
                alpha *= backtrack_factor
            
            # Re-evaluate with actual function
            h_func = segment_to_function(segments, boundaries, dx)
            current_c5 = evaluate_erdos_solution(h_func, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_segments = segments.copy()
                best_boundaries = boundaries.copy()
                
        # Width optimization  
        for _ in range(max_iter_width):
            if time() - start_time > budget_s * 0.95:
                break
            grad_x = np.zeros_like(boundaries)
            eps = 1e-5
            
            # Compute gradients for each boundary
            for i in range(1, len(boundaries) - 1):
                boundaries_plus = boundaries.copy()
                boundaries_plus[i] += eps
                boundaries_plus = np.clip(boundaries_plus, 0, 2)
                boundaries_plus = np.sort(boundaries_plus)
                
                # Maintain constraint that boundaries sum to 2
                diff = 2 - boundaries_plus[-1]
                boundaries_plus[-1] = 2
                boundaries_plus[:-1] += diff * (boundaries_plus[:-1] - boundaries_plus[0]) / (boundaries_plus[-1] - boundaries_plus[0])
                
                # Normalize
                boundaries_plus = np.clip(boundaries_plus, 0, 2)
                boundaries_plus = np.sort(boundaries_plus)
                
                try:
                    # Compute overlap approximation with new boundaries
                    new_segments = np.zeros(len(boundaries) - 1)
                    for j in range(len(boundaries) - 1):
                        if boundaries_plus[j+1] > boundaries_plus[j]:
                            new_segments[j] = np.mean(segment_to_function(segments, boundaries_plus, dx)[int(boundaries_plus[j]/dx):int(boundaries_plus[j+1]/dx)])
                    overlap_plus = compute_overlap_approximation(new_segments, boundaries_plus, dx)
                except:
                    overlap_plus = base_overlap
                    
                try:
                    overlap_current = compute_overlap_approximation(segments, boundaries, dx)
                except:
                    overlap_current = base_overlap
                
                grad_x[i] = (overlap_plus - overlap_current) / eps

            # Update boundaries with backtracking line search
            alpha = step_size
            for _ in range(max_backtrack):
                boundaries_new = boundaries - alpha * grad_x
                boundaries_new = np.clip(boundaries_new, 0, 2)
                boundaries_new = np.sort(boundaries_new) 
                if boundaries_new[0] < 0 or boundaries_new[-1] > 2:
                    break
                    
                # Now reoptimize heights with new boundaries
                try:
                    new_segments = []
                    for j in range(len(boundaries_new) - 1):
                        if boundaries_new[j+1] > boundaries_new[j]:
                            start_idx = int(boundaries_new[j]/dx)
                            end_idx = int(boundaries_new[j+1]/dx)
                            if end_idx > start_idx:
                                new_segments.append(np.mean(segment_to_function(segments, boundaries_new, dx)[start_idx:end_idx]))
                            else:
                                new_segments.append(0)
                    new_segments = np.array(new_segments)
                    
                    # Project to feasible
                    new_segments = np.clip(new_segments, 0, 1)
                    if np.sum(new_segments * np.diff(boundaries_new)) > 0:
                        new_segments = new_segments * (np.sum(segments * np.diff(boundaries)) / 
                                                       np.sum(new_segments * np.diff(boundaries_new)))
                    
                    # Check new overlap 
                    try:
                        overlap_new = compute_overlap_approximation(new_segments, boundaries_new, dx)
                        if overlap_new < base_overlap:
                            boundaries = boundaries_new
                            segments = new_segments
                            break
                    except:
                        pass
                except:
                    pass
                alpha *= backtrack_factor
                
            # Re-evaluate function
            h_func = segment_to_function(segments, boundaries, dx)
            current_c5 = evaluate_erdos_solution(h_func, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_segments = segments.copy()
                best_boundaries = boundaries.copy()

        iteration += 1
        # Refinement logic
        if iteration % useRefinement == 0 and len(boundaries) < max_segments:
            # Split longest segments
            segment_lengths = np.diff(boundaries)
            split_indices = np.argsort(segment_lengths)[-2:]  # split 2 longest
            new_boundaries = []
            for i in range(len(boundaries) - 1):
                if i in split_indices:
                    mid_point = (boundaries[i] + boundaries[i+1]) / 2
                    new_boundaries.extend([boundaries[i], mid_point, boundaries[i+1]])
                else:
                    new_boundaries.append(boundaries[i])
            new_boundaries.append(boundaries[-1])
            boundaries = np.array(new_boundaries)
            
            # Reinitialize segments
            m = len(boundaries) - 1
            segments = np.array([np.mean(segment_to_function(best_segments, best_boundaries, dx)[int(boundaries[i]/dx):int(boundaries[i+1]/dx)]) for i in range(m)])
            segments = np.clip(segments, 0, 1)
            segments = segments * (np.sum(best_segments * np.diff(best_boundaries)) / 
                                   np.sum(segments * np.diff(boundaries)))

    # Final refinement
    for _ in range(final_iter):
        if time() - start_time > budget_s * 0.95:
            break
        h_func = segment_to_function(best_segments, best_boundaries, dx)
        current_c5 = evaluate_erdos_solution(h_func, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Convert to required format
    final_h = segment_to_function(best_segments, best_boundaries, dx)
    return (list(final_h), best_c5, len(final_h))
