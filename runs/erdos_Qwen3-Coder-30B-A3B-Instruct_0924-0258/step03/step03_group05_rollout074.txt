import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple starting points to avoid local minima
    best_c5 = float('inf')
    best_h = None
    
    for attempt in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with different random patterns each time
        if attempt == 0:
            h = h_start.copy()
        else:
            # Slight random perturbation of the current best if available
            if best_h is not None:
                h = best_h + np.random.normal(0, 0.05, n_points)
            else:
                h = np.random.rand(n_points) * 0.5
                
        # Clamp to [0,1] and renormalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        
        # Local optimization with trust-constr
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h, method='trust-constr', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'gtol': 1e-6, 'xtol': 1e-6})
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:

                    best_c5 = c5_opt
        except Exception as e:

    if best_h is not None:

        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 50})
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception as e:

    # Ensure final normalization
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        h_sum = np.sum(best_h)

    if best_h is not None:

            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

                    best_c5 = c5_final
        except Exception as e:
            pass

        # Fallback to initial construction
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
