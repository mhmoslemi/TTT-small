def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segment_correlation(w, x, dx):
    """Compute correlation for piecewise constant function with given segments"""
    # Build step function from segments
    h_vals = np.zeros(len(x) - 1)
    for i in range(len(w)):
        h_vals[i] = w[i]
    
    # Compute overlap
    corr = np.correlate(h_vals, 1 - h_vals, mode='full')
    return np.max(corr) * dx
