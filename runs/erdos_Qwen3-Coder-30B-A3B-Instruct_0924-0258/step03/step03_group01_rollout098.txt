import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_start = 0.9
    w_end = 0.2
    c1, c2 = 1.5, 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(h.copy())
        personal_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            curr_c5 = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if curr_c5 < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = curr_c5
            
            # Update global best
            if curr_c5 < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = curr_c5
                
                if global_best_value < best_c5:
                    best_h = global_best.copy()
                    best_c5 = global_best_value
        
        # Local refinement every few iterations
        if iteration % 5 == 0 and iteration > 0:
            # SQP refinement of top candidates
            top_indices = np.argsort(personal_best_values)[:3]
            for idx in top_indices:
                h_refined = personal_best[idx].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    # Use SLSQP for refinement
                    res = minimize(objective, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        h_sqp = res.x
                        c5_sqp = objective(h_sqp)
                        if c5_sqp < best_c5:
                            best_h = h_sqp
                            best_c5 = c5_sqp
                except:
                    pass
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
