import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 20
    population = []
    for _ in range(population_size):
        perturbation = np.random.normal(0, 0.02, n_points)
        h_candidate = h_start + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Simulated Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    
    # Main optimization loop
    iteration = 0
    temp = initial_temp
    max_iterations = 500
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        if iteration % 20 == 0:
            temp = initial_temp * (cooling_rate ** (iteration // 20))
            if temp < min_temp:
                temp = min_temp
            
        # Evaluate current population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find current best
        current_best_idx = np.argmin(fitness)
        current_best_h = population[current_best_idx]
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_h = current_best_h.copy()
            best_c5 = current_best_c5
        
        # Create new population
        new_population = []
        for h in population:
            # Apply perturbation
            idx = np.random.randint(0, n_points)
            perturbation = np.random.normal(0, 0.01 * temp, n_points)
            h_new = h + perturbation
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # SQP refinement
            def objective(h_refine):
                return evaluate_erdos_solution(h_refine, dx)
            
            def constraint_sum(h_refine):
                return np.sum(h_refine) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_new, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    new_population.append(h_refined)
                else:
                    new_population.append(h_new)
            except:
                new_population.append(h_new)
        
        # Accept/reject based on Metropolis criterion
        population = new_population
        iteration += 1
    
    # Final refinement on best solution using gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h_refine):
            return evaluate_erdos_solution(h_refine, dx)
        
        def constraint_sum(h_refine):
            return np.sum(h_refine) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Refine using multiple SLSQP restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
