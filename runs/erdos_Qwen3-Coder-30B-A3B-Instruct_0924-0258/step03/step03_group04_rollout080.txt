import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic pattern as starting point
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with perturbations
    max_iter = 100000
    lr = 1e-3
    p_perturb = 0.1
    perturb_mag = 0.05
    patience = 100
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)  # Convert to actual shift
        
        # Compute subgradient for current max shift
        grad = np.zeros(n_points)
        if max_shift >= 0:  # h[i] * (1 - h[i + shift])
            for i in range(n_points - max_shift):
                grad[i] += (1 - best_h[i + max_shift])
        else:  # h[i + shift] * (1 - h[i])
            for i in range(-max_shift, n_points):
                grad[i] += -(1 - best_h[i])
                
        # Update h using gradient step
        h_new = best_h - lr * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Occasionally apply random perturbation
        if np.random.random() < p_perturb:
            perturb = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h_perturbed = best_h + perturb
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                
        # Early stopping
        if no_improve_count >= patience:
            break

    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
