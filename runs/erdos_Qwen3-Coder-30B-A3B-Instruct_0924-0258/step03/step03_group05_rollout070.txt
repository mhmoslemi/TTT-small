import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization approach
    max_iter = 10000
    patience = 100
    patience_counter = 0
    best_c5_so_far = best_c5
    
    # Helper functions for gradient computation
    def compute_gradient(h, dx):
        # Compute correlations for all shifts
        h_extended = np.concatenate([h, np.zeros(n_points)])
        grad = np.zeros(n_points)
        
        for k in range(1, n_points):
            if k >= n_points:
                break
            # For shift k, we look at h[i] * (1 - h[i+k])
            # Gradient w.r.t. h[i] is (1 - h[i+k]) for i < n_points-k
            # Gradient w.r.t. h[i+k] is -h[i] for i < n_points-k
            valid_len = n_points - k
            if valid_len <= 0:
                break
            
            # Gradient contributions
            # d/dh_i of sum_{j=0}^{valid_len-1} h_j * (1 - h_{j+k})
            # For j = i, we get (1 - h_{i+k})  
            # For j = i+k, we get -h_i
            if i < n_points - k:
                grad[i] += (1 - h[i+k]) * dx
            if i+k < n_points:
                grad[i+k] -= h[i] * dx
        
        return -grad  # negative because we minimize
    
    # Actual optimization
    h_current = best_h.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute gradient with respect to current h
        try:
            grad = np.zeros_like(h_current)
            # Re-compute for each iteration
            for k in range(1, n_points):
                if k >= n_points:
                    break
                valid_len = n_points - k
                if valid_len <= 0:
                    break
                for i in range(valid_len):
                    if i+k < n_points:
                        grad[i] += (1 - h_current[i+k]) * dx
                    if i < n_points - k:
                        grad[i+k] -= h_current[i] * dx
                
            # Gradient descent step
            learning_rate = 0.01 / (1.0 + 0.001 * it)
            h_new = h_current - learning_rate * grad
            
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to satisfy integral constraint
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            if new_c5 < best_c5_so_far:
                best_c5_so_far = new_c5
                h_current = h_new.copy() 
                patience_counter = 0
            else:
                patience_counter += 1
                
            if patience_counter >= patience:
                # Perturb periodically to escape local minima
                h_current = np.clip(h_current + np.random.normal(0, 0.01, n_points), 0, 1)
                h_sum = np.sum(h_current)
                if h_sum > 0:
                    h_current = h_current * (n_points / 2) / h_sum
                patience_counter = 0
                
        except Exception as e:
            # Fallback to previous good solution
            pass
            
    # Final refinement with scipy optimization
    try:
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_current, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5_so_far:
                h_current = h_final
                best_c5_so_far = final_c5
    except:
        pass
    
    # Ensure final constraints
    h_current = np.clip(h_current, 0, 1)
    h_sum = np.sum(h_current)
    if h_sum > 0:
        h_current = h_current * (n_points / 2) / h_sum
    
    final_c5 = evaluate_erdos_solution(h_current, dx)
    
    return (list(h_current), final_c5, n_points)
