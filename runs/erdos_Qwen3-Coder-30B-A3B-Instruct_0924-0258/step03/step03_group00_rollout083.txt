import numpy as np
import time
from scipy.optimize import minimize
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    max_iter = 10000
    temp = 1.0
    cool_rate = 0.9995
    tabu_list = []
    tabu_size = 10
    
    # Generate initial configuration
    current_h = h_start.copy()
    current_c5 = best_c5
    
    def perturb_h(h, i):
        h_new = h.copy()
        # Try different values: 0, 0.5, 1
        candidates = [0.0, 0.5, 1.0]
        # Remove current value
        candidates = [v for v in candidates if abs(v - h[i]) > 1e-6]
        if candidates:
            h_new[i] = random.choice(candidates)
        else:
            # If all values same, keep as is or choose randomly
            h_new[i] = random.choice([0.0, 0.5, 1.0])
        return h_new
    
    def normalize_h(h):
        h_sum = np.sum(h)
        if h_sum > 0:
            h_norm = h * (n_points / 2) / h_sum
        else:
            h_norm = h
        return np.clip(h_norm, 0, 1)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Randomly select a point
        idx = random.randint(0, n_points - 1)
        
        # Perturb
        proposed_h = perturb_h(current_h, idx)
        proposed_h = normalize_h(proposed_h)
        proposed_c5 = evaluate_erdos_solution(proposed_h, dx)
        
        # Accept or reject
        if proposed_c5 < current_c5:
            current_h = proposed_h
            current_c5 = proposed_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Accept with probability based on temperature
            delta = proposed_c5 - current_c5
            if random.random() < np.exp(-delta / temp):
                current_h = proposed_h
                current_c5 = proposed_c5
                
        # Annealing
        temp *= cool_rate
        
        # Tabu list
        tabu_list.append(tuple(current_h.round(6)))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
    
    # Local refinement with Nelder-Mead
    if time.time() - start_time < budget_s * 0.98:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple random restarts
        for _ in range(20):
            if time.time() - start_time > budget_s * 0.98:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_restart, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    # Final validation and normalization
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
