import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe algorithm
    h_current = h_start.copy()
    max_iter = 10000
    step_size = 0.5
    perturbation_rate = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current maximum overlap
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute linear approximation direction
        # We want to minimize sum(h * (1 - h_shifted_k))
        # The gradient is approximately (1 - h_shifted_k)
        h_shifted = np.roll(h_current, k_max)
        grad = 1 - h_shifted
        
        # Linear subproblem: find binary extreme point that minimizes <grad, h>
        # Heuristic: pick top n/2 indices where grad is largest (negative)
        indices = np.argsort(-grad)
        indices = indices[:n_points//2]
        h_extreme = np.zeros(n_points)
        h_extreme[indices] = 1
        
        if np.sum(h_extreme) != n_points/2:
            # If we have too few or too many ones, adjust
            ones = np.where(h_extreme == 1)[0]
            if len(ones) > n_points//2:
                # Remove some
                remove_indices = ones[n_points//2:]
                h_extreme[remove_indices] = 0
            elif len(ones) < n_points//2:
                # Add more
                zero_indices = np.where(h_extreme == 0)[0]
                add_indices = zero_indices[:n_points//2 - len(ones)]
                h_extreme[add_indices] = 1
                
        # Frank-Wolfe update
        h_current = (1 - step_size) * h_current + step_size * h_extreme
        
        # Project to feasibility
        h_current = project_to_feasible(h_current, n_points)
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.random() < perturbation_rate:
            perturbation = np.random.normal(0, 0.01, n_points)
            h_current += perturbation
            h_current = project_to_feasible(h_current, n_points)
        
        # Evaluation
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        # Adaptive step size
        if iteration % 10 == 0 and iteration > 10:
            if 'prev_c5' in locals() and prev_c5 < current_c5:
                step_size = max(0.01, step_size * 0.9)
            elif current_c5 < prev_c5:
                step_size = min(1.0, step_size * 1.05)
            prev_c5 = current_c5
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        def obj_func(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        from scipy.optimize import minimize
        res = minimize(obj_func, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
