import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5 = current_c5
    best_h = h_start.copy()
    best_n_points = n_points
    
    # Gradient descent optimization
    max_iter = 500
    step_size = 0.01
    eps = 1e-5
    
    def compute_gradient(h, dx, eps=1e-5):
        grad = np.zeros_like(h)

            h_plus = h.copy()

    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:

        grad = compute_gradient(best_h, dx, eps)
        
        # Update with gradient ascent (minimize via negative gradient)
        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)

            best_c5 = perturbed_c5
