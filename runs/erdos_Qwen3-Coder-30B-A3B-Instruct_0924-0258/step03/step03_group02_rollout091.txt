import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple gradient-based refinement approach
    # Start with a large number of iterations for local optimization
    max_iterations = 1000
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    h_current = best_h.copy()
    
    # Define the objective function for optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint function for SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Gradient descent - simple approach with annealing
    step_size = 0.01
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current objective
        current_c5 = objective(h_current)
        
        # Compute gradient approximation (simplified finite difference)
        grad = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_to_feasible(h_minus, n_points)
            
            grad[j] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        
        # Update using gradient descent
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Accept change based on objective
        new_c5 = objective(h_new)
        if new_c5 < current_c5:
            h_current = h_new
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            # Reduce step size if no improvement
            step_size *= 0.995
            
        # Early stopping if no improvement for many steps
        if no_improvement_count > max_no_improvement:
            break
            
        iteration += 1
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            res = minimize(objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final projection and evaluation
    best_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
