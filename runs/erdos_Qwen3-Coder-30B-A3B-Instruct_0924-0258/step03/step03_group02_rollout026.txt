import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Initialize with the best known solution + noise
    for i in range(n_particles):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    last_improvement = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    last_improvement = iteration
                    
        # Early stopping if no improvement
        if iteration - last_improvement > 100:
            break
    
    # Final refinement with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and global_best_score < best_c5:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, global_best, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < global_best_score:
                    global_best = h_final
                    global_best_score = c5_final
        except Exception:
            pass
    
    return (list(global_best), global_best_score, n_points)
