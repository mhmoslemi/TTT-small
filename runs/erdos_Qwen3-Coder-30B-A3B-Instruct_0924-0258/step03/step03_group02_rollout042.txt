import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    k_values = np.arange(0, 2 * n_points - 1)
    
    # Precompute overlap vectors for each k
    overlap_vectors = []
    for k in k_values:
        if k < n_points:
            # For positive k
            vec = np.concatenate([np.zeros(k), np.ones(n_points - k)])
        else:
            # For negative k
            vec = np.concatenate([np.ones(k - n_points + 1), np.zeros(2 * n_points - k - 1)])
        overlap_vectors.append(vec)
    
    overlap_vectors = np.array(overlap_vectors)
    
    # Cluster the shifts
    n_clusters = 10
    kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
    cluster_assignments = kmeans.fit_predict(overlap_vectors)
    
    h_current = best_h.copy()
    
    # Gradient-based optimization for clusters
    max_iter = 1000
    iter_count = 0
    stagnation_count = 0
    best_c5_history = [best_c5]
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Randomly shuffle cluster order
        cluster_order = np.random.permutation(n_clusters)
        
        # Process each cluster
        for cluster_id in cluster_order:
            if (time() - start_time) > budget_s * 0.95:
                break
                
            # Get indices for current cluster
            indices = np.where(cluster_assignments == cluster_id)[0]
            if len(indices) == 0:
                continue
                
            # Compute overlap for selected indices
            def objective(h):
                total = 0
                for k_idx in indices:
                    k = k_values[k_idx]
                    if k == 0:
                        total += np.sum(h * (1 - h)) * dx
                    elif k > 0:
                        total += np.sum(h[:-k] * (1 - h[k:])) * dx
                    else:
                        k_abs = -k
                        total += np.sum(h[k_abs:] * (1 - h[:-k_abs])) * dx
                return total
            
            def constraint(h):
                return np.sum(h) - n_points / 2
            
            # Try to update only this cluster's component
            h_old = h_current.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': constraint}
            
            try:
                # Gradient descent update (simple approach)
                step_size = 0.01
                for _ in range(10):
                    h_trial = h_current + np.random.normal(0, 0.001, n_points)
                    h_trial = project_to_feasible(h_trial, n_points)
                    c5_trial = evaluate_erdos_solution(h_trial, dx)
                    if c5_trial < evaluate_erdos_solution(h_current, dx):
                        h_current = h_trial.copy()
            except:
                pass
                
        # Update the best solution seen so far
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            best_c5_history = [best_c5]
            stagnation_count = 0
        else:
            stagnation_count += 1
            if len(best_c5_history) >= 2:
                if abs(best_c5_history[-1] - best_c5_history[-2]) < 1e-8:
                    stagnation_count += 1
            best_c5_history.append(best_c5)
        
        if stagnation_count > 100:
            break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            final_h = best_h.copy()
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                final_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
