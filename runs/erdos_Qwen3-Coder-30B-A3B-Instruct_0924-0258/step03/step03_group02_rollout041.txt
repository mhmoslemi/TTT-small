import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def cluster_shifts(shifts, n_clusters=10):
    """Cluster shifts based on their overlap patterns"""
    kmeans = KMeans(n_clusters=n_clusters, random_state=42, n_init=10)
    clusters = kmeans.fit_predict(shifts)
    return clusters, kmeans

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iterations = 100
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Precompute all possible shifts
    all_shifts = np.arange(0, n_points)
    
    # Initialize clustering
    shift_vectors = []
    for k in all_shifts:
        if k == 0:
            vector = np.ones(n_points)
        else:
            vector = np.concatenate([np.ones(n_points - k), np.zeros(k)])
        shift_vectors.append(vector)
    shift_vectors = np.array(shift_vectors)
    
    n_clusters = 10
    clusters, kmeans = cluster_shifts(shift_vectors, n_clusters)
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update shifts for clustering
        shift_vectors = []
        for k in all_shifts:
            if k == 0:
                vector = np.ones(n_points)
            else:
                vector = np.concatenate([np.ones(n_points - k), np.zeros(k)])
            shift_vectors.append(vector)
        shift_vectors = np.array(shift_vectors)
        
        # Update clusters
        if iteration % 25 == 0:
            clusters, kmeans = cluster_shifts(shift_vectors, n_clusters)
        
        h_current = best_h.copy()
        improved = False
        
        # Process each cluster
        for cluster_id in range(n_clusters):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Get indices of shifts in this cluster
            cluster_indices = np.where(clusters == cluster_id)[0]
            
            if len(cluster_indices) == 0:
                continue
                
            # Define cluster-specific optimization
            def cluster_objective(h):
                if time() - start_time > budget_s * 0.95:
                    return float('inf')
                result = 0
                for k in cluster_indices:
                    if k < n_points:
                        # For positive k
                        for i in range(n_points - k):
                            result += h[i] * (1 - h[i + k])
                    else:
                        # For negative k, so k is effectively negative shift
                        k_actual = k - n_points
                        for i in range(-k_actual, n_points):
                            if i >= 0 and i + k_actual < n_points:
                                result += h[i] * (1 - h[i + k_actual])
                return -result * dx  # Negative because we maximize
            
            # Local optimization: gradient-based approach
            try:
                res = minimize(cluster_objective, h_current, method='L-BFGS-B',
                              bounds=[(0, 1) for _ in range(n_points)],
                              options={'maxiter': 100, 'ftol': 1e-4})
                if res.success:
                    h_candidate = project_to_feasible(res.x, n_points)
                    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
                        improved = True
                        no_improvement_count = 0
            except:
                pass
                
            # Convex relaxation for cluster
            try:
                h_var = cp.Variable(n_points)
                constraints = [cp.sum(h_var) == n_points / 2, h_var >= 0, h_var <= 1]
                h_var.value = h_current
                
                # Approximate convex relaxation using McCormick envelopes (simplified)
                for k in cluster_indices:
                    if k < n_points:
                        # Convex relaxation of: sum h[i] * (1 - h[i + k])
                        # This is a crude approximation of the overlap integral
                        for i in range(n_points - k):
                            if i >= 0 and i + k < n_points:
                                pass  # Linear relaxation would be very complex here
                # We make one big convex approximation to all constraints
                # Simplified: 0.5 * n_points * dx approximates the total sum constraint
                # We just ensure h is in bounds and sum equals target
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
                if prob.status == cp.OPTIMAL and h_var.value is not None:
                    h_relaxed = project_to_feasible(h_var.value, n_points)
                    c5_relaxed = evaluate_erdos_solution(h_relaxed, dx)
                    if c5_relaxed < best_c5:
                        best_h = h_relaxed.copy()
                        best_c5 = c5_relaxed
                        improved = True
                        no_improvement_count = 0
            except:
                pass
        
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final check
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
