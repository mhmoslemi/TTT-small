import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization parameters
    n_particles = 50
    inertia_start = 0.9
    inertia_end = 0.4
    c1, c2 = 1.5, 1.5
    max_iterations = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Start with the initial solution
    particles.append(h_start.copy())
    pbest_positions.append(h_start.copy())
    pbest_values.append(best_c5)
    
    # Initialize rest randomly
    for i in range(1, n_particles):
        h = np.random.rand(n_points) * 0.5 + 0.25  # Cluster around 0.5
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        pbest_positions.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
    
    # Velocities initialized to small random values
    for i in range(n_particles):
        v = np.random.rand(n_points) * 0.1 - 0.05
        velocities.append(v.copy())
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
        inertia = inertia_start - (inertia_start - inertia_end) * iteration / max_iterations
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - particles[i]) +
                            c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new position
            new_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest_values[i] = new_value
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if new_value < gbest_value:
                gbest_value = new_value
                gbest_position = particles[i].copy()
                if new_value < best_c5:
                    best_c5 = new_value
                    best_h = particles[i].copy()
        
        # Early termination on plateau
        if iteration > 100 and abs(gbest_value - pbest_values[gbest_idx]) < 1e-8:
            break
    
    # Refinement using local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            h_refined = best_h.copy()
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
