import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set up problem
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Initialize population
    population_size = 20
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Genetic algorithm main loop
    max_generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Selection (top 50%)
        selected_indices = np.argsort(fitness)[:population_size//2]
        selected = [population[i] for i in selected_indices]
        selected_fitness = [fitness[i] for i in selected_indices]
        
        # Keep best solution
        current_best_idx = np.argmin(fitness)
        current_best_h = population[current_best_idx]
        if fitness[current_best_idx] < best_c5:
            best_h = current_best_h.copy()
            best_c5 = fitness[current_best_idx]
        
        # Create new population through crossover and mutation
        new_population = [best_h.copy()]  # Elitism
        
        # Crossover
        while len(new_population) < population_size * crossover_rate:
            if len(selected) < 2:
                break
            i, j = random.sample(range(len(selected)), 2)
            parent1, parent2 = selected[i], selected[j]
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        # Mutation
        while len(new_population) < population_size:
            if len(selected) == 0:
                h = np.ones(n_points) * 0.5
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = random.choice(selected).copy()
                noise = np.random.normal(0, 0.03, n_points)
                h += noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            new_population.append(h)
        
        population = new_population
        
        # Local refinement with SQP every few generations
        if generation % sqp_interval == 0:
            refined_population = []
            for i, h in enumerate(population):
                if i < 3:  # Refine top candidates
                    def objective(h_vec):
                        return evaluate_erdos_solution(h_vec, dx)
                    def constraint_sum(h_vec):
                        return np.sum(h_vec) - n_points / 2.0
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    
                    try:
                        res = minimize(objective, h, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 100, 'ftol': 1e-6})
                        if res.success:
                            h_refined = res.x
                        else:
                            h_refined = h
                    except:
                        h_refined = h
                else:
                    h_refined = h
                refined_population.append(h_refined)
            population = refined_population
    
    # Final evaluation
    final_fitness = [evaluate_erdos_solution(h, dx) for h in population]
    final_best_idx = np.argmin(final_fitness)
    if final_fitness[final_best_idx] < best_c5:
        best_h = population[final_best_idx]
        best_c5 = final_fitness[final_best_idx]
    
    return (list(best_h), best_c5, n_points)
