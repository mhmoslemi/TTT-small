import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 50
    generations = 200
    mutation_rate = 0.2
    tournament_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Renormalize
        return np.clip(h, 0, 1)
    
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    # Genetic operations
    def tournament_selection(population, fitnesses, k=tournament_size):
        idxs = random.sample(range(len(population)), k)
        best_idx = idxs[0]
        for i in idxs[1:]:
            if fitnesses[i] < fitnesses[best_idx]:
                best_idx = i
        return population[best_idx]
    
    def crossover(parent1, parent2):
        # One-point crossover
        crossover_point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(individual, rate=mutation_rate):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < rate:
                mutated[i] += np.random.normal(0, 0.1)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # Evaluate fitness
    def evaluate_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Main GA loop
    stagnant_generations = 0
    best_fitness_history = []
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            stagnant_generations = 0
        else:
            stagnant_generations += 1
            
        if stagnant_generations > 100:
            break
            
        # Create new population
        new_population = []
        # Elitism: keep best individual
        best_idx = fitnesses.index(min(fitnesses))
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:population_size]
    
    # Fine-tune best solution
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # If we still need more time, try larger resolution
    if time() - start_time < budget_s * 0.9:
        n_points_double = n_points * 2
        dx_double = 2.0 / n_points_double
        h_double = np.interp(np.linspace(0, 2, n_points_double), 
                             np.linspace(0, 2, n_points), best_h)
        h_double = h_double * (n_points_double / 2) / np.sum(h_double)
        
        def objective_double(h_vec):
            return evaluate_erdos_solution(h_vec, dx_double)
        
        cons_double = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_double / 2}]
        bounds_double = [(0, 1) for _ in range(n_points_double)]
        
        try:
            res = minimize(objective_double, h_double, method='SLSQP',
                           bounds=bounds_double, constraints=cons_double,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points_double / 2) / np.sum(h_final)
                final_c5 = evaluate_erdos_solution(h_final, dx_double)
                if final_c5 < best_c5:
                    best_h = h_final
                    best_c5 = final_c5
                    n_points = n_points_double
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
