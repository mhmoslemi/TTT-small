import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Cluster-based optimization algorithm
    # Parameters
    n_clusters = 10
    max_iter = 100
    local_steps = 20
    convex_steps = 30
    recluster_freq = 50
    
    # Initialize
    h_current = best_h.copy()
    
    # Get all shift vectors
    shift_vectors = []
    for k in range(2 * n_points - 1):
        if k < n_points:
            # Positive shift
            v = np.zeros(n_points)
            v[:n_points - k] = 1
            v = v[:-k]  # Make it match the convolution length
        else:
            # Negative shift  
            v = np.zeros(n_points)
            v[k - n_points + 1:] = 1
            v = v[k - n_points + 1:]  # Make it match the convolution length
        # Make it length n_points - 1
        if len(v) > 0:
            shift_vectors.append(v[:n_points-1])
    
    shift_vectors = np.array(shift_vectors)
    
    # K-means clustering of shifts
    kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
    kmeans.fit(shift_vectors)
    cluster_labels = kmeans.labels_
    
    # Iterative optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        improved = False
        
        # For each cluster
        for c in range(n_clusters):
            # Get indices in cluster
            indices = np.where(cluster_labels == c)[0]
            
            if len(indices) == 0:
                continue
                
            # Create overlap constraint matrix for this cluster
            # Use a simple approach: QP for small subset
            if len(indices) <= 5:
                # Local optimization on small cluster
                # Try several methods to refine
                
                def obj_func(h):
                    # Random selection of indices to optimize
                    temp_h = h.copy()
                    overlaps = []
                    for i in indices[:5]:  # Limit to 5 for speed
                        if i < n_points:
                            # Positive shift
                            overlap = np.sum(temp_h[:n_points-i] * (1 - temp_h[i:]))
                        else:
                            # Negative shift
                            overlap = np.sum(temp_h[i-n_points+1:] * (1 - temp_h[:n_points-(i-n_points+1)]))
                        overlaps.append(overlap)
                    return -np.max(overlaps)  # Minimize negative to maximize max
                
                # Gradient-based or grid search in this local space
                try:
                    bounds = [(0, 1) for _ in range(n_points)]
                    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                    
                    res = minimize(obj_func, h_current, method='SLSQP', bounds=bounds, constraints=constraints,
                                  options={'maxiter': 10, 'ftol': 1e-6})
                    if res.success:
                        h_test = project_to_feasible(res.x, n_points)
                        c5_test = evaluate_erdos_solution(h_test, dx)
                        if c5_test < best_c5:
                            best_h = h_test.copy()
                            best_c5 = c5_test
                            h_current = h_test.copy()
                            improved = True
                except:
                    pass
    
        # Resample clusters with a more robust approach
        # Occasionally recluster based on current solution
        if iteration % recluster_freq == 0 and iteration > 0:
            try:
                # Recompute shift vectors with current h
                # Use better sampling approach instead of full convolution
                n_shifts = min(20, len(shift_vectors))
                sample_indices = np.random.choice(len(shift_vectors), n_shifts, replace=False)
                
                # Compute actual overlaps for re-clustering
                current_overlaps = []
                test_h = h_current
                for i in range(n_shifts):
                    k = sample_indices[i]
                    if k < n_points:
                        overlap = np.sum(test_h[:n_points-k] * (1 - test_h[k:]))
                    else:
                        overlap = np.sum(test_h[k-n_points+1:] * (1 - test_h[:n_points-(k-n_points+1)]))
                    current_overlaps.append(overlap)
                
                # Apply simple clustering based on overlaps
                cluster_centers = kmeans.cluster_centers_
                current_vec = np.array(current_overlaps)
                distances = [np.linalg.norm(current_vec - center) for center in cluster_centers]
                cluster_labels = np.array([np.argmin(distances)])
                
            except:
                # Fall back to fixed approach
                pass
        
        # Standard improvement with projection
        h_current = project_to_feasible(h_current, n_points)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            improved = True
            
        # Apply simple perturbation if no improvement seen recently
        if iteration > 0 and iteration % 10 == 0 and not improved:
            h_current += np.random.normal(0, 0.01, n_points)
            h_current = project_to_feasible(h_current, n_points)
            
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
