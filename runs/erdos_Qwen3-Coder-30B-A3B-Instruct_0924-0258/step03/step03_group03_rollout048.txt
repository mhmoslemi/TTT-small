import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, beta=100):
    """Smooth approximation to max function"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def compute_gradient_wrt_height(h, dx, beta=100):
    """Compute gradient with respect to h using finite differences"""
    grad = np.zeros_like(h)
    eps = 1e-5
    base_score = smooth_max(np.correlate(h, 1-h, mode='full'), beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = smooth_max(np.correlate(h_plus, 1-h_plus, mode='full'), beta)
        grad[i] = (score_plus - base_score) / eps
    return grad

def compute_gradient_wrt_boundaries(boundaries, heights, dx, beta=100):
    """Compute gradient with respect to boundaries"""
    grad = np.zeros_like(boundaries)
    eps = 1e-5
    base_score = smooth_max(np.correlate(heights, 1-heights, mode='full'), beta)
    
    for i in range(1, len(boundaries)-1):  # Skip first and last
        b_plus = boundaries.copy()
        b_plus[i] += eps
        # Reconstruct heights for new boundaries
        new_heights = reconstruct_heights(boundaries, heights, b_plus)
        score_plus = smooth_max(np.correlate(new_heights, 1-new_heights, mode='full'), beta)
        grad[i] = (score_plus - base_score) / eps
    return grad

def reconstruct_heights(old_boundaries, old_heights, new_boundaries):
    """Reconstruct height values for new boundaries"""
    new_heights = np.zeros(len(new_boundaries)-1)
    for i in range(len(new_heights)):
        for j in range(len(old_heights)):
            if old_boundaries[j] >= new_boundaries[i] and old_boundaries[j+1] <= new_boundaries[i+1]:
                new_heights[i] = old_heights[j]
                break
    return new_heights

def line_search(x, direction, func, max_iter=20, alpha=1.0, beta=0.5):
    """Backtracking line search"""
    for _ in range(max_iter):
        x_new = x + alpha * direction
        if func(x_new) < func(x):
            return alpha
        alpha *= beta
    return alpha
