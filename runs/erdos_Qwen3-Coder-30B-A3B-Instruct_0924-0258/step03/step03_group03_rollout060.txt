import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_objective(h, beta):
    """Smooth approximation to the max overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    exp_vals = np.exp(beta * corr)
    return (1 / beta) * np.log(exp_vals.sum())

def gradient(h, beta, eps=1e-5):
    """Approximate gradient via finite differences"""
    grad = np.zeros_like(h)
    base_obj = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_obj) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    # Optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute smoothed objective and gradient
        grad = gradient(h, beta)
        
        # Try a step
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = smooth_objective(h_new, beta)
        
        # Accept the step if improvement
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Update beta for sharper approximation
        if i % 100 == 0 and i > 0:
            beta *= 1.5
            
        # Check for convergence
        if abs(best_c5 - smooth_objective(h, beta)) < tolerance:
            break
    
    # Final evaluation with exact computation
    final_c5 = evaluate_erdos_solution(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (list(best_h), best_c5, n_points)
