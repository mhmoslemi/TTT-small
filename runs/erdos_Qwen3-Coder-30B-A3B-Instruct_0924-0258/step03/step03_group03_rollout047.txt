import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        
    # Initialize with current best
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_n_points = n_points

    # Use a different optimization approach: simulated annealing with local search
    # This attempts to escape local minima more effectively
    
    def neighbor(h, dx):
        """Generate a neighbor solution"""
        h_new = h.copy()
        # Perturb one element randomly
        idx = np.random.randint(0, len(h_new))
        h_new[idx] += np.random.normal(0, 0.05)
        h_new = project_to_feasible(h_new, len(h_new))
        return h_new

    def acceptance_probability(old_cost, new_cost, temperature):
        """Calculate acceptance probability"""
        if new_cost < old_cost:
            return 1.0
        return np.exp((old_cost - new_cost) / temperature)
    
    # Optimized parameters for SA
    max_iter = 5000
    initial_temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-8
    
    temperature = initial_temp
    current_h = best_h.copy()
    current_c5 = best_c5
    
    sa_iter = 0
    while sa_iter < max_iter and time() - start_time < budget_s * 0.95:
        # Generate neighbor
        new_h = neighbor(current_h, dx)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if acceptance_probability(current_c5, new_c5, temperature) > np.random.rand():
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                best_n_points = n_points
                
        # Cool down
        temperature *= cooling_rate
        sa_iter += 1
        
        # Occasionally also do a local search with gradient descent
        if sa_iter % 100 == 0 and time() - start_time < budget_s * 0.95:
            h_refined = current_h.copy()
            for _ in range(50):
                if time() - start_time > budget_s * 0.95:
                    break
                grad = np.zeros_like(h_refined)
                eps = 1e-5
                base_score = evaluate_erdos_solution(h_refined, dx)
                
                for i in range(len(h_refined)):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    score_plus = evaluate_erdos_solution(h_plus, dx)
                    grad[i] = (score_plus - base_score) / eps
                
                step_size = 0.001
                h_refined = h_refined - step_size * grad
                h_refined = project_to_feasible(h_refined, n_points)
                
                score_refined = evaluate_erdos_solution(h_refined, dx)
                if score_refined < current_c5:
                    current_h = h_refined.copy()
                    current_c5 = score_refined
                    
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
                        best_n_points = n_points
                        
    return (list(best_h), best_c5, best_n_points)
