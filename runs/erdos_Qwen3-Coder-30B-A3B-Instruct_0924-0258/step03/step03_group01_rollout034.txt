import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convex feasibility approach for initial bounds
    # Try to find a better bound via convex optimization
    def solve_convex_feasibility(target_c5):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # For all possible shifts, add constraint
        # correlation length is 2*n_points-1
        for k in range(2 * n_points - 1):
            corr_sum = cp.sum(cp.multiply(
                h[:(n_points-k)] if k < n_points else h[k-n_points+1:],
                1 - h[k:] if k < n_points else h[:n_points-(k-n_points+1)]
            ))
            # Only add constraint if shift makes sense
            if k < n_points and (n_points - k) > 0:
                constraints.append(corr_sum <= target_c5 * dx)
            elif k >= n_points and (n_points - (k - n_points + 1)) > 0:
                constraints.append(corr_sum <= target_c5 * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            return prob.status == cp.OPTIMAL, h.value
        except Exception:
            return False, None
    
    # Binary search to find tight bound
    low_bound = 0.37
    high_bound = 0.38085
    best_feasible = high_bound
    best_feasible_h = None
    
    iteration = 0
    max_iterations = 50
    
    while low_bound < high_bound and iteration < max_iterations:
        if time() - start_time > budget_s * 0.95:
            break
        mid_bound = (low_bound + high_bound) / 2
        feasible, h_sol = solve_convex_feasibility(mid_bound)
        
        if feasible:
            best_feasible = mid_bound
            best_feasible_h = h_sol
            high_bound = mid_bound
        else:
            low_bound = mid_bound
        iteration += 1
    
    if best_feasible_h is not None:
        # Perform refinement using scipy optimizer
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Refine with multiple restarts
        best_refined = best_feasible_h
        best_c5_refined = evaluate_erdos_solution(best_feasible_h, dx)
        
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slightly perturb
            h_start_refine = best_refined + np.random.normal(0, 0.01, n_points)
            h_start_refine = np.clip(h_start_refine, 0, 1)
            h_start_refine = h_start_refine * (n_points/2) / np.sum(h_start_refine)
            
            try:
                res = minimize(objective, h_start_refine, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5_refined:
                        best_c5_refined = c5_refined
                        best_refined = h_refined
            except:
                pass
                
        best_h = best_refined
        best_c5 = best_c5_refined
    
    return (list(best_h), best_c5, n_points)
