import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradients(h, dx):
    """Compute finite difference gradients of C_5 with respect to h values"""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-5
    base_score = evaluate_erdos_solution(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - base_score) / eps
    
    return grad
