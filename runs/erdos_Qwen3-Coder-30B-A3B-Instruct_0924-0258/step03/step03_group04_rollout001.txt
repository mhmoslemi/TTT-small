```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with sinusoidal pattern with periodicity
        h_start = 0.5 * np.sin(np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent refinement
    def obj_func(h):
        h_proj = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h_proj, dx)
    
    # Iterative refinement with adaptive constraint tightening
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Consider only non-zero shifts
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = corr
        # Get indices of 10% worst overlaps (by value)
        worst_idx = np.argsort(-overlaps)[:(max(1, n_points // 10))]
        # Get corresponding shifts
        worst_shifts = shifts[worst_idx]
        
        # Build constraints on worst shifts
        constraints = [
            cp.sum(cp.Variable(n_points)) == n_points / 2,
            cp.Variable(n_points) >= 0,
            cp.Variable(n_points) <= 1
        ]
        
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add only worst shifts to avoid over-constraining
        for k in worst_shifts:
            if k == 0:
                continue
            # Normalize shift index to valid correlation indices
            corr_sum = 0
            valid_len = n_points - abs(k)
            if valid_len > 0:
                if k > 0:
                    corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:valid_len]))
            constraints.append(corr_sum <= best_c5 * dx * valid_len)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feas = h_var.value
                if h_feas is not None:
                    # Use this as starting point for gradient descent
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    res = minimize(refinement_obj, h_feas, method='L-BFGS-B', 
                                 bounds=[(0,1) for _ in range(n_points)], 
                                 options={'maxiter': 50})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
        except:
            pass
        
        # Also try random perturbation to escape local optima
        if iteration % 5 == 0:
            h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        res = minimize(obj_func, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
```
