import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting points
    start_points = [h_start]
    
    # Add perturbed versions
    for i in range(5):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        start_points.append(h_pert)
    
    # Try different resolutions
    resolutions = [35]
    if budget_s > 400:
        resolutions.append(70)
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res == 70:
            n_points = 70
        else:
            n_points = 35
            
        dx = 2.0 / n_points
        
        # Try all starting points
        for h_seed in start_points:
            h_current = h_seed.copy()
            
            # Local optimization
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Try multiple local searches
            for attempt in range(3):
                try:
                    res = minimize(objective, h_current, method='SLSQP', 
                                   bounds=bounds, constraints=cons, 
                                   options={'maxiter': 50, 'ftol': 1e-7})
                    if res.success:
                        h_refined = np.clip(res.x, 0, 1)
                        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                    else:
                        h_refined = h_current
                except:
                    h_refined = h_current
                
                # Perturb for next attempt
                if attempt < 2:
                    h_current = h_refined + np.random.normal(0, 0.02, n_points)
                    h_current = np.clip(h_current, 0, 1)
                    h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            # Early termination if we're getting close to target
            if best_c5 < 0.3807:
                break
                
        # If we've used most of the budget and found a good solution, break
        if time() - start_time > budget_s * 0.9:
            break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure integral constraint
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
