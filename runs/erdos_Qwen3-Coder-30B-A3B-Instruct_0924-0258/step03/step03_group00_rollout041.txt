import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default: 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier basis approach with 5 terms
    d = 5  # Number of cosine terms
    
    def h_from_coeffs(a):
        # a: [a0, a1, ..., ad]
        x = np.linspace(0, 2, n_points)
        g = a[0] + sum(a[k+1] * np.cos((k+1) * np.pi * x) for k in range(d))
        h = 1 / (1 + np.exp(-g))
        # Normalize to unit integral
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def obj_func(a):
        if time.time() - start_time > budget_s * 0.95:
            return np.inf
        h = h_from_coeffs(a)
        return evaluate_erdos_solution(h, dx)
    
    # Optimizer bounds
    bounds = [(-10, 10)] * (d + 1)
    
    # Try multiple random restarts
    num_restarts = 10
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Initialize randomly
        a_init = np.random.uniform(-2, 2, d + 1)
        
        res = minimize(obj_func, a_init, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 100, 'ftol': 1e-6})
        
        if res.success:
            h_test = h_from_coeffs(res.x)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test

    # Final check
    return (list(best_h), best_c5, n_points)
