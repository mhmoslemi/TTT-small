import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a reasonable starting point
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization with adaptive step-size
    h_current = h_start.copy()
    max_iter = 10000
    step_size = 0.5
    patience = 100
    patience_count = 0
    last_best = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution and find the worst shift
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        best_shift_idx = np.argmax(corr_full)
        best_shift = best_shift_idx - (n_points - 1)
        
        # Compute the gradient for the worst shift
        # h[i] * (1 - h[i+best_shift]) 
        # we want to minimize sum(h[i] * (1 - h[i+best_shift]))
        grad = np.zeros(n_points)
        if best_shift > 0:
            # h[i] * (1 - h[i+best_shift]) for i in range(n_points - best_shift)
            grad[:n_points - best_shift] = 1 - h_current[best_shift:]
        elif best_shift < 0:
            # h[i] * (1 - h[i+best_shift]) for i in range(-best_shift, n_points)
            grad[-best_shift:] = 1 - h_current[:n_points + best_shift]
        else:
            # best_shift == 0, so we compute h[i] * (1 - h[i])
            grad = 1 - h_current
            
        # Solve linear subproblem over extreme points (binary vectors with n/2 ones)
        # This is an NP-hard problem; we use a heuristic greedy approach
        # Find the h_extreme that minimizes <grad, h_extreme - h_current> 
        score = grad * (1 - h_current)
        # Set scores for current h_current as very large (infinite penalty)
        score[h_current > 0.9] = -10000
        # Perturb to encourage exploration
        perturbation = np.random.random(n_points) * 0.01
        score += perturbation
        
        # Sort indices by score descending and select top n_points//2
        ordered_indices = np.argsort(score)[::-1]
        h_extreme = np.zeros(n_points)
        h_extreme[ordered_indices[:n_points//2]] = 1
        
        # Update with Frank-Wolfe step
        h_new = (1 - step_size) * h_current + step_size * h_extreme
        h_projected = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_projected, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_projected.copy()
            patience_count = 0
            # Try to decrease step-size if improvement is smooth
            step_size = max(0.1, step_size * 0.98)
        else:
            patience_count += 1
            # Increase step size if no improvement for a while
            step_size = min(1.0, step_size * 1.02)
            
        h_current = h_projected
        
        # Perturb occasionally to escape local minima
        if iteration % 200 == 0:
            perturb_indices = np.random.choice(n_points, size=n_points//5, replace=False)
            h_current[perturb_indices] = np.random.random(n_points//5) * 0.2 + 0.4
            
        # Early stopping
        if patience_count > patience:
            break
            
    # Final refinement with L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
            
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                
    return (list(best_h), best_c5, n_points)
