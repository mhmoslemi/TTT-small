import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Create initial population
    population_size = 20
    population = [h_start.copy()]
    
    # Generate diverse candidates
    for _ in range(population_size - 1):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    # Simulated Annealing with SQP refinement
    temp = 0.1
    cooling_rate = 0.99
    max_iterations = 30
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Process each candidate in population
        for i in range(len(population)):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_current = population[i].copy()
            
            # SQP refinement
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                    population[i] = h_refined
            except:
                pass
            
            # Simulated Annealing perturbation
            h_perturb = h_current + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            
            # Accept or reject based on temperature
            if c5_perturb < best_c5 or np.random.rand() < np.exp(-(c5_perturb - best_c5) / temp):
                population[i] = h_perturb
                if c5_perturb < best_c5:
                    best_h = h_perturb
                    best_c5 = c5_perturb
        
        temp *= cooling_rate
    
    return (list(best_h), best_c5, n_points)
