import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default: 35 points with uniform distribution
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based parametrization with 5 cosine terms
    def get_h_from_coeffs(coeffs):
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + coeffs[1]*np.cos(np.pi*x) + coeffs[2]*np.cos(2*np.pi*x) + \
            coeffs[3]*np.cos(3*np.pi*x) + coeffs[4]*np.cos(4*np.pi*x) + \
            coeffs[5]*np.cos(5*np.pi*x)
        h = 1 / (1 + np.exp(-g))
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def objective(coeffs):
        h = get_h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    # Optimization settings
    bounds = [(-10, 10)] * 6  # 6 coefficients for 0 to 5 cos terms
    max_restarts = 20
    restart_interval = budget_s / max_restarts
    
    for i in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initial guess for coefficients
        x0 = np.random.uniform(-2, 2, 6)
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                          options={'maxiter': 500, 'ftol': 1e-6})
            if res.success:
                c5_test = objective(res.x)
                if c5_test < best_c5:
                    best_h = get_h_from_coeffs(res.x)
                    best_c5 = c5_test
        except:
            continue
    
    # Final refinement with local optimization from best solution
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(coeffs):
            h = get_h_from_coeffs(coeffs)
            return evaluate_erdos_solution(h, dx)
        
        x0 = np.zeros(6)
        try:
            res = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                          options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = get_h_from_coeffs(res.x)
                    best_c5 = c5_test
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
