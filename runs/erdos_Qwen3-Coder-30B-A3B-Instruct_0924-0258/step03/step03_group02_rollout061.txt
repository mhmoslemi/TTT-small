import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a combination of convex relaxation and local refinement
    max_iterations = 50
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try convex relaxation approach
        try:
            # Create variable for h
            h_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Add McCormick bounds approach
            # This is a rough heuristic - just checking a few key overlaps
            # Find subsets that are more likely to be active
            for k in range(1, n_points//2 + 1):
                if time() - start_time > budget_s * 0.95:
                    break
                # For each k, create a rough estimate of the overlap constraint
                # We'll use a simplified approach: add constraints on pairwise products
                if k < n_points:
                    # Simple product bound using McCormick bounds (relaxed)
                    # For now, we use just a subset of constraints for efficiency
                    if k <= n_points//3:  # simpler constraints
