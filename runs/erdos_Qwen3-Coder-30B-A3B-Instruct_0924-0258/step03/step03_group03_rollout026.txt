import numpy as np
from cvxpy import *
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create symmetric initialization: starts with smooth variation
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 * (1 + np.sin(np.pi * x))
        h_start = project_to_feasible(h_start, n_points)
    
    best_c5 = float('inf')
    best_h = h_start.copy()
    best_n_points = n_points
    
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_start, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_start.copy()
        best_n_points = n_points

    # For optimization we'll try a hybrid approach:
    # 1. Use CVXPY for convex relaxation
    # 2. Projected gradient descent refinement
    
    def solve_convex_relaxation(h0, iterations=50):
        # Construct a feasible convex relaxation problem
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1, sum(h) == n_points/2]
        
        # We are minimizing max_k ∫ h(x)(1-h(x+k)) dx
        # For a convex approximation, we consider not just the maximum,
        # but use a smooth surrogate like the expected value with a uniform
        # distribution over feasible k values.
        # Simplification: directly optimize for a good solution
        
        # This is a high-level approach: we use the initial solution and
        # fine-tune using proximal gradient method with feasible projections
        h_sol = h0.copy()
        step_size = 1e-4
        
        for _ in range(iterations):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Approximate gradient via finite difference
            grad = np.zeros(n_points)
            eps = 1e-6
            base_score = evaluate_erdos_solution(h_sol, dx)
            for i in range(n_points):
                h_plus = h_sol.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (score_plus - base_score) / eps

            # Update with gradient step
            h_sol = h_sol - step_size * grad
            
            # Project to feasible region
            h_sol = project_to_feasible(h_sol, n_points)
            
            # Update if better
            new_score = evaluate_erdos_solution(h_sol, dx)
            if new_score < best_c5:
                best_c5 = new_score
                best_h = h_sol.copy()
                best_n_points = n_points

        return h_sol

    # Try to solve convex relaxation
    h_refined = solve_convex_relaxation(h_start.copy(), 100)
    
    # Grid refinement with more points if needed
    if best_c5 > 0.381 and time() - start_time < budget_s * 0.7:
        for n_try in range(40, 100, 5):
            if time() - start_time > budget_s * 0.95:
                break
            h_interp = np.interp(np.linspace(0, 2, n_try), 
                                np.linspace(0, 2, n_points), 
                                best_h)
            h_interp = project_to_feasible(h_interp, n_try)
            dx_try = 2.0 / n_try
            
            score_try = evaluate_erdos_solution(h_interp, dx_try)
            if score_try < best_c5:
                best_c5 = score_try
                best_h = h_interp.copy()
                best_n_points = n_try
                
    # Final refinement via gradient steps
    if time() - start_time < budget_s * 0.9:
        for _ in range(200):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros_like(best_h)
            eps = 1e-5
            base_score = evaluate_erdos_solution(best_h, dx)
            
            for i in range(len(best_h)):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (score_plus - base_score) / eps

            # Update with small step
            step_size = 0.0005
            best_h = best_h - step_size * grad
            best_h = project_to_feasible(best_h, best_n_points)

            score_refined = evaluate_erdos_solution(best_h, dx)
            if score_refined < best_c5:
                best_c5 = score_refined

    return (list(best_h), best_c5, best_n_points)
