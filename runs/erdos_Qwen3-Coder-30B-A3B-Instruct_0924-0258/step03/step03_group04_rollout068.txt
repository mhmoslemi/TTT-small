import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use sinusoidal initialization
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    iterations = 0
    accept_count = 0
    max_accept = 10000
    
    while time() - start_time < budget_s * 0.95 and iterations < max_accept:
        # Perturb the current solution
        h_candidate = best_h.copy()
        
        # Select random subset of points to modify (5-10%)
        n_modify = max(1, min(n_points // 10, 3))
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        # Perturb the selected points
        for i in indices:
            delta = np.random.normal(0, 0.1)
            h_candidate[i] += delta
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject the candidate
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            accept_count += 1
        else:
            accept_count = max(0, accept_count - 1)
        
        # Update temperature
        if iterations % 100 == 0:
            if accept_count < 10:
                T *= 0.9  # Cool faster if few accepts
            else:
                T *= cooling_rate  # Normal cooling
            T = max(T, min_T)
            
        iterations += 1
        
        # Occasionally refine with local optimization
        if iterations % 500 == 0:
            def obj(h_local):
                return evaluate_erdos_solution(project_to_feasible(h_local, n_points), dx)
            
            try:
                res = minimize(obj, best_h, method='L-BFGS-B', 
                             bounds=[(0,1) for _ in range(n_points)], 
                             options={'maxiter': 20})
                if res.success:
                    h_refined = project_to_feasible(res.x, n_points)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except:
                pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        def obj_final(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        try:
            res = minimize(obj_final, best_h, method='L-BFGS-B',
                          bounds=[(0,1) for _ in range(n_points)],
                          options={'maxiter': 50})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
