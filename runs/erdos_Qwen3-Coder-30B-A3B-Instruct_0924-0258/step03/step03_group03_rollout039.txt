import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_objective(h, beta):
    """Smooth approximation to the max overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    exp_vals = np.exp(beta * corr)
    return (1/beta) * np.log(exp_vals.sum())

def compute_gradient(h, beta, eps=1e-5):
    """Compute gradient via finite differences"""
    grad = np.zeros_like(h)
    base_val = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_val) / eps
    return grad

    start_time = time()
    
    # Initialize

    dx = 2.0 / n_points
    h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    current_c5 = evaluate_erdos_solution(h, dx)
    best_c5 = current_c5
    
    # Optimization parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-5
    max_iter = 1000
    
    # Trust-region optimization
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(h, beta)
        
        # Try step
        h_trial = h - step_size * grad
        h_trial = project_to_feasible(h_trial, n_points)
        
        # Evaluate
        trial_c5 = smooth_objective(h_trial, beta)
        
        # Accept or reject step
        if trial_c5 < current_c5:
            h = h_trial
            current_c5 = trial_c5
            if trial_c5 < best_c5:
                best_h = h.copy()
                best_c5 = trial_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
        
        # Update beta (make it tighter as we progress)
        if it % 100 == 0 and it > 0:
            beta *= 1.5
            
        # Check for convergence (stagnation)
        if abs(current_c5 - smooth_objective(h, beta)) < tolerance:
            break
    
    # Final evaluation (correct way to get C5)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Refine if needed with additional gradient steps
    for _ in range(200):
        if time() - start_time > budget_s * 0.95:
            break
        grad = compute_gradient(best_h, beta)
        best_h = best_h - 0.005 * grad
