import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    
    # Initialize swarm
    swarm = np.random.random((swarm_size, n_points)) * 0.5 + 0.25  # Between 0.25 and 0.75
    for i in range(swarm_size):
        swarm[i] = project_to_feasible(swarm[i], n_points)
    
    # Initialize personal best
    pbest = swarm.copy()
    pbest_scores = np.array([evaluate_erdos_solution(p, dx) for p in pbest])
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Velocity
    velocity = np.zeros((swarm_size, n_points))
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocity[i] = (inertia * velocity[i] + 
                          cognitive * r1 * (pbest[i] - swarm[i]) + 
                          social * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] += velocity[i]
            
            # Apply constraints
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate new position
            current_score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest[i] = swarm[i].copy()
                pbest_scores[i] = current_score
                
                # Update global best
                if current_score < gbest_score:
                    gbest = swarm[i].copy()
                    gbest_score = current_score
        
        # Update best solution found
        if gbest_score < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_score
    
    # Local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        result = minimize(final_obj, best_h, method='L-BFGS-B',
                         bounds=[(0,1) for _ in range(n_points)],
                         options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
        
        if result.success:
            h_final = project_to_feasible(result.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
