import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def normalize(h):
        h_sum = np.sum(h)
        if h_sum > 0:
            return h * (n_points / 2) / h_sum
        else:
            return h
    
    # Genetic algorithm parameters
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    elite_size = 2
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        return normalize(h)
    
    population = [create_individual() for _ in range(population_size - 1)]
    if 'initial_h_values' in globals():
        population.append(h_start)
    else:
        population.append(create_individual())
    
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        return fitness
    
    def tournament_selection(pop, fitness, k):
        selected = []
        for _ in range(len(pop)):
            tournament = random.sample(list(zip(pop, fitness)), k)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        return selected
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return normalize(child1), normalize(child2)
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.1)
        return np.clip(mutated, 0, 1)
    
    # Optimization loop
    generation = 0
    no_improvement = 0
    last_best = float('inf')
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate current population
        fitness = evaluate_population(population)
        
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitness.index(current_best)].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 100:
            break
            
        # Selection
        selected = tournament_selection(population, fitness, tournament_size)
        
        # Create new population
        new_population = []
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        generation += 1
    
    # Final refinement with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
        
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
