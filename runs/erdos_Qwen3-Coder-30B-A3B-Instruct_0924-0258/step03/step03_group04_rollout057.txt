import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic pattern: sine wave with zero mean
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with random perturbations
    max_iter = 50000
    lr = 1e-3
    decay_rate = 0.9999
    p_perturb = 0.1
    perturb_magnitude = 0.05
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute the correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute subgradient with respect to h
        subgrad = np.zeros(n_points)
        valid_len = n_points - abs(max_k - n_points)
        if max_k < n_points:  # Positive shift
            subgrad[:valid_len] = 1 - best_h[max_k:max_k+valid_len]
        else:  # Negative shift
            shift = max_k - n_points
            subgrad[shift:shift+valid_len] = 1 - best_h[:valid_len]
            
        # Update h using gradient descent
        new_h = best_h - lr * subgrad
        new_h = project_to_feasible(new_h, n_points)
        
        # Random perturbations
        if np.random.random() < p_perturb:
            noise = np.random.normal(0, perturb_magnitude, n_points)
            new_h = project_to_feasible(new_h + noise, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            # Reset learning rate on improvement
            lr = 1e-3
            
        # Decay learning rate
        lr *= decay_rate
        
        # Early stopping on no improvement
        if i > 100 and np.abs(best_c5 - evaluate_erdos_solution(best_h, dx)) < 1e-8:
            break
    
    return (list(best_h), best_c5, n_points)
