import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    # Initialize
    dx = 2.0 / n_points
    h = h_start.copy()
    h = h * (n_points / 2) / np.sum(h)  # Ensure integral constraint
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    patience = 100
    patience_counter = 0
    learning_rate = 0.01
    momentum = 0.9
    
    # Gradient approximation via finite differences
    def compute_gradient(h_vals, dx):
        n = len(h_vals)
        grad = np.zeros(n)
        c5_base = evaluate_erdos_solution(h_vals, dx)
        
        eps = 1e-6
        for i in range(n):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5_base) / eps
            
        return grad
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute approximate gradient
        try:
            grad = compute_gradient(h, dx)
        except:
            break
            
        # Update using momentum-based gradient descent
        velocity = momentum * velocity + learning_rate * grad if 'velocity' in locals() else learning_rate * grad
        h_new = h - velocity
        
        # Project into feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            patience_counter = 0
            # Reduce learning rate to avoid overshooting once we get close
            learning_rate = max(0.001, learning_rate * 0.995)
        else:
            patience_counter += 1
            if patience_counter >= patience:
                # Reset learning rate and try a random perturbation
                learning_rate = 0.01
                h_new = h + np.random.normal(0, 0.01, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
                    patience_counter = 0
        
        h = h_new.copy()
        
    # Final refinement using scipy minimize
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
