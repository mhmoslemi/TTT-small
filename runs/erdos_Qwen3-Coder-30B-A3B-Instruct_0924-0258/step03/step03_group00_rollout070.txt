import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Place a spike at center to satisfy constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = []
    max_no_improvement = 50
    no_improvement_count = 0
    
    def project_h(h):
        """Project h to satisfy integral constraint"""
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return np.clip(h, 0, 1)
    
    def get_neighbors(h, num_neighbors=10):
        neighbors = []
        for _ in range(num_neighbors):
            h_new = h.copy()
            # Perturb ~20% of indices
            indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
            for i in indices:
                delta = np.random.uniform(-0.1, 0.1)
                h_new[i] += delta
            h_new = project_h(h_new)
            neighbors.append(h_new)
        return neighbors
    
    # Main tabu search loop
    num_iterations = 0
    while time.time() - start_time < budget_s * 0.7:
        num_iterations += 1
        neighbors = get_neighbors(best_h)
        
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_neighbor in neighbors:
            # Check if neighbor is in tabu list
            neighbor_tuple = tuple(h_neighbor.round(6))
            if neighbor_tuple in tabu_list:
                continue
            
            c5 = evaluate_erdos_solution(h_neighbor, dx)
            if c5 < best_neighbor_c5:
                best_neighbor = h_neighbor
                best_neighbor_c5 = c5
        
        # Update tabu list
        if len(tabu_list) >= tabu_list_size:
            tabu_list.pop(0)
        tabu_list.append(tuple(best_h.round(6)))
        
        if best_neighbor is not None:
            no_improvement_count = 0
            best_h = best_neighbor
            if best_neighbor_c5 < best_c5:
                best_c5 = best_neighbor_c5
        else:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
    
    # Local refinement
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run local optimizers to refine solution
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
