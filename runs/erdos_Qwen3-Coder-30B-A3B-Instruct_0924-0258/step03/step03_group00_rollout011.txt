import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 500
    generations = 200
    elite_size = 20
    mutation_rate = 0.05
    crossover_rate = 0.8
    num_parents = 100
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def create_binary_individual():
        # Create binary segment function
        segments = np.random.choice([0, 1], size=n_points//2)
        if np.sum(segments) == 0:
            segments[0] = 1
        h = np.repeat(segments, 2)
        if len(h) > n_points:
            h = h[:n_points]
        elif len(h) < n_points:
            h = np.pad(h, (0, n_points - len(h)))
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                mutated[i] = np.clip(mutated[i] + np.random.normal(0, 0.05), 0, 1)
        h_sum = np.sum(mutated)
        if h_sum > 0:
            mutated = mutated * (n_points / 2) / h_sum
        return mutated
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        # Normalize
        h1_sum = np.sum(child1)
        if h1_sum > 0:
            child1 = child1 * (n_points / 2) / h1_sum
        h2_sum = np.sum(child2)
        if h2_sum > 0:
            child2 = child2 * (n_points / 2) / h2_sum
        return child1, child2
    
    # Initialize population
    population = [create_individual() for _ in range(pop_size)]
    
    # Eval and sort
    fitness = []
    for ind in population:
        f = evaluate_erdos_solution(ind, dx)
        fitness.append(f)
    
    # Sort by fitness
    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]
    fitness = [fitness[i] for i in sorted_indices]
    
    # Main genetic loop
    for g in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Selection
        parents = population[:num_parents]
        
        # Elitism
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection
            parent1 = parents[np.random.randint(0, len(parents))]
            parent2 = parents[np.random.randint(0, len(parents))]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim population size
        population = new_population[:pop_size]
        
        # Re-evaluate fitness
        fitness = []
        for ind in population:
            f = evaluate_erdos_solution(ind, dx)
            fitness.append(f)
        
        # Sort
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
    
    # Local refinement with multiple restarts
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    num_restarts = 20
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation of best solution
        h_perturb = best_h + np.random.normal(0, 0.02, n_points)
        res = minimize(obj_func, h_perturb, method='SLSQP', 
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-6})
        
        if res.success:
            c5_test = obj_func(res.x)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = res.x.copy()
    
    return (list(best_h), best_c5, n_points)
