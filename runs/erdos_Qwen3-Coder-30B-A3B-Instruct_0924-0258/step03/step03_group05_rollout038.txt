import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Direct optimization approach with local search
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimizer with adaptive sampling
    def optimize_with_adaptive_sampling():
        best_h_local = best_h.copy()
        best_c5_local = best_c5
        
        # Start with random perturbations
        for _ in range(20):
            if time() - start_time > budget_s * 0.9:
                break
                
            # Perturb solution
            h_perturbed = best_h_local + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Renormalize
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Local optimization
            cons = [{'type': 'eq', 'fun': constraint_sum}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_perturbed, method='SLSQP', 
                                bounds=bounds, constraints=cons, 
                                options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5_local:
                        best_h_local = h_refined
                        best_c5_local = c5_refined
            except:
                pass
                
            if best_c5_local < best_c5:
                best_h = best_h_local.copy()
                best_c5 = best_c5_local
                
        return best_h_local, best_c5_local
    
    # Try various methods
    h_opt, c5_opt = optimize_with_adaptive_sampling()
    
    # Try another variation: basin hopping approach
    if time() - start_time < budget_s * 0.8:
        # Try with more aggressive mutation
        h_basin = h_opt.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            # Generate a new point
            trial = np.random.normal(h_basin, 0.02)
            trial = np.clip(trial, 0, 1)
            # Renormalize
            trial = trial * (n_points / 2) / np.sum(trial)
            
            # Check improvement
            c5_trial = evaluate_erdos_solution(trial, dx)
            if c5_trial < c5_opt:
                h_basin = trial
                c5_opt = c5_trial
    
    # Final optimization
    if time() - start_time < budget_s * 0.95:
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_basin, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                final_c5 = evaluate_erdos_solution(h_final, dx)
                if final_c5 < c5_opt:
                    h_basin = h_final
                    c5_opt = final_c5
        except:
            pass
    
    return (list(h_basin), c5_opt, n_points)
