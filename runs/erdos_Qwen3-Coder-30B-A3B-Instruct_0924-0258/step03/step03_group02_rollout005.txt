import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution_with_refinement(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # DE parameters
    pop_size = 50
    F = 0.5
    CR = 0.9
    max_iters = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    last_improvement_iter = 0
    current_iter = 0
    
    while time() - start_time < budget_s * 0.85 and current_iter < max_iters:
        current_iter += 1
        
        # Evaluate current population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current population
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            last_improvement_iter = current_iter
        
        # Check for stagnation
        if current_iter - last_improvement_iter > 100:
            F = min(1.0, F * 1.1)  # Increase mutation for more exploration
            if F > 0.9:
                break  # Stop if too high
        
        # DE mutation and crossover
        new_population = []
        for i in range(pop_size):
            # Select three different individuals
            idxs = [idx for idx in range(pop_size) if idx != i]
            a, b, c = [population[j] for j in random.sample(idxs, 3)]
            
            # Mutation
            mutant = a + F * (b - c)
            mutant = project_to_feasible(mutant, n_points)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < CR, mutant, population[i])
            trial = project_to_feasible(trial, n_points)
            
            # Selection
            trial_fitness = evaluate_erdos_solution(trial, dx)
            old_fitness = fitness[i]
            
            if trial_fitness < old_fitness:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Refinement with SLSQP if time allows
    if time() - start_time < budget_s * 0.95:
        try:
            # Refine the best solution found
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return differential_evolution_with_refinement(seed, budget_s)
