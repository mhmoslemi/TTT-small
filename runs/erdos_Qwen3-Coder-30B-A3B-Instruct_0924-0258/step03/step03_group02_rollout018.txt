import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_overlap_vectors(h, n_points, dx):
    """Compute overlap vectors for all shifts"""
    overlap_vectors = []
    for k in range(2 * n_points - 1):
        if k < n_points:
            overlap = np.sum(h[:n_points-k] * (1 - h[k:])) * dx
        else:
            overlap = np.sum(h[k-n_points+1:] * (1 - h[:n_points-(k-n_points+1)])) * dx
        overlap_vectors.append(overlap)
    return np.array(overlap_vectors)

                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
