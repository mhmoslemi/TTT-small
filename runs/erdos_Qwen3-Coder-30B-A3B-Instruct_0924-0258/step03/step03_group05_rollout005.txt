import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Population size
    pop_size = 50
    max_generations = 500
    n_points = 35
    
    # Initialize population
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
        population.append(h_start)
    
    # Fill with random individuals
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Genetic algorithm loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for all individuals in population
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Select top 50% to reproduce
        elite_count = pop_size // 2
        selected = population[:elite_count]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        # Create offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = selected[random.randint(0, len(selected)-1)]
            parent2 = selected[random.randint(0, len(selected)-1)]
            
            # Crossover: single point
            crossover_point = random.randint(1, n_points-1)
            offspring = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation: small Gaussian noise
            mutation_rate = 0.1
            for i in range(n_points):
                if random.random() < mutation_rate:
                    offspring[i] += np.random.normal(0, 0.1)
            
            # Ensure feasibility
            offspring = np.clip(offspring, 0, 1)
            offspring = offspring / np.sum(offspring) * (n_points / 2)
            
            new_population.append(offspring)
        
        population = new_population
    
    # Final refinement using scipy.optimize
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    if best_h is not None:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                best_h = np.clip(res.x, 0, 1)
                best_h = best_h * (n_points / 2) / np.sum(best_h)
                final_c5 = evaluate_erdos_solution(best_h, dx)
                if final_c5 < best_c5:
                    best_c5 = final_c5
        except:
            pass
    
    # Ensure final solution meets the requirement: sum(h) * dx = 1
    if best_h is not None:
        h_sum = np.sum(best_h)
        if h_sum > 0:
            best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
