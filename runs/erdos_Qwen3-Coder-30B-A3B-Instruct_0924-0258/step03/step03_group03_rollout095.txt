import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5 = current_c5
    best_h = h_start.copy()
    best_n_points = n_points
    
    # Gradient descent optimization
    max_iter = 500
    step_size = 0.01
    tolerance = 1e-6
    
    h_current = h_start.copy()
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        base_score = evaluate_erdos_solution(h_current, dx)
        
        for i in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
        
        # Update with gradient descent
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        score_new = evaluate_erdos_solution(h_new, dx)
        if score_new < best_c5:
            best_c5 = score_new
            best_h = h_new.copy()
            best_n_points = n_points
            
        h_current = h_new
        
        # Early stopping if improvement is small
        if abs(base_score - score_new) < tolerance:
            break
    
    # Try different resolutions
    resolutions = [30, 40, 45]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        dx_try = 2.0 / res
        h_interp = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h_start)
        h_interp = project_to_feasible(h_interp, res)
        c5_try = evaluate_erdos_solution(h_interp, dx_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_interp.copy()
            best_n_points = res
            
        # Local refinement
        h_refined = h_interp.copy()
        for _ in range(50):
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx_try)
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, res)
                score_plus = evaluate_erdos_solution(h_plus, dx_try)
                grad[i] = (score_plus - base_score) / eps
            h_refined = h_refined - 0.01 * grad
            h_refined = project_to_feasible(h_refined, res)
            score_refined = evaluate_erdos_solution(h_refined, dx_try)
            if score_refined < best_c5:
                best_c5 = score_refined
                best_h = h_refined.copy()
                best_n_points = res
            if time() - start_time > budget_s * 0.95:
                break
    
    return (list(best_h), best_c5, best_n_points)
