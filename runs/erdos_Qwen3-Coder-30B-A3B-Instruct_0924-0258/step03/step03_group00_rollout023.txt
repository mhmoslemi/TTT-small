import numpy as np
from scipy.optimize import minimize
import time
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple spike to meet integral constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = deque(maxlen=tabu_list_size)
    visited = set()
    
    # Local search parameters
    local_iter = 10
    perturb_rate = 0.2
    
    iteration = 0
    no_improvement = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        if no_improvement >= max_no_improvement:
            break
            
        # Generate neighbors by perturbing 20% of indices
        neighbor_indices = np.random.choice(n_points, int(n_points * perturb_rate), replace=False)
        neighbors = []
        
        for _ in range(10):  # Generate 10 neighbors
            h_new = best_h.copy()
            for i in neighbor_indices:
                # Add small random perturbation
                delta = np.random.normal(0, 0.05)
                h_new[i] += delta
                # Clamp to [0,1]
                h_new[i] = np.clip(h_new[i], 0, 1)
            
            # Project to satisfy integral constraint
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            
            neighbors.append(h_new.copy())
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Check if in tabu
            h_tuple = tuple(np.round(h_candidate, decimals=6))
            if h_tuple in visited:
                continue
                
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = c5_candidate
        
        # Update if we found a better neighbor
        if best_neighbor is not None and best_neighbor_c5 < best_c5:
            best_h = best_neighbor.copy()
            best_c5 = best_neighbor_c5
            no_improvement = 0
            
            # Add to tabu list and visited set
            h_tuple = tuple(np.round(best_h, decimals=6))
            tabu_list.append(h_tuple)
            visited.add(h_tuple)
        else:
            no_improvement += 1
        
        # Occasionally perform local refinement
        if iteration % 5 == 0 and time.time() - start_time < budget_s * 0.95:
            def obj_func(h):
                h_clipped = np.clip(h, 0, 1)
                h_sum = np.sum(h_clipped)
                if h_sum > 0:
                    h_normalized = h_clipped * (n_points / 2) / h_sum
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Run local optimization
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                pass
    
    # Final local refinement
    if time.time() - start_time < budget_s * 0.05:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_perturb = best_h + np.random.normal(0, 0.001, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                final_c5 = obj_func(res.x)
                if final_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
