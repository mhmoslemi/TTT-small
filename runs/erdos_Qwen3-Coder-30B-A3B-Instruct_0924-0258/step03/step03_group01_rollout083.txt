import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Helper to generate neighbor solutions for local search
    def perturb_h(h, noise_level=0.05):
        h_new = h + np.random.normal(0, noise_level, len(h))
        h_new = np.clip(h_new, 0, 1)
        # Adjust to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Start with initial solution and refine using gradient-based methods
    remaining_time = budget_s - (time() - start_time)
    
    # Use local optimization with different start points
    for restart in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate alternative starting points
        h_restart = perturb_h(best_h, 0.02)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Convex relaxation refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Create a small subset of overlap constraints to maintain tractability
        # Only add constraints for small shifts
        for k in range(1, min(11, n_points)):  # small range to keep problem size manageable
            # Forward shift
            corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            constraints.append(corr_sum <= best_c5 * dx)
            
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
