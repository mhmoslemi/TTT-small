import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    max_iter = 1000
    adapt_lr = 0.01
    patience = 100
    patience_counter = 0
    
    # Alternative optimization loop with better perturbation
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute gradient approximation
        h_current = best_h.copy()
        eps = 1e-5
        grad = np.zeros(n_points)
        
        # Finite difference gradient estimation
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            
            # Small perturbations
            h_plus[j] = min(1.0, h_plus[j] + eps)
            h_minus[j] = max(0.0, h_minus[j] - eps)
            
            # Ensure integral constraint after perturbations
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c_plus = evaluate_erdos_solution(h_plus, dx)
            c_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[j] = (c_plus - c_minus) / (2 * eps)
        
        # SGD step with adaptive learning rate
        h_new = h_current - adapt_lr * grad
        
        # Projection back to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to satisfy integral constraint
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            patience_counter = 0
            adapt_lr = max(1e-5, adapt_lr * 0.995)  # Gradually reduce learning rate
        else:
            patience_counter += 1
            if patience_counter > patience:
                # Random perturbation to escape local minimum
                noise = np.random.normal(0, 0.01, n_points)
                h_perturbed = best_h + noise
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed_sum = np.sum(h_perturbed)
                if h_perturbed_sum > 0:
                    h_perturbed = h_perturbed * (n_points / 2) / h_perturbed_sum
                c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
                if c5_perturbed < best_c5:
                    best_h = h_perturbed
                    best_c5 = c5_perturbed
                patience_counter = 0
                adapt_lr = max(1e-5, adapt_lr * 1.05)  # Increase again after perturbation
    
    # Final refinement with scipy
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
