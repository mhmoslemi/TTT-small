import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initialization if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use periodic sinusoidal pattern as better initialization
        h_start = 0.5 + 0.5 * np.sin(np.pi * np.arange(n_points) / n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Main optimization loop
    max_iter = 50
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst shifts (top 10%)
        h_current = project_to_feasible(best_h, n_points)
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        # Only consider shifts from 1 to n_points-1 (non-zero)
        shift_overlaps = corr[n_points-1:2*n_points-1] * dx
        # Get indices of top 10% worst overlaps
        n_worst = max(1, int(0.1 * (n_points - 1)))
        worst_indices = np.argpartition(-shift_overlaps, range(n_worst))[:n_worst]
        worst_shifts = worst_indices + 1  # shifts from 1 to n_points-1
        
        # Convex feasibility problem using only worst shifts
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints for worst shifts only
        for k in worst_shifts:
            # Compute correlation for shift k
            corr_sum = 0
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
            constraints.append(corr_sum <= best_c5 * dx * valid_len)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_var.value
                if h_feasible is not None:
                    # Refine using gradient descent
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    res = minimize(refinement_obj, h_feasible, method='L-BFGS-B', 
                                 bounds=[(0,1) for _ in range(n_points)], 
                                 options={'maxiter': 50})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
        except:
            pass
        
        # Handle early stopping
        if iteration % 5 == 0:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time < 10:
                break
    
    # Final refinement with gradient descent on best known solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        h_final = project_to_feasible(best_h, n_points)
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, h_final, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
