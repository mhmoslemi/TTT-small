import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create an initial guess with two clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    population_size = 20
    iterations = 30
    initial_temp = 0.1
    decay_rate = 0.99
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Optimization loop
    for iter_num in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Refine each individual using SQP
        refined_population = []
        for h in population:
            if time() - start_time > budget_s * 0.95:
                break
                
            def objective(x):
                return evaluate_erdos_solution(x, dx)
            
            def constraint_sum(x):
                return np.sum(x) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h.copy(), method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                else:
                    refined_h = h
            except:
                refined_h = h
            
            # Ensure feasibility after refinement
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points/2) / np.sum(refined_h)
            refined_population.append(refined_h)
        
        population = refined_population
        
        # Simulated Annealing step
        temp = initial_temp * (decay_rate ** iter_num)
        for _ in range(population_size):
            if time() - start_time > budget_s * 0.95:
                break
                
            idx = random.randint(0, population_size - 1)
            h_old = population[idx]
            
            # Create neighbor by perturbing
            h_new = h_old + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_old = evaluate_erdos_solution(h_old, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept with simulated annealing probability
            if c5_new < c5_old or random.random() < np.exp((c5_old - c5_new) / temp):
                population[idx] = h_new
    
    # Final refinement using global method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Try multiple restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
