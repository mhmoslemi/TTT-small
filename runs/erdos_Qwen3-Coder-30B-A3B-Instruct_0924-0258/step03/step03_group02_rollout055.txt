import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution Parameters
    pop_size = 50
    F = 0.5
    CR = 0.9
    max_iter = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values) + np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_fitness = evaluate_erdos_solution(population[0], dx)
    stagnation_count = 0
    last_improvement = 0
    
    # DE loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        new_population = []
        fitness_history = []
        
        for i in range(pop_size):
            # Select three different individuals
            candidates = list(range(pop_size))
            candidates.remove(i)
            a, b, c = random.sample(candidates, 3)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if random.random() < CR or j == (i % n_points):
                    trial[j] = mutant[j]
            
            # Projection
            trial = project_to_feasible(trial, n_points)
            
            # Selection
            trial_fitness = evaluate_erdos_solution(trial, dx)
            current_fitness = evaluate_erdos_solution(population[i], dx)
            
            if trial_fitness < current_fitness:
                new_population.append(trial)
                fitness_history.append(trial_fitness)
            else:
                new_population.append(population[i])
                fitness_history.append(current_fitness)
        
        population = new_population
        current_best = min(fitness_history)
        
        if current_best < best_fitness:
            best_fitness = current_best
            # Find best individual in current population
            best_idx = np.argmin(fitness_history)
            best_h = population[best_idx].copy()
            stagnation_count = 0
            last_improvement = iteration
        else:
            stagnation_count += 1
        
        if stagnation_count > stagnation_limit:
            F = min(1.0, F * 1.1)  # Increase mutation if stagnating
            
    # Post-processing refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final validation
    best_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), float(final_c5), int(n_points))
