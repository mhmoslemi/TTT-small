import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use the coordinate descent strategy
    max_iterations = 500
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Shuffle indices for randomized coordinate descent
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        h_current = best_h.copy()
        
        # Coordinate descent on shuffled indices
        for i in indices:
            if time() - start_time > budget_s * 0.9:
                break
                
            current_h = h_current.copy()
            current_val = current_h[i]
            
            # Simple grid search for optimal value (100 steps)
            best_val = current_val
            best_overlap = best_c5
            
            # Try to improve this specific coefficient
            # Create a grid around the current value
            search_range = 0.1
            if i == 0:
                search_points = np.linspace(max(0, current_val - search_range), 
                                          min(1, current_val + search_range), 100)
            else:
                search_points = np.linspace(max(0, current_val - search_range), 
                                          min(1, current_val + search_range), 100)
            
            for candidate_val in search_points:
                current_h[i] = candidate_val
                # Project to maintain feasibility with sum constraint
                project_h = project_to_feasible(current_h, n_points)
                overlap = evaluate_erdos_solution(project_h, dx)
                
                if overlap < best_overlap:
                    best_overlap = overlap
                    best_val = candidate_val
                    improved = True
            
            # Update the current value
            current_h[i] = best_val
            h_current = project_to_feasible(current_h, n_points)
        
        # Update best solution
        if improved:
            best_h = h_current.copy()
            best_c5 = evaluate_erdos_solution(best_h, dx)
            
            # Early stopping if improvement is small
            if iteration > 10 and abs(best_c5 - evaluate_erdos_solution(h_start, dx)) < tolerance:
                break
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
