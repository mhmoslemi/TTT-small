import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(h, beta):
    """Smooth approximation of max function for better gradient computation"""
    corr = np.correlate(h, 1 - h, mode='full')
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def gradient(h, beta, eps=1e-5):
    """Compute gradient of smooth objective using finite differences"""
    grad = np.zeros_like(h)
    base_val = smooth_max(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_max(h_plus, beta) - base_val) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with initial solution if available
    n_points = 35
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        if len(h) != n_points:
            # Interpolate to desired size
            old_x = np.linspace(0, 2, len(h))
            new_x = np.linspace(0, 2, n_points)
            h = np.interp(new_x, old_x, h)
    else:
        h = np.ones(n_points) * 0.5 # fallback
    h = project_to_feasible(h, n_points)
    
    dx = 2.0 / n_points
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = gradient(h, beta)
        
        # Line search
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.2
        else:
            step_size *= 0.5
        
        # Update beta for smoother approximation
        if i % 50 == 0 and i > 0:
            beta *= 1.1
            
        # Check for convergence
        if np.abs(best_c5 - evaluate_erdos_solution(h, dx)) < tolerance:
            break
        
        # Adaptive refinement if progress plateaus
        if i > 200 and i % 100 == 0:
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5 * 0.9999:
                continue # No refinement needed
            # Try to refine resolution
            if n_points < 100:
                n_points_try = min(n_points * 2, 100)
                old_h = h.copy()
                old_x = np.linspace(0, 2, len(old_h))
                new_x = np.linspace(0, 2, n_points_try)
                h = np.interp(new_x, old_x, old_h)
                h = project_to_feasible(h, n_points_try)
                n_points = n_points_try
                dx = 2.0 / n_points
                # Reinitialize for new resolution
                step_size = 0.01
                beta = 100.0
    
    # Final refinement using standard gradient descent with small step and given tolerance
    last_c5 = best_c5 + 1.0  # initialize to enter loop
    step_size = 0.001
    for i in range(500):
        if time() - start_time > budget_s * 0.95:
            break
        grad = gradient(h, beta)
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
        else:
            step_size *= 0.8
        if abs(best_c5 - last_c5) < 1e-8:
            break
        last_c5 = best_c5

    return (list(best_h), best_c5, n_points)
