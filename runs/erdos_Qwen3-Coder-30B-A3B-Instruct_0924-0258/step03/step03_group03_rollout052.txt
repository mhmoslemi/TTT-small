import numpy as np
from scipy.signal import correlate
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    else:

        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()

    # Local optimization with gradient-based method
    max_iter = 2000
    lr = 0.001
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Approximate gradient by finite differences
        h_current = best_h.copy()
        dx = 2.0 / best_n_points
        base_score = evaluate_erdos_solution(h_current, dx)
        
        grad = np.zeros_like(h_current)

            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, best_n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
            
        # Update step
        h_new = h_current - lr * grad
        h_new = project_to_feasible(h_new, best_n_points)
        
        score_new = evaluate_erdos_solution(h_new, dx)
        if score_new < best_c5:
            best_h = h_new.copy()
            best_c5 = score_new

    # Refinement with adaptive steps and cooling
    if time() - start_time < budget_s * 0.9:
        adaptive_lr = lr
        for iter in range(500):
            if time() - start_time > budget_s * 0.95:

            # Adaptive step size based on gradient norm

            h_new = h_current - adaptive_lr * grad

            if score_new < best_c5:
                best_h = h_new.copy()
                best_c5 = score_new
                adaptive_lr = max(adaptive_lr * 0.99, 1e-6)  # Slow down as we approach solution

    return (list(best_h), best_c5, best_n_points)
