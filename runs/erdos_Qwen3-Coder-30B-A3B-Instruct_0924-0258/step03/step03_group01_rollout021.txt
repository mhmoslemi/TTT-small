import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_max
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial values
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Uniform random
            h = np.random.rand(n_points)
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gb_index = np.argmin(personal_best_values)
    global_best = personal_best[gb_index].copy()
    global_best_value = personal_best_values[gb_index]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        inertia_weight = w_min + (w_max - w_min) * (1 - iter_count / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] +
                            c1 * r1 * (personal_best[i] - swarm[i]) +
                            c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_c5 = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_c5
            
            # Update global best
            if current_c5 < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = current_c5
                
        # Refinement every 5 iterations
        if iter_count % 5 == 0 and iter_count > 0:
            # Refine top solutions with SQP
            top_indices = np.argsort(personal_best_values)[:3]
            
            for idx in top_indices:
                h_refined = personal_best[idx].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, h_refined, method='SLSQP',
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < global_best_value:
                            global_best = h_refined
                            global_best_value = c5_refined
                except:
                    pass
        
        # Update best solution found so far
        if global_best_value < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_value
    
    return (list(best_h), best_c5, n_points)
