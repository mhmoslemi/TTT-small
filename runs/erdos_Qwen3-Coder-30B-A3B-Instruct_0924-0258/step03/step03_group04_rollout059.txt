import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a more sophisticated initial pattern
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach with perturbations
    max_iter = 100000
    learning_rate = 1e-3
    p_perturb = 0.1
    perturb_magnitude = 0.05
    patience = 100
    
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Compute overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Compute subgradient for the max overlap
        subgradient = np.zeros(n_points)
        if max_k < n_points:
            # Positive shift case (h[i] * (1-h[i+k]))
            for i in range(n_points - max_k):
                subgradient[i] += (1 - best_h[i + max_k])
        else:
            # Negative shift case (h[i+k] * (1-h[i]))
            shift = max_k - n_points + 1
            for i in range(shift, n_points):
                subgradient[i] += (1 - best_h[i - shift])
        
        # Update step
        new_h = best_h - learning_rate * subgradient
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Track best solution
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Perturb with probability
        if np.random.rand() < p_perturb:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            new_h = best_h + perturb
            new_h = project_to_feasible(new_h, n_points)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
        
        # Adaptive learning rate decay
        if no_improvement_count > 50:
            learning_rate *= 0.95
        
        # Early stopping
        if no_improvement_count > patience:
            break
            
        iter_count += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
