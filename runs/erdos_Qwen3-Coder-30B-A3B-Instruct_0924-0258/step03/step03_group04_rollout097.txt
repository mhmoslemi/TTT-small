import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Smooth initialization
        h_start = 0.5 + 0.3 * np.sin(np.linspace(0, 2*np.pi, n_points))
        h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    max_iter = int(1e5)
    eta = 0.5  # Step size
    tol = 1e-8
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current maximum overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient approximation
        h_shifted = np.concatenate([best_h[-max_k:], best_h[:-max_k]] if max_k > 0 else 
                                 [best_h[max_k:], best_h[:max_k]]) if max_k != 0 else best_h.copy()
        grad = 1 - h_shifted
        
        # Solve linear subproblem by selecting top indices
        # We want to minimize sum(h * (1 - h_shifted)) 
        # Equivalent to minimizing sum(h * (-h_shifted)) 
        # So we want h where (-h_shifted) is large -> h_shifted small
        indices = np.argsort(grad)[::-1][:n_points//2]
        h_extreme = np.zeros(n_points)
        h_extreme[indices] = 1.0
        
        # Frank-Wolfe update
        new_h = (1 - eta) * best_h + eta * h_extreme
        new_h = project_to_feasible(new_h, n_points)
        
        # Perturbation to escape local optima
        if i % 50 == 0 and i > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            new_h = project_to_feasible(new_h + perturbation, n_points)
            
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            eta = min(eta * 1.02, 0.99)  # Adapt step size
        else:
            eta = max(eta * 0.95, 0.001)  # Reduce step size
            
    # Final refinement
    def final_obj(h):
        return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 500})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
