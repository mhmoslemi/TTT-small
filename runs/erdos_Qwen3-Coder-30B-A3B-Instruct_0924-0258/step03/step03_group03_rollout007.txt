import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(x, eps=1e-6):
    """Smooth approximation to max function"""
    return np.log(np.sum(np.exp(x / eps))) * eps

def trust_region_step(h_prev, grad, trust_radius, n_points, dx):
    """Perform a trust region step with box constraints"""
    # This is a simplified trust region update
    # We'll restrict to approximate step within trust radius
    step = -grad * 0.1
    h_new = h_prev + step
    
    # Project to feasible region
    h_new = project_to_feasible(h_new, n_points)
    
    # Ensure trust region constraint
    step_norm = np.linalg.norm(step)
    if step_norm > trust_radius:
        h_new = h_prev + step * trust_radius / step_norm
    
    h_new = project_to_feasible(h_new, n_points)
    return h_new
