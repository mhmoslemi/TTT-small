import numpy as np
import time
from scipy.optimize import minimize
from math import inf

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to avoid cycling
    tabu_list = []
    tabu_size = 10
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.995
    min_temperature = 1e-6
    max_iter_per_temp = 100
    
    iter_count = 0
    
    while time.time() - start_time < budget_s * 0.95 and temperature > min_temperature:
        # Simple local search with perturbations
        current_h = best_h.copy()
        
        # Randomly perturb one element
        idx = np.random.randint(0, n_points)
        old_val = current_h[idx]
        delta = np.random.uniform(-0.1, 0.1)
        current_h[idx] = np.clip(old_val + delta, 0, 1)
        
        # Renormalize
        h_sum = np.sum(current_h)
        if h_sum > 0:
            current_h = current_h * (n_points / 2) / h_sum
        
        # Check if this state is tabu
        state_tuple = tuple(np.round(current_h, decimals=6))
        if state_tuple in tabu_list:
            continue
            
        # Evaluate
        c5_new = evaluate_erdos_solution(current_h, dx)
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5:
            best_h = current_h.copy()
            best_c5 = c5_new
            # Add to tabu list
            if len(tabu_list) >= tabu_size:
                tabu_list.pop(0)
            tabu_list.append(state_tuple)
        else:
            # Accept with probability based on temperature
            delta_e = c5_new - best_c5
            if np.random.rand() < np.exp(-delta_e / temperature):
                best_h = current_h.copy()
                best_c5 = c5_new
                # Add to tabu list
                if len(tabu_list) >= tabu_size:
                    tabu_list.pop(0)
                tabu_list.append(state_tuple)
        
        # Cool down
        temperature *= cooling_rate
        iter_count += 1
        
        # Occasionally run a local optimizer
        if iter_count % 20 == 0 and time.time() - start_time < budget_s * 0.95:
            def obj_func(h):
                h_clipped = np.clip(h, 0, 1)
                h_sum = np.sum(h_clipped)
                if h_sum > 0:
                    h_normalized = h_clipped * (n_points / 2) / h_sum
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Random restart
            h_restart = np.random.rand(n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_sum = np.sum(h_restart)
            if h_sum > 0:
                h_restart = h_restart * (n_points / 2) / h_sum
                
            try:
                res = minimize(obj_func, h_restart, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                pass
    
    # Final validation and refinement
    if time.time() - start_time < budget_s * 0.05:
        # Try a hill climb from initial construction
        h_current = h_start.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Perturb small parts of the solution to see if it helps
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
                
            # Select a random segment to modify
            i = np.random.randint(0, n_points)
            # Adjust by a small value, normalize, and evaluate
            step = np.random.uniform(-0.05, 0.05)
            h_new = h_current.copy()
            h_new[i] = np.clip(h_new[i] + step, 0, 1)
            
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                
        if c5_current < best_c5:
            best_h = h_current
            best_c5 = c5_current
    
    return (list(best_h), best_c5, n_points)
