import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 100
    inertia_weight_start = 0.9
    inertia_weight_end = 0.2
    c1 = c2 = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial solution
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random uniform solution
            h = np.random.uniform(0, 1, n_points)
        
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        swarm.append(h.copy())
    
    # Track personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_values = [evaluate_erdos_solution(swarm[i], dx) for i in range(swarm_size)]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iteration
    for iter_idx in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia weight
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (iter_idx / max_iter)
        
        for i in range(swarm_size):
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocity = (inertia_weight * (swarm[i] - personal_best[i]) + 
                       c1 * r1 * (personal_best[i] - swarm[i]) + 
                       c2 * r2 * (global_best - swarm[i]))
            
            # Limit velocity
            velocity = np.clip(velocity, -velocity_max, velocity_max)
            
            # Update position
            new_h = swarm[i] + velocity
            
            # Repair feasibility
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            # Update personal best
            if new_c5 < personal_best_values[i]:
                personal_best[i] = new_h.copy()
                personal_best_values[i] = new_c5
                
                # Update global best if needed
                if new_c5 < global_best_value:
                    global_best = new_h.copy()
                    global_best_value = new_c5
            
            # Update swarm particle
            swarm[i] = new_h.copy()
        
        # Occasionally refine the global best with SQP
        if iter_idx % 5 == 0 and iter_idx > 0:
            best_candidates = [global_best]
            for i in range(swarm_size):
                if personal_best_values[i] < global_best_value * 1.05:
                    best_candidates.append(personal_best[i])
            
            # Refine top candidates
            for candidate in best_candidates:
                if time() - start_time > budget_s * 0.95:
                    break
                    
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                try:
                    res = minimize(obj, candidate, method='SLSQP',
                                   bounds=[(0, 1) for _ in range(n_points)],
                                   constraints={'type': 'eq', 'fun': constraint_sum},
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    
                    if res.success:
                        refined = res.x
                        refined_c5 = evaluate_erdos_solution(refined, dx)
                        if refined_c5 < global_best_value:
                            global_best = refined
                            global_best_value = refined_c5
                except:
                    pass
    
    # Final refinement
    def final_obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def final_constraint(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        res = minimize(final_obj, global_best, method='SLSQP',
                       bounds=[(0, 1) for _ in range(n_points)],
                       constraints={'type': 'eq', 'fun': final_constraint},
                       options={'maxiter': 50, 'ftol': 1e-8})
        
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < global_best_value:
                global_best = final_h
                global_best_value = final_c5
    except:
        pass
    
    return (list(global_best), global_best_value, n_points)
