import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a sinusoidal pattern to break symmetry
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
        # Make sure sum is exactly n_points/2
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization with adaptive step size
    h_current = h_start.copy()
    max_iter = 10000
    step_size = 0.5
    stagnation_counter = 0
    prev_best = best_c5 + 1.0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current maximum overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_shift = np.argmax(corr)
        max_overlap = corr[max_shift] * dx
        
        # Linear subproblem: find the extreme point that most decreases gradient
        # This selects a binary vector with n/2 ones that minimizes the linearized objective
        # The linearized gradient is the shift k that maximized correlation
        # So we solve a greedy selection problem
        
        # Compute linear approximation around current h for shift k
        h_shifted_k = np.concatenate([h_current[max_shift:], np.zeros(max_shift)])
        linear_grad = 1 - h_shifted_k
        
        # Select indices where linear_grad is largest (to minimize 1-h_shifted_k)
        # This is a greedy selection for the binary extreme point
        indices_sorted = np.argsort(linear_grad)[::-1]
        binary_h = np.zeros(n_points)
        ones_indices = indices_sorted[:n_points//2]
        binary_h[ones_indices] = 1.0
        
        # Update h using Frank-Wolfe rule
        h_new = (1 - step_size) * h_current + step_size * binary_h
        
        # Project back to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            stagnation_counter = 0
            # Adapt step size when improvement happens
            if step_size < 1.0:
                step_size += 0.05
        else:
            stagnation_counter += 1
            if stagnation_counter > 100:
                # Random perturbation to escape local optima
                perturbation = np.random.normal(0, 0.01, n_points)
                h_new = h_current + perturbation
                h_new = project_to_feasible(h_new, n_points)
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = new_c5
                # Reset stagnation
                stagnation_counter = 0
                
        h_current = h_new
        
        # Adaptive step size decay for stability
        if step_size > 0.05 and (new_c5 - prev_best) > 1e-7:
            step_size = max(0.05, step_size * 0.99)
        
        prev_best = new_c5

    # Final verification
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    # If final result is worse, return best found
    if final_c5 >= best_c5:
        final_h, final_c5 = best_h, best_c5
        
    return (list(final_h), final_c5, n_points)
