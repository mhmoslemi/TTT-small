import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def generate_candidate(h, n_points, perturb_frac=0.1, perturb_strength=0.1):
    """Generate a candidate solution by perturbing a subset of points"""
    h_new = h.copy()
    n_perturb = max(1, int(n_points * perturb_frac))
    indices = np.random.choice(n_points, n_perturb, replace=False)
    
    for idx in indices:
        # Random walk with bounded step
        delta = np.random.uniform(-perturb_strength, perturb_strength)
        h_new[idx] += delta
        
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a smooth sinusoidal starting point
        n_points = 35
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temp = 0.1
    temp_decay = 0.99
    min_temp = 1e-8
    acceptance_count = 0
    total_accepted = 0
    adaptive_interval = 100
    max_iter = int(1e5)
    perturb_frac = 0.1
    perturb_strength = 0.1
    
    # SA loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        h_candidate = generate_candidate(best_h, n_points, perturb_frac, perturb_strength)
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        c5_current = evaluate_erdos_solution(best_h, dx)
        
        # Compute delta
        delta = c5_candidate - c5_current
        
        # Accept or reject
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = h_candidate
            best_c5 = c5_candidate
            total_accepted += 1
            if delta < 0:
                acceptance_count = 0
            else:
                acceptance_count += 1
        else:
            acceptance_count += 1
            
        # Adaptive temperature updates
        if i % adaptive_interval == 0 and i > 0:
            if acceptance_count > 0.5 * adaptive_interval:
                temp *= temp_decay
            elif acceptance_count < 0.2 * adaptive_interval:
                temp *= 1.1
            else:
                temp *= 0.99
            acceptance_count = 0
            
            # Ensure temperature doesn't get too small
            temp = max(temp, min_temp)
        
        # Adaptive perturbation
        if total_accepted % 500 == 0 and total_accepted > 0:
            perturb_strength *= 0.95

    # Final evaluation to ensure we have the right value
    best_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), best_c5, n_points)
