import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    inertia_start = 0.9
    inertia_end = 0.2
    c1 = c2 = 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial solution
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random solution
            h = np.random.uniform(0, 1, n_points)
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_bests = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_scores = [evaluate_erdos_solution(swarm[i], dx) for i in range(swarm_size)]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_bests[i] = swarm[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = swarm[i].copy()
                global_best_score = score
        
        # Periodic local refinement
        if iteration % 5 == 0 and iteration > 0:
            for i in range(min(3, swarm_size)):
                h_refine = personal_bests[i].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_final = res.x
                        c5_final = objective(h_final)
                        if c5_final < global_best_score:
                            global_best = h_final
                            global_best_score = c5_final
                except:
                    pass
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(global_best, dx)
    
    return (list(global_best), best_c5, n_points)
