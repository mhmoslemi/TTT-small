import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use Fourier basis approach with logistic transformation
    # Parameters for Fourier representation
    n_freq = 5  # Number of cosine terms
    
    def h_from_coeffs(coeffs):
        # Generate h(x) from Fourier coefficients
        x = np.linspace(0, 2, n_points)
        # Cosine basis functions
        basis = np.array([[np.cos(k * np.pi * xi) for xi in x] for k in range(n_freq+1)])
        # Linear combination
        g = np.dot(np.array([1] + list(coeffs)), basis)
        # Logistic transform to ensure [0,1]
        h = 1.0 / (1.0 + np.exp(-g))
        # Normalize integral to 1
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def obj_func(coeffs):
        """Objective function for optimization"""
        h = h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(coeffs):
        """Constraint to enforce integral = 1"""
        h = h_from_coeffs(coeffs)
        return np.sum(h) - n_points/2
    
    # Bounds for coefficients
    bounds = [(-10, 10) for _ in range(n_freq)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Do multiple random restarts
    n_restarts = 20
    for i in range(n_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random starting coefficients
        coeffs_start = np.random.uniform(-2, 2, n_freq)
        
        try:
            result = minimize(
                obj_func, 
                coeffs_start, 
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            
            if result.success:
                h_new = h_from_coeffs(result.x)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    
        except Exception as e:
            continue
    
    # Final refinement if time allows
    if time.time() - start_time < budget_s * 0.95:
        def full_obj(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_normalized = h_clipped * (n_points / 2) / np.sum(h_clipped)
            return evaluate_erdos_solution(h_normalized, dx)
        
        bounds_final = [(0, 1) for _ in range(n_points)]
        cons_final = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        
        # Use best solution as starting point
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        
        try:
            res = minimize(
                full_obj, 
                h_perturb,
                method='SLSQP',
                bounds=bounds_final,
                constraints=cons_final,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if res.success:
                c5_final = full_obj(res.x)
                if c5_final < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
