import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(arr, radius=0.01):
    """Smooth approximation to max function"""
    if len(arr) == 0:
        return 0
    max_val = np.max(arr)
    # Exponential smoothing to handle numerical issues
    smoothed = np.log(np.sum(np.exp((arr - max_val) / radius))) * radius + max_val
    return smoothed

def compute_correlation_gradient(h, dx, shifted_h):
    """Compute gradient of correlation objective with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    # This is a simplified approximation
    # In a full implementation this would compute partials properly
    # For now, use finite differences
    eps = 1e-6
    base_corr = evaluate_erdos_solution(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        corr_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (corr_plus - base_corr) / eps
    
    return grad

def trust_region_optimize(h_start, max_iter=200, trust_radius=0.1):
    """Optimize with trust region method"""
    h = h_start.copy()
    n = len(h)
    dx = 2.0 / n
    
    for iter_num in range(max_iter):
        # Compute current value and gradient
        current_c5 = evaluate_erdos_solution(h, dx)
        try:
            grad = compute_correlation_gradient(h, dx, 1 - h)
        except:
            grad = np.zeros(n)
        
        # Trust region update
        # Simple step: move in gradient direction within trust radius
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            step = -grad / grad_norm * trust_radius
            h_new = h + step
        else:
            h_new = h.copy()
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n)
        
        # Evaluate new value
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement
        if new_c5 < current_c5:
            h = h_new.copy()
            trust_radius = min(trust_radius * 1.5, 0.5)  # Expand
        else:
            trust_radius = max(trust_radius * 0.5, 1e-6)  # Shrink
            
        if trust_radius < 1e-6:
            break
            
    return h
