import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_piecewise_segments(h, n_points):
    """Convert step function to piecewise segments"""
    # For simplicity, we work with a 15 segment approximation
    # This would be replaced with actual segment boundaries in a proper implementation
    if n_points == len(h):
        return h
    return np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
