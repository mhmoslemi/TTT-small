import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(conv_vals, beta):
    """Smooth approximation of max function"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * conv_vals)))

def smooth_objective(h, beta):
    """Compute smooth objective value"""
    corr = np.correlate(h, 1 - h, mode='full')
    return smooth_max(corr, beta)

def gradient(h, beta, eps=1e-5):
    """Compute gradient via finite differences"""
    grad = np.zeros_like(h)
    base_val = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_val) / eps
    return grad

    # Initial evaluation
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Optimization parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = gradient(h, beta)
        
        # Try step
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject step
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Adaptive β
        if iter % 100 == 0 and iter > 0:
            beta *= 1.5
            
        # Early stopping if improvement is small
        if iter > 10 and abs(best_c5 - evaluate_erdos_solution(h, dx)) < tolerance:
            break
            
        # Resolution adjustment
        if iter > 500 and n_points < 100:
            n_points_try = min(n_points * 2, 100)
            h_interp = np.interp(np.linspace(0, 2, n_points_try),

    final_c5 = evaluate_erdos_solution(best_h, dx)
