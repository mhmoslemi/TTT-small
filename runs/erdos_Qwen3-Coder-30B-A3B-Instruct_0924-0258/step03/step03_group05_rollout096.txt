import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize variables for optimization
    h_current = best_h.copy()
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Add entropy-based perturbations to avoid local optima
    def perturb_h(h, factor=0.05):
        # Small random perturbation while maintaining sum constraint
        pert = np.random.normal(0, factor, len(h))
        pert = pert - np.mean(pert)  # Ensure mean is 0
        h_new = h + pert
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Gradient-based optimization with projected updates
    def project_h(h):
        return np.clip(h, 0, 1)
    
    def normalize_h(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Track iterations and convergence
    no_improvement_count = 0
    max_no_improvement = 100
    last_best = best_c5
    
    # Optimization loop
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Compute gradients using finite differences
        def compute_gradient(h_vec, eps=1e-4):
            grad = np.zeros_like(h_vec)
            base_c5 = evaluate_erdos_solution(h_vec, dx)
            
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_minus = h_vec.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            return grad
        
        # Gradient descent step
        learning_rate = 1e-3 / (1 + iteration / 100.0)
        grad = compute_gradient(h_current)
        h_new = h_current - learning_rate * grad
        h_new = project_h(h_new)
        h_new = normalize_h(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept improvement or occasionally accept worse solutions for escape
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count >= 50 or iteration % 100 == 0:
            h_current = perturb_h(h_current, 0.02)

        h_current = normalize_h(h_new)
        
        # Check for convergence

            break

        # Occasionally perturb the solution
        if iteration % 100 == 0:
            h_current = perturb_h(h_current, 0.01)
    
    # Final refinement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]

            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
