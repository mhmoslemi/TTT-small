import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based optimization
    n_freq = 5
    bounds = [(-10, 10)] * n_freq
    
    def fourier_to_h(coeffs):
        x = np.linspace(0, 2, n_points)
        g = np.zeros_like(x)
        for i in range(n_freq):
            g += coeffs[i] * np.cos(i * np.pi * x)
        h = 1.0 / (1.0 + np.exp(-g))
        # Normalize to unit integral
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def objective(coeffs):
        if time.time() - start_time > budget_s * 0.95:
            return float('inf')
        h = fourier_to_h(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    # Random restarts
    n_restarts = 20
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        initial_coeffs = np.random.uniform(-5, 5, n_freq)
        
        try:
            res = minimize(objective, initial_coeffs, method='SLSQP', 
                          bounds=bounds, options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                h_candidate = fourier_to_h(res.x)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
