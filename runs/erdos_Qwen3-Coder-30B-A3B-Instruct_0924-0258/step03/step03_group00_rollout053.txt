import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085
    iterations = 15
    tolerance = 1e-5
    
    # Time allocation: 60% for feasibility checks, 40% for local refinement
    feasibility_budget = 0.6 * budget_s
    refinement_budget = 0.4 * budget_s
    
    # Track time for feasibility checks
    feasibility_start = time.time()
    
    for _ in range(iterations):
        if time.time() - start_time > feasibility_budget:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Define convex optimization problem
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1]
        
        # Sum constraint
        constraints.append(sum(h) * dx == 1.0)
        
        # Overlap constraints for all shifts
        for k in range(1, n_points):  # Skip k=0
            # k-shifted correlation sum
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = sum(h[i] * (1 - h[i + k]) for i in range(valid_len)) * dx
                constraints.append(corr_sum <= c_mid)
        
        # Create and solve the problem
        objective = Minimize(0)  # Dummy objective - we just want feasibility
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000, feastol=1e-8, abstol=1e-8)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                # Problem is feasible
                h_solution = h.value
                if h_solution is not None:
                    h_solution = np.clip(h_solution, 0, 1)
                    h_sum = np.sum(h_solution)
                    if h_sum > 0:
                        h_solution = h_solution * (n_points / 2) / h_sum
                    c5_new = evaluate_erdos_solution(h_solution, dx)
                    if c5_new < best_c5:
                        best_h = h_solution.copy()
                        best_c5 = c5_new
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
    
    # Local refinement
    local_start = time.time()
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations with different starting points
        best_local = best_h.copy()
        best_local_c5 = best_c5
        num_restarts = 15
        
        for i in range(num_restarts):
            if time.time() - local_start > refinement_budget:
                break
            if time.time() - start_time > budget_s * 0.95:
                break
            # Perturb current best
            sigma = 0.1 * (1 - i / num_restarts)  # Decreasing noise scale
            h_perturb = best_h + np.random.normal(0, sigma, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_local_c5:
                        best_local = res.x.copy()
                        best_local_c5 = c5_test
            except:
                continue
        
        # Final validation
        if best_local_c5 < best_c5:
            best_h = best_local
            best_c5 = best_local_c5
    
    return (list(best_h), best_c5, n_points)
