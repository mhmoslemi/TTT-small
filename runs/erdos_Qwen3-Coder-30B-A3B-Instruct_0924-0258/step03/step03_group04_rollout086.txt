import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia_weight = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points) * 0.5 + 0.25
    for i in range(swarm_size):
        swarm[i] = project_to_feasible(swarm[i], n_points)
    
    # Initialize personal bests
    personal_best = swarm.copy()
    personal_best_c5 = np.array([evaluate_erdos_solution(p, dx) for p in swarm])
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            inertia = inertia_weight * swarm[i]
            cognitive_term = cognitive * r1 * (personal_best[i] - swarm[i])
            social_term = social * r2 * (global_best - swarm[i])
            velocity = inertia + cognitive_term + social_term
            
            # Update position
            swarm[i] += velocity
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate and update personal best
            c5 = evaluate_erdos_solution(swarm[i], dx)
            if c5 < personal_best_c5[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_c5[i] = c5
                
                # Update global best if needed
                if c5 < global_best_c5:
                    global_best = swarm[i].copy()
                    global_best_c5 = c5
                    if global_best_c5 < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_c5
    
    # Local refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refinement_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(refinement_obj, global_best, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = project_to_feasible(res.x, n_points)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
