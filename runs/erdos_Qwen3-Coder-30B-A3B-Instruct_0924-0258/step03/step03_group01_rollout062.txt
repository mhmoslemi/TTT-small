import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Create a better initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two clusters to avoid trivial solutions
        cluster_size = int(3.0 / dx)  # about 3 units
        h_start[5:5+cluster_size] = 1.0
        h_start[25:25+cluster_size] = 1.0
        # Normalize to sum to 1
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with Simulated Annealing
    population_size = 20
    population = [h_start.copy()]
    for i in range(1, population_size):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    temp = 0.1
    cooling_rate = 0.99
    iterations = 30
    
    for iteration in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate and sort population by fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Generate new population
        new_population = [population[0].copy()]  # Keep best
        
        # Mutation and crossover for rest of population
        for i in range(1, population_size):
            # Random parent
            parent = population[np.random.randint(0, len(population)//2)]
            
            # Perturb parent
            h_mut = parent + np.random.normal(0, temp, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points/2) / np.sum(h_mut)
            
            # Local optimization with SQP
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def cons_sum(h):
                return np.sum(h) - n_points / 2.0
                
            def cons_bounds(h):
                return np.concatenate([h, 1 - h])
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum_eq = {'type': 'eq', 'fun': cons_sum}
            
            try:
                res = minimize(obj, h_mut, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum_eq, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    new_population.append(h_refined)
                else:
                    new_population.append(h_mut)
            except:
                new_population.append(h_mut)
        
        population = new_population
        temp *= cooling_rate
    
    # Final refinement of best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def cons_sum(h):
        return np.sum(h) - n_points / 2.0

    cons_sum_eq = {'type': 'eq', 'fun': cons_sum}

        res = minimize(obj, best_h, method='SLSQP',

        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
