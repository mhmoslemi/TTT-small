def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

                cp.sum(h_ref) == n_points / 2.0,

                h_ref <= 1
            ]

            prob = cp.Problem(cp.Minimize(0), constraints_ref)
