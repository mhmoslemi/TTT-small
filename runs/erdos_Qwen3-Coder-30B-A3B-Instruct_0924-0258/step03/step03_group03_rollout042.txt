import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5 = current_c5
    best_h = h_start.copy()
    best_n_points = n_points
    
    # Try different numbers of points to escape local minima
    candidate_sizes = [25, 30, 35, 40, 45]
    
    for size in candidate_sizes:
        if time() - start_time > budget_s * 0.9:
            break
            
        # Interpolate to new size
        if size != n_points:
            x_old = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, size)
            h_interp = np.interp(x_new, x_old, h_start)
            h_new = project_to_feasible(h_interp, size)
            n_points = size
        else:
            h_new = h_start.copy()
        
        # Local optimization using gradient-like method with finite differences
        for iter_num in range(100):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute derivative using finite differences
            grad = np.zeros_like(h_new)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_new, dx)
            
            for i in range(len(h_new)):
                h_plus = h_new.copy()

            if new_score < best_c5:

                best_h = h_new.copy()
                best_n_points = n_points
                
            # Early stopping for stagnation
            if np.abs(new_score - base_score) < 1e-8:
                break

    if time() - start_time < budget_s * 0.95:
        h_final = best_h.copy()
        dx_final = 2.0 / best_n_points
