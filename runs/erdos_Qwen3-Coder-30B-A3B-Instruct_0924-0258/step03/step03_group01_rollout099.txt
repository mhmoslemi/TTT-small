import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(h.copy())
        personal_best_scores.append(evaluate_erdos_solution(h, dx))
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_max - (w_max - w_min) * iter_num / max_iter
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best[i] - swarm[i])
                           + c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = swarm[i].copy()
                global_best_score = score
        
        # Local refinement every 5 iterations
        if iter_num % 5 == 0 and iter_num > 0:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time < 10:
                break
            # Refine top 3 particles
            top_indices = np.argsort(personal_best_scores)[:3]
            for idx in top_indices:
                h_refine = personal_best[idx].copy()
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def con_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': con_sum}
                
                try:
                    res = minimize(obj, h_refine, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < global_best_score:
                            global_best = h_refined
                            global_best_score = c5_refined
                except:
                    pass
    
    # Final evaluation of global best
    final_c5 = evaluate_erdos_solution(global_best, dx)
    
    if final_c5 < best_c5:
        best_h = global_best
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
