import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try several starting points to avoid local minima
    h_candidates = [h_start]
    
    # Add perturbed versions
    for _ in range(4):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        h_candidates.append(h_pert)
    
    # Gradient-based optimization approach
    def project_and_normalize(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec
    
    def compute_gradient(h_vec, c5_val):
        # For a given h, compute gradient of max overlap
        # This is approximating gradient of max correlation
        # We'll sample at a few key shifts and compute
        n = len(h_vec)
        grad = np.zeros(n)
        
        # Efficient gradient computation using convolution properties
        # Gradient of h[i] w.r.t. max correlation
        for i in range(n):
            # Estimate partial derivative by finite difference
            eps = 1e-6
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus[i] = max(0.0, h_minus[i] - eps)
            
            c_plus = evaluate_erdos_solution(h_plus, dx)
            c_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        return grad
    
    # Main optimization loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Define optimization trajectory with decreasing step sizes
    learning_rates = [0.01 * (0.95 ** i) for i in range(200)]
    
    # Start with best candidate
    current_h = project_and_normalize(h_start.copy())
    current_c5 = evaluate_erdos_solution(current_h, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Use adaptive learning rate
        lr = learning_rates[iteration % len(learning_rates)]
        
        # Compute gradient
        try:
            grad = compute_gradient(current_h, current_c5)
            # Update with gradient descent
            current_h_new = current_h - lr * grad
            
            # Project into feasible region
            current_h_new = project_and_normalize(current_h_new)
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(current_h_new, dx)

                # Random perturbation to escape local minima

            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Ensure final result is valid
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
