import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Annealing parameters
    max_iter = 30
    temp = 0.1
    cool_rate = 0.99
    perturb_mag = 0.01
    
    # Population of solutions
    pop_size = 20
    population = []
    for _ in range(pop_size):
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    def optimize_single(h_in):
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_in, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                return res.x
        except:
            pass
        return h_in
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        for i, h in enumerate(population):
            curr_c5 = evaluate_erdos_solution(h, dx)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
                
            # Refine with SQP
            refined = optimize_single(h)
            refined_c5 = evaluate_erdos_solution(refined, dx)
            if refined_c5 < curr_c5:
                population[i] = refined
                if refined_c5 < best_c5:
                    best_h = refined.copy()
                    best_c5 = refined_c5
        
        # Generate new population with perturbations and simulated annealing
        new_population = []
        for h in population:
            # Perturb
            h_pert = h + np.random.normal(0, perturb_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            # Accept with probability based on temperature
            old_c5 = evaluate_erdos_solution(h, dx)
            new_c5 = evaluate_erdos_solution(h_pert, dx)
            if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
                new_population.append(h_pert)
            else:
                new_population.append(h)
        
        population = new_population
        temp *= cool_rate
    
    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
