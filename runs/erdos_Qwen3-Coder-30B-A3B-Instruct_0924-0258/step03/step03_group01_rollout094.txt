import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize from existing construction or create simple start
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization
    population_size = 20
    mutation_rate = 0.3
    crossover_rate = 0.7
    generations = 30
    sqp_refine_freq = 1
    
    # Initialize population
    population = []
    population.append(h_start.copy())
    for _ in range(population_size - 1):
        h_mut = h_start + np.random.normal(0, 0.05, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        population.append(h_mut)
    
    # GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Selection: top 50%
        selected = population[:population_size//2]
        
        # Crossover: create new individuals
        new_population = selected.copy()
        while len(new_population) < population_size:
            if random.random() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                parent = random.choice(selected)
                child = parent + np.random.normal(0, 0.03, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Update population
        population = new_population[:population_size]
        
        # Local refinement with SQP every few generations
        if gen % sqp_refine_freq == 0:
            for i in range(min(3, len(population))):
                h_refine = population[i].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def con_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': con_sum}
                
                res = minimize(obj, h_refine, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    refined = res.x
                    refined = np.clip(refined, 0, 1)
                    refined = refined * (n_points / 2) / np.sum(refined)
                    c5_refined = evaluate_erdos_solution(refined, dx)
                    if c5_refined < fitness_scores[i]:
                        population[i] = refined
                        if c5_refined < best_c5:
                            best_h = refined
                            best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
