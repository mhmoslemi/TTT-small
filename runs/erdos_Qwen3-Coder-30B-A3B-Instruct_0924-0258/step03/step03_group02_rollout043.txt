import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization with clustering approach
    n_clusters = 10
    max_iter = 100
    cluster_iter = 0
    
    h_current = best_h.copy()
    no_improve_count = 0
    
    # Precompute shift vectors for clustering
    shift_vectors = []
    for k in range(n_points):
        # Create overlap vector for shift k
        if k == 0:
            v = np.ones(n_points)
        else:
            v = np.concatenate([np.ones(n_points - k), np.zeros(k)])
        shift_vectors.append(v)
    shift_vectors = np.array(shift_vectors)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Cluster shifts based on their impact vectors
        kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
        cluster_assignments = kmeans.fit_predict(shift_vectors)
        
        # For each cluster, optimize the corresponding overlaps
        for cluster_id in range(n_clusters):
            if time() - start_time > budget_s * 0.95:
                break
                
            indices = np.where(cluster_assignments == cluster_id)[0]
            if len(indices) == 0:
                continue
                
            # Randomly sample a few indices for efficient local optimization
            if len(indices) > 10:
                indices = np.random.choice(indices, size=10, replace=False)
            
            # Local optimization
            h_local = h_current.copy()
            
            # Simple gradient step for each shift in the cluster
            for k in indices:
                # Create approximate gradient information
                # This is a simplified model: not truly gradient but model-based
                if k == 0:
                    idx = slice(None)
                    idx_back = slice(None)
                else:
                    idx = slice(k, None)
                    idx_back = slice(None, -k)
                
                grad = np.zeros_like(h_local)
                if k > 0:
                    grad[:-k] = h_local[k:] - h_local[:-k]  # Simplified effect
                else:
                    grad = h_local - h_local  # Zero effect for k=0
                    
                # Update with small step size
                h_local = h_local - 0.01 * grad
                h_local = project_to_feasible(h_local, n_points)
            
            # Convex relaxation step for cluster (simplified version)
            # Since full convex relaxation of C5 is complex, we do a simpler approach:
            # minimize-overlapping-components using convex relaxation for bounding
            try:
                # Reconstruct the overlap approximation as a linearized system
                h_vars = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_vars) == n_points / 2,
                    h_vars >= 0,
                    h_vars <= 1
                ]
                
                # Simplified convex approximation for few key overlaps
                for k in indices[:5]:  # Limit to some shifts for performance
                    if k == 0:
                        # Overlap with self
                        overlap = h_vars @ (1 - h_vars)
                    else:
                        # For overlap with shift k, we compute a simple approximation
                        # Simplified: only part of the data
                        sum_part = 0
                        for i in range(n_points - k):
                            sum_part += h_vars[i] * (1 - h_vars[i+k])
                        overlap = sum_part
                    
                    constraints.append(overlap <= 0.4)  # Loose upper bound
                    
                # Solve it
                prob = cp.Problem(cp.Minimize(cp.sum(h_vars)), constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_new = h_vars.value
                    if h_new is not None:
                        h_new = project_to_feasible(h_new, n_points)
                        h_test = h_new.copy()
                        c5_test = evaluate_erdos_solution(h_test, dx)
                        if c5_test < best_c5:
                            best_h = h_test.copy()
                            best_c5 = c5_test
                            no_improve_count = 0
            except:
                pass
            
            # Final update if we did better
            h_current = best_h.copy()
        
        # Refresh basis every 10 iterations to improve quality
        if iteration % 10 == 0:
            h_current = project_to_feasible(h_current, n_points)
            
        # Check for stagnation
        no_improve_count += 1
        if iteration > 0 and iteration % 20 == 0:
            # Re-evaluate how we're doing
            current_c5 = evaluate_erdos_solution(h_current, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                no_improve_count = 0
                
        if no_improve_count > 50:
            break
    
    # Post-processing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
