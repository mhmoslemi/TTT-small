import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def create_individual():
        # Create a random individual with correct sum
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # normalize
        return h
    
    def create_population(size):
        # Create population, ensure one individual is from initial
        pop = []
        if 'initial_h_values' in globals():
            pop.append(h_start.copy())
            size -= 1
        for _ in range(size):
            pop.append(create_individual())
        return pop
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def selection(population, fitnesses, k=5):
        # Tournament selection
        selected = []
        for _ in range(len(population)):
            tournament = random.sample(list(zip(population, fitnesses)), k)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        return selected
    
    def crossover(parent1, parent2):
        # One-point crossover
        point = random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Normalize children
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def mutate(individual, rate=0.2, sigma=0.1):
        # Gaussian mutation
        for i in range(len(individual)):
            if random.random() < rate:
                individual[i] += np.random.normal(0, sigma)
        # Clip to [0, 1]
        individual = np.clip(individual, 0, 1)
        # Renormalize
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    pop_size = 50
    population = create_population(pop_size)
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Track best
        min_fit = min(fitnesses)
        if min_fit < best_c5:
            best_c5 = min_fit
            best_h = population[fitnesses.index(min_fit)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                new_population.append(mutate(child1))
                new_population.append(mutate(child2))
            else:
                new_population.append(mutate(selected[i]))
        
        # Keep some of the best from old population
        sorted_indices = np.argsort(fitnesses)[:pop_size//2]
        for i in sorted_indices:
            new_population.append(population[i].copy())
            
        # Trim to population size
        population = new_population[:pop_size]
        
        generation += 1
        
    # Final refinement with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
