import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess with two clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for simulated annealing
    temp_start = 0.1
    temp_decay = 0.99
    perturb_magnitude = 0.01
    pop_size = 20
    iterations = 30
    
    # Generate initial population
    population = []
    for _ in range(pop_size):
        h_pert = h_start + np.random.normal(0, perturb_magnitude, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        scale = (n_points/2) / np.sum(h_pert)
        h_pert *= scale
        population.append(h_pert)
    
    # Main optimization loop
    temp = temp_start
    for iteration in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current population
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Refine each member using SQP
        new_population = []
        for h in population:
            # SQP refinement
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
                
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                else:
                    h_refined = h
            except:
                h_refined = h
                
            # Ensure constraints are met after optimization
            h_refined = np.clip(h_refined, 0, 1)
            scale = (n_points/2) / np.sum(h_refined)
            h_refined *= scale
            
            new_population.append(h_refined)
        
        population = new_population
        
        # Simulated Annealing step
        for i in range(pop_size):
            h_current = population[i]
            h_new = h_current + np.random.normal(0, perturb_magnitude, n_points)
            h_new = np.clip(h_new, 0, 1)
            scale = (n_points/2) / np.sum(h_new)
            h_new *= scale
            
            c5_old = evaluate_erdos_solution(h_current, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_old or np.random.rand() < math.exp(-(c5_new - c5_old) / temp):
                population[i] = h_new
        
        temp *= temp_decay
    
    # Final refinement with scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
