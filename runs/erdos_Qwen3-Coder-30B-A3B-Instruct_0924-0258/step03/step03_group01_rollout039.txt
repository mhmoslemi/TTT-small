import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to sum to n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with SQP refinement
    population_size = 20
    max_iterations = 30
    temperature = 0.1
    decay_rate = 0.99
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h_candidate = h_start + np.random.normal(0, 0.01, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all candidates
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Find best in population
        best_in_pop = min(fitnesses)
        if best_in_pop < best_c5:
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_c5 = best_in_pop
            
        # Refine each candidate with SQP
        for i in range(population_size):
            h_current = population[i]
            
            # Define objective for this candidate
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points / 2.0
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < fitnesses[i]:
                        population[i] = h_refined
                        fitnesses[i] = c5_refined
            except:
                pass
        
        # Simulated Annealing step
        for i in range(population_size):
            h_current = population[i]
            # Perturb
            h_perturbed = h_current + np.random.normal(0, temperature, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            
            # Accept or reject
            if c5_perturbed < fitnesses[i]:
                population[i] = h_perturbed
                fitnesses[i] = c5_perturbed
            else:
                # Accept with some probability
                delta = c5_perturbed - fitnesses[i]
                if np.random.rand() < math.exp(-delta / temperature):
                    population[i] = h_perturbed
                    fitnesses[i] = c5_perturbed
                    
        temperature *= decay_rate
        
        # Update global best
        if min(fitnesses) < best_c5:
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_c5 = min(fitnesses)
    
    # Final refinement of best solution
    def obj_final(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum_final(h):
        return np.sum(h) - n_points / 2.0
        
    bounds_final = [(0, 1) for _ in range(n_points)]
    cons_sum_final = {'type': 'eq', 'fun': con_sum_final}
    
    try:
        res = minimize(obj_final, best_h, method='SLSQP', 
                       bounds=bounds_final, constraints=cons_sum_final, 
                       options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = obj_final(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
