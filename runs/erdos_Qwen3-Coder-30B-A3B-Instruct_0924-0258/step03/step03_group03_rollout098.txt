import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segment_params(h, n_points, dx):
    """Compute segment heights from step function"""
    return h.copy()

def compute_step_from_segments(segments, boundaries, n_points, dx):
    """Reconstruct step function from segments and boundaries"""
    h = np.zeros(n_points)
    for i in range(len(segments)):
        start_idx = int(boundaries[i] / dx)
        end_idx = int(boundaries[i+1] / dx)
        h[start_idx:end_idx] = segments[i]
    return h

def log_sum_exp(x, beta=100):
    """Numerically stable log-sum-exp for smoothing max"""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(beta * (x - x_max)))) / beta

    grad_w = np.zeros(n_segs)

        h_plus = h.copy()

        base_score = evaluate_erdos_solution(h, dx)
        score_plus = evaluate_erdos_solution(h_plus_reconstructed, dx)
        grad_w[i] = (score_plus - base_score) / eps
