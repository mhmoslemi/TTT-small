import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based approach
    d = 5  # number of cosine terms
    
    def construct_h(coeffs):
        # Construct h from cosine coefficients using logistic transform
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + np.sum(coeffs[1:] * np.cos(np.arange(1, d+1) * np.pi * x), axis=0)
        h = 1.0 / (1.0 + np.exp(-g))
        # Normalize to sum to 1
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def objective(coeffs):
        h = construct_h(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(coeffs):
        h = construct_h(coeffs)
        return np.sum(h) - n_points / 2
    
    # Optimization parameters
    bounds = [(-10, 10)] * (d + 1)
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Multiple random restarts
    best_coeffs = np.zeros(d + 1)
    num_restarts = 20
    
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random starting point
        coeffs_start = np.random.uniform(-5, 5, d + 1)
        
        try:
            # Optimize using SLSQP
            res = minimize(objective, coeffs_start, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            
            if res.success:
                h_new = construct_h(res.x)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_coeffs = res.x.copy()
                    best_h = h_new.copy()
        except:
            continue
    
    # Final refinement
    if time.time() - start_time < budget_s * 0.95:
        h_refined = construct_h(best_coeffs)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    return (list(best_h), best_c5, n_points)
