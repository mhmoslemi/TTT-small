import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    def gradient_descent_optimization(h_init, max_iter=1000):
        h_current = h_init.copy()
        learning_rate = 1e-3
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Calculate overlaps for all shifts
            corr_full = np.correlate(h_current, 1 - h_current, mode='full')
            # Get maximum overlap index
            max_idx = np.argmax(corr_full)
            # Get the actual overlap value
            max_overlap = corr_full[max_idx] * dx
            
            # Compute gradient approximation (simplified approach)
            grad = np.zeros_like(h_current)
            for i in range(len(h_current)):
                # Compute finite difference gradient for this position
                h_pert = h_current.copy()
                h_pert[i] += 1e-6
                corr_plus = np.correlate(h_pert, 1 - h_pert, mode='full')
                h_minus = h_current.copy()
                h_minus[i] -= 1e-6
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                
                grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) * 1e6 * dx
            
            # Update with gradient and adapt learning rate
            h_new = h_current - learning_rate * grad
            
            # Project to valid bounds [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to satisfy integral constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)

            new_c5 = evaluate_erdos_solution(h_new, dx)

            if new_c5 < best_c5:

                learning_rate = min(learning_rate * 1.01, 1e-2)
            else:
                learning_rate *= 0.99
                
            h_current = h_new

            if iteration % 50 == 0 and iteration > 0:

                perturb = np.random.normal(0, 1e-4, n_points)
                h_current += perturb

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_current.copy()

    h_best, c5_best = gradient_descent_optimization(best_h)
