import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a different approach: direct optimization with constraints
    def optimize_direct():
        nonlocal best_h, best_c5
        
        #Initial guess
        h_init = best_h.copy()
        
        # Function to compute the objective
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Constraint functions
        def integral_constraint(h_vec):
            return np.sum(h_vec) * dx - 1.0
            
        def bounds_low(h_vec):
            return h_vec
            
        def bounds_high(h_vec):
            return 1 - h_vec
            
        # Constraints for SLSQP
        cons = [
            {'type': 'eq', 'fun': integral_constraint},
            {'type': 'ineq', 'fun': bounds_low},
            {'type': 'ineq', 'fun': bounds_high}
        ]
        
        # Bounds for variables
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run minimization
        try:
            res = minimize(objective, h_init, method='SLSQP',

            if res.success:
                h_opt = np.clip(res.x, 0, 1)
