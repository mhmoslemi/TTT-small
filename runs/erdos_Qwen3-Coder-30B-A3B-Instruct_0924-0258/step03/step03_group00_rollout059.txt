import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search for local optimization
    tabu_list = []
    tabu_length = 10
    
    # Simulated annealing parameters
    T = 1.0
    cooling_rate = 0.99
    min_T = 1e-6
    
    # Perturbation types
    perturbations = [0.0, 1.0, 0.5]
    
    # Additional local search parameters
    max_iter = int(5e5)
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        iter_count += 1
        if (iter_count % 1000) == 0:
            if (time.time() - start_time) > budget_s * 0.9:
                break
                
        # Pick a random segment to modify
        i = np.random.randint(0, n_points)
        
        # Get current value
        current_value = best_h[i]
        
        # Try different perturbations
        new_values = [v for v in perturbations if abs(v - current_value) > 1e-6]
        if not new_values:
            new_values = [1.0 if current_value < 0.5 else 0.0]
            
        new_value = np.random.choice(new_values)
        
        # Create candidate
        candidate = best_h.copy()
        candidate[i] = new_value
        
        # Ensure integral constraint is maintained
        # We'll normalize the sum to satisfy the constraint exactly
        current_sum = np.sum(candidate)
        if current_sum > 0:
            candidate = candidate * (n_points / 2) / current_sum
        
        # Check tabu list
        candidate_tuple = tuple(candidate)
        if candidate_tuple in tabu_list:
            continue
            
        # Evaluate candidate
        c5_new = evaluate_erdos_solution(candidate, dx)
        
        # Accept if better or use simulated annealing
        if c5_new < best_c5:
            best_h = candidate.copy()
            best_c5 = c5_new
            tabu_list.append(candidate_tuple)
            if len(tabu_list) > tabu_length:
                tabu_list.pop(0)
        else:
            # Simulated annealing acceptance probability
            if T > min_T:
                delta = c5_new - best_c5
                if np.random.rand() < np.exp(-delta / T):
                    best_h = candidate.copy()
                    best_c5 = c5_new
                    tabu_list.append(candidate_tuple)
                    if len(tabu_list) > tabu_length:
                        tabu_list.pop(0)
        
        # Cool down
        T *= cooling_rate
    
    # Final local refinement
    if (time.time() - start_time) < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple local minimizations
        for i in range(10):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            # Start from best solution
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
