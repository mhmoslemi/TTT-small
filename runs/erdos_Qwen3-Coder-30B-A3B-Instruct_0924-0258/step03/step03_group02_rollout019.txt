import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_overlap_vectors(h, dx, n_points):
    """Compute overlap vectors for each shift k"""
    overlap_vectors = []
    for k in range(n_points):
        if k == 0:
            overlap = np.sum(h * (1 - h)) * dx
        else:
            overlap = np.sum(h[:-k] * (1 - h[k:])) * dx
        overlap_vectors.append(overlap)
    return np.array(overlap_vectors)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Clustering approach
    max_iters = 200
    cluster_count = 10
    stagnation_count = 0
    last_improvement = 0
    
    # Initial clustering
    overlap_vectors = []
    for k in range(n_points):
        if k == 0:
            overlap = np.sum(h_start * (1 - h_start)) * dx
        else:
            overlap = np.sum(h_start[:-k] * (1 - h_start[k:])) * dx
        overlap_vectors.append(overlap)
    overlap_vectors = np.array(overlap_vectors).reshape(-1, 1)
    
    kmeans = KMeans(n_clusters=cluster_count, n_init=10, random_state=seed)
    kmeans.fit(overlap_vectors)
    cluster_labels = kmeans.labels_
    
    h_current = h_start.copy()
    
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Apply clustering-based optimization
        for cluster_id in range(cluster_count):
            # Get indices in this cluster
            cluster_indices = np.where(cluster_labels == cluster_id)[0]
            
            if len(cluster_indices) == 0:
                continue
                
            # Local optimization for this cluster
            h_prev = h_current.copy()
            
            # Create a simpler optimization for this subset of shifts
            def cluster_objective(h_subset):
                total_overlap = 0.0
                for k in cluster_indices:
                    if k == 0:
                        total_overlap += np.sum(h_subset * (1 - h_subset)) * dx
                    else:
                        total_overlap += np.sum(h_subset[:-k] * (1 - h_subset[k:])) * dx
                return total_overlap
                
            # Simple gradient-based local update
            h_updated = h_current.copy()
            for _ in range(10):
                h_candidates = []
                for i in range(len(h_current)):
                    h_test = h_current.copy()
                    h_test[i] = max(0, min(1, h_current[i] + 0.01))
                    h_candidates.append((h_test, cluster_objective(h_test)))
                    
                # Find best neighbor
                best_h_candidate = min(h_candidates, key=lambda x: x[1])[0]
                if cluster_objective(best_h_candidate) < cluster_objective(h_updated):
                    h_updated = best_h_candidate
                    
            h_current = h_updated

        # Apply feasibility projection
        h_current = project_to_feasible(h_current, n_points)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Periodic re-clustering
        if iteration % 50 == 0 and iteration > 0:
            overlap_vectors = []
            for k in range(n_points):
                if k == 0:
                    overlap = np.sum(h_current * (1 - h_current)) * dx
                else:
                    overlap = np.sum(h_current[:-k] * (1 - h_current[k:])) * dx
                overlap_vectors.append(overlap)
            overlap_vectors = np.array(overlap_vectors).reshape(-1, 1)
            kmeans.fit(overlap_vectors)
            cluster_labels = kmeans.labels_

        if stagnation_count > 100:
            break
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5 and best_c5 > 0.37:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 20, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
