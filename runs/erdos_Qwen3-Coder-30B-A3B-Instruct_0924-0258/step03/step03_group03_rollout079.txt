import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.2 * np.sin(2 * np.pi * x) - 0.1 * np.sin(4 * np.pi * x)
        h_start = np.clip(h_start, 0, 1)
        
    dx = 2.0 / n_points
    h_start = project_to_feasible(h_start, n_points)
    
    # Optimize using a hybrid approach combining convex relaxation and gradient-based method
    h_current = h_start.copy()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # First, try simple gradient descent to refine the solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
