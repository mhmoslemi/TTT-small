import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define a more effective search approach with genetic algorithm
    # Generate population of candidate solutions
    def random_h():
        # Generate a random valid h that sums to 1
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return np.clip(h, 0, 1)
    
    def mutate_h(h, rate=0.1):
        # Mutate h with given rate (bit-flip for binary, small perturbation for continuous)
        h_new = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                h_new[i] = np.random.rand()  # Continuous mutation
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        return np.clip(h_new, 0, 1)
    
    def crossover_h(h1, h2):
        # Uniform crossover
        h_new = np.zeros_like(h1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h_new[i] = h1[i]
            else:
                h_new[i] = h2[i]
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        return np.clip(h_new, 0, 1)
    
    # Genetic algorithm settings
    population_size = 200
    generations = 50
    elite_size = 20
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population
    population = [random_h() for _ in range(population_size)]
    # Ensure first individual is from starting point
    population[0] = h_start.copy()
    
    # Main evolutionary loop
    for generation in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Track best solution so far
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                # Select two parents
                parents = np.random.choice(elite_size, 2, replace=False)
                h1 = population[parents[0]]
                h2 = population[parents[1]]
                child = crossover_h(h1, h2)
                child = mutate_h(child, mutation_rate)
                new_population.append(child)
            else:
                # Mutation only
                parent = population[np.random.randint(elite_size)]
                child = mutate_h(parent, mutation_rate)
                new_population.append(child)
        
        population = new_population
    
    # Final local refinement with a simple gradient descent approach
    def obj_func(x):
        h = np.clip(x, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h_norm = h * (n_points / 2) / h_sum
        else:
            h_norm = h
        return evaluate_erdos_solution(h_norm, dx)

    def obj_grad(x):
        # Approximate gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(x)
        base_val = obj_func(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            grad[i] = (obj_func(x_plus) - base_val) / eps
        return grad

    if time.time() - start_time < budget_s * 0.95:
        # Run refine with current best
        h_refined = best_h.copy()
        def refine_obj(x):
            h = np.clip(x, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h_norm = h * (n_points / 2) / h_sum
            else:
                h_norm = h
            return evaluate_erdos_solution(h_norm, dx)

        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start with small perturbation to avoid local minima
        for _ in range(5):
            h_perturb = h_refined + np.random.normal(0, 0.02, n_points)
            try:
                res = minimize(refine_obj, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = refine_obj(res.x)
                    if c5_test < best_c5:
                        h_refined = res.x.copy()
                        best_c5 = c5_test
            except:
                continue

        if best_c5 < evaluate_erdos_solution(best_h, dx):
            best_h = h_refined.copy()

    return (list(best_h), best_c5, n_points)
