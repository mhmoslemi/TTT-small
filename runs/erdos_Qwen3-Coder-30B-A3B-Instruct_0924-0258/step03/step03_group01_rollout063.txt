import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing setup
    temperature = 0.1
    cooling_rate = 0.99
    max_iter = 30
    population_size = 20
    perturbation_std = 0.01
    
    def refine_solution(h):
        """Refine solution using SLSQP"""
        def objective(x):
            return evaluate_erdos_solution(x, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                return res.x
        except:
            pass
        return h
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_pert = h_start + np.random.normal(0, perturbation_std, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate and refine population
        for i, h in enumerate(population):
            refined_h = refine_solution(h)
            c5_refined = evaluate_erdos_solution(refined_h, dx)
            
            if c5_refined < best_c5:
                best_h = refined_h
                best_c5 = c5_refined
                
            population[i] = refined_h
        
        # Simulated Annealing step
        current_best = best_h
        for _ in range(5):  # Multiple annealing steps per iteration
            if time() - start_time > budget_s * 0.95:
                break
                
            # Generate neighbor
            neighbor = current_best + np.random.normal(0, perturbation_std, n_points)
            neighbor = np.clip(neighbor, 0, 1)
            neighbor = neighbor * (n_points/2) / np.sum(neighbor)
            
            c5_neighbor = evaluate_erdos_solution(neighbor, dx)
            
            # Accept with probability based on temperature
            if c5_neighbor < best_c5 or np.random.random() < math.exp(-(c5_neighbor - best_c5) / temperature):
                current_best = neighbor
                if c5_neighbor < best_c5:
                    best_h = neighbor
                    best_c5 = c5_neighbor
                    
        # Cool down
        temperature *= cooling_rate
    
    # Final refinement of the best solution
    final_h = refine_solution(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
