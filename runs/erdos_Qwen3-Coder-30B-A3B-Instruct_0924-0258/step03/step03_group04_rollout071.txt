import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    h_current = h_start.copy()
    max_iter = int(1e5)
    step_size = 0.5
    perturbation_rate = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current maximum overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Linear approximation around current point
        # Find shift that gives max overlap
        if max_k < n_points:
            k = max_k
            h_shifted = np.concatenate([h_current[k:], np.zeros(k)])
        else:
            k = max_k - n_points + 1
            h_shifted = np.concatenate([np.zeros(k), h_current[:-k]])
        
        # Linear objective: sum of (1 - h_shifted[i]) * h_current[i] 
        # But we look at the gradient of the correlation, which should be
        # approximately (1 - h_shifted[i]) when we look at the shift k 
        linear_grad = 1 - h_shifted
        # We want to minimize sum(h_current[i] * (1 - h_shifted[i]))
        # So we pick indices i where (1 - h_shifted[i]) is largest
        
        # Subproblem: find extreme point that minimizes h^T * (1-h_shifted)
        # Extreme points are binary vectors with exactly n/2 ones
        
        # Greedy approach to find binary extreme point
        # Sort indices by (1 - h_shifted[i])
        indices = np.argsort(linear_grad)[::-1]
        h_extreme = np.zeros(n_points)
        h_extreme[indices[:n_points//2]] = 1.0
        
        # Update with Frank-Wolfe step
        h_next = (1 - step_size) * h_current + step_size * h_extreme
        
        # Project to feasible region
        h_next = project_to_feasible(h_next, n_points)
        
        # Evaluate
        c5_current = evaluate_erdos_solution(h_next, dx)
        if c5_current < best_c5:
            best_h = h_next.copy()
            best_c5 = c5_current
            
        # Safety adjustment to prevent getting stuck
        if iteration % 20 == 0 and np.random.random() < perturbation_rate:
            idx = np.random.randint(n_points)
            h_next[idx] = np.random.random()
            h_next = project_to_feasible(h_next, n_points)
            
        h_current = h_next
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
