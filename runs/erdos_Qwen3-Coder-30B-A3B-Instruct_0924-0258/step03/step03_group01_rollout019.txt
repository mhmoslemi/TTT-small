import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 20
    num_generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_refinement_interval = 1
    
    # Initialize population
    population = []
    population_fitness = []
    
    # Start with initial solution
    h_scaled = h_start * (n_points / 2) / np.sum(h_start)
    population.append(h_scaled)
    population_fitness.append(evaluate_erdos_solution(h_scaled, dx))
    
    # Add random perturbations to create diverse initial population
    for i in range(population_size - 1):
        if time() - start_time > budget_s * 0.9:
            break
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
        population_fitness.append(evaluate_erdos_solution(h_pert, dx))
    
    # GA main loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(population_fitness)
        population = [population[i] for i in sorted_indices]
        population_fitness = [population_fitness[i] for i in sorted_indices]
        
        # Keep top 50%
        elite_size = population_size // 2
        elite = population[:elite_size]
        elite_fitness = population_fitness[:elite_size]
        
        # Generate offspring
        new_population = elite.copy()
        new_population_fitness = elite_fitness.copy()
        
        # Crossover
        num_crossovers = int(population_size * crossover_rate)
        for _ in range(num_crossovers):
            if time() - start_time > budget_s * 0.95:
                break
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.zeros(n_points)
            child[:crossover_point] = parent1[:crossover_point]
            child[crossover_point:] = parent2[crossover_point:]
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
            new_population_fitness.append(evaluate_erdos_solution(child, dx))
        
        # Mutation
        num_mutations = int(population_size * mutation_rate)
        for _ in range(num_mutations):
            if time() - start_time > budget_s * 0.95:
                break
            idx = np.random.randint(0, len(new_population))
            child = new_population[idx].copy()
            # Add Gaussian noise
            noise = np.random.normal(0, 0.03, n_points)
            child += noise
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            new_population[idx] = child
            new_population_fitness[idx] = evaluate_erdos_solution(child, dx)
        
        population = new_population
        population_fitness = new_population_fitness
        
        # Local SQP refinement (every few generations to save time)
        if generation % sqp_refinement_interval == 0:
            for i in range(min(3, len(population))):
                if time() - start_time > budget_s * 0.95:
                    break
                h_init = population[i]
                
                # Define constraint function for sum == 17.5
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                # Define bounds for each variable
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                # Try to refine with SQP
                try:
                    result = minimize(
                        evaluate_erdos_solution,
                        h_init,
                        args=(dx,),
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons_sum,
                        options={'maxiter': 100, 'ftol': 1e-6}
                    )
                    if result.success:
                        h_refined = result.x
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < population_fitness[i]:
                            population[i] = h_refined
                            population_fitness[i] = c5_refined
                except:
                    pass  # Ignore failures and continue
        
        # Update best solution
        min_idx = np.argmin(population_fitness)
        if population_fitness[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = population_fitness[min_idx]
    
    # Make sure we return a feasible solution
    best_h = np.clip(best_h, 0, 1) 
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
