import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing Parameters
    max_iter = 300
    temp_init = 0.1
    temp_decay = 0.99
    perturbation_scale = 0.01
    
    # Population-based refinement
    population_size = 20
    population = []
    for _ in range(population_size):
        h_candidate = h_start + np.random.normal(0, perturbation_scale, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Track best in population
    for h_candidate in population:
        c5_val = evaluate_erdos_solution(h_candidate, dx)
        if c5_val < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_val

    # Main optimization loop
    temp = temp_init
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample from population
        for _ in range(2):
            if time() - start_time > budget_s * 0.95:
                break
            h_current = population[np.random.randint(0, population_size)].copy()
            
            # Perturb the candidate
            h_perturbed = h_current + np.random.normal(0, perturbation_scale, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            # SQP refinement
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
            
            # Acceptance probability for simulated annealing
            c5_current = evaluate_erdos_solution(h_current, dx)
            if c5_current > 0:
                accept_prob = math.exp(-abs(c5_refined - c5_current) / temp)
                if np.random.rand() < accept_prob:
                    h_current = h_refined.copy()
            
            # Replace worst in population
            worst_idx = np.argmax([evaluate_erdos_solution(h, dx) for h in population])
            population[worst_idx] = h_current.copy()
            
            # Update global best
            if evaluate_erdos_solution(h_current, dx) < best_c5:
                best_h = h_current.copy()
                best_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Reduce temperature
        temp *= temp_decay
    
    # Final cleanup to make sure it's feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
