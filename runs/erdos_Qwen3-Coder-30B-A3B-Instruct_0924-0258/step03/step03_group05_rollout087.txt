import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Wavelet-inspired approach using Haar wavelets for basis
    # Create Haar wavelet basis
    def create_haar_basis(n):
        basis = []
        # Scaling function (constant)
        phi = np.ones(n)
        basis.append(phi / np.linalg.norm(phi))
        
        # Wavelet functions
        for j in range(1, int(np.log2(n)) + 1):
            for k in range(2**(j-1)):
                psi = np.zeros(n)
                start = 2**(j-1) * k
                end = start + 2**(j-1)
                mid = (start + end) // 2
                psi[start:mid] = 1
                psi[mid:end] = -1
                basis.append(psi / np.linalg.norm(psi))
        return np.array(basis)
    
    # Generate basis
    if n_points >= 4:
        try:
            basis = create_haar_basis(n_points)
            basis = basis[:n_points]  # Limit to n_points
        except:
            # Fallback to simple basis
            basis = np.eye(n_points)
    else:
        basis = np.eye(n_points)

    max_iterations = 100
    
    for iteration in range(max_iterations):

            # Reconstruct h from coefficients
