import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    velocity_bounds = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize with existing construction
    if 'initial_h_values' in globals():
        particles.append(h_start.copy())
        particles[-1] = project_to_feasible(particles[-1], n_points)
        pbest.append(particles[-1].copy())
        pbest_vals.append(evaluate_erdos_solution(particles[-1], dx))
    else:
        particles.append(h_start.copy())
        particles[-1] = project_to_feasible(particles[-1], n_points)
        pbest.append(particles[-1].copy())
        pbest_vals.append(evaluate_erdos_solution(particles[-1], dx))
    
    # Initialize rest randomly
    for i in range(n_particles - 1):
        h = np.random.rand(n_points) * 0.5
        h = project_to_feasible(h, n_points)
        particles.append(h)
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # Initialize velocities
    for i in range(n_particles):
        vel = (np.random.rand(n_points) - 0.5) * 2 * velocity_bounds
        velocities.append(vel)
    
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = fitness
                if fitness < gbest_val:
                    gbest = pbest[i].copy()
                    gbest_val = fitness
                    iter_no_improve = 0
            
            # Check for stagnation
            iter_no_improve += 1
            if iter_no_improve > 100:
                break
        
        if iter_no_improve > 100:
            break
            
    # Refine with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and gbest_val < best_c5:
        h_refined = gbest.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
