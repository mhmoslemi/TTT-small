import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def build_segmented_h(segments, segment_widths):
    """Build piecewise constant h from segments and widths"""
    h = np.zeros(len(segment_widths))
    for i in range(len(segments)):
        h[i] = segments[i]
    return h

def smooth_max_correlation(h, dx, smoothing_radius=0.1):
    """Compute a smooth approximation of the maximum correlation"""
    # This smooth approximation mimics the behavior of max_k int h(x)(1-h(x+k))dx
    # by convolution with a smoothed kernel
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    
    # Smooth the correlation to make it differentiable
    # Use a simple approach: replace max with a soft maximum
    soft_max = np.log(np.sum(np.exp(corr / smoothing_radius))) * smoothing_radius
    return soft_max * dx

def trust_region_optimize(h0, n_points, dx, budget_s):
    """Trust-region optimization for minimizing C5"""
    max_iter = 1000
    trust_radius = 0.1
    radius_decay = 0.5
    radius_min = 1e-6
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0, dx)
    start_time = time()

    # Add small perturbation to avoid degenerate starting point
    h_current = h0 + np.random.normal(0, 1e-5, n_points)
    h_current = project_to_feasible(h_current, n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        base_val = evaluate_erdos_solution(h_current, dx)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (val_plus - base_val) / eps
            
        # Update using projected gradient descent within trust region
        step = -0.001 * grad
        step_norm = np.linalg.norm(step)
        
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
            
        new_h = h_current + step
        new_h = project_to_feasible(new_h, n_points)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept step if better
        if new_c5 < current_c5:
            h_current = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
        else:
            trust_radius *= radius_decay
            if trust_radius < radius_min:
                trust_radius = radius_min
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with initial solution if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        best_c5 = current_c5
        best_n_points = n_points
    else:
        # Rough symmetric initialization
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points

    # Try a few simplified configurations first
    candidates = [30, 32, 35]
    for n in candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n
        h_try = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_start)), h_start)
        h_try = project_to_feasible(h_try, n)
        c5_try = evaluate_erdos_solution(h_try, dx_try)
        
        if c5_try < best_c5:
            best_h = h_try.copy()
            best_c5 = c5_try
            best_n_points = n

    # Apply trust-region optimization to the best configuration
    if time() - start_time < budget_s * 0.9:
        best_h, best_c5 = trust_region_optimize(best_h, best_n_points, 2.0 / best_n_points, budget_s * 0.9)

    # Final refinement with gradient descent
    if time() - start_time < budget_s * 0.95:
        h_refined = best_h.copy()
        dx_refined = 2.0 / best_n_points
        for _ in range(200):
            if time() - start_time > budget_s * 0.99:
                break
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_refined, dx_refined)
            
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx_refined)
                grad[i] = (score_plus - base_score) / eps

            step_size = 0.0005
            h_refined = h_refined - step_size * grad
            h_refined = project_to_feasible(h_refined, best_n_points)

            score_refined = evaluate_erdos_solution(h_refined, dx_refined)
            if score_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = score_refined

    # Ensure final state
    best_h = project_to_feasible(best_h, best_n_points)
    
    return (list(best_h), best_c5, best_n_points)
