import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with projected updates
    def project_h(h_vec):
        # Ensure h is in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Renormalize to satisfy integral constraint
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec
    
    # Simple gradient descent approach
    max_iter = 10000
    patience = 100
    
    # Initialize learning rate
    lr = 0.01
    
    # Track improvement
    last_improvement = 0
    best_c5_so_far = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation
        h_current = best_h.copy()
        h_current = project_h(h_current)
        
        # Use finite differences for gradient
        eps = 1e-5
        gradient = np.zeros_like(h_current)
        
        for j in range(len(h_current)):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            c_plus = evaluate_erdos_solution(h_plus, dx)
            
            h_minus = h_current.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            c_minus = evaluate_erdos_solution(h_minus, dx)
            
            gradient[j] = (c_plus - c_minus) / (2 * eps)
        
        # Update with gradient
        new_h = h_current - lr * gradient
        
        # Project and normalize
        new_h = project_h(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            last_improvement = i
            best_c5_so_far = min(best_c5_so_far, new_c5)
        
        # Adaptive learning rate
        if i % 10 == 0:
            lr *= 0.995
        
        # Early stopping if no improvement
        if i - last_improvement > patience:
            break
    
    # Final refinement with scipy minimize
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Constraints
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Use best solution from gradient descent as initial point
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    best_h = project_h(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
