import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_objective(h, beta):
    """Smooth approximation to the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    exp_vals = np.exp(beta * corr)
    return (1/beta) * np.log(exp_vals.sum())

def gradient(h, beta, eps=1e-5):
    """Numerical gradient of the smooth objective"""
    grad = np.zeros_like(h)
    base_val = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_val) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction or a simple one

        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        
    # Set up parameters
    dx = 2.0 / n_points
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-5
    max_iter = 1000
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = smooth_objective(h, beta)
    
    h = project_to_feasible(h, n_points)
    
    # Inner loop for optimization
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient and step
        grad = gradient(h, beta)
        new_h = h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)
        new_c5 = smooth_objective(new_h, beta)
        
        # Accept step if improved
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Adapt beta
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.5
            
        # Check stagnation
        if np.abs(best_c5 - smooth_objective(h, beta)) < tolerance:
            break
    
    # Final evaluation with true objective
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Possibly refine resolution
    if time() - start_time < budget_s * 0.8:
        best_h_refined = best_h.copy()
        best_c5_refined = final_c5
        
        # Try increasing resolution
        n_points_refine = min(n_points * 2, 100)
        if n_points_refine > n_points:
            # Interpolate
            old_x = np.linspace(0, 2, n_points)
            new_x = np.linspace(0, 2, n_points_refine)
            h_interp = np.interp(new_x, old_x, best_h_refined)
            h_interp = project_to_feasible(h_interp, n_points_refine)
            c5_refined = evaluate_erdos_solution(h_interp, 2.0 / n_points_refine)
            
            if c5_refined < best_c5_refined:
                best_h_refined = h_interp
                best_c5_refined = c5_refined
                n_points_refine = n_points_refine
                
        best_h = best_h_refined
        final_c5 = best_c5_refined
        n_points = n_points_refine
    
    return (list(best_h), final_c5, n_points)
