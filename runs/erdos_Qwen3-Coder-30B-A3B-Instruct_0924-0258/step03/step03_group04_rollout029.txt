import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def periodic_correlation(h, k, n_points):
    """Compute periodic correlation sum for shift k"""
    if k == 0:
        return 0
    valid_len = n_points - k
    if valid_len <= 0:
        return 0
    corr = np.sum(h[:valid_len] * (1 - h[k:k+valid_len]))
    return corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = 0.5 * np.sin(np.pi * np.arange(n_points) / n_points) + 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    h_last = best_h.copy()
    
    # Early loop with adaptive constraint selection
    n_iter = 0
    while time() - start_time < budget_s * 0.95:
        n_iter += 1
        # Identify worst shifts
        shifts = []
        for k in range(1, n_points):
            corr = periodic_correlation(h_last, k, n_points)
            shifts.append((corr, k))
        shifts.sort(reverse=True)
        worst_shifts = [k for _, k in shifts[:max(5, n_points // 20)]]
        
        # Convex feasibility with only worst shifts
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add specific constraints for worst shifts
        for k in worst_shifts:
            if k == 0:
                continue
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                constraints.append(corr_sum <= best_c5 * dx * valid_len)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_var.value
                if h_feasible is not None:
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    res = minimize(refinement_obj, h_feasible, method='L-BFGS-B', 
                                 bounds=[(0,1) for _ in range(n_points)], 
                                 options={'maxiter': 50, 'disp': False})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                            h_last = h_refined.copy()
                        else:
                            h_last = h_feasible.copy()
                    else:
                        h_last = h_feasible.copy()
                else:
                    h_last = h_start.copy()
            else:
                h_last = h_start.copy()
        except:
            h_last = h_start.copy()
    
    # Final pure gradient refinement  
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'disp': False})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
