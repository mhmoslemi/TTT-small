import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: convex relaxation with refinement
    # First convex relaxation with McCormick bounds
    remaining_time = budget_s * 0.7
    
    # Generate initial solution for convex relaxation
    h_refined = best_h.copy()
    
    # Use the relaxation approach with McCormick bounds to get better start
    while time() - start_time < remaining_time and remaining_time > 1:
        # Build convex model for relaxation
        try:
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # For each possible shift k, add constraint based on correlation
            for k in range(n_points):
                if k == 0:
                    # h[i] * (1 - h[i]) terms
                    overlap = cp.sum(cp.multiply(h_var, 1 - h_var)) * dx
                else:
                    # Sum over valid indices
                    overlap = cp.sum(cp.multiply(h_var[:-k], 1 - h_var[k:])) * dx
                constraints.append(overlap <= 1.0)  # Placeholder, to be updated

            # Solve problem with a simple tightened version
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_var.value
                if h_feasible is not None:
                    h_feasible = project_to_feasible(h_feasible, n_points)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_feasible
        except:
            pass
        remaining_time = budget_s - (time() - start_time)
    
    # Refined optimization with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final validation
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
