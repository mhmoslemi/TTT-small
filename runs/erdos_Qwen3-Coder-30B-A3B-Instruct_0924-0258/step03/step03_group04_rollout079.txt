import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Generate a sinusoidal pattern as initialization
        h_start = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / n_points)) 
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Track the number of iterations for early stopping
    iteration_count = 0
    
    while time() - start_time < budget_s * 0.95:
        iteration_count += 1
        # Identify worst shifts (top 20%)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Only consider non-zero shifts (k != 0)
        shifts = range(-n_points+1, n_points) 
        overlap_values = [corr[n_points-1 + k] for k in shifts if k != 0]
        worst_indices = np.argpartition(overlap_values, -int(0.2 * len(overlap_values)))[-int(0.2 * len(overlap_values)):]
        
        # Constraints on worst shifts only
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        for idx in worst_indices:
            k = shifts[idx + 1] if idx < len(shifts) / 2 else shifts[idx]  # Avoid including index 0
            if k == 0:
                continue
            valid_len = n_points - abs(k)
            if valid_len > 0:
                if k > 0:
                    corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:valid_len]))
                constraints.append(corr_sum <= best_c5 * dx * valid_len)
        
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_var.value
                if h_feasible is not None:
                    # Refine using gradient descent
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    res = minimize(refinement_obj, h_feasible, method='L-BFGS-B', 
                                 bounds=[(0,1) for _ in range(n_points)], 
                                 options={'maxiter': 30})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
        except:
            pass

    # Final refinement with gradient descent 
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
