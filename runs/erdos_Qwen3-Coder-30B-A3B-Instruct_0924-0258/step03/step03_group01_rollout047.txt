import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_max
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for _ in range(swarm_size):
        # Start with initial solution and perturb it
        h_init = h_start.copy()
        if 'initial_h_values' in globals():
            h_init = h_start + np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        val = evaluate_erdos_solution(h_init, dx)
        personal_best.append(h_init.copy())
        personal_best_vals.append(val)
    
    # Global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        inertia_weight = w_min + (w_max - w_min) * (1 - iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel_update = (c1 * r1 * (personal_best[i] - particles[i]) + 
                          c2 * r2 * (global_best - particles[i]))
            velocities[i] = inertia_weight * velocities[i] + vel_update
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = current_val
                
            # Update global best
            if current_val < global_best_val:
                global_best = particles[i].copy()
                global_best_val = current_val
                
        # Local refinement every few iterations
        if iteration % 5 == 0 and iteration > 0:
            refined_count = 0
            for i in range(min(3, swarm_size)):
                h_restart = particles[i].copy()
                h_restart = np.clip(h_restart + np.random.normal(0, 0.01, n_points), 0, 1)
                h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < global_best_val:
                            global_best = h_refined
                            global_best_val = c5_refined
                            refined_count += 1
                except:
                    pass
                    
        if global_best_val < best_c5:
            best_h = global_best.copy()
            best_c5 = global_best_val
    
    return (list(best_h), best_c5, n_points)
