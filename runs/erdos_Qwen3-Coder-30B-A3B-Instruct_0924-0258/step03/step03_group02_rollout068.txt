import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Cluster-based adaptive optimization
    C = 10  # Number of clusters
    max_iter = 200
    stagnation_count = 0
    prev_best = best_c5
    
    # Initialize clusters
    shifts = np.arange(0, n_points)
    overlap_vectors = []
    
    # Precompute overlap structure
    for k in range(n_points):
        if k == 0:
            overlap_vectors.append(np.ones(n_points))
        else:
            v = np.zeros(n_points)
            v[k:] = 1
            overlap_vectors.append(v)
    
    overlap_vectors = np.array(overlap_vectors)
    
    # Run optimization loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update clusters every 20 iterations or if needed
        if iter_num % 20 == 0:
            # Get current overlaps for each shift
            current_overlaps = []
            for k in range(n_points):
                if k == 0:
                    overlap_val = np.sum(best_h * (1 - best_h)) * dx
                else:
                    overlap_val = np.sum(best_h[:-k] * (1 - best_h[k:])) * dx
                current_overlaps.append(overlap_val)
            
            # Use KMeans to group shifts
            kmeans = KMeans(n_clusters=C, random_state=seed + iter_num).fit(
                np.array(current_overlaps).reshape(-1, 1)
            )
            cluster_labels = kmeans.labels_

            cluster_centers = kmeans.cluster_centers_.flatten()

        for cluster_id in range(C):
            # Get indices for this cluster
            cluster_indices = np.where(cluster_labels == cluster_id)[0]
            if len(cluster_indices) == 0:
                continue
                
            # Local optimization on cluster shifts
            # Copy current h
            h_current = best_h.copy()
            
            # Define objective for this cluster
            def cluster_obj(h):
                total_overlap = 0.0
                for k in cluster_indices:
                    if k == 0:
                        overlap = np.sum(h * (1 - h)) * dx
                    else:
                        overlap = np.sum(h[:-k] * (1 - h[k:])) * dx
                    total_overlap = max(total_overlap, overlap)
                return total_overlap
            
            # Gradient-based optimization for this cluster
            def objective(h):
                return cluster_obj(h)
                
            def constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint}

            return evaluate_erdos_solution(h, dx)

        try:
            res = minimize(final_obj, h_final, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
            
    return (list(best_h), best_c5, n_points)
