import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with a periodic pattern, ensure sum = n_points/2
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    h = h_start.copy()
    max_iter = 10000
    eta = 0.5
    eta_decay = 0.999
    tol = 1e-6
    no_improve_count = 0
    max_no_improve = 500
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the shift that maximizes overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Find the gradient direction for that shift
        # We compute the gradient of the objective function
        # for the shift that gave the maximum overlap
        shift = k_max - (n_points - 1)
        # Build the h shifted for that shift
        h_shifted = np.roll(h, shift)
                
        # Linear approximation of the gradient
        grad = 1 - 2 * h  # This is the gradient d/dh of integral h*(1-h)
        
        # Solve linear subproblem
        # We want to find the extreme point that minimizes the gradient
        # For Frank-Wolfe: we want to minimize grad^T h over extreme points
        # since we are minimizing over binary vectors with exactly n/2 ones.
        # The gradient is 1 - 2 * h where h implies that optimal extreme  point
        # will assign 1 to indices with largest (1 - 2 * h)
        # conceptually: indices for which (1 - 2 * h) > 0 are more beneficial to set to 1
        
        # In practice, we construct the binary vector that minimizes the linear objective
        indices_sorted = np.argsort(1 - 2 * h)[::-1]
        h_extreme = np.zeros(n_points)
        h_extreme[indices_sorted[:n_points//2]] = 1.0
        
        # Update step
        h_new = (1 - eta) * h + eta * h_extreme
        h = project_to_feasible(h_new, n_points)
        
        # Adaptive step size
        eta *= eta_decay
        
        # Evaluate current solution
        c5_new = evaluate_erdos_solution(h, dx)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                break
    
    # Final optimization with scipy if possible
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h_vec):
            h_proj = project_to_feasible(h_vec, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Near-optimal starting point, try to refine
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
