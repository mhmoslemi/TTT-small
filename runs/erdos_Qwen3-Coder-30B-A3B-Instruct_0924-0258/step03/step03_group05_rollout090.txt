import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try higher resolution if time permits
    if budget_s > 400:
        n_points = 70
        dx = 2.0 / n_points
        # Start with scaled version of previous solution
        h_start = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, len(h_start)), 
                           h_start)
    
    # Initialize h to satisfy constraint ∫ h dx = 1
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define the dual optimization
    def solve_feasibility(h_guess, c_val, n_points, dx):
        # Define variables
        h = cp.Variable(n_points)
        # Constraints for h
        constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1]
        
        # Add overlap constraints for all possible shifts
        for k in range(n_points):  # Each shift k from 0 to n_points-1
            if k == 0:
                continue  # Skip shift 0 (identity)
            
            # For shift k: integral of h(x) * (1 - h(x+k)) dx
            # Valid indices for convolution
            valid_len = n_points - k
            if valid_len <= 0:
                break
            
            # Sum-form of the overlap integral
            overlap_sum = 0
            for i in range(valid_len):
                overlap_sum += h[i] * (1 - h[i + k])
            
            # Add constraint: overlap_sum * dx <= c_val
            overlap_constraint = overlap_sum * dx <= c_val
            constraints.append(overlap_constraint)
        
        # Solve the feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, max_iters=1000, feastol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            # Try again with different solver
            try:
                prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    return h.value
            except:
                pass
        return None
    
    # Iterative refinement with binary search on C5
    c_low = 0.370
    c_high = 0.381
    tol = 1e-5
    max_iter = 20
    
    # Perturbation parameters
    pert_magnitude = 0.02
    max_perturb_trials = 20
    
    # Start with a coarse search to find a starting C5
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        h_feasible = solve_feasibility(best_h, c_mid, n_points, dx)

            c_low = c_mid
            best_h = h_feasible

                return evaluate_erdos_solution(h_vec, dx)
            
            # Constraints
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:

        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Renormalize
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate new solution
        try:
            current_c5 = evaluate_erdos_solution(h_perturbed, dx)
            if current_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = current_c5
        except:
            pass
    
    # Final local improvement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Validate solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
