import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent approach
    max_iterations = 500
    tolerance = 1e-6
    prev_c5 = float('inf')
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Random order for variable updates
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for i in indices:
            if time() - start_time > budget_s * 0.9:
                break
                
            # Save current h
            h_current = best_h.copy()
            
            # Compute current overlap contributions
            old_overlap = 0.0
            for k in range(n_points):
                if k == 0:
                    continue
                # Convolve h with 1-h with shift k
                if k < n_points:
                    overlap = np.sum(h_current[:-k] * (1 - h_current[k:]))
                else:
                    overlap = 0
                old_overlap = max(old_overlap, overlap * dx)
            
            # One-dimensional search for optimal h[i]
            best_h_i = h_current[i]
            best_overlap = old_overlap
            
            # Simple line search around current value
            search_range = np.linspace(max(0, h_current[i] - 0.1), 
                                     min(1, h_current[i] + 0.1), 50)
            
            for h_i_try in search_range:
                h_temp = h_current.copy()
                h_temp[i] = h_i_try
                
                # Project to feasible region
                h_projected = project_to_feasible(h_temp, n_points)
                
                # Evaluate this candidate
                c5_try = evaluate_erdos_solution(h_projected, dx)
                if c5_try < best_overlap:
                    best_overlap = c5_try
                    best_h_i = h_i_try
                    
            # Update h[i] if improved
            if abs(best_h_i - h_current[i]) > 1e-8:
                best_h[i] = best_h_i
                improved = True
                
        # Project to feasible after all updates
        best_h = project_to_feasible(best_h, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_erdos_solution(best_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            
        # Check for convergence
        if abs(prev_c5 - current_c5) < tolerance:
            break
            
        prev_c5 = current_c5
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Use scipy to fine-tune
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        # Optimize with bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
