import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

            h_start[i] = 0.5 + 0.2 * np.sin(i * 2 * np.pi / n_points)
        h_start = project_to_feasible(h_start, n_points)

    best_c5 = float('inf')
    best_h = None
    
    # Try to find a better solution by optimizing the current construction
    # Use a simple gradient descent approach with projection
    h = h_start.copy()
    dx = 2.0 / n_points

    if current_c5 < best_c5:

    # We'll use a smoothing approximation to make the objective more manageable

    decay_rate = 0.995

        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
