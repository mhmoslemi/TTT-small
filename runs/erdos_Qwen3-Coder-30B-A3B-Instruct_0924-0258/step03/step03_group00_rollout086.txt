import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction: 50 points
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    pop_size = 500
    generations = 100
    elite_size = 50
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        # Create piecewise constant step function
        h = np.random.rand(n_points)
        # Normalize to sum = 1
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return np.clip(h, 0, 1)
    
    def create_individual_binary():
        # Create binary-valued function (0 or 1)
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
        # Adjust to make sum = 1
        h_sum = np.sum(h)
        if h_sum == 0:
            h[0] = 1
        elif h_sum > 1:
            # Turn some 1s to 0s
            ones_indices = np.where(h == 1)[0]
            to_zero = int(h_sum - 1)
            zero_indices = np.random.choice(ones_indices, to_zero, replace=False)
            h[zero_indices] = 0
        elif h_sum < 1:
            # Turn some 0s to 1s
            zeros_indices = np.where(h == 0)[0]
            to_one = int(1 - h_sum)
            one_indices = np.random.choice(zeros_indices, to_one, replace=False)
            h[one_indices] = 1
        return h
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        return child1, child2
    
    def mutate(individual):
        # Randomly change some values
        for i in range(n_points):
            if random.random() < mutation_rate:
                # Randomly flip to 0 or 1 if it's binary
                individual[i] = np.random.rand()
        # Normalize again
        h_sum = np.sum(individual)
        if h_sum > 0:
            individual = individual * (n_points / 2) / h_sum
        return np.clip(individual, 0, 1)
    
    # Initialize the population
    population = [create_individual() for _ in range(pop_size)]
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            c5 = evaluate_erdos_solution(individual, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection (tournament)
            parent1_idx = random.randint(0, elite_size - 1)
            parent2_idx = random.randint(0, elite_size - 1)
            
            child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Truncate to population size
        population = new_population[:pop_size]
    
    # Local refinement with SLSQP
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try a few refinements with different starting points
    for i in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
        
        # Start from best individual
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
