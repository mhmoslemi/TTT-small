import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to store recently visited configurations
    tabu_list = []
    tabu_size = 10
    
    # Simulated annealing
    T = 1.0
    cooling_rate = 0.999
    min_T = 1e-6
    
    # Optimization loop
    max_iter = 50000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Create a neighbor solution
        h_new = best_h.copy()
        # Choose a random segment to modify
        idx = np.random.randint(0, n_points)
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1)
        h_new[idx] += perturbation
        # Clip to [0, 1]
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Ensure integral constraint
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        
        # Check tabu list
        h_tuple = tuple(h_new.round(6))
        if h_tuple in tabu_list:
            # Accept with small probability
            if np.random.rand() > 0.1:
                continue
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Add to tabu list
            tabu_list.append(h_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            # Accept with probability based on temperature
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T):
                best_h = h_new.copy()
                best_c5 = c5_new
                # Add to tabu list
                tabu_list.append(h_tuple)
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        
        # Cool down temperature
        T = max(min_T, T * cooling_rate)
    
    # Final local refinement using scipy minimize
    if (time.time() - start_time) < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Run multiple local optimizations
        for _ in range(20):
            if (time.time() - start_time) > budget_s * 0.95:
                break
            # Start from current best with small random perturbation
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
