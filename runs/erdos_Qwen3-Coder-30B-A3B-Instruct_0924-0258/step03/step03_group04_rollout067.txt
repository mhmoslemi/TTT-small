import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a sinusoidal pattern
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    max_iter = 10000
    iter_count = 0
    
    # Initial neighbor size 
    neighbor_size = max(1, n_points // 20)
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Generate candidate by perturbing some points
        h_candidate = best_h.copy()
        
        # Select random points to perturb
        indices = random.sample(range(n_points), min(neighbor_size, n_points))
        
        # Perturb these points
        for i in indices:
            # Small random change
            change = np.random.normal(0, 0.05)
            h_candidate[i] += change
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            # Adjust neighbor size if improvement is large
            if delta < -1e-4:
                neighbor_size = max(1, min(n_points // 4, neighbor_size + 1))
            else:
                neighbor_size = max(1, neighbor_size - 1)
        
        # Update temperature
        temperature *= cooling_rate
        if temperature < min_temp:
            temperature = min_temp
            
        iter_count += 1
        
        # Occasionally refine the best solution
        if iter_count % 500 == 0:
            def refinement_obj(h):
                h_proj = project_to_feasible(h, n_points)
                return evaluate_erdos_solution(h_proj, dx)
            
            res = minimize(refinement_obj, best_h, method='L-BFGS-B', 
                         bounds=[(0,1) for _ in range(n_points)], 
                         options={'maxiter': 20})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined

    # Final refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
