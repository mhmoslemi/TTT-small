import numpy as np
from scipy.optimize import minimize
from time import time
import itertools

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create structured initial guess with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            for i in range(start, end):
                h_start[i] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n_points / 2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    
    # Store best solution and its value
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Nelder-Mead parameters
    population_size = 15
    max_iter = 100
    
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def projection(h):
        """Project h to feasible region"""
        h = np.clip(h, 0, 1)
        # Adjust to meet constraint sum(h) = n_points/2
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h

    # Generate initial population
    def generate_population(n_pop, h_base):
        pop = [h_base.copy()]
        for _ in range(n_pop - 1):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = projection(h)
            pop.append(h)
        return pop

    def nelder_mead_step(pop, obj_func):
        # Sort population by objective value
        pop_with_vals = [(h, obj_func(h)) for h in pop]
        pop_with_vals.sort(key=lambda x: x[1])
        
        # Best and worst
        best_h = pop_with_vals[0][0]
        worst_h = pop_with_vals[-1][0]
        second_worst_h = pop_with_vals[-2][0]
        
        # Compute centroid (excluding worst)
        centroid = np.mean([h for h, _ in pop_with_vals[:-1]], axis=0)
        
        # Reflection
        r = 1.0
        reflected_h = centroid + r * (centroid - worst_h)
        reflected_h = projection(reflected_h)
        reflected_val = obj_func(reflected_h)
        
        # If reflected is better than best, expand
        if reflected_val < pop_with_vals[0][1]:
            e = 2.0
            expanded_h = centroid + e * (reflected_h - centroid)
            expanded_h = projection(expanded_h)
            expanded_val = obj_func(expanded_h)
            if expanded_val < reflected_val:
                pop_with_vals[-1] = (expanded_h, expanded_val)
            else:
                pop_with_vals[-1] = (reflected_h, reflected_val)
        elif reflected_val < pop_with_vals[-2][1]:
            pop_with_vals[-1] = (reflected_h, reflected_val)
        else:
            # Contract or shrink
            if reflected_val < pop_with_vals[-1][1]:
                c = 0.5
                contracted_h = centroid + c * (reflected_h - centroid)
            else:
                c = 0.5
                contracted_h = centroid + c * (worst_h - centroid)
            contracted_h = projection(contracted_h)
            contracted_val = obj_func(contracted_h)
            pop_with_vals[-1] = (contracted_h, contracted_val)
        
        # Re-sort
        pop_with_vals.sort(key=lambda x: x[1])
        return [h for h, _ in pop_with_vals]
    
    # Initialize population
    population = generate_population(population_size, h_start)

    # Main loop
    iteration = 0
    last_improvement_iter = 0
    patience = 10
    
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        iteration += 1
        population = nelder_mead_step(population, obj_func)
        
        # Check if improvement was made
        new_best_c5 = obj_func(population[0])
        if new_best_c5 < best_c5:
            best_c5 = new_best_c5
            best_h = population[0].copy()
            last_improvement_iter = iteration
            
        # Early stopping if no improvement for a while
        if iteration - last_improvement_iter > patience:
            break
            
    # Refinement using scipy minimize on top candidates
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Select top 3 candidates
        top_candidates = population[:3]
        
        for h_restart in top_candidates:
            if time() - start_time > budget_s * 0.95:
                break
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj_func, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj_func(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
                
    return (list(best_h), best_c5, n_points)
