import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    random.seed(seed)
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_generations = 1000
    stagnation_limit = 100
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = np.random.permutation(initial_h_values)
        else:
            h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_fitness_history = []
    stagnation_count = 0
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitnesses.append(fitness)
        
        # Track best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count >= stagnation_limit:
            # Reset stagnation with new random population
            for i in range(population_size):
                h = np.random.rand(n_points)
                h = project_to_feasible(h, n_points)
                population[i] = h
            stagnation_count = 0
        
        # Create new population
        new_population = []
        for i in range(population_size):
            # Select three different individuals
            idxs = random.sample(range(population_size), 3)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < crossover_prob
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(0, n_points)] = True
            trial = np.where(crossover_mask, mutant, population[i])
            
            # Ensure feasibility and add to new population
            trial = project_to_feasible(trial, n_points)
            new_population.append(trial)
        
        population = new_population
    
    # Final refinement with SLSQP
    refinement_time = budget_s - (time() - start_time)
    if refinement_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h.copy(), method='SLSQP',

                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
