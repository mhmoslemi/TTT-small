import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_max
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        inertia_weight = w_min + (w_max - w_min) * (max_iter - iteration) / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] 
                           + c1 * r1 * (pbest[i] - swarm[i]) 
                           + c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            curr_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if curr_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = curr_val
            
            # Update global best
            if curr_val < gbest_val:
                gbest = swarm[i].copy()
                gbest_val = curr_val
                
        # Check if global best improved
        if gbest_val < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_val
                
        # Local refinement every 5 iterations
        if iteration % 5 == 0 and iteration > 0:
            for j in range(min(3, swarm_size)):
                idx = np.argsort(pbest_vals)[:3][j]
                h_refined = pbest[idx]
                h_restart = h_refined + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                 bounds=bounds, constraints=cons_sum, 
                                 options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        h_final = res.x
                        c5_final = objective(h_final)
                        if c5_final < best_c5:
                            best_h = h_final
                            best_c5 = c5_final
                except:
                    pass
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
