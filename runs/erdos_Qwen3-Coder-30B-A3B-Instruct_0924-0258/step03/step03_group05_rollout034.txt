import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 50
    num_generations = 1000
    mutation_rate = 0.2
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize sum to n_points/2
        return np.clip(h, 0, 1)
    
    def create_population():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            population.append(create_individual())
        return population
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def selection(population):
        tournament_size = 5
        selected = []
        for _ in range(population_size):
            tournament = random.sample(population, tournament_size)
            winner = min(tournament, key=fitness)
            selected.append(winner.copy())
        return selected
    
    def crossover(parent1, parent2):
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Renormalize
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return np.clip(child1, 0, 1), np.clip(child2, 0, 1)
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.05)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # Main GA Loop
    population = create_population()
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and generation < num_generations:
        # Evaluate all individuals
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best individual
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_c5 = fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Selection
        selected = selection(population)
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        generation += 1
    
    # Final fine-tuning with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
