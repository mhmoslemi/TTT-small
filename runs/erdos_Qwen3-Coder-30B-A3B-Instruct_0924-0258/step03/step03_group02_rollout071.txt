import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate Descent approach
    max_iterations = 500
    tolerance = 1e-6
    max_iter_time = budget_s * 0.7  # 70% for coordinate descent
    
    h_current = best_h.copy()
    
    for iteration in range(max_iterations):
        if time() - start_time > max_iter_time:
            break
            
        # Random order update
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for i in indices:
            if time() - start_time > max_iter_time:
                break
                
            h_prev = h_current.copy()
            
            # One-dimensional search for optimal h[i]
            best_h_i = h_current[i]
            best_c5_i = best_c5
            
            # Grid search for h[i] in [0,1]
            grid = np.linspace(0, 1, 100)
            for val in grid:
                h_current[i] = val
                # Project to maintain constraints
                h_current = project_to_feasible(h_current, n_points)
                
                c5 = evaluate_erdos_solution(h_current, dx)
                if c5 < best_c5_i:
                    best_c5_i = c5
                    best_h_i = val
                    
            # Update the value
            h_current[i] = best_h_i
            
            # Project to maintain constraints
            h_current = project_to_feasible(h_current, n_points)
            
            if best_c5_i < best_c5:
                improved = True
                best_c5 = best_c5_i
                best_h = h_current.copy()
                
        # Early stopping if no improvement
        if not improved and iteration > 10:
            break
    
    # Refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
