import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier parametrization approach
    # Use cosine basis functions to represent h(x) in a smooth way
    d = 5  # Number of cosine terms
    
    def h_from_params(a):
        # a: array of coefficients [a0, a1, ..., ad]
        x = np.linspace(0, 2, n_points)
        g = a[0] + sum(a[k+1] * np.cos((k+1) * np.pi * x) for k in range(d))
        h = 1.0 / (1.0 + np.exp(-g))
        return h
    
    def obj_func(a):
        h = h_from_params(a)
        # Normalize h to satisfy the integral constraint
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h = h * (1.0 / h_sum)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(a):
        h = h_from_params(a)
        h_sum = np.sum(h) * dx
        return h_sum - 1.0
    
    # Optimization bounds for coefficients
    bounds = [(-10.0, 10.0) for _ in range(d+1)]
    
    # Random restarts
    num_restarts = 20
    best_a = None
    best_c5_local = float('inf')
    
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        a0 = np.random.uniform(-2, 2, d+1)
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj_func, a0, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5_local:
                    best_c5_local = c5_test
                    best_a = res.x.copy()
        except:
            continue
    
    # Final refinement
    if best_a is not None and time.time() - start_time < budget_s * 0.95:
        # Apply the final solution
        h_final = h_from_params(best_a)
        h_sum = np.sum(h_final) * dx
        if h_sum > 0:
            h_final = h_final * (1.0 / h_sum)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final.copy()
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
