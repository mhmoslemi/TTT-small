import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_cluster(boundary_points, initial_h_values, n_points):
    """Cluster segments with adaptive boundary placement"""
    # Create clusters based on initial h
    h_seg = np.array(initial_h_values)
    dx = 2.0 / n_points
    
    # Identify significant jumps
    thresholds = [0.1, 0.2, 0.3]
    segments = []
    last_idx = 0

    for t in thresholds:
        for i in range(1, len(h_seg) - 1):
            if abs(h_seg[i] - h_seg[i-1]) > t or abs(h_seg[i+1] - h_seg[i]) > t:
                if i > last_idx + 2:  # Prevent too small segments
                    segments.append(i)
                    last_idx = i
        if segments:
            break
    
    # Evenly distribute boundaries
    if not segments:
        segments = [i for i in range(5, n_points-5, 5)]
    
    segments = [0] + segments + [n_points]
    return np.array(segments) * dx

def get_segment_data(x_boundaries, h_values):
    """Get segment widths and heights"""
    widths = np.diff(x_boundaries)
    heights = h_values
    return widths, heights

def compute_objective_with_buffer(h_values, x_boundaries, beta=100):
    """Differentiable approximation of max correlation with buffer"""
    h = np.array(h_values)
    corr = np.correlate(h, 1-h, mode='full')  # This will be our "correlation data"
    return np.log(np.sum(np.exp(beta * corr))) / beta

def project_segments(heights, widths):
    """Ensure 0 <= heights <= 1 and integral = 1"""
    total_width = np.sum(widths)
    total_area = np.sum(heights * widths)
    if total_area > 0:
        heights = heights * (1.0 / total_area)
    heights = np.clip(heights, 0, 1)
    return heights

def update_bounds(boundaries, step_size, idx):
    """Update one boundary, maintaining order"""
    new_bounds = boundaries.copy()
    new_bounds[idx] += step_size
    # Keep within bounds
    if idx > 0 and new_bounds[idx] < new_bounds[idx-1]:
        new_bounds[idx] = new_bounds[idx-1] + 1e-10
    if idx < len(new_bounds) - 1 and new_bounds[idx] > new_bounds[idx+1]:
        new_bounds[idx] = new_bounds[idx+1] - 1e-10
    return new_bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with existing construction or create one
    if 'initial_h_values' in globals():
        orig_h = np.array(initial_h_values)
        n_points = len(orig_h)
    else:
        orig_h = np.zeros(35) + 0.5
        n_points = 35
        
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(orig_h, dx)
    best_h = orig_h.copy()
    best_n_points = n_points
    
    # Hyperparameters
    max_iter = 1000
    initial_refinements = 20
    refinement_interval = 25
    
    # Segment optimization approach
    x_boundaries = np.linspace(0, 2, n_points+1)
    seg_widths = np.diff(x_boundaries)  # segment widths
    seg_heights = np.copy(orig_h)
    
    # Normalize to integral = 1
    total_area = np.sum(seg_heights * seg_widths)
    seg_heights = seg_heights * (1.0 / total_area)
    
    last_improvement = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # optimize heights
        step_size = 0.01
        grad_h = np.zeros_like(seg_heights)
        eps = 1e-5
        base_score = evaluate_erdos_solution(seg_heights, dx)
        
        for i in range(len(seg_heights)):
            h_plus = seg_heights.copy()
            h_plus[i] += eps
            h_plus = project_segments(h_plus, seg_widths)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad_h[i] = (score_plus - base_score) / eps
            
        # Update heights
        seg_heights -= step_size * grad_h
        seg_heights = project_segments(seg_heights, seg_widths)
        
        # optimize boundaries
        grad_b = np.zeros_like(x_boundaries)
        eps = 1e-5
        base_score2 = evaluate_erdos_solution(seg_heights, dx)
        
        for i in range(1, len(x_boundaries)-1):
            x_plus = x_boundaries.copy()
            x_plus[i] += eps
            x_plus = update_bounds(x_plus, eps, i)
            h_trial = np.interp(np.linspace(0, 2, n_points), x_plus[:-1] + np.diff(x_plus) / 2, seg_heights)
            h_trial = project_to_feasible(h_trial, n_points)
            score_plus = evaluate_erdos_solution(h_trial, dx)
            grad_b[i] = (score_plus - base_score2) / eps
            
        # Update boundaries
        x_boundaries[1:-1] -= step_size * grad_b[1:-1]
        x_boundaries = np.clip(x_boundaries, 0, 2)
        x_boundaries = np.sort(x_boundaries)
        
        # Recompute widths and heights
        seg_widths = np.diff(x_boundaries)
        seg_heights = np.interp(np.linspace(0, 2, n_points), x_boundaries[:-1] + np.diff(x_boundaries) / 2, seg_heights)
        seg_heights = project_segments(seg_heights, seg_widths)
        
        # Evaluate
        candidate_h = project_to_feasible(seg_heights, n_points)
        score = evaluate_erdos_solution(candidate_h, dx)
        if score < best_c5:
            best_c5 = score
            best_h = candidate_h.copy()
            last_improvement = iteration
    
    # Final refinement if needed
    if time() - start_time < budget_s * 0.9 and last_improvement != iteration:
        for _ in range(25):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros_like(seg_heights)
            eps = 1e-5
            base_score = evaluate_erdos_solution(seg_heights, dx)
            for i in range(len(seg_heights)):
                h_plus = seg_heights.copy()
                h_plus[i] += eps
                h_plus = project_segments(h_plus, seg_widths)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (score_plus - base_score) / eps
                
            step_size = 0.01
            seg_heights -= step_size * grad
            seg_heights = project_segments(seg_heights, seg_widths)
            
            candidate_h = project_to_feasible(seg_heights, n_points)
            score = evaluate_erdos_solution(candidate_h, dx)
            if score < best_c5:
                best_c5 = score
                best_h = candidate_h.copy()
    
    return (list(best_h), best_c5, best_n_points)
