import numpy as np
import time
from scipy.optimize import minimize
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    max_iter = 10000
    tabu_length = 10
    tabu_list = deque(maxlen=tabu_length)
    
    # Start with initial configuration
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.9995
    min_temp = 1e-6
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Make a random perturbation
        idx = np.random.randint(0, n_points)
        old_value = current_h[idx]
        
        # Try different perturbations
        perturbations = [0.0, 1.0, 0.5]
        new_value = old_value
        if old_value not in perturbations:
            new_value = np.random.choice(perturbations)
        else:
            # Choose from other options
            candidates = [x for x in perturbations if x != old_value]
            new_value = np.random.choice(candidates)
            
        current_h[idx] = new_value
        
        # Ensure sum constraint
        h_sum = np.sum(current_h)
        if h_sum > 0:
            current_h = current_h * (n_points / 2) / h_sum
        
        # Evaluate new overlap
        new_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5:
            current_c5 = new_c5
            # Update best if better
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Accept with probability based on temperature (if not in tabu)
            if temp > min_temp:
                delta = new_c5 - current_c5
                accept_prob = np.exp(-delta / temp)
                if np.random.rand() < accept_prob:
                    current_c5 = new_c5
                else:
                    # Revert
                    current_h[idx] = old_value
                    # Ensure sum constraint again
                    h_sum = np.sum(current_h)
                    if h_sum > 0:
                        current_h = current_h * (n_points / 2) / h_sum
            else:
                # Revert if we're below minimum temperature
                current_h[idx] = old_value
                # Ensure sum constraint again
                h_sum = np.sum(current_h)
                if h_sum > 0:
                    current_h = current_h * (n_points / 2) / h_sum
        
        # Update temperature
        temp *= cooling_rate
        
        # Keep track of visited configurations to avoid cycling
        config_key = tuple(current_h.round(6))
        if config_key in tabu_list:
            # If we've been here before, we might want to restart
            pass
        else:
            tabu_list.append(config_key)
    
    # Final refinement with local optimization if time allows
    if time.time() - start_time < budget_s * 0.98:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations with different starting points
        num_restarts = 10
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.98:
                break

            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
