import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_iterations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        init_h = project_to_feasible(h_start, n_points)
        population[0] = init_h
    
    last_improvement = 0
    iteration = 0
    
    while time() - start_time < budget_s * 0.85 and iteration < max_iterations:
        # Evaluate all candidates in the population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best solution so far
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Stop if no improvement for a while
        if last_improvement > 100:
            break
        
        # Create next generation
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < crossover_prob
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n_points)] = True
            
            trial = np.where(crossover_mask, mutant, population[i])
            
            # Ensure feasibility
            trial = project_to_feasible(trial, n_points)
            new_population.append(trial)
        
        population = new_population
        iteration += 1
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Extract the best solution so far
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
