import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    n_clusters = 10
    max_iter = 200
    max_no_improve = 100
    recluster_freq = 50
    local_steps = 20
    convex_steps = 30
    
    # Initialize clusters
    shifts = np.arange(0, 2 * n_points - 1)
    overlap_vectors = []
    
    # Precompute overlap vectors for each k
    for k in range(2 * n_points - 1):
        if k < n_points:
            v = np.concatenate([np.zeros(k), np.ones(n_points - k)])
        else:
            v = np.concatenate([np.ones(k - n_points + 1), np.zeros(2 * n_points - 1 - k)])
        overlap_vectors.append(v)
    
    overlap_vectors = np.array(overlap_vectors)
    
    # K-means clustering
    kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
    cluster_labels = kmeans.fit_predict(overlap_vectors)
    
    h_current = best_h.copy()
    
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update clusters every recluster_freq iterations
        if iteration % recluster_freq == 0 and iteration > 0:
            # Recompute overlap vectors based on current h
            current_overlap_vectors = []
            for k in range(2 * n_points - 1):
                # Compute overlap for shift k
                if k <= n_points - 1:
                    overlap = np.sum(h_current[:n_points-k] * (1 - h_current[k:]))
                else:
                    overlap = np.sum(h_current[k-n_points+1:] * (1 - h_current[:2*n_points-k-1]))
                current_overlap_vectors.append(overlap)
            
            # Update cluster assignment using current overlaps
            kmeans.fit_predict(np.array(current_overlap_vectors).reshape(-1,1))
            cluster_labels = kmeans.labels_
        
        # Optimizing each cluster
        for cluster_id in range(n_clusters):
            cluster_indices = np.where(cluster_labels == cluster_id)[0]
            
            # Local search for this cluster
            h_old = h_current.copy()
            for _ in range(local_steps):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Gradient approximation of overlap for this cluster
                grad = np.zeros(n_points)
                
                # Compute gradients for overlapping shifts
                for k_idx in cluster_indices:
                    k = k_idx - (n_points - 1)
                    if k < 0:
                        # h[i] * (1 - h[i-k]) for i = 0..n_points+k-1
                        for i in range(n_points + k):
                            grad[i] += 1 - h_current[i - k]
                            grad[i - k] -= h_current[i]
                    else:
                        # h[i+k] * (1 - h[i]) for i = 0..n_points-k-1
                        for i in range(n_points - k):
                            grad[i] -= h_current[i + k]
                            grad[i + k] += 1 - h_current[i]
                
                # Gradient descent step
                step_size = 0.01
                h_current = h_current - step_size * grad
                h_current = project_to_feasible(h_current, n_points)
            
            # Convex relaxation for this cluster
            for _ in range(convex_steps):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Relaxation using convex hull
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Add constraints from current cluster
                for k_idx in cluster_indices:
                    k = k_idx - (n_points - 1)
                    if k < 0:
                        # Convex constraints for negative shifts
                        for i in range(n_points + k):
                            # h[i] * (1 - h[i-k]) <= z[i]
                            z = cp.Variable()
                            constraints.append(h_var[i] + z <= 1)
                            constraints.append(h_var[i-k] <= 1)
                            constraints.append(z >= h_var[i] - h_var[i-k])
                            constraints.append(z >= 0)
                    else:
                        # Convex constraints for positive shifts
                        for i in range(n_points - k):
                            z = cp.Variable()
                            constraints.append(h_var[i] + z <= 1)
                            constraints.append(h_var[i+k] <= 1)
                            constraints.append(z >= h_var[i] - h_var[i+k])
                            constraints.append(z >= 0)
                
                prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
                try:
                    prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6)
                    if prob.status == cp.OPTIMAL and h_var.value is not None:
                        h_current = h_var.value
                        h_current = project_to_feasible(h_current, n_points)
                except:
                    pass
                    
        # Evaluate
        c5 = evaluate_erdos_solution(h_current, dx)
        if c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= max_no_improve:
            break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
