import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iterations = 2000
    patience = 100
    patience_counter = 0
    
    # Normalize initial solution
    h_norm = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(h_norm, 0, 1)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def compute_gradient(h_vec, dx):
        n = len(h_vec)
        grad = np.zeros(n)
        # For each position, compute derivative w.r.t. h_i
        # This needs to compute gradient of max_k Corr_k w.r.t. h
        # We'll do this via finite differences for simplicity
        
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_k_idx = np.argmax(corr)
        max_correlation = corr[max_k_idx]
        
        # For small perturbations
        eps = 1e-6
        for i in range(n):
            h_pert = h_vec.copy()
            h_pert[i] += eps
            if h_pert[i] > 1:
                h_pert[i] = 1
            elif h_pert[i] < 0:
                h_pert[i] = 0
                
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            max_k_idx_pert = np.argmax(corr_pert)
            max_correlation_pert = corr_pert[max_k_idx_pert]
            
            grad[i] = (max_correlation_pert - max_correlation) / eps
        
        return grad
    
    def project_and_normalize(h_vec):
        # Project to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Normalize to sum to n_points/2
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec
    
    # Main optimization loop
    for iter_num in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient (simplified version)
        current_grad = compute_gradient(best_h, dx)
        
        # Update using adaptive learning rate
        lr = 1e-3 * (1.0 - iter_num / max_iterations)
        new_h = best_h - lr * current_grad
        
        # Project and normalize
        new_h = project_and_normalize(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Add random exploration periodically to escape local minima
        if iter_num % 50 == 0 and patience_counter > 10:
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            new_h = best_h + perturbation
            new_h = project_and_normalize(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                patience_counter = 0
        
        # Early stopping if no improvement
        if patience_counter > patience:
            break
    
    # Final refinement with scipy optimizer
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Ensure final bounds
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    # Local optimization
    try:
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
