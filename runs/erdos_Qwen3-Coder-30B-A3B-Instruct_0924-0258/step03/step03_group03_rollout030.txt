import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(arr, smoothing=0.1):
    """Numerically stable smooth approximation of maximum"""
    return np.log(np.sum(np.exp(arr / smoothing))) * smoothing

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if provided
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points
    
    # Optimizer parameters
    max_iter = 1000
    trust_radius = 0.1
    trust_decay = 0.5
    max_trust_iter = 20
    learning_rate = 0.01
    max_inner_iter = 100
    
    h_current = best_h.copy()
    dx_current = 2.0 / n_points
    
    # Trust region optimization
    for trust_iter in range(max_trust_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create a more structured initial proposal
        if trust_iter == 0:
            # Start with piecewise constant structure
            h_new = h_current.copy()
            # Apply small random perturbations with some structure
            perturbation = np.random.normal(0, 0.05, n_points)
            h_new += perturbation
            h_new = project_to_feasible(h_new, n_points)
                
            # Use gradient-based refinement
            h_refined = h_new.copy()
            prev_score = evaluate_erdos_solution(h_refined, dx_current)
            
            for inner_iter in range(max_inner_iter):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Finite difference gradient approximation
                grad = np.zeros_like(h_refined)
                eps = 1e-5
                
                for i in range(len(h_refined)):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    score_plus = evaluate_erdos_solution(h_plus, dx_current)
                    grad[i] = (score_plus - prev_score) / eps
                    
                # Update
                h_refined = h_refined - learning_rate * grad
                h_refined = project_to_feasible(h_refined, n_points)
                
                new_score = evaluate_erdos_solution(h_refined, dx_current)
                if new_score < prev_score:
                    prev_score = new_score
                    h_current = h_refined.copy()
                else:
                    break
                    
        else:
            # Random refinement
            h_refined = h_current + np.random.normal(0, trust_radius, n_points)
            h_refined = project_to_feasible(h_refined, n_points)
            
        current_score = evaluate_erdos_solution(h_refined, dx_current)
        
        if current_score < best_c5:
            best_h = h_refined.copy()
            best_c5 = current_score
            best_n_points = n_points
            
        # Decrease trust region if no improvement
        if current_score >= best_c5:
            trust_radius *= trust_decay
            
    # Final precise refinement
    if time() - start_time < budget_s * 0.9:
        h_fine = project_to_feasible(best_h, n_points)
        best_score = best_c5
        
        # Projected gradient descent
        for _ in range(200):
            if time() - start_time > budget_s * 0.95:
                break
                
            grad = np.zeros_like(h_fine)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_fine, dx_current)
            
            for i in range(len(h_fine)):
                h_plus = h_fine.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx_current)
                grad[i] = (score_plus - base_score) / eps

            # Update with gradient
            step_size = 0.005
            h_fine = h_fine - step_size * grad
            h_fine = project_to_feasible(h_fine, n_points)
            
            score_fine = evaluate_erdos_solution(h_fine, dx_current)
            if score_fine < best_score:
                best_score = score_fine
                best_h = h_fine.copy()
                
    return (list(best_h), best_c5, best_n_points)
