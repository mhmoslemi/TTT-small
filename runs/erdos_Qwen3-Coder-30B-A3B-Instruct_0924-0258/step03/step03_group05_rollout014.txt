import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try several optimization strategies
    strategies = []
    
    # Strategy 1: Direct optimization with bounds
    def strategy_1():
        h0 = best_h.copy()
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Constraints for integration and bounds
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize
        try:
            res = minimize(objective, h0, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_val = evaluate_erdos_solution(h_refined, dx)
                return h_refined, c5_val
        except:
            pass
        return h0, best_c5
    
    # Strategy 2: CVX feasibility with iterative reduction
    def strategy_2():
        def solve_feasibility(h_guess, c_val):
            h = cp.Variable(n_points)
            constraints = [
                h >= 0,
                h <= 1,
                cp.sum(h) * dx == 1
            ]
            
            # Add overlap constraints for all shifts
            for k in range(1, n_points):  # skip k=0
                valid_len = n_points - k
                if valid_len <= 0:
                    break
                overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_val
                constraints.append(overlap_constraint)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, max_iters=2000, feastol=1e-6)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    return h.value
            except:
                pass
            return None

        c_low = 0.37
        c_high = best_c5

                break

            h_feasible = solve_feasibility(best_h, c_mid)
            if h_feasible is not None:
                c_low = c_mid
                best_h = h_feasible
            else:
                c_high = c_mid
        
        # Final optimization of found h
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, best_h, method='SLSQP',

            return evaluate_erdos_solution(h_vec, dx)

        try:
            # Use L-BFGS-B for better handling of bounds
            res = minimize(objective, h0, method='L-BFGS-B',

            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)

        except:
            pass
        return h0, best_c5

                best_c5 = c5_new

            return evaluate_erdos_solution(h_vec, dx)

                if final_c5 < best_c5:

                    best_c5 = final_c5
        except:
            pass

    return (list(best_h), best_c5, n_points)
