import numpy as np
from scipy.signal import correlate
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_segments(n_segments, initial_h=None):
    """Initialize segment values and boundaries"""
    if initial_h is not None and len(initial_h) >= n_segments:
        # Interpolate from existing solution
        x_full = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_segments + 1)
        h_vals = np.interp(x_new, x_full, initial_h)
        h_vals = np.clip(h_vals, 0, 1)
    else:
        # Create initial symmetric pattern
        h_vals = np.ones(n_segments) * 0.5
        
    # Segment boundaries
    x_bounds = np.linspace(0, 2, n_segments + 1)
    
    return h_vals, x_bounds

def smooth_max_correlation(h_vals, x_bounds, dx):
    """Compute a smooth approximation to the maximum correlation"""
    n_segments = len(h_vals)
    max_corr = 0
    # For each segment, compute overlap with all shifted segments
    for i in range(n_segments):
        for j in range(n_segments):
            # Overlap distance
            d = x_bounds[j+1] - x_bounds[i]  # Relative shift
            # Value product
            overlap_val = h_vals[i] * h_vals[j]
            if overlap_val > 0:
                # Calculate actual overlap region
                overlap_start = max(x_bounds[i], x_bounds[j])
                overlap_end = min(x_bounds[i+1], x_bounds[j+1])
                if overlap_start < overlap_end:
                    overlap_width = overlap_end - overlap_start
                    # Add to maximum if larger
                    max_corr = max(max_corr, overlap_val * overlap_width)
    return max_corr

def create_piecewise_representation(h_vals, x_bounds):
    """Convert segment representation to piecewise function values"""
    n_points = 100
    x_fine = np.linspace(0, 2, n_points)
    h_fine = np.zeros(n_points)
    for i in range(len(h_vals)):
        mask = (x_fine >= x_bounds[i]) & (x_fine < x_bounds[i+1])
        h_fine[mask] = h_vals[i]
    return h_fine

def trust_region_optimization(seed=42, budget_s=600):
    """Main optimization routine with trust region approach"""
    np.random.seed(seed)
    start_time = time()
    
    # Parameters for trust region
    n_initial_segments = 20
    max_segments = 40
    trust_region_radius = 0.1
    trust_region_decay = 0.5
    max_trust_region_steps = 100
    max_inner_iter = 500
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
        dx = 2.0 / n_points
        initial_c5 = evaluate_erdos_solution(initial_h, dx)
        best_h = initial_h.copy()
        best_c5 = initial_c5
        best_n_points = n_points
    else:
        initial_h = None
        best_h = None
        best_c5 = float('inf')
        best_n_points = 0
    
    # Initialize and try different segmentation schemes
    n_segments = n_initial_segments
    try_count = 0
    while time() - start_time < budget_s * 0.95 and try_count < 10:
        try:
            try_count += 1
            h_vals, x_bounds = initialize_segments(n_segments, initial_h)
            x_vals = (x_bounds[1:] + x_bounds[:-1]) / 2  # Midpoints for segment centers
            dx_segment = np.diff(x_bounds)
            
            # Number of parameters: values + boundaries
            n_params = n_segments + n_segments  # values and segment widths for trust region
            
            def objective(params):
                # Decode parameters: 0 to n_segments-1 are h_vals, n_segments to 2*n_segments-1 are boundary dx adjustments
                h_vals_new = np.clip(params[:n_segments], 0, 1)
                # Adjust boundary widths (with positive constraints)
                dx_adj = np.clip(params[n_segments:], -0.5, 0.5)
                x_bounds_new = np.zeros(n_segments + 1)
                x_bounds_new[0] = 0
                x_bounds_new[1:] = np.cumsum(dx_segment + dx_adj)  # Partition config
                x_bounds_new = np.clip(x_bounds_new, 0, 2)
                
                # Fix final constraint: x_bounds_new[n_segments] = 2.0
                x_bounds_new[-1] = 2.0
                
                # Ensure monotonically increasing
                x_bounds_new = np.maximum.accumulate(x_bounds_new)
                
                # Check if finite
                if not np.all(np.isfinite(x_bounds_new)):
                    return 1e10
                
                # Convert piecewise representation to function and evaluate
                try:
                    h_fine = create_piecewise_representation(h_vals_new, x_bounds_new)
                except:
                    return 1e10
                
                dx_val = 2.0 / len(h_fine)
                try:
                    corr = np.correlate(h_fine, 1 - h_fine, mode='full')
                    c5 = np.max(corr) * dx_val
                except:
                    return 1e10
                    
                return c5
            
            def constraint_norm(params):
                """Constraint: integral = 1"""
                # Decode h_vals part of params
                h_vals_new = np.clip(params[:n_segments], 0, 1)
                dx_adj = params[n_segments:] if len(params) > n_segments else np.zeros(n_segments)
                dx_segment_new = dx_segment + dx_adj
                dx_segment_new = np.clip(dx_segment_new, 0, 2)  # Should stay within 0,2 range
                integral = np.sum(h_vals_new * dx_segment_new)
                return integral - 1.0  # Target is 1
            
            # Initialize the trust region search
            current_params = np.concatenate([h_vals, np.zeros(n_segments)])
            best_params = current_params.copy()
            best_value = objective(current_params)
            
            # For trust region
            for tr_step in range(max_trust_region_steps):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Use scipy minimize with trust region
                # Since we're modifying both h_vals and x_bounds, we need a suitable objective
                # Use projected gradient descent method with constraint
                bounds = [(0, 1) for _ in range(n_segments)] + [(-0.5, 0.5) for _ in range(n_segments)]
                
                # Inner optimization
                result = minimize(objective, current_params, method='L-BFGS-B', bounds=bounds, 
                                 options={'maxiter': max_inner_iter, 'ftol': 1e-8})
                
                if result.success and np.isfinite(result.fun):
                    if result.fun < best_value:
                        best_params = result.x.copy()
                        best_value = result.fun
                        if best_value < best_c5:
                            # Extract the best solution
                            h_vals_sol = np.clip(best_params[:n_segments], 0, 1)
                            dx_adj_sol = best_params[n_segments:]
                            x_bounds_sol = np.zeros(n_segments + 1)
                            x_bounds_sol[0] = 0
                            x_bounds_sol[1:] = np.cumsum(dx_segment + dx_adj_sol)
                            x_bounds_sol = np.clip(x_bounds_sol, 0, 2)
                            x_bounds_sol[-1] = 2.0
                            x_bounds_sol = np.maximum.accumulate(x_bounds_sol)
                            
                            # Construct solution and re-evaluate
                            sol_h = create_piecewise_representation(h_vals_sol, x_bounds_sol)
                            sol_dx = 2.0 / len(sol_h)
                            sol_c5 = evaluate_erdos_solution(sol_h, sol_dx)
                            
                            if sol_c5 < best_c5:
                                best_h = sol_h.copy()
                                best_c5 = sol_c5
                                best_n_points = len(sol_h)
                                
                    # Update current solution
                    current_params = result.x
                else:
                    # Exceeds iterations, try again
                    # Perturb current and restart with fresh random parameters
                    current_params = best_params + np.random.normal(0, trust_region_radius, n_params)
                    
                # Reduce trust region size on failure but no excessive decrease
                if tr_step % 10 == 0 and tr_step > 0:
                    trust_region_radius *= trust_region_decay
                    
            # If we didn't find good improvement, increase n_segments
            if n_segments < max_segments and best_value >= 0.381:
                n_segments += 5
                
        except Exception as e:
            # Skip this attempt if error in optimization
            continue
    
    if best_h is None:
        # Fallback: return a basic construction
        best_h = np.ones(30) * 0.5
        dx = 2.0 / 30
        best_c5 = evaluate_erdos_solution(best_h, dx)
        best_n_points = 30
    
    return (list(best_h), best_c5, best_n_points)

def run(seed=42, budget_s=600, **kwargs):
    return trust_region_optimization(seed, budget_s)
