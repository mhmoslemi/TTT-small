import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # PSO parameters
    n_particles = 50
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Start with initial solution
    h0 = project_to_feasible(h_start, n_points)
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO main loop
    g_best = particles[0].copy()
    g_best_val = evaluate_erdos_solution(g_best, dx)
    
    stagnation_count = 0
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            p_best = particles[i]
            p_best_val = evaluate_erdos_solution(p_best, dx)
            
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (p_best - particles[i]) +
                c2 * r2 * (g_best - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate 
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_val:
                particles[i] = particles[i].copy()
                
            # Update global best
            if current_val < g_best_val:
                g_best = particles[i].copy()
                g_best_val = current_val
                stagnation_count = 0
            else:
                stagnation_count += 1
            
            if stagnation_count > 100:
                break
                
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        # Optimize with bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, g_best, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < g_best_val:
                    g_best = h_final
                    g_best_val = c5_final
        except Exception:
            pass
    
    return (list(g_best), g_best_val, n_points)
