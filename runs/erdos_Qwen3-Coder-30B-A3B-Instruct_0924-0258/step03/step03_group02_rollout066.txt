import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_overlap_vectors(h, n_points, dx):
    """Compute overlap vectors for all k-shifts"""
    overlap_vectors = []
    for k in range(n_points):
        if k == 0:
            overlap = np.sum(h * (1 - h)) * dx
        else:
            overlap = np.sum(h[:-k] * (1 - h[k:])) * dx
        overlap_vectors.append(overlap)
    return np.array(overlap_vectors)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Clustering-based approach
    max_clusters = 10
    kmeans = KMeans(n_clusters=max_clusters, n_init=10, random_state=seed)
    max_iterations = 100
    no_improvement_count = 0
    
    # Initialize with uniform clustering
    X_shifts = np.arange(n_points).reshape(-1, 1)
    cluster_labels = kmeans.fit_predict(X_shifts)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get current overlap vector
        current_overlaps = compute_overlap_vectors(best_h, n_points, dx)
        current_overlaps = np.abs(current_overlaps)  # Only absolute values for clustering
        
        # Re-cluster shifts based on current overlap pattern
        if iteration % 20 == 0 and iteration > 0:
            # Cluster shifts based on estimated values
            cluster_indices = np.argsort(current_overlaps)
            # Re-cluster based on rank of overlaps
            kmeans = KMeans(n_clusters=max_clusters, n_init=5, random_state=seed)
            X_shifts = np.argsort(current_overlaps).reshape(-1, 1)
            cluster_labels = kmeans.fit_predict(X_shifts)
        
        # Optimize each cluster
        for cluster_id in range(max_clusters):
            cluster_mask = cluster_labels == cluster_id
            cluster_shifts = np.where(cluster_mask)[0]
            
            if len(cluster_shifts) < 1:
                continue
                
            # Local optimization for this cluster
            h_current = best_h.copy()
            # SLSQP for this subset
            bounds = [(0, 1) for _ in range(n_points)]
            
            def cluster_objective(h_test):
                # Compute only for shifts in this cluster
                total_overlap = 0
                for k in cluster_shifts:
                    if k == 0:
                        total_overlap += np.sum(h_test * (1 - h_test))
                    else:
                        total_overlap += np.sum(h_test[:-k] * (1 - h_test[k:]))
                return -total_overlap * dx  # Negative because we minimize
            
            def cluster_constraint(h_test):
                return np.sum(h_test) - n_points / 2
            
            constraints = {'type': 'eq', 'fun': cluster_constraint}
            
            try:
                res = minimize(cluster_objective, h_current, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 10, 'ftol': 1e-6})
                if res.success:
                    h_test = project_to_feasible(res.x, n_points)
                    test_c5 = evaluate_erdos_solution(h_test, dx)
                    if test_c5 < best_c5:
                        best_h = h_test.copy()
                        best_c5 = test_c5
                        no_improvement_count = 0
            except:
                pass
                
        # Periodic projection to maintain feasibility
        best_h = project_to_feasible(best_h, n_points)
        
        # Check for stagnation
        no_improvement_count += 1
        if no_improvement_count > 50:
            # Reset cluster assignment and reduce steps
            kmeans = KMeans(n_clusters=max_clusters, n_init=5, random_state=seed)
            X_shifts = np.arange(n_points).reshape(-1, 1)
            cluster_labels = kmeans.fit_predict(X_shifts)
            no_improvement_count = 0
            
    # Final refinement with L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='L-BFGS-B',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    best_h = project_to_feasible(best_h, n_points)
    return (list(best_h), best_c5, n_points)
