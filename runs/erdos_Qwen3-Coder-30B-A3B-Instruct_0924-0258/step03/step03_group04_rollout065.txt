import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic-like initialization
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    max_iter = 10000
    iter_per_temp = 100
    perturb_ratio = 0.1
    perturb_strength = 0.2
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Annealing loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate
        candidate_h = current_h.copy()
        n_perturb = max(1, int(n_points * perturb_ratio))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for idx in indices:
            # Random perturbation
            delta = np.random.normal(0, perturb_strength)
            candidate_h[idx] += delta
            
        # Project to feasible region
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        # Compute change in objective
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        delta_c5 = candidate_c5 - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Update temperature
        if iteration % iter_per_temp == 0:
            T = max(T_min, T * alpha)
            
        # Adaptive perturbation
        if iteration % 1000 == 0 and iteration > 0:
            # Increase perturbation strength if no improvement recently
            if current_c5 >= best_c5:
                perturb_strength *= 1.1
                
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
