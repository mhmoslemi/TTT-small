import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available and if it's a well-formed solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Global initialization: sinusoidal pattern that sums to n/2
        h_start = np.abs(np.sin(np.linspace(0, 2*np.pi, n_points)))
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe algorithm with line search
    max_iter = 10000
    eta = 0.5  # Step size
    perturbation_rate = 0.1  # Perturbation half the time
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation to find worst overlap shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        # Convert index back to shift value k
        shift_idx = max_k - (n_points - 1)
        shift_idx = abs(shift_idx)  # For consistency
        
        # Form the shifted vector for linearization
        h_shifted = np.zeros(n_points)
        if shift_idx == 0:
            h_shifted = 1 - best_h
        else:
            h_shifted[shift_idx:] = 1 - best_h[:-shift_idx]
            
        # Linearized objective (approximation of derivative)
        linear_grad = h_shifted
        
        # Find the extreme point (binary pattern) that minimizes linearized obj
        # The "extreme point" is one that would minimize the linear objective
        # We do this heuristically by finding n/2 positions where (1-h) is largest
        # but we'll keep a few numbers that are closer to 0.5 to avoid perfect binary
        sorted_indices = np.argsort(linear_grad)[::-1]
        binary_support = set()
        for i in range(n_points // 2):
            binary_support.add(sorted_indices[i]) 
        
        # Construct the candidate binary vector
        h_candidate = np.zeros(n_points)
        for i in binary_support:
            h_candidate[i] = 1.0
            
        # Adjust to ensure sum = n/2
        actual_sum = np.sum(h_candidate)
        if actual_sum != n_points / 2:
            # Just increase some entries to meet constraint (heuristic)
            h_candidate += (n_points / 2 - actual_sum) / n_points
            
        # Perturbation: randomly mix in a small amount of non-binary values
        if random.random() < perturbation_rate:
            h_candidate = h_candidate * (1 - 0.1) + np.random.rand(n_points) * 0.1
            
        # Update with Frank-Wolfe step
        new_h = (1 - eta) * best_h + eta * h_candidate
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        c5_val = evaluate_erdos_solution(new_h, dx)
        if c5_val < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_val
            
        # Maybe adjust step size based on improvement
        # if iteration % 10 == 0 and iteration > 0:
        #     eta = max(0.05, eta * 0.99)
    
    # Final refinement with L-BFGS if within time
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                
    return (list(best_h), best_c5, n_points)
