import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Randomized improvement strategy
    max_iter = 50
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample a new initialization
        h_init = np.random.rand(n_points) * 0.5 + 0.25
        h_init = project_to_feasible(h_init, n_points)
        
        # Local refinement
        def obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
            
        res = minimize(obj, h_init, method='L-BFGS-B', 
                      bounds=[(0,1) for _ in range(n_points)], 
                      options={'maxiter': 30})
        
        if res.success:
            h_refined = project_to_feasible(res.x, n_points)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
                
        # Try periodic pattern
        h_periodic = 0.5 * (1 + np.sin(np.pi * np.arange(n_points) / n_points))
        h_periodic = project_to_feasible(h_periodic, n_points)
        c5_periodic = evaluate_erdos_solution(h_periodic, dx)
        if c5_periodic < best_c5:
            best_h = h_periodic.copy()
            best_c5 = c5_periodic
            
        # Try a single bump
        h_bump = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 4
        for i in range(max(0, mid - width), min(n_points, mid + width)):
            h_bump[i] = 1.0
        h_bump = project_to_feasible(h_bump, n_points)
        c5_bump = evaluate_erdos_solution(h_bump, dx)
        if c5_bump < best_c5:
            best_h = h_bump.copy()
            best_c5 = c5_bump
    
    # Final optimization step
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
            
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                      bounds=[(0,1) for _ in range(n_points)],
                      options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
