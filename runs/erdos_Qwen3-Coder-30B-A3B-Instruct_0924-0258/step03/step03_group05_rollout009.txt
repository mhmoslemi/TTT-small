import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    n_points = 35
    dx = 2.0 / n_points
    population_size = 50
    generations = 0
    stagnation_count = 0
    max_stagnation = 100
    
    # Create initial population
    population = []
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        population.append(h_start)
    else:
        h_start = np.ones(n_points) * 0.5
        population.append(h_start)
    
    # Fill population with random valid solutions
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # GA parameters
    mutation_rate = 0.2
    crossover_rate = 0.8
    
    def tournament_selection(pop, k=5):
        selected = random.sample(pop, min(k, len(pop)))
        return min(selected, key=lambda h: evaluate_erdos_solution(h, dx))
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += random.gauss(0, 0.1)
        # Clip and renormalize
        mutated = np.clip(mutated, 0, 1)
        h_sum = np.sum(mutated)
        if h_sum > 0:
            mutated = mutated * (n_points / 2) / h_sum
        return mutated
    
    def evolve_population(pop):
        new_pop = []
        # Keep best individual
        pop.sort(key=lambda h: evaluate_erdos_solution(h, dx))
        new_pop.append(pop[0].copy())
        
        # Create offspring
        while len(new_pop) < population_size:
            parent1 = tournament_selection(pop)
            parent2 = tournament_selection(pop)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
            
        return new_pop[:population_size]
    
    # Main GA loop
    while time() - start_time < budget_s * 0.95:
        population = evolve_population(population)
        population.sort(key=lambda h: evaluate_erdos_solution(h, dx))
        current_best = population[0]
        current_c5 = evaluate_erdos_solution(current_best, dx)
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= max_stagnation:
            # Increase mutation rate if stagnating
            mutation_rate = min(mutation_rate * 1.1, 0.5)
            stagnation_count = 0
            
        generations += 1
    
    # Final local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
