import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use simulated annealing with tabu search
    # Limit iterations to allow for multiple restarts
    max_iter = 10000
    tabu_list = []
    tabu_length = 10
    
    # Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.999
    min_temp = 1e-6
    
    # Track best solution found
    current_h = best_h.copy()
    current_c5 = best_c5
    temp = initial_temp
    iter_count = 0
    
    # Precompute indices for efficient evaluation
    indices = np.arange(n_points)
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.9:
        iter_count += 1
        
        # Make a small perturbation
        h_new = current_h.copy()
        # Choose a random index to perturb
        idx = np.random.randint(0, n_points)
        
        # Perturb by adding noise
        noise = np.random.normal(0, temp)
        h_new[idx] += noise
        
        # Clip to bounds
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Adjust to maintain integral constraint
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        
        # Compute new C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < current_c5:
            current_h = h_new.copy()
            current_c5 = c5_new
        else:
            # With probability, accept worse solution
            delta = c5_new - current_c5
            if np.random.random() < np.exp(-delta / temp):
                current_h = h_new.copy()
                current_c5 = c5_new
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Update temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Update tabu list
        tabu_list.append(tuple(current_h.round(6)))
        if len(tabu_list) > tabu_length:
            tabu_list.pop(0)
    
    # Perform a final optimization with local search
    if (time.time() - start_time) < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts
        num_restarts = 10
        for i in range(num_restarts):
            if (time.time() - start_time) > budget_s * 0.95:
                break
                
            # Perturb current best
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
