import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Helper function to project solution to feasible region
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h

    # Nelder-Mead optimization with feasible restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Create initial simplex
    M = n_points
    simplex = np.array([h_start] + [project_feasible(h_start + np.random.normal(0, 0.05, n_points)) 
                                    for _ in range(M)])
    
    # Nelder-Mead iteration
    max_iter = 150
    iter_count = 0
    restart_count = 0
    no_improve_count = 0
    max_no_improve = 10
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.9:
        # Evaluate all points in simplex
        values = [objective(p) for p in simplex]
        # Order the simplex by objective
        order = np.argsort(values)
        simplex = simplex[order]
        values = [values[i] for i in order]
        
        # Check for improvement
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex[0].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > max_no_improve:
            # Restart the simplex
            simplex = np.array([best_h] + [project_feasible(best_h + np.random.normal(0, 0.05, n_points)) 
                                          for _ in range(M)])
            no_improve_count = 0
            restart_count += 1
            
        # Stop if we haven't improved for too long
        if no_improve_count > max_no_improve:
            break
            
        # Perform Nelder-Mead operations
        x_old = np.mean(simplex[:-1], axis=0)
        x_reflect = x_old + (x_old - simplex[-1])  # Reflection
        x_reflect = project_feasible(x_reflect)
        f_reflect = objective(x_reflect)
        
        if values[0] <= f_reflect < values[-2]:
            # Accept reflection
            simplex[-1] = x_reflect
        elif f_reflect < values[0]:
            # Expansion
            x_expand = x_old + 2*(x_reflect - x_old)
            x_expand = project_feasible(x_expand)
            f_expand = objective(x_expand)
            if f_expand < f_reflect:
                simplex[-1] = x_expand
            else:
                simplex[-1] = x_reflect
        else:
            # Contraction
            x_contract = x_old + 0.5*(simplex[-1] - x_old)
            x_contract = project_feasible(x_contract)
            f_contract = objective(x_contract)
            if f_contract < values[-1]:
                simplex[-1] = x_contract
            else:
                # Shrink
                simplex = simplex[0:1] + [simplex[0] + 0.5*(p - simplex[0]) for p in simplex[1:]]
            
        iter_count += 1

    # Local refinement using gradient-based optimization
    def refine_solution(h_seed):
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        con = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj, h_seed, method='SLSQP', bounds=bounds, constraints=con, 
                           options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                return h_refined, c5_refined
        except:
            pass
        return h_seed, obj(h_seed)

    # Perform local refinement on top candidates
    top_candidates = [best_h] + [simplex[i] for i in range(min(3, len(simplex)))]
    
    for i, h_candidate in enumerate(top_candidates):
        if (time() - start_time) > budget_s * 0.95:
            break
        h_refined, c5_refined = refine_solution(h_candidate)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined

    return (list(best_h), best_c5, n_points)
