import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a more aggressive refinement approach
    remaining_time = budget_s * 0.8  # Leave some buffer
    
    # Try a variety of starting points
    start_points = [h_start]
    if 'initial_h_values' in globals():
        start_points.append(np.array(initial_h_values))
    
    # Add random perturbations
    for i in range(3):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        start_points.append(project_to_feasible(h_pert, n_points))
    
    for h_init in start_points:
        if time() - start_time > remaining_time:
            break
            
        h_refined = h_init.copy()
        
        # Initial convex relaxation with McCormick bounds
        try:
            # Define variables
            h_vars = cp.Variable(n_points)
            
            # Define constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Apply approximate McCormick relaxation for overlaps
            # This is a heuristic: we'll just check that overlaps don't get too large
            # instead of full relaxation
            
            # For optimization, we'll use local refinement
            def objective_func(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_func(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_func}
            
            # Local optimization with SLSQP
            res = minimize(objective_func, h_refined, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
                    
        except Exception:
            continue
    
    # Final SLSQP refinement with best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            h_final = best_h.copy()
            def objective_func(h_vec):
                return evaluate_erdos_solution(h_vec, dx)

                return np.sum(h_vec) - n_points / 2

    # Ensure the final result is valid

    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
