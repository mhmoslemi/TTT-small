import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def get_breakpoints_and_values(h, n_points, dx):
    """Convert h to breakpoints and segment values"""
    # Simple approach: Use h values to determine approximately where to place breaks
    # But we'll simplify by modeling it as piecewise constant function
    # This is a key point: we model our optimization over piecewise constant fields
    # with explicit breakpoints
    
    # For simplicity, we'll model it as a binary segment selection + position optimization
    # But we aren't necessarily constrained to only binary values in segment fields
    # We'll start from a reasonable parameterization
    
    # This is a 1D parameterization approach: optimize position of values 
    # and their widths through coordinate descent
    values = np.array(h)
    return values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local refinement approach
    # We'll try to get better results with more direct optimization
    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Generate population, refine, optimize
    pop_size = 200
    max_iter = 100
    pop = []
    
    # Initialize population
    for i in range(pop_size):
        individual = np.random.rand(n_points)
        # Normalize
        h_sum = np.sum(individual)
        if h_sum > 0:
            individual = individual * (n_points / 2) / h_sum
        pop.append(individual.copy())
    
    # Add the best solution at start
    pop.append(best_h.copy())
        
    # Evaluate population
    fitness = []
    for h in pop:
        fitness.append(objective(h))
    
    # Main optimization loop
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s * 0.95:
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        pop = [pop[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check if we improved
        if fitness[0] < best_c5:
            best_h = pop[0].copy()
            best_c5 = fitness[0]
        
        # Create new generation
        new_pop = [pop[0].copy()]  # Elitism
        
        # Selection and crossover
        num_to_create = pop_size - 1
        for _ in range(num_to_create):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Tournament selection
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            parent1 = pop[parent1_idx]
            parent2 = pop[parent2_idx]
            
            # Blend crossover (BLX-0)
            alpha = 0.5
            child = np.zeros(n_points)
            for i in range(n_points):
                d = abs(parent1[i] - parent2[i])
                lower = min(parent1[i], parent2[i]) - alpha * d
                upper = max(parent1[i], parent2[i]) + alpha * d
                child[i] = np.random.uniform(lower, upper)
            
            # Random mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Normalize
            h_sum = np.sum(child)
            if h_sum > 0:
                child = child * (n_points / 2) / h_sum
            new_pop.append(child.copy())
        
        # Replace old population
        pop = new_pop
        
        # Recalculate fitness
        fitness = []
        for h in pop:
            fitness.append(objective(h))
        
        iter_count += 1
    
    # Final SLSQP refinement of best solutions
    if time.time() - start_time < budget_s * 0.98:
        # Let's do SLSQP for top 5, since we're running version without constraints
        best_idx = np.argmin(fitness)
        h_start = pop[best_idx].copy()
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.98:
                break
                
            h_perturb = h_start + np.random.normal(0, 0.02, n_points)
            try:
                res = minimize(objective, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    c5_test = objective(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    # Final normalization check
    h_final = np.clip(best_h, 0, 1)
    h_sum = np.sum(h_final)
    if h_sum > 0:
        h_final = h_final * (n_points / 2) / h_sum
    actual_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), actual_c5, n_points)
