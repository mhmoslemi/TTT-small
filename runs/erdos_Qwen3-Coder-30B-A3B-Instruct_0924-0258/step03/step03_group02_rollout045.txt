import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate Descent with random order
    max_iterations = 500
    tolerance = 1e-6
    search_steps = 100
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Randomize the order
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        h_current = best_h.copy()
        
        for i in indices:
            if time() - start_time > budget_s * 0.95:
                break
            
            # One-dimensional search for h[i]
            current_h = h_current[i]
            current_c5 = evaluate_erdos_solution(h_current, dx)
            
            best_h_i = current_h
            best_c5_i = current_c5
            
            # Try multiple values for h[i]
            for j in range(search_steps):
                # Linear search on [0, 1]
                test_h_i = j / (search_steps - 1.0)
                h_test = h_current.copy()
                h_test[i] = test_h_i
                
                # Project to feasible region
                h_test = project_to_feasible(h_test, n_points)
                
                test_c5 = evaluate_erdos_solution(h_test, dx)
                
                if test_c5 < best_c5_i:
                    best_c5_i = test_c5
                    best_h_i = test_h_i
            
            # Update current solution
            h_current[i] = best_h_i
            
            if abs(best_c5_i - current_c5) > tolerance:
                improved = True
                
        # Project to feasible region after all updates
        h_current = project_to_feasible(h_current, n_points)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
        
        # Early termination if no improvement
        if not improved:
            break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
