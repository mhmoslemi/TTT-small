import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a binary pattern with exactly n_points/2 ones
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe algorithm for non-convex optimization
    max_iterations = 50000
    step_size = 0.5
    perturb_rate = 0.1
    
    h_current = h_start.copy()
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for current h
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Convert to shift in range [1, n_points-1], ignore shift 0
        if max_shift == 0:
            valid_shifts = list(range(1, n_points))
            max_shift = valid_shifts[np.argmax([corr[n_points-1+k] for k in valid_shifts])]
        
        # For shift k, we want to compute correlation over overlapping region
        k = max_shift
        if k > 0:
            # indices for h_current on left, 1-h_current on right
            left_indices = np.arange(n_points - k)
            right_indices = np.arange(k, n_points)
            # compute linear approximation: gradient of overlap with respect to h
            grad = 1 - h_current[right_indices] - h_current[left_indices]
            # create extreme point by choosing indices with highest descent
            extreme_indices = np.argsort(-grad)[:n_points//2]
        else:
            # k < 0
            k = -k
            left_indices = np.arange(k, n_points)
            right_indices = np.arange(n_points - k)
            grad = 1 - h_current[right_indices] - h_current[left_indices]
            extreme_indices = np.argsort(-grad)[:n_points//2]
        
        # Construct extreme point (binary vector with chosen ones)
        h_extreme = np.zeros(n_points)
        h_extreme[extreme_indices] = 1.0
        
        # Frank-Wolfe update
        h_new = (1 - step_size) * h_current + step_size * h_extreme
        h_projected = project_to_feasible(h_new, n_points)
        
        # Test for improvement
        c5_new = evaluate_erdos_solution(h_projected, dx)
        if c5_new < best_c5:
            best_h = h_projected.copy()
            best_c5 = c5_new
            step_size = min(0.99, step_size * 1.01)  # adaptive step size
        else:
            step_size = max(0.01, step_size * 0.99)
            
        # Perturb if no improvement (escape local optima)
        if np.random.random() < perturb_rate:
            h_perturbed = h_projected + np.random.normal(0, 0.005, n_points)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
                
        h_current = h_projected
        
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
