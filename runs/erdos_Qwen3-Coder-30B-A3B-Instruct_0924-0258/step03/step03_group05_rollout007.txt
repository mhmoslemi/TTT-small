import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    pop_size = 50
    generations = 200
    mutation_rate = 0.2
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def create_population():
        pop = [h_start.copy()]
        for _ in range(pop_size - 1):
            pop.append(create_individual())
        return pop
    
    def tournament_select(pop, fitnesses):
        selected = []
        for _ in range(pop_size):
            tournament_indices = random.sample(range(pop_size), tournament_size)
            tournament_fits = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fits)]
            selected.append(pop[winner_idx].copy())
        return selected
    
    def crossover(parent1, parent2):
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(individual):
        for i in range(n_points):
            if random.random() < mutation_rate:
                individual[i] += random.gauss(0, 0.1)
        # Renormalize and clip
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    def evaluate_population(pop):
        return [evaluate_erdos_solution(ind, dx) for ind in pop]
    
    population = create_population()
    
    # Run GA
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        fitnesses = evaluate_population(population)
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Selection
        selected = tournament_select(population, fitnesses)
        
        # Create new population
        new_population = []
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Fine-tune best solution
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Try finer resolution if time permits
    if time() - start_time < budget_s * 0.8 and n_points == 35:
        # Use best solution from coarse resolution as starting point for fine grid
        h_coarse = best_h
        
        n_points_fine = 70
        dx_fine = 2.0 / n_points_fine
        h_fine = np.interp(np.linspace(0, 2, n_points_fine), 
                           np.linspace(0, 2, n_points), h_coarse)
        # Normalize fine grid
        h_fine = h_fine * (n_points_fine / 2) / np.sum(h_fine)
        
        # Local optimization on fine grid
        def objective_fine(h_vec):
            return evaluate_erdos_solution(h_vec, dx_fine)
        
        cons_fine = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_fine / 2}]
        bounds_fine = [(0, 1) for _ in range(n_points_fine)]
        
        try:
            res = minimize(objective_fine, h_fine, method='SLSQP', 
                           bounds=bounds_fine, constraints=cons_fine, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final_fine = np.clip(res.x, 0, 1)
                h_final_fine = h_final_fine * (n_points_fine / 2) / np.sum(h_final_fine)
                final_c5_fine = evaluate_erdos_solution(h_final_fine, dx_fine)
                if final_c5_fine < best_c5:
                    best_h = h_final_fine
                    best_c5 = final_c5_fine
                    n_points = n_points_fine
        except:
            pass
    
    # Ensure final solution meets constraints
    if np.sum(best_h) != n_points / 2:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
