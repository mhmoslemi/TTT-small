import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm approach
    pop_size = 500
    num_generations = 100
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def mutate(individual):
        new_individual = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                new_individual[i] = np.random.rand()
        # Normalize
        h_sum = np.sum(new_individual)
        if h_sum > 0:
            new_individual = new_individual * (n_points / 2) / h_sum
        return new_individual
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy()
        crossover_point = np.random.randint(1, n_points)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        # Normalize
        h_sum = np.sum(child)
        if h_sum > 0:
            child = child * (n_points / 2) / h_sum
        return child
    
    def evaluate_individual(individual):
        return evaluate_erdos_solution(individual, dx)
    
    # Create initial population
    population = [create_individual() for _ in range(pop_size)]
    
    # Main GA loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate population
        fitness_scores = [evaluate_individual(ind) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    # Final local refinement on best solution
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Local optimization
        try:
            res = minimize(obj_func, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
