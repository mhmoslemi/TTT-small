import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal initial guess
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    iter_per_temp = 100
    acceptance_rate = 0.5
    max_iter = int(budget_s * 500)  # Rough estimate
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Track recent improvements
    recent_improvements = []
    max_recent = 10
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        candidate_h = current_h.copy()
        
        # Choose subset of indices to perturb (5-10%)
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Perturb selected indices
        for i in indices:
            # Small random change
            delta = np.random.normal(0, 0.05)
            candidate_h[i] += delta
            
        # Project to feasible region
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        
        if delta_c5 < 0 or np.random.random() < math.exp(-delta_c5 / temp):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Temperature update
        if iteration % iter_per_temp == 0 and iteration > 0:
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
                
        # Track recent improvements to adapt cooling rate
        if delta_c5 < 0:
            recent_improvements.append(-delta_c5)
            if len(recent_improvements) > max_recent:
                recent_improvements.pop(0)
                
    return (list(best_h), best_c5, n_points)
