import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_c5 = current_c5
    best_h = h_start.copy()
    best_n_points = n_points
    
    # Use trust region approach with alternating optimization
    max_iter = 1000
    trust_radius = 0.1
    trust_radius_decay = 0.5
    min_trust_radius = 1e-6
    
    # Initialize with appropriate gradient descent approach
    h_current = h_start.copy()
    
    # Coordinate descent with projected gradient
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            best_n_points = n_points
            
        # Evaluate gradient approximately using finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        base_score = current_c5
        
        for i in range(len(h_current)):
            if time() - start_time > budget_s * 0.95:
                break
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)

        step_size = 0.01

        new_score = evaluate_erdos_solution(h_new, dx)
        if new_score < current_c5:

    # Attempt local optimization with scipy minimize
    if time() - start_time < budget_s * 0.9:
        bounds = [(0, 1) for _ in range(n_points)]

            x = np.array(x)
            x = project_to_feasible(x, n_points)
            return evaluate_erdos_solution(x, dx)

            x = np.array(x)
            return np.sum(x) * dx - 1.0
