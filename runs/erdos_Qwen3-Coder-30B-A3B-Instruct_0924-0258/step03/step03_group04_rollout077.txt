import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create periodic sinusoidal pattern for initialization
        n_points = 35
        i_vals = np.arange(n_points)
        h_start = 0.5 * np.sin(np.pi * i_vals / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize the optimization loop
    max_iter = 100
    iter_count = 0
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Compute the current overlap values for all shifts
        corr_all = np.correlate(best_h, 1 - best_h, mode='full')
        # Extract the relevant values (non-zero shifts)
        corr_values = corr_all[len(best_h)-1:]  # Only positive shifts
        # Identify worst 10% of shifts
        num_worst = max(1, int(len(corr_values) * 0.1))
        worst_indices = np.argsort(corr_values)[-num_worst:]
        
        # Convex feasibility problem with only worst shifts constrained
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Only constrain the worst shifts
        for k in worst_indices:
            if k == 0:
                continue
            # For shift k, the correlation is over valid window
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                constraints.append(corr_sum <= best_c5 * dx * valid_len)
        
        # No objective needed for feasibility check
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            # Solve with ECOS
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_var.value
                if h_feasible is not None:
                    # Refine using gradient descent
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    # Use h_feasible as starting point for gradient descent
                    res = minimize(refinement_obj, h_feasible, method='L-BFGS-B', 
                                   bounds=[(0,1) for _ in range(n_points)], 
                                   options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6})
                    
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                            
        except:
            pass  # If optimization fails, continue with current solution
        
        iter_count += 1
        
        # Break early if macro improvement is observed
        if best_c5 < 0.38085 and iter_count > 5:
            break
    
    # Final refinement with gradient descent on best solution so far
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-7})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
