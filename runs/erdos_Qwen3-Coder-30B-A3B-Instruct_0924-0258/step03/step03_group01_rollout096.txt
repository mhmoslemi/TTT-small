import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iterations = 30
    w_start = 0.9
    w_end = 0.2
    c1 = c2 = 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial solution
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random solution
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
    
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    if gbest_value < best_c5:
        best_h = gbest.copy()
        best_c5 = gbest_value
    
    # PSO iteration
    for iter_count in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        
        w = w_start - (w_start - w_end) * (iter_count / max_iterations)
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - swarm[i]) + 
                           c2 * r2 * (gbest - swarm[i]))
            
            swarm[i] += velocities[i]
            
            # Ensure feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = swarm[i].copy()
                gbest_value = current_value
                
                if gbest_value < best_c5:
                    best_h = gbest.copy()
                    best_c5 = gbest_value
        
        # Occasionally refine with SQP
        if iter_count % 5 == 0 and iter_count > 0:
            refine_indices = np.argsort(pbest_values)[:3]
            for idx in refine_indices:
                h_refine = pbest[idx].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_final = res.x
                        c5_final = obj(h_final)
                        if c5_final < best_c5:
                            best_h = h_final
                            best_c5 = c5_final
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
