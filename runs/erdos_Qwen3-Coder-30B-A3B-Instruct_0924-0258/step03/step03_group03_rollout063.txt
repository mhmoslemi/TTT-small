import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_maximum(values, beta):
    """Smooth approximation of maximum"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * values)))

def gradient_smooth_objective(h, beta, eps=1e-5):
    """Compute gradient of smooth objective using finite differences"""
    n = len(h)
    grad = np.zeros(n)
    base_value = smooth_maximum(np.correlate(h, 1 - h, mode='full'), beta)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        plus_value = smooth_maximum(np.correlate(h_plus, 1 - h_plus, mode='full'), beta)
        grad[i] = (plus_value - base_value) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize

        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = project_to_feasible(h, n_points)
    
    # Parameters
    beta = 100.0
    step_size = 0.01
    tol = 1e-6
    max_iter = 1000
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization loop
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = gradient_smooth_objective(h, beta)
        
        # Line search
        step = h - step_size * grad
        step = project_to_feasible(step, n_points)
        new_c5 = evaluate_erdos_solution(step, dx)
        
        if new_c5 < best_c5:
            h = step.copy()
            best_c5 = new_c5
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Adjust beta
        if iter % 100 == 0 and iter > 0:
            beta *= 1.5
            
        # Check for convergence
        if iter > 10 and np.abs(best_c5 - evaluate_erdos_solution(h, dx)) < tol:
            break
    
    # Final refinement
    h_refined = h.copy()
    for _ in range(200):
        if time() - start_time > budget_s * 0.95:
            break
        grad = gradient_smooth_objective(h_refined, beta)
        step = h_refined - 0.001 * grad
        step = project_to_feasible(step, n_points)
        new_c5 = evaluate_erdos_solution(step, dx)
        if new_c5 < best_c5:
            h_refined = step.copy()
            best_c5 = new_c5
        else:
            break

    # Final check
    final_c5 = evaluate_erdos_solution(h_refined, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_refined.copy()
    
    return (list(best_h), best_c5, n_points)
