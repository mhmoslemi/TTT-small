import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available

        n_points = len(h_start)

    best_c5 = float('inf')
    best_h = None
    best_n_points = None
    
    # Initial evaluation
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_start, dx)
    if current_c5 < best_c5:
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points

    # Try different resolutions
    resolutions = [25, 30, 35]
    
    for n_points_try in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        if n_points_try != n_points:
            # Interpolate to new resolution
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_points_try)
            n_points = n_points_try
        
        # Local optimization with gradient descent
        h_current = h_start.copy()
        max_iter = 150
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient using finite differences
            grad = np.zeros_like(h_current)
            eps = 1e-4
            base_score = evaluate_erdos_solution(h_current, dx_try)
            
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)

                grad[i] = (score_plus - base_score) / eps

            # Update with gradient descent
            step_size = 0.002
            h_new = h_current - step_size * grad
            h_new = project_to_feasible(h_new, n_points)
            
            new_score = evaluate_erdos_solution(h_new, dx_try)
            
            # Accept if better
            if new_score < base_score:
                h_current = h_new
                if new_score < best_c5:
                    best_h = h_current.copy()
                    best_c5 = new_score
                    best_n_points = n_points
            else:
                # Try backtracking line search
                step_size *= 0.5
                h_new = h_current - step_size * grad
                h_new = project_to_feasible(h_new, n_points)
                new_score = evaluate_erdos_solution(h_new, dx_try)
                if new_score < base_score:
                    h_current = h_new
                    if new_score < best_c5:
                        best_h = h_current.copy()
                        best_c5 = new_score

            h_refined = project_to_feasible(h_refined, best_n_points)

            score_refined = evaluate_erdos_solution(h_refined, dx_refined)
            if score_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = score_refined

    return (list(best_h), best_c5, best_n_points)
