import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.2
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # normalize to sum to n_points/2
        return np.clip(h, 0, 1)
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    no_improvement_count = 0
    
    while time() - start_time < budget_s * 0.9 and generation < max_generations:
        # Tournament selection
        def tournament_select():
            tournament = random.sample(list(enumerate(population)), tournament_size)
            winner_idx = min(tournament, key=lambda x: fitness[x[0]])[0]
            return population[winner_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection + Crossover
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            # Single point crossover
            crossover_point = random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.1)
                if random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.1)
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Evaluate new population
        population = new_population[:population_size]
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        generation += 1
        
        if no_improvement_count >= 100:
            break
    
    # Final local refinement with SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_refined = np.clip(res.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            final_c5 = evaluate_erdos_solution(h_refined, dx)
            if final_c5 < best_c5:
                best_h = h_refined
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
