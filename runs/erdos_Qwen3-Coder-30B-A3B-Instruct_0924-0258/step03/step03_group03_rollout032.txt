import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Quick initialization with segment-based approach
    n_points = 35  # Use the provided initial solution size if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create a symmetric construction that should be reasonably good
        h_start = np.ones(n_points) * 0.5
        # Perturb to break symmetry slightly
        h_start += np.random.normal(0, 0.01, n_points)
        h_start = project_to_feasible(h_start, n_points)
    
    # Initial evaluation
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_n_points = n_points
    
    # Use trust region approach with projected gradient
    max_iter = 1000
    trust_radius = 0.1
    max_trust_radius = 0.5
    min_trust_radius = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-5
        base_score = evaluate_erdos_solution(best_h, dx)
        
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps

        # Compute trust region step
        # Simple projected gradient step with trust region
        step = -grad * 0.001  # Learning rate
        # Apply trust region
        step_norm = np.linalg.norm(step)
        if step_norm > trust_radius:
            step = step * trust_radius / step_norm
        
        # Update h
        new_h = best_h + step
        new_h = project_to_feasible(new_h, n_points)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept the step if it improves the objective
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            trust_radius = min(trust_radius * 1.2, max_trust_radius)
        else:
            trust_radius = max(trust_radius * 0.5, min_trust_radius)
    
    # Final refinement with gradient descent
    for _ in range(500):
        if time() - start_time > budget_s * 0.95:
            break
        grad = np.zeros_like(best_h)
        eps = 1e-5
        base_score = evaluate_erdos_solution(best_h, dx)
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
        step_size = 0.0005
        best_h = best_h - step_size * grad
        best_h = project_to_feasible(best_h, n_points)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure final feasibility
    best_h = project_to_feasible(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, best_n_points)
