import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Heuristic search with better local refinement
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimizer settings
    max_iter_local = 100
    n_restarts = 15
    max_time_local = budget_s * 0.8
    time_per_restart = max_time_local / n_restarts
    
    # Try multiple restarts with different starting points
    for restart in range(n_restarts):
        if time.time() - start_time > max_time_local:
            break
            
        # Start with perturbed version of current best
        h_perturb = best_h + np.random.normal(0, 0.02, n_points)
        
        # Ensure it's within bounds, and normalize
        h_perturb = np.clip(h_perturb, 0, 1)
        h_sum = np.sum(h_perturb)
        if h_sum > 0:
            h_perturb = h_perturb * (n_points / 2) / h_sum
            
        # Local optimization
        try:
            # Use L-BFGS-B for better convergence 
            res = minimize(obj_func, h_perturb, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter_local, 'ftol': 1e-8})
            
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
    
    # Final fine-tuning with SLSQP if still time
    if time.time() - start_time < budget_s * 0.95:
        try:
            # Try again with more stringent settings
            res = minimize(obj_func, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-10})
            
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_sum = np.sum(h_final)
    if h_sum > 0:
        h_final = h_final * (n_points / 2) / h_sum
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
