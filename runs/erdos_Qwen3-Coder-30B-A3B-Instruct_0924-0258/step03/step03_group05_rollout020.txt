import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iter = 10000
    patience = 100
    patience_counter = 0
    best_c5_seen = best_c5
    
    # Initial step size and decay
    lr = 0.01
    decay = 0.9999
    
    # Optimizer loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute correlations for current h
        h = best_h.copy()
        
        # Compute gradient of correlation with respect to h
        # This is approximated via finite differences
        # Correlation: sum h_i * (1 - h_{i+k}) for all valid k,i
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = len(corr) // 2
        
        # Gradient approximation - simplify by considering only max correlation path
        # We treat C5 as the maximum of correlations over all shifts
        # So we compute subgradient wrt h for max shift
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - max_k
        max_overlap = corr[max_shift_idx] * dx
        
        # Simple gradient approx - finite difference
        grad = np.zeros_like(h)
        epsilon = 1e-4
        
        # Compute approximate gradient using finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1.0, h[i] + epsilon)
            h_minus[i] = max(0.0, h[i] - epsilon)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            
            grad[i] = (corr_plus[max_shift_idx] - corr_minus[max_shift_idx]) * dx / (2 * epsilon)
        
        # Apply gradient
        h_new = h - lr * grad
        
        # Project to [0,1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to integral constraint
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        # Compute new C5 value
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Update if improved
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            patience_counter = 0
            best_c5_seen = best_c5
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        lr *= decay
        
        # Escape local minima with random perturbations
        if iteration % 50 == 0 and iteration > 0:
            h_perturbed = best_h.copy()
            # Random perturbation to escape local minima
            perturb = np.random.normal(0, 0.005, n_points)
            h_perturbed += perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum > 0:
                h_perturbed = h_perturbed * (n_points / 2) / h_perturbed_sum
            perturbed_c5 = evaluate_erdos_solution(h_perturbed, dx)
            if perturbed_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = perturbed_c5
                
        # Early stopping if no improvement
        if patience_counter >= patience:
            break
    
    # Final refinement with scipy minimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Ensure final solution meets constraints exactly
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
