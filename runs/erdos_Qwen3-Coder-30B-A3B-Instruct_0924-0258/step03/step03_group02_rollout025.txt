import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        c5 = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_c5.append(c5)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # PSO main loop
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                if c5 < best_c5:
                    best_h = particles[i].copy()
                    best_c5 = c5
    
    # Post-processing - fine-tuning with optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            # Use SLSQP to refine the best solution
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
