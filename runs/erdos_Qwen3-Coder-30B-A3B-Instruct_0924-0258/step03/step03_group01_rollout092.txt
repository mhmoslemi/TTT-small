import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for GA
    population_size = 20
    num_generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_refinement_interval = 1
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Keep top 50%
        top_indices = sorted_indices[:population_size // 2]
        top_population = [population[i] for i in top_indices]
        
        # Generate new population through crossover and mutation
        new_population = []
        
        # Keep top individuals
        new_population.extend(top_population)
        
        # Crossover
        while len(new_population) < population_size:
            if random.random() < crossover_rate:
                parent1 = random.choice(top_population)
                parent2 = random.choice(top_population)
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = random.choice(top_population)
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += np.random.normal(0, 0.03)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Ensure we have exactly population_size individuals
        population = new_population[:population_size]
        
        # Local refinement with SQP every few generations
        if gen % sqp_refinement_interval == 0:
            for i in range(min(3, len(population))):
                h_candidate = population[i]
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_candidate, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = objective(refined_h)
                        if refined_c5 < fitness[i]:
                            population[i] = refined_h
                            fitness[i] = refined_c5
                except:
                    pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50 and best_c5 > 0.3808:
        # Try further optimization with a different approach
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        current_h = best_h.copy()
        max_iter = 50
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.98:
                break
            try:
                res = minimize(objective, current_h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    current_h = res.x
                    current_c5 = objective(current_h)
                    if current_c5 < best_c5:
                        best_h = current_h.copy()
                        best_c5 = current_c5
            except:
                break
    
    return (list(best_h), best_c5, n_points)
