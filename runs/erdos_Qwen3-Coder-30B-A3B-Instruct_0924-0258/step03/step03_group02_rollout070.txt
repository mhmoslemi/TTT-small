import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent approach
    max_iterations = 500
    tolerance = 1e-6
    
    # Initialize current h
    h_current = h_start.copy()
    current_c5 = best_c5
    
    n_iter = 0
    last_improvement = 0.0
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Randomize order to avoid bias
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        improved = False
        
        # Update each variable in random order
        for i in indices:
            if time() - start_time > budget_s * 0.9:
                break
                
            # Save current value for restoration
            old_value = h_current[i]
            
            # Evaluate change for different values of h[i]
            best_value = old_value
            best_c5_local = current_c5
            
            # Simple grid search for optimal h[i]
            # Test 100 values in neighborhood of current value
            if old_value == 0:
                test_values = np.linspace(0, 1, 50)
            elif old_value == 1:
                test_values = np.linspace(0, 1, 50)
            else:
                # Search around current value
                range_start = max(0, old_value - 0.2)
                range_end = min(1, old_value + 0.2)
                test_values = np.linspace(range_start, range_end, 50)
                
            for test_val in test_values:
                # Temporarily update h[i]
                h_current[i] = test_val
                
                # Project to feasible region (enforce constraints)
                h_current = project_to_feasible(h_current, n_points)
                
                # Evaluate new overlap
                new_c5 = evaluate_erdos_solution(h_current, dx)
                
                if new_c5 < best_c5_local:
                    best_c5_local = new_c5
                    best_value = test_val
                    
            # Restore best value found
            h_current[i] = best_value
            h_current = project_to_feasible(h_current, n_points)
            
            # Update if improvement
            if abs(best_c5_local - current_c5) > 1e-10:
                current_c5 = best_c5_local
                improved = True
                
            # Move to next index
            if time() - start_time > budget_s * 0.9:
                break
                
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Break if no significant improvement
        if iteration > 5 and abs(last_improvement - current_c5) < tolerance:
            # Check for stagnation
            if iteration > 10:
                break
        last_improvement = current_c5
        
        n_iter += 1
        
        # Occasionally refine with SLSQP
        if iteration % 20 == 0 and iteration > 0:
            # SLSQP refinement
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            try:
                res = minimize(refine_objective, h_current, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 20, 'ftol': 1e-8})
                if res.success:
                    h_current = project_to_feasible(res.x, n_points)
                    current_c5 = evaluate_erdos_solution(h_current, dx)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_current.copy()
            except:
                pass
    
    # Final refinement with SLSQP using best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
                        
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
