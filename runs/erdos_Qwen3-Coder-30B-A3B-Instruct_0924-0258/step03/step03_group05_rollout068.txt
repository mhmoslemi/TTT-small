import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [35]
    if budget_s > 400:
        resolutions.append(70)
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Initialize and improve solution
        h_current = best_h.copy() if n_points == 35 else np.interp(
            np.linspace(0, 2, n_points), 
            np.linspace(0, 2, len(best_h)), 
            best_h
        )
        if n_points > len(h_current):
            h_current = np.pad(h_current, (0, n_points - len(h_current)), 'constant')
        h_current = h_current[:n_points]
        
        # Normalize
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Optimization: Use a structural approach with input validation
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple local optimizations with different starting points
        for trial in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb current solution
            h_trial = h_current.copy()
            h_trial += np.random.normal(0, 0.02, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            try:
                res = minimize(objective, h_trial, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined
            except:
                pass
        
        # Attempt advanced dual approach
        c_low = 0.37
        c_high = 0.38085
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            c_mid = (c_low + c_high) / 2.0
            # Simple feasibility check through projection
            h_test = best_h.copy()
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            c5_test = evaluate_erdos_solution(h_test, dx)
            
            if c5_test <= c_mid:
                c_high = c_mid
                best_c5 = min(best_c5, c5_test)
            else:
                c_low = c_mid
        
        # Final check with solver
        try:
            h_best = best_h.copy()
            h_best = np.clip(h_best, 0, 1)
            h_best = h_best * (n_points / 2) / np.sum(h_best)
            final_c5 = evaluate_erdos_solution(h_best, dx)
            if final_c5 < best_c5:
                best_h = h_best
                best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
