import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    w = 0.729
    c1 = 1.49445
    c2 = 1.49445
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    
    for _ in range(swarm_size):
        h = project_to_feasible(np.random.rand(n_points) * 0.5 + 0.25, n_points)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest_positions.append(h.copy())
        pbest_scores.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Apply constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_scores[i] = current_score
            
            # Update global best
            if current_score < gbest_score:
                gbest_position = particles[i].copy()
                gbest_score = current_score
        
        # Early stopping
        if iteration > 50 and gbest_score < 0.38085:
            break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def objective(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
            
        # Gradient-based refinement
        h_refine = gbest_position.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient approximation
            epsilon = 1e-4
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += epsilon
                h_plus = project_to_feasible(h_plus, n_points)
                
                h_minus = h_refine.copy()
                h_minus[i] -= epsilon
                h_minus = project_to_feasible(h_minus, n_points)
                
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
            
            # Update with gradient
            learning_rate = 1e-3
            h_refine = h_refine - learning_rate * grad
            h_refine = project_to_feasible(h_refine, n_points)
            
            # Check improvement
            current_score = objective(h_refine)
            if current_score < best_c5:
                best_h = h_refine.copy()
                best_c5 = current_score
    
    return (list(best_h), best_c5, n_points)
