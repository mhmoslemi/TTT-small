import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            # Add small random noise
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = project_to_feasible(h, n_points)
        else:
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
        population.append(h)
    
    # DE loop
    last_improvement = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
            last_improvement = iteration
        
        # Stop if no improvement for a while
        if iteration - last_improvement > 100:
            break
            
        new_population = []
        
        for i in range(population_size):
            # Select three different individuals
            idxs = list(range(population_size))
            idxs.remove(i)
            a, b, c = random.sample(idxs, 3)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if random.random() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Project to feasible region
            trial = project_to_feasible(trial, n_points)
            
            # Selection: keep better of trial and original
            trial_fitness = evaluate_erdos_solution(trial, dx)
            original_fitness = fitness[i]
            
            if trial_fitness < original_fitness:
                new_population.append(trial)
            else:
                new_population.append(population[i])
                
        population = new_population
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:

            return evaluate_erdos_solution(h, dx)

                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
