import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize PSO
    positions = np.random.uniform(0, 1, (swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    personal_best_pos = positions.copy()
    personal_best_scores = np.full(swarm_size, np.inf)
    global_best_pos = None
    global_best_score = np.inf
    
    # Initialize swarm
    for i in range(swarm_size):
        h = project_to_feasible(positions[i], n_points)
        score = evaluate_erdos_solution(h, dx)
        personal_best_scores[i] = score
        if score < global_best_score:
            global_best_score = score
            global_best_pos = h.copy()
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best_pos[i] - positions[i]) +
                           social * r2 * (global_best_pos - positions[i]))
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = project_to_feasible(positions[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(positions[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_pos[i] = positions[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_pos = positions[i].copy()
    
    # Refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Convert global best to a refineable format
        refined_h = global_best_pos.copy()
        
        def refinement_objective(h):
            h_projected = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_projected, dx)
        
        # Use L-BFGS-B for refinement
        res = minimize(refinement_objective, refined_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            final_h = project_to_feasible(res.x, n_points)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
