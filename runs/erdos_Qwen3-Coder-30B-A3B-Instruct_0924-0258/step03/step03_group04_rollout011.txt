import numpy as np
from scipy.optimize import minimize
from time import time
from scipy import fftpack

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_scores.append(score)
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best[i] - swarm[i]) + 
                           social * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Keep within bounds
            swarm[i] = project_to_feasible(swarm[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = swarm[i].copy()
                global_best_score = score
        
        # Occasionally refine with gradient descent
        if iter_count % 5 == 0 and time() - start_time < budget_s * 0.9:
            # Refine the best solution found by PSO
            def refine_obj(h):
                h_proj = project_to_feasible(h, n_points)
                return evaluate_erdos_solution(h_proj, dx)
            
            res = minimize(refine_obj, global_best, method='L-BFGS-B',
                          bounds=[(0,1) for _ in range(n_points)],
                          options={'maxiter': 30})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < global_best_score:
                    global_best = h_refined.copy()
                    global_best_score = c5_refined
    
    # Final refinement with gradient descent if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, global_best, method='L-BFGS-B',
                      bounds=[(0,1) for _ in range(n_points)],
                      options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < global_best_score:
                global_best = h_final.copy()
                global_best_score = c5_final
    
    return (list(global_best), global_best_score, n_points)
