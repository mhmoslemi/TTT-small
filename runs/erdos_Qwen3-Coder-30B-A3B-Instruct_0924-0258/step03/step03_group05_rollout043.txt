import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Perturbation and refinement parameters
    max_perturbations = 200
    perturbation_magnitude = 0.05
    max_local_iters = 50
    max_refinements = 10
    
    # Helper function for perturbing h while maintaining constraints
    def perturb_h(h_vec, magnitude):
        h_new = h_vec + np.random.normal(0, magnitude, len(h_vec))
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Try perturbation-based refinement
    for iteration in range(max_perturbations):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_perturbed = perturb_h(best_h, perturbation_magnitude)
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed
    
    # Local optimization refinement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    for iteration in range(max_refinements):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': max_local_iters, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Ensure final constraints are met
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    best_h = np.clip(best_h, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
