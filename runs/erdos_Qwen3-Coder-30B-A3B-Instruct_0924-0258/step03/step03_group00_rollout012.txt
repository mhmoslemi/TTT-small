import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    runtime_limit = budget_s * 0.95

    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use PSO instead of gradient descent for better exploration
    def pso_optimization():
        # PSO parameters
        n_particles = 50
        max_iter = 200
        w = 0.7
        c1 = 1.5
        c2 = 1.5
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initialize particles
        particles = [np.random.rand(n_points) for _ in range(n_particles)]
        velocities = [np.random.randn(n_points) for _ in range(n_particles)]
        personal_best = [p.copy() for p in particles]
        personal_best_scores = [evaluate_erdos_solution(p, dx) for p in particles]
        global_best = deepcopy(personal_best[np.argmin(personal_best_scores)])
        global_best_score = min(personal_best_scores)
        
        for _ in range(max_iter):
            if time.time() - start_time > runtime_limit:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best[i] - particles[i]) + 
                                c2 * r2 * (global_best - particles[i]))
                
                # Update position
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                
                # Apply sum constraint
                h_sum = np.sum(particles[i])
                if h_sum > 0:
                    particles[i] = particles[i] * (n_points / 2) / h_sum
                
                # Evaluate fitness
                score = evaluate_erdos_solution(particles[i], dx)
                
                # Update personal best
                if score < personal_best_scores[i]:
                    personal_best[i] = particles[i].copy()
                    personal_best_scores[i] = score
                    
                    if score < global_best_score:
                        global_best = particles[i].copy()
                        global_best_score = score
        
        return global_best, global_best_score
    
    def local_search(h_init):
        # Use a hybrid approach: local search around good solutions
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Try multiple restarts
            best_local = h_init.copy()
            best_local_c5 = obj_func(h_init)
            num_restarts = 10
            
            for i in range(num_restarts):
                if time.time() - start_time > runtime_limit:
                    break
                # Perturb current best
                h_perturb = np.clip(h_init + np.random.normal(0, 0.02, n_points), 0, 1)
                h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb) if np.sum(h_perturb) > 0 else h_perturb
                
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_local_c5:
                        best_local = res.x.copy()
                        best_local_c5 = c5_test
                        
            return best_local, best_local_c5
        
        except:
            return h_init, obj_func(h_init)
    
    # Main optimization loop
    for iteration in range(50):
        if time.time() - start_time > runtime_limit:
            break
            
        # Start with best solution
        h_start = best_h.copy()
        
        # PSO to explore global
        pso_h, pso_c5 = pso_optimization()
        
        # Local search on best
        refined_h, refined_c5 = local_search(pso_h)
        
        # Selection
        if refined_c5 < best_c5:
            best_h = refined_h.copy()
            best_c5 = refined_c5
        
        # Occasionally reset with new random sample
        if np.random.rand() < 0.1:
            h_new = np.random.rand(n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new) if np.sum(h_new) > 0 else h_new
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
    
    return (list(best_h), best_c5, n_points)
