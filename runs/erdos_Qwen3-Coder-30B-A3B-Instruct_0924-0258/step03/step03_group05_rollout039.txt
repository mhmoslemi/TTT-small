import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a gradient-based approach with careful bounds
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Project to valid range [0,1]
    def project_h(h_vec):
        return np.clip(h_vec, 0, 1)
    
    # Normalize to satisfy integral constraint
    def normalize_h(h_vec):
        h_sum = np.sum(h_vec)
        if h_sum > 0:
            return h_vec * (n_points / 2) / h_sum
        else:
            return h_vec
    
    # Local optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    # Fine-tune with multiple restarts
    max_restarts = 20
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = project_h(h_perturbed)
        h_perturbed = normalize_h(h_perturbed)
        
        try:
            res = minimize(objective, h_perturbed, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_candidate = project_h(res.x)
                h_candidate = normalize_h(h_candidate)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass
    
    # Additional refinement using SLSQP
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = project_h(res.x)
            h_final = normalize_h(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Ensure final solution
    best_h = project_h(best_h)
    best_h = normalize_h(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
