import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.2
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize
        return np.clip(h, 0, 1)
    
    def create_population():
        population = [h_start.copy()]
        for _ in range(population_size - 1):
            individual = create_individual()
            population.append(individual)
        return population
    
    # Fitness evaluation
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    # Tournament selection
    def tournament_selection(population, fitnesses, k=5):
        tournament_indices = np.random.choice(len(population), k)
        tournament_fitness = [fitnesses[i] for i in tournament_indices]
        best_index = tournament_indices[np.argmin(tournament_fitness)]
        return population[best_index]
    
    # Crossover
    def crossover(parent1, parent2):
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.1)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # Main GA loop
    population = create_population()
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Find best individual
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final optimization with SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
