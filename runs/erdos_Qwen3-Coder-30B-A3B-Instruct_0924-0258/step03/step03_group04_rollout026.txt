import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create periodic sinusoidal pattern
        x = np.linspace(0, 2*np.pi, n_points, endpoint=False)
        h_start = 0.5 * (1 + np.sin(x))  # Maps to [0,1]
        # Ensure sum constraint
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start optimization with a loop for better exploration
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify the worst shifts (those contributing most to C5)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Only consider non-zero shifts (not including 0 shift)
        corr_worst = corr[n_points:]  # Exclude first element which is 0 shift
        # Select top 15% worst shifts
        k_worst = np.argsort(corr_worst)[-int(0.15 * n_points):]
        
        # Create a convex feasibility problem with only worst shifts constrained
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        for k_idx in k_worst:
            # For shift k (k_idx), compute correlation
            k = k_idx + 1  # since k_idx=0 corresponds to k=1
            corr_sum = 0
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
            constraints.append(corr_sum <= best_c5 * dx * valid_len)
            
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feas = h_var.value
                if h_feas is not None:
                    # Refine with gradient descent to improve C5
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    # Use the feasible solution as starting point for gradient descent
                    res = minimize(refinement_obj, h_feas, method='L-BFGS-B', 
                                   bounds=[(0,1) for _ in range(n_points)], 
                                   options={'maxiter': 50, 'ftol': 1e-9, 'gtol': 1e-9})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
        except:
            pass
        
        # Occasionally add random search for escape from local optima
        if iteration % 20 == 0 and iteration > 0:
            # Perturb best solution
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            h_perturb = project_to_feasible(h_perturb, n_points)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5_perturb
    
    # Final refinement  
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-10})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
