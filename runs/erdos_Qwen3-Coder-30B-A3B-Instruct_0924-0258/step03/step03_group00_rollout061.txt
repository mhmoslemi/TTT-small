import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for genetic algorithm
    population_size = 500
    generations = 100
    elite_size = 20
    mutation_rate = 0.05
    crossover_rate = 0.8
    num_parents = 20
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def create_population():
        return [create_individual() for _ in range(population_size)]
    
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    def selection(population, fitnesses):
        sorted_indices = np.argsort(fitnesses)
        return [population[i] for i in sorted_indices[:num_parents]]
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Simple uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        
        # Normalize
        sum1 = np.sum(child1)
        if sum1 > 0:
            child1 = child1 * (n_points / 2) / sum1
        sum2 = np.sum(child2)
        if sum2 > 0:
            child2 = child2 * (n_points / 2) / sum2
            
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                mutated[i] = np.random.rand()
        sum_m = np.sum(mutated)
        if sum_m > 0:
            mutated = mutated * (n_points / 2) / sum_m
        return mutated
    
    # Local refinement
    def refine_individual(h):
        def obj_func(x):
            h_clipped = np.clip(x, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_func, h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                return res.x
        except:
            pass
        return h
    
    population = create_population()
    best_in_generation = []
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        fitnesses = [evaluate_individual(ind) for ind in population]
        best_fitness = min(fitnesses)
        best_in_generation.append(best_fitness)
        
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[np.argmin(fitnesses)].copy()
        
        # Selection
        parents = selection(population, fitnesses)
        
        # Create new population
        new_population = deepcopy(parents[:elite_size])
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final local refinement on best individual
    if time.time() - start_time < budget_s * 0.95:
        refined = refine_individual(best_h)
        refined_c5 = evaluate_erdos_solution(refined, dx)
        if refined_c5 < best_c5:
            best_h = refined
            best_c5 = refined_c5
    
    # Ensure final constraints
    h_final = best_h.copy()
    h_sum = np.sum(h_final)
    if h_sum > 0:
        h_final = h_final * (n_points / 2) / h_sum
    
    return (list(h_final), best_c5, n_points)
