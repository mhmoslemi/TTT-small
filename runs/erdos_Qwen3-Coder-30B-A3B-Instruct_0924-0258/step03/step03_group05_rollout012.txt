import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate a wavelet basis (Haar-like)
    def get_haar_basis(n):
        # Simple 1D Haar basis
        basis = []
        for i in range(n):
            wavelet = np.zeros(n)
            wavelet[i] = 1.0
            basis.append(wavelet)
        return np.array(basis)
    
    # Iterative refinement with coordinate descent
    max_iterations = 500
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try to optimize each coefficient
        current_h = best_h.copy()
        for i in range(n_points):
            if time() - start_time > budget_s * 0.95:
                break
            # Try a range of values for this coefficient
            original_val = current_h[i]
            best_val = original_val
            best_overlap = best_c5
            
            # Small grid search around current value
            test_values = np.linspace(max(0, original_val - 0.1), 
                                    min(1, original_val + 0.1), 11)
            
            for test_val in test_values:
                current_h[i] = test_val
                # Normalize to satisfy integral constraint
                current_h = current_h * (n_points / 2) / np.sum(current_h)
                current_h = np.clip(current_h, 0, 1)
                overlap = evaluate_erdos_solution(current_h, dx)
                if overlap < best_overlap:
                    best_overlap = overlap
                    best_val = test_val
            
            # Accept the best value found for this coordinate
            current_h[i] = best_val
            # Renormalize
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            current_h = np.clip(current_h, 0, 1)
            
            # Update if this improved the overall solution
            if best_overlap < best_c5:
                best_h = current_h
                best_c5 = best_overlap
        
        # After a few iterations, try to "flatten" the overlap spectrum
        if iteration % 10 == 0:
            # Refine with convex optimization for overlap constraints
            def solve_feasibility(h_guess, c_val):
                h = cp.Variable(n_points)
                constraints = [
                    h >= 0,
                    h <= 1,
                    cp.sum(h) * dx == 1
                ]
                
                # Add overlap constraints for all shifts
                for k in range(n_points):
                    if k == 0:
                        continue
                    valid_len = n_points - k
                    if valid_len <= 0:
                        break
                    overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_val
                    constraints.append(overlap_constraint)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6)
                    if prob.status in ["optimal", "optimal_inaccurate"]:
                        return h.value
                except:
                    pass
                return None
            
            # Try a projection if we're close to a good bound
            h_feasible = solve_feasibility(best_h, best_c5)
            if h_feasible is not None:
                best_h = np.clip(h_feasible, 0, 1)
                new_c5 = evaluate_erdos_solution(best_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
    
    # Final optimization using scipy
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
