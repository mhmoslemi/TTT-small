import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    # Differential Evolution parameters
    popsize = 50
    F = 0.5
    CR = 0.9
    maxiter = 1000
    
    # Initialize population
    population = []
    for _ in range(popsize):
        if 'initial_h_values' in globals():

            h += np.random.normal(0, 0.05, n_points)

        population.append(h)

    best_individual = best_h.copy()
    best_fitness = best_c5

    iter_no_improve = 0

        if time() - start_time > budget_s * 0.85:
            break

        fitnesses = []

        current_best_idx = np.argmin(fitnesses)

            mutant = population[a] + F * (population[b] - population[c])

            crossover_mask = np.random.rand(n_points) < CR
            trial = np.where(crossover_mask, mutant, population[i])

            new_population.append(trial)
