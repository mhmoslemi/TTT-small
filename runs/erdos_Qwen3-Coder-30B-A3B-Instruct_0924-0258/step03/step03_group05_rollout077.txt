import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try a refined search using a wider range of points
    for n_points_try in [35, 50, 70]:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx_try = 2.0 / n_points_try
        h_try = np.interp(np.linspace(0, 2, n_points_try), 
                          np.linspace(0, 2, len(best_h)), 
                          best_h)
        
        # Local optimization with SLSQP
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx_try)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_try / 2}]
        bounds = [(0, 1) for _ in range(n_points_try)]
        
        try:
            res = minimize(objective, h_try, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points_try / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx_try)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    n_points = n_points_try
        except:
            pass

    # Final refinement using a hybrid approach
    if time() - start_time < budget_s * 0.9:
        # Objective function
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple initializations to avoid local minima
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Perturb the current best solution
            h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            try:
                res = minimize(objective, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass

    # Ensuring constraints are met at the end
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    c5_bound = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_bound, n_points)
