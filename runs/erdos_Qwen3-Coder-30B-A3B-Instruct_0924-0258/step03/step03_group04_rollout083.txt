import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def approximate_gradient(h, k_star, n_points, dx):
    """Approximate subgradient for the k_star shift"""
    grad = np.zeros(n_points)
    if k_star == 0:
        return grad
    
    valid_len = n_points - k_star
    if valid_len > 0:
        # Only consider valid overlap computation
        if k_star > 0:
            # Compute contribution from h[i] * (1 - h[i + k_star])
            grad[:valid_len] += (1 - h[k_star:k_star+valid_len])
        else:
            # For negative lag, which doesn't apply here
            pass
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with perturbations
    lr_start = 1e-3
    max_iter = int(1e5)
    p_perturb = 0.1
    perturb_magnitude = 0.05
    
    h = h_start.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all k
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        # Center at zero shift, so we're looking at the full correlation, k from -n+1 to n-1
        k_star = max_idx - (n_points - 1)
        
        # Compute gradient for k_star
        grad = approximate_gradient(h, k_star, n_points, dx)
        
        # Update h with gradient step
        h_new = h - lr_start * grad
        
        # Perturb randomly
        if np.random.random() < p_perturb:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_new += perturb
        
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
