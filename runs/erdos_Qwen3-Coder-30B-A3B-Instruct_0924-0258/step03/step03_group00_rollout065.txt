import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization using Fourier-based parametrization
    # We'll use a cosine series to parameterize h(x)
    num_fourier_terms = 5
    
    def h_from_coeffs(coeffs):
        # Map coefficients to h values using cosine series + logistic
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + np.sum([coeffs[i+1] * np.cos((i+1) * np.pi * x) 
                               for i in range(num_fourier_terms)], axis=0)
        h_vals = 1.0 / (1.0 + np.exp(-g))
        # Normalize so that integral equals 1
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return np.clip(h_vals, 0, 1)
    
    def objective(coeffs):
        h_vals = h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h_vals, dx)
    
    # Bounds for coefficients
    bounds = [(-10, 10) for _ in range(num_fourier_terms + 1)]
    
    # Run multiple random restarts
    num_restarts = 20
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Random starting point
        coeffs_start = np.random.uniform(-2, 2, num_fourier_terms + 1)
        
        try:
            result = minimize(objective, coeffs_start, method='SLSQP',
                            bounds=bounds, options={'maxiter': 100, 'ftol': 1e-6})
            
            if result.success:
                h_test = h_from_coeffs(result.x)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
        except:
            continue
    
    # Final refinement
    if time.time() - start_time < budget_s * 0.95:
        # Try local optimization starting from best solution
        coeffs_best = np.zeros(num_fourier_terms + 1)
        h_refined = h_from_coeffs(coeffs_best)
        # Simple gradient-free optimization
        for _ in range(50):
            if time.time() - start_time > budget_s * 0.98:
                break
            # Random perturbation
            perturbation = np.random.normal(0, 0.1, num_fourier_terms + 1)
            coeffs_perturbed = coeffs_best + perturbation
            h_test = h_from_coeffs(coeffs_perturbed)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                coeffs_best = coeffs_perturbed
                best_h = h_test.copy()
                best_c5 = c5_test
    
    return (list(best_h), best_c5, n_points)
