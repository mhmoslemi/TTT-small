import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 20
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # GA parameters
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1
    
    for gen in range(30):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Selection (top 50%)
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:population_size//2]]
        
        # Crossover
        new_population = selected.copy()
        for _ in range(population_size // 2):
            if random.random() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                noise = np.random.normal(0, 0.03, n_points)
                new_population[i] = new_population[i] + noise
                new_population[i] = np.clip(new_population[i], 0, 1)
                new_population[i] = new_population[i] * (n_points / 2) / np.sum(new_population[i])
                
        population = new_population
        
        # Local SQP refinement
        if gen % sqp_interval == 0:
            for i in range(min(3, len(population))):
                h_init = population[i].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
