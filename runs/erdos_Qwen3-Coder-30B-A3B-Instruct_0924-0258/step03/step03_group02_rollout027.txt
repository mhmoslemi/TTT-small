import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    for i in range(n_particles):
        particles[i] = project_to_feasible(particles[i], n_points)
    
    velocities = np.random.uniform(-v_max, v_max, (n_particles, n_points))
    
    # Personal best
    p_best_positions = particles.copy()
    p_best_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Global best
    global_best_idx = np.argmin(p_best_values)
    g_best_position = p_best_positions[global_best_idx].copy()
    g_best_value = p_best_values[global_best_idx]
    
    # PSO iterations
    no_improvement_count = 0
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best_positions[i] - particles[i]) + 
                           c2 * r2 * (g_best_position - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best_values[i] = current_value
                p_best_positions[i] = particles[i].copy()
                
                # Update global best
                if current_value < g_best_value:
                    g_best_value = current_value
                    g_best_position = particles[i].copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        
        # Early stopping if no improvement
        if no_improvement_count > 100:
            break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, g_best_position, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < g_best_value:
                    g_best_position = h_final
                    g_best_value = c5_final
        except Exception:
            pass
    
    # Ensure best_h meets the constraint
    best_h = project_to_feasible(g_best_position, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
