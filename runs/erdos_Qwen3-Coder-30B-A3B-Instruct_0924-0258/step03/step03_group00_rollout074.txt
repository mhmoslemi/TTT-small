import numpy as np
import time
from scipy.optimize import minimize
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple initial point
    
    # Ensure the integral constraint
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5 = min(best_c5, 0.38085)  # Upper bound
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = deque(maxlen=tabu_list_size)
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Perturbation parameters
    perturb_rate = 0.2
    perturb_magnitude = 0.1
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.85:
        if no_improvement_count >= max_no_improvement:
            break
            
        # Generate neighbors by perturbing random elements
        neighbors = []
        for _ in range(20):  # Generate 20 neighbors
            h_neighbor = best_h.copy()
            
            # Perturb random elements
            num_to_perturb = max(1, int(perturb_rate * n_points))
            indices_to_perturb = np.random.choice(n_points, num_to_perturb, replace=False)
            
            for i in indices_to_perturb:
                perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude)
                h_neighbor[i] += perturb
                h_neighbor[i] = np.clip(h_neighbor[i], 0, 1)
            
            # Project to satisfy constraint
            h_sum = np.sum(h_neighbor)
            if h_sum > 0:
                h_neighbor = h_neighbor * (n_points / 2) / h_sum
            
            neighbors.append(h_neighbor)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_test in neighbors:
            # Check if this neighbor is in tabu list
            config_tuple = tuple(np.round(h_test, decimals=6))
            if config_tuple in tabu_list:
                # Use a small random escape cost
                escape_cost = np.random.uniform(0.01, 0.1)
                test_c5 = evaluate_erdos_solution(h_test, dx) + escape_cost
            else:
                test_c5 = evaluate_erdos_solution(h_test, dx)
            
            if test_c5 < best_neighbor_c5:
                best_neighbor = h_test
                best_neighbor_c5 = test_c5
        
        # Update tabu list and solution
        if best_neighbor is not None:
            tabu_list.append(tuple(np.round(best_h, decimals=6)))
            best_h = best_neighbor.copy()
            
            if best_neighbor_c5 < best_c5:
                best_c5 = best_neighbor_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
    
    # Local refinement with gradient descent
    if time.time() - start_time < budget_s * 0.98:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations
        for _ in range(10):
            if time.time() - start_time > budget_s * 0.98:
                break
            try:
                res = minimize(obj_func, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    test_c5 = obj_func(res.x)
                    if test_c5 < best_c5:
                        best_h = res.x.copy()
                        best_c5 = test_c5
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
