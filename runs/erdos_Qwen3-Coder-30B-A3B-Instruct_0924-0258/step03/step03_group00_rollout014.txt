import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default: 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm for piecewise constant functions
    pop_size = 500
    generations = 100
    num_parents = int(0.2 * pop_size)
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    def generate_individual():
        # Create a binary vector for whether segment is 0 or 1
        v = np.random.choice([0, 1], size=n_points)
        # Create breakpoints with random positions
        bp = np.sort(np.random.rand(n_points - 1)) * 2.0
        bp = np.concatenate([[0], bp, [2.0]])
        return bp, v
    
    def individual_to_h(bp, v):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = v[i]
        return h
    
    def fitness(individual):
        bp, v = individual
        h = individual_to_h(bp, v)
        return evaluate_erdos_solution(h, dx)
    
    def crossover(parent1, parent2):
        bp1, v1 = parent1
        bp2, v2 = parent2
        
        # Crossover breakpoints
        if np.random.rand() < crossover_rate:
            idx = np.random.randint(1, n_points - 1)
            bp1_new = np.concatenate([bp1[:idx], bp2[idx:]])
            bp2_new = np.concatenate([bp2[:idx], bp1[idx:]])
        else:
            bp1_new, bp2_new = bp1.copy(), bp2.copy()
        
        # Crossover values
        if np.random.rand() < crossover_rate:
            idx = np.random.randint(1, n_points)
            v1_new = np.concatenate([v1[:idx], v2[idx:]])
            v2_new = np.concatenate([v2[:idx], v1[idx:]])
        else:
            v1_new, v2_new = v1.copy(), v2.copy()
            
        return (bp1_new, v1_new), (bp2_new, v2_new)
    
    def mutate(individual):
        bp, v = individual
        # Mutate breakpoints
        for i in range(1, n_points - 1):
            if np.random.rand() < mutation_rate:
                bp[i] += np.random.normal(0, 0.05)
                bp[i] = max(0, min(2, bp[i]))
        
        # Sort breakpoints
        bp = np.sort(bp)
        
        # Mutate values
        for i in range(n_points):
            if np.random.rand() < 0.05:
                v[i] = 1 - v[i]
        return (bp, v)
    
    def init_population():
        population = []
        for _ in range(pop_size):
            individual = generate_individual()
            population.append(individual)
        return population
    
    def repair(individual):
        bp, v = individual
        # Ensure bounds
        bp = np.clip(bp, 0, 2)
        bp = np.sort(bp)
        bp[0] = 0
        bp[-1] = 2
        
        # Ensure valid shape
        h = individual_to_h(bp, v)
        
        # Renormalize 
        int_sum = np.sum(h) * dx
        if int_sum > 0:
            h = h * (1.0 / int_sum)
        
        return (bp, v)
    
    # Main GA loop
    population = init_population()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness_scores = [(individual, fitness(individual)) for individual in population]
        fitness_scores.sort(key=lambda x: x[1])
        
        # Store best individual so far
        if fitness_scores[0][1] < best_c5:
            best_h = individual_to_h(fitness_scores[0][0][0], fitness_scores[0][0][1])
            best_c5 = fitness_scores[0][1]
        
        # Selection: keep top 20%
        parents = [ind[0] for ind in fitness_scores[:num_parents]]
        
        # Create new population
        new_population = parents[:]  # Elitism
        
        # Add mutated individuals
        while len(new_population) < pop_size:
            # Parent selection
            p1 = parents[np.random.randint(0, len(parents))]
            p2 = parents[np.random.randint(0, len(parents))]
            
            # Crossover
            c1, c2 = crossover(p1, p2)
            
            # Mutate
            c1 = mutate(c1)
            c2 = mutate(c2)
            
            new_population.extend([c1, c2])
            
        # Trim to correct size
        population = new_population[:pop_size]
    
    # Local refinement of best
    if time.time() - start_time < budget_s * 0.95:
        def local_obj(h):
            h_clip = np.clip(h, 0, 1)
            norm_sum = np.sum(h_clip) * dx
            if norm_sum > 0:
                h_norm = h_clip * (1.0 / norm_sum)
            else:
                h_norm = h_clip
            return evaluate_erdos_solution(h_norm, dx)
        
        def constraint_sum(h):
            return np.sum(h) * dx - 1.0
        
        # Refine the best solution found
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        x0 = best_h.copy()
        try:
            res = minimize(local_obj, x0, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_refined = local_obj(res.x)
                if c5_refined < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
