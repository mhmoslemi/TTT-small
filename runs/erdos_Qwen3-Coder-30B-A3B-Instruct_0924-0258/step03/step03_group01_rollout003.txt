import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Clustered initialization
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to sum to n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with Nelder-Mead variations
    population_size = 15
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    max_iter = 20
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all candidates
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Create new population using variation of Nelder-Mead
        new_population = [population[0].copy()]  # Best always survives
        
        # Generate variants
        for _ in range(population_size - 1):
            # Pick two distinct parents
            parents = np.random.choice(len(population), size=2, replace=False)
            h1, h2 = population[parents[0]], population[parents[1]]
            
            # Create offspring
            alpha = np.random.rand()
            child = alpha * h1 + (1 - alpha) * h2
            
            # Add noise
            noise = np.random.normal(0, 0.03, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
    
    # Local refinement with gradient-based methods
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine top solutions
        for i in range(min(3, len(population))):
            h_ref = population[i]
            h_ref = h_ref * (n_points/2) / np.sum(h_ref)  # Ensure sum constraint
            
            try:
                res = minimize(objective, h_ref, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
