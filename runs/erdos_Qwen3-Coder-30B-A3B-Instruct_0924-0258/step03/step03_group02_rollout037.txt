import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting points and refinements
    attempts = 0
    max_attempts = 5
    
    while attempts < max_attempts and (time() - start_time) < budget_s * 0.9:
        # Use a variation of the starting point for diversity
        h_init = h_start.copy()
        if attempts > 0:
            # Add small Gaussian noise for exploration
            noise = np.random.normal(0, 0.05, n_points)
            h_init = h_init + noise

        # Direct convex optimization approach with McCormick bounds

            return evaluate_erdos_solution(h, dx)
