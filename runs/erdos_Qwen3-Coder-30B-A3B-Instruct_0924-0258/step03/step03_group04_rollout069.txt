import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with sinusoidal and adjust to satisfy sum constraint
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temperature = 0.1
    min_temp = 1e-8
    cooling_rate = 0.99
    min_acceptance_rate = 0.05
    max_acceptance_rate = 0.5
    adapt_interval = 100
    
    # Main SA loop
    iteration = 0
    accept_count = 0
    total_iter = 0
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        total_iter += 1
        
        # Generate candidate
        h_candidate = best_h.copy()
        
        # Perturb about 5-10% of points
        num_perturb = max(1, int(n_points * 0.05 + random.random() * 0.05 * n_points))
        indices = random.sample(range(n_points), num_perturb)
        
        for idx in indices:
            # Add random perturbation
            perturbation = random.uniform(-0.2, 0.2)
            h_candidate[idx] += perturbation
            
        # Project to feasible set
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        delta = c5_candidate - best_c5
        
        # Accept or reject based on Metropolis criterion
        if delta < 0 or random.random() < np.exp(-delta / temperature):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            accept_count += 1
            
        # Update temperature
        if iteration % adapt_interval == 0:
            acceptance_rate = accept_count / adapt_interval
            if acceptance_rate < min_acceptance_rate:
                temperature *= 0.9
            elif acceptance_rate > max_acceptance_rate:
                temperature *= 1.1
            accept_count = 0
            
        # Cool temperature
        temperature *= cooling_rate
        if temperature < min_temp:
            temperature = min_temp
    
    # Final refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
