import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 50
    max_generations = 500
    mutation_rate = 0.2
    crossover_rate = 0.8
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Renormalize
        return np.clip(h, 0, 1)
    
    def create_pop():
        pop = [h_start.copy()] if 'initial_h_values' in globals() else []
        while len(pop) < pop_size:
            pop.append(create_individual())
        return pop
    
    # Tournament selection
    def tournament_selection(pop, fitnesses, k=5):
        selected_idx = random.randint(0, len(pop)-1)
        for _ in range(k-1):
            candidate_idx = random.randint(0, len(pop)-1)
            if fitnesses[candidate_idx] < fitnesses[selected_idx]:
                selected_idx = candidate_idx
        return pop[selected_idx]
    
    # One-point crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        return child1, child2
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.1)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # Evaluate fitness (lower is better)
    def evaluate_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Main GA loop
    population = create_pop()
    generation = 0
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
        generation += 1
    
    # Final refinement with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Try multiple restarts
        for _ in range(5):
            x0 = best_h + np.random.normal(0, 0.01, n_points)
            x0 = np.clip(x0, 0, 1)
            x0 = x0 * (n_points / 2) / np.sum(x0)
            
            res = minimize(objective, x0, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
