import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic initialization: sin-like but with proper sum constraint
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    max_iter = 10000
    iter_count = 0
    accept_count = 0
    accept_rate = 0.5
    current_h = h_start.copy()
    current_c5 = best_c5
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Generate neighbor solution
        h_candidate = current_h.copy()
        
        # Select random subset to perturb (5-10%)
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected indices
        for i in indices:
            # Random perturbation within feasible range
            delta = np.random.normal(0, 0.1)
            h_candidate[i] += delta
            
        # Project to feasible space
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_c5 = c5_candidate - current_c5
        if delta_c5 < 0 or np.random.rand() < math.exp(-delta_c5 / T):
            current_h = h_candidate.copy()
            current_c5 = c5_candidate
            accept_count += 1
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Update temperature
        iter_count += 1
        if iter_count % 100 == 0:
            if accept_count > 0:
                accept_rate = accept_count / 100.0
                if accept_rate < 0.05:
                    T *= 0.9  # Lower temperature if acceptance rate too low
                elif accept_rate > 0.3:
                    T *= 1.1  # Raise temperature if too high
            accept_count = 0
            
            # Avoid temperature from getting too small
            T = max(T_min, T * alpha)
            
            # Break early if temperature too small
            if T < T_min:
                break
    
    # Final L-BFGS refinement if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and best_c5 > 0.375:
        def refinement_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        try:
            res = minimize(refinement_obj, best_h, method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 100})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
