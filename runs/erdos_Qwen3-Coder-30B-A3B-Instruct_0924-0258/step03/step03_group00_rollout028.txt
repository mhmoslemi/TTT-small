import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Alternate search strategy:
    # Direct local optimization with adaptive restarts
    max_restarts = 20
    max_iter_per_restart = 100
    
    # Try different restart strategies
    restart_strategies = [
        lambda: h_start.copy(),
        lambda: np.random.uniform(0, 1, n_points),
        lambda: np.ones(n_points) * 0.5,
        lambda: np.clip(np.random.normal(0.5, 0.2, n_points), 0, 1),
    ]
    
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_restarts:
        # Select restart strategy
        strategy_idx = iter_count % len(restart_strategies)
        h_current = restart_strategies[strategy_idx]()
        
        # Normalize if needed
        h_current_sum = np.sum(h_current)
        if h_current_sum > 0:
            h_current = h_current * (n_points / 2) / h_current_sum
        
        # Local optimization with SLSQP
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run optimization
        res = minimize(obj_func, h_current, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter_per_restart, 'ftol': 1e-6})
        
        if res.success:
            c5_test = obj_func(res.x)
            if c5_test < best_c5:
                best_h = res.x.copy()
                best_c5 = c5_test
        iter_count += 1
    
    # Final round of refinement using the best solution so far
    if time.time() - start_time < budget_s * 0.99:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Fine-tune with small perturbations
        h_perturb = best_h + np.random.normal(0, 0.005, n_points)
        res = minimize(obj_func, h_perturb, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 50, 'ftol': 1e-7})
        
        if res.success:
            c5_test = obj_func(res.x)
            if c5_test < best_c5:
                best_h = res.x.copy()
                best_c5 = c5_test
    
    return (list(best_h), best_c5, n_points)
