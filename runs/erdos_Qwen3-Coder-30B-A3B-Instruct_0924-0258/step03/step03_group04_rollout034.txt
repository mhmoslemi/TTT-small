import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_subgradient(h, k, n_points):
    """Compute subgradient for shift k"""
    grad = np.zeros(n_points)
    if k == 0:
        return grad
    
    # For shift k, we look at the convolution term
    # The contribution to the correlation comes from h[i] * (1 - h[i+k])
    # which gives gradient term: -1 if i+k < n_points and h[i+k] is active
    
    # When k is positive, we use the correlation in forward direction:
    # For i = 0,...,n_points-k-1, we have correlation contribution:
    # h[i] * (1 - h[i+k])
    # Gradient wrt h[i] is - (1 - h[i+k])
    # Gradient wrt h[i+k] is - h[i]
    
    # But this is for maximum, so we need to be careful to identify 
    # which part of h contributes to the maximum.
    
    # For simplicity let's compute gradient for the entire convolution
    # and then compute its gradient contribution correctly.
    
    # For the problem at hand, it's better to do pairwise gradient
    # with respect to the identified maximum correlation
    # But we can't compute the gradient directly without knowing which term 
    # is the maximum. Let's compute numerically with small perturbations later.
    
    # Let's return zero gradient and handle the one in the optimization function
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Generate a simple but reasonable initial pattern
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Keep track of recent improvements for early stopping
    last_improvement = 0
    iter_count = 0
    
    # Use of a learning rate that decays with time
    base_lr = 1e-3
    lr_decay = 1e-5
    
    # Main optimization loop
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Every 50 iterations, try to restart with a random perturbation
        if iter_count % 50 == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturb = best_h + perturb
            h_perturb = project_to_feasible(h_perturb, n_points)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_c5 = c5_perturb
                best_h = h_perturb.copy()
                last_improvement = iter_count
                continue
                
        # Compute current overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_corr_val = corr[max_k]
        
        # Compute gradient of correlation for the selected shift
        # By approximation, what contributes most is the point where 
        # h[i] * (1 - h[i+k]) reaches its maximum
        
        # Create gradient for the max shift k
        grad = np.zeros(n_points)
        k = max_k - (n_points - 1)  # shift in range [-n+1, n-1]
        
        # For shift k:
        # grad[i] = d/dh[i] (sum over valid j, h[j] * (1 - h[j+k]))
        # which for j = i: - (1 - h[i+k]) 
        # Which we simplify using code more efficiently
        # Let's take a per-direction derivative approach
        
        # When we compute correlation, there are two parts:
        # Contribution to h[i] is - (1 - h[i + k]) when there's a valid i+k
        # Contribution to h[i + k] is - h[i] when there's a valid i 
        
        # We plug in the region where Correlation_{i} = h[i] * (1 - h[i+k]) is
        # maximized. This happens for the term corresponding to maximum exponential
        
        # Simple heuristic: for shift k (defining h in [0,2] with n_points),
        # positions in valid range are [0, n-k-1] for h[i]*h[i+k] 
        # So:
        valid_len = n_points - abs(k)
        if k > 0:
            # decreasing shift: from left, h[i] contributes 1-h[i+k]
            indices = np.arange(0, valid_len)
            if len(indices) > 0:
                grad[indices] = -(1 - best_h[indices+k])
        elif k < 0:
            # increasing shift: from right h[i+k] contributes -h[i] 
            indices = np.arange(-k, n_points)
            if len(indices) > 0:
                grad[indices] = -best_h[indices+k]
        else:
            # k = 0 case, ignore
            pass
                
        # Apply gradient update with weighted decrease
        lr = base_lr * np.exp(-lr_decay * iter_count)
        h_new = best_h - lr * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate the new configuration
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iter_count
        else:
            # Every 100 iter, or if no improvement in last 30 steps
            if iter_count % 100 == 0 or iter_count - last_improvement > 30:
                # Stochastic change
                h_rand = best_h.copy() + np.random.normal(0, 0.05, n_points)
                h_rand = project_to_feasible(h_rand, n_points)
                c5_rand = evaluate_erdos_solution(h_rand, dx)
                if c5_rand < best_c5:
                    best_c5 = c5_rand
                    best_h = h_rand.copy()
                    last_improvement = iter_count
    
    # Final verification with scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
