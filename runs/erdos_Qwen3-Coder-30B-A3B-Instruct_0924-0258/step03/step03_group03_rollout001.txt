import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_n_points = n_points

    # Use CVXPY for global relaxation (if feasible)
    # For large n_points it's too expensive; we'll use smaller sizes
    if n_points <= 50:
        try:
            # Create a candidate solution using CVXPY relaxation
            h_cvx = cp.Variable(n_points, name='h')
            # Define objective using a surrogate form

            constraints = [h_cvx >= 0, h_cvx <= 1]

            constraints.append(cp.sum(h_cvx) == n_points/2.0)
            
            # Define a simple convex approximation
            # We don't have a direct convex form for the full problem,

            # on a structured subset
            
            # Since exact CVXPY formulation for convolution max is challenging,
            # just use the current h_start to get neighborhood solution
            
        except:
            pass

    # Try gradient-based refinement on the best solution found so far
    max_iters = 1000
    step_size = 0.001
    
    # Define a smooth approximator to compute gradients
    def compute_gradient(h, dx):
        # Simple finite difference gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h)

    for iteration in range(max_iters):

            break

        gradient = compute_gradient(best_h, dx)
        
        # Update with gradient descent
        new_h = best_h - step_size * gradient
        new_h = project_to_feasible(new_h, n_points)
