import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Define segment count
    n_segments = 20
    n_points = 100
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():

        h = project_to_feasible(h, len(h))
    else:
        # Initialize symmetric construction
        h = np.ones(n_points) * 0.5
        # Slightly asymmetric to avoid trivial solutions
        h[:n_points//4] = 0.7
        h[n_points//4:3*n_points//4] = 0.3
        h[3*n_points//4:] = 0.5
        
    # Refine with optimization
    dx = 2.0 / n_points
    
    # Use a simple gradient descent with projection
    max_iter = 10000
    step_size = 0.001
    
    best_c5 = float('inf')
    best_h = None
    
    # Start optimization loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        score = evaluate_erdos_solution(h, dx)
        if score < best_c5:
            best_h = h.copy()
            best_c5 = score
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        base_score = evaluate_erdos_solution(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
            
        # Gradient step
        h = h - step_size * grad
        
        # Project back to feasible set
        h = project_to_feasible(h, n_points)
        
        # Early termination for better performance
        if it > 5000 and abs(score - best_c5) < 1e-6:

    # Refine with local search if possible
    if time() - start_time < budget_s * 0.95:

        for _ in range(20):

            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            score = evaluate_erdos_solution(h_perturbed, dx)
            if score < best_c5:
                best_h = h_perturbed
                best_c5 = score
    
    # Final evaluation
    if best_h is None:
        best_h = h
        
    final_score = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_score, n_points)
