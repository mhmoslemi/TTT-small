import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic-like pattern as initial guess
        h_start = 0.5 + 0.2 * np.sin(2 * np.pi * np.arange(n_points) / n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize learning rate and parameters
    lr = 1e-3
    decay_factor = 0.9995
    p_perturb = 0.1
    perturb_mag = 0.05
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        # Compute all correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Compute subgradient for the max shift
        subgradient = np.zeros(n_points)
        if max_shift >= 0:
            # Shift forward: h[i] contributes to (1-h[i+shift])
            valid_len = n_points - max_shift
            if valid_len > 0:
                subgradient[:valid_len] = 1 - best_h[max_shift:max_shift+valid_len]
        else:
            # Shift backward: h[i-shift] contributes to (1-h[i])
            valid_len = n_points + max_shift
            if valid_len > 0:
                subgradient[-max_shift:] = 1 - best_h[:-max_shift]
        
        # Update h with subgradient
        h_new = best_h - lr * subgradient
        h_new = project_to_feasible(h_new, n_points)
        
        # Possibly add random perturbation
        if np.random.random() < p_perturb:
            perturbation = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h_new = project_to_feasible(h_new + perturbation, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Learning rate decay
        lr *= decay_factor
        
        # Early termination
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement with gradient descent 
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
