import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 20
    num_generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_refinement_interval = 1
    
    # Initialize population
    def create_individual():
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    population = [create_individual() for _ in range(population_size)]
    
    # Genetic algorithm main loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: keep top 50%
        num_selected = population_size // 2
        selected = population[:num_selected]
        
        # Crossover and mutation to create new population
        new_population = selected.copy()
        
        # Crossover
        for i in range(num_selected, population_size):
            if random.random() < crossover_rate:
                parent1, parent2 = random.sample(selected, 2)
                crossover_point = random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = selected[random.randint(0, num_selected-1)].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_strength = 0.03
                noise = np.random.normal(0, mutation_strength, n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
            
            # Maintain constraint
            child = child * (n_points / 2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Local refinement with SQP every few generations
        if generation % sqp_refinement_interval == 0:
            for i in range(min(3, len(population))):
                h_candidate = population[i]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_candidate, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-6})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = objective(refined_h)
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
                        # Update population with better solution
                        population[i] = refined_h
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
