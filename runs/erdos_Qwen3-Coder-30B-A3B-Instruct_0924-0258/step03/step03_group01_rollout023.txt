import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iterations = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_vals.append(evaluate_erdos_solution(h, dx))
        
        if personal_best_vals[-1] < best_c5:
            best_h = h.copy()
            best_c5 = personal_best_vals[-1]
    
    # Global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO iterations
    for iter_num in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        
        w = w_max - (w_max - w_min) * iter_num / max_iterations
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = current_val
                
                if current_val < best_c5:
                    best_h = swarm[i].copy()
                    best_c5 = current_val
            
            # Update global best
            if current_val < global_best_val:
                global_best = swarm[i].copy()
                global_best_val = current_val
        
        # Local refinement every 5 iterations
        if iter_num % 5 == 0 and iter_num > 0:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time > 30:
                # Refine top 3 particles
                top_indices = np.argsort(personal_best_vals)[:3]
                for idx in top_indices:
                    h_refined = personal_best[idx].copy()
                    
                    def obj(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons = {'type': 'eq', 'fun': constraint_sum}
                    
                    try:
                        res = minimize(obj, h_refined, method='SLSQP', 
                                       bounds=bounds, constraints=cons, 
                                       options={'maxiter': 100, 'ftol': 1e-6})
                        if res.success:
                            h_refined = res.x
                            c5_refined = obj(h_refined)
                            if c5_refined < best_c5:
                                best_h = h_refined
                                best_c5 = c5_refined
                    except:
                        pass
    
    return (list(best_h), best_c5, n_points)
