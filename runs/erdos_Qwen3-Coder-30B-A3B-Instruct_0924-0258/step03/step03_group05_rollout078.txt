import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Increase resolution for better precision
    if best_c5 > 0.38:
        n_points = 70
        dx = 2.0 / n_points
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a simpler but effective optimization approach
    # First, try to improve the starting solution with direct optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use differential evolution for global search to avoid local minima
    try:
        from scipy.optimize import differential_evolution
        res = differential_evolution(objective, bounds, constraints=cons, 
                                   seed=seed, maxiter=100, popsize=15)
        if res.success:
            h_opt = np.clip(res.x, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except:
        pass
    
    # Local refinement with SLSQP if better solution not found
    if best_c5 > 0.38:
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_local = np.clip(res.x, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                c5_local = evaluate_erdos_solution(h_local, dx)
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
        except:
            pass
    
    # Try one more refinement if time allows
    if time() - start_time < budget_s * 0.7:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_lbfgs = np.clip(res.x, 0, 1)
                h_lbfgs = h_lbfgs * (n_points / 2) / np.sum(h_lbfgs)
                c5_lbfgs = evaluate_erdos_solution(h_lbfgs, dx)
                if c5_lbfgs < best_c5:
                    best_h = h_lbfgs
                    best_c5 = c5_lbfgs
        except:
            pass

    # Final bound checking
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
