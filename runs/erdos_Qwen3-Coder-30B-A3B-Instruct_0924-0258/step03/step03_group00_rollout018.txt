import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based optimization approach
    # Parameters
    d = 5  # Number of Fourier terms
    max_restarts = 20
    max_iter_per_restart = 30
    
    def fourier_to_h(coeffs):
        # Convert Fourier coefficients to function values
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + sum(coeffs[i+1] * np.cos((i+1) * np.pi * x) for i in range(d))
        h = 1.0 / (1.0 + np.exp(-g))
        return h
    
    def obj_func(coeffs):
        if time.time() - start_time > budget_s * 0.95:
            return np.inf
        
        h = fourier_to_h(coeffs)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for coefficients
    bounds = [(-10, 10) for _ in range(d + 1)]
    
    # Try multiple random restarts
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initial guess for Fourier coefficients
        coeffs_start = np.random.uniform(-5, 5, d + 1)
        
        try:
            res = minimize(obj_func, coeffs_start, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': max_iter_per_restart})
            
            if res.success:
                h_trial = fourier_to_h(res.x)
                h_trial = np.clip(h_trial, 0, 1)
                h_sum = np.sum(h_trial)
                if h_sum > 0:
                    h_trial = h_trial * (n_points / 2) / h_sum
                c5_trial = evaluate_erdos_solution(h_trial, dx)
                
                if c5_trial < best_c5:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                    
        except Exception:
            continue
    
    # Final refinement with local optimization
    if time.time() - start_time < budget_s * 0.95:
        def obj_func_local(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations with different starting points
        best_local = best_h.copy()
        best_local_c5 = best_c5
        num_restarts = 10
        
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.98:
                break
            # Generate a new random starting point
            h_perturb = np.random.rand(n_points) * 0.5 + 0.25
            try:
                res = minimize(obj_func_local, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func_local(res.x)
                    if c5_test < best_local_c5:
                        best_local = res.x.copy()
                        best_local_c5 = c5_test
            except:
                continue
        
        # Final validation
        if best_local_c5 < best_c5:
            best_h = best_local
            best_c5 = best_local_c5
    
    return (list(best_h), best_c5, n_points)
