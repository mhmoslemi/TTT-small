import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(values, beta=100):
    """Numerically stable implementation of smooth maximum"""
    values = np.array(values)
    max_val = np.max(values)
    exp_vals = np.exp(beta * (values - max_val))
    return max_val + np.log(np.sum(exp_vals)) / beta

def compute_gradients(h, dx, beta=100):
    """Compute gradients for a given function"""
    eps = 1e-5
    base_score = evaluate_erdos_solution(h, dx)
    
    grad_w = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad_w[i] = (score_plus - base_score) / eps

    return grad_w

def adaptive_refinement(h, dx, beta=100):
    """Refine the step function with additional segments"""
    # Determine where the function has highest variation
    # For this implementation, we'll split based on gradient magnitude
    grad_w = compute_gradients(h, dx, beta)
    
    # Find segments with highest gradient magnitude to split
    split_indices = np.argsort(-np.abs(grad_w))[:min(2, len(h)-1)]
    
    # Interpolate new values in between
    new_h = np.zeros(len(h) + len(split_indices))
    new_h[::2] = h
    for i, idx in enumerate(sorted(split_indices)):
        # Insert new point in between
        new_h[2*idx + 1] = (h[idx] + h[idx+1]) / 2
        
    # Project to feasible region again
    new_h = project_to_feasible(new_h, len(new_h))
    return new_h
