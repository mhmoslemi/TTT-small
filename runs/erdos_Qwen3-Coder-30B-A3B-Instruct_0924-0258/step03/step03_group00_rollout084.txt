import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    current_h = h_start.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_erdos_solution(current_h, dx)
    
    # Parameter settings
    max_iterations = 5000
    tabu_tenure = 10
    tabu_list = []
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Annealing schedule
        if temp < min_temp:
            temp = 1.0
            
        # Make a random move
        idx = np.random.randint(0, n_points)
        old_val = current_h[idx]
        
        # Try a new value (0, 0.5, or 1)
        new_val = np.random.choice([0.0, 0.5, 1.0])
        
        # Apply the change
        current_h[idx] = new_val
        
        # Check if this state is tabu
        state_hash = tuple(np.round(current_h, decimals=3))
        if state_hash in tabu_list:
            # Accept with low probability
            if np.random.rand() > 0.1:
                # Revert if tabu
                current_h[idx] = old_val
                continue
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(current_h, dx)
        
        # Accept or reject
        if c5_new < best_c5:
            # Accept improvement
            best_c5 = c5_new
            best_h = current_h.copy()
        else:
            # Accept with probability based on temperature
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                # Accept worse solution
                pass
            else:
                # Revert
                current_h[idx] = old_val
        
        # Update tabu list
        if len(tabu_list) >= tabu_tenure:
            tabu_list.pop(0)
        tabu_list.append(state_hash)
        
        # Update temperature
        temp *= cooling_rate
    
    # Final refinement with local search
    if time.time() - start_time < budget_s * 0.98:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple restarts
        for _ in range(20):
            if time.time() - start_time > budget_s * 0.98:
                break
            h_rand = np.random.rand(n_points)
            try:
                res = minimize(obj_func, h_rand, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = res.x.copy()
            except:
                continue
    
    # Final normalization
    h_normalized = np.clip(best_h, 0, 1)
    h_sum = np.sum(h_normalized)
    if h_sum > 0:
        h_normalized = h_normalized * (n_points / 2) / h_sum
    
    return (list(h_normalized), best_c5, n_points)
