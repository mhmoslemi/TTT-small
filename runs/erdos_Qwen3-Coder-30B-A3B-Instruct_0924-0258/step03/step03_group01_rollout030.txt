```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple local optimization with restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple restarts
    restarts = 10
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize with a slightly perturbed version of best_h
        h_restart = best_h + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Occasionally restart with a different approach
        if restart % 3 == 0:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 30, 'ftol': 1e-5})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Try convex feasibility with a tighter bound to improve
    c_mid = best_c5 * 0.99  # Start with a slightly tighter bound
    max_iterations = 5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Convex feasibility problem with current bound
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints - this is the critical step
        # Due to computational complexity, we use a selective approach
        try:
            # Sample some key shifts instead of all
            shifts_to_check = list(range(0, n_points, max(1, n_points//8))) + \
                             list(range(n_points//2 - 2, n_points//2 + 3))
            for k in shifts_to_check:
                # Calculate overlap for shift k
                if k < n_points:
                    # Forward shift
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    # Backward shift
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= c_mid * dx)
            
            # Create and solve the optimization problem
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
                    c_mid = c5_feasible * 0.98  # Tighten constraint for next iteration
        except Exception:
            pass
            
        # Try annealing-style decrease in constraint bound
        c_mid = max(c_mid * 0.95, best_c5 * 0.9)
        
    # Final refinement once more
    if time() - start_time < budget_s * 0.95:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
```
