import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_bounds = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Warm up with initial solution if available
    if 'initial_h_values' in globals():
        particles.append(h_start.copy())
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
        pbest.append(h_start.copy())
        pbest_values.append(evaluate_erdos_solution(h_start, dx))
    else:
        particles.append(np.random.random(n_points) * 0.5)
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
        pbest.append(particles[0].copy())
        pbest_values.append(evaluate_erdos_solution(particles[0], dx))
    
    # Initialize other particles
    for i in range(1, n_particles):
        h = np.random.random(n_points) * 0.5
        particles.append(h)
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
    
    # Track best global solution
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iteration
    iter_no_improve = 0
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        w = w_start - (w_start - w_end) * (it / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_value
                iter_no_improve = 0
            else:
                iter_no_improve += 1
                
        if iter_no_improve >= 100:
            break
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_final = gbest.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_final, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
            else:
                best_h = h_final
                best_c5 = evaluate_erdos_solution(h_final, dx)
        except Exception:
            best_h = h_final
            best_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(best_h), best_c5, n_points)
