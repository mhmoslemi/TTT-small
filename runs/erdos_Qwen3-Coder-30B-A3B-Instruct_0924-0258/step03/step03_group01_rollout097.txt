import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_max
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial construction
            particle = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random initialization
            particle = np.random.uniform(0, 1, n_points)
        
        # Ensure feasibility
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        
        swarm.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(evaluate_erdos_solution(particle, dx))
    
    # Global best tracking
    gbest_index = np.argmin(pbest_values)
    gbest = pbest[gbest_index].copy()
    gbest_value = pbest_values[gbest_index]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update inertia weight linearly
        inertia_weight = w_max - (w_max - w_min) * (iteration / max_iter)
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                inertia_weight * velocities[i] +
                c1 * r1 * (pbest[i] - swarm[i]) +
                c2 * r2 * (gbest - swarm[i])
            )
            
            # Update position
            swarm[i] += velocities[i]
            
            # Enforce bounds and normalization
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate solution
            current_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = swarm[i].copy()
                gbest_value = current_value
                if gbest_value < best_c5:
                    best_h = gbest.copy()
                    best_c5 = gbest_value
        
        # Local refinement every few iterations
        if iteration % 5 == 0 and iteration > 0:
            for i in range(min(3, swarm_size)):
                # Pick top particles for refinement
                idx = np.argsort(pbest_values)[:3][i]
                h_init = pbest[idx].copy()
                
                # SQP refinement
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
