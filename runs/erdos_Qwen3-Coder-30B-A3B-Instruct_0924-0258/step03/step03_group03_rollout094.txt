import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_grid_refinement(grid_x, values, gradient_magnitudes, max_segments=40):
    """Refine grid by splitting segments with highest gradient"""
    if len(grid_x) >= max_segments:
        return grid_x, values
    
    # Find indices of largest gradient magnitudes
    sorted_indices = np.argsort(gradient_magnitudes)[::-1]
    split_indices = sorted_indices[:min(2, len(sorted_indices))]
    
    # Split segments at midpoints of largest gradients
    new_grid_x = grid_x.copy()
    new_values = values.copy()
    
    # Insert new points at midpoints of selected segments
    insert_positions = []
    for idx in split_indices:
        if idx < len(grid_x) - 1:
            mid_x = (grid_x[idx] + grid_x[idx+1]) / 2
            insert_positions.append((mid_x, idx))
    
    # Sort by x position and insert
    insert_positions.sort()
    for i, (mid_x, idx) in enumerate(insert_positions):
        insert_pos = np.searchsorted(new_grid_x, mid_x)
        new_grid_x = np.insert(new_grid_x, insert_pos, mid_x)
        new_values = np.insert(new_values, insert_pos, new_values[idx])
    
    return new_grid_x, new_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initialize from parent construction or default

        n_points = len(h_start)

    else:
        n_points = 35
        values = np.ones(n_points) * 0.5
        x_points = np.linspace(0, 2, n_points + 1)
    
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    best_n_points = None
    
    # Initial evaluation
    h_current = project_to_feasible(values, n_points)
    current_c5 = evaluate_erdos_solution(h_current, dx)
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
        best_n_points = n_points
    
    # Gradient descent parameters
    max_iter = 1000
    learning_rate = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute finite difference gradients
        grad_w = np.zeros_like(values)
        eps = 1e-5
        
        base_score = evaluate_erdos_solution(h_current, dx)
        
        # Approximate gradient w.r.t. values
        for i in range(len(values)):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad_w[i] = (score_plus - base_score) / eps
        
        # Update
        values = values - learning_rate * grad_w
        values = project_to_feasible(values, n_points)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(values, dx)
        if current_c5 < best_c5:
            best_h = values.copy()
            best_c5 = current_c5
            best_n_points = n_points
            
        # Adaptive refinement every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            # Simple approximation for gradient magnitudes
            grad_mag = np.abs(grad_w)
            x_points, values = adaptive_grid_refinement(x_points, values, grad_mag, 40)
            n_points = len(values)

    # Final optimization with higher precision refinement
    if best_h is not None and time() - start_time < budget_s * 0.9:
        # Use a simple continuation approach with different learning rates
        learning_rate = 0.001
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            h_current = best_h.copy()
            base_score = evaluate_erdos_solution(h_current, dx)
            grad = np.zeros_like(h_current)

            h_current = h_current - learning_rate * grad
            h_current = project_to_feasible(h_current, len(h_current))
            
            score = evaluate_erdos_solution(h_current, dx)
            if score < best_c5:
                best_h = h_current.copy()
                best_c5 = score

    return (list(best_h), best_c5, len(best_h))
