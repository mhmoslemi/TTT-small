import numpy as np
import time
from scipy.optimize import minimize
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple spike to satisfy sum constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list_size = 20
    tabu_list = deque(maxlen=tabu_list_size)
    memory = set()
    
    # Perturbation parameters
    perturb_rate = 0.2
    perturb_magnitude = 0.1
    
    # Local refinement
    local_refine_steps = 10
    
    # Main tabu search loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s * 0.95:
        # Generate neighbors
        neighbors = []
        
        # Sample a few candidates
        for _ in range(20):
            h_candidate = best_h.copy()
            
            # Perturb some indices
            indices_to_perturb = np.random.choice(n_points, 
                                                 size=int(n_points * perturb_rate),
                                                 replace=False)
            for idx in indices_to_perturb:
                h_candidate[idx] += np.random.normal(0, perturb_magnitude)
            
            # Clamp to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Normalize to satisfy integral constraint
            h_sum = np.sum(h_candidate)
            if h_sum > 0:
                h_candidate = h_candidate * (n_points / 2) / h_sum
            
            neighbors.append(h_candidate)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Check tabu list
            h_tuple = tuple(np.round(h_candidate, decimals=6))
            if h_tuple in memory:
                continue
                
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            if c5_candidate < best_neighbor_c5:
                best_neighbor_c5 = c5_candidate
                best_neighbor = h_candidate
        
        # Update best solution
        if best_neighbor is not None and best_neighbor_c5 < best_c5:
            best_h = best_neighbor.copy()
            best_c5 = best_neighbor_c5
            no_improvement_count = 0
            
            # Add to tabu list
            h_tuple = tuple(np.round(best_h, decimals=6))
            if h_tuple not in memory:
                tabu_list.append(h_tuple)
                memory.add(h_tuple)
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        iteration += 1
        
        # Local refinement
        if iteration % 3 == 0:
            def obj_func(h):
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_points / 2) / h_sum
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            for _ in range(local_refine_steps):
                if time.time() - start_time > budget_s * 0.95:
                    break
                
                h_perturb = best_h + np.random.normal(0, 0.01, n_points)
                try:
                    res = minimize(obj_func, h_perturb, method='SLSQP', 
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        c5_test = obj_func(res.x)
                        if c5_test < best_c5:
                            best_h = res.x.copy()
                            best_c5 = c5_test
                except:
                    continue
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
