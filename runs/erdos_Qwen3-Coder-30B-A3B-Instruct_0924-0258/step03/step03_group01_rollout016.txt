import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 20
    elite_size = 10
    mutation_rate = 0.3
    crossover_rate = 0.7
    generations = 30
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection
        elites = population[:elite_size]
        
        # Crossover and Mutation
        new_population = elites.copy()
        
        while len(new_population) < population_size:
            # Crossover
            if random.random() < crossover_rate and len(new_population) + 1 < population_size:
                parent1 = elites[random.randint(0, len(elites)-1)]
                parent2 = elites[random.randint(0, len(elites)-1)]
                crossover_point = random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                new_population.extend([child1, child2])
            else:
                # Mutation
                parent = elites[random.randint(0, len(elites)-1)]
                child = parent + np.random.normal(0, 0.03, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Keep population size
        population = new_population[:population_size]
        
        # Local refinement with SQP for top individuals (every 2 generations)
        if gen % 2 == 0:
            for i in range(min(3, len(population))):
                h0 = population[i]
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_sum}
                try:
                    res = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        refined = res.x
                        c5_refined = obj(refined)
                        if c5_refined < fitness[i]:
                            population[i] = refined
                            fitness[i] = c5_refined
                except:
                    pass
    
    # Final refinement of best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
