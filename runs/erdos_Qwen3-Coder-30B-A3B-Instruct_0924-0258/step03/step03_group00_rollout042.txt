import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier representation approach with optimization
    d = 5  # number of Fourier terms
    max_restarts = 20
    trial_time = budget_s / max_restarts
    
    def fourier_to_h(coeffs):
        """Convert Fourier coefficients to h(x) using logistic transform"""
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + sum(coeffs[i+1] * np.cos((i+1) * np.pi * x) for i in range(d))
        h = 1 / (1 + np.exp(-g))
        # Normalize to ensure integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def obj_func(coeffs):
        h = fourier_to_h(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    # Optimization bounds
    bounds = [(-10, 10) for _ in range(d + 1)]
    
    # Try multiple random restarts
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization of Fourier coefficients
        coeffs0 = np.random.uniform(-2, 2, d + 1)
        
        try:
            # Use SLSQP for optimization with bounds
            res = minimize(obj_func, coeffs0, method='SLSQP', bounds=bounds,
                          options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-4})
            
            if res.success:
                h_candidate = fourier_to_h(res.x)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            pass
    
    # Final refinement with local optimization
    if time.time() - start_time < budget_s * 0.95:
        def refined_obj(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        # Perturb current best solution for local refinement
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_sum = np.sum(h_perturb)
        if h_sum > 0:
            h_perturb = h_perturb * (n_points / 2) / h_sum
        
        try:
            res = minimize(refined_obj, h_perturb, method='SLSQP', 
                          bounds=[(0, 1) for _ in range(n_points)],
                          options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                c5_test = refined_obj(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
