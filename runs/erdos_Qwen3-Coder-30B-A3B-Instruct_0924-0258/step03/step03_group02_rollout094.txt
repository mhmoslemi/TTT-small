import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    n_clusters = 10
    max_iter = 100
    local_steps = 10
    conv_steps = 20
    recluster_freq = 25
    
    # Initial clustering
    shift_indices = np.arange(n_points)
    overlap_vectors = []
    for k in range(n_points):
        if k == 0:
            overlap_vectors.append(np.ones(n_points) * 0.5)  # Dummy
        else:
            v = np.zeros(n_points)
            v[k:] = 1
            overlap_vectors.append(v[:n_points-k])
    overlap_vectors = np.array(overlap_vectors)
    
    # Clustering
    kmeans = KMeans(n_clusters=n_clusters, n_init=10, random_state=seed)
    cluster_labels = kmeans.fit_predict(overlap_vectors[:n_clusters])
    
    h_current = h_start.copy()
    
    iter_count = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Update clustering every recluster_freq iterations
        if iter_count % recluster_freq == 0:
            # Recompute overlap vectors for current h
            overlap_vectors = []
            for k in range(n_points):
                if k == 0:
                    overlap = np.ones(n_points)
                else:
                    overlap = np.convolve(h_current, np.ones(n_points-k), mode='valid')[:n_points-k]
                overlap_vectors.append(overlap)
            overlap_vectors = np.array(overlap_vectors)
            cluster_labels = kmeans.fit_predict(overlap_vectors[:n_clusters])
        
        # Optimize for each cluster
        for c in range(n_clusters):
            # Get indices in this cluster
            indices = np.where(cluster_labels == c)[0]
            if len(indices) == 0:
                continue
            
            # Local search step
            for step in range(local_steps):
                if time() - start_time > budget_s * 0.95:
                    break
                grad = np.zeros(n_points)
                for k in indices:
                    if k < n_points:
                        # k-shift overlap computation
                        if k == 0:
                            # Direct computation without shift
                            grad += 1 - 2 * h_current
                        else:
                            # For k > 0, gradient calculation via convolution
                            grad[:n_points-k] += 1 - 2 * h_current[k:]
                            grad[k:] += 1 - 2 * h_current[:-k]

                # Gradient step
                h_current += 0.01 * grad
                h_current = project_to_feasible(h_current, n_points)
                
            # Convex relaxation step for cluster
            for step in range(conv_steps):
                if time() - start_time > budget_s * 0.95:
                    break
                # Simplified convex relaxation
                try:
                    h_var = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_var) == n_points / 2,
                        h_var >= 0,
                        h_var <= 1
                    ]
                    # Add some basic overlap constraints
                    for k in indices[:5]:  # Limit constraints
                        if k < n_points:
                            if k == 0:
                                # Identity case would be problematic, skip
                                continue
                            corr_constraint = cp.sum(cp.multiply(h_var[:-k], 1 - h_var[k:])) * dx <= 0.5
                            constraints.append(corr_constraint)
                    
                    prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
                    prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_new = h_var.value
                        if h_new is not None:
                            h_current = project_to_feasible(h_new, n_points)
                except:
                    pass
                    
        # Check progress
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > 100:
                break

    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-10})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception as e:
            pass
    
    return (list(best_h), best_c5, n_points)
