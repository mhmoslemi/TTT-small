import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create periodic initialization
        n_points = 35
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    max_iter = 10000
    iter_count = 0
    
    # Track acceptance rate
    accept_count = 0
    total_count = 0
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Generate neighbor: perturb a few random points
        h_candidate = h_current.copy()
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Perturb each selected index
        for i in indices:
            # Gaussian perturbation
            delta = np.random.normal(0, 0.1)
            h_candidate[i] += delta
            
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - c5_current
        total_count += 1
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_candidate
            c5_current = c5_candidate
            accept_count += 1
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        
        # Temperature update - adaptive based on acceptance rate
        if iter_count % 100 == 0:
            accept_rate = accept_count / max(1, total_count)
            if accept_rate > 0.8:
                T *= 0.95  # decrease temperature more slowly
            elif accept_rate < 0.2:
                T *= 1.2   # increase temperature more quickly
            else:
                T *= alpha  # normal decay
            
            # Ensure temperature doesn't go too low
            T = max(T_min, T)
            
            # Reset counters
            accept_count = 0
            total_count = 0
    
    # Final refinement with local search
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        from scipy.optimize import minimize
        
        def refinement_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(refinement_obj, best_h, method='L-BFGS-B', 
                      bounds=[(0,1) for _ in range(n_points)], 
                      options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
