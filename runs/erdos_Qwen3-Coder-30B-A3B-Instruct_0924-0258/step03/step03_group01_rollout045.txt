import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # PSO Parameters
    swarm_size = 20
    max_iter = 30
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = lambda iter: w_max - (w_max - w_min) * iter / max_iter
    
    # Initialize swarm
    swarm = []
    for i in range(swarm_size):
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append((h.copy(), evaluate_erdos_solution(h, dx), h.copy(), np.inf))
    
    # Global best
    best_h = swarm[0][0].copy()
    best_c5 = swarm[0][1]
    best_global = best_h.copy()
    best_c5_global = best_c5
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        w = inertia_weight(iteration)
        
        for i, (h, c5, pbest_h, pbest_c5) in enumerate(swarm):
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = w * (h - pbest_h) + c1 * r1 * (best_global - h) + c2 * r2 * (best_global - h)
            new_h = h + v
            
            # Enforce feasibility
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(new_h, dx)
            
            # Update particle best
            if new_c5 < pbest_c5:
                swarm[i] = (new_h, new_c5, new_h, new_c5)
            else:
                swarm[i] = (new_h, new_c5, pbest_h, pbest_c5)
            
            # Update global best
            if new_c5 < best_c5_global:
                best_global = new_h.copy()
                best_c5_global = new_c5
                if best_c5_global < best_c5:
                    best_h = new_h.copy()
                    best_c5 = best_c5_global

        # Local refinement every few iterations
        if iteration % 5 == 0:
            for _ in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                # Select top candidate
                candidates = sorted(swarm, key=lambda x: x[1])
                h_refine = candidates[0][0].copy()
                
                # SQP refinement
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
