import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segment_boundaries(h_segments, x_starts):
    """Combined boundaries from segments"""
    x_boundaries = np.concatenate(([0], x_starts, [2]))
    return x_boundaries

def compute_h_from_segments(h_segments, x_boundaries):
    """Compute step function from segments"""
    h_values = np.zeros(len(x_boundaries) - 1)
    for i in range(len(h_segments)):
        h_values[i] = h_segments[i]
    return h_values

def fit_segments_to_h(h, n_segments):
    """Fit h to piecewise constant segments"""
    # Simple average within cells
    dx = 2.0 / len(h)
    segments = []
    boundary_points = []
    for i in range(n_segments):
        start = i * len(h) // n_segments
        end = (i+1) * len(h) // n_segments
        avg = np.mean(h[start:end])
        segments.append(avg)
        boundary_points.append((start + end) / 2 * dx)
    return segments, boundary_points

        max_corr_idx = np.argmax(corr)

        grad = np.zeros(len(h_segments))
