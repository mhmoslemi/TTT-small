import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use sinusoidal initialization
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    max_iter = int(1e4)
    iter_count = 0
    
    # Keep track of recent acceptance rate
    recent_accepts = []
    max_recent = 100
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Generate candidate
        h_candidate = best_h.copy()
        
        # Randomly select a subset of points to perturb (5-10%)
        n_perturb = max(1, int(n_points * (0.05 + 0.05 * np.random.random())))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb selected points
        for i in indices:
            # Small random change
            delta = np.random.normal(0, 0.1)
            h_candidate[i] += delta
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        c5_current = evaluate_erdos_solution(best_h, dx)
        
        # Calculate change in C5
        delta_c5 = c5_candidate - c5_current
        
        # Accept or reject
        accept = False
        if delta_c5 < 0:
            accept = True
        else:
            # Accept with probability
            if np.random.random() < math.exp(-delta_c5 / T):
                accept = True
        
        if accept:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            recent_accepts.append(1)
        else:
            recent_accepts.append(0)
            
        # Update temperature and adapt
        if iter_count % 100 == 0:
            # Adjust temperature based on recent acceptance rate
            if len(recent_accepts) > 0:
                acc_rate = sum(recent_accepts) / len(recent_accepts)
                if acc_rate < 0.05:
                    T *= 0.9  # Cool down fast if too few accepts
                elif acc_rate > 0.5:
                    T *= 1.1  # Warm up if too many accepts
                
                # Cap temperature
                T = max(T_min, min(T, 1.0))
         
        # Keep window of recent accepts
        if len(recent_accepts) > max_recent:
            recent_accepts = recent_accepts[-max_recent:]
        
        # Occasionally refine with local search
        if iter_count % 500 == 0:
            # Try gradient descent local refinement
            from scipy.optimize import minimize
            def obj(h):
                return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
            
            res = minimize(obj, best_h, method='L-BFGS-B',
                          bounds=[(0,1) for _ in range(n_points)], 
                          options={'maxiter': 20})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
    
    # Final check of solution
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
