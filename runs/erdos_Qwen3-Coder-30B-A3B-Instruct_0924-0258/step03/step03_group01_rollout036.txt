import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a structured initial guess
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization with local refinement
    population_size = 20
    temperature = 0.1
    cooling_rate = 0.99
    max_iterations = 30
    
    # Generate initial population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        min_c5_idx = np.argmin(population_c5)
        if population_c5[min_c5_idx] < best_c5:
            best_h = population[min_c5_idx].copy()
            best_c5 = population_c5[min_c5_idx]
        
        # Refine each individual in population with SQP
        refined_population = []
        for h in population:
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                else:
                    h_refined = h
            except:
                h_refined = h
                
            # Ensure feasibility
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            refined_population.append(h_refined)
        
        population = refined_population
        
        # Simulated Annealing step
        for i in range(population_size):
            h = population[i]
            # Generate neighbor
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # Calculate energy difference
            c5_old = evaluate_erdos_solution(h, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            delta = c5_new - c5_old
            
            # Accept with probability
            if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
                population[i] = h_new
                
        temperature *= cooling_rate
    
    # Final refinement using gradient-based method
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use multiple restarts for final refinement
    for restart in range(3):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
