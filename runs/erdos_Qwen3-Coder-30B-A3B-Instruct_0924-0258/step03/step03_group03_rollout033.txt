import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def adaptive_segmentation(h, n_segments):
    """Create adaptive segmentation from step function"""
    # Simple approach: divide into segments of roughly equal width
    segments = []
    segment_width = 2.0 / n_segments
    for i in range(n_segments):
        start_idx = int(i * segment_width * len(h) / 2)
        end_idx = int((i + 1) * segment_width * len(h) / 2)
        if end_idx <= start_idx:
            end_idx = start_idx + 1
        segment_values = h[start_idx:end_idx]
        segments.append(np.mean(segment_values) if len(segment_values) > 0 else 0.5)
    return np.array(segments)

def smooth_maximum(arr, radius=0.1):
    """Smoothed maximum using cubic envelope"""
    # Simple smoothing with linear interpolation 
    smooth_arr = np.zeros_like(arr)
    n = len(arr)
    for i in range(n):
        start = max(0, i - int(radius * n))
        end = min(n, i + int(radius * n) + 1)
        smooth_arr[i] = np.max(arr[start:end])
    return smooth_arr

def trust_region_optimize(f, grad_f, x0, trust_radius=0.1, max_iter=1000):
    """Simple trust region optimization for single variable"""
    x = x0.copy()
    for iteration in range(max_iter):
        if iteration > 0 and trust_radius < 1e-6:
            break
        try:
            # Evaluate function and gradient
            fx = f(x)
            gx = grad_f(x)
            
            # Simple quadratic model: m(p) = fx + p^T gx + 0.5 * p^T H p
            # For now, using just gradient descent with trust region
            step = -gx * trust_radius / (np.linalg.norm(gx) + 1e-8)
            
            # Project to feasible region
            new_x = x + step
            new_x = np.clip(new_x, 0, 1)
            new_x = new_x * (len(new_x) / 2) / np.sum(new_x)  # Sum constraint
            
            new_f = f(new_x)
            if new_f < fx:
                x = new_x
                trust_radius *= 1.5  # Increase trust radius
            else:
                trust_radius *= 0.5  # Decrease trust radius
        except:
            # In case of numerical issues, reduce trust region and continue
            trust_radius *= 0.5
            if trust_radius < 1e-6:
                break
    return x
