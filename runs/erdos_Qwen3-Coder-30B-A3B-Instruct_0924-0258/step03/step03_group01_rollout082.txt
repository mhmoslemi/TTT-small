import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population with modified versions of h_start
    population_size = 20
    population = [h_start.copy()]
    for i in range(1, population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Training iterations
    max_iterations = 100
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        best_candidates = [population[i] for i in sorted_indices[:5]]
        best_fitness = [fitness[i] for i in sorted_indices[:5]]
        
        if best_fitness[0] < best_c5:
            best_h = best_candidates[0].copy()
            best_c5 = best_fitness[0]
        
        # Generate new population through mutations
        new_population = []
        for i in range(population_size):
            # Select two parents
            parent1 = best_candidates[np.random.randint(0, 5)]
            parent2 = best_candidates[np.random.randint(0, 5)]
            
            # Crossover and mutation
            alpha = np.random.random()
            h = alpha * parent1 + (1 - alpha) * parent2
            h += np.random.normal(0, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            new_population.append(h)
        
        population = new_population
    
    # Convex refinement for best candidate
    h_refined = best_h.copy()
    try:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints
        corr = cp.Variable(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_refined = h.value
    except:
        pass
    
    refined_c5 = evaluate_erdos_solution(h_refined, dx)
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = h_refined
    
    # Final local refinement using scipy minimize
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except:
                pass

    return (list(best_h), best_c5, n_points)
