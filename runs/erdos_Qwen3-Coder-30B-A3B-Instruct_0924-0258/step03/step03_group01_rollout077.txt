import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
    else:
        n_points = 50
        dx = 2.0 / n_points
        # Initialize with structured pattern
        h_start = np.zeros(n_points)
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate population of candidates
    def create_population(h_init, n_pop):
        population = [h_init]
        for _ in range(n_pop - 1):
            h_pert = h_init + np.random.normal(0, 0.05, len(h_init))
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            population.append(h_pert)
        return population
    
    # Nelder-Mead optimization with population
    max_iter = 15
    n_pop = 15
    population = create_population(h_start, n_pop)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate population
        values = []
        for i, h in enumerate(population):
            val = evaluate_erdos_solution(h, dx)
            values.append((val, i))
        
        # Sort by objective value
        values.sort()
        best_idx = values[0][1]
        best_value = values[0][0]
        
        if best_value < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = best_value
        
        # Nelder-Mead step (simplified version)
        # Keep top 3
        top_indices = [v[1] for v in values[:3]]
        top_pop = [population[i] for i in top_indices]
        
        # Simple gradient refinement
        for i in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_current = top_pop[i].copy()
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_current, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
        
        # Create new generation with diversity
        new_population = [population[best_idx]]
        for _ in range(n_pop - 1):
            h1 = population[np.random.randint(0, n_pop)]
            h2 = population[np.random.randint(0, n_pop)]
            h_new = (h1 + h2) / 2.0 + np.random.normal(0, 0.05, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_population.append(h_new)
            
        population = new_population
        
    # Final pair-wise refinement if time allows
    if time() - start_time < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        for _ in range(5):
            if time() - start_time > budget_s * 0.98:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                cons = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
