import numpy as np
import time
from scipy.optimize import minimize
from math import sqrt

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to prevent cycling
    tabu_list = []
    tabu_length = 10
    
    # Simulated annealing parameters
    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-6
    
    # Local search parameters
    max_local_iter = 50
    
    def perturb_h(h, idx, value):
        h_new = h.copy()
        h_new[idx] = value
        return h_new
    
    def neighbor_h(h, idx):
        # Try different values for neighbor: 0, 0.5, 1
        candidates = [0, 0.5, 1]
        current = h[idx]
        candidates = [v for v in candidates if v != current]
        if not candidates:
            return h
        return perturb_h(h, idx, np.random.choice(candidates))
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        # Simulated Annealing step
        # Pick a random index
        idx = np.random.randint(0, n_points)
        # Perturb it
        h_new = neighbor_h(best_h, idx)
        
        # Ensure sum constraint
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on simulated annealing
        delta = c5_new - best_c5
        if delta < 0 or np.exp(-delta / temperature) > np.random.rand():
            best_h = h_new.copy()
            best_c5 = c5_new
            
            # Update tabu list
            h_key = tuple(np.round(h_new, decimals=3))
            tabu_list.append(h_key)
            if len(tabu_list) > tabu_length:
                tabu_list.pop(0)
        
        # Cool down
        if temperature > min_temperature:
            temperature *= cooling_rate
        
        # Occasionally do local search around current solution
        if iter_count % 20 == 0 and time.time() - start_time < budget_s * 0.9:
            # Local search
            h_local = best_h.copy()
            c5_local = best_c5
            
            # Try greedy descent
            for _ in range(max_local_iter):
                if time.time() - start_time > budget_s * 0.95:
                    break
                    
                improvements = 0
                for idx in np.random.permutation(n_points):
                    for val in [0, 0.5, 1]:
                        h_test = perturb_h(h_local, idx, val)
                        h_sum = np.sum(h_test)
                        if h_sum > 0:
                            h_test = h_test * (n_points / 2) / h_sum
                        c5_test = evaluate_erdos_solution(h_test, dx)
                        if c5_test < c5_local:
                            h_local = h_test
                            c5_local = c5_test
                            improvements += 1
                if improvements == 0:
                    break
                    
            if c5_local < best_c5:
                best_h = h_local
                best_c5 = c5_local
    
    # Final local refinement
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Final gradient descent
    res = minimize(obj_func, best_h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 100, 'ftol': 1e-6})
    
    if res.success:
        h_final = res.x
        h_sum = np.sum(h_final)
        if h_sum > 0:
            h_final = h_final * (n_points / 2) / h_sum
        final_c5 = evaluate_erdos_solution(h_final, dx)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5

    return (list(best_h), best_c5, n_points)
