import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization parameters
    max_iterations = 50
    perturb_magnitude = 0.05
    
    # Start with a single resolution
    current_n = n_points
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try to improve using dual optimization approach
        # Create a simplified version that focuses on key constraints
        try:
            # Define CVX variables for the current solution
            h_vars = cp.Variable(current_n)
            
            # Constraint for integral
            integral_constraint = cp.sum(h_vars) * dx == 1
            
            # Initialize bounds
            bounds = [h_vars >= 0, h_vars <= 1]
            
            # Create overlap constraints for limited shifts
            # Only consider relevant shifts for computational efficiency
            max_shift = min(current_n, 20)  # Limit number of shifts
            overlap_constraints = []
            
            for k in range(1, max_shift):
                # Compute valid range for this shift
                valid_len = current_n - k
                if valid_len <= 0:
                    break
                # Overlap constraint as sum(h[i] * (1 - h[i+k])) dx <= C5
                overlap_sum = cp.sum(cp.multiply(h_vars[:valid_len], 1 - h_vars[k:k+valid_len])) * dx
                overlap_constraints.append(overlap_sum <= best_c5)
            
            # Create and solve the problem
            prob = cp.Problem(
                cp.Minimize(0),
                bounds + [integral_constraint] + overlap_constraints
            )
            
            # Solve
            prob.solve(solver=cp.GLPK, max_iters=500, feastol=1e-6)
            
            if prob.status in ["optimal", "optimal_inaccurate"] and h_vars.value is not None:
                new_h = np.array(h_vars.value)
                # Ensure bounds
                new_h = np.clip(new_h, 0, 1)
                # Ensure integral constraint is met
                h_sum = np.sum(new_h)
                if h_sum > 0:
                    new_h = new_h * (current_n / 2) / h_sum
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    
                # Local refinement using SLSQP
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - current_n / 2}]
                bounds_list = [(0, 1) for _ in range(current_n)]
                
                res = minimize(objective, new_h, method='SLSQP', 
                               bounds=bounds_list, constraints=cons, 
                               options={'maxiter': 30, 'ftol': 1e-7})
                
                if res.success:
                    refined_h = np.clip(res.x, 0, 1)
                    refined_h = refined_h * (current_n / 2) / np.sum(refined_h)
                    refined_c5 = evaluate_erdos_solution(refined_h, dx)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
                        
        except Exception as e:
            pass
        
        # Random perturbation to escape local optima
        if time() - start_time > budget_s * 0.1:
            h_perturbed = best_h.copy()
            # Add small random noise
            noise = np.random.normal(0, perturb_magnitude * np.mean(best_h), current_n)
            h_perturbed += noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Renormalize
            h_sum = np.sum(h_perturbed)
            if h_sum > 0:
                h_perturbed = h_perturbed * (current_n / 2) / h_sum
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
            
        # Occasionally double resolution if we have time
        if time() - start_time > 400 and current_n == 35:
            current_n = 70
            # Convert best_h to new resolution by interpolation
            old_h = best_h
            new_h = np.interp(np.linspace(0, 2, current_n), 
                             np.linspace(0, 2, len(old_h)), old_h)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (current_n / 2) / np.sum(new_h)
            best_h = new_h
            dx = 2.0 / current_n
            best_c5 = evaluate_erdos_solution(best_h, dx)
            # Ensure equal constraint is met
            h_sum = np.sum(best_h)
            if h_sum > 0:
                best_h = best_h * (current_n / 2) / h_sum
            best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final verification and adjustment
    if time() - start_time < budget_s * 0.95:
        # Final improvement using SLSQP
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - current_n / 2}]
        bounds = [(0, 1) for _ in range(current_n)]
        
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            final_h = final_h * (current_n / 2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    else:
        # Quick final adjustment to maintain constraint
        h_sum = np.sum(best_h)
        if h_sum > 0:
            best_h = best_h * (current_n / 2) / h_sum
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, current_n)
