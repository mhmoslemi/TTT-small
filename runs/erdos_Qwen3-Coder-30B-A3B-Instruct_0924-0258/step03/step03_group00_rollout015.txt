import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based parametrization
    d = 5  # number of cosine terms
    max_restarts = 20
    max_iter_per_restart = 30
    
    def get_h_from_coeffs(coeffs):
        # coeffs: [a0, a1, ..., ad]
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + sum(coeffs[k+1] * np.cos((k+1) * np.pi * x) for k in range(d))
        h = 1.0 / (1.0 + np.exp(-g))
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def obj_func(coeffs):
        h = get_h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    bounds = [(-10, 10)] * (d + 1)
    
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        coeffs0 = np.random.uniform(-1, 1, d + 1)
        
        try:
            res = minimize(obj_func, coeffs0, method='SLSQP',
                          bounds=bounds, 
                          options={'maxiter': max_iter_per_restart, 'ftol': 1e-6})
            
            if res.success:
                h_candidate = get_h_from_coeffs(res.x)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except:
            continue
    
    # Final refinement with local search
    def obj_for_local(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    num_restarts = 10
    
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.98:
            break
        h_perturb = best_h + np.random.normal(0, 0.02, n_points)
        try:
            res = minimize(obj_for_local, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_for_local(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
