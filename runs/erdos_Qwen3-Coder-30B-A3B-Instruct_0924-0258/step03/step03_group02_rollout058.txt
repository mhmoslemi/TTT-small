import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Initialize population using differential evolution
    pop_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    
    # Create initial population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        pop_size -= 1
    
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    stagnation_count = 0
    max_stagnation = 100
    
    # DE parameters
    F = mutation_factor
    CR = crossover_prob
    
    max_iterations = 1000
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stagnation_count = 0
        
        # Evolve population
        new_population = []
        for i in range(len(population)):
            # Mutation
            idxs = [idx for idx in range(len(population)) if idx != i]
            a, b, c = population[np.random.choice(idxs)], population[np.random.choice(idxs)], population[np.random.choice(idxs)]
            mutated = a + F * (b - c)
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial.append(mutated[j])
                else:
                    trial.append(population[i][j])
            
            # Project to feasible region
            trial = project_to_feasible(np.array(trial), n_points)
            
            # Selection
            trial_c5 = evaluate_erdos_solution(trial, dx)
            if trial_c5 < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
        
        stagnation_count += 1
        if stagnation_count >= max_stagnation:
            F = min(1.0, F * 1.1)
            stagnation_count = 0
    
    # Post-processing: refine best solution
    if best_h is not None:
        try:
            # Use scipy to fine-tune
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Ensure final result is feasible
    if best_h is not None:
        best_h = project_to_feasible(best_h, n_points)
        final_c5 = evaluate_erdos_solution(best_h, dx)
        return (list(best_h), final_c5, n_points)
    
    # If no improvement, return initial solution
    initial_c5 = evaluate_erdos_solution(h_start, dx)
    return (list(h_start), initial_c5, n_points)
