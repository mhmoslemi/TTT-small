import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Initialize with periodic binary pattern
        h_start = np.zeros(n_points)
        ones_count = n_points // 2
        for i in range(ones_count):
            h_start[i*2] = 1.0
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe optimization
    max_iter = 10000
    step_size = 0.5
    perturbation_rate = 0.1
    
    for iteration in range(max_iter):
        elapsed = time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Find the maximizer k
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        # Shift to get the actual shift index
        shift = k_max - (n_points - 1)
        
        # Linear approximation: gradient of overlap at shift k
        h_shifted = np.roll(1 - best_h, shift)
        grad = h_shifted
        
        # Find h_extreme that minimizes grad^T h
        # This is a greedy selection of the n/2 largest values of (1 - h_k)
        # We need the least h that minimizes <grad, h>
        # This is the h with most zeros where grad is positive
        indices = np.argsort(-grad)  # descending
        h_extreme = np.zeros(n_points)
        for i in range(n_points // 2):
            h_extreme[indices[i]] = 1.0
            
        # Update with Frank-Wolfe step
        new_h = (1 - step_size) * best_h + step_size * h_extreme
        
        # Project to feasible region
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            step_size = min(0.99 * step_size, 0.8)
        else:
            # Try perturbation
            if np.random.rand() < perturbation_rate:
                perturbation = np.random.normal(0, 0.01, n_points)
                perturbed_h = best_h + perturbation
                perturbed_h = project_to_feasible(perturbed_h, n_points)
                perturbed_c5 = evaluate_erdos_solution(perturbed_h, dx)
                if perturbed_c5 < best_c5:
                    best_h = perturbed_h.copy()
                    best_c5 = perturbed_c5
                    
    # Final refinement with scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
