import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try increasing the resolution to 70 points
    if budget_s > 300:
        n_points = 70
        dx = 2.0 / n_points
        # Initialize with upsampling the existing solution
        h_start = np.repeat(h_start, 2) * 0.5
    
    # Refine the initial solution
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_refined = np.clip(res.x, 0, 1)
            # Renormalize
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Use a hybrid approach: solve feasibility problems with different target values
    c_low = 0.37
    c_high = 0.38085
    max_iterations = 10
    
    # Convert to CVX optimization to find a tighter bound
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        
        # Define CVX variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1
        ]
        
        # Add overlap constraints based on all shifts
        for k in range(1, n_points):
            valid_len = n_points - k
            if valid_len <= 0:
                break
            # Constraint: sum(h[i] * (1 - h[i+k])) * dx <= c_mid
            overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_mid
            constraints.append(overlap_constraint)
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_feasible = h.value
                # Now refine this solution
                def objective_local(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                # Apply bounds and constraints
                cons_local = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
                bounds_local = [(0, 1) for _ in range(n_points)]
                
                # Local optimization
                try:
                    res = minimize(objective_local, h_feasible, method='SLSQP',
                                   bounds=bounds_local, constraints=cons_local,
                                   options={'maxiter': 50, 'ftol': 1e-7})
                    if res.success:
                        h_refined = np.clip(res.x, 0, 1)
                        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                        c5_local = evaluate_erdos_solution(h_refined, dx)
                        if c5_local < best_c5:
                            best_h = h_refined
                            best_c5 = c5_local
                except:
                    pass
                c_low = c_mid
            else:
                c_high = c_mid
        except:
            c_high = c_mid
    
    # Final local improvement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
