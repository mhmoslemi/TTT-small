import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth periodic pattern
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(np.pi * x)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Configure optimization parameters
    learning_rate = 1e-3
    perturb_prob = 0.1
    perturb_magnitude = 0.05
    max_iter = 100000
    patience = 100
    patience_counter = 0
    
    # Initialize h
    h = project_to_feasible(h_start, n_points)
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(h, 1 - h, mode='full')
        
        # Find shift k with maximum overlap
        max_corr_idx = np.argmax(corr)
        max_k = max_corr_idx - (n_points - 1)  # Convert to shift index
        
        # Compute subgradient for the max shift
        subgradient = np.zeros(n_points)
        if max_k >= 0:
            # h[i] contributes to corr at index i + max_k
            valid_len = n_points - max_k
            if valid_len > 0:
                subgradient[:valid_len] += (1 - h[max_k:max_k + valid_len])
        else:
            # h[i] contributes to corr at index i - max_k in the other direction
            valid_len = n_points + max_k
            if valid_len > 0:
                subgradient[-max_k:] += (1 - h[:valid_len])
        
        # Update h using subgradient descent
        h_new = h - learning_rate * subgradient
        
        # Add random perturbations occasionally
        if np.random.random() < perturb_prob:
            perturbation = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h_new += perturbation
            
        # Project to feasible region
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept better solution or with probability based on temperature
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Decay learning rate
        learning_rate *= 0.99995
        
        # Early stopping if no improvement
        if patience_counter >= patience:
            break
            
    # Final refinement with L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 200, 'gtol': 1e-6})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
