import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.2
    elite_size = 5
    
    # Initialize population
    def initialize_population(size, n_points):
        population = []
        if 'initial_h_values' in globals():
            # Include initial solution
            h0 = np.array(initial_h_values)
            population.append(h0.copy())
            size -= 1
        for _ in range(size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)  # Renormalize to sum = n_points/2
            population.append(h)
        return population
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
        
    def tournament_selection(population, fitnesses, k=5):
        selected = random.sample(list(zip(population, fitnesses)), k)
        selected.sort(key=lambda x: x[1])
        return selected[0][0]
        
    def crossover(parent1, parent2):
        if random.random() < 0.5:
            # One-point crossover
            point = random.randint(1, len(parent1) - 1)
            child1 = np.concatenate([parent1[:point], parent2[point:]])
            child2 = np.concatenate([parent2[:point], parent1[point:]])
        else:
            # Uniform crossover
            mask = np.random.rand(len(parent1)) < 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
        
        return child1, child2
        
    def mutate(h, rate):
        h_mut = h.copy()
        for i in range(len(h)):
            if random.random() < rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.1)
                h_mut[i] += noise
        # Clip to [0, 1] and renormalize
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    # Main GA loop
    population = initialize_population(population_size, n_points)
    generation = 0
    
    while generation < generations and time() - start_time < budget_s * 0.95:
        # Evaluate fitness for all individuals
        fitnesses = [fitness(h) for h in population]
        
        # Find best solution in current generation
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitnesses[min_fitness_idx]
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        generation += 1
    
    # Fine-tune with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Try with the best from GA
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
        
    # If no improvement, return the best from GA
    return (list(best_h), best_c5, n_points)
