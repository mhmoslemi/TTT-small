import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_c5 = min(best_c5, 0.38085)  # Cap at known upper bound
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    max_no_improve = 50
    no_improve_count = 0
    
    # Local refinement parameters
    local_iter = 10
    
    # Main search loop
    while time.time() - start_time < budget_s * 0.95:
        # Generate neighbors
        neighbors = []
        for _ in range(200):  # Generate many neighbors
            if time.time() - start_time > budget_s * 0.95:
                break
            h_new = best_h.copy()
            
            # Perturb 20% of indices
            indices_to_perturb = random.sample(range(n_points), max(1, int(0.2 * n_points)))
            for i in indices_to_perturb:
                delta = random.uniform(-0.1, 0.1)
                h_new[i] = max(0, min(1, h_new[i] + delta))
            
            # Project to satisfy integral constraint
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            
            neighbors.append(h_new)
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            # Check if in tabu list (using hashable representation)
            candidate_tuple = tuple(h_candidate.round(6))
            if candidate_tuple in tabu_list:
                # Escape with low probability
                if random.random() > 0.7:
                    pass
                else:
                    continue
            
            c5 = evaluate_erdos_solution(h_candidate, dx)
            if c5 < best_neighbor_c5:
                best_neighbor = h_candidate
                best_neighbor_c5 = c5
        
        # If no valid neighbor found, break
        if best_neighbor is None:
            break
            
        # Update best solution
        if best_neighbor_c5 < best_c5:
            best_h = best_neighbor
            best_c5 = best_neighbor_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Add to tabu list
        tabu_list.append(tuple(best_h.round(6)))
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)

        if time.time() - start_time < budget_s * 0.95:

                h_clipped = np.clip(h, 0, 1)

                return np.sum(h) - n_points / 2

            for _ in range(local_iter):

    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
