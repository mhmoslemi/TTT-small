```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better initial pattern
        h_start = 0.5 * np.sin(np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try adaptive refinement 
    max_iter = 100
    adaptive_step = 5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Identify worst shifts by evaluating correlation with different offsets
        # Only check a subset of shifts to reduce computation
        shifts_to_check = list(range(1, n_points))
        overlaps = []
        for k in shifts_to_check:
            corr = np.correlate(best_h, 1 - best_h, mode='valid')
            if len(corr) > k:
                overlap = corr[k] * dx
                overlaps.append((overlap, k))
                
        # Sort by overlap to find worst shifts
        overlaps.sort(reverse=True)
        worst_k = [k for _, k in overlaps[:max(1, len(shifts_to_check)//10)]]
        
        # Create convex feasibility problem with worst shifts as constraints
        h_var = cp.Variable(n_points)
        
        # Basic constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add constraints for worst shifts only
        for k in worst_k:
            if k > 0:
                # For shift k, compute valid correlation indices
                valid_len = n_points - k
                if valid_len > 0:
                    corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                    constraints.append(corr_sum <= best_c5 * dx * valid_len)
        
        # Try solving
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feasible = h_var.value
                if h_feasible is not None:
                    # Refine with gradient descent
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    # Random restart if needed
                    if np.random.random() < 0.2:
                        h_init = np.random.uniform(0, 1, n_points)
                        h_init = project_to_feasible(h_init, n_points)
                    else:
                        h_init = h_feasible
                    
                    try:
                        res = minimize(refinement_obj, h_init, method='L-BFGS-B', 
                                     bounds=[(0,1) for _ in range(n_points)], 
                                     options={'maxiter': 50})
                        if res.success:
                            h_refined = project_to_feasible(res.x, n_points)
                            c5_refined = evaluate_erdos_solution(h_refined, dx)
                            if c5_refined < best_c5:
                                best_h = h_refined.copy()
                                best_c5 = c5_refined
                    except:
                        pass
        except:
            pass
            
        # Occasionally do random search
        if i % adaptive_step == 0 and i > 0:
            h_random = np.random.uniform(0, 1, n_points)
            h_random = project_to_feasible(h_random, n_points)
            c5_random = evaluate_erdos_solution(h_random, dx)
            if c5_random < best_c5:
                best_h = h_random.copy()
                best_c5 = c5_random
    
    # Final fine tuning with L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
```
