import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement parameters
    max_refinements = 3
    n_points_list = [35, 50, 70]  # small, medium, large
    
    for iter_count, current_n in enumerate(n_points_list):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new h for this resolution
        if iter_count == 0:
            h_current = best_h.copy()
        else:
            # Interpolate from previous solution
            h_current = np.interp(np.linspace(0, 2, current_n), 
                                np.linspace(0, 2, len(best_h)), 
                                best_h)
        
        dx_current = 2.0 / current_n
        
        # Local optimization
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx_current)
        
        # Constraints for optimization
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - current_n / 2}]
        bounds = [(0, 1) for _ in range(current_n)]
        
        # Try multiple starting points to avoid local minima
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            h_init = h_current + np.random.normal(0, 0.05, current_n)
            h_init = np.clip(h_init, 0, 1)
            
            # Renormalize
            h_init = h_init * (current_n / 2) / np.sum(h_init)
            
            try:
                res = minimize(objective, h_init, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    # Renormalize
                    h_refined = h_refined * (current_n / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx_current)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        n_points = current_n
            except:
                continue
                
        # Use updated h for next resolution
        best_h = np.interp(np.linspace(0, 2, current_n),

    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), final_c5, n_points)
