import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal initialization that's relaxed and feasible
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    iter_per_temp = 100
    max_iter = int(budget_s * 1000)  # Rough estimate for iterations
    
    # Track acceptance rate for adaptive cooling
    acceptance_count = 0
    total_attempts = 0
    
    # Initialize with a good baseline if available
    current_h = h_start.copy()
    current_c5 = best_c5
    
    iteration = 0
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Generate a neighbor by perturbing a random subset of points
        h_candidate = current_h.copy()
        
        # Perturb about 5-10% of points
        num_perturb = max(1, int(n_points * 0.05 + np.random.random() * n_points * 0.05))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            delta = np.random.normal(0, 0.1)
            h_candidate[idx] += delta
            
        # Project candidate to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Compute overlap for candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Metropolis criterion
        delta_c5 = c5_candidate - current_c5
        total_attempts += 1
        
        if delta_c5 < 0 or np.random.random() < math.exp(-delta_c5 / max(temperature, 1e-10)):
            current_h = h_candidate.copy()
            current_c5 = c5_candidate
            acceptance_count += 1
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Update temperature with adaptive cooling
        iteration += 1
        if iteration % iter_per_temp == 0:
            # Adaptive cooling based on acceptance ratio
            accept_ratio = acceptance_count / max(1, total_attempts)
            if accept_ratio < 0.05:
                temperature *= 0.9  # Cool faster
            elif accept_ratio > 0.2:
                temperature *= 1.05  # Cool slower
                
            temperature = max(temperature, min_temp)
            acceptance_count = 0
            total_attempts = 0
            
            # Occasionally refine with local search
            if iteration % (iter_per_temp * 2) == 0:
                # Refine best solution periodically
                h_refined = project_to_feasible(best_h, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined

    # Final local optimization 
    if (time() - start_time) < budget_s * 0.98:
        def obj_func(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        # Try a few optimization steps
        try:
            from scipy.optimize import minimize
            res = minimize(obj_func, best_h, method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 100})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass  # Continue if optimization fails

    return (list(best_h), best_c5, n_points)
