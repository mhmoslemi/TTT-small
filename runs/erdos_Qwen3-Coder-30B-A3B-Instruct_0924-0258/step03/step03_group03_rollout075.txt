import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points
    else:
        # Start with a symmetric construction
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        best_h = h_start.copy()
        best_c5 = current_c5
        best_n_points = n_points

    # Try to improve using convex relaxation and gradient descent
    # Try different resolution settings
    resolutions = [30, 35, 40]
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res != n_points:
            h_interp = np.interp(np.linspace(0, 2, res), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, res)
            n_points = res
            dx = 2.0 / n_points

        # Try a smooth construction based on Fourier modes
        # Initialize with a construction that has related structure
        h_candidate = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        
        # Try adding sawtooth pattern to better cover the domain
        h_candidate += 0.5 + 0.2 * np.sin(np.pi * x)
        h_candidate = project_to_feasible(h_candidate, n_points)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            best_n_points = n_points
            
        # More aggressive refinement: gradient descent
        max_iter = 300
        step_size = 0.005
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Gradient approximation via finite differences
            grad = np.zeros_like(h_candidate)
            eps = 1e-4
            base_score = evaluate_erdos_solution(h_candidate, dx)
            
            for i in range(len(h_candidate)):
                h_plus = h_candidate.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (score_plus - base_score) / eps

            # Update with small step and projected
            h_candidate = h_candidate - step_size * grad
            h_candidate = project_to_feasible(h_candidate, n_points)

            # Evaluation and update best
            score_refined = evaluate_erdos_solution(h_candidate, dx)
            if score_refined < best_c5:
                best_h = h_candidate.copy()
                best_c5 = score_refined
                best_n_points = n_points

    # Refine again with a deterministic heuristic
    # Now we try a more structured pattern based on double-sawtooth
    h_final = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    h_final = 0.5 + 0.15 * np.sin(np.pi*x) + 0.1 * np.sin(2*np.pi*x) 
    h_final = project_to_feasible(h_final, n_points)
    c5_test = evaluate_erdos_solution(h_final, dx)
    
    if c5_test < best_c5:
        best_h = h_final.copy()
        best_c5 = c5_test
        best_n_points = n_points

    return (list(best_h), best_c5, best_n_points)
