import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 20
    num_generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_refinement_interval = 1
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Genetic algorithm
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Sort population by fitness
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs]
        fitness = [fitness[i] for i in idxs]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection (top 50%)
        selected = population[:population_size // 2]
        
        # Crossover
        new_population = selected.copy()
        while len(new_population) < population_size:
            if random.random() < crossover_rate and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Repair child
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                if len(selected) > 0:
                    parent = random.choice(selected)
                    child = parent + np.random.normal(0, 0.03, n_points)
                    # Repair child
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
                else:
                    # Fallback - random individual
                    child = np.random.rand(n_points)
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
        
        population = new_population[:population_size]
        
        # SQP refinement every few generations
        if generation % sqp_refinement_interval == 0:
            for i in range(min(3, len(population))):
                h = population[i]
                def objective(h_val):
                    return evaluate_erdos_solution(h_val, dx)
                def constraint_sum(h_val):
                    return np.sum(h_val) - n_points / 2.0
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                try:
                    res = minimize(objective, h, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = objective(refined_h)
                        if refined_c5 < evaluate_erdos_solution(population[i], dx):
                            population[i] = refined_h
                except:
                    pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        best_h_refined = best_h.copy()
        best_c5_refined = best_c5
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            def objective(h_val):
                return evaluate_erdos_solution(h_val, dx)
            def constraint_sum(h_val):
                return np.sum(h_val) - n_points / 2.0
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            try:
                res = minimize(objective, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5_refined:
                        best_h_refined = h_refined
                        best_c5_refined = c5_refined
            except:
                pass
        best_h = best_h_refined
        best_c5 = best_c5_refined

    return (list(best_h), best_c5, n_points)
