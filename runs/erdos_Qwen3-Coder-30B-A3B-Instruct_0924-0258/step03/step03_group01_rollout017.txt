import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    population = []
    
    # Add initial solution
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    population.append(h_start.copy())
    
    # Add perturbed solutions
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Keep track of best solution
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Genetic algorithm parameters
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1  # Apply SQP every N generations
    
    generation = 0
    while time() - start_time < budget_s * 0.95:
        if generation % sqp_interval == 0:
            # SQP refinement for top 3 candidates
            population_copy = [h.copy() for h in population]
            population_copy.sort(key=lambda x: evaluate_erdos_solution(x, dx))
            top_candidates = population_copy[:3]
            
            for h in top_candidates:
                if time() - start_time > budget_s * 0.95:
                    break
                def objective(x):
                    return evaluate_erdos_solution(x, dx)
                
                def constraint_sum(x):
                    return np.sum(x) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                except:
                    pass
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Selection: keep top half
        selected_indices = np.argsort(fitnesses)[:population_size // 2]
        selected = [population[i] for i in selected_indices]
        
        # Crossover
        new_population = selected.copy()
        while len(new_population) < population_size:
            if random.random() < crossover_rate and len(selected) >= 2:
                i, j = random.sample(range(len(selected)), 2)
                parent1, parent2 = selected[i], selected[j]
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                child = new_population[i].copy()
                noise = np.random.normal(0, 0.03, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population[i] = child
                
        population = new_population
        
        # Update best solution
        for h in population:
            if time() - start_time > budget_s * 0.95:
                break
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
        generation += 1
        
        if generation > 30:
            break

    return (list(best_h), best_c5, n_points)
