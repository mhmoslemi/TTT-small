import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.random.rand(n_points) * 0.5 + 0.25
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.2 - 0.1)
        val = evaluate_erdos_solution(h, dx)
        pbest.append(h.copy())
        pbest_vals.append(val)
        if val < best_c5:
            best_h = h.copy()
            best_c5 = val
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    stagnation_count = 0
    last_improvement = 0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * (iter_num / max_iter)

            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] +

                           c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = val
                
                if val < gbest_val:
                    gbest = particles[i].copy()

                c5_final = evaluate_erdos_solution(h_final, dx)

                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
