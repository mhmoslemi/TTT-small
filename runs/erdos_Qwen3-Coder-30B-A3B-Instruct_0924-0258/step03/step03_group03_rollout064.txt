import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_objective(h, beta):
    """Smooth approximation of the maximum overlap"""
    corr = np.correlate(h, 1 - h, mode='full')
    exp_vals = np.exp(beta * corr)
    return (1.0 / beta) * np.log(exp_vals.sum())

def compute_gradient(h, beta, eps=1e-5):
    """Compute gradient using finite differences"""
    grad = np.zeros_like(h)
    base_obj = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_obj) / eps
    return grad

def geometric_mean(a, b):
    """Geometric mean to control step size"""
    return np.sqrt(a * b)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize

        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    best_n_points = n_points
    
    # Optimization parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    # Main optimization loop
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(h, beta)
        
        # Perform step with projection
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new objective
        new_c5 = smooth_objective(h_new, beta)
        new_c5_eval = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject step
        if new_c5_eval < best_c5:
            h = h_new.copy()
            best_c5 = new_c5_eval
            best_h = h.copy()
            best_n_points = n_points
            
            # Increase step size if beneficial
            if step_size < 1e-2:
                step_size *= 1.2

        else:
            step_size *= 0.5  # Decrease step size
            
        # Update beta for sharper approximation (simpler approach)
        if iter % 100 == 0 and iter > 0:
            beta *= 1.2

        # Check for convergence
        if np.abs(new_c5 - smooth_objective(h, beta)) < tolerance:
            break

    # Resolution refinement - try coarser grids for better results
    if time() - start_time < budget_s * 0.8:
        for n_try in [25, 30, 40, 50]:
            if time() - start_time > budget_s * 0.95:
                break
            try:
                h_interp = np.interp(np.linspace(0, 2, n_try), 
                                    np.linspace(0, 2, len(h)),

                h_interp = project_to_feasible(h_interp, n_try)
                dx_try = 2.0 / n_try
                c5_try = evaluate_erdos_solution(h_interp, dx_try)
                
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_interp.copy()
                    best_n_points = n_try
            except:
                continue

    return (list(best_h), best_c5, best_n_points)
