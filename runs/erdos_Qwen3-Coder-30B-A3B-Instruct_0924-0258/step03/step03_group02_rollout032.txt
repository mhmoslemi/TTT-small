import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def differential_evolution_optimize(h_start, n_points, dx, budget_s):
    """Differential evolution optimization with proper constraints"""
    # DE parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_probability = 0.9
    max_iterations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Keep track of best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    start_time = time()
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # DE evolution
        new_population = []
        for i in range(population_size):
            # Select three distinct individuals
            indices = list(range(population_size))
            indices.remove(i)
            a, b, c = random.sample(indices, 3)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = np.zeros(n_points)
            for j in range(n_points):
                if random.random() < crossover_probability:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Projection to constraints
            trial = project_to_feasible(trial, n_points)
            
            # Selection
            trial_c5 = evaluate_erdos_solution(trial, dx)
            if trial_c5 < fitness[i]:
                new_population.append(trial)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
        
        population = new_population
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential evolution
    h_de, c5_de = differential_evolution_optimize(h_start, n_points, dx, budget_s)
    if c5_de < best_c5:
        best_h = h_de.copy()
        best_c5 = c5_de
    
    # Refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
