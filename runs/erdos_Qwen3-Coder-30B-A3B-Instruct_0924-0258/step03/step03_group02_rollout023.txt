import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate Descent with Random Updates
    max_iterations = 1000
    tolerance = 1e-6
    
    h_current = best_h.copy()
    prev_c5 = best_c5 + tolerance * 10
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random permutation of indices
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for i in indices:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Current value
            old_val = h_current[i]
            
            # Compute current overlap contributions
            old_overlap = 0.0
            for k in range(n_points):
                if i + k < n_points:
                    old_overlap += h_current[i] * (1 - h_current[i + k])
                if i - k >= 0:
                    old_overlap += h_current[i] * (1 - h_current[i - k])
            
            # One-dimensional search to find optimal h[i]
            best_h_val = old_val
            best_overlap = old_overlap
            
            # Try several candidate values
            search_points = np.linspace(0, 1, 50)
            for candidate in search_points:
                if time() - start_time > budget_s * 0.95:
                    break
                h_current[i] = candidate
                new_overlap = 0.0
                for k in range(n_points):
                    if i + k < n_points:
                        new_overlap += h_current[i] * (1 - h_current[i + k])
                    if i - k >= 0:
                        new_overlap += h_current[i] * (1 - h_current[i - k])
                        
                if new_overlap < best_overlap:
                    best_overlap = new_overlap
                    best_h_val = candidate
            
            # Apply the best value
            h_current[i] = best_h_val
            # Project to feasibility
            h_current = project_to_feasible(h_current, n_points)
            
            # Compute new C5
            new_c5 = evaluate_erdos_solution(h_current, dx)
            if new_c5 < best_c5 - tolerance:
                best_c5 = new_c5
                best_h = h_current.copy()
                improved = True
                
        # Early termination if no improvement
        if not improved and abs(prev_c5 - best_c5) < tolerance:
            break
        prev_c5 = best_c5
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
