import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    max_no_improve = 50
    no_improve_count = 0
    escape_cost_range = [0.01, 0.1]
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.7:
        # Generate neighbors by perturbing 20% of points
        neighbor_indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        neighbors = []
        
        for _ in range(10):  # Generate multiple neighbors
            h_new = best_h.copy()
            for i in neighbor_indices:
                delta = np.random.uniform(-0.1, 0.1)
                h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            
            # Project to satisfy sum constraint
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            
            neighbors.append(h_new.copy())
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_candidate in neighbors:
            h_tuple = tuple(h_candidate)
            if h_tuple in tabu_list:
                # Apply escape mechanism
                if np.random.random() < 0.1:  # 10% chance to escape
                    pass
                else:
                    continue
            
            c5 = evaluate_erdos_solution(h_candidate, dx)
            if c5 < best_neighbor_c5:
                best_neighbor_c5 = c5

        if best_neighbor is not None:
            # Update best solution
            if best_neighbor_c5 < best_c5:
                best_h = best_neighbor.copy()
                best_c5 = best_neighbor_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
            
            # Manage tabu list
            tabu_list.append(tuple(best_h))
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
        else:
            no_improve_count += 1
        
        if no_improve_count > max_no_improve:
            break

    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)

            return np.sum(h) - n_points / 2

                h_perturb = h_perturb * (n_points / 2) / h_sum

                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
