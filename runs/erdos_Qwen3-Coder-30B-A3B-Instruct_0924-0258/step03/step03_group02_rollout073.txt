import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iterations = 500
    tolerance = 1e-6
    improvement = float('inf')
    
    # Coordinate Descent with Random Order
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.9:
        h_old = best_h.copy()
        indices = list(range(n_points))
        random.shuffle(indices)
        
        for i in indices:
            if (time() - start_time) > budget_s * 0.9:
                break
            # Try different values for h[i] to minimize max overlap
            best_h_i = h_old[i]
            best_c5_i = best_c5
            
            # Simple grid search - if we have more time, we might increase resolution
            grid_points = 50
            h_candidates = np.linspace(0, 1, grid_points)
            
            for candidate in h_candidates:
                h_test = h_old.copy()
                h_test[i] = candidate
                
                # Project to feasible region to maintain constraints
                h_test = project_to_feasible(h_test, n_points)
                c5_test = evaluate_erdos_solution(h_test, dx)
                
                if c5_test < best_c5_i:
                    best_c5_i = c5_test
                    best_h_i = candidate
            
            # Update only if there's improvement
            if abs(best_h_i - h_old[i]) > 1e-6:
                h_old[i] = best_h_i
                h_old = project_to_feasible(h_old, n_points)
                c5_current = evaluate_erdos_solution(h_old, dx)
                if c5_current < best_c5:
                    best_h = h_old.copy()
                    best_c5 = c5_current
        
        # Check if improvement is below tolerance
        improvement = np.abs(best_c5 - evaluate_erdos_solution(h_old, dx))
        if improvement < tolerance:
            break
        
        iteration += 1
    
    # Final SLSQP refinement
    if (time() - start_time) < budget_s * 0.95:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
