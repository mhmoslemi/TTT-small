import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Better initialization with sinusoidal pattern
        n_points = 35
        h_start = 0.5 * np.sin(np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Early refinement with gradient descent
    def obj(h):
        h_proj = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h_proj, dx)
    
    res = minimize(obj, h_start, method='L-BFGS-B', 
                   bounds=[(0,1) for _ in range(n_points)], 
                   options={'maxiter': 100})
    if res.success:
        h_refined = project_to_feasible(res.x, n_points)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_refined

    iterations = 0
    while time() - start_time < budget_s * 0.95:
        iterations += 1
        if iterations > 100:
            break
            
        # Identify worst shifts affecting the current C5
        c5_current = evaluate_erdos_solution(best_h, dx)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Warm up the correlation to avoid recomputation
        shifts = np.arange(1, len(corr))  # excluding zero shift
        overlaps = corr[1:] * dx  # for each shift k
        # Focus on worse overlaps (top 10%)
        worst_indices = np.argsort(overlaps)[-max(1, len(shifts) // 10):]
        worst_shifts = shifts[worst_indices] 

        # Formulate feasibility problem with constraint only on worst shifts
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        for k in worst_shifts:
            # a1 = h[0:(n-k)]
            # a2 = 1 - h[k:n]
            # corr_sum = a1 @ a2
            # Create the correlation explicitly for shift k
            corr_sum = 0
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
            
            # Constraint: correlation for this shift <= 0.38 * dx * valid_len (a guess)
            constraints.append(corr_sum <= (c5_current + 1e-6) * dx * valid_len)
        
        # Solve convex feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feas = h_var.value
                if h_feas is not None:
                    # Refine using gradient descent from feasible point
                    def refinement_func(x):
                        return evaluate_erdos_solution(project_to_feasible(x, n_points), dx)
                    
                    res = minimize(refinement_func, h_feas, method='L-BFGS-B', 
                                 bounds=[(0,1) for _ in range(n_points)], 
                                 options={'maxiter': 50})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                else:
                    break  # If infeasible, give up this iteration
            else:
                # If infeasible with current constraints
                # Break and fallback to local search
                break
        except:
            break

    # Final line search refinement on best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
