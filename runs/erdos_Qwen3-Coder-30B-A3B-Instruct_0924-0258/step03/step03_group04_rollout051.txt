import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic structure to avoid local minima
        h_start = 0.5 + 0.2 * np.sin(np.arange(n_points) * np.pi / n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize with a better candidate if possible
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, n_points)
        c0 = evaluate_erdos_solution(h0, dx)
        if c0 < best_c5:
            best_h = h0.copy()
            best_c5 = c0

    # Hyperparameters
    max_iter = 100
    patience = 10
    patience_counter = 0
    best_c5_so_far = best_c5

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shifts (those contributing most to C5)
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        # Only consider non-zero shifts
        shift_indices = np.arange(1, len(corr_full))
        corr_vals = corr_full[shift_indices] 
        worst_shifts = np.argsort(corr_vals)[-min(20, len(shift_indices)):]
        
        # Define a convex feasibility problem over only worst shifts
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add only worst shift constraints
        for k in worst_shifts:
            # For shift k, compute correlation over valid indices
            corr_sum = 0
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
            constraints.append(corr_sum <= best_c5_so_far * dx * valid_len)
            
        # No objective needed for feasibility check
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            # Solve with a more robust solver
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-5, abstol=1e-5)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feas = h_var.value
                if h_feas is not None:
                    # Refine with projected gradient descent
                    h_refined = project_to_feasible(h_feas, n_points)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        patience_counter = 0
                        best_c5_so_far = best_c5
                    else:
                        patience_counter += 1
                else:
                    patience_counter += 1
            else:
                patience_counter += 1
        except:
            patience_counter += 1

        # Occasionally add more constraints if we're stuck
        if patience_counter >= patience // 2:
            # Add more constraints from top 10% of shifts
            top_10_percent = int(len(shift_indices) * 0.1)
            top_indices = np.argsort(corr_vals)[-top_10_percent:]
            for k in top_indices:
                corr_sum = 0
                valid_len = n_points - k
                if valid_len > 0:
                    corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                constraints.append(corr_sum <= best_c5_so_far * dx * valid_len)
            patience_counter = 0
        
        # Random perturbation occasionally
        if iteration % 20 == 0 and iteration > 0:
            # Introduce a small random perturbation to escape local minima
            pert = np.random.normal(0, 0.01, n_points)
            pert = project_to_feasible(best_h + pert, n_points)
            c5_pert = evaluate_erdos_solution(pert, dx)
            if c5_pert < best_c5:
                best_h = pert.copy()
                best_c5 = c5_pert

    # Final optimization with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'gtol': 1e-6})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final

    return (list(best_h), best_c5, n_points)
