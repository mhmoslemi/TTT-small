import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Better optimization approach: Gradient-based with adaptive refinement
    best_c5 = float('inf')
    best_h = None
    best_n_points = None
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        if current_c5 < best_c5:
            best_h = h_start.copy()
            best_c5 = current_c5
            best_n_points = n_points
    else:
        # Default initialization
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        if current_c5 < best_c5:
            best_h = h_start.copy()
            best_c5 = current_c5
            best_n_points = n_points

    # First, try a simple gradient-based optimization with coordinate descent
    h_current = best_h.copy() if best_h is not None else h_start.copy()
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use gradient approximations with finite differences
    max_iter = 500
    step_size = 0.005
    eps = 1e-4
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute finite difference gradients
        h_current = project_to_feasible(h_current, n_points)
        base_score = evaluate_erdos_solution(h_current, dx)
        
        grad = np.zeros_like(h_current)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps

        # Update with gradient descent
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        new_score = evaluate_erdos_solution(h_new, dx)
        if new_score < best_c5:
            best_h = h_new.copy()
            best_c5 = new_score
            
        h_current = h_new.copy()
        
        # Occasionally refine and restart with improved h
        if it % 100 == 0 and it > 0:
            h_current = project_to_feasible(best_h, n_points)

    # Additional local search refinement
    h_refined = best_h.copy()
    dx_refined = 2.0 / n_points
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_refined = project_to_feasible(h_refined, n_points)
        base_score = evaluate_erdos_solution(h_refined, dx_refined)
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        for i in range(len(h_refined)):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx_refined)
            grad[i] = (score_plus - base_score) / eps

        step_size = 0.001
        h_refined = h_refined - step_size * grad
        h_refined = project_to_feasible(h_refined, n_points)

        new_score = evaluate_erdos_solution(h_refined, dx_refined)
        if new_score < best_c5:
            best_h = h_refined.copy()
            best_c5 = new_score

    return (list(best_h), best_c5, n_points)
