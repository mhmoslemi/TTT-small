import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    w = 0.729
    c1 = 1.49445
    c2 = 1.49445
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        h = project_to_feasible(np.random.rand(n_points), n_points)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Keep within bounds and project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = current_val
        
        if gbest_val < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_val
            
    # Refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def gradient_descent_refinement(h0):
            h = h0.copy()
            lr = 1e-3
            for _ in range(100):
                if time() - start_time > budget_s * 0.95:
                    break
                # Compute gradient using finite differences
                grad = np.zeros_like(h)
                eps = 1e-6
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = project_to_feasible(h_plus, n_points)
                    val_plus = evaluate_erdos_solution(h_plus, dx)
                    
                    h_minus = h.copy()
                    h_minus[i] -= eps
                    h_minus = project_to_feasible(h_minus, n_points)
                    val_minus = evaluate_erdos_solution(h_minus, dx)
                    
                    grad[i] = (val_plus - val_minus) / (2 * eps)
                
                # Update
                h -= lr * grad
                h = project_to_feasible(h, n_points)
                
                # Evaluate
                current_val = evaluate_erdos_solution(h, dx)
                if current_val < best_c5:
                    best_h[:] = h[:]
                    best_c5 = current_val
            
            return h, best_c5
        
        best_h, best_c5 = gradient_descent_refinement(best_h)
    
    return (list(best_h), best_c5, n_points)
