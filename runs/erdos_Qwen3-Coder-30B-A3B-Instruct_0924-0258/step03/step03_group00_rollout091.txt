import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different optimizers and strategies
    max_restarts = 20
    num_restarts = 0
    
    # Strategy 1: Direct optimization with different starting points
    while num_restarts < max_restarts and time.time() - start_time < budget_s * 0.95:
        # Generate random perturbation
        h_perturb = h_start + np.random.normal(0, 0.05, n_points)
        
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
        
        num_restarts += 1
        
        # Strategy 2: Use the fact that we can combine functions with shifts

            h_shifted = np.zeros(n_points)

            half_points = n_points // 2
            h_shifted[:half_points] = 1.0
            h_shifted[half_points:] = 0.0
            
            # Add some smoothing to prevent overfitting
            h_smooth = np.convolve(h_shifted, np.ones(5)/5, mode='same')
            
            # Normalize
            h_norm = h_smooth * (n_points / 2) / np.sum(h_smooth)
            
            c5_test = evaluate_erdos_solution(h_norm, dx)
            if c5_test < best_c5:
                best_h = h_norm.copy()
                best_c5 = c5_test

            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)

            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)

            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to further refine with local search
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP',

                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:

                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
