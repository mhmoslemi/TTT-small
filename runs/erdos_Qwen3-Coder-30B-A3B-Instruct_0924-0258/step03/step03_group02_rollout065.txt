import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Set up clustering
    n_clusters = 10
    max_iter = 100
    n_shifts = 2 * n_points - 1
    shift_indices = np.arange(n_shifts)
    
    # Initialize h
    h = h_start.copy()
    
    # Precompute all overlap patterns for clustering
    overlap_patterns = []
    for k in range(n_shifts):
        if k < n_points:
            pattern = np.array([1 if i < n_points - k else 0 for i in range(n_points)])
        else:
            pattern = np.array([1 if i >= k - n_points + 1 else 0 for i in range(n_points)])
        overlap_patterns.append(pattern)
    
    overlap_patterns = np.array(overlap_patterns)
    
    # Cluster shifts based on overlap patterns
    kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
    cluster_labels = kmeans.fit_predict(overlap_patterns)
    
    # Group shifts by cluster
    cluster_shifts = []
    for c in range(n_clusters):
        cluster_shifts.append(np.where(cluster_labels == c)[0])
    
    iter_count = 0
    stagnation_count = 0
    min_improvement = 1e-6
    
    while time() - start_time < budget_s * 0.95:
        old_best_c5 = best_c5
        iter_count += 1
        
        # For each cluster, optimize h
        for c in range(n_clusters):
            if time() - start_time > budget_s * 0.95:
                break
                
            shifts = cluster_shifts[c]
            
            # Optimization for this cluster
            def cluster_objective(h_val):
                h_current = h_val.copy()
                h_current = project_to_feasible(h_current, n_points)
                total_overlaps = 0
                
                # Compute overlap for shifts in this cluster
                for k in shifts:
                    # Compute overlap for shift k
                    if k < n_points:
                        overlap = np.sum(h_current[:n_points-k] * (1 - h_current[k:]))
                    else:
                        overlap = np.sum(h_current[k-n_points+1:] * (1 - h_current[:n_points-(k-n_points+1)]))
                    total_overlaps += overlap
                
                # Return average overlap (since we're minimizing max, we minimize average with penalty)
                return total_overlaps / len(shifts) * dx
            
            # Consider even local search with SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            
            attempt = 0
            while attempt < 2 and time() - start_time < budget_s * 0.95:
                try:
                    res = minimize(cluster_objective, h.copy(), method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
                    if res.success:
                        h_new = project_to_feasible(res.x, n_points)
                        new_c5 = evaluate_erdos_solution(h_new, dx)
                        if new_c5 < best_c5:
                            best_h = h_new.copy()
                            best_c5 = new_c5
                    break
                except Exception:
                    attempt += 1
                    h_new = h + 0.01 * (np.random.rand(n_points) - 0.5)
                    h_new = project_to_feasible(h_new, n_points)
                    new_c5 = evaluate_erdos_solution(h_new, dx)
                    if new_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = new_c5
                    break
        
        # Update progress
        if abs(best_c5 - old_best_c5) < min_improvement:
            stagnation_count += 1
        else:
            stagnation_count = 0
        
        # Periodically re-cluster if stagnated
        if stagnation_count > 50 and iter_count % 20 == 0:
            # Recompute overlap patterns based on current h
            overlap_patterns_new = []
            h_full = np.concatenate([h, np.zeros(n_points-1)])
            for k in range(n_shifts):
                pattern = np.zeros(n_points)
                if k < n_points:
                    pattern[:n_points-k] = h_full[k:k+n_points]
                else:
                    pattern[k-n_points+1:] = h_full[:n_points-(k-n_points+1)]
                overlap_patterns_new.append(pattern)
            overlap_patterns_new = np.array(overlap_patterns_new)
            
            cluster_labels = kmeans.fit_predict(overlap_patterns_new)
            cluster_shifts = []
            for c in range(n_clusters):
                cluster_shifts.append(np.where(cluster_labels == c)[0])
        
        if stagnation_count > 100:
            break

    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_objective(h_val):
            return evaluate_erdos_solution(h_val, dx)
        
        def final_constraint(h_val):
            return np.sum(h_val) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': final_constraint}
        
        try:
            # Try a few restarts for better convergence
            for _ in range(3):
                if time() - start_time > budget_s * 0.95:
                    break
                h_restart = np.random.rand(n_points) * 0.5 + 0.25
                h_restart = project_to_feasible(h_restart, n_points)
                res = minimize(final_objective, h_restart, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_final = project_to_feasible(res.x, n_points)
                    c5_final = evaluate_erdos_solution(h_final, dx)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
