import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Use a sinusoidal initialization as it often works well for such problems
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temperature = 0.1
    cooling_rate = 0.99
    min_temp = 1e-8
    max_iter = 10000
    iter_count = 0
    accept_count = 0
    max_accept = 100
    
    # Perform simulated annealing
    for iteration in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        h_candidate = best_h.copy()
        
        # Perturb a random subset of points (5-10%)
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Apply small random perturbations
        for i in indices:
            # Move in a random direction
            delta = np.random.normal(0, 0.05)
            h_candidate[i] += delta
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            accept_count += 1
            
        # Update temperature
        iter_count += 1
        if iter_count % 100 == 0:
            temperature *= cooling_rate
            
        # Safety check to prevent getting stuck
        if accept_count >= max_accept:
            accept_count = 0
            temperature *= 0.95
    
    # Final evaluation with projection
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
