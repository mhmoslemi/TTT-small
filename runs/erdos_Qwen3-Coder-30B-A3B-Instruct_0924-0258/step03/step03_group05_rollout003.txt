import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try higher resolution if time allows
    if budget_s > 400:
        n_points = 70
        dx = 2.0 / n_points
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
        h_start = np.clip(h_start, 0, 1)
        # Renormalize
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local refinement with SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    max_iter = 50
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 30, 'ftol': 1e-7, 'eps': 1e-6})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                # Renormalize
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
            else:
                break
        except:
            break
    
    # Additional local search exploration
    if time() - start_time < budget_s * 0.9:
        # Try different starting points
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            # Random perturbation
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            try:
                res = minimize(objective, h_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 20, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
    
    # Final verification and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
        
    return (list(best_h), final_c5, n_points)
