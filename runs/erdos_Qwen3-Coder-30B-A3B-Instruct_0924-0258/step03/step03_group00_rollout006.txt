import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search with perturbation-based optimization
    max_iterations = 10000
    tabu_list = []
    tabu_length = 10
    temp = 1.0
    cooling_rate = 0.9995
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Make a random perturbation
        h_pert = best_h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_pert[idx] += delta
        h_pert[idx] = np.clip(h_pert[idx], 0, 1)
        
        # Renormalize to maintain integral constraint
        h_sum = np.sum(h_pert)
        if h_sum > 0:
            h_pert = h_pert * (n_points / 2) / h_sum
        
        # Evaluate new configuration
        c5_new = evaluate_erdos_solution(h_pert, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_pert.copy()
            best_c5 = c5_new
        else:
            # Accept with some probability if not in tabu list
            h_hash = hash(tuple(h_pert.round(6)))
            if h_hash not in tabu_list:
                prob = np.exp(- (c5_new - best_c5) / temp)
                if np.random.random() < prob:
                    best_h = h_pert.copy()
                    best_c5 = c5_new
                    
        # Update tabu list
        tabu_list.append(h_hash)
        if len(tabu_list) > tabu_length:
            tabu_list.pop(0)
            
        # Cooling
        temp *= cooling_rate
    
    # Final local refinement
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run several local optimizations with different starting points
    num_restarts = 20
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.98:
            break
        
        h_perturb = best_h + np.random.normal(0, 0.02, n_points)
        
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
