import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def normalize_h(h, n_points):
    """Normalize h to satisfy sum(h) * dx = 1"""
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return np.clip(h, 0, 1)

def crossover(parent1, parent2):
    """Perform one-point crossover between two parents"""
    n = len(parent1)
    point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, sigma=0.1):
    """Apply Gaussian mutation to h"""
    h = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += np.random.normal(0, sigma)
    return np.clip(h, 0, 1)

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection"""
    tournament_indices = random.sample(range(len(population)), k)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

    pop_size = 50
