import numpy as np
from scipy.signal import correlate
from time import time
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(arr, eps=1e-8):
    """Smooth approximation of max function"""
    return np.log(np.sum(np.exp(arr / eps))) * eps

def smooth_max_correlation(h, dx):
    """Smooth objective using log-sum-exp for correlation maximum"""
    corr = np.correlate(h, 1 - h, mode='full')
    # Estimate maximum by summing around the peak
    n_corr = len(corr)
    idx_max = np.argmax(corr)
    # Use a few points around max
    half_window = min(10, n_corr // 2)
    start_idx = max(0, idx_max - half_window)
    end_idx = min(n_corr, idx_max + half_window + 1)
    window = corr[start_idx:end_idx]
    # Smooth max of window
    smoothed_max = smooth_max(window)
    return smoothed_max * dx

def solve_with_trust_region(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    h_start = project_to_feasible(h_start, n_points)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    
    # Quick 2D grid search for better starting point
    if time() - start_time < budget_s * 0.1:
        n_samples = 10
        for a in np.linspace(0.3, 0.7, n_samples):
            for b in np.linspace(0.3, 0.7, n_samples):
                h_test = h_start.copy()
                mid = n_points // 2
                h_test[mid-1:mid+1] = [a, b]
                h_test = project_to_feasible(h_test, n_points)
                c5_test = evaluate_erdos_solution(h_test, 2.0 / n_points)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test

    # Parametric trust-region optimization on h values directly
    # Use harder constraint handling and more iterations

    dx = 2.0 / n_points
    max_iter = 500
    trust_radius = 0.1
    trust_radius_min = 1e-6
    min_improvement = 1e-8

    # Variables to optimize: h values
    h_opt = best_h.copy()
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_val = evaluate_erdos_solution(h_opt, dx)
        
        # Compute gradient via finite difference
        grad = np.zeros_like(h_opt)
        eps = 1e-4
        
        for i in range(len(h_opt)):
            h_plus = h_opt.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (val_plus - current_val) / eps

        # Compute step based on gradient
        step = -grad
        step_norm = np.linalg.norm(step)
        
        if step_norm > 0:
            step = step / step_norm * trust_radius
            
        # Try step and project
        h_new = h_opt + step
        h_new = project_to_feasible(h_new, n_points)
        
        new_val = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement, adjust trust radius
        if new_val < current_val:
            h_opt = h_new.copy()
            if current_val - new_val > min_improvement:
                trust_radius = min(trust_radius * 1.5, 1.0)
            else:
                trust_radius = max(trust_radius * 0.5, trust_radius_min)
        else:
            trust_radius = max(trust_radius * 0.5, trust_radius_min)
            
        if new_val < best_c5:
            best_c5 = new_val
            best_h = h_opt.copy()
        
        # Occasionally rescale to maintain diversity
        if iter_num % 50 == 0 and iter_num > 0:
            h_opt = project_to_feasible(h_opt, n_points)

    # Final gradient refinement
    if time() - start_time < budget_s * 0.95:
        # Use smoother objective for better gradient estimates
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            eps = 1e-5
            grad = np.zeros_like(h_opt)
            base_val = evaluate_erdos_solution(h_opt, dx)
            for i in range(len(h_opt)):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                val_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (val_plus - base_val) / eps
            
            step_size = 0.01
            h_opt = h_opt - step_size * grad
            h_opt = project_to_feasible(h_opt, n_points)
            new_val = evaluate_erdos_solution(h_opt, dx)
            if new_val < best_c5:
                best_c5 = new_val
                best_h = h_opt.copy()

    return (list(best_h), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return solve_with_trust_region(seed, budget_s)
