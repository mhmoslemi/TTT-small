        if np.random.random() > rate:
            return h1.copy(), h2.copy()

        alpha = np.random.random()
        h_new1 = alpha * h1 + (1 - alpha) * h2
        h_new2 = (1 - alpha) * h1 + alpha * h2
        h_new1 = np.clip(h_new1, 0, 1)
        h_new2 = np.clip(h_new2, 0, 1)
        return h_new1, h_new2
