import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(corr, beta):
    """Smooth approximation to max function"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def smooth_objective(h, beta):
    """Smoothed version of the C5 objective"""
    corr = np.correlate(h, 1 - h, mode='full')
    return smooth_max(corr, beta)

def compute_gradient(h, beta, eps=1e-5):
    """Compute gradient using finite differences"""
    grad = np.zeros_like(h)
    base_value = smooth_objective(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta) - base_value) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(h, beta)
        
        # Update step
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            step_size *= 0.5
            
        # Update beta for sharper approximation
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.5
            
        # Early stopping based on tolerance
        if np.abs(best_c5 - evaluate_erdos_solution(h, dx)) < tolerance:
            break
            
    # Resolution refinement if we've made progress
    if iteration > 500:
        n_points_try = min(n_points * 2, 100)
        if n_points_try > n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_try), 
                                np.linspace(0, 2, n_points), 
                                best_h)
            h_interp = project_to_feasible(h_interp, n_points_try)
            refined_c5 = evaluate_erdos_solution(h_interp, 2.0 / n_points_try)
            if refined_c5 < best_c5:
                best_h = h_interp.copy()
                best_c5 = refined_c5
                n_points = n_points_try
    
    return (list(best_h), best_c5, n_points)
