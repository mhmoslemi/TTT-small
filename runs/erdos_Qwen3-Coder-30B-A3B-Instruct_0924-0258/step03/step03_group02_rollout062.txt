import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple strategies
    strategies = [
        # Strategy 1: Convex relaxation with McCormick bounds
        'convex_relaxation',
        # Strategy 2: Direct optimization from the initial solution
        'direct_optimization',
        # Strategy 3: Random perturbations of current best
        'random_perturbations'
    ]
    
    for strategy in strategies:
        if time() - start_time > budget_s * 0.9:
            break
            
        h_current = best_h.copy()
        
        if strategy == 'convex_relaxation':
            # Use convex relaxation approach
            try:
                # Create variable for optimization
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Compute overlap constraints using convex relaxation
                # This is a simplified McCormick relaxation approach
                # For each k, we want to bound sum of h[i] * (1 - h[i+k])
                # We add constraints via a specific approach: 
                # bound each term with linear constraints if possible.
                
                # Simplified convex approach: optimize with some margin
                prob = cp.Problem(cp.Minimize(cp.sum(h_var**2)), constraints)
                prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
                
                if prob.status == cp.OPTIMAL:
                    h_opt = h_var.value
                    h_feasible = project_to_feasible(h_opt, n_points)
                    new_c5 = evaluate_erdos_solution(h_feasible, dx)
                    if new_c5 < best_c5:
                        best_h = h_feasible
                        best_c5 = new_c5
                        
            except Exception:
                pass
                
        elif strategy == 'direct_optimization':
            # Direct local optimization from current solution
            try:
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                res = minimize(objective, h_current, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_feasible = project_to_feasible(res.x, n_points)
                    new_c5 = evaluate_erdos_solution(h_feasible, dx)
                    if new_c5 < best_c5:
                        best_h = h_feasible
                        best_c5 = new_c5
                        
            except Exception:
                pass
                
        elif strategy == 'random_perturbations':
            # Try random perturbations around the current solution
            for _ in range(5):
                if time() - start_time > budget_s * 0.9:
                    break
                h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
                h_feasible = project_to_feasible(h_perturbed, n_points)
                new_c5 = evaluate_erdos_solution(h_feasible, dx)
                if new_c5 < best_c5:
                    best_h = h_feasible
                    best_c5 = new_c5
    
    # Final refinement with SLSQP on the best found solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5 and best_c5 > 0.37:
        try:
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-9})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
                    
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
