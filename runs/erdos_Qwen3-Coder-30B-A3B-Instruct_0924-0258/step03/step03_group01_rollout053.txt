import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        dx = 2.0 / n_points
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization
    population_size = 15
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
    
    iterations = 0
    while time() - start_time < budget_s * 0.95:
        iterations += 1
        # Evaluate population
        evals = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            evals.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(evals)
        population = [population[i] for i in sorted_indices]
        evals = [evals[i] for i in sorted_indices]
        
        if evals[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = evals[0]
        
        # Continue optimization with the best solutions
        if iterations > 50:
            break
            
        # Simplex-like behavior: mutate top solutions
        new_population = [population[0]]  # Keep best
        for i in range(1, population_size):
            h_new = population[0] + 0.5 * (population[i] - population[0])
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_population.append(h_new)
        
        population = new_population
    
    # Local refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            try:
                res = minimize(objective, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
