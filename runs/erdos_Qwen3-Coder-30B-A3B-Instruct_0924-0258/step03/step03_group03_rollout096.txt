import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Number of points to use
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize h: use initial_h_values if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = project_to_feasible(h_current, n_points)
    else:
        h_current = np.ones(n_points) * 0.5
        h_current = project_to_feasible(h_current, n_points)

    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Use a discrete gradient descent approach combined with perturbations
    max_iter = 10000
    step_size = 0.01
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sample gradient via finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-4
        base_score = evaluate_erdos_solution(h_current, dx)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps

        # Update with gradient descent
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Accept the new solution if it improves the objective
        new_score = evaluate_erdos_solution(h_new, dx)
        if new_score < best_c5:
            best_h = h_new.copy()
            best_c5 = new_score
        
        h_current = h_new
        # Occasionally add small random perturbations to escape local minima
        if it % 100 == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h_current = h_current + perturb
            h_current = project_to_feasible(h_current, n_points)
            
    return (list(best_h), best_c5, n_points)
