import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization parameters
    max_iter = 100
    patience = 10
    patience_count = 0
    prev_best = best_c5
    
    # Start with the initial solution
    current_h = best_h.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation to explore
        perturbation = np.random.normal(0, 0.02, n_points)
        perturbed_h = current_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        h_sum = np.sum(perturbed_h)
        if h_sum > 0:
            perturbed_h = perturbed_h * (n_points / 2) / h_sum
        
        # Try local optimization on perturbed solution
        try:
            res = minimize(objective, perturbed_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 20, 'ftol': 1e-7})
            if res.success:
                refined_h = np.clip(res.x, 0, 1)
                h_sum = np.sum(refined_h)
                if h_sum > 0:
                    refined_h = refined_h * (n_points / 2) / h_sum
                c5_new = evaluate_erdos_solution(refined_h, dx)
                if c5_new < best_c5:
                    best_h = refined_h
                    best_c5 = c5_new
                    patience_count = 0
                else:
                    patience_count += 1
            else:
                patience_count += 1
        except:
            patience_count += 1
            
        # Check if we're making progress
        if abs(prev_best - best_c5) < 1e-6:
            patience_count += 1
        else:
            prev_best = best_c5
            patience_count = 0
            
        # If we're stuck, try a different approach
        if patience_count > patience:
            # Random restart
            current_h = np.random.rand(n_points) * 0.5 + 0.25
            h_sum = np.sum(current_h)
            if h_sum > 0:
                current_h = current_h * (n_points / 2) / h_sum
            patience_count = 0
        else:
            current_h = best_h.copy()
    
    # Final optimization
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            h_sum = np.sum(final_h)
            if h_sum > 0:
                final_h = final_h * (n_points / 2) / h_sum
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
