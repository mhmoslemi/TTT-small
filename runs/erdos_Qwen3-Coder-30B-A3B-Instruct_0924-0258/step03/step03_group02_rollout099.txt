import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with random order
    max_iterations = 500
    tolerance = 1e-6
    search_grid_size = 100
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Randomize order for this iteration
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        h_current = best_h.copy()
        
        for i in indices:
            if time() - start_time > budget_s * 0.95:
                break
            
            # Create a copy of current h
            h_temp = h_current.copy()
            
            # Grid search on h[i]
            old_val = h_temp[i]
            best_val = old_val
            best_c5_local = best_c5
            
            # Create search range
            search_min = max(0, old_val - 0.2)
            search_max = min(1, old_val + 0.2)
            
            if search_min >= search_max:
                continue
                
            # Grid search
            search_points = np.linspace(search_min, search_max, search_grid_size)
            
            for candidate_val in search_points:
                h_temp[i] = candidate_val
                h_temp = project_to_feasible(h_temp, n_points)
                
                c5_test = evaluate_erdos_solution(h_temp, dx)
                if c5_test < best_c5_local:
                    best_c5_local = c5_test
                    best_val = candidate_val
                    improved = True
                    
            # Update h_temp using best value
            h_temp[i] = best_val
            h_temp = project_to_feasible(h_temp, n_points)
            
            # Update current solution
            h_current = h_temp.copy()
            
            # Check if we improved the global solution
            if best_c5_local < best_c5:
                best_c5 = best_c5_local
                best_h = h_current.copy()
        
        # If no improvement in this iteration, break
        if not improved:
            break
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
