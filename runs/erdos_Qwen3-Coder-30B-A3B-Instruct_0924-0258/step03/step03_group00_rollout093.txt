import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try Fourier-based search with periodic functions
    def fourier_to_h(coeffs, n_points):
        # Create a Fourier basis with cosines
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + coeffs[1] * np.cos(np.pi * x) + coeffs[2] * np.cos(2 * np.pi * x) + \
            coeffs[3] * np.cos(3 * np.pi * x) + coeffs[4] * np.cos(4 * np.pi * x) + \
            coeffs[5] * np.cos(5 * np.pi * x)
        # Apply logistic transform to map to [0,1]
        h = 1 / (1 + np.exp(-g))
        # Normalize to sum to n_points/2 (i.e., integral = 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Optimization parameters
    num_restarts = 20
    max_iter = 30
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random starting point for Fourier coefficients
        coeffs_start = np.random.uniform(-5, 5, 6)
        
        def objective(coeffs):
            h_values = fourier_to_h(coeffs, n_points)
            c5 = evaluate_erdos_solution(h_values, dx)
            return c5
        
        # Bounds for coefficients
        bounds = [(-10, 10)] * 6
        
        try:
            result = minimize(objective, coeffs_start, method='SLSQP',
                            bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-6})
            
            if result.success:
                h_final = fourier_to_h(result.x, n_points)
                c5_new = evaluate_erdos_solution(h_final, dx)
                
                if c5_new < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_new
                    
        except:
            continue
            
    # Final refinement step with local optimization
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run local optimizations with different starting points
        num_restarts = 10
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Perturb current best
            h_perturb = best_h + np.random.normal(0, 0.05, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
