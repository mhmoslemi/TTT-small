import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a periodic starting pattern
        n_points = 35
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
        # Ensure sum = n_points/2
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with random perturbations
    max_iter = 10000
    lr = 1e-3
    decay_rate = 0.9999
    p_perturb = 0.05
    perturb_magnitude = 0.03
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps for all lag shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)  # Convert to actual shift
        
        # Compute subgradient for the maximum overlap
        subgrad = np.zeros(n_points)
        if max_shift >= 0:
            # For shift k, derivative wrt h[i] is (1 - h[i+k]) 
            # valid range is from 0 to n_points - k - 1
            valid_range = n_points - max_shift
            subgrad[:valid_range] += 1 - best_h[max_shift:max_shift + valid_range]
        else:
            # For shift k < 0, derivative wrt h[i] is -(1 - h[i-k])
            valid_range = n_points + max_shift
            subgrad[-valid_range:] -= 1 - best_h[-max_shift:valid_range - max_shift]
            
        # Gradient descent step
        new_h = best_h - lr * subgrad
        new_h = project_to_feasible(new_h, n_points)
        
        # Occasionally add random perturbation to escape local minima
        if np.random.random() < p_perturb:
            pert = np.random.normal(0, perturb_magnitude, n_points)
            new_h = project_to_feasible(new_h + pert, n_points)
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept improvement or with some probability
        if new_c5 < best_c5 or np.random.random() < 0.01:
            best_h = new_h.copy()
            best_c5 = new_c5
        
        lr *= decay_rate
    
    # Final refinement with scipy optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 200})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
