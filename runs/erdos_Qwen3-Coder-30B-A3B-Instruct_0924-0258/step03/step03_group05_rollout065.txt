import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different resolutions
    resolutions = [35, 70]
    best_solution = (best_h.copy(), best_c5, n_points)
    
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
            
        dx = 2.0 / n_points
        
        # Initialize with best solution so far
        h_current = best_solution[0].copy()
        if n_points > len(h_current):
            # Interpolate to higher resolution
            h_current = np.interp(np.linspace(0, 2, n_points), 
                                np.linspace(0, 2, len(h_current)), h_current)
        else:
            h_current = h_current[:n_points]
        
        # Normalize to correct integral
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Try several random perturbations to escape local minima
        num_perturbations = min(50, max(10, int(100000 / n_points)))
        for _ in range(num_perturbations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Create random perturbation
            pert = np.random.normal(0, 0.02, n_points)
            h_perturbed = h_current + pert
            
            # Project to feasible region
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Renormalize
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(h_perturbed, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_perturbed.copy()
                best_solution = (best_h.copy(), best_c5, n_points)
        
        # Fine optimization with SLSQP
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Starting from the best solution found
        h_initial = best_h.copy()
        
        try:
            res = minimize(objective, h_initial, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
                    best_solution = (best_h.copy(), best_c5, n_points)
        except:
            pass
    
    return (list(best_solution[0]), best_solution[1], best_solution[2])
