import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a smooth periodic initial pattern
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based refinement with momentum
    h_curr = h_start.copy()
    velocity = np.zeros(n_points)
    learning_rate = 1e-3
    momentum = 0.9
    decay_factor = 0.9995
    perturb_prob = 0.1
    perturb_magnitude = 0.05
    
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    while time() - start_time < budget_s * 0.95:
        # Check for early termination
        if no_improve_count > max_no_improve:
            break
            
        # Compute all overlap integrals
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        max_idx = np.argmax(corr)
        # Get the shift that maximizes overlap
        max_shift = max_idx - (n_points - 1)
        
        # Create subgradient (manual computation)
        subgrad = np.zeros(n_points)
        if max_shift >= 0:
            # Elements from h_curr contribute to h_curr[0:n-shift] with 1-h_curr[shift:n]
            for i in range(n_points - max_shift):
                subgrad[i] += (1 - h_curr[i + max_shift])
        else:
            # Elements from h_curr contribute to h_curr[-shift:n] with 1-h_curr[0:n+shift]
            neg_shift = -max_shift
            for i in range(n_points - neg_shift):
                subgrad[i] += (1 - h_curr[i])
                
        # Update with momentum
        velocity = momentum * velocity - learning_rate * subgrad
        h_curr = h_curr + velocity
        
        # Apply bounds and normalization
        h_curr = project_to_feasible(h_curr, n_points)
        
        # Random perturbation
        if np.random.random() < perturb_prob:
            pert = np.random.normal(0, perturb_magnitude, n_points)
            h_curr += pert
            h_curr = project_to_feasible(h_curr, n_points)
        
        # Decrease learning rate
        learning_rate *= decay_factor
        
        # Update best if improved
        new_c5 = evaluate_erdos_solution(h_curr, dx)
        if new_c5 < best_c5:
            best_h = h_curr.copy()
            best_c5 = new_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        iter_count += 1
        
    # Final optimization using scipy minimize
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj_func(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        try:
            res = minimize(obj_func, best_h, method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                final_h = project_to_feasible(res.x, n_points)
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
