import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradients(h, dx, max_overlap):
    """Compute gradients of max_overlap w.r.t. h values"""
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-6
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, n)
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - max_overlap) / eps
    
    return grad

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_c5 = current_c5
    
    # Gradient-based optimization
    max_iter = 500
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradients(best_h, dx, best_c5)
        
        # Update with gradient descent
        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = new_h.copy()

    # Final refinement with more aggressive gradient descent
    for _ in range(200):

        new_h = best_h - step_size * grad
        new_h = project_to_feasible(new_h, n_points)
