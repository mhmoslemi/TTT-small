import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points) * 0.5 + 0.25  # Random in [0.25, 0.75]
    velocities = np.random.rand(swarm_size, n_points) * 0.01
    personal_best = swarm.copy()
    personal_best_scores = np.array([evaluate_erdos_solution(project_to_feasible(p, n_points), dx) 
                                   for p in swarm])
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            h_proj = project_to_feasible(swarm[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(h_proj, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = h_proj.copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = h_proj.copy()
                global_best_score = score
                
                if global_best_score < best_c5:
                    best_h = global_best.copy()
                    best_c5 = global_best_score
    
    # Final gradient-based refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        # Use gradient descent from the best solution found by PSO
        res = minimize(final_obj, best_h.copy(), method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'gtol': 1e-6})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
