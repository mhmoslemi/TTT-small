import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Start with a sinusoidal construction
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    iterations = 0
    accepted = 0
    total = 0
    max_iter = int(1e4)
    
    # Main SA loop
    while iterations < max_iter and (time() - start_time) < budget_s * 0.95:
        # Generate candidate by perturbing a random subset
        h_candidate = best_h.copy()
        
        # Select random points to perturb (5-10%)
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected points
        for i in indices:
            delta = np.random.normal(0, 0.05)
            h_candidate[i] += delta
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Metropolis criterion
        delta = c5_candidate - best_c5
        total += 1
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            accepted += 1
        
        # Update temperature
        iterations += 1
        if iterations % 100 == 0:
            if accepted > 0:
                acceptance_rate = accepted / total
                if acceptance_rate < 0.05:
                    T *= 0.9
                elif acceptance_rate > 0.1:
                    T *= 1.1
            accepted = 0
            total = 0
            T = max(T_min, T * alpha)
    
    # Final refinement with L-BFGS
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
