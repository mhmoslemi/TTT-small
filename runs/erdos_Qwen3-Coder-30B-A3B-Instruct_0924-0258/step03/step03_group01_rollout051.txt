import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Structured initialization - place clusters to minimize overlap
    h_init = np.zeros(n_points)
    cluster_size = 5
    cluster_centers = [0.6, 1.6]  # Positions to place clusters
    for c in cluster_centers:
        start = max(0, int((c - cluster_size/2)/dx))
        end = min(n_points, int((c + cluster_size/2)/dx))
        h_init[start:end] = 1.0 / (2*cluster_size)
    # Normalize to ensure sum is n/2
    scale = (n_points/2) / np.sum(h_init)
    h_init *= scale
    h_start = h_init
    
    # Nelder-Mead optimization with restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Evaluate optimal
    current_c5 = evaluate_erdos_solution(h_start, dx)
    if current_c5 < best_c5:
        best_h = h_start.copy()
        best_c5 = current_c5
    
    # Nelder-Mead loop
    remaining_time = budget_s - (time() - start_time)
    if remaining_time <= 10.0:
        return (list(best_h), best_c5, n_points)
    
    # Start with downsampled version
    n_reduced = min(35, n_points)
    dx_reduced = 2.0 / n_reduced
    h_reduced = h_start[:n_reduced]
    
    # Refine with more iterations
    for iteration in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Random perturbations for diversity
        h_trial = h_reduced + np.random.normal(0, 0.02, n_reduced)
        h_trial = project_feasible(h_trial)
        
        c5_trial = evaluate_erdos_solution(h_trial, dx_reduced)
        if c5_trial < best_c5:
            best_h = h_trial.copy()
            best_c5 = c5_trial

    # Final gradient descent refinement
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    con = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try several restarts
    for i in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Start from best solution
        h_restart = best_h.copy()
        
        # Add small noise
        h_restart += np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        res = minimize(objective, h_restart, method='SLSQP', 
                       bounds=bounds, constraints=con, 
                       options={'maxiter': 100, 'ftol': 1e-6})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
