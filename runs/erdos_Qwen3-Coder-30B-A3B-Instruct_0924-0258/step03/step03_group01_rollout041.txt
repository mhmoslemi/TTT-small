import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1
    
    generation = 0
    max_generations = 30
    
    while generation < max_generations and time() - start_time < budget_s * 0.95:
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: top 50%
        selected = population[:population_size//2]
        
        # Crossover: create offspring
        offspring = []
        while len(offspring) < population_size - len(selected):
            if random.random() < crossover_rate:
                parent1, parent2 = random.sample(selected, 2)
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                offspring.append(child)
            else:
                # Mutation: add noise
                parent = random.choice(selected)
                child = parent + np.random.normal(0, 0.03, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                offspring.append(child)
        
        # Repair offspring to maintain feasibility
        for i in range(len(offspring)):
            h = offspring[i]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            offspring[i] = h
        
        # Create new population
        population = selected + offspring
        
        # Local refinement
        if generation % sqp_interval == 0:
            for i in range(min(3, len(population))):
                h = population[i].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    # Use a small random perturbation to avoid being too close to previous
                    h_perturbed = h + np.random.normal(0, 0.01, n_points)
                    h_perturbed = np.clip(h_perturbed, 0, 1)
                    h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                    
                    res = minimize(objective, h_perturbed, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 20, 'ftol': 1e-5})
                    if res.success:
                        new_h = res.x
                        new_c5 = objective(new_h)
                        if new_c5 < fitness[i]:
                            population[i] = new_h
                except:
                    pass
        
        generation += 1
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final refinement with SLSQP
    for _ in range(5):
        if time() - start_time > budget_s * 0.98:
            break
            
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_restart, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
