import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    max_iter = 10000
    patience = 100
    patience_counter = 0
    best_c5_so_far = best_c5
    
    # Gradient-based optimization with constraint handling
    def compute_gradient(h_vec, dx):
        """Compute gradient of max overlap with respect to h"""
        # Compute overlaps for all shifts
        h = h_vec.copy()
        n = len(h)
        grad = np.zeros(n)
        
        # For each shift k, compute gradient contribution
        for k in range(1, n):  # skip k=0
            # valid indices for shift k
            valid_len = n - k
            if valid_len <= 0:
                continue
            # Gradient of h[i] * (1 - h[i+k]) w.r.t. h[i] is (1 - h[i+k])
            # Gradient w.r.t. h[i+k] is -h[i]
            for i in range(valid_len):
                grad[i] += (1 - h[i + k]) * dx
                if i + k < n:
                    grad[i + k] -= h[i] * dx
        
        return grad
    
    def project_h(h_vec):
        """Project h to feasible region"""
        h_proj = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_proj)
        if h_sum > 0:
            h_proj = h_proj * (n_points / 2) / h_sum
        return h_proj
    
    # Main optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(best_h, dx)
        
        # Simple gradient descent step
        step_size = 0.01 / (1.0 + i / 1000.0)
        h_new = best_h - step_size * grad
        
        # Project to feasible region
        h_new = project_h(h_new)
        
        # Check improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            patience_counter = 0
            if best_c5 < best_c5_so_far:
                best_c5_so_far = best_c5
        else:
            patience_counter += 1
            
        # Random perturbation every 50 iterations
        if i % 50 == 0 and patience_counter > 30:
            perturbation = np.random.normal(0, 0.02, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = project_h(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        # Early stopping if no improvement for too long
        if patience_counter > patience:
            break
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final refinement with scipy optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
