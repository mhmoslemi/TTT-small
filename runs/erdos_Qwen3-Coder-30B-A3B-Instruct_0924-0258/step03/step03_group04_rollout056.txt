import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with perturbations
    max_iter = 100000
    lr = 1e-3
    decay = 0.9999
    p_perturb = 0.1
    perturb_magnitude = 0.05
    
    h = best_h.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute subgradient
        subgrad = np.zeros(n_points)
        if max_k > 0:
            # h[i] contributes to overlap with h[i+max_k]
            subgrad[:n_points - max_k] = 1 - h[max_k:]
        else:
            # h[i] contributes to overlap with h[i-max_k]
            subgrad[-max_k:] = 1 - h[:n_points + max_k]
        
        # Apply gradient step
        h_new = h - lr * subgrad
        
        # Add random perturbations occasionally
        if random.random() < p_perturb:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_new += perturb
            
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Update current solution
        h = h_new.copy()
        lr *= decay
        
        # Early stopping if no improvement in last 100 iter
        if iteration > 100 and iteration % 100 == 0:
            if abs(c5_new - best_c5) < 1e-8:
                break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5 and best_c5 > 0.375:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
