import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available, else use default
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    c_low = 0.37
    c_high = 0.38085
    iterations = 15
    
    for _ in range(iterations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        c_mid = (c_low + c_high) / 2.0
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1]
        
        # Sum constraint
        constraints.append(sum(h) * dx == 1.0)
        
        # Overlap constraints for all k shifts
        # Note: Only considering k from 1 to n-1 due to symmetric nature and indexing
        max_k = n_points - 1
        for k in range(1, max_k):
            valid_len = n_points - k
            if valid_len > 0:
                corr_sum = sum(h[i] * (1 - h[i + k]) for i in range(valid_len)) * dx
                constraints.append(corr_sum <= c_mid)
        
        objective = Minimize(0)  # Dummy objective, we just want feasibility
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_solution = h.value
                if h_solution is not None:
                    h_solution = np.clip(h_solution, 0, 1)
                    h_sum = np.sum(h_solution)
                    if h_sum > 0:
                        h_solution = h_solution * (n_points / 2) / h_sum
                    c5_new = evaluate_erdos_solution(h_solution, dx)
                    if c5_new < best_c5:
                        best_h = h_solution.copy()
                        best_c5 = c5_new
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
    
    # Local refinement with multiple restarts
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Different restarts with improved strategies
        num_restarts = 20
        for i in range(num_restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            # Strategy 1: Start from best solution with noise
            h_perturb = best_h + np.random.normal(0, 0.02, n_points)
            
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue

    return (list(best_h), best_c5, n_points)
