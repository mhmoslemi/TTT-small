import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Configuration
    max_segments = 40
    max_iter = 200
    initial_segments = 15
    lambda_penalty = 1000.0
    initial_step_size = 0.1
    backtrack_factor = 0.5
    min_improvement = 1e-8
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        segments = max(10, min(max_segments, n_points // 2))
    else:
        n_points = 30
        segments = initial_segments
        h_start = np.ones(n_points) * 0.5

    # Piecewise constant representation
    def to_piecewise(h_vals, n_segments):
        # Initialize segment boundaries
        boundaries = np.linspace(0, 2, n_segments + 1)
        # Initialize segment values
        values = np.ones(n_segments) * 0.5
        # Assign initial values from h_vals
        for i in range(n_segments):
            start_idx = int(i * n_points / n_segments)
            end_idx = int((i + 1) * n_points / n_segments)
            if end_idx > start_idx:
                values[i] = np.mean(h_vals[start_idx:end_idx])
            else:
                values[i] = 0
        return boundaries, values

    def from_piecewise(boundaries, values):
        # Reconstruct h from piecewise representation
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        for i in range(len(values)):
            start_idx = int(boundaries[i] / dx)
            end_idx = int(boundaries[i+1] / dx)
            if end_idx > start_idx:
                h[start_idx:end_idx] = values[i]
        return h
    
    def compute_constraints(boundaries, values):
        total_integral = 0
        for i in range(len(values)):
            width = boundaries[i+1] - boundaries[i]
            total_integral += values[i] * width
        return total_integral
    
    def project_piecewise(boundaries, values):
        # Ensure values are in [0,1]
        values = np.clip(values, 0, 1)
        # Rescale to meet integral constraint
        integral = sum(values[i] * (boundaries[i+1] - boundaries[i]) for i in range(len(values)))
        if integral > 0:
            scale = (2.0 / 2.0) / integral  # target integral is 1
            values = values * scale
        return boundaries, values

    def compute_gradient(boundaries, values):
        # Compute finite differences for gradient approx
        h = from_piecewise(boundaries, values)
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        grad_w = np.zeros_like(values)
        grad_x = np.zeros_like(boundaries)
        
        eps = 1e-6
        for i in range(len(values)):
            # Perturb value
            values_perturbed = values.copy()
            values_perturbed[i] += eps
            _, values_proj = project_piecewise(boundaries, values_perturbed)
            h_perturbed = from_piecewise(boundaries, values_proj)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            score_perturbed = np.max(corr_perturbed) * dx
            
            grad_w[i] = (score_perturbed - max_overlap) / eps
            
            # Perturb boundary (left Neighbor)
            boundaries_perturbed = boundaries.copy()
            boundaries_perturbed[i] += eps
            _, values_proj = project_piecewise(boundaries_perturbed, values)
            h_perturbed = from_piecewise(boundaries_perturbed, values_proj)
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            score_perturbed = np.max(corr_perturbed) * dx
            
            grad_x[i] = (score_perturbed - max_overlap) / eps
            
        return grad_w, grad_x

    # Start with piecewise version
    boundaries, values = to_piecewise(h_start, segments)
    
    best_c5 = float('inf')
    best_original_h = None
    best_n_points = None
    
    # Optimization loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Current solution
        h_current = from_piecewise(boundaries, values)
        dx_current = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_current, dx_current)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_original_h = h_current.copy()
            best_n_points = n_points
            
        # Compute gradients
        try:
            grad_w, grad_x = compute_gradient(boundaries, values)
        except:
            break
        
        # Update values with respect to weights
        step_size = initial_step_size
        new_values = values - step_size * grad_w
        new_bounds = boundaries - step_size * grad_x
        
        new_bounds, new_values = project_piecewise(new_bounds, new_values)
        
        # Backtracking line search
        while True:
            h_try = from_piecewise(new_bounds, new_values)
            h_try = project_to_feasible(h_try, n_points)
            dx_try = 2.0 / n_points
            try:
                c5_try = evaluate_erdos_solution(h_try, dx_try)
                if c5_try < current_c5 - min_improvement:
                    break
                step_size *= backtrack_factor
                if step_size < 1e-10:
                    break
                new_values = values - step_size * grad_w
                new_bounds = boundaries - step_size * grad_x
                new_bounds, new_values = project_piecewise(new_bounds, new_values)
            except:
                break
        
        values = new_values
        boundaries = new_bounds
        
        # Refine segments every few iterations
        if it % 50 == 0 and len(values) < max_segments:
            # Split segments with high gradient
            max_grad_idx = np.argmax(np.abs(grad_w))
            if len(values) < max_segments:
                # Insert a new boundary at midpoint of segment
                old_len = len(values)
                values = np.insert(values, max_grad_idx, values[max_grad_idx])
                boundaries = np.insert(boundaries, max_grad_idx+1, 
                                     (boundaries[max_grad_idx] + boundaries[max_grad_idx+1]) / 2)
                # Renormalize if needed 
                new_bounds, new_values = project_piecewise(boundaries, values)
                boundaries = new_bounds 
                values = new_values

    # Final gradient descent
    if best_original_h is not None and time() - start_time < budget_s * 0.9:
        h_final = best_original_h.copy()
        dx_final = 2.0 / best_n_points
        
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Project to feasible
            h_current = project_to_feasible(h_final, best_n_points)
            dx_current = 2.0 / best_n_points
            c5_current = evaluate_erdos_solution(h_current, dx_current)
            
            # Approximate gradient
            grad = np.zeros_like(h_current)
            eps = 1e-5
            
            for i in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, best_n_points)
                c5_plus = evaluate_erdos_solution(h_plus, dx_current)
                grad[i] = (c5_plus - c5_current) / eps
            
            # Update
            step_size = 0.001
            h_final = h_current - step_size * grad
            h_final = project_to_feasible(h_final, best_n_points)
            
            c5_new = evaluate_erdos_solution(h_final, dx_current)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_original_h = h_final.copy()

    return (list(best_original_h), best_c5, best_n_points)
