def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(arr, eps=1e-8):
    """Smooth approximation to max for gradient-based optimization"""
    return np.log(np.sum(np.exp(arr / eps))) * eps

def construct_piecewise_segments(n_segments, x_boundary=None, values=None):
    """Construct piecewise segments"""
    if x_boundary is None:
        x_boundary = np.linspace(0, 2, n_segments + 1)
    if values is None:
        values = np.ones(n_segments) * 0.5
    return x_boundary, values

def compute_overlap_segments(x_boundary, values, dx):
    """Compute overlap for piecewise segments"""
    n_segments = len(values)
    overlaps = []
    # For each shift k, compute overlap
    # We will sample multiple k values including 0, 0.1, 0.2, ...
    k_values = np.arange(0, 2.1, 0.1)
    for k in k_values:
        overlap = 0
        for i in range(n_segments):
            for j in range(n_segments):
                x_i_start, x_i_end = x_boundary[i], x_boundary[i+1]
                x_j_start, x_j_end = x_boundary[j], x_boundary[j+1]
                # Overlap of segments when segment j is shifted by k
                overlap_start = max(x_i_start, x_j_start + k)
                overlap_end = min(x_i_end, x_j_end + k)
                if overlap_end > overlap_start:
                    overlap += values[i] * (1 - values[j]) * (overlap_end - overlap_start)
        overlaps.append(overlap)
    return max(overlaps)
