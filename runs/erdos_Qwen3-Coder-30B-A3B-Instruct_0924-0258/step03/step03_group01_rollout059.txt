import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try to improve with a hybrid approach:
    # 1. Start with good initial solution
    # 2. Use particle swarm optimization for global exploration
    # 3. Refine using convex optimization
    
    def particle_swarm_optimization():
        # PSO parameters
        n_particles = 20
        max_iter = 300
        w = 0.7  # inertia weight
        c1 = 1.5  # cognitive parameter
        c2 = 1.5  # social parameter
        
        # Initialize particles
        particles = []
        velocities = []
        
        for _ in range(n_particles):
            # Start near initial solution with some perturbation
            h_init = h_start + np.random.normal(0, 0.05, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)  # normalize sum
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        
        # Global best 
        gbest = best_h.copy()
        gbest_c5 = best_c5
        gbest_idx = 0
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update each particle
            for i in range(n_particles):
                h = particles[i]
                v = velocities[i]
                
                # Update velocity
                r1, r2 = np.random.rand(2)
                cognitive = c1 * r1 * (h_start - h)  # inertia toward initial
                social = c2 * r2 * (gbest - h)
                v = w * v + cognitive + social
                
                # Update position
                h_new = h + v
                
                # Keep within bounds
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate
                c5_new = evaluate_erdos_solution(h_new, dx)
                
                # Update particle's best
                current_c5 = evaluate_erdos_solution(h, dx)
                if c5_new < current_c5:
                    particles[i] = h_new.copy()
                
                # Update global best
                if c5_new < gbest_c5:
                    gbest = h_new.copy()
                    gbest_c5 = c5_new
                    gbest_idx = i
                    
            # Update inertia weight
            w = 0.9 - (0.7 * iteration / max_iter)
            
        return gbest, gbest_c5

    # Run PSO
    try:
        h_pso, c5_pso = particle_swarm_optimization()
        if c5_pso < best_c5:
            best_h = h_pso
            best_c5 = c5_pso
    except Exception:
        pass

    # Refine with convex optimization (if time allows)
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            # Convex optimization approach
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Create an approximate upper bound constraint
            # Maximize the C5 bound 
            # Use lower bound for feasible problem
            # We don't know a tight bound, use a reasonable estimate
            c5_upper = best_c5 + 0.01
            
            # Process overlaps for a range of shifts that matter most
            corr_max = 0.0
            for k in range(1, n_points):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                
                constraints.append(corr_sum <= c5_upper * dx)
            
            # Convex optimization to try to find minimal C5
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            pass
            
    # Final saturate refinement
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        # Refine using scipy minimize with SLSQP
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine using a simple restart strategy
        for restart in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    except:
        pass

    return (list(best_h), best_c5, n_points)
