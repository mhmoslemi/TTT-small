import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_length = 10
    max_iter = 10000
    temp = 1.0
    cooling_rate = 0.9995
    
    # Perturbations for binary search
    perturbations = [0.0, 1.0, 0.5]
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Make a copy of current solution
        h_current = best_h.copy()
        
        # Select random segment to modify
        segment_idx = np.random.randint(0, n_points)
        
        # Get current value
        current_val = h_current[segment_idx]
        
        # Try different perturbations
        candidates = []
        for p in perturbations:
            if abs(p - current_val) > 1e-6:  # Only consider different values
                h_test = h_current.copy()
                h_test[segment_idx] = p
                c5_test = evaluate_erdos_solution(h_test, dx)
                candidates.append((h_test, c5_test))
        
        if not candidates:
            continue
            
        # Find best candidate
        best_candidate = min(candidates, key=lambda x: x[1])
        h_new, c5_new = best_candidate
        
        # Accept if better or with simulated annealing probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            # Update best if improved
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                
            # Update tabu list
            tabu_list.append(segment_idx)
            if len(tabu_list) > tabu_length:
                tabu_list.pop(0)
                
        # Cooling schedule
        temp *= cooling_rate
        
        # Local refinement with gradient descent
        if iteration % 100 == 0 and iteration > 0:
            if time.time() - start_time > budget_s * 0.95:
                break
                
            def obj_func(h):
                h_clipped = np.clip(h, 0, 1)
                h_sum = np.sum(h_clipped)
                if h_sum > 0:
                    h_normalized = h_clipped * (n_points / 2) / h_sum
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)

            def constraint_sum(h):
                return np.sum(h) - n_points / 2

            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Run local optimization
            try:
                res = minimize(obj_func, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
