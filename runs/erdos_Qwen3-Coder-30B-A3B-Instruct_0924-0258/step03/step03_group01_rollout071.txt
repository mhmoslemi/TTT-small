import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 100
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_max
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial solution
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random solution
            h = np.random.uniform(0, 1, n_points)
        # Normalize to sum = 17.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
        
        if pbest_values[-1] < best_c5:
            best_h = h.copy()
            best_c5 = pbest_values[-1]
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        inertia_weight = w_min + (w_max - w_min) * (1 - iteration / max_iter)
        
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Repair feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                if current_value < best_c5:
                    best_h = particles[i].copy()
                    best_c5 = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_value
                gbest_idx = i
        
        # Local refinement every 5 iterations
        if iteration % 5 == 0 and iteration > 0:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time < 30:
                break
                
            # Refine top 3 particles
            top_indices = np.argsort(pbest_values)[:3]
            for idx in top_indices:
                if time() - start_time > budget_s * 0.95:
                    break
                    
                h_restart = pbest[idx].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
