import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Periodic initialization
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    iterations = 0
    max_iter = 10000
    acceptance_count = 0
    max_acceptance = 100
    
    # Initial temperature based on random walk
    current_h = h_start.copy()
    current_c5 = best_c5
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iter:
        # Perturb current solution
        h_candidate = current_h.copy()
        
        # Select a random subset of indices to perturb (5-10%)
        num_perturb = max(1, int(0.05 * n_points))
        indices = random.sample(range(n_points), num_perturb)
        
        # Apply small random changes
        for i in indices:
            delta = random.uniform(-0.1, 0.1)
            h_candidate[i] += delta
        
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta_c5 = c5_candidate - current_c5
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / temp):
            current_h = h_candidate.copy()
            current_c5 = c5_candidate
            acceptance_count += 1
            
            # Update best solution if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Update temperature
        iterations += 1
        if iterations % 100 == 0:
            temp *= cooling_rate
            temp = max(temp, min_temp)
        
        # Adapt temperature based on acceptance rate
        if acceptance_count > max_acceptance:
            temp *= 0.95
            acceptance_count = 0
        elif acceptance_count == 0:
            temp *= 1.05
    
    # Final refinement with LBFGS
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        from scipy.optimize import minimize
        
        def objective(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
