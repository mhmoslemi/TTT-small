import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different starting configurations
    for attempt in range(3):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate random starting points with better structure
        if attempt == 0:
            h_trial = h_start.copy()
        elif attempt == 1:
            # More structured initial guess 
            h_trial = np.ones(n_points) * 0.5
            # Add some structure - peaks in middle
            mid = n_points // 2
            h_trial[mid-2:mid+3] = 0.8
        else:
            # Random with constraints
            h_trial = np.random.rand(n_points) * 0.8 + 0.1
        
        # Normalize to satisfy integral constraint
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Local optimization with bounds
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue

    # Further refinement using a global optimizer approach
    # Using a simple evolutionary approach for higher quality
    if time() - start_time < budget_s * 0.8:
        # Random search nearby the best
        for i in range(50):
            if time() - start_time > budget_s * 0.95:
                break
            h_trial = best_h + np.random.normal(0, 0.02, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                
    # Final verification that we stay within bounds
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum

    return (list(best_h), best_c5, n_points)
