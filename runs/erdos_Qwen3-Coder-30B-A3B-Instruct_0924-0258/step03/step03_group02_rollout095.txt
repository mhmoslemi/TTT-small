import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_overlaps(h, dx):
    """Compute all overlap integrals for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with random order
    max_iterations = 500
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Get random order for coordinate update
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        prev_c5 = best_c5
        
        for i in indices:
            # Save current h
            h_current = best_h.copy()
            
            # Try to optimize h[i] in isolation
            # We need to assess impact on C5, so we compute the change
            # due to varying h[i] while keeping others fixed
            h_optimal = h_current.copy()
            
            # Simple grid search for h[i] in [0,1]
            best_h_i = h_current[i]
            min_c5 = prev_c5
            
            # Try values in a grid around current value
            test_values = np.linspace(0, 1, 50)
            for val in test_values:
                h_optimal[i] = val
                h_optimal = project_to_feasible(h_optimal, n_points)
                c5_val = compute_overlaps(h_optimal, dx)
                if c5_val < min_c5:
                    min_c5 = c5_val
                    best_h_i = val
                    
            # Update h[i] to best found
            h_current[i] = best_h_i
            h_current = project_to_feasible(h_current, n_points)
            
            new_c5 = compute_overlaps(h_current, dx)
            if new_c5 < best_c5 - tolerance:
                best_h = h_current.copy()
                best_c5 = new_c5
                improved = True
            
        # If no significant improvement, break
        if not improved and iteration > 10:
            break
    
    # Refinement with SLSQP
    if time() - start_time < budget_s * 0.9:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
