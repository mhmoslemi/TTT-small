import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Structured initialization with clustered regions
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Nelder-Mead with restarts for global search
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Various heuristic restarts
    restarts = []
    for i in range(10):
        h_restart = h_start.copy() + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        restarts.append(h_restart)
    
    # Refinement using scipy minimize with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Main optimization loop
    max_iters = 20
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current candidates
        candidates = []
        for i in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            h_current = restarts[i]
            c5_current = objective(h_current)
            candidates.append((h_current, c5_current))
            
        # Sort by objective
        candidates.sort(key=lambda x: x[1])
        best_candidate = candidates[0]
        
        # Update global best
        if best_candidate[1] < best_c5:
            best_h = best_candidate[0]
            best_c5 = best_candidate[1]
        
        # Gradient descent refinement on top candidates
        for i in range(min(3, len(candidates))):
            if time() - start_time > budget_s * 0.95:
                break
            try:
                h_refined = minimize(objective, candidates[i][0], 
                                    method='SLSQP', bounds=bounds, 
                                    constraints=cons_sum, 
                                    options={'maxiter': 50, 'ftol': 1e-6})
                if h_refined.success:
                    c5_refined = objective(h_refined.x)
                    if c5_refined < best_c5:
                        best_h = h_refined.x
                        best_c5 = c5_refined
            except:
                continue
        
        # Generate new candidates via perturbation
        new_restarts = []
        for h in candidates[:5]:
            h_new = h[0] + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_restarts.append(h_new)
        restarts = new_restarts
    
    return (list(best_h), best_c5, n_points)
