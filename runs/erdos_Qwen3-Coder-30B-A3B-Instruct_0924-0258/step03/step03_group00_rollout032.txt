import time
from scipy.optimize import minimize
from scipy.spatial.distance import pdist

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    tabu_list = []
    tabu_length = 10
    
    # Local search with simulated annealing
    max_iter = 50000
    temp = 1.0
    cooling_rate = 0.9995
    stagnation_count = 0
    max_stagnation = 1000
    
    current_h = h_start.copy()
    current_c5 = best_c5

        h_perturbed = current_h.copy()

        perturbations = [0.0, 0.1, -0.1, 0.05, -0.05]

        best_pert_c5 = current_c5

            h_test = h_perturbed.copy()
            h_test[idx] += delta
            h_test[idx] = np.clip(h_test[idx], 0, 1)

            h_sum = np.sum(h_test)
            if h_sum > 0:
                h_test = h_test * (n_points / 2) / h_sum
            
            c5_test = evaluate_erdos_solution(h_test, dx)

                stagnation_count = 0
            else:
                stagnation_count += 1

        temp *= cooling_rate

        if stagnation_count > max_stagnation:
            break
