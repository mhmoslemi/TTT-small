import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trying different optimization approaches
    max_iterations = 100
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Generate a new candidate using randomized approach
        h_new = np.random.uniform(0, 1, n_points)
        h_new = project_to_feasible(h_new, n_points)
        
        # Random search refinement
        def obj_func(h):
            return evaluate_erdos_solution(h, dx)
        
        # Simple random walk around current best
        for _ in range(10):
            if time() - start_time > budget_s * 0.9:
                break
            h_trial = h_new + np.random.normal(0, 0.01, n_points)
            h_trial = project_to_feasible(h_trial, n_points)
            c5_trial = obj_func(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                h_new = h_trial.copy()
        
        # SLSQP refinement of best current
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_func(h):
            return np.sum(h) - n_points / 2
        
        try:
            res = minimize(obj_func, best_h, method='SLSQP',
                          bounds=bounds, constraints={'type': 'eq', 'fun': constraint_func},
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = obj_func(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
