import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 30
    w_start, w_end = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(swarm_size):
        # Start with perturbed initial solution or random
        if i < swarm_size // 2:
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - swarm[i]) + 
                            c2 * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Ensure feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_val < pbest_vals[i]:
                pbest[i] = swarm[i].copy()
                pbest_vals[i] = current_val
                
                # Update global best
                if current_val < gbest_val:
                    gbest = swarm[i].copy()
                    gbest_val = current_val
        
        # Perform local refinement every 5 iterations
        if iteration % 5 == 0 and iteration > 0:
            for i in range(min(3, swarm_size)):
                # Take top candidates
                ref_idx = np.argsort(pbest_vals)[:3][i]
                h_ref = pbest[ref_idx].copy()
                
                # Local refinement using SLSQP
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_ref, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    # Final refine best solution if needed
    if best_c5 > gbest_val:
        best_h = gbest
        best_c5 = gbest_val
    
    return (list(best_h), best_c5, n_points)
