import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhancement: Use a different optimization approach
    # Start with a more structured approach using wavelet-like basis
    
    # Try a few different initializations
    initial_configs = []
    
    # Config 1: Smoothed version of the starting point
    h1 = best_h.copy()
    h1 = np.clip(h1, 0, 1)
    h1 = h1 * (n_points / 2) / np.sum(h1)
    initial_configs.append(h1)
    
    # Config 2: Piecewise constant with alternating values
    h2 = np.ones(n_points)
    h2[::2] = 0.2
    h2[1::2] = 0.8
    h2 = h2 * (n_points / 2) / np.sum(h2)
    initial_configs.append(h2)
    
    # Config 3: Two-piece step function
    h3 = np.ones(n_points)
    mid = n_points // 2
    h3[:mid] = 0.3
    h3[mid:] = 0.7
    h3 = h3 * (n_points / 2) / np.sum(h3)
    initial_configs.append(h3)
    
    for h_test in initial_configs:
        if time() - start_time > budget_s * 0.95:
            break
        c5 = evaluate_erdos_solution(h_test, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_test.copy()
    
    # Refinement using a more sophisticated optimization
    # Perform a two-stage approach:
    # Stage 1: Each point is optimized with constraints
    # Stage 2: Try a global smoother
    
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    # Constrained optimization for the best solution so far
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different methods
    methods = ['SLSQP', 'L-BFGS-B']
    for method in methods:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            res = minimize(objective, best_h, method=method, 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                # Renormalize
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue
    
    # Additional tuned smoothing approach
    if time() - start_time < budget_s * 0.9:
        # Try to improve using a gradient-free approach with multiple restarts
        best_h_local = best_h.copy()
        best_c5_local = best_c5
        
        # Perturb the solution slightly and optimize
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            h_perturb = best_h_local + np.random.normal(0, 0.02, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            
            try:
                res = minimize(objective, h_perturb, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_new = np.clip(res.x, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5_local:
                        best_c5_local = c5_new
                        best_h_local = h_new
            except:
                continue
        
        if best_c5_local < best_c5:
            best_h = best_h_local
            best_c5 = best_c5_local

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
