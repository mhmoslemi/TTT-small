import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iters = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = np.random.uniform(0, 1, (n_particles, n_points))
    for i in range(n_particles):
        particles[i] = project_to_feasible(particles[i], n_points)
    
    # Personal best positions and their fitness
    personal_best = particles.copy()
    personal_best_fitness = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Global best
    best_idx = np.argmin(personal_best_fitness)
    global_best = personal_best[best_idx].copy()
    global_best_fitness = personal_best_fitness[best_idx]
    
    if global_best_fitness < best_c5:
        best_h = global_best.copy()
        best_c5 = global_best_fitness
    
    # Velocities
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # PSO iterations
    for iteration in range(max_iters):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iters
        
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best[i] - particles[i])
                           + c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = fitness
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best = particles[i].copy()
                    global_best_fitness = fitness
                    
                    if global_best_fitness < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_fitness
    
    # Post-processing: refine with SLSQP if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            # Refine the global best
            res = minimize(refine_objective, global_best, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
