import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(h, beta):
    """Approximate max with smooth surrogate"""
    corr = np.correlate(h, 1 - h, mode='full')
    exp_vals = np.exp(beta * corr)
    return (1/beta) * np.log(exp_vals.sum())

def gradient(h, beta, eps=1e-5):
    """Approximate gradient via finite differences"""
    grad = np.zeros_like(h)
    base_val = smooth_max(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_max(h_plus, beta) - base_val) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Project to feasible region
    h = project_to_feasible(h, n_points)
    
    # Parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    
    # Initial evaluation
    best_c5 = smooth_max(h, beta)
    best_h = h.copy()
    best_n_points = n_points
    
    # Gradient descent loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = gradient(h, beta)
        
        # Line search step
        trial_h = h - step_size * grad
        trial_h = project_to_feasible(trial_h, n_points)
        trial_c5 = smooth_max(trial_h, beta)
        
        # Accept step if improvement
        if trial_c5 < best_c5:
            h = trial_h.copy()
            best_c5 = trial_c5
            # Increase step size if small
            if step_size < 1e-3:
                step_size *= 1.5
        else:
            # Reduce step size
            step_size *= 0.5
            
        # Update beta for sharper approximation
        if iteration % 100 == 0 and iteration > 0:
            beta *= 1.2
            
        # Check for convergence
        if iteration > 50 and abs(best_c5 - smooth_max(h, beta)) < tolerance:
            break
            
    # Final evaluation with exact maximum
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Try resolution refinement if beneficial and time allows
    if time() - start_time < budget_s * 0.9:
        for n_mult in [1.5, 2.0]:
            n_try = int(n_points * n_mult)
            if n_try > 100:
                break
            try:
                h_try = np.interp(np.linspace(0, 2, n_try), 
                                 np.linspace(0, 2, n_points), 
                                 best_h)
                h_try = project_to_feasible(h_try, n_try)
                c5_try = evaluate_erdos_solution(h_try, 2.0 / n_try)
                if c5_try < final_c5:
                    final_c5 = c5_try
                    best_h = h_try
                    best_n_points = n_try
            except:
                continue
    
    return (list(best_h), final_c5, best_n_points)
