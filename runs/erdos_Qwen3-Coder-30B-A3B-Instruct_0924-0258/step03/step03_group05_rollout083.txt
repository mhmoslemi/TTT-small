import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def normalize(h):
        """Ensure sum(h) * dx == 1"""
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def mutate(h, rate=0.2):
        """Apply mutation with given rate"""
        h_mut = h.copy()
        for i in range(len(h)):
            if random.random() < rate:
                h_mut[i] += np.random.normal(0, 0.1)
        h_mut = np.clip(h_mut, 0, 1)
        return normalize(h_mut)
    
    def crossover(h1, h2):
        """One-point crossover"""
        if len(h1) != len(h2):
            return h1.copy()
        point = random.randint(1, len(h1) - 1)
        h_new = np.concatenate([h1[:point], h2[point:]])
        return normalize(h_new)
    
    def tournament_selection(pop, fitnesses, k=5):
        """Select one individual using tournament selection"""
        idxs = random.sample(range(len(pop)), min(k, len(pop)))
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        return pop[best_idx].copy()
    
    # Population-based GA
    pop_size = 50
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h_rand = np.random.random(n_points)
        h_rand = normalize(h_rand)
        population.append(h_rand)
    
    # Fitness evaluation
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitnesses[best_idx]
    
    stagnation_counter = 0
    generations = 0
    max_generations = 1000
    
    # Run GA
    while time() - start_time < budget_s * 0.95 and generations < max_generations:
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite = [population[i] for i in sorted_indices[:pop_size//4]]
        
        new_population = elite.copy()
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child = crossover(parent1, parent2)
            child = mutate(child)
            
            new_population.append(child)
        
        # Evaluate new population
        population = new_population[:pop_size]
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        generations += 1
        
        if stagnation_counter > 100:
            break
    
    # Final refinement with local optimization
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
