import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier parameterization approach
    d = 5  # Number of cosine terms
    max_iter = 20
    max_evals = 100
    
    def fourier_to_h(coeffs):
        # Convert Fourier coefficients to h values via logistic transform
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + sum(coeffs[i+1] * np.cos((i+1) * np.pi * x) for i in range(d))
        h_vals = 1.0 / (1.0 + np.exp(-g))
        # Normalize to integral 1
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return np.clip(h_vals, 0, 1)
    
    def objective(coeffs):
        h_vals = fourier_to_h(coeffs)
        return evaluate_erdos_solution(h_vals, dx)
    
    # Start with the initial construction
    best_coeffs = np.zeros(d + 1)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random restarts
        best_local = float('inf')
        best_local_coeffs = best_coeffs.copy()
        
        for _ in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Perturb initial coeffs
            coeffs0 = best_coeffs + np.random.normal(0, 0.5, d + 1)
            
            # Optimize with bounds
            bounds = [(-10, 10) for _ in range(d + 1)]
            
            res = minimize(objective, coeffs0, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': max_evals})
            
            if res.success:
                val = res.fun
                if val < best_local:
                    best_local = val
                    best_local_coeffs = res.x.copy()
        
        # Update global best
        if best_local < best_c5:
            best_c5 = best_local
            best_coeffs = best_local_coeffs.copy()
            best_h = fourier_to_h(best_coeffs)
        
        # Early stopping if improvement is minimal
        if best_c5 < 0.3807:
            break
    
    return (list(best_h), best_c5, n_points)
