import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    max_iter = 1000
    step_size = 0.001
    n_points = 35  # Start with the given resolution
    
    # Use initialization if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, n_points)
    else:
        # Create symmetric initial guess
        h = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        h = project_to_feasible(h, n_points)
    
    # Initialize best solution
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_n_points = n_points
    
    # Gradient-based optimization with projected gradients
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Estimate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_score = evaluate_erdos_solution(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
            
        # Update with gradient descent
        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        h = h_new
        
        # Evaluate new solution
        score = evaluate_erdos_solution(h, dx)
        if score < best_c5:
            best_h = h.copy()
            best_c5 = score
            best_n_points = n_points

    # Refine at higher resolution if time allows
    if time() - start_time < budget_s * 0.8:
        n_points_high = 50
        dx_high = 2.0 / n_points_high
        h_high = np.interp(np.linspace(0, 2, n_points_high),
                          np.linspace(0, 2, len(best_h)),
                          best_h)
        h_high = project_to_feasible(h_high, n_points_high)
        best_c5_high = evaluate_erdos_solution(h_high, dx_high)
        
        if best_c5_high < best_c5:
            best_h = h_high
            best_c5 = best_c5_high
            best_n_points = n_points_high
            
        # Further refine at high resolution
        for iteration in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros_like(h_high)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_high, dx_high)
            
            for i in range(len(h_high)):
                h_plus = h_high.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points_high)
                score_plus = evaluate_erdos_solution(h_plus, dx_high)
                grad[i] = (score_plus - base_score) / eps
                
            h_high = h_high - step_size * grad
            h_high = project_to_feasible(h_high, n_points_high)
            score = evaluate_erdos_solution(h_high, dx_high)
            if score < best_c5:
                best_h = h_high.copy()
                best_c5 = score
                best_n_points = n_points_high
                
    return (list(best_h), best_c5, best_n_points)
