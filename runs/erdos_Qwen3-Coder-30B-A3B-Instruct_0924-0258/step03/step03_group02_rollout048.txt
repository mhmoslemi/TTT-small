import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iterations = 500
    tolerance = 1e-6
    iteration = 0
    
    # Coordinate Descent with random order
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.9:
        h_current = best_h.copy()
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for i in indices:
            if (time() - start_time) > budget_s * 0.9:
                break
                
            # Current h values
            h_temp = h_current.copy()
            
            # Grid search for optimal value of h[i]
            best_h_i = h_temp[i]
            best_overlap = best_c5
            
            # Search range for h[i], conservatively
            search_range = np.linspace(max(0, h_temp[i] - 0.2), 
                                     min(1, h_temp[i] + 0.2), 50)
            
            # Try to maximize overlap contributions
            for h_i_try in search_range:
                h_temp[i] = h_i_try
                h_temp = project_to_feasible(h_temp, n_points)
                overlap = evaluate_erdos_solution(h_temp, dx)
                
                if overlap < best_overlap:
                    best_overlap = overlap
                    best_h_i = h_i_try
            
            h_current[i] = best_h_i
            h_current = project_to_feasible(h_current, n_points)
            
            # Check if this was an improvement
            if best_overlap < best_c5:
                improved = True
                best_c5 = best_overlap
                best_h = h_current.copy()
        
        if not improved:
            break
    
        iteration += 1
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
