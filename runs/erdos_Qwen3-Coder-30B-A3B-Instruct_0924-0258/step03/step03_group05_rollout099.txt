import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with constraint handling
    def objective_and_gradient(h_vec):
        # Normalize
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        h_norm = np.clip(h_norm, 0, 1)
        
        # Compute overlap
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        max_overlap = np.max(corr) * dx
        
        # Gradient computation
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            for k in range(1, n_points):  # avoid k=0
                if i + k < n_points:
                    grad[i] += dx * (1 - 2 * h_norm[i + k])
        
        return max_overlap, grad
    
    # Main optimization loop
    max_iter = 10000
    iteration = 0
    patience = 0
    best_c5_so_far = best_c5
    learning_rate = 0.01
    
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Create a new candidate
        h_new = best_h.copy()
        
        # Gradient step
        current_c5, grad = objective_and_gradient(h_new)
        
        # Apply gradient step
        h_new = h_new - learning_rate * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new candidate
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            patience = 0
            best_c5_so_far = min(best_c5_so_far, new_c5)
        else:
            patience += 1
            if patience > 20:
                # Reduce learning rate and continue
                patience = 0
                learning_rate *= 0.95
                
        # Occasionally add noise to escape local optima
        if iteration % 50 == 0:
            h_noisy = best_h + np.random.normal(0, 0.001, n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points / 2) / np.sum(h_noisy)
            noisy_c5 = evaluate_erdos_solution(h_noisy, dx)
            if noisy_c5 < best_c5:
                best_h = h_noisy
                best_c5 = noisy_c5
                
        iteration += 1
        
        # Occasionally reset learning rate
        if iteration % 1000 == 0:
            learning_rate = max(0.0001, learning_rate)
    
    # Final refinement using scipy minimize
    def refine_objective(h_vec):
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        return evaluate_erdos_solution(h_norm, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n_points / 2}
    
    try:
        result = minimize(refine_objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            best_h = np.clip(result.x, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
