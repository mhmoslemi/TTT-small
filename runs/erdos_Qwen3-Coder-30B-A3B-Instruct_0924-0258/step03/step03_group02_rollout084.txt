import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = [project_to_feasible(np.random.rand(n_points), n_points) 
                  for _ in range(population_size)]
    population[0] = best_h  # Ensure we start with a good solution
    
    # Function to compute fitness (negative because we minimize)
    def compute_fitness(h):
        return -evaluate_erdos_solution(h, dx)
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [compute_fitness(individual) for individual in population]
        
        # Find best individual so far
        best_idx = np.argmin(fitnesses)
        current_best = population[best_idx]
        current_c5 = -fitnesses[best_idx]
        
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
            
        new_population = []
        
        for i in range(population_size):
            # Select three different individuals
            idxs = random.sample(range(population_size), 3)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_factor * (b - c)
            
            # Crossover
            trial = []
            for j in range(n_points):
                if random.random() < crossover_prob:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            
            # Project to feasible space
            trial = project_to_feasible(np.array(trial), n_points)
            new_population.append(trial)
            
        population = new_population
    
    # Refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final projection to make sure it's feasible
    best_h = project_to_feasible(best_h, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
