import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a more direct optimization approach:
    # 1. Start with initial solution
    # 2. Use a mix of random perturbations and local optimization
    
    h_current = h_start.copy()
    h_current = project_to_feasible(h_current, n_points)
    
    # Try multiple random starts
    for restart in range(5):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation of current solution
        h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        
        # Local refinement
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint(h):
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint}
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 30, 'ftol': 1e-8})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_new = evaluate_erdos_solution(h_refined, dx)
                if c5_new < best_c5:
                    best_h = h_refined
                    best_c5 = c5_new
                    h_current = h_refined.copy()
        except Exception:
            pass
    
    # Additional deterministic search with auto-generated sequences
    # Try a few more structured sequences to see if we can improve
    patterns = [
        np.concatenate([np.ones(n_points//2), np.zeros(n_points - n_points//2)]) * 2,
        np.concatenate([np.zeros(n_points//4), np.ones(n_points//2), np.zeros(n_points - n_points//4 - n_points//2)]) * 2,
        np.ones(n_points) * 0.5,
    ]
    
    for pattern in patterns:
        pattern = project_to_feasible(pattern, n_points)
        c5_pattern = evaluate_erdos_solution(pattern, dx)
        if c5_pattern < best_c5:
            best_h = pattern
            best_c5 = c5_pattern
    
    # Final refinement step
    def final_objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def final_constraint(h):
        return np.sum(h) - n_points / 2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': final_constraint}
    
    try:
        res = minimize(final_objective, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 50, 'ftol': 1e-9})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
