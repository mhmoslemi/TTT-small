import numpy as np
from scipy.optimize import minimize
import time
from math import floor
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def get_breakpoints_segments(h):
        # Determine breakpoints and segments for a piecewise constant function
        # 1. each segment has value 0 or 1
        # 2. ensure integral = 1
        # 3. determine segment lengths and values
        segments = []
        breakpoints = [0.0]
        current_seg_start = 0.0
        current_seg_val = h[0] > 0.5  # binary segmentation
        seg_length = 0.0
        for i in range(n_points):
            seg_length += dx
            if h[i] > 0.5:
                if not current_seg_val:
                    breakpoints.append(current_seg_start)
                    segments.append((current_seg_start, seg_length, current_seg_val))
                    current_seg_start = current_seg_start + seg_length
                    current_seg_val = True
                    seg_length = 0
            else:
                if current_seg_val:
                    breakpoints.append(current_seg_start)
                    segments.append((current_seg_start, seg_length, current_seg_val))
                    current_seg_start = current_seg_start + seg_length
                    current_seg_val = False
                    seg_length = 0
        if seg_length > 0.0:
            breakpoints.append(current_seg_start)
            segments.append((current_seg_start, seg_length, current_seg_val))
        breakpoints.append(2.0)
        return breakpoints, segments

    def compute_overlap_from_segments(bp, segs):
        # Compute overlap integral using the segments
        # take all shift values k from 0 to 2*dx
        overlaps = []
        total_weight = 0.0
        for i in range(len(segs)):
            total_weight += segs[i][2] * segs[i][1]
            
        for k in range(1, len(bp)):  # skip k=0
            overlap = 0.0
            for i in range(len(segs)):
                start1, len1, val1 = segs[i]
                for j in range(len(segs)):
                    start2, len2, val2 = segs[j] 
                    # compute overlap between segments i and j shifted by k*dx
                    # given segments [start1, start1+len1) and [start2-k*dx, start2+len2-k*dx)
                    # overlap is min(overlap length, len1, len2)
                    start_a = start1
                    end_a = start1 + len1
                    start_b = start2 - (k-1)*dx
                    end_b = start2 + len2 - (k-1)*dx
                    if end_a <= start_b or end_b <= start_a:
                        continue  # no overlap
                    overlap_start = max(start_a, start_b)
                    overlap_end = min(end_a, end_b)
                    overlap_len = overlap_end - overlap_start
                    if overlap_len > 0:
                        overlap += overlap_len * val1 * (1 - val2)  # integral of v1*(1-v2) over overlap
            overlaps.append(overlap)
        return max(overlaps) if overlaps else 0.0

    def form_h_from_breakpoints_segments(bp, segs):
        h_vec = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            for j in range(len(segs)):
                start, length, val = segs[j]
                if start <= x < start + length:
                    h_vec[i] = val
                    break
        return h_vec

    def constraint_sum(h):
        # Ensure that sum(h)*dx == 1
        return np.sum(h) * dx - 1.0

    def optimize_h_via_adaptive_local_search(h_init, max_evals=500):
        # Starts with the current h and perturbs it, evaluating the configuration
        current_h = h_init.copy()
        min_c5 = evaluate_erdos_solution(current_h, dx)
        best_h = current_h.copy()
        for _ in range(max_evals):
            if time.time() - start_time > budget_s * 0.95:
                break
            # Slightly perturb h and see if we improve
            candidate_h = current_h + np.random.normal(0, 0.01, n_points)
            candidate_h = np.clip(candidate_h, 0, 1)
            # Normalize to maintain sum(h)*dx = 1
            h_sum = np.sum(candidate_h)
            if h_sum > 0:
                candidate_h = candidate_h * (n_points / 2) / h_sum
            c5_val = evaluate_erdos_solution(candidate_h, dx)
            if c5_val < min_c5:
                min_c5 = c5_val
                best_h = candidate_h.copy()
                current_h = candidate_h.copy()
        return best_h, min_c5

    # Start with original construction
    current_h = h_start.copy()
    current_c5 = best_c5
    max_generations = 50
    pop_size = 200
    elite_count = 10
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Start a population of h vectors
        population = []
        for _ in range(pop_size):
            h_temp = np.random.rand(n_points) * 0.2 + 0.9  # slightly random
            h_sum = np.sum(h_temp)
            if h_sum > 0:
                h_temp = h_temp * (n_points / 2) / h_sum
            population.append(h_temp)
        
        # Evaluate the population
        scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            scores.append(c5)
        
        # Get best individuals
        sorted_indices = np.argsort(scores)
        top_individuals = [population[i] for i in sorted_indices[:elite_count]]
        best_in_gen = scores[sorted_indices[0]]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = top_individuals[0].copy()
        
        # Generate new population by selection + cross-over + mutation
        new_pop = []
        
        # Elitism
        new_pop.extend(top_individuals)
        
        # Uniform crossover + mutation for rest
        while len(new_pop) < pop_size:
            # Select two parents randomly
            p1_idx, p2_idx = random.sample(range(len(top_individuals)), 2)
            p1, p2 = top_individuals[p1_idx], top_individuals[p2_idx]
            
            # Uniform crossover
            child = np.zeros(n_points)
            mask = np.random.rand(n_points) > 0.5
            child[mask] = p1[mask]
            child[~mask] = p2[~mask]
            
            # Add small random mutation
            mut_mask = np.random.rand(n_points) < 0.05
            child[mut_mask] += np.random.normal(0, 0.02, np.sum(mut_mask))
            
            # Clip and normalize
            child = np.clip(child, 0, 1)
            h_sum = np.sum(child)
            if h_sum > 0:
                child = child * (n_points / 2) / h_sum  
                
            new_pop.append(child)
        
        population = new_pop

    # Final refinement
    for _ in range(10):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_refined, c5_refined = optimize_h_via_adaptive_local_search(best_h, max_evals=50)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    return (list(best_h), best_c5, n_points)
