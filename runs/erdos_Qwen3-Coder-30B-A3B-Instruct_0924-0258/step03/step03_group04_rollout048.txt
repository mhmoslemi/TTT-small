import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def find_best_binary_vector(h_shifted_k, n_points):
    """Find binary vector h_extreme that minimizes linear approximation"""
    # Find n/2 indices with largest (1 - h_shifted_k) values
    top_indices = np.argsort(1 - h_shifted_k)[::-1][:n_points//2]
    h_extreme = np.zeros(n_points)
    h_extreme[top_indices] = 1.0
    return h_extreme

def frank_wolfe_update(h, n_points, dx):
    """Perform one Frank-Wolfe step"""
    # Compute correlation to find worst overlap shift
    corr = correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n_points - 1)  # shift value
    
    # Compute shifted version of h for the worst case
    h_shifted_k = np.zeros(n_points)
    if k >= 0:
        h_shifted_k[k:] = h[:-k]
    else:
        h_shifted_k[:k] = h[-k:]
        
    # Find the extreme point that minimizes linear approximation
    h_extreme = find_best_binary_vector(h_shifted_k, n_points)
    
    # Frank-Wolfe update with step size 0.5
    h_new = 0.5 * h + 0.5 * h_extreme
    
    # Project back to feasible set
    h_new = project_to_feasible(h_new, n_points)
    
    return h_new
