import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with the initial solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Simulated Annealing approach with local refinement
    # Temperature schedule
    temp = 1.0
    temp_min = 1e-6
    alpha = 0.95
    max_iter = 1000
    
    # Number of iterations at each temperature
    iter_per_temp = 10
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # For each temperature, do multiple iterations
        for _ in range(iter_per_temp):
            # Create a new candidate by perturbing current solution
            h_new = current_h.copy()
            
            # Randomly modify some values
            num_changes = np.random.randint(1, max(1, n_points // 5))
            indices = np.random.choice(n_points, num_changes, replace=False)
            
            for idx in indices:
                # Random move in [0,1]
                delta = np.random.uniform(-0.1, 0.1)
                h_new[idx] += delta
                # Clip to [0,1]
                h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Rescale to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept or reject based on simulated annealing
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
            else:
                # Accept with probability based on temperature
                delta = c5_new - current_c5
                if np.random.uniform() < np.exp(-delta / temp):
                    current_h = h_new
                    current_c5 = c5_new
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temp = max(temp_min, temp * alpha)
        
        # Local refinement using convex optimization every few iterations
        if iteration % 5 == 0:
            # Convex feasibility problem to reduce C5 further
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add correlation constraints
            # For all shifts from 0 to n_points-1
            for k in range(n_points):
                corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_vars.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except:
                pass
    
    # Final local optimization using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
