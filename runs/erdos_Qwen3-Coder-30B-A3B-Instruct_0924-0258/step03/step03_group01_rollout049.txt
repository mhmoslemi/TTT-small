import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iterations = 30
    w_initial = 0.9
    w_final = 0.2
    c1, c2 = 1.5, 1.5
    inertia_weight = w_initial
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial solution
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Uniform random
            h = np.random.uniform(0, 1, n_points)
        
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Track personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_values = [evaluate_erdos_solution(swarm[i], dx) for i in range(swarm_size)]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    if global_best_value < best_c5:
        best_h = global_best.copy()
        best_c5 = global_best_value
    
    # PSO main loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        inertia_weight = w_initial - (w_initial - w_final) * (iteration / max_iterations)
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (inertia_weight * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
                    
                    if global_best_value < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_value
        
        # Every 5 iterations, refine top particles with SQP
        if iteration % 5 == 0 and iteration > 0:
            remaining_time = budget_s - (time() - start_time)
            if remaining_time < 10:
                break
                
            # Sort by current value and take top 3
            sorted_indices = np.argsort(personal_best_values)[:3]
            for idx in sorted_indices:
                h_start_refine = personal_best[idx].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_start_refine, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
