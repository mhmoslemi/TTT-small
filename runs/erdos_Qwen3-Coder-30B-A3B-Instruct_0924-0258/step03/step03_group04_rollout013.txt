import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_overlap_fft(h):
    """Compute overlap using FFT for efficiency"""
    n = len(h)
    h_padded = np.pad(h, (0, n-1), mode='constant')
    one_minus_h_padded = np.pad(1-h, (0, n-1), mode='constant')
    corr = np.fft.irfft(np.fft.rfft(h_padded) * np.fft.rfft(one_minus_h_padded))
    return np.max(corr[:n])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    velocity_max = 0.1
    
    # Initialize PSO
    positions = np.random.rand(swarm_size, n_points) * 0.5 + 0.25
    velocities = np.random.rand(swarm_size, n_points) * 0.02 - 0.01
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(swarm_size, np.inf)
    
    # Apply constraints
    for i in range(swarm_size):
        positions[i] = project_to_feasible(positions[i], n_points)
        score = evaluate_erdos_solution(positions[i], dx)
        personal_best_scores[i] = score
        if score < best_c5:
            best_c5 = score
            best_h = positions[i].copy()
    
    global_best_position = best_h.copy()
    global_best_score = best_c5
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (personal_best_positions[i] - positions[i]) +
                           social * r2 * (global_best_position - positions[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            positions[i] += velocities[i]
            
            # Apply constraints
            positions[i] = project_to_feasible(positions[i], n_points)
            
            # Evaluate
            score = evaluate_erdos_solution(positions[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best_positions[i] = positions[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_position = positions[i].copy()
                if score < best_c5:
                    best_c5 = score
                    best_h = positions[i].copy()
    
    # Final refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
