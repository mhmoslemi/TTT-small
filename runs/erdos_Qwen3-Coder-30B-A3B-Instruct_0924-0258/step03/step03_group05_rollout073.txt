import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with perturbations
    def compute_gradient(h_vec):
        n = len(h_vec)
        grad = np.zeros(n)
        # Compute gradient w.r.t. each component
        for i in range(n):
            # Approximate gradient via finite differences
            eps = 1e-6
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Renormalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            val_plus = evaluate_erdos_solution(h_plus, dx)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * eps)
        
        return grad
    
    # Optimization loop
    max_iter = 2000
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    # Initialize learning rate
    lr = 0.01
    
    current_h = best_h.copy()
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Update learning rate
        lr = max(1e-6, lr * 0.999)
        
        # Compute gradient
        try:
            grad = compute_gradient(current_h)
        except:
            # Degenerate case, revert to best
            current_h = best_h.copy()
            continue
        
        # Update step
        update = lr * grad
        new_h = current_h - update
        
        # Clip to bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Renormalize
        h_sum = np.sum(new_h)
        if h_sum > 0:
            new_h = new_h * (n_points / 2) / h_sum
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            no_improve_count = 0
            # Accelerate learning sometimes
            lr = min(0.1, lr * 1.1)
        else:
            no_improve_count += 1
            
        # Perturb occasionally
        if iter_count % 50 == 0:
            # Add noise
            noise = np.random.normal(0, 0.001, n_points)
            perturbed_h = new_h + noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            h_sum = np.sum(perturbed_h)
            if h_sum > 0:
                perturbed_h = perturbed_h * (n_points / 2) / h_sum
            perturbed_c5 = evaluate_erdos_solution(perturbed_h, dx)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
        
        # Check for termination
        if no_improve_count > max_no_improve:
            # Try a different approach
            current_h = best_h.copy()
            # Add random noise to escape local minima
            noise = np.random.laplace(0, 0.01, n_points)
            current_h += noise
            current_h = np.clip(current_h, 0, 1)
            h_sum = np.sum(current_h)
            if h_sum > 0:
                current_h = current_h * (n_points / 2) / h_sum
            no_improve_count = 0
        
        current_h = new_h
    
    # Final refinement
    def obj(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
