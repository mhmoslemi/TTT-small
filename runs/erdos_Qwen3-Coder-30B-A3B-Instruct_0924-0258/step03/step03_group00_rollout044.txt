import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based optimization approach
    d = 5  # Number of cosine terms
    max_restarts = 10
    max_iter = 50
    
    # Helper functions
    def fourier_to_h(a_coeffs):
        x = np.linspace(0, 2, n_points)
        g = np.zeros(n_points)
        for k in range(d + 1):
            g += a_coeffs[k] * np.cos(k * np.pi * x)
        h = 1 / (1 + np.exp(-g))
        return h
    
    def objective(a_coeffs):
        h = fourier_to_h(a_coeffs)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h_normalized = h * (n_points / 2) / h_sum
        else:
            h_normalized = h
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(a_coeffs):
        h = fourier_to_h(a_coeffs)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h_normalized = h * (n_points / 2) / h_sum
        else:
            h_normalized = h
        return np.sum(h_normalized) - (n_points / 2)
    
    # Optimization bounds
    bounds = [(-10, 10) for _ in range(d + 1)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Run multiple randomized restarts
    for restart in range(max_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        a_start = np.random.uniform(-2, 2, d + 1)
        
        try:
            result = minimize(
                objective,
                a_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-6}
            )
            
            if result.success:
                h_candidate = fourier_to_h(result.x)
                h_candidate = np.clip(h_candidate, 0, 1)
                h_sum = np.sum(h_candidate)
                if h_sum > 0:
                    h_candidate = h_candidate * (n_points / 2) / h_sum
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except Exception:
            continue
    
    # Final local refinement
    if time.time() - start_time < budget_s * 0.98:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run several local optimizations with different restarts
        for i in range(5):
            if time.time() - start_time > budget_s * 0.98:
                break
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            try:
                res = minimize(obj_func, h_perturb, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
