import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max(corr, beta):
    """Smooth approximation to max function"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))

def smooth_objective(h, beta, dx):
    """Compute smoothed objective"""
    corr = np.correlate(h, 1 - h, mode='full')
    return smooth_max(corr, beta) * dx

def gradient(h, beta, eps=1e-5):
    """Approximate gradient via finite differences"""
    n = len(h)
    grad = np.zeros(n)
    base_obj = smooth_objective(h, beta, 1.0)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_objective(h_plus, beta, 1.0) - base_obj) / eps
    return grad

def line_search(h, grad, beta, step_size, dx):
    """Perform line search to avoid bad steps"""
    while True:
        candidate = h - step_size * grad
        candidate = project_to_feasible(candidate, len(h))
        candidate_score = smooth_objective(candidate, beta, dx)
        if candidate_score < smooth_objective(h, beta, dx):
            return candidate, candidate_score
        step_size *= 0.5
        if step_size < 1e-8:
            break
    return h, smooth_objective(h, beta, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize parameters
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-5
    max_iters = 1000
    
    # Initialize

        n_points = len(h)
    else:

    dx = 2.0 / n_points
    
    # Project to feasible region
    h = project_to_feasible(h, n_points)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization loop
    for it in range(max_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = gradient(h, beta)
        
        # Line search for descent
        h_new, new_score = line_search(h, grad, beta, step_size, dx)
        
        # Update
        if new_score < best_c5:
            h = h_new.copy()
            best_c5 = new_score
            if step_size < 1e-3:
                step_size *= 1.5
        
        # Step size adaptation
        step_size *= 0.99
        
        # Update beta (smoothing)
        if it % 100 == 0 and it > 0:
            beta *= 1.2
            
        # Check for stagnation
        if it > 10 and abs(best_c5 - smooth_objective(h, beta, dx)) < tolerance:
            break
    
    # Final refinement
    if time() - start_time < budget_s * 0.9:
        # Try higher resolution
        n_points_high = min(n_points * 2, 100)
        if n_points_high > n_points:
            h_interp = np.interp(np.linspace(0, 2, n_points_high), 
                                np.linspace(0, 2, n_points), h)
            h = project_to_feasible(h_interp, n_points_high)
            n_points = n_points_high
            dx = 2.0 / n_points
            
            # Refine with smaller steps
            step_size = 0.001
            for _ in range(500):
                if time() - start_time > budget_s * 0.95:
                    break
                grad = gradient(h, beta)
                h_new, new_score = line_search(h, grad, beta, step_size, dx)
                if new_score < best_c5:
                    h = h_new.copy()
                    best_c5 = new_score
                else:
                    step_size *= 0.5
                    if step_size < 1e-6:
                        break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (list(best_h), best_c5, n_points)
