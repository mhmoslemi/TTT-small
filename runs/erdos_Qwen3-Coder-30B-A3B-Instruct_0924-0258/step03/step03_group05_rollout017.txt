import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try to improve with higher resolution
    if budget_s > 400:
        n_points = 70
        dx = 2.0 / n_points
        # Start with rescaled version of best solution
        h_coarse = best_h
        h_start = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, len(h_coarse)), 
                            h_coarse)
    
    # Local search parameters
    max_local_iters = 100
    perturb_mag = 0.05
    
    for iteration in range(max_local_iters):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Clone current best and perturb
        h_current = best_h.copy()
        # Small random perturbations
        perturb = np.random.normal(0, perturb_mag, n_points)
        h_perturbed = h_current + perturb
        
        # Project back to feasible region
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_h = np.sum(h_perturbed)
        if sum_h > 0:
            h_perturbed = h_perturbed * (n_points / 2) / sum_h
        
        # Evaluate the perturbed solution
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
            
        # Occasionally run a local optimization 
        if iteration % 10 == 0 and time() - start_time < budget_s * 0.9:
            # Solve using SLSQP to refine the current solution
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 10, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
    
    # Final validation with refined solution
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 20, 'ftol': 1e-8})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
