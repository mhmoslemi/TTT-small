import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from sklearn.cluster import KMeans

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Clustering approach with K-means
    # Create shift vectors for clustering (for all valid k values)
    k_values = list(range(0, n_points))  # Valid k values for correlations
    
    # Build overlap vectors for clustering
    overlap_vectors = []
    for k in k_values:
        # For shift k, compute overlap vector of size (n_points - k)
        if k == 0:
            # Self-overlap, constant 1
            ov = np.ones(n_points)
        else:
            ov = np.ones(n_points - k)
        overlap_vectors.append(ov)
    
    # Reshape and cluster
    overlap_matrix = np.array(overlap_vectors).T  # shape: (n_points - max_k, n_k_values) 
    # Trim to the size we can actually compute overlaps for
    effective_n_points = n_points
    max_k = n_points - 1
    
    # Use final effective overlap matrix for clustering
    if overlap_matrix.shape[0] > 0:
        # Normalize the overlap vectors for K-means
        overlap_matrix = overlap_matrix / (np.linalg.norm(overlap_matrix, axis=1, keepdims=True) + 1e-10)
        
        # Select number of clusters
        n_clusters = min(10, len(k_values) // 2)
        if n_clusters < 2:
            n_clusters = 2
        
        try:
            kmeans = KMeans(n_clusters=n_clusters, random_state=seed, n_init=10)
            cluster_labels = kmeans.fit_predict(overlap_matrix.T)
            cluster_centers = kmeans.cluster_centers_
        except Exception:
            # Fallback to simple uniform assignment
            cluster_labels = np.arange(len(k_values)) % n_clusters
            cluster_centers = np.zeros((n_clusters, overlap_matrix.shape[0]))
    else:
        cluster_labels = np.arange(len(k_values)) % 2  # Fallback for small cases
        n_clusters = 2
    
    # Create cluster mapping
    clusters = []
    for i in range(n_clusters):
        clusters.append(np.where(cluster_labels == i)[0])
    
    # Main optimization loop
    last_improvement = 0
    iteration = 0
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # For each cluster, optimize for its k shifts
        h_current = best_h.copy()
        
        for c_idx in range(n_clusters):
            if time() - start_time > budget_s * 0.95:
                break
                
            if len(clusters[c_idx]) == 0:
                continue
                
            # Get shifts belonging to this cluster
            k_indices = clusters[c_idx]
            
            # Define a simple function to optimize for this cluster
            def cluster_objective(h):
                # Compute overlap for this cluster's shifts
                total_overlap = 0.0
                # For each k in cluster
                for ki in k_indices:
                    # a) handle indices for shift ki
                    if ki == 0:
                        # Hardcoded overlap for k=0: just sum of h * (1 - h) 
                        term = np.dot(h, (1 - h))
                    else:
                        # For positive k: corresponding term
                        if ki < n_points:
                            # 0 to n_points - ki
                            term = np.sum(h[:-ki] * (1 - h[ki:]))
                        else:
                            term = 0.0
                    total_overlap += term
                return -total_overlap * dx  # we minimize negative to maximize
                
            # Approximate a gradient-based optimization with projection
            # Use fixed number of gradient steps
            h_local = h_current.copy()
            for step in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                # Approximate gradient using finite differences
                epsilon = 1e-6
                grad = np.zeros_like(h_local)
                obj_base = cluster_objective(h_local)
                for i in range(len(h_local)):
                    h_test = h_local.copy()
                    h_test[i] += epsilon
                    obj_plus = cluster_objective(h_test)
                    grad[i] = (obj_plus - obj_base) / epsilon
                # Gradient ascent step
                step_size = 0.01
                h_local = h_local + step_size * grad
                # Project to feasible
                h_local = project_to_feasible(h_local, n_points)
                
            h_current = h_local.copy()
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_current, dx)
        if new_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = new_c5
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early exit if stable
        if last_improvement > 50:
            break
    
    # Final refinement with scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 30, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
