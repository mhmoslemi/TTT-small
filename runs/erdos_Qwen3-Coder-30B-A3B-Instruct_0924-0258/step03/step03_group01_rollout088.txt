import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for simulated annealing
    temp = 0.1
    cooling_rate = 0.99
    max_iter = 30
    population_size = 20
    
    def perturb(h, temp_val):
        """Perturb solution with noise."""
        pert = np.random.normal(0, temp_val, len(h))
        h_new = h + pert
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Renormalize to sum to n/2
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    def refine_solution(h):
        """Refine a solution using local optimization."""
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                return res.x
        except:
            pass
        return h
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate population
        population = [best_h.copy()]
        for _ in range(population_size - 1):
            h_new = perturb(best_h, temp)
            population.append(h_new)
        
        # Evaluate and refine each candidate
        for h_candidate in population:
            h_refined = refine_solution(h_candidate)
            c5_candidate = evaluate_erdos_solution(h_refined, dx)
            if c5_candidate < best_c5:
                best_h = h_refined
                best_c5 = c5_candidate
        
        # Simulated annealing step
        h_candidate = perturb(best_h, temp)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept if better or with probability exp(-delta/T)
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / temp):
            best_h = h_candidate
            best_c5 = c5_candidate
        
        temp *= cooling_rate
    
    # Final refinement using scipy minimize with multiple restarts
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.98:
                break
                
            h_start_refine = best_h + np.random.normal(0, 0.02, n_points)
            h_start_refine = np.clip(h_start_refine, 0, 1)
            h_start_refine = h_start_refine * (n_points/2) / np.sum(h_start_refine)
            
            try:
                res = minimize(objective, h_start_refine, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
