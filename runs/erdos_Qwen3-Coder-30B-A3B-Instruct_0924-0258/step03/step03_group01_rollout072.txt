import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 20
    max_iter = 50
    w_max, w_min = 0.9, 0.2
    c1, c2 = 1.5, 1.5
    
    # Initialize swarm
    swarm = []
    for i in range(swarm_size):
        if i < swarm_size // 2:
            # Perturb initial construction
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random construction
            h = np.random.rand(n_points)
        
        # Clamp and normalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
    
    # Personal bests
    personal_best = [h.copy() for h in swarm]
    personal_best_scores = [evaluate_erdos_solution(h, dx) for h in swarm]
    
    # Global best
    best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]
    
    # Velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_max - (w_max - w_min) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - swarm[i]) +
                c2 * r2 * (global_best - swarm[i])
            )
            
            # Update position
            swarm[i] += velocities[i]
            
            # Repair feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = swarm[i].copy()
                
                # Update global best if needed
                if score < global_best_score:
                    global_best_score = score
                    global_best = swarm[i].copy()
        
        # Occasionally refine with SQP
        if iteration % 5 == 0 and iteration > 0:
            for k in range(3):  # Top 3 particles
                if time() - start_time > budget_s * 0.95:
                    break
                    
                h_refine = personal_best[k].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 30, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < global_best_score:
                            global_best_score = c5_refined
                            global_best = h_refined
                except:
                    pass
    
    return (list(global_best), global_best_score, n_points)
