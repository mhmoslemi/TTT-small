import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm
    generation = 0
    max_generations = 30
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection (top 50%)
        selected = population[:population_size//2]
        
        # Crossover
        new_offspring = []
        for _ in range(10):
            parent1, parent2 = random.sample(selected, 2)
            alpha = random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            new_offspring.append(child)
        
        # Mutation
        for i in range(len(new_offspring)):
            if random.random() < 0.3:
                mutation_strength = 0.03
                noise = np.random.normal(0, mutation_strength, n_points)
                new_offspring[i] += noise
                new_offspring[i] = np.clip(new_offspring[i], 0, 1)
                new_offspring[i] = new_offspring[i] * (n_points / 2) / np.sum(new_offspring[i])
        
        # Form new population
        population = selected + new_offspring
        
        # Local refinement with SQP every 2 generations
        if generation % 2 == 0:
            for i in range(min(3, len(population))):
                h_ref = population[i]
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, h_ref, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_improved = res.x
                        c5_improved = obj(h_improved)
                        if c5_improved < best_c5:
                            best_h = h_improved
                            best_c5 = c5_improved
                        # Update the population
                        population[i] = h_improved
                except:
                    pass
        
        generation += 1
    
    # Final refinement of best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
