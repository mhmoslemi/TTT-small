import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a sinusoidal initialization
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    max_iter = int(1e4)
    iter_per_temp = 100
    perturb_count = max(1, n_points // 10)  # about 10% of points
    
    # Track acceptance rate
    accept_count = 0
    total_count = 0
    
    h_current = h_start.copy()
    c5_current = best_c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate solution
        h_candidate = h_current.copy()
        
        # Select random indices to perturb
        perturb_indices = np.random.choice(n_points, size=perturb_count, replace=False)
        
        # Add small random changes
        for idx in perturb_indices:
            delta = np.random.normal(0, 0.1)
            h_candidate[idx] += delta
            
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Metropolis criterion
        delta = c5_candidate - c5_current
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h_current = h_candidate
            c5_current = c5_candidate
            accept_count += 1
            
            # Update best solution
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
        total_count += 1
        
        # Temperature update
        if (i + 1) % iter_per_temp == 0:
            if total_count > 0:
                accept_rate = accept_count / total_count
                if accept_rate < 0.05:
                    T *= 1.2  # Increase temperature
                elif accept_rate > 0.5:
                    T *= 0.8  # Decrease temperature
                accept_count = 0
                total_count = 0
            T = max(T_min, T * alpha)
    
    # Final refinement with local search
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        from scipy.optimize import minimize
        
        def refinement_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(refinement_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_refined = project_to_feasible(res.x, n_points)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
