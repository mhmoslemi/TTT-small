import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    population_size = 50
    mutation_factor = 0.5
    crossover_prob = 0.9
    max_iterations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Store best solution found
    best_individual = best_h.copy()
    best_score = best_c5
    
    # Differential Evolution main loop
    stagnation_count = 0
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate current population
        scores = []
        for h in population:
            score = evaluate_erdos_solution(h, dx)
            scores.append(score)
        
        # Track best solution
        min_score_idx = np.argmin(scores)
        if scores[min_score_idx] < best_score:
            best_individual = population[min_score_idx].copy()
            best_score = scores[min_score_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Stop if stagnated
        if stagnation_count > 100:
            break
            
        # Create next generation
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = random.sample(candidates, 3)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = random.randint(0, n_points - 1)
            for j in range(n_points):
                if random.random() < crossover_prob or j == j_rand:
                    trial[j] = mutant[j]
            
            # Project to feasible region
            trial = project_to_feasible(trial, n_points)
            
            # Selection
            trial_score = evaluate_erdos_solution(trial, dx)
            if trial_score < scores[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        # Start from the best solution found by DE
        h_refined = best_individual.copy()
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_score:
                    best_individual = h_final
                    best_score = c5_final
        except Exception:
            pass
    
    return (list(best_individual), best_score, n_points)
