import numpy as np
from scipy.optimize import minimize
from time import time
from sklearn.cluster import KMeans
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Clustering approach
    cluster_count = 10
    max_iter = 200
    
    # Precompute overlap vectors for all k
    shift_vectors = []
    for k in range(n_points):
        v = np.zeros(n_points)
        if k < n_points:
            v[k:] = 1
        shift_vectors.append(v)
    
    # Initialize clusters with simple approach
    h_current = best_h.copy()
    h_current = project_to_feasible(h_current, n_points)
    
    # Main loop
    no_improvement_count = 0
    for iteration in range(max_iter):
        current_time = time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Cluster shifts based on overlap patterns
        # Instead of full correlation, we create a distance matrix of shift patterns
        overlap_matrix = []
        for k in range(n_points):
            vec = np.zeros(n_points)
            if k < n_points:
                vec[k:] = 1
            overlap_matrix.append(vec)
        
        overlap_matrix = np.array(overlap_matrix)
        try:
            kmeans = KMeans(n_clusters=cluster_count, random_state=seed+iteration, n_init=10)
            cluster_labels = kmeans.fit_predict(overlap_matrix)
        except:
            cluster_labels = np.arange(n_points) % cluster_count
        
        # Optimize each cluster
        for cluster_id in range(cluster_count):
            if current_time - start_time > budget_s * 0.95:
                break
                
            cluster_indices = np.where(cluster_labels == cluster_id)[0]
            if len(cluster_indices) == 0:
                continue
                
            # Approximate optimal solution for cluster
            # This is a heuristic restart: try to solve a bounded version
            h_trial = h_current.copy()
            # Update h for this cluster loop with finite sum constraint
            # Simple box-constrained optimization within cluster indices
            trial_h = h_trial.copy()
            
            def obj_func(h_trial):
                # Approximate the objective for the cluster
                total_overlap = 0
                for k in cluster_indices:
                    if k == 0:
                        # Starts at 0
                        total_overlap += np.sum(trial_h * (1 - trial_h)) * dx
                    else:
                        # Convolution-like constraint
                        overlap_part = np.sum(trial_h[:-k] * (1 - trial_h[k:])) * dx
                        total_overlap = max(total_overlap, overlap_part)
                return total_overlap
            
            # Simple rollout with 2D constraints (more elegant would be constrained optimizer)
            # For each actual shift k in cluster, we'll mark what it depends on,
            # and apply small adjustments to local area
            
            # This is a greedy approach that loosely uses a variation of alternating optimization with ADMM style updates
            # We make an intermediate update using clustered local heuristics
            
            # Introduce some random moves in cluster regions
            updated_indices = cluster_indices
            if len(updated_indices) > 0:
                # Scramble minimal for exploration
                for _ in range(2):
                    indices_to_update = np.random.choice(updated_indices, min(3, len(updated_indices)), replace=False)

                        # Small perturbation
                        new_h = h_trial[i] + np.random.normal(0, 0.1)
                        h_trial[i] = np.clip(new_h, 0, 1)
                
                # Project before test
                h_trial = project_to_feasible(h_trial, n_points)
                test_c5 = evaluate_erdos_solution(h_trial, dx)

                    best_h = h_trial.copy()
                    no_improvement_count = 0

                    no_improvement_count += 1
        
        # Sometimes additional random moves or restarts
        if iteration % 5 == 0 and iteration > 0:
            # New random initialization within bounds
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = project_to_feasible(h_rand, n_points)
            test_c5 = evaluate_erdos_solution(h_rand, dx)
            if test_c5 < best_c5:
                best_c5 = test_c5
                best_h = h_rand.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
    
    # Final SLSQP refinement if time allows
    if time() - start_time < budget_s - 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
