import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a gradient-based approach with initial feasibility check
    def objective(h):
        h_proj = project_to_feasible(h, n_points)
        return evaluate_erdos_solution(h_proj, dx)
    
    # Constraint function for scipy.optimize
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Gradient descent loop
    max_iter = 500
    learning_rate = 1e-3
    patience = 50
    best_iter = 0
    
    h_current = h_start.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient of overlap w.r.t. h using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[j] += eps
            h_minus[j] -= eps
            
            f_plus = objective(h_plus)
            f_minus = objective(h_minus)
            grad[j] = (f_plus - f_minus) / (2 * eps)
        
        # Project gradient to feasible set
        grad = project_to_feasible(h_current - learning_rate * grad, n_points) - h_current
        
        # Line search to find optimal step size
        step_size = 1.0
        h_new = h_current - step_size * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Ensure sufficient improvement
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            h_current = h_new
            best_c5 = c5_new
            best_iter = i
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.9
        
        # Early stopping if no improvement in patience iterations
        if i - best_iter > patience:
            break
    
    # Final polishing using L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            res = minimize(objective, h_current, method='L-BFGS-B',
                          bounds=[(0, 1) for _ in range(n_points)],
                          options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
