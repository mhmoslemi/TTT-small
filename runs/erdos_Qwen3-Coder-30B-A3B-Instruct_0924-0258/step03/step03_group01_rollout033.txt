import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate initial population of diverse candidates
    population_size = 20
    population = [h_start.copy()]
    for i in range(1, population_size):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Learning rate and parameters
    learning_rate = 0.01
    discount_factor = 0.99
    entropy_coeff = 0.1
    max_iterations = 60
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Evaluate current population
        population_c5 = [evaluate_erdos_solution(h, dx) for h in population]
        best_idx = np.argmin(population_c5)
        if population_c5[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = population_c5[best_idx]
        
        # Convex refinement for elite candidates
        elite_count = min(5, len(population))
        elite_indices = np.argsort(population_c5)[:elite_count]
        
        for idx in elite_indices:
            h = population[idx]
            
            # Create convex optimization problem
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points / 2.0,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Add overlap constraints
            # This is a simplified version - full constraints would be computationally intensive
            # For brevity, we'll focus on a subset of shift indices
            shift_indices = list(range(1, min(10, n_points)))
            for k in shift_indices:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_var[k-n_points+1:], 1 - h_var[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-8)
            
            # Minimize C5 via convex optimization
            try:
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    refined_h = h_var.value
                    refined_c5 = evaluate_erdos_solution(refined_h, dx)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
            except:
                pass
        
        # Generate new candidates through perturbations
        new_population = []
        
        for h in population:
            # Apply structured perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_population.append(h_new)
        
        # Combine old and new populations
        population = population + new_population
        population_c5 = population_c5 + [evaluate_erdos_solution(h, dx) for h in new_population]
        
        # Retain best individuals
        sorted_indices = np.argsort(population_c5)
        population = [population[i] for i in sorted_indices[:population_size]]
        population_c5 = [population_c5[i] for i in sorted_indices[:population_size]]
    
    # Final refinement using scipy minimize with restarts
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]

            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
