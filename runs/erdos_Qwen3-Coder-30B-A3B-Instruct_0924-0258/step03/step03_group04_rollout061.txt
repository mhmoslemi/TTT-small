import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try optimization with different methods:
    # Method 1: Gradient-based optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj_grad(h):
            h_feas = project_to_feasible(h, n_points)
            c5 = evaluate_erdos_solution(h_feas, dx)
            return c5, np.zeros_like(h)  # Simplified gradient
        
        # Try with different starting points
        for i in range(3):
            if time() - start_time > budget_s * 0.9:
                break
            h_init = np.random.rand(n_points) * 0.5 + 0.25
            h_init = project_to_feasible(h_init, n_points)
            
            res = minimize(lambda h: evaluate_erdos_solution(project_to_feasible(h, n_points), dx),
                          h_init, method='L-BFGS-B',
                          bounds=[(0,1) for _ in range(n_points)],
                          options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
    
    # Method 2: Local search with perturbations
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Simply try a few random perturbations around the current best
        for i in range(100):
            if time() - start_time > budget_s * 0.9:
                break
            # Generate small random perturbation
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = project_to_feasible(h_pert, n_points)
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
    
    # Final method: Convex relaxation approach
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        try:
            # Use a simplified convex approximation
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Add a few key constraints
            for k in range(1, n_points//3 + 1): 
                valid_len = n_points - k
                if valid_len > 0:
                    corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[k:k+valid_len]))
                    constraints.append(corr_sum <= 0.38 * dx * valid_len)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-5, abstol=1e-5)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] and h_var.value is not None:
                h_feas = project_to_feasible(h_var.value, n_points)
                c5_feas = evaluate_erdos_solution(h_feas, dx)
                if c5_feas < best_c5:
                    best_h = h_feas.copy()
                    best_c5 = c5_feas
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
