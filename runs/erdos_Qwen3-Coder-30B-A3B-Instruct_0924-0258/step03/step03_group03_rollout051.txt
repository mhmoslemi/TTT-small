import numpy as np
from scipy.signal import correlate
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 35  # Start with the provided initial solution size
    dx = 2.0 / n_points
    max_iter = 1000
    lr = 0.01
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        # Create initial symmetric solution
        h_current = np.ones(n_points) * 0.5
        # Slightly perturb to avoid trivial case
        h_current = h_current + np.random.normal(0, 0.01, n_points)
    
    h_current = project_to_feasible(h_current, n_points)
    best_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    best_n_points = n_points
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient approximation via finite differences
        grad = np.zeros_like(h_current)
        eps = 1e-5
        base_score = evaluate_erdos_solution(h_current, dx)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            score_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (score_plus - base_score) / eps
            
        # Update with gradient descent step
        h_new = h_current - lr * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        new_score = evaluate_erdos_solution(h_new, dx)
        
        if new_score < best_c5:
            best_h = h_new.copy()
            best_c5 = new_score
            best_n_points = n_points
            
        h_current = h_new
        lr *= 0.999  # Adaptive learning rate decay
        
        # Occasionally restart with better solution
        if iteration % 200 == 0 and iteration > 0:
            h_current = project_to_feasible(best_h + np.random.normal(0, 0.01, n_points), n_points)
            
    return (list(best_h), best_c5, best_n_points)
