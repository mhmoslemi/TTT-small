import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def normalize_h(h, target_sum):
    """Normalize h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.2, noise_std=0.1):
    """Apply mutation to the step function h"""
    h_mut = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, noise_std)
    # Clip to [0,1]
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover_h(h1, h2):
    """Perform crossover between two step functions"""
    if len(h1) != len(h2):
        raise ValueError("H vectors must have same length")
    point = random.randint(1, len(h1)-1)
    h_new = np.concatenate([h1[:point], h2[point:]])
    return h_new

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection"""
    tournament_indices = random.sample(range(len(population)), k)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    population_size = 50
    generations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = normalize_h(h, n_points / 2)
        population.append(h)
    
    # GA loop
    while time() - start_time < budget_s * 0.95:
        generations += 1
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check termination condition
        if no_improvement_count > max_no_improvement:
            break
            
        # Create new population
        new_population = [population[min_fitness_idx].copy()]  # Keep best
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child = crossover_h(parent1, parent2)
            
            # Mutation
            child = mutate_h(child)
            
            # Normalize to target sum
            child = normalize_h(child, n_points / 2)
            
            new_population.append(child)
        
        population = new_population
    
    # Final local refinement with SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = normalize_h(h_final, n_points / 2)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
