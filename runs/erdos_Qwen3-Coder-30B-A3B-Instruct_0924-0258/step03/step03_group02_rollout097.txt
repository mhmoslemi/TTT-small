import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    max_iterations = 1000
    tolerance = 1e-6
    improvement = float('inf')
    
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.9:
        h_old = h_current.copy()
        h_current = project_to_feasible(h_current, n_points)
        
        # Random order for coordinate descent
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        for i in indices:
            if (time() - start_time) > budget_s * 0.9:
                break
                
            # Current h value
            h_old_i = h_current[i]
            
            # Evaluate the objective function values for a range of possible values for h[i]
            best_h_i = h_old_i
            best_c5_i = best_c5
            
            # Check a dense range around current value
            # Use 100 steps in range [0, 1]
            test_values = np.linspace(0, 1, 100)
            for test_h_i in test_values:
                h_test = h_current.copy()
                h_test[i] = test_h_i
                # Project to feasibility and evaluate
                h_test = project_to_feasible(h_test, n_points)
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5_i:
                    best_c5_i = c5_test
                    best_h_i = test_h_i
            
            # Update h_current with the best value found
            h_current[i] = best_h_i
            
        # Evaluate the entire solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Check for improvement
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
        
        # Check for convergence
        improvement = np.abs(c5_current - best_c5)
        if improvement < tolerance:
            break
            
        iteration += 1
    
    # Final refinement with SLSQP if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_final = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_final, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
