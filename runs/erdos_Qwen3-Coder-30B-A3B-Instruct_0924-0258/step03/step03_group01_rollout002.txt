import numpy as np
from scipy.optimize import minimize
from time import time
import itertools

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Initialize with structured pattern
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = int(5 * dx * n_points)
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size*dx/2)/dx))
            end = min(n_points, int((c + cluster_size*dx/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize
        h_start *= (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based search with Nelder-Mead and local refinement
    # Initialize population
    population_size = 15
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    max_iterations = 20  # Limit iterations based on budget
    no_improvement_count = 0
    max_no_improvement = 5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Order population by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                # Reset population if no improvement for several iterations
                population = [h_start.copy()]
                for _ in range(population_size - 1):
                    h_pert = h_start + np.random.normal(0, 0.02, n_points)
                    h_pert = np.clip(h_pert, 0, 1)
                    h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                    population.append(h_pert)
                no_improvement_count = 0
        
        # Create new population via mutation
        new_population = [population[0]]  # Keep best
        for i in range(1, population_size):
            # Mutation: add random noise to a random individual
            parent_idx = np.random.randint(0, len(population) // 2)
            h_new = population[parent_idx] + np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_population.append(h_new)
        
        population = new_population
        
        # Local refinement on top candidates
        refined_count = 0
        for i in range(min(3, len(population))):
            if time() - start_time > budget_s * 0.95:
                break
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            try:
                res = minimize(objective, population[i], method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                    # Replace in population if better
                    if c5_refined < fitness[i]:
                        population[i] = h_refined
                        fitness[i] = c5_refined
                    refined_count += 1
            except:
                pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
