import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic pattern with some randomness
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    lr = 1e-3
    decay_factor = 0.999
    p_perturb = 0.1
    perturb_magnitude = 0.05
    max_iter = int(1e5)
    patience = 100
    
    # Precompute shifts for faster correlation calculation
    shifts = np.arange(n_points)
    
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Every 100 iterations, reduce learning rate
        if iter_count % 100 == 0:
            lr *= decay_factor
        
        # Compute all correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Find shift with maximum overlap
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute subgradient
        grad = np.zeros_like(best_h)
        
        if max_shift > 0:
            # h[i] * (1 - h[i + max_shift]) term contributes to gradient
            for i in range(n_points - max_shift):
                grad[i] += (1 - best_h[i + max_shift])
        elif max_shift < 0:
            # h[i] * (1 - h[i - max_shift]) term contributes to gradient
            for i in range(-max_shift, n_points):
                grad[i] += (1 - best_h[i + max_shift])
        else:
            # Skip for shift 0
            pass
        
        # Update h
        h_new = best_h - lr * grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Occasionally perturb
        if np.random.random() < p_perturb:
            perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h_new = project_to_feasible(h_new + perturb, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
        elif iter_count - last_improvement > patience:
            # Early stopping if no improvement
            break
    
    # Final refinement using L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 200})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
