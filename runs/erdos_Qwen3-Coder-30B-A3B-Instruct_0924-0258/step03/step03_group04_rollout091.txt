import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create periodic-like initial solution
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing Parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    acceptance_threshold = 0.05
    max_iter = 10000
    
    # Track acceptance rate
    accepted = 0
    total = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution
        h_candidate = best_h.copy()
        
        # Perturb about 5-10% of the points
        num_perturb = max(1, n_points // 20)
        indices = random.sample(range(n_points), num_perturb)
        
        for idx in indices:
            # Perturb slightly
            perturbation = random.uniform(-0.1, 0.1)
            h_candidate[idx] += perturbation
            
        # Project to feasible region
        h_candidate = project_to_feasible(h_candidate, n_points)
        
        # Evaluate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        delta = c5_candidate - best_c5
        total += 1
        
        if delta < 0 or random.random() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            accepted += 1
            
        # Update temperature based on acceptance rate
        if total % 100 == 0:
            if total > 0:
                acceptance_rate = accepted / total
                if acceptance_rate < acceptance_threshold:
                    T *= 0.9
                elif acceptance_rate > 0.2:
                    T *= 1.1
                accepted = 0
                total = 0
                
            # Ensure T doesn't go too low
            if T < min_T:
                T = min_T
                
        # Cool down
        if iteration % 100 == 0:
            T *= cooling_rate
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        def objective(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
            
        # Use a few iterations of optimization with different restarts
        best_refined = best_h.copy()
        best_c5_refined = best_c5
        
        for _ in range(10):
            # Random restart
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = project_to_feasible(h_restart, n_points)
            
            res = minimize(objective, h_restart, method='L-BFGS-B', 
                          bounds=[(0,1) for _ in range(n_points)], 
                          options={'maxiter': 20})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5_refined:
                    best_c5_refined = c5_final
                    best_refined = h_final.copy()
                    
        best_h = best_refined
        best_c5 = best_c5_refined
    
    return (list(best_h), best_c5, n_points)
