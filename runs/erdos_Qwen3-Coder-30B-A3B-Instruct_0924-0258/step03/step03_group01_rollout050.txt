import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create structured initialization
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Nelder-Mead optimization for global search
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Generate initial population
    def generate_population(n_pop, h_init, dx):
        pop = [h_init.copy()]
        for _ in range(n_pop - 1):
            h_new = h_init + np.random.normal(0, 0.05, n_points)
            h_new = np.clip(h_new, 0, 1)
            # Normalize
            scale = (n_points/2) / np.sum(h_new)
            h_new *= scale
            pop.append(h_new)
        return pop
    
    # Nelder-Mead step
    def nelder_mead_step(pop, obj_func):
        # Sort population by objective
        pop.sort(key=obj_func)
        # Simplex
        simplex = pop[:n_points+1]
        # Compute centroid (excluding worst)
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        r = 1 + 1/n_points
        reflect = centroid + r * (centroid - simplex[-1])
        reflect = np.clip(reflect, 0, 1)
        scale = (n_points/2) / np.sum(reflect)
        reflect *= scale
        
        f_r = obj_func(reflect)
        f_best = obj_func(simplex[0])
        f_worst = obj_func(simplex[-1])
        
        if f_best <= f_r < obj_func(simplex[-2]):
            # Expansion
            e = 2
            expand = centroid + e * (reflect - centroid)
            expand = np.clip(expand, 0, 1)
            scale = (n_points/2) / np.sum(expand)
            expand *= scale
            if obj_func(expand) < f_r:
                simplex[-1] = expand
            else:
                simplex[-1] = reflect
        elif f_r < f_best:
            # Outside contraction
            oc = 0.5
            contract = centroid + oc * (reflect - centroid)
            contract = np.clip(contract, 0, 1)
            scale = (n_points/2) / np.sum(contract)
            contract *= scale
            if obj_func(contract) < f_r:
                simplex[-1] = contract
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
        else:
            # Inside contraction
            ic = 0.5
            contract = centroid + ic * (simplex[-1] - centroid)
            contract = np.clip(contract, 0, 1)
            scale = (n_points/2) / np.sum(contract)
            contract *= scale
            if obj_func(contract) < f_worst:
                simplex[-1] = contract
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
        
        return simplex
    
    # Iterative Nelder-Mead
    n_pop = 15
    pop = generate_population(n_pop, h_start, dx)
    
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        try:
            pop = nelder_mead_step(pop, objective)
            best_candidate = pop[0]
            val = objective(best_candidate)
            if val < best_c5:
                best_h = best_candidate
                best_c5 = val
        except Exception:
            pass
            
        if iteration > 50:  # Prevent infinite loop
            break
    
    # Local refinement with gradient-based methods
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Refine using SLSQP
        num_refinements = 5
        for i in range(min(num_refinements, 10)):
            h_refine = best_h.copy()
            # Add noise to avoid local minima
            noise_factor = 0.1
            h_refine += np.random.normal(0, noise_factor, n_points)
            h_refine = np.clip(h_refine, 0, 1)
            scale = (n_points/2) / np.sum(h_refine)
            h_refine *= scale
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_refine, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    c5_refined = objective(res.x)
                    if c5_refined < best_c5:
                        best_h = res.x
                        best_c5 = c5_refined
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
