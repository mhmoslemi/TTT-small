import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic pattern with more oscillations to improve chances
        h_start = 0.5 * (1 + np.sin(2 * np.pi * np.arange(n_points) / n_points))
    
    dx = 2.0 / n_points
    
    # Initialize
    h_current = project_to_feasible(h_start, n_points)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Optimizer hyperparameters
    learning_rate = 1e-3
    decay_factor = 0.9995
    perturb_prob = 0.1
    perturb_magnitude = 0.05
    max_iter = 100000
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Compute correlations
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_k = np.argmax(corr)
        
        # Compute subgradient (simplified approach)
        h_grad = np.zeros_like(h_current)
        
        # Start from the max overlap offset
        valid_len = n_points - abs(max_k - n_points)
        if valid_len > 0:
            if max_k < n_points:
                # Positive shift case
                h_grad[:valid_len] += (1 - h_current[max_k:max_k+valid_len])
            else:
                # Negative shift case  
                h_grad[max_k - n_points:] += (1 - h_current[:valid_len])
        
        # Apply gradient step
        h_new = h_current - learning_rate * h_grad
        h_new = project_to_feasible(h_new, n_points)
        
        # Random perturbation with some probability
        if np.random.random() < perturb_prob:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_new = h_new + perturb
            h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept new solution if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update current solution
        h_current = h_new.copy()
        learning_rate *= decay_factor
        
        # Early stopping if no improvement
        if no_improvement_count > max_no_improvement:
            break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
