import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

        # Try different constructions

        h3 = np.zeros(n)
        h3[n//4:3*n//4] = 1.0  # broad step function
        h4 = np.ones(n) * 0.5
        h4[:n//3] = 1.0
        h4[2*n//3:] = 1.0  # two peaks
        
        # Try different random seeds for initialization
        np.random.seed(seed + 10)
        h5 = np.random.rand(n) * 0.5 + 0.25
        h5 = project_to_feasible(h5, n)
        
        return [h1, h2, h3, h4, h5]
    
    # Multiple restarts with different strategies
    initial_h_list = generate_initial_h(n_points)

    best_h = h_start.copy()

            h_proj = project_to_feasible(h, n_points)

    h_current = best_h.copy()

    max_cd_iter = 500

    return (list(best_h), best_c5, n_points)
