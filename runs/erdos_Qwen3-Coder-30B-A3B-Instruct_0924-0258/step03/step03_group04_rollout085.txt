import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a hybrid approach: focus more on efficient global search
    # and basic gradient refinement
    
    # PSO-like optimization with refinement
    max_iter = 500
    swarm_size = 60
    inertia = 0.729
    cognitive = 1.49445
    social = 1.49445
    
    # Initialize swarm
    swarm = []
    best_particle = None
    best_particle_c5 = float('inf')
    
    for _ in range(swarm_size):
        h_init = project_to_feasible(np.random.rand(n_points) * 0.5 + 0.25, n_points)
        swarm.append(h_init)
        
    # PSO iterations
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        for i, h in enumerate(swarm):
            # Evaluate current particle
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_particle_c5:
                best_particle_c5 = c5
                best_particle = h.copy()
            
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            if no_improvement_count > max_no_improvement:
                break

        if no_improvement_count > max_no_improvement:
            break
            
        # Update particles with PSO update rules
        for i, h in enumerate(swarm):
            # Compute velocity update
            if best_particle is not None:
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                # Update velocity (simplified PSO)
                vel = inertia * (h - h) + \
                      cognitive * r1 * (best_particle - h) + \
                      social * r2 * (best_particle - h)
                h_new = h + vel
                h_new = project_to_feasible(h_new, n_points)
                swarm[i] = h_new
    
    # Refinement with local search
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        # Start from best solution
        if best_particle is not None:
            res = minimize(final_obj, best_particle, method='L-BFGS-B',
                           bounds=[(0,1) for _ in range(n_points)],
                           options={'maxiter': 100})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
