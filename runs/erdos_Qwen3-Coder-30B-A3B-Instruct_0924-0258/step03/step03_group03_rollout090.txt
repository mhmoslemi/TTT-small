import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def gradient_descent_step(h, dx, lr=0.01):
    """Compute gradient and perform one step"""
    eps = 1e-6
    base_score = evaluate_erdos_solution(h, dx)
    grad = np.zeros_like(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - base_score) / eps

    h_new = h - lr * grad
    h_new = project_to_feasible(h_new, len(h))
    return h_new

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Try a few different small resolutions to get better starting points
    for n_try in [20, 25, 30, 35]:
        if time() - start_time > budget_s * 0.8:
            break
        dx_try = 2.0 / n_try
        if n_try != n_points:
            h_interp = np.interp(np.linspace(0, 2, n_try), 
                                np.linspace(0, 2, len(h_start)), 
                                h_start)
            h_start = project_to_feasible(h_interp, n_try)
            n_points = n_try
            
        # Local optimization with gradient descent
        h_current = h_start.copy()
        for _ in range(50):
            if time() - start_time > budget_s * 0.95:
                break
            h_current = gradient_descent_step(h_current, dx_try, lr=0.05)
            current_c5 = evaluate_erdos_solution(h_current, dx_try)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
                best_n_points = n_points

    # Final refinement with more gradient descent
    if time() - start_time < budget_s * 0.95:
