import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        n_points = len(h_current)
    else:
        n_points = 50
        h_current = np.zeros(n_points)
        h_current[25] = 1.0  # Simple initial construction
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_size = 20
    max_no_improvement = 50
    no_improvement_count = 0
    
    # Main loop
    while time.time() - start_time < budget_s * 0.95:
        # Generate neighbors by perturbing 20% of indices
        neighbor_indices = np.random.choice(n_points, int(0.2 * n_points), replace=False)
        neighbors = []
        
        for _ in range(5):  # Generate 5 neighbors per iteration
            h_neighbor = h_current.copy()
            for i in neighbor_indices:
                # Add small random perturbation
                delta = np.random.normal(0, 0.05)
                h_neighbor[i] += delta
                # Clamp to [0,1]
                h_neighbor[i] = np.clip(h_neighbor[i], 0, 1)
            
            # Project to satisfy constraint sum(h) * dx = 1
            h_sum = np.sum(h_neighbor)
            if h_sum > 0:
                h_neighbor = h_neighbor * (n_points / 2) / h_sum
            
            neighbors.append(h_neighbor.copy())
        
        # Evaluate neighbors
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for h_neighbor in neighbors:
            # Check tabu list
            h_tuple = tuple(h_neighbor.round(6))
            if h_tuple in tabu_list:
                continue
                
            c5_neighbor = evaluate_erdos_solution(h_neighbor, dx)
            if c5_neighbor < best_neighbor_c5:
                best_neighbor = h_neighbor
                best_neighbor_c5 = c5_neighbor
        
        # If no valid neighbor found, escape with probability
        if best_neighbor is None:
            h_tuple = tuple(h_current.round(6))
            if h_tuple not in tabu_list:
                tabu_list.append(h_tuple)
            if len(tabu_list) > tabu_size:
                tabu_list.pop(0)
            no_improvement_count += 1
            continue
        
        # Accept best neighbor
        h_current = best_neighbor
        h_tuple = tuple(h_current.round(6))
        
        if h_tuple in tabu_list:
            tabu_list.remove(h_tuple)
        tabu_list.append(h_tuple)
        
        if len(tabu_list) > tabu_size:
            tabu_list.pop(0)
        
        # Local refinement with SLSQP
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        for _ in range(10):
            if time.time() - start_time > budget_s * 0.95:
                break
            try:
                res = minimize(obj_func, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10, 'ftol': 1e-6})
                if res.success:
                    h_current = res.x.copy()
                    h_sum = np.sum(h_current)
                    if h_sum > 0:
                        h_current = h_current * (n_points / 2) / h_sum
                    c5_new = evaluate_erdos_solution(h_current, dx)
                    if c5_new < best_c5:
                        best_h = h_current.copy()
                        best_c5 = c5_new
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                else:
                    no_improvement_count += 1
            except:
                no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                break
        
        # Update best solution
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
    
    return (list(best_h), best_c5, n_points)
