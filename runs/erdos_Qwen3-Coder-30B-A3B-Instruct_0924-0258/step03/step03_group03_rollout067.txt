import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_gradients(h, dx):
    """Compute gradients of the overlap function"""
    # This is a simplified gradient computation using finite differences
    eps = 1e-5
    base_score = evaluate_erdos_solution(h, dx)
    
    grad = np.zeros_like(h)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        score_plus = evaluate_erdos_solution(h_plus, dx)
        grad[i] = (score_plus - base_score) / eps
    
    return grad

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h_start, dx)
    best_h = h_start.copy()
    best_c5 = current_c5
    best_n_points = n_points

    # Gradient descent with adaptive step size
    max_iter = 500
    step_size = 0.01
    tolerance = 1e-6
    decay_factor = 0.99

        if time() - start_time > budget_s * 0.95:
            break
            
        grad = compute_gradients(best_h, dx)

        new_h = best_h - step_size * grad

        if new_c5 < best_c5:

            best_c5 = new_c5

        # Early stopping
        if abs(best_c5 - current_c5) < tolerance:
            break

    # Works on a refined version
    best_h_refined = best_h.copy()
