import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 35
    dx = 2.0 / n_points
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create a reasonable initial guess with a few peaks
        h_start = np.ones(n_points) * 0.5
        # Create a simple construction that's already somewhat optimal
        h_start[5:10] = 1.0
        h_start[20:25] = 1.0
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initialize population
    population_size = 500
    population = []
    
    # Generate initial population
    for i in range(population_size):
        if time.time() - start_time > budget_s * 0.8:
            break
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum if h_sum > 0 else h
        population.append(h.copy())
    
    # Sort initial population
    population.sort(key=objective)
    best_in_gen = population[0].copy()
    best_c5_in_gen = objective(best_in_gen)
    if best_c5_in_gen < best_c5:
        best_h = best_in_gen.copy()
        best_c5 = best_c5_in_gen
    
    # Evolution parameters
    generations = 500
    elite_count = 50
    crossover_rate = 0.8
    mutation_rate = 0.05
    
    # Main evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Selection: keep top 20%
        elite = population[:elite_count]
        
        # Create new population
        new_population = deepcopy(elite)
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if time.time() - start_time > budget_s * 0.9:
                break
                
            # Tournament selection
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover (uniform)
            if np.random.random() < crossover_rate:
                child = np.copy(parent1)
                mask = np.random.random(n_points) < 0.5
                child[mask] = parent2[mask]
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Mutate some points
                mut_count = max(1, int(n_points * 0.05))
                indices = np.random.choice(n_points, mut_count, replace=False)
                for idx in indices:
                    child[idx] = np.clip(child[idx] + np.random.normal(0, 0.05), 0, 1)
            
            # Normalize
            child_sum = np.sum(child)
            if child_sum > 0:
                child = child * (n_points / 2) / child_sum
            new_population.append(child)
        
        # Sort and continue
        population = new_population
        population.sort(key=objective)
        
        # Update best in generation
        best_in_gen = population[0].copy()
        best_c5_in_gen = objective(best_in_gen)
        if best_c5_in_gen < best_c5:
            best_h = best_in_gen.copy()
            best_c5 = best_c5_in_gen
    
    # Local refinement on best solutions
    num_refinements = 10
    for i in range(num_refinements):
        if time.time() - start_time > budget_s * 0.99:
            break
        h_perturb = best_h + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_sum = np.sum(h_perturb)
        if h_sum > 0:
            h_perturb = h_perturb * (n_points / 2) / h_sum
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                c5_test = objective(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
