import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    max_iter = 1000
    patience = 100
    patience_counter = 0
    best_c5_so_far = best_c5
    
    # Initialize learning rate
    learning_rate = 0.01
    
    def compute_gradient(h, dx):
        """Compute gradient of the overlap integral w.r.t. h"""
        n = len(h)
        grad = np.zeros(n)
        
        # Compute overlap integrals for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        
        # For each shift k, compute gradient contribution
        for k in range(n):
            if k == 0:
                continue
            # Contributions to gradient
            valid_len = n - k
            if valid_len <= 0:
                break
            
            # Gradient of overlap w.r.t. h[i] is (1 - h[i+k]) if i+k < n
            for i in range(valid_len):
                if i+k < n:
                    grad[i] += (1 - h[i+k]) * dx
        
        # Add gradient from the terms h[i+k] * 1
        for k in range(n):
            if k == 0:
                continue
            valid_len = n - k
            if valid_len <= 0:
                break
            for i in range(valid_len):
                if i+k < n:
                    grad[i+k] += -h[i] * dx
        
        return grad
    
    # Project h to [0,1] range
    def project_h(h):
        return np.clip(h, 0, 1)
    
    # Normalize h to satisfy constraint
    def normalize_h(h):
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Optimization loop
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(best_h, dx)
        
        # Update h with gradient decent
        updated_h = best_h - learning_rate * grad
        
        # Project updated_h back to [0,1]
        updated_h = project_h(updated_h)
        
        # Normalize to maintain measure constraint
        updated_h = normalize_h(updated_h)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(updated_h, dx)
        
        # Update if improvement
        if new_c5 < best_c5:
            best_h = updated_h.copy()
            best_c5 = new_c5
            patience_counter = 0
            best_c5_so_far = min(best_c5_so_far, best_c5)
        else:
            patience_counter += 1
            
        # Reduce learning rate over time
        learning_rate *= 0.995
        
        # Reset if stuck
        if patience_counter > patience:
            # Add small random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            best_h = project_h(best_h + noise)
            best_h = normalize_h(best_h)
            patience_counter = 0
        
        # Occasionally apply larger perturbation to escape local minima
        if i % 50 == 0 and i > 0:
            noise = np.random.normal(0, 0.02, n_points)
            best_h = project_h(best_h + noise)
            best_h = normalize_h(best_h)
    
    # Final refinement using scipy minimize
    def obj(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            refined_h = np.clip(res.x, 0, 1)
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
