import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Structured initialization with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to sum to n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization with Nelder-Mead
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_feasible(h):
        # Ensure bounds and sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Generate initial population
    population_size = 15
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h_start + perturbation
        h_perturbed = project_feasible(h_perturbed)
        population.append(h_perturbed)
    
    # Nelder-Mead iterations
    max_iterations = 20
    iteration = 0
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.9:
        # Evaluate all candidates
        fitness = [objective(h) for h in population]
        # Sort by fitness
        indices = np.argsort(fitness)
        population = [population[i] for i in indices]
        fitness = [fitness[i] for i in indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new simplex vertices
        # Best (0) is kept, others are constructed from it
        new_population = [population[0].copy()]
        for i in range(1, population_size):
            # Reflect or generate new points
            # Simplex method with random vectors
            if i == 1:
                # Perturb the best with a random vector
                new_h = population[0] + 0.5 * (population[0] - population[1]) + np.random.normal(0, 0.02, n_points)
            else:
                # Average of previous best points
                ref_point = np.mean([population[0], population[i-1]], axis=0)
                new_h = ref_point + np.random.normal(0, 0.02, n_points) * (1 + i/10)
            
            new_h = project_feasible(new_h)
            new_population.append(new_h)
        
        population = new_population
        iteration += 1
    
    # Gradient descent refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Refine top candidates
        for i in range(min(3, len(population))):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = population[i].copy()
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
