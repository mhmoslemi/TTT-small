import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple random restarts to avoid local optima
    n_restarts = max(1, int(600 / (budget_s * 0.1)))
    n_restarts = min(n_restarts, 10)
    
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        h_init = np.random.rand(n_points) * 0.5 + 0.25
        # Normalize to integral = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Constraints
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Minimize with SLSQP
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = np.clip(res.x, 0, 1)
                # Renormalize
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Perform final refinement with local optimization 
    if time() - start_time < budget_s * 0.9:
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                final_c5 = evaluate_erdos_solution(h_final, dx)
                if final_c5 < best_c5:
                    best_h = h_final
                    best_c5 = final_c5
        except:
            pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
