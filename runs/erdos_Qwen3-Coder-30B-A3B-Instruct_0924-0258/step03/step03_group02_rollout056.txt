import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Differential Evolution parameters
    pop_size = 50
    F = 0.5
    CR = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Store best overall
    current_best = best_h.copy()
    current_best_c5 = best_c5
    
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # DE mutation and crossover
        new_population = []
        for i in range(pop_size):
            # Select three distinct individuals
            idxs = [idx for idx in range(pop_size) if idx != i]
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                    
            # Ensure feasibility
            trial = project_to_feasible(trial, n_points)
            new_population.append(trial)
            
        # Selection: keep better of old and new
        for i in range(pop_size):
            new_c5 = evaluate_erdos_solution(new_population[i], dx)
            if new_c5 < fitness[i]:
                population[i] = new_population[i]
                
        # Update best
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < current_best_c5:
                current_best = h.copy()
                current_best_c5 = c5
                iter_no_improve = 0
        iter_no_improve += 1
        if iter_no_improve > 100:
            break
    
    # Final SLSQP refinement on best candidate
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        h_refined = current_best.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < current_best_c5:
                    current_best = h_final
                    current_best_c5 = c5_final
        except Exception:
            pass
    
    # Ensure best solution is feasible
    final_h = project_to_feasible(current_best, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
