import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple gradient descent approach with restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_h(h):
        # Ensure h is in [0,1] and sum is correct
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Use a variation of simulated annealing approach
    # Start with initial solution
    current_h = h_start.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    # Simulated annealing parameters
    T = 1.0
    T_min = 1e-6
    alpha = 0.95
    max_iter = 1000
    iter_count = 0
    
    while T > T_min and (time() - start_time) < budget_s * 0.95:
        # Take a random step
        new_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        step_size = 0.05 * T
        new_h[idx] += np.random.uniform(-step_size, step_size)
        new_h = project_h(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on simulated annealing criteria
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down
        T *= alpha
        iter_count += 1
        
        # Occasionally do convex refinement
        if iter_count % 50 == 0:
            # Project to proper domain
            h_proj = project_h(best_h)
            
            # Use convex optimization for local refinement
            h_cvx = cp.Variable(n_points)
            constraints = [
                cp.sum(h_cvx) == n_points / 2.0,
                h_cvx >= 0,
                h_cvx <= 1
            ]
            
            # Add one overlap constraint (not all for performance)
            shift_idx = np.random.randint(0, n_points)
            if shift_idx < n_points:
                corr_sum = cp.sum(cp.multiply(h_cvx[:n_points-shift_idx], 1 - h_cvx[shift_idx:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_cvx[shift_idx-n_points+1:], 1 - h_cvx[:n_points-(shift_idx-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_cvx.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except:
                pass
                
        if (time() - start_time) > budget_s * 0.95:
            break
    
    # Final refinement with scipy optimizer
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple restarts
        for restart in range(5):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = project_h(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
