import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def generate_initial_h(n_points):
    """Generate initial h values from global initial_h_values if available"""
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Default: uniform distribution
        return np.ones(n_points) * 0.5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Generate initial construction
    h_start = generate_initial_h(35)
    n_points = len(h_start)
    dx = 2.0 / n_points
    
    # Current best
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Run local optimization with multiple restarts
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multiple restarts
    num_restarts = 20
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb current best
        h_perturb = best_h + np.random.normal(0, 0.05, n_points)
        
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    # Refine with additional gradient descent
    if time.time() - start_time < budget_s * 0.98:
        h_refined = best_h.copy()
        # Ensure proper normalization
        h_refined = np.clip(h_refined, 0, 1)
        h_sum = np.sum(h_refined)
        if h_sum > 0:
            h_refined = h_refined * (n_points / 2) / h_sum
        
        # Local optimization with bounds
        def obj_func_local(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        try:
            res = minimize(obj_func_local, h_refined, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-7})
            if res.success:
                final_c5 = obj_func_local(res.x)
                if final_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = final_c5
        except:
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (n_points / 2) / final_sum
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
