import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 20
    population = []
    
    # Create initial population around the starting point
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    
    # Optimization iterations
    max_iterations = 30
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Refine each individual with SQP
        refined_population = []
        for h in population:
            # SQP refinement
            def objective(h_ref):
                return evaluate_erdos_solution(h_ref, dx)
            
            def constraint_sum(h_ref):
                return np.sum(h_ref) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                else:
                    h_refined = h
            except:
                h_refined = h
            
            # Ensure feasibility
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            refined_population.append(h_refined)
        
        # Move to refined population
        population = refined_population
        
        # Simulated Annealing step
        for i in range(len(population)):
            h_current = population[i]
            h_candidate = h_current.copy()
            
            # Perturb
            perturbation = np.random.normal(0, 0.01, n_points)
            h_candidate += perturbation
            
            # Clamp and renormalize
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate
            c5_current = evaluate_erdos_solution(h_current, dx)
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            # Accept based on temperature
            if c5_candidate < c5_current or np.random.random() < np.exp((c5_current - c5_candidate) / temp):
                population[i] = h_candidate
        
        # Update temperature
        temp = max(min_temp, temp * cooling_rate)
    
    return (list(best_h), best_c5, n_points)
