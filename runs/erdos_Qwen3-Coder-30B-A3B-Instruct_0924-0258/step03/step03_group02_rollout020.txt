import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate Descent with random order
    max_iterations = 500
    tolerance = 1e-6
    max_iterations_per_search = 100

    # Precompute overlap contributions for efficiency
    def compute_overlap_contribution(h, i, k):
        # This is a simplified version since we'll compute full correlation anyway
        return h[i] * (1 - h[i + k]) if i + k < len(h) else 0

    last_improvement = 0
    iteration_count = 0

    def evaluate_c5_for_h(h):
        return evaluate_erdos_solution(h, dx)
    
    while time() - start_time < budget_s * 0.9 and iteration_count < max_iterations:
        # Shuffle indices for random update order
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        for idx in indices:
            if time() - start_time > budget_s * 0.9:
                break
                
            current_h = best_h.copy()
            current_c5 = evaluate_c5_for_h(current_h)
            
            # Grid search for optimal value
            best_h_i = current_h[idx]
            best_val = current_c5
            
            # Trying candidate values
            candidates = np.linspace(0, 1, max_iterations_per_search)
            for candidate in candidates:
                # Try replacing h[idx] with candidate
                h_trial = current_h.copy()
                h_trial[idx] = candidate
                h_trial = project_to_feasible(h_trial, n_points)
                trial_c5 = evaluate_c5_for_h(h_trial)
                
                if trial_c5 < best_val:
                    best_val = trial_c5
                    best_h_i = candidate
            
            # Update if we found a better value
            if best_val < current_c5 - 1e-10:
                current_h[idx] = best_h_i
                current_h = project_to_feasible(current_h, n_points)
                new_c5 = evaluate_c5_for_h(current_h)
                if new_c5 < best_c5 - 1e-10:
                    best_h = current_h
                    best_c5 = new_c5
                    improved = True
                    last_improvement = iteration_count
        
        iteration_count += 1
        if not improved and iteration_count - last_improvement > 50:
            break

    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': refine_constraint}]
            
            res = minimize(refine_objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass

    return (list(best_h), best_c5, n_points)
