import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Early stopping if time is nearly up
    if time() - start_time > budget_s * 0.95:
        return (list(best_h), best_c5, n_points)
    
    # Try to initialize from simple construction (simpler approach)
    def simple_construction(n):
        h = np.zeros(n)
        # Place a tall, narrow spike in the middle
        mid = n // 2
        h[mid] = 1.0
        # Spread it out slightly to use the whole interval
        for i in range(1, min(5, n//2)):
            h[mid-i] = 0.1
            h[mid+i] = 0.1
        # Normalize to sum to n/2
        h = h * (n/2) / np.sum(h)
        return h
    
    # Get a simple construction to start with
    h_simple = simple_construction(n_points)
    h_simple = project_to_feasible(h_simple, n_points)
    c5_simple = evaluate_erdos_solution(h_simple, dx)
    
    if c5_simple < best_c5:
        best_h = h_simple.copy()
        best_c5 = c5_simple
    
    # Make a small number of refined steps
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Try refining the better solution
    for _ in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Refine with SLSQP
        try:
            res = minimize(objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
        # Also try a simple gradient descent variant from current best
        h_new = best_h.copy()
        # Stochastic gradient-like approach
        for i in range(n_points):
            h_test = h_new.copy()
            h_test[i] += 0.01
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                h_new = h_test
                best_c5 = c5_test
                best_h = h_new.copy()
                
        # Directly check a few variations from current solution
        for i in range(n_points):
            h_test = best_h.copy()
            h_test[i] = 1.0 - h_test[i]  # flip one value
            h_test = project_to_feasible(h_test, n_points)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test
                best_c5 = c5_test
        
        if time() - start_time > budget_s:
            break
    
    # Final validation
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
