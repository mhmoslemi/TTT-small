import numpy as np
import time
from scipy.optimize import minimize
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to avoid cycling
    tabu_list = deque(maxlen=10)
    
    # Simulated annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    iter_per_temp = 20
    
    def perturb_h(h, idx, delta=0.1):
        h_new = h.copy()
        h_new[idx] = np.clip(h_new[idx] + np.random.uniform(-delta, delta), 0, 1)
        return h_new
    
    def get_neighbors(h, idx, delta=0.1):
        neighbors = []
        for d in [-delta, 0, delta]:
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + d, 0, 1)
            neighbors.append(h_new)
        return neighbors
    
    def rescale_h(h):
        h_sum = np.sum(h)
        if h_sum > 0:
            return h * (n_points / 2) / h_sum
        return h
    
    current_h = best_h.copy()
    current_c5 = best_c5
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Check if we've exhausted the budget
        if iteration % 100 == 0 and time.time() - start_time > budget_s * 0.9:
            break
            
        # Perturb a random index
        idx = np.random.randint(0, n_points)
        h_new = perturb_h(current_h, idx)
        h_new = rescale_h(h_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Metropolis acceptance criterion
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            # Accept with probability based on temperature
            delta = c5_new - current_c5
            if np.random.rand() < np.exp(-delta / temp):
                current_h = h_new
                current_c5 = c5_new
                
        # Update temperature
        if iteration % iter_per_temp == 0:
            temp = max(temp * cooling_rate, min_temp)
    
    # Local optimization
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different starting points for local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h_perturb = best_h + np.random.normal(0, 0.05, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
