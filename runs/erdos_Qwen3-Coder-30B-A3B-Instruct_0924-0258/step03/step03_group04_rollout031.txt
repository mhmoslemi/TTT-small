import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with random perturbations
    max_iter = 100000
    learning_rate = 1e-3
    p_perturb = 0.1
    perturb_magnitude = 0.05
    patience = 100
    patience_counter = 0
    
    h_current = h_start.copy()
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)  # Convert to shift index
        
        # Compute subgradient for the maximum overlap
        subgradient = np.zeros(n_points)
        if max_shift > 0:
            # Shift positive: h[i] contributes to overlap with h[i+shift]
            subgradient[:n_points - max_shift] = 1 - h_current[max_shift:]
        elif max_shift < 0:
            # Shift negative: h[i] contributes to overlap with h[i-shift]
            subgradient[-max_shift:] = 1 - h_current[:n_points + max_shift]
        else:
            # Zero shift - this shouldn't happen given our arrangement
            pass
            
        # Update with negative gradient
        h_new = h_current - learning_rate * subgradient
        
        # Project to feasibility
        h_new = project_to_feasible(h_new, n_points)
        
        # Random perturbation with low probability
        if np.random.random() < p_perturb:
            perturbation = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h_new = h_new + perturbation
            h_new = project_to_feasible(h_new, n_points)
            
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        h_current = h_new.copy()
        
        # Early stopping if no improvement for a while
        if patience_counter >= patience:
            break
            
        # Adaptive learning rate decay
        if iteration % 1000 == 0 and iteration > 0:
            learning_rate *= 0.95
    
    # Final optimization using scipy minimize
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def objective(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Optimize starting from our best solution
        res = minimize(objective, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
