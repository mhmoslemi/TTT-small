import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a two-stage approach: genetic algorithm + local refinement
    population_size = 500
    generations = 50
    elite_size = 10
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population with random variations of the best solution
    def create_individual():
        # Start with perturbed version of best solution
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def tournament_selection(population, fitnesses, k=3):
        indices = np.random.choice(len(population), size=k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Simple uniform crossover
        mask = np.random.rand(n_points) < 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        
        # Normalize
        s1 = np.sum(child1)
        s2 = np.sum(child2)
        if s1 > 0:
            child1 = child1 * (n_points / 2) / s1
        if s2 > 0:
            child2 = child2 * (n_points / 2) / s2
        return child1, child2
    
    def mutate(individual):
        mutation_mask = np.random.rand(n_points) < mutation_rate
        individual = individual.copy()
        if np.any(mutation_mask):
            individual[mutation_mask] += np.random.normal(0, 0.05, np.sum(mutation_mask))
            individual = np.clip(individual, 0, 1)
            s = np.sum(individual)
            if s > 0:
                individual = individual * (n_points / 2) / s
        return individual
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    generation = 0
    
    while generation < generations and (time.time() - start_time < budget_s * 0.8):
        # Evaluate fitness
        fitnesses = [fitness(ind) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        current_best = population[0]
        current_c5 = fitnesses[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Create next generation
        new_population = population[:elite_size]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        generation += 1
        if time.time() - start_time > budget_s * 0.95:
            break
    
    # Local refinement on the best solution
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run local optimization with several restarts
    num_restarts = 50
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.98:
            break
        # Perturb the current best solution
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return (list(best_h), best_c5, n_points)
