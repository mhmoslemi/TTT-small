import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def smooth_max_approx(h, beta):
    """Smooth approximation of max function for convolution values"""
    corr = np.correlate(h, 1 - h, mode='full')
    return (1 / beta) * np.log(np.sum(np.exp(beta * corr)))

def compute_gradient(h, beta, eps=1e-5):
    """Compute numerical gradient using finite differences"""
    grad = np.zeros_like(h)
    base_val = smooth_max_approx(h, beta)
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h))
        grad[i] = (smooth_max_approx(h_plus, beta) - base_val) / eps
    return grad

    best_h = h.copy()
    
    # Parameters for optimization
    beta = 100.0
    step_size = 0.01
    tolerance = 1e-6
    max_iter = 1000
    decay_rate = 0.95
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute gradient
        grad = compute_gradient(h, beta)

        h_new = h - step_size * grad
        h_new = project_to_feasible(h_new, n_points)

        if new_c5 < best_c5:

        else:

        # Update beta for smoother approximation
        if iter % 50 == 0 and iter > 0:

        if iter % 200 == 0 and iter > 0 and n_points < 100:
