import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_segments(h, dx):
    """Convert step function to segments"""
    segments = []
    for i in range(len(h)):
        if i == 0:
            segments.append((0, h[i]))
        else:
            segments.append((i*dx, h[i]))
    return segments

def fit_segments_to_h(segments, n_points, dx):
    """Convert segments back to h vector"""
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        # Find segment containing x
        for j in range(len(segments)-1):
            if segments[j][0] <= x < segments[j+1][0]:
                h[i] = segments[j][1]
                break
        else:
            h[i] = segments[-1][1]
    return h

def compute_segment_widths(segments):
    """Get segment widths"""
    widths = []
    for i in range(len(segments)-1):
        widths.append(segments[i+1][0] - segments[i][0])
    return np.array(widths)

def compute_integral(segments):
    """Compute integral of segments"""
    integral = 0
    for i in range(len(segments)-1):
        integral += segments[i][1] * (segments[i+1][0] - segments[i][0])
    return integral

def project_segments(segments, n_points, dx):
    """Project segments to satisfy integral constraint and [0,1] bounds"""
    # Clip segment heights
    for i in range(len(segments)):
        segments[i] = (segments[i][0], max(0, min(1, segments[i][1])))
    
    # Adjust heights to satisfy integral constraint
    integral = compute_integral(segments)
    if integral > 0:
        scale = (n_points / 2) / integral
        for i in range(len(segments)):
            segments[i] = (segments[i][0], segments[i][1] * scale)
    
    return segments

def boundary_gradient_descent(segments, n_points, dx, max_iter=50):
    """Optimize segment boundaries"""
    beta = 100
    step_size = 0.01
    epsilon = 1e-5
    
    def compute_smooth_max(segments):
        h = fit_segments_to_h(segments, n_points, dx)
        corr = np.correlate(h, 1 - h, mode='full')
        smooth_max = np.sum(np.exp(beta * corr)) / beta
        return smooth_max, corr
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        smooth_max, corr = compute_smooth_max(segments)
        
        # Gradient w.r.t. x coordinates
        grad = np.zeros(len(segments)-1)  # Only boundaries between segments
        for i in range(len(segments)-1):
            h = fit_segments_to_h(segments, n_points, dx)
            x_plus = segments.copy()
            x_plus[i+1] = (segments[i+1][0] + epsilon, segments[i+1][1])
            # Ensure boundaries are sorted
            x_plus = sorted(x_plus, key=lambda x: x[0])
            h_plus = fit_segments_to_h(x_plus, n_points, dx)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.sum(np.exp(beta * corr_plus)) - np.sum(np.exp(beta * corr))) / (beta * epsilon)
        
        # Update boundaries
        for i in range(len(segments)-1):
            segments[i+1] = (segments[i+1][0] - step_size * grad[i], segments[i+1][1])
        
        # Ensure sorted and within bounds
        segments = sorted(segments, key=lambda x: x[0])
        segments[0] = (0, segments[0][1])
        segments[-1] = (2, segments[-1][1])
        
        segments = project_segments(segments, n_points, dx)
    
    return segments
