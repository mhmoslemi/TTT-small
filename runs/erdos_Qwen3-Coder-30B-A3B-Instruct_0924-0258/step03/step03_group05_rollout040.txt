import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tune parameters
    max_iterations = 20
    perturbation_magnitude = 0.05
    
    # Scale resolution if time permits
    if budget_s > 400:
        n_points = 70
        dx = 2.0 / n_points
        # Initialize with scaled version of the best solution from previous run
        h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)

    # Dual optimization approach
    def solve_feasibility(h_guess, c_val):
        # Define CVX variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            if k == 0:
                continue  # skip shift 0
            # Consider valid indices for this shift
            valid_len = n_points - k
            if valid_len <= 0:
                break
            # Constraint: sum(h[i] * (1 - h[i+k])) * dx <= c_val
            overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_val
            constraints.append(overlap_constraint)
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, max_iters=1000, feastol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        return None

    # Binary search approach with dual feasibility checks
    c_low = 0.37
    c_high = 0.38085
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        c_mid = (c_low + c_high) / 2.0
        h_feasible = solve_feasibility(best_h, c_mid)
        
        if h_feasible is not None:
            c_low = c_mid
            best_h = h_feasible.copy()
            # Refine with local search
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(objective, h_feasible, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
        else:
            c_high = c_mid

    # Post-processing: local refinement with perturbations
    max_perturbations = 100
    pert_count = 0
    
    while time() - start_time < budget_s * 0.95 and pert_count < max_perturbations:
        h_perturbed = best_h + np.random.normal(0, perturbation_magnitude, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Check if this solution is better
        c5_pert = evaluate_erdos_solution(h_perturbed, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_perturbed
        
        # Soft check to avoid waiting if too long
        pert_count += 1
        if time() - start_time > budget_s * 0.9:
            break

    # Final optimization step
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
