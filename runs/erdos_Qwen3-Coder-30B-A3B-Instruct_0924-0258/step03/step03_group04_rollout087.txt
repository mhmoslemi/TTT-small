import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simple gradient-based approach
    max_iter = 1000
    learning_rate = 0.01
    patience = 50
    
    h_current = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Gradient approximation
        epsilon = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, n_points)
            
            h_minus = h_current.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, n_points)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update step
        h_new = h_current - learning_rate * grad
        h_new = project_to_feasible(h_new, n_points)
        
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < current_c5:
            h_current = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_new
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.995
    
    # Final optimization with scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def objective(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Random perturbations for better exploration
        best_h_scipy = best_h.copy()
        best_c5_scipy = best_c5
        
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            h_perturbed = best_h_scipy + np.random.normal(0, 0.01, n_points)
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            
            res = minimize(objective, h_perturbed, method='L-BFGS-B',
                          bounds=[(0,1) for _ in range(n_points)],
                          options={'maxiter': 50})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5_scipy:
                    best_h_scipy = h_refined.copy()
                    best_c5_scipy = c5_refined
        
        if best_c5_scipy < best_c5:
            best_h = best_h_scipy
            best_c5 = best_c5_scipy
    
    return (list(best_h), best_c5, n_points)
