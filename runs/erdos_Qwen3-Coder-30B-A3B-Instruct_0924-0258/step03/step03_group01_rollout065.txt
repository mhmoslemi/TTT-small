import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_population(n_points, initial_h, pop_size=20):
    """Initialize a diverse population around the initial solution"""
    population = []
    for i in range(pop_size):
        if i == 0:
            # Use initial solution
            h = initial_h.copy()
        else:
            # Add random perturbation
            h = initial_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def crossover(parent1, parent2):
    """Perform uniform crossover"""
    mask = np.random.random(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(individual, mutation_rate=0.3, std=0.03):
    """Apply Gaussian mutation"""
    if random.random() < mutation_rate:
        noise = np.random.normal(0, std, len(individual))
        individual = individual + noise
        individual = np.clip(individual, 0, 1)
        # Renormalize
        individual = individual * (len(individual) / 2) / np.sum(individual)
    return individual

def repair(individual):
    """Ensure constraints are satisfied"""
    individual = np.clip(individual, 0, 1)
    individual = individual * (len(individual) / 2) / np.sum(individual)
    return individual

def genetic_algorithm_with_local_refinement(n_points, initial_h, budget_s):
    """Main GA with local refinement routine"""
    np.random.seed(42)
    start_time = time()
    
    # Parameters
    pop_size = 20
    generations = 30
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1  # Refine every generation
    
    # Initialize
    population = initialize_population(n_points, initial_h, pop_size)
    best_h = None
    best_c5 = float('inf')
    dx = 2.0 / n_points
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Genetic algorithm loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size // 2):
            # Tournament selection
            tournament_size = 3
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Create new population
        new_population = selected[:]
        while len(new_population) < pop_size:
            if random.random() < crossover_rate and len(selected) >= 2:
                p1, p2 = random.sample(selected, 2)
                child = crossover(p1, p2)
                child = repair(child)
            else:
                child = random.choice(selected).copy()
            child = mutate(child, mutation_rate)
            child = repair(child)
            new_population.append(child)
        
        population = new_population[:pop_size]
        
        # Evaluate all
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Local SQP refinement on top individuals (every generation)
        if gen % sqp_interval == 0:
            for idx in range(min(3, len(population))):
                h = population[idx]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_c5 = c5_refined
                            best_h = h_refined.copy()
                except:
                    pass
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Run genetic algorithm with local refinement
    h_opt, c5_bound = genetic_algorithm_with_local_refinement(n_points, h_start, budget_s)
    
    return (list(h_opt), c5_bound, n_points)
