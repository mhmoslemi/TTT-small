import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based simulated annealing with SQP refinement
    population_size = 20
    iterations = 30
    temp_init = 0.1
    temp_decay = 0.99
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    T = temp_init
    
    for iter_num in range(iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Find best in population
        best_in_pop_idx = np.argmin(fitness)
        if fitness[best_in_pop_idx] < best_c5:
            best_h = population[best_in_pop_idx].copy()
            best_c5 = fitness[best_in_pop_idx]
        
        # Refine each solution with SQP
        new_population = []
        for h in population:
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constr_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constr_sum}
            
            try:
                res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                               constraints=cons_sum, options={'maxiter': 20})
                if res.success:
                    h_refined = res.x
                else:
                    h_refined = h
            except:
                h_refined = h
            
            # Ensure feasibility
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            new_population.append(h_refined)
        
        population = new_population
        
        # Simulated Annealing update
        for i in range(population_size):
            h_old = population[i].copy()
            # Perturb
            h_new = h_old + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_old = evaluate_erdos_solution(h_old, dx)
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < c5_old or np.random.random() < np.exp((c5_old - c5_new) / T):
                population[i] = h_new
        
        T *= temp_decay
    
    # Final refinement on best solution
    def obj(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constr_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constr_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', bounds=bounds, 
                       constraints=cons_sum, options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            best_h = res.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
