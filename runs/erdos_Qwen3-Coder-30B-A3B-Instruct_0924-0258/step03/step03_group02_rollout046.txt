import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate Descent approach
    max_iterations = 500
    tolerance = 1e-6
    improvement = tolerance + 1
    iter_count = 0
    
    # Initialize list of indices to iterate through
    indices = list(range(n_points))
    
    while improvement > tolerance and (time() - start_time) < budget_s * 0.9:
        iter_count += 1
        prev_c5 = best_c5
        np.random.shuffle(indices)
        
        for i in indices:
            # Compute current overlaps
            h_temp = best_h.copy()
            
            # Optimize h[i] to minimize maximum overlap
            # For a single parameter, we'll do a grid search
            best_h_i = h_temp[i]
            best_val = best_c5
            
            # Grid search for optimal value of h[i]
            grid_points = 50
            h_range = 1.0
            for j in range(grid_points):
                new_h_i = j * h_range / (grid_points - 1)
                h_temp[i] = new_h_i
                val = evaluate_erdos_solution(h_temp, dx)
                if val < best_val:
                    best_val = val
                    best_h_i = new_h_i
                    
            # Update the value
            h_temp[i] = best_h_i
            h_temp = project_to_feasible(h_temp, n_points)
            updated_val = evaluate_erdos_solution(h_temp, dx)
            
            # Accept the change if it improves the objective
            if updated_val < best_c5:
                best_h = h_temp
                best_c5 = updated_val
                
            # Check time out
            if (time() - start_time) > budget_s * 0.9:
                break
        
        improvement = abs(prev_c5 - best_c5)
        
        # Small chance per iteration to explore other indexes
        if iter_count % 50 == 0 and (time() - start_time) < budget_s * 0.9:
            # Refine with SLSQP
            def refine_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def refine_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = {'type': 'eq', 'fun': refine_constraint}
            
            try:
                res = minimize(refine_objective, best_h, method='SLSQP',
                              bounds=bounds, constraints=constraints,
                              options={'maxiter': 20, 'ftol': 1e-8})
                if res.success:
                    h_refined = project_to_feasible(res.x, n_points)
                    new_c5 = evaluate_erdos_solution(h_refined, dx)
                    if new_c5 < best_c5:
                        best_h = h_refined
                        best_c5 = new_c5
            except:
                pass
    
    # Final refinement with SLSQP if still time
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
