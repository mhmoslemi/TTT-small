import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 50
    num_generations = 1000
    mutation_rate = 0.2
    elite_size = 5
    tournament_size = 5
    
    # Initialize population with diversity
    def create_individual():
        ind = np.random.rand(n_points)
        # Normalize to satisfy constraint sum(h) * dx = 1
        ind = ind * (n_points / 2) / np.sum(ind)
        return np.clip(ind, 0, 1)
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        population.append(create_individual())
    
    # Add initial solution to population
    population = [h_start.copy()] + population[1:pop_size]
    
    # GA main loop
    no_improvement_count = 0
    best_found = best_c5
    
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_found:
            best_found = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 100:
            break
        
        # Selection using tournament
        def tournament_selection():
            best_idx = random.randint(0, pop_size - 1)
            for _ in range(tournament_size - 1):
                idx = random.randint(0, pop_size - 1)
                if fitness[idx] < fitness[best_idx]:
                    best_idx = idx
            return population[best_idx]
        
        # Crossover: one-point
        def crossover(parent1, parent2):
            point = random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:point], parent2[point:]])
            child2 = np.concatenate([parent2[:point], parent1[point:]])
            
            # Renormalize to keep constraint
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            return np.clip(child1, 0, 1), np.clip(child2, 0, 1)
        
        # Mutation: Gaussian noise
        def mutate(individual):
            mutated = individual.copy()
            for i in range(n_points):
                if random.random() < mutation_rate:
                    mutated[i] += np.random.normal(0, 0.1)
            
            # Clip and renormalize
            mutated = np.clip(mutated, 0, 1)
            if np.sum(mutated) > 0:
                mutated = mutated * (n_points / 2) / np.sum(mutated)
            return mutated
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final improvement with local search
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        # Try with best found solution
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, dx)
            if final_c5 < best_c5:
                best_h = h_final
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
