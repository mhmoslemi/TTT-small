import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with perturbations
    max_iter = 100000
    learning_rate = 1e-3
    p_perturb = 0.1
    perturb_magnitude = 0.05
    patience = 100
    patience_counter = 0
    
    h_current = h_start.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_shift = max_idx - (n_points - 1)
        
        # Compute subgradient for max shift
        subgradient = np.zeros(n_points)
        if max_shift >= 0:
            # h[i] * (1 - h[i + max_shift])
            for i in range(n_points - max_shift):
                subgradient[i] += (1 - h_current[i + max_shift])
        else:
            # h[i + abs(max_shift)] * (1 - h[i])
            for i in range(-max_shift, n_points):
                subgradient[i] += (1 - h_current[i + max_shift])
        
        # Update h
        h_new = h_current - learning_rate * subgradient
        h_new = project_to_feasible(h_new, n_points)
        
        # Perturbation
        if np.random.random() < p_perturb:
            perturbation = np.random.normal(0, perturb_magnitude, n_points)
            h_new = h_new + perturbation
            h_new = project_to_feasible(h_new, n_points)
        
        # Accept the new point if it improves C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        h_current = h_new.copy()
        
        # Early termination if no improvement
        if patience_counter >= patience:
            break
            
        # Adjust learning rate
        if i % 1000 == 0 and i > 0:
            learning_rate *= 0.95
    
    return (list(best_h), best_c5, n_points)
