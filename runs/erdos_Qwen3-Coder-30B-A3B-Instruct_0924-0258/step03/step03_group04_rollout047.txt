import numpy as np
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe algorithm
    h_curr = h_start.copy()
    max_iter = 10000
    eta = 0.5  # step size
    perturb_rate = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute shifts
        h_shifted = np.zeros((n_points, n_points))
        for k in range(n_points):
            h_shifted[k, k:] = h_curr[:n_points-k]
        
        # Compute correlation
        corr = np.zeros(n_points)
        for k in range(n_points):
            corr[k] = np.sum(h_curr[:n_points-k] * (1 - h_curr[k:]))
        
        # Find maximum shift
        max_k = np.argmax(corr)
        
        # Linear objective for Frank-Wolfe
        linear_obj = 1 - h_curr[max_k:]  # corresponds to 1 - h[i+k] 
        if max_k > 0:
            linear_obj = np.concatenate([linear_obj, np.zeros(max_k)]) 
        
        # Subproblem: find extreme point that minimizes linear_obj
        # This is a ranking-based heuristic: select indices where (1-h) is largest
        diffs = np.zeros(n_points)
        for i in range(n_points):
            diffs[i] = h_curr[i] * linear_obj[i] if linear_obj[i] < 0 else (1 - h_curr[i]) * linear_obj[i]
        
        # Simple heuristic: select indices where 1-h is large
        h_extreme = np.zeros(n_points)
        indices_sorted = np.argsort(linear_obj)[::-1]  # descending order
        ones_count = int(n_points / 2)
        h_extreme[indices_sorted[:ones_count]] = 1.0
        h_extreme = project_to_feasible(h_extreme, n_points)
        
        # Frank-Wolfe update
        h_new = (1 - eta) * h_curr + eta * h_extreme
        h_curr = project_to_feasible(h_new, n_points)
        
        # Evaluate
        c5_curr = evaluate_erdos_solution(h_curr, dx)
        if c5_curr < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5_curr
            
        # Perturb occasionally to escape local optima
        if iteration % 100 == 0 and iteration > 0:
            if np.random.random() < perturb_rate:
                h_curr += np.random.normal(0, 0.01, n_points)
                h_curr = project_to_feasible(h_curr, n_points)
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and best_c5 > 0.37:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        # Gradient descent refinement
        from scipy.optimize import minimize
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
                
    return (list(best_h), best_c5, n_points)
