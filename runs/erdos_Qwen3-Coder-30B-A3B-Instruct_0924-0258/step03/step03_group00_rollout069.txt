import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based parametrization approach
    
    # Number of Fourier coefficients to optimize
    num_coeffs = 5
    
    def get_h_from_coeffs(coeffs):
        # Create Fourier series
        x = np.linspace(0, 2, n_points)
        g = np.zeros_like(x)
        for i in range(num_coeffs):
            g += coeffs[i] * np.cos(i * np.pi * x)
        # Apply logistic transform to keep in [0,1]
        h = 1 / (1 + np.exp(-g))
        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def objective(coeffs):
        h = get_h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(coeffs):
        h = get_h_from_coeffs(coeffs)
        return np.sum(h) - n_points / 2
    
    # Bounds for coefficients
    bounds = [(-10, 10) for _ in range(num_coeffs)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Random restarts
    num_restarts = 20
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Random starting point in coefficient space
        initial_coeffs = np.random.uniform(-2, 2, num_coeffs)
        
        try:
            result = minimize(
                objective,
                initial_coeffs,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-6}
            )
            
            if result.success:
                h_candidate = get_h_from_coeffs(result.x)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Final refinement with direct optimization on the discrete values
    if time.time() - start_time < budget_s * 0.95:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Gradient-based local refinement
        num_refinements = 10
        for i in range(num_refinements):
            if time.time() - start_time > budget_s * 0.98:
                break
            try:
                res = minimize(obj_func, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < best_c5:
                        best_h = res.x.copy()
                        best_c5 = c5_test
            except:
                break
    
    return (list(best_h), best_c5, n_points)
