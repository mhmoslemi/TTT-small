import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-start gradient descent with adaptive sampling
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Main optimization loop
    max_iterations = 100
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate using diverse perturbations
        h_candidate = h_start.copy()
        # Randomized shuffle with small perturbations
        indices = np.random.permutation(n_points)
        for i in range(0, n_points, 3):  # Every 3rd for variety
            if i < n_points:
                idx = indices[i]
                perturbation = np.random.normal(0, 0.05)
                h_candidate[idx] = np.clip(h_candidate[idx] + perturbation, 0, 1)
        
        # Normalize to satisfy sum constraint
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Try local optimization with multiple initializations
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Multiple restarts for SLSQP
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = h_candidate + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
        
        # Keep the candidate for next iteration as base
        h_start = best_h.copy()
    
    return (list(best_h), best_c5, n_points)
