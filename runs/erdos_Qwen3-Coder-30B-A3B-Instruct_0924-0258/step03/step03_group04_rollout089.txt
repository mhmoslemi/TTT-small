import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 200
    inertia = 0.729
    cogni = 1.49445
    social = 1.49445
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    for _ in range(swarm_size):
        h = np.random.uniform(0, 1, n_points)
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                           cogni * r1 * (pbest[i] - particles[i]) + 
                           social * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate and update personal best
            new_val = evaluate_erdos_solution(particles[i], dx)
            if new_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_val
                
                # Update global best
                if new_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = new_val
                    
        if gbest_val < best_c5:
            best_h = gbest.copy()
            best_c5 = gbest_val
    
    # Refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refinement_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        # Simple gradient descent
        step_size = 1e-3
        max_steps = 500
        current_h = best_h.copy()
        current_c5 = best_c5
        
        for step in range(max_steps):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_proj = project_to_feasible(current_h, n_points)
            new_c5 = evaluate_erdos_solution(h_proj, dx)
            
            if new_c5 < current_c5:
                current_c5 = new_c5
                best_h = h_proj.copy()
                best_c5 = current_c5
            
            # Simple gradient approximation
            eps = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_proj.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                val_plus = evaluate_erdos_solution(h_plus, dx)
                
                grad[i] = (val_plus - new_c5) / eps
            
            # Update with gradient
            current_h -= step_size * grad
            current_h = project_to_feasible(current_h, n_points)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
    
    # Final check
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    if final_c5 < best_c5:
        best_h = final_h.copy()
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
