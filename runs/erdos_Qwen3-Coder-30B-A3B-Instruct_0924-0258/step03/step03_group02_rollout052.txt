import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iterations = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial solution as one particle
    h_init = project_to_feasible(h_start, n_points)
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize remaining particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points) * 0.8 + 0.1  # [0.1, 0.9]
        h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_best = [evaluate_erdos_solution(p, dx) for p in particles]
    personal_best_h = [p.copy() for p in particles]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best)
    global_best = personal_best[global_best_idx]
    global_best_h = personal_best_h[global_best_idx].copy()
    
    # PSO loop
    iterations_no_improvement = 0
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Adaptive inertia weight
        w = w_start - (w_start - w_end) * (iteration / max_iterations)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_h[i] - particles[i]) +
                           c2 * r2 * (global_best_h - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply feasibility constraints
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate fitness
            current_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_c5 < personal_best[i]:
                personal_best[i] = current_c5
                personal_best_h[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < global_best:
                    global_best = current_c5
                    global_best_h = particles[i].copy()
                    iterations_no_improvement = 0
                else:
                    iterations_no_improvement += 1
            
        # Early stopping if no improvement
        if iterations_no_improvement > 100:
            break
    
    # Post-processing: refine best solution with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, global_best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < global_best:
                    global_best_h = h_final
                    global_best = c5_final
        except Exception:
            pass
    
    return (list(global_best_h), global_best, n_points)
