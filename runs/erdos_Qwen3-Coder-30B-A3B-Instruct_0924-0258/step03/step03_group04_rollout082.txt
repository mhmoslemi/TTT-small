import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_subgradient(h, max_shift, n_points, dx):
    """Compute subgradient for the maximum overlap constraint"""
    subgrad = np.zeros(n_points)
    if max_shift == 0:
        return subgrad
        
    # Compute contribution for shift k
    valid_len = n_points - max_shift
    if valid_len > 0:
        # Gradient of h[i] in the constraint
        for i in range(valid_len):
            subgrad[i] += (1 - h[i + max_shift]) * dx
        for i in range(max_shift, n_points):
            subgrad[i] += (-h[i - max_shift]) * dx
    
    return subgrad

    # Ensure initial sum is correct
    h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent settings
    lr = 1e-3
   decay_rate = 0.999
    p_perturb = 0.1
    perturb_magnitude = 0.05
    max_iter = int(1e6)
    
    # For tracking no improvement iterations
    no_improvement_count = 0
    max_no_improvement = 100

        # Compute overlap and find maximum shift
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Returns a vector of length 2*n_points - 1, centered at n_points-1
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (n_points - 1)
        max_overlap_value = corr[max_shift_idx] * dx
        
        # Check if we improved
        if max_overlap_value < best_c5:
            best_c5 = max_overlap_value
            best_h = best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                break
        
        # Compute subgradient of the maximum overlap constraint
        subgrad = compute_subgradient(best_h, max_shift, n_points, dx)
        
        # Update with subgradient descent
        h_new = best_h - lr * subgrad
        h_new = project_to_feasible(h_new, n_points)
        
        # Occasionally perturb
        if np.random.random() < p_perturb:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_new = h_new + perturb
            h_new = project_to_feasible(h_new, n_points)
        
        best_h = h_new.copy()
        lr *= decay_rate
