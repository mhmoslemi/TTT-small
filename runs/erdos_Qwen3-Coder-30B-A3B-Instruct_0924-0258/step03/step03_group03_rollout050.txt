import numpy as np
from scipy.signal import correlate
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Try multiple resolutions and approaches
    resolutions = [25, 30, 35, 40]
    best_c5 = float('inf')
    best_h = None
    best_n_points = None
    
    # Try initialization from initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        dx = 2.0 / n_points
        current_c5 = evaluate_erdos_solution(h_start, dx)
        if current_c5 < best_c5:
            best_h = h_start.copy()
            best_c5 = current_c5
            best_n_points = n_points
    
    # Helper function to generate smooth symmetric construction
    def generate_smooth_symmetric(n):
        x = np.linspace(0, 2, n)
        # Create a smooth symmetric function that integrates to 1
        h = np.exp(-((x - 1)**2) / 0.1) / np.sum(np.exp(-((x - 1)**2) / 0.1)) * (n / 2)
        return np.clip(h, 0, 1)
    
    # Helper function to generate piecewise constant with fine resolution
    def generate_piecewise_const(n):
        # Divide the interval into groups, with values near 0 or 1
        h = np.ones(n)
        # Leave some portions at 0 or low values
        for i in range(n):
            if i < n//3 or i > 2*n//3:
                h[i] = 0.0
            elif i < n//2:
                h[i] = 1.0
            else:
                h[i] = 0.5
        # Normalize
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n / 2) / h_sum
        return h
    
    # Try starting from smooth constructions
    for n_points in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        dx = 2.0 / n_points
        
        # Try smooth symmetric construction
        h_smooth = generate_smooth_symmetric(n_points)
        h_smooth = project_to_feasible(h_smooth, n_points)
        c5_smooth = evaluate_erdos_solution(h_smooth, dx)
        if c5_smooth < best_c5:
            best_h = h_smooth.copy()
            best_c5 = c5_smooth
            best_n_points = n_points
            
        # Try piecewise constant
        h_piece = generate_piecewise_const(n_points)
        h_piece = project_to_feasible(h_piece, n_points)
        c5_piece = evaluate_erdos_solution(h_piece, dx)
        if c5_piece < best_c5:
            best_h = h_piece.copy()
            best_c5 = c5_piece
            best_n_points = n_points

    # Refine using quasi-Newton with projected gradient if we have a current best solution
    if best_h is not None and time() - start_time < budget_s * 0.8:
        h_ref = best_h.copy()
        n_points = best_n_points
        dx = 2.0 / n_points
        
        # Use Newton's method with projected gradient for refinement
        max_iter = 500
        step_size = 0.01
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Approximate gradient via finite differences
            grad = np.zeros_like(h_ref)
            eps = 1e-5
            base_score = evaluate_erdos_solution(h_ref, dx)
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                score_plus = evaluate_erdos_solution(h_plus, dx)
                grad[j] = (score_plus - base_score) / eps
                
            # Update step
            h_ref = h_ref - step_size * grad
            h_ref = project_to_feasible(h_ref, n_points)
            
            new_score = evaluate_erdos_solution(h_ref, dx)
            if new_score < best_c5:
                best_c5 = new_score
                best_h = h_ref.copy()
                best_n_points = n_points
                
            # Adaptive step size
            if np.linalg.norm(grad) < 1e-6:
                break

    return (list(best_h), best_c5, best_n_points)
