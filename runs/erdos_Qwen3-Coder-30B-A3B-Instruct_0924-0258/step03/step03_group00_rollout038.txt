import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def create_breakpoint_function(breakpoints, values, n_points):
    """Convert breakpoints and values to step function"""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    for i in range(len(breakpoints)-1):
        start_idx = int(breakpoints[i] / dx)
        end_idx = int(breakpoints[i+1] / dx)
        if end_idx > start_idx:
            h[start_idx:end_idx] = values[i]
    return h

def validate_solution(h, dx):
    """Ensure the solution satisfies constraints"""
    # Clamp values to [0,1]
    h = np.clip(h, 0, 1)
    # Ensure area constraint
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (1.0 / integral) * (n_points / 2)
    return h

def generate_initial_solution(n_points):
    """Generate initial solution from initial_h_values or default"""
    if 'initial_h_values' in globals():

        h = validate_solution(h, 2.0 / n_points)
    else:

def genetic_algorithm_run(n_points, budget_s):
    """Run genetic algorithm to find improved solution"""
    start_time = time.time()
    dx = 2.0 / n_points
    
    # Parameters for GA
    population_size = 500
    generations = 200
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    fitness = []
    
    for _ in range(population_size):
        h = generate_initial_solution(n_points)
        h = validate_solution(h, dx)
        c5 = evaluate_erdos_solution(h, dx)
        population.append(h.copy())
        fitness.append(c5)
    
    # Genetic algorithm main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20% and generate new population
        elite_size = population_size // 5
        new_population = population[:elite_size]
        new_fitness = fitness[:elite_size]
        
        # Generate rest via crossover and mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate:
                # Tournament selection
                parent1_idx = np.random.choice(elite_size)
                parent2_idx = np.random.choice(elite_size)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Simple uniform crossover
                child = np.zeros_like(parent1)
                mask = np.random.random(n_points) < 0.5
                child[mask] = parent1[mask]
                child[~mask] = parent2[~mask]
            else:
                # Select one parent
                parent_idx = np.random.choice(elite_size)
                child = population[parent_idx].copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                mutation_idx = np.random.choice(n_points)
                child[mutation_idx] = np.random.random()
            
            child = validate_solution(child, dx)
            child_c5 = evaluate_erdos_solution(child, dx)
            
            new_population.append(child)
            new_fitness.append(child_c5)
        
        population = new_population
        fitness = new_fitness
    
    # Final evaluation of top solutions with local refinement
    best_idx = np.argmin(fitness)
    best_h = population[best_idx]
    best_c5 = fitness[best_idx]
    
    # Local refinement on top 5 solutions
    top_k = min(5, len(population))
    for i in range(top_k):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        h_start = population[i].copy()
        def objective(h):
            h = np.clip(h, 0, 1)
            integral = np.sum(h) * dx
            if integral > 0:
                h = h * (1.0 / integral) * (n_points / 2)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - (n_points / 2)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50})
            if res.success:
                c5_test = objective(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            continue
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use reasonable number of points to balance speed and accuracy
    n_points = 35
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm_run(n_points, budget_s)
    
    # Final validation
    dx = 2.0 / n_points
    validated_h = validate_solution(best_h, dx)
    final_c5 = evaluate_erdos_solution(validated_h, dx)
    
    return (list(validated_h), final_c5, n_points)
