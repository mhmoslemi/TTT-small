import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Create population of candidates
    pop_size = 15
    population = [h_start.copy()]
    
    # Generate some diverse initial candidates
    for i in range(1, pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Nelder-Mead simplex approach
    def simplex_iteration():
        nonlocal population, best_h, best_c5
        
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Simplex operations
        # Replicate the best, reflect, expand, contract as needed
        # For simplicity, we'll just do a random mutation based on top few
        top_candidates = population[:3]
        
        # Create new candidate by combining top ones with noise
        h_new = np.mean(top_candidates, axis=0)
        h_new += np.random.normal(0, 0.02, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Replace worst member of population
        population[-1] = h_new
        
    # Local refinement with gradient descent
    def refine_solution(h_in):
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def cons_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': cons_sum}
        
        try:
            res = minimize(obj, h_in, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                return h_refined, c5_refined
        except:
            pass
        return h_in, evaluate_erdos_solution(h_in, dx)
    
    # Main loop
    max_iter = 300
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Nelder-Mead step
        simplex_iteration()
        
        # Local refinement on best few
        for i in range(min(3, len(population))):
            refined_h, refined_c5 = refine_solution(population[i])
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
            population[i] = refined_h
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Refine the best solution further
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        
        try:
            res = minimize(evaluate_erdos_solution, best_h, 
                          args=(dx,), method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                best_h = res.x
                best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
