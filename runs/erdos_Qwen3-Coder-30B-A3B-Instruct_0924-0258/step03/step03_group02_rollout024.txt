import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coordinate descent with random order updates
    max_iterations = 500
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Shuffle indices for random order update
        indices = list(range(n_points))
        random.shuffle(indices)
        
        improved = False
        h_current = best_h.copy()
        
        for i in indices:
            if time() - start_time > budget_s * 0.9:
                break
                
            # Store original value
            original_h_i = h_current[i]
            
            # Simple line search: try several values for h[i]
            best_h_i = original_h_i
            best_c5_local = best_c5
            
            # Search in a small neighborhood around current value
            steps = 20
            for j in range(steps + 1):
                # Try values uniformly in [0, 1]
                new_h_i = j / steps
                if new_h_i < 0 or new_h_i > 1:
                    continue
                    
                h_current[i] = new_h_i
                h_current = project_to_feasible(h_current, n_points)
                
                # Check if this improves C5
                c5_new = evaluate_erdos_solution(h_current, dx)
                
                if c5_new < best_c5_local - tolerance:
                    best_c5_local = c5_new
                    best_h_i = new_h_i
                    
            # If we found an improvement, update
            if abs(best_h_i - original_h_i) > 1e-8:
                h_current[i] = best_h_i
                improved = True
                
        # Project to feasible at the end of iteration
        h_current = project_to_feasible(h_current, n_points)
        
        # Update best solution if this improved
        c5_current = evaluate_erdos_solution(h_current, dx)
        if c5_current < best_c5 - tolerance:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Early termination if no improvement
        if not improved:
            break
    
    # Post-refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        # Objective function
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Constraint function
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        # Optimize
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
