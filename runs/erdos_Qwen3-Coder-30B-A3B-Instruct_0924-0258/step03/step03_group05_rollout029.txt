import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fine-tune initial solution
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_start = np.clip(res.x, 0, 1)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            best_c5 = evaluate_erdos_solution(h_start, dx)
            best_h = h_start.copy()
    except:
        pass

    # Increase resolution if time allows
    if budget_s > 300 and n_points == 35:
        n_points = 70
        dx = 2.0 / n_points
        h_start = np.interp(np.linspace(0, 2, n_points), 
                            np.linspace(0, 2, 35), h_start)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(h_start, dx)
        
        # Further optimization at higher resolution
        try:
            res = minimize(objective, h_start, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_start = np.clip(res.x, 0, 1)
                h_start = h_start * (n_points / 2) / np.sum(h_start)
                c5_new = evaluate_erdos_solution(h_start, dx)
                if c5_new < best_c5:
                    best_h = h_start.copy()
                    best_c5 = c5_new
        except:
            pass

    # Adaptive local search in the remaining time
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and n_points == 70:
        # Simpler gradient-based approach
        for _ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random perturbation
            pert = np.random.normal(0, 0.01, n_points)
            pert = np.clip(pert, -0.1, 0.1)
            h_pert = best_h + pert
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert

    # Final validation and refinement
    final_h = best_h.copy()
    final_c5 = best_c5
    
    # Final optimization pass
    try:
        res = minimize(objective, final_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < final_c5:
                final_h = h_final
                final_c5 = c5_final
    except:
        pass
    
    return (list(final_h), final_c5, n_points)
