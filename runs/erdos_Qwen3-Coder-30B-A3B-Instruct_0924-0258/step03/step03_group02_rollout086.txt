import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a variant of alternating optimization
    max_iter = 50
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convex relaxation approach: try to find a better solution
        h_relaxed = best_h.copy()
        
        # Alternate between different initialization strategies
        if i % 5 == 0:
            # Try random perturbation
            h_try = h_relaxed + np.random.normal(0, 0.05, n_points)
            h_try = project_to_feasible(h_try, n_points)
        elif i % 3 == 0:
            # Try a directional perturbation
            h_try = h_relaxed.copy()
            idx = np.random.randint(0, n_points)
            h_try[idx] = np.clip(h_try[idx] + np.random.normal(0, 0.1), 0, 1)
            h_try = project_to_feasible(h_try, n_points)
        else:
            h_try = h_relaxed
        
        # Try to improve with local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint}
        
        try:
            res = minimize(objective, h_try, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': 1e-7})
            if res.success:
                h_candidate = project_to_feasible(res.x, n_points)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except Exception:
            pass
    
    # Final fine-tuning
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5:
        try:
            h_final = best_h.copy()
            def final_objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def final_constraint(h):
                return np.sum(h) - n_points / 2
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': final_constraint}
            
            res = minimize(final_objective, h_final, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 30, 'ftol': 1e-8})
            
            if res.success:
                h_opt = project_to_feasible(res.x, n_points)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
