import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 20
    mutation_rate = 0.3
    crossover_rate = 0.7
    sqp_interval = 1
    
    # Initialize population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Genetic Algorithm
    generation = 0
    max_generations = 30
    
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select top 50%
        selected = population[:population_size//2]
        
        # Crossover
        new_population = selected.copy()
        while len(new_population) < population_size:
            if np.random.random() < crossover_rate and len(selected) >= 2:
                i, j = np.random.choice(len(selected), 2, replace=False)
                parent1, parent2 = selected[i], selected[j]
                alpha = np.random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
            else:
                # Mutation
                if len(selected) > 0:
                    parent = selected[np.random.randint(0, len(selected))]
                    child = parent + np.random.normal(0, 0.03, n_points)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                    new_population.append(child)
        
        # Keep population size in check
        population = new_population[:population_size]
        
        # Local refinement with SQP
        if generation % sqp_interval == 0 and generation > 0:
            for i in range(min(3, len(population))):
                h_initial = population[i].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_initial, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        refined_h = res.x
                        c5_refined = objective(refined_h)
                        if c5_refined < best_c5:
                            best_h = refined_h
                            best_c5 = c5_refined
                        # Update in population
                        population[i] = refined_h
                except:
                    pass
        
        generation += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try refining best solution
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                refined_h = res.x
                c5_refined = objective(refined_h)
                if c5_refined < best_c5:
                    best_h = refined_h
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
