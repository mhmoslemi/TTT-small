import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based refinement using scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Start with the best starting point
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Local refinement parameters
    max_refinements = 100
    refinement_count = 0
    
    # Use SLSQP with multiple restarts
    while time() - start_time < budget_s * 0.95 and refinement_count < max_refinements:
        # Try multiple random restarts
        best_restart_c5 = current_c5
        best_restart_h = current_h.copy()
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Small random perturbation
            h_restart = current_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            # Fix sum constraint
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            # Optimize this restart
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_candidate = res.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_restart_c5:
                        best_restart_c5 = c5_candidate
                        best_restart_h = h_candidate
            except:
                continue
                
        # Update current solution
        if best_restart_c5 < current_c5:
            current_h = best_restart_h
            current_c5 = best_restart_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # No improvement, perturb slightly and try again
            current_h = current_h + np.random.normal(0, 0.005, n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            current_c5 = evaluate_erdos_solution(current_h, dx)
            
        refinement_count += 1
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final convex refinement for best solution
    if time() - start_time < budget_s * 0.98:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints
            corr_offsets = []
            for k in range(2 * n_points - 1):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                corr_offsets.append(corr_sum)
                
            # We want to minimize max of these correlations
            max_corr = cp.Variable()
            for corr_sum in corr_offsets:
                constraints.append(corr_sum <= max_corr * dx)
                
            prob = cp.Problem(cp.Minimize(max_corr), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
