import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    random.seed(seed)
    start_time = time()

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 50
    generations = 1000
    mutation_rate = 0.2
    crossover_rate = 0.8
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Renormalize to sum to 1
        return np.clip(h, 0, 1)
    
    def create_population():
        pop = []
        if 'initial_h_values' in globals():
            pop.append(h_start.copy())
        for _ in range(population_size - len(pop)):
            pop.append(create_individual())
        return pop
    
    # Selection with tournament
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(len(population)):
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_index])
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Renormalize
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return np.clip(child1, 0, 1), np.clip(child2, 0, 1)
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.1)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return np.clip(mutated, 0, 1)
    
    # GA main loop
    population = create_population()
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
    
    # Final verification and refinement
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Local refinement
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try to improve with this final solution
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = np.clip(res.x, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < final_c5:
                best_h = h_refined
                final_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), final_c5, n_points)
