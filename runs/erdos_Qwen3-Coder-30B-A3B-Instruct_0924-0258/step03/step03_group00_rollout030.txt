import numpy as np
import time
from scipy.optimize import minimize
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu list to avoid revisiting recent solutions
    tabu_list = deque(maxlen=10)
    
    # Simulated annealing parameters
    temp_start = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    
    # Start of simulated annealing
    current_h = h_start.copy()
    current_c5 = best_c5
    temp = temp_start
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Generate neighbor by perturbing one point
        neighbor_h = current_h.copy()
        idx = np.random.randint(n_points)
        # Try small perturbations
        perturb = np.random.choice([-0.05, -0.02, 0.02, 0.05])
        neighbor_h[idx] += perturb
        # Ensure bounds
        neighbor_h[idx] = np.clip(neighbor_h[idx], 0, 1)
        
        # Normalize to satisfy integral constraint
        h_sum = np.sum(neighbor_h)
        if h_sum > 0:
            neighbor_h = neighbor_h * (n_points / 2) / h_sum
            
        # Avoid tabu configurations
        neighbor_tuple = tuple(neighbor_h.round(6))
        if neighbor_tuple in tabu_list:
            continue
            
        # Evaluate neighbor
        neighbor_c5 = evaluate_erdos_solution(neighbor_h, dx)
        
        # Accept or reject based on simulated annealing
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
            tabu_list.append(neighbor_tuple)
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            # Accept worse solution with probability
            delta = neighbor_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temp):
                current_h = neighbor_h
                current_c5 = neighbor_c5
                tabu_list.append(neighbor_tuple)
        
        # Cool down temperature
        temp = max(temp * cooling_rate, min_temp)
        
        # Occasionally run local optimization to refine current solution
        if iteration % 50 == 0:
            def obj_func(h):
                h_clipped = np.clip(h, 0, 1)
                h_sum = np.sum(h_clipped)
                if h_sum > 0:
                    h_normalized = h_clipped * (n_points / 2) / h_sum
                else:
                    h_normalized = h_clipped
                return evaluate_erdos_solution(h_normalized, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Local optimization with current solution as starting point
            try:
                res = minimize(obj_func, current_h, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    c5_test = obj_func(res.x)
                    if c5_test < current_c5:
                        current_h = res.x.copy()
                        current_c5 = c5_test
                        if current_c5 < best_c5:
                            best_h = current_h.copy()
                            best_c5 = current_c5
            except:
                pass
    
    # Final local optimization on best solution
    if time.time() - start_time < budget_s * 0.99:
        def obj_func(h):
            h_clipped = np.clip(h, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_func, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                final_c5 = obj_func(res.x)
                if final_c5 < best_c5:
                    best_h = res.x.copy()
                    best_c5 = final_c5
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
