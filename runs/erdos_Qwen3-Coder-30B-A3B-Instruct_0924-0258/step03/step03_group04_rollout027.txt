import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic initialization
        h_start = 0.5 + 0.3 * np.sin(np.pi * np.arange(n_points) / n_points)
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    iterations = 0
    max_iterations = 50
    
    while time() - start_time < budget_s * 0.9 and iterations < max_iterations:
        iterations += 1
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate overlaps for all shifts
        h_eval = project_to_feasible(best_h, n_points)
        corr = np.correlate(h_eval, 1 - h_eval, mode='full')
        corr = corr * dx
        
        # Find worst overlaps (top 10% of shifts)
        worst_indices = np.argsort(-corr)[0:max(1, n_points // 10)]
        
        # Keep only the worst shifts
        # This is a rigorous constraint tightening approach 
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add correlations for worst shifts only
        for k in worst_indices:
            # Convert full correlation index back to actual shift
            if k >= n_points:
                shift = k - n_points + 1
            else:
                shift = k
                
            if shift == 0:
                continue
                
            # Compute valid indices for this shift
            valid_len = n_points - shift
            if valid_len > 0:
                # Note: For zero-shift, we skip since we want k > 0
                corr_sum = cp.sum(cp.multiply(h_var[:valid_len], 1 - h_var[shift:shift+valid_len]))
                # Constrain this correlation
                constraints.append(corr_sum <= (best_c5 + 1e-6) * valid_len * dx)
        
        # Convex feasibility problem with subset of constraints
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_feas = h_var.value
                if h_feas is not None:
                    # Refine with gradient descent
                    def refinement_obj(h):
                        h_proj = project_to_feasible(h, n_points)
                        return evaluate_erdos_solution(h_proj, dx)
                    
                    res = minimize(refinement_obj, h_feas, method='L-BFGS-B', 
                                 bounds=[(0,1) for _ in range(n_points)], 
                                 options={'maxiter': 50, 'gtol': 1e-6, 'ftol': 1e-6})
                    if res.success:
                        h_refined = project_to_feasible(res.x, n_points)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                            # Early exit if we've reached target
                            if best_c5 <= 0.3807:
                                break
        except Exception:
            pass
    
    # Final refinement with gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100, 'gtol': 1e-6, 'ftol': 1e-6})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
