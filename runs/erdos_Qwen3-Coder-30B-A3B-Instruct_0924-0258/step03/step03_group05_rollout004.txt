import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    c_low = 0.37
    c_high = 0.38085

    def solve_feasibility(h_guess, c_val):
        # Define CVX variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1
        ]
        
        # Add overlap constraints for all shifts
        for k in range(1, n_points):
            # Consider valid indices for this shift
            valid_len = n_points - k
            if valid_len <= 0:
                break
            # Constraint: sum(h[i] * (1 - h[i+k])) * dx <= c_val
            overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_val
            constraints.append(overlap_constraint)
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000, feastol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        return None

    for iteration in range(max_iterations):
