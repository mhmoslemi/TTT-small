import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a direct optimization approach with simulated annealing inspired method
    # Initially use convex relaxation for good starting point
    
    # First, let's solve a relaxed version using linear programming
    try:
        # Define variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add the most restrictive overlap constraint (k=0 for initial estimate)
        # This will be conservative but allow us to use convex relaxation
        obj = cp.Maximize(cp.sum(cp.multiply(h_vars, 1 - h_vars)) * dx)
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-8, abstol=1e-8)
        
        if prob.status == cp.OPTIMAL:
            h_init = h_vars.value
            h_init = project_to_feasible(h_init, n_points)
            c5_val = evaluate_erdos_solution(h_init, dx)
            if c5_val < best_c5:
                best_h = h_init.copy()
                best_c5 = c5_val
    except:
        pass

    # Reinitialize with the better solution and refine with SLSQP
    h_refined = best_h.copy()
    
    # Define optimization problem
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Try multiple random restarts to avoid local optima
    num_restarts = 5
    for restart in range(num_restarts):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Random perturbation around current best
        perturbation = np.random.normal(0, 0.05, n_points)
        h_trial = h_refined + perturbation
        h_trial = project_to_feasible(h_trial, n_points)
        
        try:
            res = minimize(objective, h_trial, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
        
        h_refined = best_h.copy()
    
    # Final validation
    final_h = project_to_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
