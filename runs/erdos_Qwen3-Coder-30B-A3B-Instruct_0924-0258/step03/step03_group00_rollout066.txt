import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier approach with 5 terms
    d = 5  # number of cosine terms
    
    def h_from_coeffs(coeffs):
        x = np.linspace(0, 2, n_points)
        g = coeffs[0] + np.sum([coeffs[k+1] * np.cos((k+1) * np.pi * x) for k in range(d)], axis=0)
        h = 1.0 / (1.0 + np.exp(-g))
        # Normalize to have integral = 1
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def obj_func(coeffs):
        h = h_from_coeffs(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    # Bounds for coefficients
    bounds = [(-10, 10)] * (d + 1)
    
    # Multiple random restarts
    num_restarts = 10
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        coeffs_start = np.random.uniform(-1, 1, d + 1)
        
        try:
            res = minimize(obj_func, coeffs_start, method='SLSQP', 
                          bounds=bounds, options={'maxiter': 100, 'ftol': 1e-6})
            
            if res.success:
                h_candidate = h_from_coeffs(res.x)
                c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                    
        except:
            continue
    
    # Final local refinement
    if time.time() - start_time < budget_s * 0.98:
        def obj_func_local(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Start from our best solution and do gradient-based refinement
        try:
            res = minimize(obj_func_local, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func_local(res.x)
                if c5_test < best_c5:
                    best_h = res.x.copy()
                    best_c5 = c5_test
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
