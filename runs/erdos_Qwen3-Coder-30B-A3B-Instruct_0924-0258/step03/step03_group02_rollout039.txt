import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use a local search approach to improve the construction
    max_iterations = 100
    patience = 10
    no_improvement_count = 0
    current_h = h_start.copy()
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_perturbed = current_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_feasible(h_perturbed, n_points)
        
        # Evaluate and accept if better
        c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_perturbed
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # If no improvement for too long, reduce learning rate
        if no_improvement_count > patience:
            no_improvement_count = 0
            if i > 0:
                current_h = current_h + 0.5 * (best_h - current_h)
                current_h = project_to_feasible(current_h, n_points)
        else:
            current_h = h_perturbed
          
    # SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and best_c5 > 0.38:
        h_refined = best_h.copy()
        
        def refine_objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def refine_constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': refine_constraint}
        
        try:
            res = minimize(refine_objective, h_refined, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
