import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create periodic initialization
        h_start = 0.5 * np.sin(2 * np.pi * np.arange(n_points) / n_points) + 0.5
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T = 0.1
    T_min = 1e-6
    alpha = 0.99
    iterations = 0
    max_iter = 10000
    accept_count = 0
    accept_limit = 200
    temp_update_freq = 100
    
    # Perturbation parameters
    perturb_frac = 0.1  # 10% of points
    max_perturb = int(n_points * perturb_frac)
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    while time() - start_time < budget_s * 0.9 and iterations < max_iter:
        # Generate candidate by perturbing some points
        candidate_h = current_h.copy()
        
        # Select random subset to perturb
        n_perturb = min(max_perturb, max(1, int(np.random.exponential(3))))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Perturb selected points
        for i in indices:
            # Randomly adjust the value
            delta = np.random.normal(0, 0.1)
            candidate_h[i] += delta
            
        # Project to feasible region
        candidate_h = project_to_feasible(candidate_h, n_points)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Acceptance criterion
        delta_c5 = candidate_c5 - current_c5
        
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
            accept_count += 1
        else:
            # Sometimes accept even if worse to escape local minima
            if np.random.random() < 0.01:  # 1% chance to accept worse
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
                accept_count += 1
        
        # Update temperature
        iterations += 1
        if iterations % temp_update_freq == 0:
            # Adaptive cooling based on acceptance rate
            if accept_count < accept_limit * 0.3:
                T *= 0.9  # Cool faster
            elif accept_count > accept_limit * 0.7:
                T *= 1.1  # Cool slower
            
            if T < T_min:
                T = T_min
            accept_count = 0
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 50})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
