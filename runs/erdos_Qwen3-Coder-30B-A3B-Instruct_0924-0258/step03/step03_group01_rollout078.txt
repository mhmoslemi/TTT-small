import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        dx = 2.0 / n_points
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Nelder-Mead optimization with population
    def objective(h):
        return evaluate_erdos_solution(h, 2.0/n_points)
    
    # Generate initial population
    population_size = 15
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = h_start + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
    
    # Nelder-Mead parameters
    max_iter = 500
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all candidates
        fitness = [objective(h) for h in population]
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Simple Nelder-Mead-like approach
        # Keep best, worst, and centroid
        best = population[0]
        worst = population[-1]
        centroid = np.mean(population[:-1], axis=0)
        
        # Reflection
        reflected = centroid + 2*(centroid - worst)
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (n_points/2) / np.sum(reflected)
        f_reflected = objective(reflected)
        
        if fitness[0] <= f_reflected < fitness[-2]:
            population[-1] = reflected
        elif f_reflected < fitness[0]:
            # Expansion
            expanded = centroid + 2*(reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (n_points/2) / np.sum(expanded)
            f_expanded = objective(expanded)
            if f_expanded < f_reflected:
                population[-1] = expanded
            else:
                population[-1] = reflected
        else:
            # Contraction
            if f_reflected < fitness[-1]:
                contracted = centroid + 0.5*(worst - centroid)
            else:
                contracted = 0.5 * (best + worst)
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (n_points/2) / np.sum(contracted)
            f_contracted = objective(contracted)
            if f_contracted < fitness[-1]:
                population[-1] = contracted
            else:
                # Shrink
                for i in range(1, len(population)):
                    population[i] = best + 0.5 * (population[i] - best)
    
    # Local refinement with gradient-based method
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine using SLSQP with restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.98:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
