import numpy as np
from scipy.optimize import minimize
import time
from collections import deque

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        h_start[25] = 1.0  # Simple spike to satisfy integral constraint
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = deque(maxlen=20)
    max_iter_no_improvement = 50
    iter_no_improvement = 0
    
    # Local search parameters
    local_iters = 10
    
    def project_h(h):
        """Project h to satisfy the integral constraint"""
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return np.clip(h, 0, 1)
    
    def get_neighbors(h, perturb_rate=0.2, perturb_mag=0.1):
        """Generate neighbors by perturbing some elements of h"""
        neighbors = []
        num_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in range(10):  # Generate 10 neighbor candidates
            h_new = h.copy()
            for idx in indices:
                # Add random perturbation
                h_new[idx] += np.random.normal(0, perturb_mag)
            h_new = project_h(h_new)
            neighbors.append(h_new)
        return neighbors
    
    def local_refinement(h, max_iter=local_iters):
        """Perform local refinement using gradient-based optimization"""
        def obj_func(h_vec):
            h_clipped = np.clip(h_vec, 0, 1)
            h_sum = np.sum(h_clipped)
            if h_sum > 0:
                h_normalized = h_clipped * (n_points / 2) / h_sum
            else:
                h_normalized = h_clipped
            return evaluate_erdos_solution(h_normalized, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj_func, h, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_result = res.x
                h_result = np.clip(h_result, 0, 1)
                h_sum = np.sum(h_result)
                if h_sum > 0:
                    h_result = h_result * (n_points / 2) / h_sum
                return h_result
        except:
            pass
        return h
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Main tabu search loop
    while time.time() - start_time < budget_s * 0.9:
        neighbors = get_neighbors(current_h)
        
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        
        for neighbor in neighbors:
            # Hash the current neighbor for tabu check
            neighbor_tuple = tuple(neighbor.round(10))
            
            # Skip if in tabu list (unless escaping)
            if neighbor_tuple in tabu_list:
                # Some escape mechanism
                escape_prob = 0.1
                if np.random.random() > escape_prob:
                    continue
            
            c5 = evaluate_erdos_solution(neighbor, dx)
            if c5 < best_neighbor_c5:
                best_neighbor = neighbor
                best_neighbor_c5 = c5
        
        # If no valid neighbor found, break out
        if best_neighbor is None:
            break
            
        # Update tabu list (add current solution)
        current_tuple = tuple(current_h.round(10))
        tabu_list.append(current_tuple)
        
        # Accept best neighbor
        current_h = best_neighbor
        current_c5 = best_neighbor_c5
        
        # Update global best if improved
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            iter_no_improvement = 0
        else:
            iter_no_improvement += 1
            
        # Stopping condition
        if iter_no_improvement > max_iter_no_improvement:
            break
        
        # Local refinement every few iterations
        if iter_no_improvement % 5 == 0:
            refined_h = local_refinement(current_h)
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < current_c5:
                current_h = refined_h
                current_c5 = refined_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
    
    # Final refinement
    if time.time() - start_time < budget_s * 0.95:
        final_h = local_refinement(best_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
