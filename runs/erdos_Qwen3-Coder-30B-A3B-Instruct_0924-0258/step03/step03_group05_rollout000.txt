import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try different discretization levels
    resolutions = [35, 50, 70]
    best_solution = (best_h, best_c5, n_points)
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res <= n_points:
            h_current = best_solution[0][:res].copy()
            h_current = h_current * (res / 2) / np.sum(h_current)
        else:
            h_current = np.interp(np.linspace(0, 1, res), 
                                  np.linspace(0, 1, n_points), 
                                  best_solution[0])
            h_current = h_current * (res / 2) / np.sum(h_current)
            
        dx_current = 2.0 / res
        n_points_current = res
        
        # Local optimization
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx_current)
            
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_current / 2}]
        bounds = [(0, 1) for _ in range(n_points_current)]
        
        try:
            res_opt = minimize(objective, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-8})
            if res_opt.success:
                h_refined = np.clip(res_opt.x, 0, 1)
                h_refined = h_refined * (n_points_current / 2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx_current)
                if c5_refined < best_solution[1]:
                    best_solution = (h_refined, c5_refined, n_points_current)
        except:
            continue
            
        # Optimization with different initializations
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
            h_init = np.random.rand(n_points_current) * 0.5 + 0.25
            h_init = h_init * (n_points_current / 2) / np.sum(h_init)
            try:
                res_opt = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                if res_opt.success:
                    h_refined = np.clip(res_opt.x, 0, 1)
                    h_refined = h_refined * (n_points_current / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx_current)
                    if c5_refined < best_solution[1]:
                        best_solution = (h_refined, c5_refined, n_points_current)
            except:
                continue
    
    final_h = best_solution[0]
    final_c5 = best_solution[1]
    final_n = best_solution[2]
    
    # Final refinement with more iterations
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, 2.0 / final_n)
        
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - final_n / 2}]
    bounds = [(0, 1) for _ in range(final_n)]
    
    try:
        res = minimize(objective, final_h, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (final_n / 2) / np.sum(h_final)
            final_c5 = evaluate_erdos_solution(h_final, 2.0 / final_n)
            final_h = h_final
    except:
        pass
        
    return (list(final_h), final_c5, final_n)
