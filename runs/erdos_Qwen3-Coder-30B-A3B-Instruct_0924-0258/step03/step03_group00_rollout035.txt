import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Generate initial population for GA
    pop_size = 500
    population = []
    
    # Create diverse initial population
    for _ in range(pop_size):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        h_new = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        population.append(h_new)
    
    # Add the starting solution
    population.append(h_start)
    
    # Genetic Algorithm loop
    num_generations = 50
    for gen in range(num_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20%
        elite_size = max(1, pop_size // 5)
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        remaining = pop_size - elite_size
        
        # Crossover (single point)
        while len(new_population) < pop_size:
            if time.time() - start_time > budget_s * 0.8:
                break
                
            parent1 = population[np.random.randint(0, pop_size // 2)]
            parent2 = population[np.random.randint(0, pop_size // 2)]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[i] = np.random.uniform(0, 1)
                if np.random.random() < mutation_rate:
                    child2[i] = np.random.uniform(0, 1)
            
            # Normalize
            h1_sum = np.sum(child1)
            if h1_sum > 0:
                child1 = child1 * (n_points / 2) / h1_sum
            new_population.append(child1)
            
            if len(new_population) < pop_size:
                h2_sum = np.sum(child2)
                if h2_sum > 0:
                    child2 = child2 * (n_points / 2) / h2_sum
                new_population.append(child2)
        
        population = new_population
        
        # Update best solution
        for h in population[:10]:  # Check top 10
            if time.time() - start_time > budget_s * 0.8:
                break
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Local refinement using SLSQP
    def obj_func(h):
        h_clipped = np.clip(h, 0, 1)
        h_sum = np.sum(h_clipped)
        if h_sum > 0:
            h_normalized = h_clipped * (n_points / 2) / h_sum
        else:
            h_normalized = h_clipped
        return evaluate_erdos_solution(h_normalized, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run several local optimizations with different starting points
    best_local = best_h.copy()
    best_local_c5 = best_c5
    num_restarts = 20
    
    for i in range(num_restarts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb current best
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        # Ensure it remains in bounds
        h_perturb = np.clip(h_perturb, 0, 1)
        
        try:
            res = minimize(obj_func, h_perturb, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                c5_test = obj_func(res.x)
                if c5_test < best_local_c5:
                    best_local = res.x.copy()
                    best_local_c5 = c5_test
        except:
            continue
    
    # Final validation
    final_h = best_local if best_local_c5 < best_c5 else best_h
    final_c5 = best_local_c5 if best_local_c5 < best_c5 else best_c5
    
    return (list(final_h), final_c5, n_points)
