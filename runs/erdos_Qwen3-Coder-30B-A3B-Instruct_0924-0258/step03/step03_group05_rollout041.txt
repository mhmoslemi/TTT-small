import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try using a more direct feasible solution approach
    # Generate candidate solutions
    def solve_feasibility(h_guess, c_val):
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) * dx == 1
        ]
        
        # Add overlap constraints for only a subset of shifts (to avoid too many constraints)
        # But make sure we don't skip any critical ones
        for k in range(0, n_points):  # Only some shifts for tractability
            if k == 0:
                continue
            valid_len = n_points - k
            if valid_len <= 0:
                break
            overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_val
            constraints.append(overlap_constraint)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK_MI, max_iters=500, feastol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        return None

    # Better gradient-based local search
    def improved_local_search(initial_h, c_val):
        h_g = initial_h.copy()
        h_g = np.clip(h_g, 0, 1)
        h_sum = np.sum(h_g)
        h_g = h_g * (n_points / 2) / h_sum if h_sum > 0 else h_g
        
        def obj(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
            
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try different optimization methods
        for method in ['L-BFGS-B', 'TNC']:
            try:
                res = minimize(obj, h_g, method=method, 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        return h_refined, c5_refined
            except:
                continue
        return h_g, evaluate_erdos_solution(h_g, dx)
    
    # Try different starting points and refinement strategies
    attempts = 0
    max_attempts = 200
    while attempts < max_attempts and (time() - start_time) < budget_s * 0.9:
        attempts += 1
        
        # Try multiple random perturbations of best solution
        h_perturbed = best_h.copy()
        perturb_magnitude = 0.05
        perturb = (np.random.random(n_points) - 0.5) * perturb_magnitude
        h_perturbed += perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_sum = np.sum(h_perturbed)
        if h_sum > 0:
            h_perturbed = h_perturbed * (n_points / 2) / h_sum
        
        # Re-optimize local region
        try:
            h_local, c5_local = improved_local_search(h_perturbed, best_c5 + 0.01)
            if c5_local < best_c5:
                best_h = h_local
                best_c5 = c5_local
        except:
            pass
            
        # Try feasibility approach with tighter bounds
        c_guess = best_c5 * 0.99  # Try a slightly better C5
        h_feasible = solve_feasibility(best_h, c_guess)
        if h_feasible is not None:
            h_feasible = np.clip(h_feasible, 0, 1)
            h_sum = np.sum(h_feasible)
            if h_sum > 0:
                h_feasible = h_feasible * (n_points / 2) / h_sum
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
        
        # Occasionally try different random initialization
        if attempts % 20 == 0 and (time() - start_time) < budget_s * 0.8:
            h_new = np.random.random(n_points) * 0.8 + 0.1
            h_sum = np.sum(h_new)
            if h_sum > 0:
                h_new = h_new * (n_points / 2) / h_sum
            h_refined_new, c5_new = improved_local_search(h_new, best_c5 + 0.01)
            if c5_new < best_c5:
                best_h = h_refined_new
                best_c5 = c5_new

    # Final improvement using consistent optimization approach
    try:
        h_final, c5_final = improved_local_search(best_h, best_c5)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
