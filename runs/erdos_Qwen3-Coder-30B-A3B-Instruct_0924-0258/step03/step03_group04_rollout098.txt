import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def frank_wolfe_step(h, n_points, dx, max_iter=100, eta=0.5):
    """Frank-Wolfe based optimization for the Erdős problem"""
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for iteration in range(max_iter):
        # Find the shift that maximizes overlap
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr) - (n_points - 1)
        
        # Generate the linearized objective for this shift k
        # We want to minimize sum(h_diff * (1 - h_shifted_k))
        h_shifted_k = np.roll(1 - h, k_max)
        h_diff = h_shifted_k - h
        h_diff = h_diff[:n_points - abs(k_max)] if k_max > 0 else h_diff[-(n_points + k_max):]
        
        # Find extreme point: binary h with exactly n_points/2 ones, chosen to minimize
        # linear objective with weights from h_diff centered at k_max
        # Simple heuristic: pick top n_points/2 indices with largest (1 - h_shifted_k)
        if k_max > 0:
            weights = (1 - h[k_max:]) - h[:-k_max]
        else:
            weights = (1 - h[:k_max]) - h[-k_max:]
            
        # Binary selection
        indices = np.argsort(weights)[::-1][:n_points // 2]
        h_extreme = np.zeros(n_points)
        h_extreme[indices] = 1.0
        
        # Normalize to sum to n_points/2
        h_extreme = h_extreme * (n_points / 2) / np.sum(h_extreme)
        
        # Frank-Wolfe update
        h_new = (1 - eta) * h + eta * h_extreme
        h_new = project_to_feasible(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new
    
    return best_h, best_c5

    # Use initial construction if available

        # Simple initialization: sigmoid-like shape with sum=17.5
        x = np.linspace(0, 2, n_points)
        h_start = 1.0 / (1.0 + np.exp(-5 * (x - 1)))  # Sigmoid centered at x=1
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe based optimization
    remaining_time = budget_s
    iterations = 0
    
    while remaining_time > 10 and iterations < 2000:
        h_new, c5_new = frank_wolfe_step(best_h, n_points, dx, max_iter=20)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        remaining_time = budget_s - (time() - start_time)
        iterations += 1
        
        # Occasionally perturb to escape local minima
        if iterations % 50 == 0 and remaining_time > 20:
            # Add small perturbation
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturbed = best_h + perturb
            h_perturbed = project_to_feasible(h_perturbed, n_points)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
    
    # Final refinement with scipy
    if remaining_time > 20:
        def obj(h):
            return evaluate_erdos_solution(project_to_feasible(h, n_points), dx)
        
        res = minimize(obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 200})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
