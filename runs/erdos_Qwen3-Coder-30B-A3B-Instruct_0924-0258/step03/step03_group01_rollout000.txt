import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a well-structured initial guess
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        
        # Place two clusters
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Position clusters to minimize overlap
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Main loop with multiple strategies
    max_iterations = 100
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Nelder-Mead optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Ensure feasibility
        def project_feasible(h):
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points/2) / h_sum
            return h
        
        # Create initial simplex with diverse points
        simplex_size = min(10, n_points + 1)
        simplex = [h_start.copy()]
        for _ in range(simplex_size - 1):
            h_try = h_start + np.random.normal(0, 0.05, n_points)
            simplex.append(project_feasible(h_try))
        
        # Nelder-Mead-like optimization
        for _ in range(10):
            # Sort by objective
            values = [objective(h) for h in simplex]
            indices = np.argsort(values)
            simplex = [simplex[i] for i in indices]
            values = [values[i] for i in indices]
            
            if values[0] < best_c5:
                best_h = simplex[0].copy()
                best_c5 = values[0]
            
            # Simplex operations
            centroid = np.mean(simplex[:-1], axis=0)
            
            # Reflection
            xr = centroid + (centroid - simplex[-1])
            xr = project_feasible(xr)
            xr_val = objective(xr)
            
            if values[0] <= xr_val < values[-2]:
                simplex[-1] = xr
            elif xr_val < values[0]:
                # Expansion
                xe = centroid + 2 * (xr - centroid)
                xe = project_feasible(xe)
                xe_val = objective(xe)
                if xe_val < xr_val:
                    simplex[-1] = xe
                else:
                    simplex[-1] = xr
            else:
                # Contraction
                xc = centroid + 0.5 * (simplex[-1] - centroid)
                xc = project_feasible(xc)
                xc_val = objective(xc)
                if xc_val < values[-1]:
                    simplex[-1] = xc
                else:
                    # Shrink simplex
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
        
        # Local refinement using SLSQP
        remaining_time = budget_s - (time() - start_time)
        if remaining_time > 5:
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            for _ in range(3):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                h_restart = simplex[0] + np.random.normal(0, 0.02, n_points)
                h_restart = project_feasible(h_restart)
                
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 20, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
