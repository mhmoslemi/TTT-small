import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a periodic binary pattern with exactly n_points/2 ones
        h_start = np.zeros(n_points)
        h_start[::2] = 1
        if n_points % 2 == 0:
            h_start[1] = 1
        h_start = project_to_feasible(h_start, n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe algorithm
    max_iter = 10000
    eta = 0.5  # Step size
    perturb_rate = 0.1  # Rate of random perturbation
    
    h = h_start.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the maximum overlap shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_shift = np.argmax(corr)
        # Compute shift for maximum overlap
        shift_k = max_shift - (n_points - 1)
        
        # Compute the linear approximation
        h_shifted = np.roll(1 - h, shift_k)
        # Compute linear approximation coefficients
        linear_term = h_shifted
        
        # For a binary extreme point, select top candidates
        # We want to minimize sum(h_diff * (1 - h_shifted_k))
        # This is equivalent to maximizing sum(h_shifted_k)
        # where h_diff is the direction of desired change
        # Instead, we select the top n_points/2 indices where h_shifted_k is largest
        idxs = np.argsort(-linear_term)
        h_extreme = np.zeros(n_points)
        top_indices = idxs[:n_points//2]
        h_extreme[top_indices] = 1.0
        
        # Frank-Wolfe update
        h = (1 - eta) * h + eta * h_extreme
        h = project_to_feasible(h, n_points)
        
        # Periodic perturbation
        if i % 100 == 0 and np.random.random() < perturb_rate:
            h = project_to_feasible(h + np.random.normal(0, 0.01, n_points), n_points)
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def final_obj(h):
            h_proj = project_to_feasible(h, n_points)
            return evaluate_erdos_solution(h_proj, dx)
        
        res = minimize(final_obj, best_h, method='L-BFGS-B',
                       bounds=[(0,1) for _ in range(n_points)],
                       options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x, n_points)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final.copy()
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
