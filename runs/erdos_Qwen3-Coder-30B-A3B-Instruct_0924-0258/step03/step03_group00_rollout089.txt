import numpy as np
from scipy.optimize import minimize
import time
from copy import deepcopy

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Default construction with 35 points
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameterize by breakpoints and weights
    def h_from_breakpoints(breaks, weights):
        h = np.zeros(n_points)
        for i in range(len(breaks)-1):
            start_idx = int(breaks[i] / dx)
            end_idx = int(breaks[i+1] / dx)
            if start_idx < n_points and end_idx <= n_points:
                h[start_idx:end_idx] = weights[i]
        return h
    
    def breakpoints_and_weights_to_h(breaks, weights):
        h = np.zeros(n_points)
        for i in range(len(breaks)-1):
            start_idx = int(breaks[i] / dx)
            end_idx = int(breaks[i+1] / dx)
            if start_idx < n_points and end_idx <= n_points:
                h[start_idx:end_idx] = weights[i]
        return h
    
    # Initialize with a structured approach
    def initialize_population(n_individuals, n_breaks=5):
        populations = []
        for _ in range(n_individuals):
            weights = np.random.choice([0, 1], size=n_breaks)
            breaks = np.linspace(0, 2, n_breaks+1)
            # Perturb slightly
            for i in range(1, n_breaks):
                breaks[i] += np.random.normal(0, 0.05)
            breaks = np.clip(breaks, 0, 2)
            breaks = np.sort(breaks)
            populations.append((breaks.copy(), weights.copy()))
        return populations
    
    def compute_c5_from_breakpoints(breaks, weights):
        h = breakpoints_and_weights_to_h(breaks, weights)
        return evaluate_erdos_solution(h, dx)
    
    # Genetic Algorithm
    n_individuals = 500
    max_generations = 100
    mutation_rate = 0.05
    crossover_rate = 0.8
    
    # Create initial population
    pop = initialize_population(n_individuals)
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for breaks, weights in pop:
            c5 = compute_c5_from_breakpoints(breaks, weights)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        pop = [pop[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elite selection
        elite = pop[:n_individuals//10]
        if fitness[0] < best_c5:
            best_h = breakpoints_and_weights_to_h(pop[0][0], pop[0][1])
            best_c5 = fitness[0]
        
        # Generate new population
        new_pop = elite.copy()
        
        # Use tournaments to select parents
        while len(new_pop) < n_individuals:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_pop) < n_individuals-1:
                parents = np.random.choice(len(elite), 2, replace=False)
                parent1 = elite[parents[0]]
                parent2 = elite[parents[1]]
                
                # SBX-like crossover for breakpoints
                n_breaks = len(parent1[0]) - 1
                if n_breaks >= 2:
                    child1_breaks = parent1[0].copy()
                    child2_breaks = parent2[0].copy()
                    
                    # Crossover at 1 point
                    point = np.random.randint(1, n_breaks)
                    for i in range(point, n_breaks):
                        child1_breaks[i] = (parent1[0][i] + parent2[0][i]) / 2
                        child2_breaks[i] = (parent1[0][i] + parent2[0][i]) / 2
                    
                    # Mutation
                    for i in range(1, len(child1_breaks)-1):
                        if np.random.rand() < mutation_rate:
                            child1_breaks[i] += np.random.normal(0, 0.05)
                        if np.random.rand() < mutation_rate:
                            child2_breaks[i] += np.random.normal(0, 0.05)
                    
                    child1_breaks = np.clip(child1_breaks, 0, 2)
                    child2_breaks = np.clip(child2_breaks, 0, 2)
                    child1_breaks = np.sort(child1_breaks)
                    child2_breaks = np.sort(child2_breaks)
                    
                    # Apply mutations to weights too
                    child1_weights = parent1[1].copy()
                    child2_weights = parent2[1].copy()
                    for i in range(len(child1_weights)):
                        if np.random.rand() < 0.05:
                            child1_weights[i] = 1 - child1_weights[i]
                        if np.random.rand() < 0.05:
                            child2_weights[i] = 1 - child2_weights[i]
                    
                    new_pop.append((child1_breaks, child1_weights))
                    new_pop.append((child2_breaks, child2_weights))
                else:
                    # Simple copy if too short
                    new_pop.append(parent1)
                    new_pop.append(parent2)
            else:
                # Random initialization
                new_pop.append(initialize_population(1)[0])
                
        pop = new_pop[:n_individuals]
    
    # Local refinement
    if time.time() - start_time < budget_s * 0.95:
        def objective_function(params):
            breaks = np.array(params[:len(params)//2])
            weights = np.array(params[len(params)//2:])
            breaks = np.clip(breaks, 0, 2)
            breaks = np.sort(breaks)
            # Ensure weights are binary
            weights = np.round(weights).astype(int)
            weights = np.clip(weights, 0, 1)
            h = breakpoints_and_weights_to_h(breaks, weights)
            # Normalize area
            area = np.sum(h) * dx
            if area > 0:
                h = h * (n_points / 2) / area
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(params):
            breaks = params[:len(params)//2]
            weights = params[len(params)//2:]
            breaks = np.clip(breaks, 0, 2)
            breaks = np.sort(breaks)
            weights = np.round(weights).astype(int)
            weights = np.clip(weights, 0, 1)
            h = breakpoints_and_weights_to_h(breaks, weights)
            area = np.sum(h) * dx
            return area - 1.0
            
        # Refine best solution found from GA
        if len(pop) > 0:
            best_breaks, best_weights = pop[0]
            # Flatten the parameters
            initial_params = np.concatenate([best_breaks, best_weights.astype(float)])
            
            bounds = [(0, 2) for _ in range(len(initial_params)//2)] + [(0, 1) for _ in range(len(initial_params)//2)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            # Local optimization
            try:
                res = minimize(objective_function, initial_params, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    refined_breaks = res.x[:len(res.x)//2]
                    refined_weights = np.round(res.x[len(res.x)//2:]).astype(int)
                    refined_breaks = np.clip(refined_breaks, 0, 2)
                    refined_breaks = np.sort(refined_breaks)
                    refined_weights = np.clip(refined_weights, 0, 1)
                    refined_h = breakpoints_and_weights_to_h(refined_breaks, refined_weights)
                    area = np.sum(refined_h) * dx
                    if area > 0:
                        refined_h = refined_h * (n_points / 2) / area
                    c5_new = evaluate_erdos_solution(refined_h, dx)
                    if c5_new < best_c5:
                        best_h = refined_h
                        best_c5 = c5_new
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
