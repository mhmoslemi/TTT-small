import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        # Create structured initialization
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]  # Positions to place clusters
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Nelder-Mead optimization with restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        # Adjust to satisfy integral constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            scale = (n_points/2) / current_sum
            h = h * scale
        return h
    
    # Generate population
    def create_population(size):
        pop = [project_to_feasible(h_start.copy())]
        for _ in range(size-1):
            h = h_start + np.random.normal(0, 0.02, n_points)
            pop.append(project_to_feasible(h))
        return pop
    
    # Nelder-Mead simplex
    def nelder_mead_step(pop, values, alpha=1.0, gamma=2.0, rho=0.5, sigma=0.5):
        # Sort population by objective values
        idx = np.argsort(values)
        pop = [pop[i] for i in idx]
        values = [values[i] for i in idx]
        
        # Centroid excluding worst
        centroid = np.mean(pop[:-1], axis=0)
        
        # Reflection
        xr = centroid + alpha * (centroid - pop[-1])
        xr = project_to_feasible(xr)
        fr = objective(xr)
        
        if values[0] <= fr < values[-2]:
            pop[-1] = xr
            values[-1] = fr
        elif fr < values[0]:
            # Expansion
            xe = centroid + gamma * (xr - centroid)
            xe = project_to_feasible(xe)
            fe = objective(xe)
            if fe < fr:
                pop[-1] = xe
                values[-1] = fe
            else:
                pop[-1] = xr
                values[-1] = fr
        else:
            # Contraction
            if fr < values[-1]:
                xc = centroid + rho * (xr - centroid)
            else:
                xc = centroid + rho * (pop[-1] - centroid)
            xc = project_to_feasible(xc)
            fc = objective(xc)
            if fc < values[-1]:
                pop[-1] = xc
                values[-1] = fc
            else:
                # Shrink
                for i in range(1, len(pop)):
                    pop[i] = pop[0] + sigma * (pop[i] - pop[0])
                    pop[i] = project_to_feasible(pop[i])
                values = [objective(h) for h in pop]
        
        return pop, values
    
    # Main optimization loop
    max_iterations = 100
    patience = 0
    best_c5_history = []

        if time() - start_time > budget_s * 0.95:
            break

        values = [objective(h) for h in pop]
        best_c5_iter = min(values)
        best_c5_history.append(best_c5_iter)

            idx = np.argmin(values)
            best_h = pop[idx].copy()
            best_c5 = best_c5_iter
            patience = 0
        else:
            patience += 1
            
        # Early stopping if no improvement
        if patience > 5:
            break
            
        # Run simplex iterations

                break
            pop, values = nelder_mead_step(pop, values)
            if values[0] < best_c5:
                best_h = pop[0].copy()
                best_c5 = values[0]
                patience = 0
                best_c5_history = []
                
    # Final gradient refinement of best solution
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
