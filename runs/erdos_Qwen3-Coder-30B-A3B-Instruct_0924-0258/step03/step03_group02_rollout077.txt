import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = np.random.rand(n_particles, n_points) * 0.5 + 0.25  # [0.25, 0.75]
    particles = np.array([project_to_feasible(p, n_points) for p in particles])
    velocities = (np.random.rand(n_particles, n_points) - 0.5) * 0.1  # [-0.05, 0.05]
    
    # Initialize personal bests
    p_best_positions = particles.copy()
    p_best_values = np.array([evaluate_erdos_solution(p, dx) for p in particles])
    
    # Initialize global best
    global_best_idx = np.argmin(p_best_values)
    g_best_position = p_best_positions[global_best_idx].copy()
    g_best_value = p_best_values[global_best_idx]
    
    if g_best_value < best_c5:
        best_h = g_best_position.copy()
        best_c5 = g_best_value
    
    # PSO main loop
    iter_no_improve = 0
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * i / max_iter
        
        for j in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v_new = (w * velocities[j] + 
                     c1 * r1 * (p_best_positions[j] - particles[j]) +
                     c2 * r2 * (g_best_position - particles[j]))
            
            # Clip velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            pos_new = particles[j] + v_new
            
            # Project to feasible region
            pos_new = project_to_feasible(pos_new, n_points)
            
            # Evaluate
            c5_new = evaluate_erdos_solution(pos_new, dx)
            
            # Update personal best
            if c5_new < p_best_values[j]:
                p_best_values[j] = c5_new
                p_best_positions[j] = pos_new.copy()
                
                # Update global best
                if c5_new < g_best_value:
                    g_best_value = c5_new
                    g_best_position = pos_new.copy()
                    iter_no_improve = 0
                    if c5_new < best_c5:
                        best_h = pos_new.copy()
                        best_c5 = c5_new
            else:
                particles[j] = pos_new.copy()
                velocities[j] = v_new.copy()
        
        iter_no_improve += 1
        if iter_no_improve > 100:
            # Increase exploration
            w = 0.8
            
    # Post-processing: refine with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        h_refined = best_h.copy()
        
        # Define optimization problem
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def con(h):
            return np.sum(h) - n_points / 2
        
        # Use bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con}
        
        try:
            res = minimize(obj, h_refined, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_final = project_to_feasible(res.x, n_points)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
