import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with coarse resolution and refine if time permits
    for resolution in [35, 70]:
        if time() - start_time > budget_s * 0.95:
            break
            
        n_points = resolution
        dx = 2.0 / n_points
        
        # Initialize with best known solution if available
        h_current = best_h.copy() if resolution == 35 else h_current.copy()
        h_current = h_current[:n_points]
        if len(h_current) < n_points:
            h_current = np.pad(h_current, (0, n_points - len(h_current)), 'constant', constant_values=0.5)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Binary search parameters
        c_low = 0.37
        c_high = 0.38085
        max_iterations = 10
        
        # Try to find feasible solution with binary search
        for iteration in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            c_mid = (c_low + c_high) / 2.0
            
            # Create CVX optimization problem
            h = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                h >= 0,
                h <= 1,
                cp.sum(h) * dx == 1
            ]
            
            # Add overlap constraints
            for k in range(n_points):
                if k == 0:
                    continue
                valid_len = n_points - k
                if valid_len <= 0:
                    break
                overlap_constraint = cp.sum(cp.multiply(h[:valid_len], 1 - h[k:k+valid_len])) * dx <= c_mid
                constraints.append(overlap_constraint)
            
            # Solve
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.GLPK, max_iters=2000, feastol=1e-6)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_feasible = h.value
                    h_feasible = np.clip(h_feasible, 0, 1)
                    # Renormalize
                    h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
                    c_low = c_mid
                else:
                    c_high = c_mid
            except Exception:
                c_high = c_mid
                
        # Refinement with local optimization
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try multiple random restarts for better optimization
        for _ in range(3):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb solution for random restart
            h_perturb = best_h + np.random.normal(0, 0.05, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            
            try:
                res = minimize(objective, h_perturb, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = np.clip(res.x, 0, 1)
                    h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
