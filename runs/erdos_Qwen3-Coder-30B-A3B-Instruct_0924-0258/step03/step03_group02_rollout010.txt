import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős overlap integral for given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimizer settings
    max_iter = 50
    opt_tolerance = 1e-6
    
    # Gradient-based local refinement
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start from current best
        h_current = best_h.copy()
        
        # Objective function for minimization
        def objective(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
            
        # Constraint function
        def constraint_sum(h_vals):
            return np.sum(h_vals) - n_points / 2
            
        # Bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Local optimization using SLSQP
            result = minimize(objective, h_current, method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 30, 'ftol': opt_tolerance, 'gtol': opt_tolerance})
            
            if result.success:
                h_new = project_to_feasible(result.x, n_points)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Attempt a different search idea:
    # Random sampling refined with SLSQP
    for _ in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        # Generate random feasible point
        h_random = np.random.random(n_points)
        h_random = project_to_feasible(h_random, n_points)
        # Refine
        def obj(h_vals):
            return evaluate_erdos_solution(h_vals, dx)
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda h_vals: np.sum(h_vals) - n_points / 2}
        try:
            res = minimize(obj, h_random, method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': opt_tolerance})
            if res.success:
                h_refined = project_to_feasible(res.x, n_points)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
