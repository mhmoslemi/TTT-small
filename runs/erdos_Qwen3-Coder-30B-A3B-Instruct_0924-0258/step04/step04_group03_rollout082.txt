import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    n_iter = 1000
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.ones(n_points) * 0.5
        # Random perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Track bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p, dx) for p in particles]
    global_best = best_h.copy()
    global_best_score = best_c5
    
    # PSO main loop
    for iter_no in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iter_no / n_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = (c1 * r1 * (personal_best[i] - particles[i]) + 
                         c2 * r2 * (global_best - particles[i]))
            velocities[i] = w * velocities[i] + vel_update
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to valid region
            particles[i] = np.clip(particles[i], 0, 1)
            # Maintain constraint sum(h) = n_points/2
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = particles[i].copy()
                global_best_score = score
                
                # Early update of best result
                if global_best_score < best_c5:
                    best_h = global_best.copy()
                    best_c5 = global_best_score
    
    # Refinement with convex optimization
    if time() - start_time < budget_s * 0.90:
        try:
            h_refine = cp.Variable(n_points)
            constraints = [
                cp.sum(h_refine) == n_points / 2.0,
                h_refine >= 0,
                h_refine <= 1
            ]
            
            # Add some overlap constraints
            corr_vals = np.correlate(global_best, 1 - global_best, mode='full')
            max_corr = np.max(corr_vals)
            
            for k in range(0, 2 * n_points - 1, 2):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_refine[:n_points-k], 1 - h_refine[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_refine[k-n_points+1:], 1 - h_refine[:n_points-(k-n_points+1)]))
                # Soft constraint around max observed value
                constraints.append(corr_sum <= max_corr * 1.05 * dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_refined = h_refine.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final SLSQP refinement if solvable
    if time() - start_time < budget_s * 0.8:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Start from refined solution
        h_init = best_h.copy()
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final validation
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / final_sum
    
    return (list(best_h), best_c5, n_points)
