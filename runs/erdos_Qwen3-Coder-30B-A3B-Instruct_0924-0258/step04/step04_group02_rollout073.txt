import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iter = int(1e5)
    T = T_initial
    delta_max = 0.2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Compute allowable change
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Adjust values while maintaining the sum
        h_candidate = best_h.copy()
        
        # Ensure the adjustments keep values in bounds
        if h_candidate[i] + delta > 1:
            delta = 1 - h_candidate[i]
        elif h_candidate[i] + delta < 0:
            delta = -h_candidate[i]
            
        delta_j = -delta
        
        h_candidate[i] += delta
        h_candidate[j] += delta_j
        
        # Normalize to sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Compute objective for candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on simulated annealing
        if candidate_c5 < best_c5 or np.random.rand() < np.exp(-(candidate_c5 - best_c5) / T):
            best_h = h_candidate.copy()
            best_c5 = candidate_c5
            
        # Cool down the temperature
        T *= alpha
        
        # Optional: decrease delta_max as T decreases to fine-tune later
        delta_max = max(0.001, delta_max * alpha)
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
