import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define optimized gradient-based optimizer
    def optimize_h(h_init, max_iter=500):
        h = h_init.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Simple gradient descent with adaptive step
        lr = 1e-3
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute objective value
            c5 = evaluate_erdos_solution(h, dx)
            
            # Gradient approximation via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = evaluate_erdos_solution(h_minus, dx)
                
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Check for improvement
            new_c5 = evaluate_erdos_solution(h, dx)
            if new_c5 < c5:
                h = h.copy()  # update
                c5 = new_c5
                if c5 < best_c5:
                    best_h[:] = h[:]
                    best_c5 = c5
                if c5 < 0.3807:
                    break
            
            # Adaptive learning rate decay
            if i % 50 == 0 and i > 0:
                lr *= 0.9
            
            # Noise injection to escape local minima
            if i % 100 == 0 and i > 0:
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        return h, best_c5
    
    # Run multiple restarts from diverse starting points
    restarts = 5
    for r in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate diversified starting point
        h_restart = h_start.copy() + np.random.normal(0, 0.05, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        h_opt, c5_opt = optimize_h(h_restart, max_iter=1000)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Final refinement with scipy
    p0 = best_h.copy()
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def cons(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': cons}
    
    try:
        res = minimize(obj, p0, method='SLSQP', bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
