import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_old, n_new):
    """Linearly resample a function from old grid to new grid"""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def project_to_constraints(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction or create one
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution grid setup
    n_coarse = 50
    n_fine = 100
    
    # Coarse grid version
    h_coarse = linear_resample(h_start, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    
    # Fine grid version
    h_fine = linear_resample(h_start, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Get bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Global coarse search phase
    coarse_steps = 100
    perturb_scale = 0.03
    for step in range(coarse_steps):
        if time() - start_time > budget_s * 0.6:
            break
            
        # Perturb solution
        h_perturbed = h_coarse + np.random.normal(0, perturb_scale, n_coarse)
        h_perturbed = project_to_constraints(h_perturbed, n_coarse)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_perturbed, dx_coarse)
        
        # Accept if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = linear_resample(h_perturbed, n_points)
            
        # Update perturbation scale (decay)
        perturb_scale *= 0.995
    
    # Refinement phase
    fine_steps = 200
    perturb_scale = 0.03
    
    for step in range(fine_steps):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start from coarse solution interpolated to fine grid
        h_init = linear_resample(h_coarse, n_fine)
        
        # Perturb
        h_perturbed = h_init + np.random.normal(0, perturb_scale, n_fine)
        h_perturbed = project_to_constraints(h_perturbed, n_fine)
        
        # Local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx_fine)
        
        def constraint_sum(h):
            return np.sum(h) - n_fine / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP',
                           bounds=[(0, 1) for _ in range(n_fine)],
                           constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    # Interpolate back to n_points resolution
                    best_h = linear_resample(h_refined, n_points)
                    # Ensure constraint is applied
                    best_h = project_to_constraints(best_h, n_points)
        except:
            pass
            
        # Update perturbation scale (decay)
        perturb_scale *= 0.995

    # Final verification
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        # Final boost optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass

    return (list(best_h), best_c5, n_points)
