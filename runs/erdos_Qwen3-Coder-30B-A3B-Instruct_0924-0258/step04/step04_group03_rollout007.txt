import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_constraints(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_bests = []
    personal_best_values = []
    
    for _ in range(swarm_size):
        h = h_start.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_constraints(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_bests.append(h.copy())
        personal_best_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_bests[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - swarm[i]) +
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to constraints
            swarm[i] = project_to_constraints(swarm[i], n_points)
            
            # Evaluate
            current_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_bests[i] = swarm[i].copy()
                personal_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = current_value
                if global_best_value < best_c5:
                    best_h = global_best.copy()
                    best_c5 = global_best_value
    
    # Convex refinement with ECOS
    if time() - start_time < budget_s * 0.9:
        try:
            h_refined = cp.Variable(n_points)
            constraints = [
                cp.sum(h_refined) == n_points / 2.0,
                h_refined >= 0,
                h_refined <= 1
            ]
            # Add additional overlap constraints in a simplified way to avoid 
            # over-constraining while still enabling convergence
            # Fix the relaxation: make it an optimization problem
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_sol = h_refined.value
                c5_sol = evaluate_erdos_solution(h_sol, dx)
                if c5_sol < best_c5:
                    best_h = h_sol
                    best_c5 = c5_sol
        except Exception:
            pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10 and len(best_h) > 0:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final validation
    best_h = project_to_constraints(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
