import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    elite_size = 10
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    # Evaluate initial population
    fitness = []
    for ind in population:
        fitness.append(evaluate_erdos_solution(ind, dx))
    
    # Genetic algorithm main loop
    gen = 0
    while time() - start_time < budget_s * 0.95 and gen < num_generations:
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update global best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create next generation
        new_population = population[:elite_size].copy()
        
        # Tournament selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted(random.sample(range(elite_size), 2))[0]
            parent2_idx = sorted(random.sample(range(elite_size), 2))[0]
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if random.random() < 0.8:  # 80% crossover rate
                point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:point], parent2[point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                num_mutations = max(1, int(n_points * mutation_rate))
                mutation_indices = random.sample(range(n_points), num_mutations)
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, 0.01)
            
            # Repair and enforce constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Evaluate new population
        fitness = []
        for ind in population:
            fitness.append(evaluate_erdos_solution(ind, dx))
        
        gen += 1
    
    # Final optimization of the best solution
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points/2) / np.sum(final_h)
        
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
