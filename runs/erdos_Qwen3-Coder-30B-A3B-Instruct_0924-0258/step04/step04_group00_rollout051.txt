import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Multi-resolution approach
    resolutions = [35, 40, 50]
    current_res = 0
    
    while time() - start_time < budget_s * 0.95 and current_res < len(resolutions):
        res = resolutions[current_res]
        if res > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), best_h)
            n_points = res
            dx = 2.0 / n_points
        else:
            h_new = best_h
            n_points = res
            dx = 2.0 / n_points
            
        # Local optimization with random restarts
        max_restarts = 20
        restart_count = 0
        h_current = h_new.copy()
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_current + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res_opt = minimize(objective, h_restart, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 200, 'ftol': 1e-8})
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0  # Reset counter on improvement
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
        
        current_res += 1
    
    # Final optimization at highest resolution
    if time() - start_time < budget_s * 0.9:
        try:
            # Projective refinement
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
