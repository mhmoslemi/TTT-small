import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Initialize dual variables
    lambda_k = np.ones(2 * n_points - 1)
    
    # Main loop
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        current_overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        
        # Update dual variables
        # We'll use a simple projection-shrinkage approach for duals
        # But instead use a gradient-like update based on past performance
        # to emphasize latest overlaps more
        max_idx = np.argmax(current_overlaps)
        current_overlaps = np.array(current_overlaps)
        lambda_k = np.maximum(lambda_k * (current_overlaps / lambda_k), 1e-6)
        
        # Form and solve convex problem with current duals
        h = cp.Variable(n_points)
        
        # Objective: minimize sum of lambda_k * C_k(h)
        objective = cp.Minimize(
            cp.sum(
                cp.multiply(
                    lambda_k,
                    [cp.sum(cp.multiply(h, cp.roll(1 - h, k))) for k in range(1 - n_points, n_points)]
                )
            )
        )
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_new = h.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    current_c5 = evaluate_erdos_solution(h_new, dx)
                    if current_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = current_c5
        except:
            pass
        
        # Occasionally try local optimization
        if iteration % 3 == 0:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_local = res.x
                    local_c5 = evaluate_erdos_solution(h_local, dx)
                    if local_c5 < best_c5:
                        best_h = h_local
                        best_c5 = local_c5
            except:
                pass
                
    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
