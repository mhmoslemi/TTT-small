import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization starting with initial solution or a random one
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Use a more refined optimization approach
    max_iters = 1000
    iter_count = 0
    
    # Start with a better guess  
    h_current = best_h.copy()
    
    # Gradient descent with adaptive learning rate
    lr = 1e-3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h_current)
    v = np.zeros_like(h_current)
    
    # Early stopping flag
    no_improvement_count = 0
    patience = 50
    
    while iter_count < max_iters and (time() - start_time) < budget_s * 0.95:
        # Compute gradient approximation via finite differences
        eps_val = 1e-4
        grad = np.zeros_like(h_current)
        f_current = objective(h_current)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps_val
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            f_plus = objective(h_plus)
            grad[i] = (f_plus - f_current) / eps_val
        
        # Adam optimization step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        
        # Update parameters
        h_current = h_current - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Enforce bounds and normalization
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Evaluate and update best solution
        c5_current = objective(h_current)
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early stopping if no improvement for a while
        if no_improvement_count > patience:
            break
        
        iter_count += 1
        
        # Occasionally inject noise to escape local minima
        if iter_count % 100 == 0 and iter_count > 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_current += noise
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Final refinement using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            for _ in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                h_restart = best_h + np.random.normal(0, 0.02, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                res = minimize(objective, h_restart, method='SLSQP', 
                             bounds=bounds, constraints=cons_sum, 
                             options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
