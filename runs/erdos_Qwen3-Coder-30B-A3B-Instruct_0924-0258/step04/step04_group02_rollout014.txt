import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for optimization
    max_iter = 500
    learning_rate = 0.01
    reg_strength = 1e-4
    
    # Initialize coefficients for Fourier representation
    # We'll parameterize h as sigmoid of a Fourier series
    # Using n_coefficients = n_points // 2 + 1 for real transform
    n_coefficients = n_points // 2 + 1
    a = np.random.normal(0, 0.1, n_coefficients)
    b = np.random.normal(0, 0.1, n_coefficients-1)  # b_0 = 0 for real-valued function
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Construct h from coefficients
        x = np.linspace(0, 2, n_points)
        # Fourier series
        h_series = a[0] * np.ones_like(x)
        for k in range(1, n_coefficients):
            h_series += a[k] * np.cos(k * np.pi * x)
            if k < len(b):
                h_series += b[k-1] * np.sin(k * np.pi * x)
        # Apply sigmoid
        h = 1 / (1 + np.exp(-h_series))
        # Scale to sum to 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate objective
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Compute gradients using finite difference approximation
        eps = 1e-5
        grad_a = np.zeros_like(a)
        grad_b = np.zeros_like(b)
        
        for k in range(n_coefficients):
            a_test = a.copy()
            a_test[k] += eps
            h_test = 1/(1 + np.exp(-(a_test[0]*np.ones_like(x) + 
                           np.sum([a_test[i] * np.cos(i*np.pi*x) for i in range(1,n_coefficients)], axis=0) + 
                           np.sum([b[i-1] * np.sin(i*np.pi*x) for i in range(1,len(b)+1)], axis=0))))
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_plus = evaluate_erdos_solution(h_test, dx)
            
            a_test = a.copy()
            a_test[k] -= eps
            h_test = 1/(1 + np.exp(-(a_test[0]*np.ones_like(x) + 
                           np.sum([a_test[i] * np.cos(i*np.pi*x) for i in range(1,n_coefficients)], axis=0) + 
                           np.sum([b[i-1] * np.sin(i*np.pi*x) for i in range(1,len(b)+1)], axis=0))))
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_minus = evaluate_erdos_solution(h_test, dx)
            
            grad_a[k] = (c5_plus - c5_minus) / (2*eps)
            
        for k in range(len(b)):
            b_test = b.copy()
            b_test[k] += eps
            h_test = 1/(1 + np.exp(-(a[0]*np.ones_like(x) + 
                           np.sum([a[i] * np.cos(i*np.pi*x) for i in range(1,n_coefficients)], axis=0) + 
                           np.sum([b_test[i-1] * np.sin(i*np.pi*x) for i in range(1,len(b_test)+1)], axis=0))))
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_plus = evaluate_erdos_solution(h_test, dx)
            
            b_test = b.copy()
            b_test[k] -= eps
            h_test = 1/(1 + np.exp(-(a[0]*np.ones_like(x) + 
                           np.sum([a[i] * np.cos(i*np.pi*x) for i in range(1,n_coefficients)], axis=0) + 
                           np.sum([b_test[i-1] * np.sin(i*np.pi*x) for i in range(1,len(b_test)+1)], axis=0))))
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_minus = evaluate_erdos_solution(h_test, dx)
            
            grad_b[k] = (c5_plus - c5_minus) / (2*eps)
        
        # Update coefficients with gradient descent
        a = a - learning_rate * (grad_a + reg_strength * a)
        b = b - learning_rate * (grad_b + reg_strength * b)
        
        # Normalize the learned coefficients
        h = 1 / (1 + np.exp(-(a[0]*np.ones_like(x) + 
                      np.sum([a[i] * np.cos(i*np.pi*x) for i in range(1,n_coefficients)], axis=0) + 
                      np.sum([b[i-1] * np.sin(i*np.pi*x) for i in range(1,len(b)+1)], axis=0))))
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Final refinement with local search
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine best solution found
        h_refined = best_h.copy()
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
