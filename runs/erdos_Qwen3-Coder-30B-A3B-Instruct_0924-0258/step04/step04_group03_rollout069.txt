import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient w.r.t. shift k
        grad = np.zeros_like(h)
        # For convolution h * (1-h) with shift k, the derivative w.r.t. h[i] is
        # 1 for positions where h[i+k] contributes to overlap
        # But need to consider that h is periodically extended (though correlation handled in numpy)
        # Just compute symbolically: for h[i] in the prefix of overlap (h * 1-h shifted by k),
        # its contribution is in the last k elements of h (when k < len(h))
        # Actually use the fact that compute gradient via the convolution operation
        
        # Instead of symbolic gradient, compute numerically using finite diff
        eps = 1e-5
        h_plus = h.copy()
        h_minus = h.copy()
        if k <= len(h):
            h_plus[k] += eps
            h_minus[k] -= eps
        # Clamp values to [0,1]
        h_plus = np.clip(h_plus, 0, 1)
        h_minus = np.clip(h_minus, 0, 1)
        # Fixed sum constraint
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
        
        overlap_plus = evaluate_erdos_solution(h_plus, dx)
        overlap_minus = evaluate_erdos_solution(h_minus, dx)
        grad[k] = (overlap_plus - overlap_minus) / (2 * eps)
        return grad
    
    # Gradient descent with adaptive learning rate
    h_current = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    max_iter = 5000
    
    # Track best solution
    best_h = h_current.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find k that maximizes overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient for this k
        grad = np.zeros_like(h_current)
        # Only compute non-zero components (simplified gradient approx)
        for i in range(len(h_current)):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            eps = 1e-6
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            overlap_plus = evaluate_erdos_solution(h_plus, dx)
            overlap_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Gradient step
        h_new = h_current - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < current_c5:
            h_current = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = new_c5
        else:
            # Decrease learning rate if no improvement
            alpha *= 0.99
            if alpha < 1e-8:
                alpha = 1e-8
                
        # Occasionally use convex refinement
        if iteration % 50 == 0 and iteration > 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                corr_full = np.correlate(h_current, 1 - h_current, mode='full')
                top_k_indices = np.argsort(corr_full)[-10:]
                for k in top_k_indices:
                    # Approximate individual overlap constraints for top k
                    if k >= n_points:
                        # pad k's contribution
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
