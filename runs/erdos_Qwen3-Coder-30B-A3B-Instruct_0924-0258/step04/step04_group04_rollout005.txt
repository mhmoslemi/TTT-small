import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a basic piecewise constant construction
        h_start[:n_points//2] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    h_current = h_start.copy()
    
    dx = 2.0 / n_points
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient descent parameters
    step_size = 0.01
    max_iter = 1000
    patience = 100
    patience_counter = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_shift_idx = np.argmax(corr)
        k_max = max_shift_idx - (n_points - 1)
        
        # Compute gradient for the max shift
        grad = np.zeros(n_points)
        if k_max == 0:
            # Gradient for shift 0
            for i in range(n_points):
                grad[i] = -h_current[i] - (1 - h_current[i])
        else:
            # For non-zero shift, gradient adjusts based on shift
            for i in range(n_points):
                j = i + k_max
                if j >= 0 and j < n_points:
                    grad[i] = -(1 - h_current[j]) - h_current[i]
                else:
                    grad[i] = -h_current[i]
        
        # Update using gradient descent
        h_new = h_current - step_size * grad
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Normalize
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if this improves the objective
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
            last_improvement = iteration
        else:
            patience_counter += 1
        
        # If no improvement in long time, reduce step size to converge
        if patience_counter > patience:
            step_size *= 0.9
            patience_counter = 0
            
        h_current = h_new
    
    # Final check with more precise optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Try some additional local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            from scipy.optimize import minimize
            
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
