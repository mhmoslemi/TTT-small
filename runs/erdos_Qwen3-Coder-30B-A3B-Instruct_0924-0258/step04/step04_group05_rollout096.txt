```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def normalize_sum(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def crossover(parent1, parent2):
    mask = np.random.random(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(individual, rate=0.1, sigma=0.05):
    mask = np.random.random(len(individual)) < rate
    individual = individual.copy()
    individual[mask] += np.random.normal(0, sigma, np.sum(mask))
    individual = np.clip(individual, 0, 1)
    return individual

def repair_solution(h, target_sum):
    h = np.clip(h, 0, 1)
    h = normalize_sum(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population_size = 40
    population = []
    fitness = []
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        for i in range(10):
            perturbed = initial_h + np.random.normal(0, 0.02, n_points)
            perturbed = repair_solution(perturbed, target_sum)
            population.append(perturbed)
    else:
        initial_h = np.ones(n_points) * 0.5
    
    # Add random valid solutions
    for i in range(10):
        h = np.random.random(n_points)
        h = repair_solution(h, target_sum)
        population.append(h)
    
    # Initialize with random solutions
    for _ in range(population_size - len(population)):
        h = np.random.random(n_points)
        h = repair_solution(h, target_sum)
        population.append(h)
    
    # Evaluate initial population
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 50
    max_stagnation = 50
    stagnation_count = 0
    
    # GA parameters
    crossover_prob = 0.7
    mutation_rate = 0.15
    
    # Genetic search loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        new_population = []
        new_fitness = []
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep best solutions
        elites = sorted_population[:5]
        elites_fitness = sorted_fitness[:5]
        
        # Fill remaining population with offspring
        while len(new_population) < population_size - len(elites):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Select parents
            if random.random() < crossover_prob:
                parent1_idx = random.randint(0, len(sorted_population)//2)
                parent2_idx = random.randint(0, len(sorted_population)//2)
                parent1 = sorted_population[parent1_idx]
                parent2 = sorted_population[parent2_idx]
                child = crossover(parent1, parent2)
            else:
                # Clone a random parent
                parent_idx = random.randint(0, len(sorted_population)//2)
                child = sorted_population[parent_idx].copy()
            
            # Mutation
            child = mutate(child, rate=mutation_rate)
            child = repair_solution(child, target_sum)
            
            # Local optimization for some children
            if random.random() < 0.1:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, child, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        child = res.x
                except:
                    pass
            else:
                child = repair_solution(child, target_sum)
            
            # Check for tabu
            if len(tabu_list) > 0:
                is_tabu = False
                for tabu in tabu_list:
                    if np.allclose(child, tabu, atol=1e-4):
                        is_tabu = True
                        break
                if not is_tabu:
                    new_population.append(child)
                    new_fitness.append(evaluate_erdos_solution(child, dx))
            else:
                new_population.append(child)
                new_fitness.append(evaluate_erdos_solution(child, dx))
        
        # Add elites
        for elite, elite_fit in zip(elites, elites_fitness):
            new_population.append(elite)
            new_fitness.append(elite_fit)
        
        # Update population
        population = new_population[:population_size]
        fitness = new_fitness[:population_size]
        
        # Check for improvement
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update tabu list
        if len(tabu_list) < tabu_tenure:
            tabu_list.append(best_h.copy())
        else:
            tabu_list.pop(0)
            tabu_list.append(best_h.copy())
        
        # Adaptive parameters
        if stagnation_count > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_tenure = max(10, tabu_tenure // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
            
        # Early stopping if target reached
        if best_c5 < 0.3807:
            break
    
    # Final refinement with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            # Refine best solution with more iterations
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
```
