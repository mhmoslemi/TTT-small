import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_old, n_new):
    # Linearly resample h_old to n_new points
    old_points = np.linspace(0, 2, len(h_old))
    new_points = np.linspace(0, 2, n_new)
    return np.interp(new_points, old_points, h_old)

def project_to_constraints(h, n_points):
    # Project to [0,1] and normalize sum to n_points/2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction or a default
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    # Setting up coarse and fine grids for alternating search
    n_coarse = 50
    n_fine = 100
    
    # Initialize coarse and fine representations
    h_coarse = linear_resample(h_init, n_coarse)
    h_fine = linear_resample(h_init, n_fine)
    
    # Project to constraints
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Best solution tracking
    best_h_coarse = h_coarse.copy()
    best_h_fine = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # To start with better initial solution
    current_best_h = best_h_fine.copy()
    current_best_c5 = best_c5_fine
    
    # Starting variables for alternating search
    iter_count = 0
    max_iters = 1000
    pert_scale = 0.03
    pert_decay = 0.99
    eval_interval = 10
    
    for i in range(max_iters):
        elapsed = time() - start_time
        if elapsed > budget_s * 0.98:
            break
            
        # Alternate between coarse and fine grids
        if i % 2 == 0:  # Coarse grid exploration
            h = h_coarse.copy()
            dx = dx_coarse
            n = n_coarse
        else:  # Fine grid refinement
            h = h_fine.copy()
            dx = dx_fine
            n = n_fine
            
        # Random perturbation with adaptive scale
        noise = np.random.normal(0, pert_scale, len(h))
        h = h + noise
        h = project_to_constraints(h, n)
        
        # Local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n)]
        
        try:
            res = minimize(objective, h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h = res.x
                c5_val = objective(h)
                if c5_val < current_best_c5:
                    current_best_c5 = c5_val
                    current_best_h = h.copy()
                    if c5_val < 0.3807:
                        break  # Early stopping if target achieved
        except:
            pass
        
        # Transfer solution to corresponding grid
        if i % 2 == 0:
            h_coarse = h.copy()
            c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
            if c5_coarse < best_c5_coarse:
                best_c5_coarse = c5_coarse
                best_h_coarse = h_coarse.copy()
        else:
            h_fine = h.copy()
            c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
            if c5_fine < best_c5_fine:
                best_c5_fine = c5_fine
                best_h_fine = h_fine.copy()
        
        # Adapt perturbation scale
        if i > 10 and i % 20 == 0:
            pert_scale *= pert_decay
            
        iter_count += 1
    
    # Final evaluation of best solution
    final_c5 = current_best_c5
    final_h = current_best_h
    
    return (list(final_h), final_c5, len(final_h))
