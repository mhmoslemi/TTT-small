import numpy as np
from cvxpy import *
import math
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a more structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Form a two-peak structure
        peak1_center = 0.5
        peak2_center = 1.5
        peak_width = 0.2
        for i in range(n_points):
            x = i * dx
            h_start[i] = max(0, 1.0 - abs(x - peak1_center) / peak_width)
            if abs(x - peak2_center) < peak_width:
                h_start[i] = max(h_start[i], 1.0 - abs(x - peak2_center) / peak_width)
        # Normalize to ensure sum is n/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Binary search for C5
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # To ensure we compute exact overlaps, we'll do a more precise evaluation
    def evaluate_c5_exact(h):
        dx = 2.0 / len(h)
        # Compute all overlap integrals for each k
        max_overlap = 0
        for k in range(len(h) * 2 - 1):
            overlap = 0
            for i in range(len(h)):
                j = i + k
                if j < len(h):
                    overlap += h[i] * (1 - h[j])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    dx = 2.0 / n_points
    
    # Start binary search
    iteration = 0
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        mid = (low + high) / 2.0
        
        # Try to find a solution using linear programming
        try:
            # Define the variable
            h = Variable(n_points)
            
            # Define constraints
            constraints = [h >= 0, h <= 1]
            
            # Sum constraint
            constraints.append(sum(h) == n_points/2)
            
            # Overlap constraints
            for k_idx in range(len(h) * 2 - 1):
                # Compute overlap for this lag
                overlap_expr = 0
                for i in range(len(h)):
                    j = i + k_idx
                    if j < len(h):
                        overlap_expr += h[i] * (1 - h[j])
                
                # Add constraint overlap_expr * dx <= mid
                constraints.append(overlap_expr * dx <= mid)
            
            # Problem
            prob = Problem(Minimize(0), constraints)
            prob.solve(solver=CVXOPT, verbose=False)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_result = h.value
                c5_result = evaluate_c5_exact(h_result)
                if c5_result < best_c5:
                    best_h = h_result.copy()
                    best_c5 = c5_result
                high = mid
            else:
                low = mid
                
        except:
            low = mid
            
        iteration += 1
    
    # Final check with direct computation
    best_c5 = evaluate_c5_exact(best_h)
    
    return (list(best_h), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap
