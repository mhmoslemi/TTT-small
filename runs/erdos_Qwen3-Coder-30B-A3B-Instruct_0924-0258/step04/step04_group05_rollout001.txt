import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Increase resolution to 40 points
    if n_points != 40:
        h_start = np.interp(np.linspace(0, 2, 40), np.linspace(0, 2, n_points), h_start)
        n_points = 40
    
    # Project to correct sum
    h_start = h_start * (n_points/2) / np.sum(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive grid search for better bounds
    grid_size = 100
    search_range = np.linspace(0.36, best_c5, grid_size)
    grid_step = max(1, len(search_range) // 20)  # Adaptive step size
    search_range = search_range[::grid_step]
    
    for c_mid in search_range:
        if time() - start_time > budget_s * 0.7:
            break
            
        # Check feasibility with CVXPY
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2.0,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add all overlap constraints
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_vars[k-n_points+1:], 1 - h_vars[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5_grid:
                    best_c5_grid = c5_feasible
                    best_h = h_feasible.copy()
        except:
            continue
    
    # Multiple restarts with different strategies for gradient-based refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Different perturbation strategies
    perturbation_quantities = [0.02, 0.03, 0.05]
    
    restart_count = 0
    max_restarts = 30
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        if restart_count == 0:
            # Start from best solution
            h_init = best_h.copy()
        else:
            # Strategy 1: Random perturbation
            perturbation = np.random.choice(perturbation_quantities)
            h_init = best_h + np.random.normal(0, perturbation, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)

        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 150, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    # Early stopping if target reached
                    if best_c5 < 0.3807:
                        break
        except:
            pass
        
        restart_count += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
