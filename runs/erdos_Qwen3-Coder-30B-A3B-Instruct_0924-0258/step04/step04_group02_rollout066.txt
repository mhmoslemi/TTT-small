import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimize using a variation of the Nelder-Mead approach
    # with multiple restarts and adaptive parameters
    
    # Different initializations to escape local minima
    initial_solutions = [h_start.copy()]
    
    # Add random perturbations from the starting solution
    for i in range(5):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_solutions.append(h_pert)
    
    # Also try a simple piecewise constant pattern
    h_piecewise = np.ones(n_points)
    mid = n_points // 2
    h_piecewise[:mid] = 0.75
    h_piecewise[mid:] = 0.25
    h_piecewise = h_piecewise * (n_points / 2) / np.sum(h_piecewise)
    initial_solutions.append(h_piecewise)
    
    # Generate some random solutions as well
    for i in range(4):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_solutions.append(h_rand)
    
    # Optimization using different strategies
    strategy_count = 0
    
    for h_init in initial_solutions:
        h_current = h_init.copy()
        c_current = evaluate_erdos_solution(h_current, dx)
        
        # Simple local search
        for it in range(500):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Slight random perturbation
            h_pert = h_current + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            c_pert = evaluate_erdos_solution(h_pert, dx)
            
            if c_pert < c_current:
                h_current = h_pert
                c_current = c_pert
                
                if c_current < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c_current
            
            # Easy escape with small chance
            if np.random.rand() < 0.002:
                h_eject = np.random.rand(n_points)
                h_eject = h_eject * (n_points / 2) / np.sum(h_eject)
                c_eject = evaluate_erdos_solution(h_eject, dx)
                if c_eject < c_current:
                    h_current = h_eject
                    c_current = c_eject
                    
        strategy_count += 1
        
        # Early stopping check
        if best_c5 < 0.3807:
            break
            
        if time() - start_time > budget_s * 0.9:
            break

    # Final local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # If we haven't reached a good enough solution, try one final 
        # refinement with SLSQP starting from best found solution
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

    return (list(best_h), best_c5, n_points)
