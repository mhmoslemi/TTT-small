import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for PSO
    swarm_size = 50
    max_iter = 1000
    inertia = 0.9
    inertia_decay = 0.999
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_best_h = [h.copy() for h in swarm]
    personal_best_c5 = [evaluate_erdos_solution(h, dx) for h in swarm]
    global_best_h = personal_best_h[np.argmin(personal_best_c5)].copy()
    global_best_c5 = np.min(personal_best_c5)
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (personal_best_h[i] - swarm[i])
            social = c2 * r2 * (global_best_h - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Update position and enforce constraints
            swarm[i] += velocities[i]
            # Clamp to bounds [0, 1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            # Adjust to meet sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate new solution
            new_c5 = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if new_c5 < personal_best_c5[i]:
                personal_best_c5[i] = new_c5
                personal_best_h[i] = swarm[i].copy()
                
                # Update global best
                if new_c5 < global_best_c5:
                    global_best_c5 = new_c5
                    global_best_h = swarm[i].copy()
        
        # Update inertia
        inertia *= inertia_decay
    
    # Convex optimization refinement
    try:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        # Build soft constraints on overlap
        for k in range(0, 2*n_points-1):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            # Add softened constraint
            constraints.append(corr_sum <= global_best_c5 * dx * 1.5)
        
        # Set the target to minimize
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < global_best_c5:
                    global_best_h = h_refined
                    global_best_c5 = c5_refined
        except Exception:
            pass
    except Exception:
        pass
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, global_best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < global_best_c5:
                    global_best_h = h_refined
                    global_best_c5 = c5_refined
        except:
            pass
    
    return (list(global_best_h), global_best_c5, n_points)
