import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 40
    population = []
    tabu_list = set()
    tabu_limit = 50
    stagnation_count = 0
    max_stagnation = 50
    
    # Multi-start initialization
    for _ in range(10):
        perturbed = h_start + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points/2) / np.sum(perturbed)
        population.append(perturbed)
    
    for _ in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Keep existing solutions for population
    for i in range(len(population), population_size):
        h_init = np.random.random(n_points)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        population.append(h_init)
    
    # Crossover and mutation functions
    def crossover(parent1, parent2, crossover_prob=0.7):
        if random.random() > crossover_prob:
            return parent1.copy()
            
        mask = np.random.random(n_points) > 0.5
        child = np.where(mask, parent1, parent2).copy()
        return child
    
    def mutate(individual, mutation_rate=0.15):
        if random.random() > mutation_rate:
            return individual.copy()
            
        mutated = individual.copy()
        mut_indices = np.random.choice(n_points, size=max(1, int(n_points * 0.1)), replace=False)
        for idx in mut_indices:
            mutated[idx] = random.random()
            
        # Repair
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points/2) / np.sum(mutated)
        return mutated
    
    # Evaluate population
    def evaluate_population(pop):
        fitness = []
        for h in pop:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        return fitness
    
    # Main GA loop
    generation = 0
    while time() - start_time < budget_s * 0.95:
        generation += 1
        if generation % 10 == 0:
            # Local refinement
            fitnesses = evaluate_population(population)
            top_indices = np.argsort(fitnesses)[:5]
            for i in top_indices:
                h_refined = population[i].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_final = res.x
                        c5_final = objective(h_final)
                        if c5_final < best_c5:
                            best_h = h_final
                            best_c5 = c5_final
                except:
                    pass
        
        # Evaluate population
        fitnesses = evaluate_population(population)
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive parameters
        if stagnation_count > max_stagnation:
            # Increase mutation, decrease crossover
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Selection and reproduction
        sorted_indices = np.argsort(fitnesses)
        next_pop = []
        
        # Elitism
        for i in range(5):
            next_pop.append(population[sorted_indices[i]].copy())
            
        # Tournament selection + crossover + mutation
        while len(next_pop) < population_size:
            # Tournament
            tournament_size = 3
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_idx]
            
            # Second parent
            tournament_indices2 = np.random.choice(len(population), tournament_size)
            tournament_fitness2 = [fitnesses[i] for i in tournament_indices2]
            winner_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
            
            parent2 = population[winner_idx2]
            
            # Crossover
            child = crossover(parent1, parent2, crossover_prob)
            # Mutation
            child = mutate(child, mutation_rate)
            
            # Add to next population
            next_pop.append(child)
            
        population = next_pop
        
        # Tabu maintenance
        if len(tabu_list) > tabu_limit:
            tabu_list.pop()
            
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
