import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_coarse, n_fine):
    h_coarse = np.array(h_coarse)
    x_coarse = np.linspace(0, 2, len(h_coarse))
    x_fine = np.linspace(0, 2, n_fine)
    return np.interp(x_fine, x_coarse, h_coarse)

def project_and_normalize(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Prepare grids
    n_coarse = 50
    n_fine = 100
    
    h_coarse = linear_resample(h_start, n_coarse)
    h_coarse = project_and_normalize(h_coarse, n_coarse / 2)
    
    h_fine = linear_resample(h_start, n_fine)
    h_fine = project_and_normalize(h_fine, n_fine / 2)
    
    # Global search on coarse grid
    for step in range(100):
        if time() - start_time > budget_s * 0.7:
            break
            
        h_pert = h_coarse + np.random.normal(0, 0.03, n_coarse)
        h_pert = project_and_normalize(h_pert, n_coarse / 2)
        
        c5_new = evaluate_erdos_solution(h_pert, 2.0/n_coarse)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = linear_resample(h_pert, n_points)
            
    # Local refinement on fine grid
    h_fine = linear_resample(best_h, n_fine)
    h_fine = project_and_normalize(h_fine, n_fine / 2)
    
    def objective(h):
        return evaluate_erdos_solution(h, 2.0 / len(h))
    
    def constraint_sum(h):
        return np.sum(h) - n_fine / 2
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_fine)]
    
    for restart in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_init = h_fine.copy()
        h_init += np.random.normal(0, 0.01, n_fine)
        h_init = project_and_normalize(h_init, n_fine / 2)
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = linear_resample(h_refined, n_points)
        except:
            pass
    
    # Final twist with convex relaxation, more constraints
    if time() - start_time < budget_s - 30:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add some overlap constraints based on coarse optimal
        corr_size = 2 * n_points - 1
        max_shift = n_points // 2
        for k in range(1, max_shift):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-5)
        
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
