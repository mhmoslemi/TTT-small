import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define objective in terms of Fourier coefficients
    # Using a sigmoid activation with Fourier coefficients
    def construct_h(coeffs, n_points, dx):
        # Extract real and imaginary parts; only real parts are used (cos terms)
        n_coeffs = len(coeffs) // 2
        a = coeffs[:n_coeffs]
        b = coeffs[n_coeffs:]
        
        # Compute a0 term separately to avoid scaling issues
        a0 = coeffs[0] if len(coeffs) > 0 else 0.0
        x = np.linspace(0, 2, n_points)
        
        # Generate Fourier series
        fft_series = a0 * np.ones_like(x)
        for k in range(1, n_coeffs):
            fft_series += a[k-1] * np.cos(k * np.pi * x) + b[k-1] * np.sin(k * np.pi * x)
        
        # Apply sigmoid activation
        h = 1 / (1 + np.exp(-fft_series))
        
        # Normalize
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        h = np.clip(h, 0, 1)
        return h
    
    # Reconstruction function
    def reconstruct_from_coeffs(coeffs, n_points, dx):
        h = construct_h(coeffs, n_points, dx)
        return h
    
    def obj_func(coeffs):
        h = reconstruct_from_coeffs(coeffs, n_points, dx)
        return evaluate_erdos_solution(h, dx)
    
    # Optimizer with adaptive learning rates
    def adaptive_gradient_descent(obj_func, init_params, max_iter=500, lr=0.01, tol=1e-6):
        params = init_params.copy()
        best_value = obj_func(params)
        best_params = params.copy()
        prev_value = best_value + 100
        decay = 0.99
        lr = lr
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.9:
                break
            
            # Compute gradient using finite differences
            eps = 1e-6
            grad = np.zeros_like(params)
            for j in range(len(params)):
                # Perturb parameter at j
                params_plus = params.copy()
                params_plus[j] += eps
                params_minus = params.copy()
                params_minus[j] -= eps
                
                # Evaluate objective function
                f_plus = obj_func(params_plus)
                f_minus = obj_func(params_minus)
                grad[j] = (f_plus - f_minus) / (2 * eps)
            
            # Update parameters
            params = params - lr * grad
            current_value = obj_func(params)
            if current_value < best_value:
                best_value = current_value
                best_params = params.copy()
            
            # Adaptive learning rate
            if abs(prev_value - current_value) < tol:
                lr *= decay
            prev_value = current_value
            
            if np.isnan(current_value) or np.isinf(current_value):
                break
                
        return best_params, best_value
    
    # Start optimization with initial parameters
    n_coeffs = 2 * (n_points // 2) + 1  # Use Fourier decomposition
    init_coeffs = np.random.normal(0, 0.1, n_coeffs)
    best_coeffs, final_c5 = adaptive_gradient_descent(obj_func, init_coeffs, max_iter=300)
    
    # Refine with local optimization
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = reconstruct_from_coeffs(best_coeffs, n_points, dx)
    
    # Try multiple random restarts for more exploration
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        restart_coeffs = np.random.normal(0, 0.1, n_coeffs)
        try:
            restart_coeffs, restart_c5 = adaptive_gradient_descent(obj_func, restart_coeffs, max_iter=200)
            if restart_c5 < best_c5:
                best_c5 = restart_c5
                best_h = reconstruct_from_coeffs(restart_coeffs, n_points, dx)
        except:
            pass
    
    # Check the current best
    # Ensure normalization
    h_norm = best_h * (n_points / 2) / np.sum(best_h)
    h_final = np.clip(h_norm, 0, 1)
    
    return (list(h_final), best_c5, n_points)
