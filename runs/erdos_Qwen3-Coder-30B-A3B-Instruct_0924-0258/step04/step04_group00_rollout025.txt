import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution refinement
    resolutions = [35, 40, 50]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
        
        if res > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h_start)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (res/2) / np.sum(h_new)
            h_start = h_new
            n_points = res
        else:
            h_start = best_h.copy()
            n_points = res
            
        dx = 2.0 / n_points
        
        # Local search with random restarts
        max_restarts = 20
        restart_count = 0
        local_iter = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_start + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(evaluate_erdos_solution, h_restart, args=(dx,), method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0  # Reset counter on improvement
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
                
            local_iter += 1
            if local_iter > 3:
                break
                
        # Convex relaxation for current resolution
        if time() - start_time < budget_s * 0.8:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add overlap constraints for selected shifts
                shifts = list(range(1, min(n_points//2, 10))) + list(range(n_points//2, n_points-1, 2))
                for k in shifts:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
                
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
