import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization approach
    swarm_size = 50
    max_iter = 1000
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for i in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            # Add some noise
            noise = np.random.normal(0, 0.02, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.uniform(0.1, 0.9, n_points)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        score = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_scores.append(score)
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_best[i] - particles[i])
            social = c2 * r2 * (global_best - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            # Normalize to maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
                    
                    if global_best_score < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_score
    
    # Convex optimization refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40 and best_c5 > 0.37:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Approximate problem using some overlap constraints to guide
            for k in range(1, min(15, 2 * n_points - 1)):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx * 1.1)  # Be optimistic
            
            prob = cp.Problem(cp.Minimize(cp.sum(h)), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts
        for restart in range(4):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
