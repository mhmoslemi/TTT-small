import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Fourier-based parameterization
    def h_from_coeffs(coeffs, n_points, x):
        a0, *ab_coeffs = coeffs
        a = ab_coeffs[::2]
        b = ab_coeffs[1::2]
        # Truncated Fourier series
        fourier = a0 + np.sum([a[k] * np.cos((k+1) * np.pi * x) + b[k] * np.sin((k+1) * np.pi * x)
                              for k in range(len(a))], axis=0)
        # Sigmoid activation
        return 1. / (1. + np.exp(-4. * fourier))
    
    def normalize_h(h):
        return h * (n_points / 2) / np.sum(h)
    
    def compute_overlaps(h):
        dx = 2.0 / n_points
        # Compute overlap using FFT
        h_fft = fft(h)
        inv_fft = fft(1 - h)
        overlap_fft = h_fft * np.conj(inv_fft)
        overlaps = ifft(overlap_fft).real
        # Only the valid overlaps (no padding)
        overlaps = overlaps[n_points-1:]  # Take the N-1 last elements
        max_overlap = np.max(overlaps) * dx
        max_idx = np.argmax(overlaps)
        return max_overlap, max_idx
    
    # Initial parameterization
    n_coeffs = 1 + 2 * (n_points // 4)  # Reduced for simplicity
    initial_coeffs = np.random.normal(0, 0.1, n_coeffs) if n_coeffs > 0 else np.array([0.])
    
    # Rescale to match initial h if provided
    if 'initial_h_values' in globals():
        x = np.linspace(0, 2, n_points, endpoint=False)
        init_h = h_from_coeffs(initial_coeffs, n_points, x)
        init_h = normalize_h(init_h)
        # Adjust coeffs to match initial_h
        def loss_coeffs(coeffs):
            h = h_from_coeffs(coeffs, n_points, x)
            h = normalize_h(h)
            return np.sum((h - init_h)**2)
        res = minimize(loss_coeffs, initial_coeffs, method='L-BFGS-B', options={'maxiter': 50})
        if res.success:
            initial_coeffs = res.x
    
    best_coeffs = initial_coeffs.copy()
    best_h = h_from_coeffs(best_coeffs, n_points, np.linspace(0, 2, n_points, endpoint=False))
    best_h = normalize_h(best_h)
    best_c5, _ = compute_overlaps(best_h)
    
    # Optimization loop
    it = 0
    max_it = 1000
    lr = 0.01
    reg = 0.001
    
    while it < max_it and (time() - start_time) < budget_s * 0.95:
        h = h_from_coeffs(best_coeffs, n_points, np.linspace(0, 2, n_points, endpoint=False))
        h = normalize_h(h)
        max_c5, max_idx = compute_overlaps(h)
        
        if max_c5 < best_c5:
            best_coeffs = best_coeffs.copy()
            best_c5 = max_c5
        
        # Gradient approximation using finite differences
        eps = 1e-4
        grad = np.zeros_like(best_coeffs)
        for i in range(len(best_coeffs)):
            coeffs_plus = best_coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = h_from_coeffs(coeffs_plus, n_points, np.linspace(0, 2, n_points, endpoint=False))
            h_plus = normalize_h(h_plus)
            _, max_idx_plus = compute_overlaps(h_plus)
            max_c5_plus, _ = compute_overlaps(h_plus)

            coeffs_minus = best_coeffs.copy()
            coeffs_minus[i] -= eps
            h_minus = h_from_coeffs(coeffs_minus, n_points, np.linspace(0, 2, n_points, endpoint=False))
            h_minus = normalize_h(h_minus)
            _, max_idx_minus = compute_overlaps(h_minus)
            max_c5_minus, _ = compute_overlaps(h_minus)

            grad[i] = (max_c5_plus - max_c5_minus) / (2 * eps)
        
        # Update with regularization
        best_coeffs -= lr * (grad + reg * best_coeffs)
        
        it += 1
    
    # One final refinement with local optimizers
    def objective(coeffs):
        h = h_from_coeffs(coeffs, n_points, np.linspace(0, 2, n_points, endpoint=False))
        h = normalize_h(h)
        return compute_overlaps(h)[0]
    
    def constraint(coeffs):
        h = h_from_coeffs(coeffs, n_points, np.linspace(0, 2, n_points, endpoint=False))
        h = normalize_h(h)
        return np.sum(h) - n_points / 2.0
    
    # Multiple local searches
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        coeffs_restart = np.random.normal(0, 0.1, len(best_coeffs))
        try:
            res = minimize(objective, coeffs_restart, method='SLSQP',
                           constraints={'type': 'eq', 'fun': constraint},
                           bounds=[(-5, 5)] * len(best_coeffs),
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_test = h_from_coeffs(res.x, n_points, np.linspace(0, 2, n_points, endpoint=False))
                h_test = normalize_h(h_test)
                c5_test = compute_overlaps(h_test)[0]
                if c5_test < best_c5:
                    best_coeffs = res.x
                    best_c5 = c5_test
        except:
            pass
    
    final_h = h_from_coeffs(best_coeffs, n_points, np.linspace(0, 2, n_points, endpoint=False))
    final_h = normalize_h(final_h)
    
    return (list(final_h), best_c5, n_points)
