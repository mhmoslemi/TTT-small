import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    swarm = np.array([h_start] * swarm_size)
    for i in range(swarm_size):
        if i > 0:
            # Perturb each particle slightly
            pert = np.random.normal(0, 0.02, n_points)
            swarm[i] = np.clip(swarm[i] + pert, 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
    
    # Initialize velocities
    velocities = np.zeros_like(swarm)
    
    # Initialize personal bests and global best
    personal_best = swarm.copy()
    personal_best_vals = np.array([evaluate_erdos_solution(h, dx) for h in swarm])
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Ensure bounds [0, 1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Renormalize to satisfy sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate new position
            current_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < global_best_val:
                    global_best = swarm[i].copy()
                    global_best_val = current_val
    
    # Convex relaxation refinement
    if time() - start_time < budget_s * 0.90:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add constraints for a few key lags to make problem more tractable
            for k in range(1, min(20, 2 * n_points - 1), 2):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                # Add a soft constraint increasing with k
                constraints.append(corr_sum <= (0.4 - 0.001*k) * dx * n_points)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < global_best_val:
                        global_best = h_refined
                        global_best_val = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final SLSQP polish
    if time() - start_time < budget_s - 10:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Multiple random restarts
        for restart in range(8):
            if time() - start_time > budget_s - 10:
                break
                
            h_restart = global_best + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < global_best_val:
                        global_best = h_refined
                        global_best_val = c5_refined
                    elif c5_refined < global_best_val + 1e-6:  # slight tolerance
                        global_best = h_refined
                        global_best_val = c5_refined
            except:
                pass
    
    return (list(global_best), global_best_val, n_points)
