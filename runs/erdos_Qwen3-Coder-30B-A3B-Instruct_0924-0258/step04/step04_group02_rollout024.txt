import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Simulated annealing
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two random points
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute maximum allowable delta
        delta_max = min(1.0 - current_h[i], current_h[j])
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Create candidate
        candidate_h = current_h.copy()
        candidate_h[i] += delta
        candidate_h[j] -= delta
        
        # Ensure bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(candidate_h) - (n_points / 2)
        if abs(sum_diff) > 1e-10:
            # Adjust all points to maintain sum
            candidate_h += sum_diff / n_points
            
        # Ensure final bounds
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(candidate_h, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool down
        T *= alpha
        
    return (list(best_h), best_c5, n_points)
