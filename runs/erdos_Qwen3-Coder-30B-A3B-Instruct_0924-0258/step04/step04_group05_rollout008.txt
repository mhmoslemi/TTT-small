import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Linearly resample h_in to n_out points"""
    n_in = len(h_in)
    x_in = np.arange(n_in)
    x_out = np.linspace(0, n_in - 1, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with best known construction or default
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        n_points = len(h_base)
    else:
        n_points = 35
        h_base = np.ones(n_points) * 0.5

    # Fine-scale representation
    h_fine = linear_resample(h_base, 100)
    n_fine = len(h_fine)
    dx_fine = 2.0 / n_fine
    
    # Coarse representation for global search
    h_coarse = linear_resample(h_base, 50)
    n_coarse = len(h_coarse)
    dx_coarse = 2.0 / n_coarse
    
    # Initialize tracking variables
    best_h_coarse = h_coarse.copy()
    best_h_fine = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx_fine)
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    
    # Use a more directed search strategy
    # Neighbor perturbation parameters
    init_perturb = 0.05
    min_perturb = 0.005
    decay = 0.995
    
    # Coarse grid search
    for step in range(50):
        if time() - start_time > budget_s * 0.3:
            break
            
        # Perturb coarse grid
        perturb_scale = max(min_perturb, init_perturb * (decay ** step))
        h_new = h_coarse + np.random.normal(0, perturb_scale, n_coarse)
        h_new = project_to_constraints(h_new, n_coarse/2)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx_coarse)
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_h_coarse = h_new.copy()
    
    # Transfer to fine grid
    h_fine = linear_resample(best_h_coarse, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine/2)
    c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    if c5_fine < best_c5:
        best_c5 = c5_fine
        best_h_fine = h_fine.copy()
    
    # Local refinement with fine grid
    def objective_fine(h):
        return evaluate_erdos_solution(h, dx_fine)
    
    def constraint_sum(h):
        return np.sum(h) - n_fine / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_fine)]
    
    # Annealed optimization
    for restart in range(5):
        if time() - start_time > budget_s * 0.8:
            break
            
        h_init = best_h_fine.copy() + np.random.normal(0, 0.02, n_fine)
        h_init = project_to_constraints(h_init, n_fine/2)
        
        try:
            res = minimize(objective_fine, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                h_refined = project_to_constraints(h_refined, n_fine/2)
                c5_refined = objective_fine(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h_fine = h_refined
                    if best_c5 < 0.3807:
                        break
        except:
            continue
    
    # A final global search approach
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 60:
        # Try coarse-to-fine interpolation multiple times
        for i in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Sample a new coarse point
            x_coarse = np.linspace(0, 1, n_coarse)
            h_sample = np.sin(np.pi * x_coarse) * 0.5 + 0.5  # sine-shaped
            h_sample = project_to_constraints(h_sample, n_coarse/2)
            
            # Interpolate to fine
            h_interp = linear_resample(h_sample, n_fine)
            h_interp = project_to_constraints(h_interp, n_fine/2)
            c5_new = evaluate_erdos_solution(h_interp, dx_fine)
            
            # Local optim with SLSQP if better
            if c5_new < best_c5:
                try:
                    res = minimize(objective_fine, h_interp, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 50, 'ftol': 1e-9})
                    if res.success:
                        h_final = res.x
                        h_final = project_to_constraints(h_final, n_fine/2)
                        c5_final = objective_fine(h_final)
                        if c5_final < best_c5:
                            best_c5 = c5_final
                            best_h_fine = h_final
                except:
                    pass
    
    # Final projection to ensure feasibility
    best_h_final = project_to_constraints(best_h_fine, n_fine/2)
    best_c5_final = evaluate_erdos_solution(best_h_final, dx_fine)
    
    return (list(best_h_final), best_c5_final, n_fine)
