import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization (PSO)
    swarm_size = 50
    max_iter = 1000
    w = 0.7  # Inertia weight
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Create initial swarm around the starting point
    for i in range(swarm_size):
        h = h_start.copy()
        if i > 0:  # Perturb other particles
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Normalize to sum to n_points/2
        h *= (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests and global best
    personal_best_positions = [h.copy() for h in swarm]
    personal_best_values = [evaluate_erdos_solution(h, dx) for h in swarm]
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best_positions[i] - swarm[i]) +
                c2 * r2 * (global_best_position - swarm[i])
            )
            
            # Update position
            swarm[i] += velocities[i]
            
            # Keep within bounds [0,1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Normalize to satisfy constraint
            swarm[i] *= (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate new solution
            new_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_best_values[i] = new_value
                personal_best_positions[i] = swarm[i].copy()
                
                # Update global best if needed
                if new_value < global_best_value:
                    global_best_value = new_value
                    global_best_position = swarm[i].copy()
        
        # Update global best if needed
        current_global_idx = np.argmin(personal_best_values)
        if personal_best_values[current_global_idx] < global_best_value:
            global_best_value = personal_best_values[current_global_idx]
            global_best_position = personal_best_positions[current_global_idx].copy()
    
    # Use the global best from PSO as starting point for refinement
    refined_h = global_best_position.copy()
    
    # Convex Relaxation refinement
    # Try to solve a linear program to minimize overlap (approximately)
    try:
        # Continuous relaxation with constraints
        h = cp.Variable(n_points)
        # Objective is just a proxy (we'll work on the function space limit directly)
        # We'll add penalty-style constraints to rule out bad solutions
        
        # Use constraint projection to maintain sum constraint and bounds
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add explicit constraints to make optimization more meaningful
        for k in range(1, min(10, 2 * n_points - 1)):
            # For each possible shift k, we enforce a constraint about overlap
            # This is a rough smoothing of what we want to achieve
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= 0.5 * dx * n_points)
        
        # Minimize a relaxed proxy of max overlap (using max of a few samples)
        # This is a linear relaxation of the problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < global_best_value:
                    refined_h = h_refined
                    global_best_value = c5_refined
        except Exception:
            pass
    except:
        pass
    
    # Final polishing with bounds-constrained optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try several random restarts and refer points from swarm
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Try updating from previous best or a swarm member
        if np.random.rand() < 0.5:
            h_restart = refined_h.copy()
        else:
            idx = np.random.randint(0, swarm_size)
            h_restart = swarm[idx].copy()
        
        # Add noise
        noise = np.random.normal(0, 0.02, n_points)
        h_restart += noise
        h_restart = np.clip(h_restart, 0, 1)
        h_restart *= (n_points/2) / np.sum(h_restart)
        
        try:
            # Use L-BFGS-B for better convergence properties
            res = minimize(objective, h_restart, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < global_best_value:
                    global_best_value = c5_opt
                    refined_h = h_opt
        except:
            pass
    
    # Final check
    final_h = refined_h.copy()
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
