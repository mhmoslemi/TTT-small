import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create 3 clusters with different heights
        cluster_starts = [0.2, 1.0, 1.8]
        cluster_width = 0.3
        heights = [0.5, 0.8, 0.4]
        for i, (start, height) in enumerate(zip(cluster_starts, heights)):
            start_idx = max(0, int((start - cluster_width/2)/dx))
            end_idx = min(n_points, int((start + cluster_width/2)/dx))
            h_start[start_idx:end_idx] = height
        
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 20
    generations = 300
    mutation_rate = 0.02
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Track the best solution found
    best_individual = h_start.copy()
    best_fitness = evaluate_erdos_solution(h_start, dx)
    
    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Track best individual
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Selection (tournament)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, population_size-1)]
            
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.01)
                    child2[j] += np.random.normal(0, 0.01)
            
            # Clamp and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Refine best individual with SQP
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj, best_individual, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                best_individual = res.x
                best_fitness = obj(best_individual)
        except:
            pass
    
    # Final check
    final_h = np.clip(best_individual, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
