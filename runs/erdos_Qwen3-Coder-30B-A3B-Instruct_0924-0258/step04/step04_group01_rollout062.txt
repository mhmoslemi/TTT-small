import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a more structured initial guess with 3 clusters
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define objective function for optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Define constraint for sum = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create constraints dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use BOBYQA for optimization
    options = {'maxiter': 500, 'catol': 1e-6, 'ftol': 1e-6, 'disp': False}
    
    try:
        result = minimize(objective, h_start, method='BOBYQA', 
                          bounds=bounds, constraints=cons, options=options)
        if result.success:
            h_opt = result.x
            c5_opt = objective(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception as e:
        pass

    # Final cleanup to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
