import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization (20 points)
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    max_iter = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        h_pert = best_h + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Accept with probability based on temperature
        old_c5 = evaluate_erdos_solution(best_h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            best_h = h_pert
            best_c5 = new_c5
            
        temp *= cool_rate
    
    # Upsample to 40 points
    if time() - start_time > budget_s * 0.95:
        return (list(best_h), best_c5, n_points)
    
    n_points = 40
    dx = 2.0 / n_points
    h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    best_h = h_upsampled.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine optimization (40 points) using SLSQP
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            best_h = res.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass

    # Final refinement with gradient-based optimization
    max_iter = 1000
    step_size = 0.001
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        orig_c5 = evaluate_erdos_solution(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - orig_c5) / eps
            
        # Update step
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            
    return (list(best_h), best_c5, n_points)
