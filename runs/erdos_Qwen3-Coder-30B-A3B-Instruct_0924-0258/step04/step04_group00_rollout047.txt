import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with adaptive step size
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_gradient(h, k_shifts):
        grad = np.zeros_like(h)
        for k in k_shifts:
            if k >= 0:
                # h[i] * (1 - h[i+k]) for valid indices
                if k == 0:
                    grad += 1 - 2 * h
                else:
                    grad[:len(h)-k] += 1 - h[k:]
                    grad[k:] -= h[:len(h)-k]
            else:
                # same logic but for negative k
                k = -k
                grad[k:] += 1 - h[:len(h)-k]
                grad[:len(h)-k] -= h[k:]
        return grad
    
    # Initialize
    h = best_h.copy()
    c5 = best_c5
    alpha = 0.01
    max_iter = 1000
    stagnation_count = 0
    max_stagnation = 50
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps to identify most critical shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        k_shifts = [max_k - len(h) + 1, max_k - len(h) + 2, max_k - len(h) + 3, 
                   max_k - len(h) + 4, max_k - len(h) + 5, max_k - len(h) + 6] if max_k < len(h) else [max_k]
        
        # Compute gradient
        grad = compute_gradient(h, k_shifts)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            alpha *= 1.2  # Increase step size on improvement
            stagnation_count = 0
        else:
            alpha *= 0.5  # Decrease step size
            stagnation_count += 1
            
        if stagnation_count >= max_stagnation:
            break
            
        # Occasionally add smoothing to prevent oscillation
        if i % 10 == 0:
            h = 0.9 * h + 0.1 * np.convolve(h, np.ones(3)/3, mode='same')
            h = project_h(h)
            c5 = evaluate_erdos_solution(h, dx)
    
    # Final projection and refinement
    h_final = project_h(h)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    # Final smoothing
    h_smooth = 0.95 * best_h + 0.05 * np.convolve(best_h, np.ones(5)/5, mode='same')
    h_smooth = project_h(h_smooth)
    c5_smooth = evaluate_erdos_solution(h_smooth, dx)
    
    if c5_smooth < best_c5:
        best_h = h_smooth
        best_c5 = c5_smooth
    
    return (list(best_h), best_c5, n_points)
