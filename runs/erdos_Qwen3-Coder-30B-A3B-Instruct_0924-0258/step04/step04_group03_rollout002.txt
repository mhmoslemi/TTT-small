import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hybrid optimization approach  
    def pairwise_perturb(h, step_size=0.05):
        h_new = h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-step_size, step_size)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        # Clamp and adjust to maintain sum constraint
        h_new = np.clip(h_new, 0, 1)
        # Project back to feasible set
        total = np.sum(h_new)
        scale = (n_points / 2) / total
        h_new *= scale
        return h_new
    
    # Simulated Annealing with adaptive steps
    current_h = h_start.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.9996
    max_steps = 20000
    
    # Track best
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while (time() - start_time) < budget_s * 0.90:
        # Adaptive step size
        step_size = max(0.005, 0.05 * temperature)
        
        # Perturb
        new_h = pairwise_perturb(current_h, step_size)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cooling
        temperature *= alpha
        step_count += 1
        
        # Periodic convex refinement
        if step_count % 500 == 0 and (time() - start_time) < budget_s * 0.85:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add overlap constraint based on current best
                # Compute max overlap constraint using recent solution 
                overlap_vals = np.correlate(best_h, 1 - best_h, mode='full')
                max_overlap = np.max(overlap_vals) * dx
                # Add a very tight constraint
                max_overlap_tight = max_overlap * 0.999
                
                # Manually pin down constraint on correlation values
                # This is a simplified version where we just try to reduce overlap
                corr_scaled = cp.sum(cp.multiply(h[:n_points-1], 1 - h[1:n_points])) 
                # Add soft regularization to keep overlapping small
                constraints.append(corr_scaled <= max_overlap_tight * dx)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final gradient-based refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 60:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Multiple random restarts
        for restart in range(10):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            # Perturb best solution slightly
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 150, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
