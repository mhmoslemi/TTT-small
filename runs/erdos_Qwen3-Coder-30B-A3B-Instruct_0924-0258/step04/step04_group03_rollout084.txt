import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    max_iter = 1000
    omega_start = 0.9
    omega_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.ones(n_points) * 0.5
        # Random perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_scores.append(score)
    
    # Global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO Main Loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        omega = omega_start - (omega_start - omega_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (omega * velocities[i] +
                           c1 * r1 * (personal_best[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particles[i].copy()
    
    # Update best if PSO did better
    if global_best_score < best_c5:
        best_h = global_best.copy()
        best_c5 = global_best_score
    
    # Convex optimization refinement
    if time() - start_time < budget_s * 0.90:
        try:
            h = cp.Variable(n_points)
            # Simplified approach using a hard constraint on convolution
            # Try to minimize a large constant times a sum of overlap terms that
            # is equivalent to standard form
            
            # Joint problem: we want to find minimum C such that for all k:
            # sum_i h[i] * (1 - h[i+k]) <= C * dx for all valid k
            # Ha, it's not a linear  constraint -- we'll use max overlap directly.
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            max_k = len(corr)
            # Just using a harder optimization to get a better feasible solution
            # We will compute the current overlap
            c5_current = evaluate_erdos_solution(best_h, dx)
            
            # Relax problem but still active a large portion of constraints
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            # Add direct constraints: We'll修为众多子程序以防止因小数挥发造成的信号损失.
            # Add approximate bounds on max overlap, since we know
            # max_overlap <= c5_current
            # This is a heuristic approach
            
            # Instead solve the same problem but for maximum overlap over all lags, solved directly
            # Use convex program to set a tighter feasible region
            prob = cp.Problem(cp.Minimize(cp.sum(h)), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-7)
                if prob.status == cp.OPTIMAL:
                    try:
                        # Use slack to bring it closer to a good minimum
                        h_refined = h.value
                        # Ensure it still meets the sum constraint exactly
                        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                    except:
                        pass
            except Exception:
                pass
        except:
            pass
    
    # Final polishing with SLSQP
    if time() - start_time < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts with better starting points
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Use the current best solution as starting point, with some variation
            h_restart = best_h * (1 + np.random.normal(0, 0.01, n_points))
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # If we have modified even a little over the initial run, return new solution
    return (list(best_h), best_c5, n_points)
