import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
        
    # Normalize to correct sum
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    T = T_initial
    max_iterations = int(1e5)
    
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Generate candidate by perturbing two points
        h_candidate = best_h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Determine max allowable delta to stay in bounds
        delta_max = min(1.0 - h_candidate[i], h_candidate[j])
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Apply delta
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Maintain sum constraint via linear adjustment
        sum_diff = np.sum(h_candidate) - (n_points / 2)
        adjustment = -sum_diff / n_points
        
        # Apply adjustment uniformly
        h_candidate += adjustment
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Re-adjust to sum exactly
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        if candidate_c5 < best_c5 or np.random.rand() < np.exp(-(candidate_c5 - best_c5) / T):
            best_h = h_candidate
            best_c5 = candidate_c5
        
        # Cool down
        T *= alpha
        iteration += 1
    
    # Final verification
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
