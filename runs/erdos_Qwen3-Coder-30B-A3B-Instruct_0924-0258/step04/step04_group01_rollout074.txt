```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create three clusters
        cluster_size = 3
        cluster_centers = [0.4, 1.0, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    tolerance = 1e-6
    max_iterations = 50
    current_c = (low + high) / 2.0
    
    def is_feasible(h_guess, c_val):
        # This checks if all the overlap integrals are <= c_val * dx
        # We formulate this as a convex problem using linearization
        # For the overlap term ∫h(x)(1-h(x+k))dx, we have a constraint
        # This can be approximated as a set of convex constraints
        # with the linearization at h_guess
        h_current = h_guess.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Try to solve with sequential convex programming
        max_iter = 5
        for _ in range(max_iter):
            # Create linearized constraints
            # For each k from -1 to 1, calculate overlap linearization
            # Linearize the overlap integral term
            corr = np.correlate(h_current, 1 - h_current, mode='full') * dx
            max_overlap = np.max(corr)
            if max_overlap <= c_val + 1e-8:
                return True, h_current
            
            # Use CVXPY solver for convex relaxation
            h_var = cp.Variable(n_points)
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points/2
            ]
            
            # Add constraints from correlations
            # Linear terms: h[i]*h[i+k] approximated as h[i]*h_k + h_k*h[i] - h_k*h_k
            # Let's hold h_k fixed, and use Lagrangian multipliers
            for k in range(-n_points+1, n_points):
                if k == 0:
                    continue
                # Compute correlation at k
                corr_k = 0.0
                if k > 0:
                    # h[i] * (1 - h[i+k]) for i = 0 to n_points-k-1
                    for i in range(n_points - k):
                        corr_k += h_current[i] * (1 - h_current[i+k])
                else:
                    # h[i] * (1 - h[i+k]) for i = -k to n_points-1, but start from 0
                    k_abs = -k
                    for i in range(n_points - k_abs):
                        corr_k += h_current[i+k_abs] * (1 - h_current[i])
                corr_k *= dx
                
                # Linearization around h_current
                linear_part = 0.0
                for i in range(n_points):
                    if i+k >= 0 and i+k < n_points:
                        linear_part += h_current[i] * (1 - h_current[i+k]) + \
                                       h_current[i+k] * (1 - h_current[i]) - \
                                       h_current[i] * h_current[i+k]
                linear_part *= dx
                
                # Add a constraint if the linear estimation precludes the value
                # For safety, we'll add one coherence constrained:
                # Just constrain the integral with an upper bound
                if corr_k > c_val * dx:
                    # Add a soft constraint with penalty
                    pass
            
            # Build the convex program
            obj = cp.Minimize(0)
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_current = h_var.value
                    h_current = np.clip(h_current, 0, 1)
                    h_current = h_current * (n_points/2) / np.sum(h_current)
                else:
                    break
            except:
                break  # Could not solve - might not be feasible
        
        # Final check of overlap
        corr = np.correlate(h_current, 1 - h_current, mode='full') * dx
        max_corr = np.max(corr)
        return max_corr <= c_val + 1e-6, h_current
    
    # Binary search loop with a bit of improvement by local search
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_c = (low + high) / 2.0
        feasible, h_feasible = is_feasible(best_h, current_c)
        
        # Local optimization step if feasible
        if feasible:
            # Setup optimization problem
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Use SLSQP with refines
            try:
                h_init = h_feasible.copy()
                res = minimize(obj, h_init, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
            
            high = current_c
            if high - low < tolerance:
                break
        else:
            low = current_c
            
        # Side step to sampling: if we're in an high-region, search with different voids
        # Try a bit of restart with better local exploration
        if i % 5 == 0 and not feasible:
            try:
                h_trial = np.zeros(n_points)
                # Different pattern
                clus = [0.3, 1.0, 1.7]
                clus_width = 3
                for c in clus:
                    start = max(0, int((c - clus_width/2)/dx))
                    end = min(n_points, int((c + clus_width/2)/dx))
                    h_trial[start:end] = 1.0 / (2*clus_width)
                scale = (n_points/2) / np.sum(h_trial)
                h_trial *= scale
                
                # Try refining
                res = minimize(lambda h: evaluate_erdos_solution(h, dx), h_trial, method='SLSQP',
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass

    # Final check on best solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), final_c, n_points)
```
