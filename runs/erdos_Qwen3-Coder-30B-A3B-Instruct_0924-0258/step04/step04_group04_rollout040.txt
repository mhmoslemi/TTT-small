import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create structured initial guess
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create two clusters
        cluster1_start = int(0.3 / dx)
        cluster1_end = int(0.7 / dx)
        cluster2_start = int(1.3 / dx)
        cluster2_end = int(1.7 / dx)
        
        if cluster1_end > cluster1_start:
            h_start[cluster1_start:cluster1_end] = 1.0
        if cluster2_end > cluster2_start:
            h_start[cluster2_start:cluster2_end] = 1.0
        
        # Normalize sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Cache for correlation values
    corr_cache = None
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Fast local search - gradient-based with bounds
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_trial = res.x
                trial_c5 = evaluate_erdos_solution(h_trial, dx)
                if trial_c5 < best_c5:
                    best_h = h_trial
                    best_c5 = trial_c5
        except:
            pass
        
        # Heuristic local perturbations
        h_pert = best_h.copy()
        # Add small random noise
        noise = np.random.normal(0, 0.005, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        pert_c5 = evaluate_erdos_solution(h_pert, dx)
        if pert_c5 < best_c5:
            best_h = h_pert
            best_c5 = pert_c5
        
        # Random restart with structured pattern
        if iteration % 50 == 0 and iteration > 0:
            h_restart = np.zeros(n_points)
            dx = 2.0 / n_points
            # Create a few peaks and valleys
            for i in range(0, n_points, 7):
                if i < n_points:
                    h_restart[i] = np.random.uniform(0.2, 0.8)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            restart_c5 = evaluate_erdos_solution(h_restart, dx)
            if restart_c5 < best_c5:
                best_h = h_restart
                best_c5 = restart_c5
    
    # Final refinement with full optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
