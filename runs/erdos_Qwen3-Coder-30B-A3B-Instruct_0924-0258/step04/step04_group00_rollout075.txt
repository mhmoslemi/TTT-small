import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Multi-resolution optimization
    resolutions = [35, 40, 45, 50]
    for res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
            
        if res > n_points:
            h_new = np.interp(np.linspace(0, 1, res), 
                             np.linspace(0, 1, n_points), 
                             best_h)
        else:
            h_new = best_h[:res]
        
        n_points = res
        dx = 2.0 / n_points
        
        # Random restarts with local search
        max_restarts = 20
        restart_count = 0
        
        while time() - start_time < budget_s * 0.9 and restart_count < max_restarts:
            h_restart = h_new + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res_min = minimize(objective, h_restart, method='SLSQP', 
                                  bounds=bounds, constraints=cons_sum, 
                                  options={'maxiter': 100, 'ftol': 1e-7, 'disp': False})
                if res_min.success:
                    h_refined = res_min.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                else:
                    restart_count += 1
            except:
                restart_count += 1
                
            # Convex relaxation
            if time() - start_time < budget_s * 0.8 and restart_count % 4 == 0:
                try:
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Add overlap constraints - full set
                    corr_vals = []
                    for k in range(1, n_points):
                        if k < n_points - k:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        else:
                            corr_sum = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
                        corr_vals.append(corr_sum)
                    
                    for corr_sum in corr_vals:
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
                except:
                    pass
                
        # Project back to valid range
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final refinement with 60-point resolution
    n_points = 60
    dx = 2.0 / n_points
    h_final = np.interp(np.linspace(0, 1, n_points), 
                       np.linspace(0, 1, len(best_h)), 
                       best_h)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    # Local refinement at final resolution
    def constraint_sum_final(h):
        return np.sum(h) - n_points / 2.0
    
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        res_final = minimize(objective, h_final, method='SLSQP', 
                            bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum_final}, 
                            options={'maxiter': 100, 'ftol': 1e-7, 'disp': False})
        if res_final.success:
            h_final = res_final.x
            best_c5 = objective(h_final)
    except:
        pass
    
    return (list(h_final), best_c5, n_points)
