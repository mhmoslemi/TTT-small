import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create initial 3-cluster structure
        cluster_size = 5
        cluster_centers = [0.6, 1.0, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    # Initialize binary search bounds
    low, high = 0.3, 0.38085
    precision = 1e-6
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # SCP-based binary search for C5
    for idx in range(50):  # binary search iterations
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2.0
        h_refined = h_start.copy()
        dx = 2.0 / n_points
        
        # SCP iterations to check feasibility for mid
        for _ in range(5):  # SCP iterations
            if time() - start_time > budget_s * 0.95:
                break
                
            # Formulate SCP problem
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1]
            
            # Sum constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Linearized overlap constraints
            # For all possible shifts k
            for k in range(n_points):
                # Approximate sum of h[i] * h[i+k]
                # h[i] * h[i+k] ~ h[i,old] * h[i+k] + h[i+k,old] * h[i] - h[i,old] * h[i+k,old]
                # So: ∑ h[i] * h[i+k] >= mid * dx
                # Linear in h: ∑ h[i] * h[i+k] >= mid * dx
                expr = cp.sum(cp.multiply(h_refined, cp.roll(h_vars, k)) + 
                              cp.multiply(h_vars, cp.roll(h_refined, k)) - 
                              h_refined[k] * h_refined[0]) * dx
                constraints.append(expr >= mid)
            
            # Objective: any feasible point is good
            prob = Problem(Minimize(0), constraints)
            
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    new_h = h_vars.value
                    h_refined = np.clip(new_h, 0, 1)
                    # Normalize
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            except:
                break
        
        # Evaluate new solution
        c5_val = evaluate_erdos_solution(h_refined, dx)
        if c5_val < best_c5:
            best_h = h_refined.copy()
            best_c5 = c5_val
            
        if c5_val <= mid:
            high = mid
        else:
            low = mid
            
        if abs(high - low) < precision:
            break
    
    # Final cleanup and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    return (list(best_h), final_c5, n_points)
