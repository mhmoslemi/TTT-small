import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize Fourier coefficients
    n_freqs = n_points // 2 + 1  # Half the points for real FFT
    a = np.random.normal(0, 0.1, n_freqs)
    b = np.random.normal(0, 0.1, n_freqs - 1)
    a[0] = np.log((n_points/2) / (1 - n_points/2))  # Initialize DC component to get correct mean
    
    def get_h():
        x = np.linspace(0, 2, n_points, endpoint=False)
        # Build Fourier series
        fourier = a[0] * np.ones_like(x)
        for k in range(1, n_freqs):
            fourier += a[k] * np.cos(k * np.pi * x) + b[k-1] * np.sin(k * np.pi * x)
        h = sigmoid(fourier)
        # Normalize
        h *= (n_points / 2) / np.sum(h)
        return h
    
    def obj_and_grad():
        h = get_h()
        # Compute overlap using FFT
        H = fft(h)
        H_conj = np.conj(H)
        one_minus_h = 1 - h
        one_minus_H = fft(one_minus_h)
        overlap_fft = H_conj * one_minus_H
        overlap = ifft(overlap_fft).real
        # We only care about the valid overlaps
        max_overlap = np.max(overlap)
        h_scaled = h * (n_points / 2) / np.sum(h)
        
        # Compute gradients using finite differences for simplicity
        eps = 1e-5
        grad_a = np.zeros_like(a)
        grad_b = np.zeros_like(b)
        
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = sigmoid(a_plus[0] * np.ones_like(x) + 
                             sum(a_plus[k] * np.cos(k * np.pi * x) + 
                                 b[k-1] * np.sin(k * np.pi * x) for k in range(1, n_freqs)))
            h_plus *= (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            
            a_minus = a.copy()
            a_minus[i] -= eps
            h_minus = sigmoid(a_minus[0] * np.ones_like(x) + 
                              sum(a_minus[k] * np.cos(k * np.pi * x) + 
                                  b[k-1] * np.sin(k * np.pi * x) for k in range(1, n_freqs)))
            h_minus *= (n_points / 2) / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = np.max(corr_minus) * dx
            
            grad_a[i] = (max_plus - max_minus) / (2 * eps)
            
        for i in range(len(b)):
            b_plus = b.copy()
            b_plus[i] += eps
            h_plus = sigmoid(a[0] * np.ones_like(x) + 
                             sum(a[k] * np.cos(k * np.pi * x) + 
                                 b_plus[k] * np.sin(k * np.pi * x) for k in range(1, n_freqs)))
            h_plus *= (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            
            b_minus = b.copy()
            b_minus[i] -= eps
            h_minus = sigmoid(a[0] * np.ones_like(x) + 
                              sum(a[k] * np.cos(k * np.pi * x) + 
                                  b_minus[k] * np.sin(k * np.pi * x) for k in range(1, n_freqs)))
            h_minus *= (n_points / 2) / np.sum(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_minus = np.max(corr_minus) * dx
            
            grad_b[i] = (max_plus - max_minus) / (2 * eps)
            
        return max_overlap, grad_a, grad_b

    # Main optimization loop
    learning_rate = 0.01
    for iter in range(10000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current objective and gradient
        try:
            obj_val, grad_a, grad_b = obj_and_grad()
            if obj_val < best_c5:
                best_c5 = obj_val
                h_current = get_h()
                best_h = h_current
                
            # Update coefficients
            if iter % 10 == 0:
                # Periodically update the optimization step size
                learning_rate = max(1e-5, learning_rate * 0.999)
                
            a -= learning_rate * grad_a
            b -= learning_rate * grad_b
            
        except Exception as e:
            # Handle any errors gracefully
            pass
    
    # Final refinement
    if time() - start_time < budget_s * 0.9:
        h_final = get_h()
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
