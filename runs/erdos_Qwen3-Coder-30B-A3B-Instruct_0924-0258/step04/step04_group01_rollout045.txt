import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with 3 clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 3
        cluster_centers = [0.4, 1.0, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search bounds
    low = 0.3
    high = 0.38085
    precision = 1e-6
    best_feasible = high
    
    # First, do a bit of gradient-based refinement on the starting solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try local optimization
    try:
        res = minimize(obj, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            h_refined = res.x
            c5_refined = obj(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Try a simple particle swarm optimization
    class PSO:
        def __init__(self, n_particles=20, n_dims=35, bounds=(0,1)):
            self.n_particles = n_particles
            self.n_dims = n_dims
            self.bounds = bounds
            self.particles = np.random.uniform(bounds[0], bounds[1], (n_particles, n_dims))
            self.velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_dims))
            self.pbest = self.particles.copy()
            self.pbest_values = np.array([evaluate_erdos_solution(p, dx) for p in self.particles])
            self.gbest_idx = np.argmin(self.pbest_values)
            self.gbest = self.pbest[self.gbest_idx]
            self.gbest_value = self.pbest_values[self.gbest_idx]
            
        def update(self):
            for i in range(self.n_particles):
                # Update velocity
                r1, r2 = np.random.random(2)
                self.velocities[i] = (0.7 * self.velocities[i] + 
                                      1.5 * r1 * (self.pbest[i] - self.particles[i]) +
                                      1.5 * r2 * (self.gbest - self.particles[i]))
                
                # Update position
                self.particles[i] += self.velocities[i]
                
                # Apply bounds
                self.particles[i] = np.clip(self.particles[i], self.bounds[0], self.bounds[1])
                
                # Normalize sum
                self.particles[i] = self.particles[i] * (n_points/2) / np.sum(self.particles[i])
                
                # Evaluate
                val = evaluate_erdos_solution(self.particles[i], dx)
                if val < self.pbest_values[i]:
                    self.pbest[i] = self.particles[i].copy()
                    self.pbest_values[i] = val
                    if val < self.gbest_value:
                        self.gbest = self.particles[i].copy()
                        self.gbest_value = val
    
    # PSO loop
    pso = PSO(n_particles=20, n_dims=n_points)
    pso_iter = 0
    max_pso_iter = 100
    while pso_iter < max_pso_iter and (time() - start_time) < budget_s * 0.7:
        pso.update()
        pso_iter += 1
        if pso.gbest_value < best_c5:
            best_h = pso.gbest.copy()
            best_c5 = pso.gbest_value

    # Binary search for tighter bound
    while (high - low > precision) and (time() - start_time) < budget_s * 0.95:
        mid = (low + high) / 2
        
        # Feasibility check using simple SNES approach
        h_trial = best_h.copy()  # Start with current best
        
        # Try to find a feasible solution with overlap <= mid
        for _ in range(10):
            # Simple gradient projection
            h_old = h_trial.copy()
            h_trial = h_trial + np.random.normal(0, 0.001, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            
            # Check if it's better
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial <= mid and c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
        
        if best_c5 <= mid:
            high = mid
            best_feasible = high
        else:
            low = mid
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
