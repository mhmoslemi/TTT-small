import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create a simple initial guess
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    
    best_h = h_start.copy()
    best_c5 = np.inf
    
    def evaluate_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def form_and_solve_lp(h_guess, target_c5, dx):
        # Variables
        h = Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        
        # Sum constraint
        constraints.append(sum(h) == n_points / 2)
        
        # Overlap constraints
        # Compute all valid correlations for all lags
        for k in range(2 * n_points - 1):
            if k < n_points:
                # h[i]*(1-h[i+k]) for i = 0..n_points-k-1
                expr = 0
                for i in range(n_points - k):
                    expr += h[i] * (1 - h[i+k])
            else:
                # h[i]*(1-h[i+k]) for i = 0..n_points-1, but with proper indexing
                # Will be essentially 0 when shifted too much
                expr = 0
                for i in range(k - n_points + 1, n_points):
                    expr += h[i] * (1 - h[i + n_points - k - 1])
            
            # Normalize for dx
            normalized_expr = expr * dx
            constraints.append(normalized_expr <= target_c5)
        
        # Solve problem
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=GLPK, verbose=False, max_iters=1000)
            if prob.status == 'optimal':
                return True, h.value
            elif prob.status == 'infeasible':
                return False, None
        except:
            return False, None

    # Binary search loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Try to solve with this mid value
        feasible, h_solution = form_and_solve_lp(h_start, mid, 2.0/n_points)
        
        if feasible:
            high = mid
            # Update best if better
            if mid < best_c5:
                best_c5 = mid
                best_h = h_solution.copy()
        else:
            # If not feasible, increase lower bound
            low = mid
            
        if high - low < precision:
            break

    # Final verification
    final_c5 = evaluate_overlap(best_h, 2.0/n_points)
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_overlap(best_h, 2.0/n_points)
    
    return (list(best_h), float(final_c5), n_points)
