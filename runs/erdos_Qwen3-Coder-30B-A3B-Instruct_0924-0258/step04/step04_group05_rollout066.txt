import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 40
    population = []
    
    # Generate diverse population
    for i in range(10):  # Perturb initial solution
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    for i in range(5):  # Add random valid solutions
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Ensure we have enough population
    while len(population) < population_size:
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Tabu search memory
    tabu_list = []
    tabu_limit = 50
    
    # Parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_counter = 0
    max_stagnation = 50
    
    def mutate(h):
        h_new = h.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h_new[i] += np.random.normal(0, 0.03)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def crossover(h1, h2):
        if np.random.random() > crossover_prob:
            return h1.copy()
        mask = np.random.random(n_points) > 0.5
        h_new = np.where(mask, h1, h2)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def local_refinement(h, max_iter=100):
        def objective(h_):
            return evaluate_erdos_solution(h_, dx)
        
        def constraint_sum(h_):
            return np.sum(h_) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': max_iter, 'ftol': 1e-7})
            if res.success:
                return res.x
        except:
            pass
        return h
    
    generation = 0
    while time() - start_time < budget_s * 0.95:
        if generation % 10 == 0:  # Refine top performers every 10 gens
            # Sort population by fitness
            population_fitness = [(h, evaluate_erdos_solution(h, dx)) for h in population]
            population_fitness.sort(key=lambda x: x[1])
            
            # Refine top 5
            for i in range(min(5, len(population_fitness))):
                h_refined = local_refinement(population_fitness[i][0], 100)
                f_refined = evaluate_erdos_solution(h_refined, dx)
                if f_refined < population_fitness[i][1]:
                    population_fitness[i] = (h_refined, f_refined)
            
            # Update population
            population = [h for h, _ in population_fitness]
            current_best = min(population_fitness, key=lambda x: x[1])
            current_c5 = current_best[1]
            
            if current_c5 < best_c5:
                best_h = current_best[0].copy()
                best_c5 = current_c5
                stagnation_counter = 0
            else:
                stagnation_counter += 1
        
        # Selection: tournament selection
        def tournament_selection(pop, k=3):
            selected = random.sample(pop, k)
            selected.sort(key=lambda h: evaluate_erdos_solution(h, dx))
            return selected[0]
        
        # Generate new population
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Update tabu list
        for h in population:
            # create a hashable representation
            h_hash = tuple(np.round(h, decimals=6))
            if h_hash not in tabu_list:
                tabu_list.append(h_hash)
                if len(tabu_list) > tabu_limit:
                    tabu_list.pop(0)
        
        # Check for stagnation and adapt parameters
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        
        # Adaptive changes
        if stagnation_counter > 0:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        generation += 1
        
        # Early exit if we get to target
        if best_c5 < 0.3807:
            break
    
    # Final refinement of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        h_final = local_refinement(best_h, 200)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
