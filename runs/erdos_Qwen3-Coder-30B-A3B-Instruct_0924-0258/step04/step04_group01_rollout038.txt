import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create structured initial guess with 3 clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 8
        cluster_centers = [0.5, 1.0, 1.5]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Direct optimization using BOBYQA
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Set up BOBYQA parameters
    options = {
        'maxiter': 500,
        'disp': False,
        'maxfev': 10000,
        'catrans': True,
        'rhobeg': 0.5,
        'rhoend': 1e-5
    }
    
    # Try BOBYQA optimization
    try:
        res = minimize(
            obj,
            h_start,
            method='BOBYQA',
            bounds=bounds,
            constraints=cons_sum,
            options=options
        )
        
        if res.success:
            h_optimized = res.x
            # Ensure constraints are met exactly
            h_optimized = np.clip(h_optimized, 0, 1)
            h_optimized = h_optimized * (n_points/2) / np.sum(h_optimized)
            c5_optimized = obj(h_optimized)
            
            if c5_optimized < best_c5:
                best_h = h_optimized.copy()
                best_c5 = c5_optimized
    except Exception as e:
        pass  # Fall back to original solution if BOBYQA fails

    # Final check and cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
