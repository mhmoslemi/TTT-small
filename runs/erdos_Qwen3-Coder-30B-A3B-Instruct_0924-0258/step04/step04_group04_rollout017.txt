import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from initial_h_values if available, otherwise create a structured construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a multi-peak structure similar to previous attempts but refined
        centers = [0.3, 1.0, 1.7]
        widths = [0.2, 0.3, 0.2]
        for i, (c, w) in enumerate(zip(centers, widths)):
            start_idx = max(0, int((c - w) / dx))
            end_idx = min(n_points, int((c + w) / dx))
            h_start[start_idx:end_idx] = 1.0 / 3  # to get smooth peaks
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    
    # Main optimization loop
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Regularization-based refinement using CVXPY
        dx = 2.0 / n_points
        
        # Set up variables
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        
        # Compute overlaps
        h_vec = h_start.copy()
        overlaps = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        max_overlap = np.max(overlaps)
        
        # Create a convex approximation using dual variables (as in strategy)
        lam = cp.Variable(len(overlaps), nonneg=True)
        # Approximate objective
        obj_expr = cp.sum(cp.multiply(lam, overlaps)) - cp.sum(cp.entr(lam) + lam)  # entropy penalty
        problem = cp.Problem(cp.Minimize(obj_expr), constraints)
        
        try:
            problem.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            if problem.status == cp.OPTIMAL:
                h_new = h.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2) / np.sum(h_new)
                curr_c5 = evaluate_erdos_solution(h_new, dx)
                if curr_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = curr_c5
        except:
            pass

        # Local improvement using scipy
        def obj(h_test):
            return evaluate_erdos_solution(h_test, dx)
        
        def con_sum(h_test):
            return np.sum(h_test) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_local = res.x
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points/2) / np.sum(h_local)
                local_c5 = evaluate_erdos_solution(h_local, dx)
                if local_c5 < best_c5:
                    best_h = h_local.copy()
                    best_c5 = local_c5
        except:
            pass
        
        h_start = best_h.copy()  # Use latest best as starting point
    
    return (list(best_h), best_c5, n_points)
