import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Use initial construction if available

    # Initialize with a better approach: binary search style
    # We'll use a smooth approximation with sigmoid-like behavior
    # and optimize the underlying parameters
    
    # Parameters for smooth approximation
    # We'll treat this as a smooth binary problem with parameters
    def smooth_binary_h(θ, beta, lambda_, n_points):
        # Create linear function with parameters
        x = np.linspace(0, 2, n_points)
        # Smooth sigmoid-like function
        h = 1.0 / (1.0 + np.exp(-10 * (x - 1.0)))  # sigmoid in middle
        h = h * (2.0 * lambda_ / np.sum(h) - 1.0) + 1.0  # scale and shift
        h = np.clip(h, 0, 1)
        return h
    
    def objective_with_params(θ_beta_lambda):
        theta = θ_beta_lambda[:-2]
        beta = θ_beta_lambda[-2]
        lam = θ_beta_lambda[-1]
        
        # Create piecewise constant-like with smooth transitions
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        
        # Use a periodic nature with a smooth transition
        for i in range(n_points):
            # Smooth approximation of a binary-like step function
            h[i] = 1.0 / (1.0 + np.exp(-10 * (x[i] - 1.0)))  # Center around 1
            h[i] = (h[i] - 0.5) * 2 * lam + beta  # scale and offset
            h[i] = np.clip(h[i], 0, 1)
            
        # Enforce integral constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= (n_points / 2) / current_sum
            
        return evaluate_erdos_solution(h, dx)

            res = minimize(objective_with_params, initial_guess,

                theta = res.x[:-2]

                best_c5 = res.fun
        except:
            pass

    return (list(best_h), best_c5, n_points)
