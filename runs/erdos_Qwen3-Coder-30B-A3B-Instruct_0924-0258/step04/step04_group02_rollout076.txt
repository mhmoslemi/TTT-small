import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # We'll use a trust-constr optimization method on reparameterized space
    # h = sigmoid(z), constraint: sum(h)*dx = 1
    
    def sigmoid(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def obj_and_grad(z):
        h = sigmoid(z)
        # Adjust for integral constraint
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h, dx)
        
        # Compute gradient w.r.t. z
        # Let’s do it numerically for stability
        eps = 1e-8
        grad = np.zeros_like(z)
        for i in range(len(z)):
            z_plus = z.copy()
            z_plus[i] += eps
            h_plus = sigmoid(z_plus)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            z_minus = z.copy()
            z_minus[i] -= eps
            h_minus = sigmoid(z_minus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return c5, grad
    
    # Try different initializations
    initial_zs = []
    
    # Start from best solution
    z_start = np.log(h_start / (1 - h_start + 1e-10))  # logit of h
    initial_zs.append(z_start.copy())
    
    # Random starts
    for _ in range(3):
        z_rnd = np.random.normal(0, 0.5, n_points)
        initial_zs.append(z_rnd)
    
    # Optimization with warm starts from different initializations
    for z_init in initial_zs:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Simple gradient descent with line search to stay feasible
            z_current = z_init.copy()
            for iter in range(500):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                obj_val, grad = obj_and_grad(z_current)
                if obj_val < best_c5:
                    best_c5 = obj_val
                    h_current = sigmoid(z_current)
                    h_current = h_current * (n_points / 2) / np.sum(h_current)
                    best_h = h_current.copy()
                
                # Line search step
                step_size = 0.1
                for _ in range(10):
                    z_new = z_current - step_size * grad
                    h_new = sigmoid(z_new)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    obj_new = evaluate_erdos_solution(h_new, dx)
                    if obj_new <= obj_val:
                        z_current = z_new.copy()
                        break
                    step_size *= 0.5

        except:
            continue # Continue with other initializations

    # Refine final solution using scipy minimize
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Convert best h to z space
        z = np.log(best_h / (1 - best_h + 1e-10))
        try:
            def objective(z):
                h = sigmoid(z)
                h = h * (n_points / 2) / np.sum(h)
                return evaluate_erdos_solution(h, dx)
            
            def gradient(z):
                h = sigmoid(z)
                h = h * (n_points / 2) / np.sum(h)
                # Numerical gradient again
                eps = 1e-6
                grad = np.zeros_like(z)
                for i in range(len(z)):
                    z_plus = z.copy()
                    z_plus[i] += eps
                    h_plus = sigmoid(z_plus)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c5_plus = evaluate_erdos_solution(h_plus, dx)

                    z_minus = z.copy()
                    z_minus[i] -= eps
                    h_minus = sigmoid(z_minus)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    c5_minus = evaluate_erdos_solution(h_minus, dx)

                    grad[i] = (c5_plus - c5_minus) / (2 * eps)
                return grad
            
            res = minimize(objective, z, method='L-BFGS-B', jac=gradient, 
                          options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10})
            if res.success:
                h_final = sigmoid(res.x)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final

        except:
            pass
    
    return (list(best_h), best_c5, n_points)
