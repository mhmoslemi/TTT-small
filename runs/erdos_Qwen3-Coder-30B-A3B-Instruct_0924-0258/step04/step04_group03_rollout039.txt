from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    selection_rate = 0.2
    num_generations = 1000

    population = []

        population.append(h_start)

        for _ in range(49):

            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)

    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break

        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)

        idx_sorted = np.argsort(fitness)
        population = [population[i] for i in idx_sorted]
        fitness = [fitness[i] for i in idx_sorted]

            parent1 = population[np.random.randint(0, len(population))]

            child = (parent1 + parent2) / 2.0

            mutation_magnitude = mutation_rate * (1.0 - generation / num_generations)
            noise = np.random.normal(0, mutation_magnitude, n_points)
            child += noise
            
            # Clip and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)

                    c5_refined = objective(h_refined)

                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
