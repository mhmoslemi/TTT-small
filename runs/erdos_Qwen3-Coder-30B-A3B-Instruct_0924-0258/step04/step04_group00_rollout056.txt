from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    num_generations = 500
    mutation_rate = 0.1
    tournament_size = 2

    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

            pop.append(create_individual())
        return pop
    
    population = create_initial_population()
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(ind, dx) for ind in population]
    best_idx = np.argmin(fitness)
    if fitness[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitness[best_idx]
    
    # GA loop
    generation = 0
    while time() - start_time < budget_s * 0.8 and generation < num_generations:
        # Tournament selection
        def tournament_selection():
            idxs = np.random.choice(population_size, tournament_size)
            best_idx = idxs[np.argmin([fitness[i] for i in idxs])]

            parent1 = tournament_selection()
            parent2 = tournament_selection()

                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            for i in range(n_points):

        population = new_population
        fitness = [evaluate_erdos_solution(ind, dx) for ind in population]

            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
