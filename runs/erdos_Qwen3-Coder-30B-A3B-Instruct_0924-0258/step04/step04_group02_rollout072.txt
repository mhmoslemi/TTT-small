import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    T = T_initial
    h_current = h_start.copy()
    c5_current = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute maximum delta that maintains bounds
        delta_max = min(h_current[i], 1.0 - h_current[j])
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Apply perturbation maintaining sum constraint
        h_candidate = h_current.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        # Renormalize to preserve sum
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject based on simulated annealing
        if c5_candidate < c5_current or np.random.rand() < np.exp(-(c5_candidate - c5_current) / T):
            h_current = h_candidate
            c5_current = c5_candidate
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
        
        # Cool down
        T *= alpha
    
    return (list(best_h), best_c5, n_points)
