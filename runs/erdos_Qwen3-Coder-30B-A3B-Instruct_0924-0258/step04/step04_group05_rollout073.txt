import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population with better diversity
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Population initialization with variations
    population_size = 40
    population = []
    
    # Add diverse variations
    for i in range(10):
        h_var = h_start + np.random.normal(0, 0.02, n_points)
        h_var = np.clip(h_var, 0, 1)
        h_var = h_var * (n_points/2) / np.sum(h_var)
        population.append(h_var)
    
    # Add random feasible solutions
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Add some structured variations
    for i in range(10):
        # Concentrated step function
        h_step = np.zeros(n_points)
        mid = n_points // 2
        width = n_points // 4
        h_step[mid-width//2:mid+width//2] = 1.0
        h_step = h_step * (n_points/2) / np.sum(h_step)
        population.append(h_step)
    
    # Add a few from varying scales
    for i in range(15):
        # Random height variation
        h_rand_var = np.random.rand(n_points)
        scale_factor = (n_points/2) / np.sum(h_rand_var)
        h_rand_var = h_rand_var * scale_factor
        h_rand_var = np.clip(h_rand_var, 0, 1)
        population.append(h_rand_var)
    
    # Ensure proper initialization
    if len(population) < population_size:
        while len(population) < population_size:
            h_e = np.random.rand(n_points)
            h_e = h_e * (n_points/2) / np.sum(h_e)
            population.append(h_e)
    
    population = population[:population_size]
    
    # Tabu list for memory
    tabu_list = []
    tabu_limit = 50
    
    # Parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_count = 0
    stagnation_limit = 50
    
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Evaluation function
    def eval(h):
        return evaluate_erdos_solution(h, dx)
    
    # Helper for constraint
    def normalize_sum(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Fitness evaluation with repair
    def get_fitness(h):
        h = np.clip(h, 0, 1)
        h = normalize_sum(h)
        return eval(h)
    
    # Genetic operators
    def crossover(p1, p2):
        if random.random() > crossover_prob:
            return p1.copy()
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, p1, p2)
        return child
    
    def mutate(h):
        if random.random() > mutation_rate:
            return h.copy()
        idx = random.randint(0, n_points-1)
        h[idx] += random.gauss(0, 0.03)
        h = np.clip(h, 0, 1)
        h = normalize_sum(h)
        return h
    
    # Main GA loop
    gen = 0
    max_gens = budget_s // 3
    
    while gen < max_gens and (time() - start_time) < budget_s * 0.95:
        # Evaluate all
        fitness = []
        for h in population:
            fit = get_fitness(h)
            fitness.append(fit)
        
        # Update best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            # Find corresponding h
            best_h_idx = fitness.index(min_fitness)
            best_h = population[best_h_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive parameters
        if stagnation_count > stagnation_limit:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            if tabu_limit > 10:
                tabu_limit = tabu_limit // 2
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Selection and new generation
        sorted_indices = np.argsort(fitness)
        top = [population[i] for i in sorted_indices[:population_size//2]]
        new_pop = top.copy()
        
        # Use top parents to generate new population
        while len(new_pop) < population_size:
            p1 = random.choice(top)
            p2 = random.choice(top)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        
        # Trim if over-sized
        population = new_pop[:population_size]
        
        # Tabu memory update
        current_solution_hash = hash(tuple(np.round(population[0], decimals=5)))
        if current_solution_hash in tabu_list:
            # If it's tabu, do a small mutation
            population[0] = mutate(population[0])
        else:
            tabu_list.append(current_solution_hash)
            if len(tabu_list) > tabu_limit:
                tabu_list.pop(0)
        
        gen += 1
        
        # Periodic local refinement
        if gen % 10 == 0:
            # Refine top solutions
            for i in range(min(5, len(population))):
                h_init = population[i].copy()
                # Local optimization
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]

                try:
                    res = minimize(objective, h_init, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[i]:
                            fitness[i] = c5_refined
                            population[i] = h_refined
                            if c5_refined < best_c5:
                                best_c5 = c5_refined
                                best_h = h_refined.copy()
                except:
                    pass
    
    # Refine final best
    if best_h is not None:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                best_h = res.x
                best_c5 = objective(best_h)
        except:
            pass
    
    # Ensure final constraints
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Return final result
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
