import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Initialize dual variables
    lambda_ = np.ones(2 * n_points - 1)
    
    def optimize_h(h_in, lambda_vals):
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(lambda_vals, 
            [cp.QuadForm(h, cp.eye(n_points)) - 
             cp.sum(cp.multiply(h, cp.ones(n_points)) - 
                    cp.convolve(h, np.ones(n_points), mode='full')[:n_points] * 2)])))
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                return h.value
        except:
            pass
        return h_in
    
    # Main optimization loop
    max_iter = 300
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update dual variables
        c_vals = np.correlate(h_start, 1 - h_start, mode='full') * dx
        lambda_ = np.maximum(lambda_ * (c_vals / lambda_), 1e-6)
        
        # Optimize h
        h_new = optimize_h(h_start, lambda_)
        if h_new is not None:
            h_temp = np.clip(h_new, 0, 1)
            h_temp = h_temp * (n_points/2) / np.sum(h_temp)
            h_start = h_temp
        
        # Evaluate and update best
        current_c5 = evaluate_erdos_solution(h_start, dx)
        if current_c5 < best_c5:
            best_h = h_start.copy()
            best_c5 = current_c5
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
