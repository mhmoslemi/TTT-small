import numpy as np
from scipy.optimize import minimize
from time import time
import math
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        h_start[5:15] = 1.0
        dx = 2.0 / n_points
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Convex optimization with penalty method
    def solve_convex_penalty(h_in, penalties, max_iter=50, learning_rate=0.1):
        h = Variable(n_points)
        constraints = [h >= 0, h <= 1, sum(h) == n_points/2]
        total_penalty = 0
        
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                return h_in, best_c5
            
            # Compute correlation matrix
            corr = np.correlate(h_in, 1 - h_in, 'full')
            corr_shifts = corr[len(corr)//2:]
            
            # Update penalties
            new_penalties = []
            for k in range(len(corr_shifts)):
                # Penalty based on current constraint violation
                penalty_increase = max(0, corr_shifts[k] - penalties[k])
                new_penalties.append(max(penalties[k] * (1 + learning_rate * penalty_increase), 1e-6))
            penalties = new_penalties
            
            # Objective: minimize sum of max over all shifts with penalty
            objective = Minimize(sum([max(corr_shifts[k] + penalties[k], 0) for k in range(len(corr_shifts))]))
            prob = Problem(objective, constraints)
            
            try:
                prob.solve(solver=ECOS, max_iters=50)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    h_in = h.value
                    h_in = np.clip(h_in, 0, 1)
                    h_in = h_in * (n_points/2) / np.sum(h_in)
                    
                    # Evaluate and update best
                    c5 = evaluate_erdos_solution(h_in, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_in.copy()
            except:
                pass
            
        return h_in, best_c5
    
    # Two-stage optimization
    # Stage 1: Coarse grid optimization
    penalties = [1.0] * (2 * n_points - 1)
    h_coarse, best_c5 = solve_convex_penalty(best_h.copy(), penalties, max_iter=100, learning_rate=0.1)
    
    # Upsample to finer grid
    n_points_fine = 40
    h_fine = np.interp(np.linspace(0, 2, n_points_fine), 
                       np.linspace(0, 2, n_points), 
                       h_coarse)
    h_fine = h_fine * (n_points_fine/2) / np.sum(h_fine)
    
    # Stage 2: Fine grid optimization with soft penalties
    penalties_fine = [1.0] * (2 * n_points_fine - 1)
    h_fine, best_c5 = solve_convex_penalty(h_fine.copy(), penalties_fine, max_iter=100, learning_rate=0.2)
    
    # Final refinement with SQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points_fine/2
            
        bounds = [(0, 1) for _ in range(n_points_fine)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_fine, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 30, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points_fine)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points_fine/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, 2.0/n_points_fine)
    
    return (list(best_h), final_c5, n_points_fine)
