import numpy as np
from cvxpy import *
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a smooth initial guess
        x = np.linspace(0, 2, n_points)
        h_start = np.ones_like(x) * 0.3
        # Add a central peak
        center = 1.0
        width = 0.3
        h_start += 0.4 * np.exp(-0.5 * ((x - center) / width) ** 2)
        h_start = np.clip(h_start, 0, 1)
        # Normalize to sum = n_points / 2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Binary search bounds for C5
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    best_h = h_start.copy()
    best_c5 = np.max(np.correlate(h_start, 1 - h_start, mode='full')) * dx
    
    # Binary search loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Build LP for feasibility check
        h = Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2.0
        ]
        
        # Overlap constraints for all lags
        for k in range(2 * n_points - 1):
            # Determine valid indices for shift k
            if k < n_points:
                # Forward shift: h[i] -> h[i+k]
                a_indices = np.arange(n_points - k)
                b_indices = np.arange(k, n_points)
            else:
                # Backward shift: h[i] -> h[i-(2*n_points-1-k)]
                shift = 2 * n_points - 1 - k
                a_indices = np.arange(shift, n_points)
                b_indices = np.arange(shift)
            
            # Build constraint: sum h[a] * (1 - h[b]) <= mid / dx
            if len(a_indices) > 0:
                expr = sum(h[i] * (1 - h[j]) for i, j in zip(a_indices, b_indices))
                constraints.append(expr <= mid / dx)
        
        # Formulate and solve the LP
        objective = Minimize(0)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=GLPK, verbose=False)
            if prob.status == 'optimal':
                # Update best solution
                h_opt = np.array([float(h[i].value) for i in range(n_points)])
                c5_val = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
                if c5_val < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_val
                
                # Update search bounds
                high = mid
                if high - low < precision:
                    break
            else:
                low = mid
        except:
            low = mid
            
    # Final check of best solution
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (list(best_h), final_c5, n_points)
