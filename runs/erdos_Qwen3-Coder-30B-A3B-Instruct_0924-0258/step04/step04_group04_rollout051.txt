import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    # Coarse optimization with simulated annealing
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Annealing parameters
    max_iter = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    population = []
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        for i, h in enumerate(population):
            curr_c5 = evaluate_erdos_solution(h, dx)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
                
        # Generate new population with perturbations and simulated annealing
        new_population = []
        for h in population:
            # Perturb
            h_pert = h + np.random.normal(0, perturb_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            # Accept with probability based on temperature
            old_c5 = evaluate_erdos_solution(h, dx)
            new_c5 = evaluate_erdos_solution(h_pert, dx)
            if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
                new_population.append(h_pert)
            else:
                new_population.append(h)
        
        population = new_population
        temp *= cool_rate

    # Upsample to higher resolution
    n_points = 40
    dx = 2.0 / n_points
    h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    # Refine at higher resolution using SLSQP
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, h_upsampled, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            best_h = res.x
            best_c5 = evaluate_erdos_solution(best_h, dx)
    except:
        pass

    # Final refinement with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        # Simple gradient-based refinement with decreasing step size
        step_size = 0.001
        for i in range(200):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros(n_points)
            # Approximate gradient via finite differences
            eps = 1e-5
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c_plus = evaluate_erdos_solution(h_plus, dx)
                grad[j] = (c_plus - best_c5) / eps
            
            # Gradient descent step
            new_h = best_h - step_size * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                
            step_size *= 0.995

    return (list(best_h), best_c5, n_points)
