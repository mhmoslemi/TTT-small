import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Logistic reparameterization
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def inv_sigmoid(y):
        return np.log(y / (1.0 - y))
    
    # Find appropriate b such that sum(h) = n_points/2
    def find_b(z):
        h = sigmoid(z)
        return np.log((n_points/2) / np.sum(h) + 1e-12)
    
    def h_from_z(z):
        b = find_b(z)
        return sigmoid(z + b)
    
    # Gradient computation for the maximum overlap
    def compute_gradients(z):
        h = h_from_z(z)
        # Find the worst case k
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        # Construct correlation matrix for this k
        # Since we just need gradient wrt h, we construct correlation explicitly
        grad = np.zeros_like(h)
        n = len(h)
        # Correlation for shift k_max (correlation = sum h[i] * (1 - h[i + k]))
        for i in range(n):
            j = (i + k_max) % n
            grad[i] += (1 - h[j])
            # Also account for the term (1 - h[i]) from gradient w.r.t h[j]
            # Gradient from i-th term is (1 - h[i + k]) for h[i]
            j = (i - k_max) % n
            grad[i] -= h[j]
        return grad * dx
    
    # Use a simple gradient-based approach with line search
    z_start = inv_sigmoid(h_start)

        if time() - start_time > budget_s * 0.9:
            break
            
        # Compute gradient
        grad = compute_gradients(z)
        
        # Take gradient step
        z_new = z - learning_rate * grad
        
        # Evaluate new point
        h_new = h_from_z(z_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Backtracking line search
        if c5_new < best_c5_opt:
            z = z_new
            best_c5_opt = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            patience_counter = 0
            # Increase rate if we make progress
            if learning_rate < 1.0:
                learning_rate *= 1.05
        else:
            # Decrease learning rate
            learning_rate *= 0.5
            patience_counter += 1

        if patience_counter >= patience:

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2.0

            cons_sum = {'type': 'eq', 'fun': constraint_sum}

                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
