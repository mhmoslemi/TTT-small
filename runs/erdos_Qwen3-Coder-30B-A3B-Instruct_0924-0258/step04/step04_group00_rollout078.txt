import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach: starting with a small resolution
    resolutions = [35, 40, 50]
    current_resolution = 0
    
    # Helper function to interpolate to higher resolution
    def interpolate_h(h, new_n_points):
        old_x = np.linspace(0, 2, len(h))
        new_x = np.linspace(0, 2, new_n_points)
        return np.interp(new_x, old_x, h)
    
    # Multi-resolution optimization
    while time() - start_time < budget_s * 0.95 and current_resolution < len(resolutions):
        n_points = resolutions[current_resolution]
        dx = 2.0 / n_points
        
        # Interpolate from previous resolution if needed
        if current_resolution > 0:
            h_start = interpolate_h(best_h, n_points)
        
        # Normalize to required integral
        h_start = h_start * (n_points/2) / np.sum(h_start)
        h_start = np.clip(h_start, 0, 1)
        
        # Local optimization with random restarts
        max_restarts = 20
        restart_count = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_start + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(evaluate_erdos_solution, h_restart, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               args=(dx,), options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0  # Reset counter on improvement
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
        
        current_resolution += 1
    
    # Final convex relaxation to tighten bounds
    try:
        h_opt = cp.Variable(n_points)
        constraints = [
            cp.sum(h_opt) == n_points / 2.0,
            h_opt >= 0,
            h_opt <= 1
        ]
        
        # Add a scaled subset of overlap constraints for efficiency
        max_shifts = min(20, n_points // 2)
        for k in range(1, max_shifts):
            if k < n_points // 2:
                corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= best_c5 * dx + 1e-6)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_feasible = h_opt.value
            c5_feasible = evaluate_erdos_solution(h_feasible, dx)
            if c5_feasible < best_c5:
                best_h = h_feasible
                best_c5 = c5_feasible
    except Exception:
        pass
    
    # Final projection
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
