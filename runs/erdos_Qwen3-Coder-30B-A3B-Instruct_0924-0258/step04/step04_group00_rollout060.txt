import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolution Strategy with Augmented Lagrangian
    population_size = 50
    mutation_strength = 0.1
    elite_size = 10
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, mutation_strength, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Lagrange multipliers
    lambda_sum = 0.0
    lambda_bounds = np.zeros(n_points)
    
    # Augmented Lagrangian function
    def augmented_lagrangian(h, lambda_sum, lambda_bounds):
        # Objective
        obj = evaluate_erdos_solution(h, dx)
        
        # Penalty terms for constraints
        penalty_sum = 0.5 * 1000 * (np.sum(h) - n_points/2)**2
        penalty_bounds = 0.5 * 1000 * np.sum(lambda_bounds * np.maximum(0, h - 1) + 
                                           lambda_bounds * np.maximum(0, -h))
        
        return obj + penalty_sum + penalty_bounds
    
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness of population
        fitness = []
        for h in population:
            fitness.append(evaluate_erdos_solution(h, dx))
        
        # Sort population by fitness
        sort_indices = np.argsort(fitness)
        population = [population[i] for i in sort_indices]
        fitness = [fitness[i] for i in sort_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, size=5, replace=False)
            winner = min(tournament, key=lambda i: fitness[i])
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1, parent2 = selected[i], selected[(i+1) % population_size]
            
            # Blend crossover
            alpha = np.random.random()
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutate
            noise1 = np.random.normal(0, mutation_strength, n_points)
            noise2 = np.random.normal(0, mutation_strength, n_points)
            
            child1 += noise1
            child2 += noise2
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Update Lagrange multipliers
        for h in population[:elite_size]:
            violation_sum = np.sum(h) - n_points/2
            lambda_sum += 10 * violation_sum  # Update with penalty parameter
            
            violation_bounds = np.maximum(0, h - 1) + np.maximum(0, -h)
            lambda_bounds += 10 * violation_bounds
            
        # Adapt mutation strength
        if generation % 10 == 0:
            diversity = np.std([np.sum((h - population[0])**2) for h in population])
            mutation_strength = max(0.001, 0.9 * mutation_strength) if diversity < 0.005 else mutation_strength
            if diversity < 0.005:
                mutation_strength = min(0.5, 1.1 * mutation_strength)
        
        generation += 1
    
    # Final refinement with local optimization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
