import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolution Strategy with Augmented Lagrangian
    pop_size = 50
    mutation_strength = 0.1
    elite_size = 10
    max_generations = 1000
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h_mut = h_start + np.random.normal(0, mutation_strength, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        population.append(h_mut)
    
    # Lagrange multipliers
    lambda_sum = 0.0
    lambda_bounds = np.zeros(n_points)
    
    def augmented_lagrangian(h, lambda_sum, lambda_bounds):
        obj = evaluate_erdos_solution(h, dx)
        penalty_sum = lambda_sum * (np.sum(h) - n_points / 2.0)
        penalty_bounds = np.sum(lambda_bounds * np.minimum(h, 0) + 
                                lambda_bounds * np.minimum(1 - h, 0))
        return obj - penalty_sum + penalty_bounds
    
    # Evolution loop
    generation = 0
    stagnation_count = 0
    max_stagnation = 50
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            f = augmented_lagrangian(h, lambda_sum, lambda_bounds)
            fitness.append(f)
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            indices = np.random.choice(pop_size, 5, replace=False)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            selected.append(population[best_idx])
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population through mutation
        new_population = elites.copy()
        for i in range(pop_size - elite_size):
            parent = selected[i]
            h_mut = parent + np.random.normal(0, mutation_strength, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points/2) / np.sum(h_mut)
            new_population.append(h_mut)
        
        population = new_population
        
        # Update Lagrange multipliers
        current_h = population[np.argmin(fitness)]
        sum_violation = np.sum(current_h) - n_points / 2.0
        lambda_sum += 0.1 * sum_violation
        
        bounds_violation = np.concatenate([np.maximum(0, current_h), 
                                          np.maximum(0, 1 - current_h)])
        lambda_bounds += 0.05 * bounds_violation
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(current_h, dx)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > max_stagnation:
            break
            
        generation += 1
    
    # Final refinement using local optimization
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
