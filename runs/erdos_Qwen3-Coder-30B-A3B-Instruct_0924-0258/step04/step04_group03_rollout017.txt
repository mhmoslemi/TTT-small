import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient for shift k
        grad = np.zeros_like(h)
        n = len(h)
        # For shift k, we have h[i] * (1 - h[i+k]) for valid indices
        # Gradient w.r.t. h[i] is (1 - h[i+k]) if i+k < n, else 0
        # But we also need to account for h[i+k] * (1 - h[i]) if i >= k
        for i in range(n):
            if i + k < n:
                grad[i] += (1 - h[i+k])
            if i >= k:
                grad[i] += (1 - h[i-k])
        return grad
    
    # BFGS-style gradient descent
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    max_iter = 5000
    stagnation_count = 0
    max_stagnation = 100
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    for iteration in range(max_iter):
        if (time() - start_time) > budget_s * 0.95:
            break
        
        # Find worst shift
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_worst = np.argmax(corr)
        max_overlap = corr[k_worst] * dx
        
        # Compute gradient for the worst shift
        grad = compute_gradient(current_h, k_worst)
        
        # Update with gradient descent
        new_h = current_h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        # Project to satisfy sum constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            stagnation_count = 0
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        else:
            stagnation_count += 1
            # Reduce learning rate if stuck
            if stagnation_count > max_stagnation:
                alpha *= 0.9
                stagnation_count = 0
                
        # Occasionally perform convex refinement
        if iteration % 50 == 0 and iteration > 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Find top violating shifts
                corr_full = np.correlate(current_h, 1 - current_h, mode='full')
                top_k = min(10, len(corr_full))
                indices = np.argsort(corr_full)[-top_k:]
                
                for k in indices:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        for restart in range(10):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
