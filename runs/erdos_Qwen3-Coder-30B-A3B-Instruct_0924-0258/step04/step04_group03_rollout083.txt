import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    n_iterations = 1000
    inertia = 0.9
    inertia_decay = (0.4 / inertia) ** (1.0 / n_iterations)
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.ones(n_points) * 0.5
        # Random perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_best_h = [p.copy() for p in particles]
    personal_best_c5 = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best_h = personal_best_h[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # PSO main loop
    for it in range(n_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (inertia * velocities[i] 
                   + c1 * r1 * (personal_best_h[i] - particles[i])
                   + c2 * r2 * (global_best_h - particles[i]))
            
            # Update position
            new_pos = particles[i] + vel
            
            # Enforce bounds and constraint
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2.0) / np.sum(new_pos)
            
            particles[i] = new_pos
            
            # Evaluate
            new_c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_c5 < personal_best_c5[i]:
                personal_best_c5[i] = new_c5
                personal_best_h[i] = particles[i].copy()
            
            # Update global best
            if new_c5 < global_best_c5:
                global_best_c5 = new_c5
                global_best_h = particles[i].copy()
                
        inertia *= inertia_decay
        
        # Keep track of best solution
        if global_best_c5 < best_c5:
            best_h = global_best_h.copy()
            best_c5 = global_best_c5
    
    # Convex optimization refinement
    if time() - start_time < budget_s * 0.9:
        try:
            h = cp.Variable(n_points)
            obj = cp.Minimize(0)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            corr_constraint = 0
            for k in range(0, n_points):
                corr_val = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                constraints.append(corr_val <= best_c5 / dx)
                corr_constraint = cp.max(corr_constraint, corr_val)
            constraints.append(corr_constraint <= best_c5 / dx)  # make known value the tightest
            
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final polishing with SLSQP
    if time() - start_time < budget_s * 0.8:
        # Set up from best found particle
        h_restart = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Restart and optimize
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
