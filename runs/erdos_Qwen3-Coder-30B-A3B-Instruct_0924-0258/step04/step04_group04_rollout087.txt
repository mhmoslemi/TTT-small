import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5  # Uniform distribution
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    tol = 1e-5
    lambda_low = 0.0
    lambda_high = 1.0
    best_feasible_h = None
    best_feasible_lambda = 0.0
    
    def check_feasibility(h, lambda_val):
        """Check if h is feasible for the given lambda"""
        # Compute all auto-correlations
        corr_full = np.correlate(h, 1 - h, mode='full')
        corr_vals = corr_full[len(h)-1:]  # Take non-negative shifts only
        min_corr = np.min(corr_vals)
        return min_corr >= lambda_val * dx
    
    def compute_gradients(h, lambda_val):
        """Compute gradient of minimum correlation w.r.t. h"""
        corr_full = np.correlate(h, 1 - h, mode='full')
        corr_vals = corr_full[len(h)-1:]
        min_idx = np.argmin(corr_vals)
        # Analytical gradient computation (simplified)
        # For the minimum correlation at shift min_idx, we want to increase it
        grad = np.zeros_like(h)
        # The gradient w.r.t. each h[i] of correlation at shift min_idx
        for i in range(len(h)):
            # Assuming we only add to h[i]
            if i - min_idx >= 0 and i - min_idx < len(h):
                grad[i] += 1
            if i + min_idx < len(h):
                grad[i] += 1
        # Normalize and scale
        grad = np.sign(grad) * 0.1
        return grad
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points/2) / h_sum
        return h
    
    def optimize_with_gradients(h_in, lambda_val, max_iter=1000, lr=0.1):
        h = h_in.copy()
        for i in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            if check_feasibility(h, lambda_val):
                return h
            # Compute gradient that would improve min correlation
            grad = compute_gradients(h, lambda_val)
            # Take gradient step
            h_new = h + lr * grad
            # Project to feasible region
            h = project_to_feasible(h_new)
        return h
    
    # Binary Search Loop
    while time() - start_time < budget_s * 0.95 and lambda_high - lambda_low > tol:
        lambda_mid = (lambda_low + lambda_high) / 2.0
        # Try to find feasible solution for lambda_mid
        try:
            # Start with the best feasible h from previous iteration
            if best_feasible_h is not None:
                h_trial = best_feasible_h.copy()
            else:
                h_trial = h_start.copy()
            
            # Optimize with gradients
            h_final = optimize_with_gradients(h_trial, lambda_mid)
            
            # Check feasibility
            if check_feasibility(h_final, lambda_mid):
                lambda_low = lambda_mid
                best_feasible_h = h_final.copy()
                best_feasible_lambda = lambda_mid
                # Update best solution
                if evaluate_erdos_solution(h_final, dx) < best_c5:
                    best_h = h_final.copy()
                    best_c5 = evaluate_erdos_solution(h_final, dx)
            else:
                lambda_high = lambda_mid
        except:
            lambda_high = lambda_mid
    
    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
