import numpy as np
from scipy.optimize import linprog
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        x = np.linspace(0, 2, n_points)
        h_start = np.exp(-((x - 1.0)**2) / 0.1) * 2.5
        h_start = np.clip(h_start, 0, 1)
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    precision = 1e-6
    max_iter = 60
    iter_count = 0
    
    while iter_count < max_iter and (high - low) > precision:
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Create optimization variables and constraints for feasibility check
        h_vars = cp.Variable(n_points)
        obj = cp.Minimize(0)
        constraints = [0 <= h_vars, h_vars <= 1]
        
        # Add integral constraint
        constraints.append(cp.sum(h_vars) == n_points / 2.0)
        
        # Add overlap constraints for this mid value
        # Since we're doing a feasibility check, we add constraints for all lags k
        # We'll approximate continuous domain with discrete convolution
        # The correlation is handled by translating the index
        # But we work in a matrix form to get all lag cases
        for k in range(n_points):
            # Compute the discrete correlation for lag k
            # For lags 0 to 2*dx our phase is at most 2-based on our discretization
            # So we compute sum h[i] * (1 - h[i+k]) for i such that both are valid
            # i should range from 0 to min(n_points-1, n_points-1-k)
            # but we use periodic or zero padding as required by n-correlation
            
            # For the lag k (in reverse pos and neg direction due to 2D correlation)
            # In numpy like correlation: valid mode for correlation gives this form:
            # result[k] = sum h[i] * (1 - h[i+k]) for i = 0 to n_points-1-k
            
            corr_sum = 0
            for i in range(n_points - k):
                corr_sum += h_vars[i] * (1 - h_vars[i + k])
            constraints.append(corr_sum <= mid * dx)
        
        # Formulate and solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-8, reltol=1e-8)
            
            if prob.status == cp.OPTIMAL:
                h_sol = h_vars.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    # Normalize manually in case of tolerance issues
                    h_sol = h_sol * (n_points/2) / np.sum(h_sol)
                    c5_val = evaluate_erdos_solution(h_sol, dx)
                    if c5_val < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_val
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except Exception as e:
            low = mid
            
        iter_count += 1

    # If the binary search did not give a good enough solution or

    # Step 1: Get current solution
    h_current = best_h.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simple hill climbing or local optimization
    # Whitelist popular optimizers that give good results for this constrained problem
    for iteration in range(50):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Small perturbation
        h_perturbed = h_current + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)

    final_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Return the final solution
    return (list(h_current), final_c5, n_points)
