import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Population parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    max_stagnation = 50
    
    # Initialize population
    population = []
    
    # Add perturbed versions of initial solution
    for _ in range(10):
        h_pert = initial_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random valid solutions
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fill up to population size
    while len(population) < population_size:
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Initialize Tabu list
    tabu_list = []
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0/n_points)
        fitness.append(fit)
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    stagnation_count = 0
    gen = 0
    
    while time() - start_time < budget_s * 0.95:
        gen += 1
        if stagnation_count > max_stagnation:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Select parents (tournament selection)
        new_population = [sorted_pop[0]]  # Keep best
        
        # Generate offspring
        while len(new_population) < population_size:
            # Crossover
            if random.random() < crossover_prob:
                parent1_idx = random.randint(0, population_size//2)
                parent2_idx = random.randint(0, population_size//2)
                parent1 = sorted_pop[parent1_idx]
                parent2 = sorted_pop[parent2_idx]
                
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Clone
                parent_idx = random.randint(0, population_size//2)
                child = sorted_pop[parent_idx].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, 
                                                   int(n_points * 0.1), 
                                                   replace=False)
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, 0.05)
                
            # Repair to feasible region
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Check tabu list
            tabu_violation = False
            for tabu_h in tabu_list:
                if np.allclose(child, tabu_h, rtol=1e-4):
                    tabu_violation = True
                    break
            
            if not tabu_violation:
                new_population.append(child)
            
            # Add to tabu list if needed
            if len(tabu_list) < tabu_limit:
                tabu_list.append(child)
            else:
                tabu_list.pop(0)
                tabu_list.append(child)
        
        # Local refinement every 10 generations
        if gen % 10 == 0:
            for i in range(min(5, len(new_population))):
                h_init = new_population[i]
                def objective(h):
                    return evaluate_erdos_solution(h, 2.0/n_points)
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        new_population[i] = h_refined
                        if c5_refined < sorted_fit[0]:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
        
        # Update population
        population = new_population
        # Recalculate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, 2.0/n_points)
            fitness.append(fit)
        
        # Adaptive parameters
        if stagnation_count > 30:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, 2.0/n_points)
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
