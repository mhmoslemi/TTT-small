import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Resample h_in to n_out points using linear interpolation."""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and 0<=h<=1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init, dx)
    
    # Multi-resolution strategy
    n_coarse = 50
    n_fine = 100
    coarse_dx = 2.0 / n_coarse
    fine_dx = 2.0 / n_fine
    
    # Initialize coarse and fine grids
    h_coarse = linear_resample(h_init, n_coarse)
    h_fine = linear_resample(h_init, n_fine)
    
    # Project to constraints
    h_coarse = project_to_constraints(h_coarse, n_coarse / 2)
    h_fine = project_to_constraints(h_fine, n_fine / 2)
    
    # Coarse-grid search
    coarse_steps = 100
    pert_scale_coarse = 0.05
    for step in range(coarse_steps):
        if time() - start_time > budget_s * 0.6:
            break
            
        # Random perturbation
        h_pert = h_coarse + np.random.normal(0, pert_scale_coarse, n_coarse)
        h_pert = project_to_constraints(h_pert, n_coarse / 2)
        
        # Evaluate
        c5_test = evaluate_erdos_solution(h_pert, coarse_dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = linear_resample(h_pert, n_points)
        
        # Update coarse solution
        h_coarse = h_pert
        pert_scale_coarse *= 0.99
    
    # Fine-grid refinement
    fine_steps = 200
    pert_scale_fine = 0.03
    for step in range(fine_steps):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb
        h_pert = h_fine + np.random.normal(0, pert_scale_fine, n_fine)
        h_pert = project_to_constraints(h_pert, n_fine / 2)
        
        # Evaluate
        c5_test = evaluate_erdos_solution(h_pert, fine_dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = linear_resample(h_pert, n_points)
        
        # Update fine solution
        h_fine = h_pert
        pert_scale_fine *= 0.995
    
    # Final local optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multiple restarts from the best found
    for restart in range(5):
        if time() - start_time > budget_s:
            break
            
        h_init = best_h + np.random.normal(0, 0.01, n_points)
        h_init = project_to_constraints(h_init, n_points / 2)
        
        try:
            res = minimize(objective, h_init, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final check and projection
    best_h = project_to_constraints(best_h, n_points / 2)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
