import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a 3-cluster structure
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_sizes = [3, 5, 3]
        for c, sz in zip(cluster_centers, cluster_sizes):
            start = max(0, int((c - sz/2)/dx))
            end = min(n_points, int((c + sz/2)/dx))
            h_start[start:end] = 1.0 / sz
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 20
    generations = 200
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.5:
            # Random perturbation of best solution
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
        else:
            # Fresh random construction
            h = np.random.rand(n_points)
            # Make cluster-like pattern
            for i in range(n_points // 7):
                idx = np.random.randint(0, n_points)
                h[max(0, idx-2):min(n_points, idx+2)] = 0.5
        # Normalize
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection (top 50%)
        selected = population[:pop_size//2]
        
        # Crossover and mutation to generate new population
        new_population = selected.copy()
        while len(new_population) < pop_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                # Crossover
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                h1, h2 = selected[i1], selected[i2]
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
                new_population.extend([child1, child2])
            else:
                # Mutation
                parent = selected[np.random.randint(len(selected))]
                child = parent.copy()
                mask = np.random.rand(n_points) < mutation_rate
                noise = np.random.normal(0, 0.02, n_points)
                child = child + noise * mask
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        # Keep only pop_size individuals
        population = new_population[:pop_size]
        
        # Local refinement with SQP for top individuals
        for i in range(min(5, len(population))):
            h_current = population[i].copy()
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < fitness[i]:
                        population[i] = h_refined.copy()
                        fitness[i] = c5_refined
            except:
                pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
