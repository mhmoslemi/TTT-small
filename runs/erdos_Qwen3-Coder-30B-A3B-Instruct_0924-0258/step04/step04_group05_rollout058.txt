import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    h_in = np.array(h_in)
    x_in = np.linspace(0, 2, len(h_in))
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_best = h_init.copy()
    c5_best = evaluate_erdos_solution(h_init, dx)
    
    # Coarse grid setup
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = linear_resample(h_init, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    
    # Fine grid setup
    n_fine = 100
    dx_fine = 2.0 / n_fine
    h_fine = linear_resample(h_init, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    # Coarse grid search
    steps_coarse = 100
    pert_scale_coarse = 0.05
    for i in range(steps_coarse):
        if time() - start_time > budget_s * 0.6:
            break
            
        h_test = h_coarse + np.random.normal(0, pert_scale_coarse, n_coarse)
        h_test = project_to_constraints(h_test, n_coarse)
        
        c5_test = evaluate_erdos_solution(h_test, dx_coarse)
        if c5_test < c5_best:
            c5_best = c5_test
            h_best = h_test.copy()
            
        h_coarse = h_test.copy()
        pert_scale_coarse *= 0.99
    
    # Fine grid refinement
    steps_fine = 200
    pert_scale_fine = 0.03
    for i in range(steps_fine):
        if time() - start_time > budget_s * 0.9:
            break
            
        h_test = h_fine + np.random.normal(0, pert_scale_fine, n_fine)
        h_test = project_to_constraints(h_test, n_fine)
        
        c5_test = evaluate_erdos_solution(h_test, dx_fine)
        if c5_test < c5_best:
            c5_best = c5_test
            h_best = h_test.copy()
            
        h_fine = h_test.copy()
        pert_scale_fine *= 0.99
    
    # Alternating search between coarse and fine grids
    steps_alt = 100
    pert_scale_alt = 0.03
    for i in range(steps_alt):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Coarse to fine
        h_fine_new = linear_resample(h_coarse, n_fine)
        h_fine_new = project_to_constraints(h_fine_new, n_fine)
        h_fine_new += np.random.normal(0, pert_scale_alt, n_fine)
        h_fine_new = project_to_constraints(h_fine_new, n_fine)
        
        c5_test = evaluate_erdos_solution(h_fine_new, dx_fine)
        if c5_test < c5_best:
            c5_best = c5_test
            h_best = h_fine_new.copy()
            h_fine = h_fine_new.copy()
        
        # Fine to coarse
        h_coarse_new = linear_resample(h_fine, n_coarse)
        h_coarse_new = project_to_constraints(h_coarse_new, n_coarse)
        h_coarse_new += np.random.normal(0, pert_scale_alt, n_coarse)
        h_coarse_new = project_to_constraints(h_coarse_new, n_coarse)
        
        c5_test = evaluate_erdos_solution(h_coarse_new, dx_coarse)
        if c5_test < c5_best:
            c5_best = c5_test
            h_best = h_coarse_new.copy()
            h_coarse = h_coarse_new.copy()
            
        pert_scale_alt *= 0.99
    
    # Final local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        h_final = h_best.copy()
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < c5_best:
                    c5_best = c5_refined
                    h_best = h_refined
        except:
            pass
    
    return (list(h_best), c5_best, n_points)
