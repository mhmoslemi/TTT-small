import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimizer: L-BFGS with constraint handling
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def optimize_with_restart(h0):
        # Normalize initial values
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h0, method='L-BFGS-B', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                return h_final, c5_final
        except:
            pass
        return h0, evaluate_erdos_solution(h0, dx)
    
    # Refinement loop
    max_refinements = 20
    refinement_count = 0
    
    while refinement_count < max_refinements and (time() - start_time) < budget_s * 0.95:
        # Random restarts
        for _ in range(3):
            if (time() - start_time) > budget_s * 0.95:
                break
                
            # Add noise to current best
            h_pert = best_h + np.random.normal(0, 0.05, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            h_refined, c5_refined = optimize_with_restart(h_pert)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        
        refinement_count += 1
        
        # Try a more systematic approach with better initialization
        if refinement_count % 4 == 0 and n_points < 50 and (time() - start_time) < budget_s * 0.9:
            # Increase resolution
            n_points = 50
            dx = 2.0 / n_points
            h_new = np.zeros(n_points)
            # Resample from current best
            for i in range(n_points):
                idx = int(i * len(best_h) / n_points)
                h_new[i] = best_h[min(idx, len(best_h)-1)]
            
            # Refine
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_refined, c5_refined = optimize_with_restart(h_new)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    
    # Final refinement with L-BFGS on best
    if (time() - start_time) < budget_s * 0.1:
        h_final, c5_final = optimize_with_restart(best_h)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
