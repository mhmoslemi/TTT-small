import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        h_start[5:15] = 1.0
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization (n_points=20)
    max_iter = 50
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.4:
            break
            
        h_pert = best_h + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        old_c5 = evaluate_erdos_solution(best_h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            best_h = h_pert
            best_c5 = new_c5
            
        temp *= cool_rate
    
    # Upsample to 40 points
    if n_points < 40:
        n_points = 40
        h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
        best_h = h_upsampled
        dx = 2.0 / n_points
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine optimization (n_points=40)
    for iteration in range(100):
        if time() - start_time > budget_s * 0.8:
            break
            
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                best_h = res.x
                best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass
    
    # Final refinement with gradient descent approach
    max_iter = 50
    step_size = 1e-3
    for iteration in range(max_iter):
        if time() - start_time > budget_s:
            break
            
        # Gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-6
        
        # Compute gradient numerically
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            
            h_minus = best_h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            
            val_plus = evaluate_erdos_solution(h_plus, dx)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            grad[i] = (val_plus - val_minus) / (2 * eps)
        
        # Update with gradient descent
        best_h = best_h - step_size * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
