import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Linearly resample h_in from old length to new length"""
    if len(h_in) == n_out:
        return h_in.copy()
    x_in = np.linspace(0, 2, len(h_in))
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] with sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Multi-resolution approach: coarse grid then fine grid
    n_coarse = 50
    n_fine = 100
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine

    # Start with initial solution
    h_solution = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_solution, dx)
    best_h = h_solution.copy()

    # Coarse grid exploration
    h_coarse = linear_resample(h_solution, n_coarse)
    h_coarse = project_feasible(h_coarse, n_coarse / 2.0)
    c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    
    if c5_coarse < best_c5:
        best_c5 = c5_coarse
        best_h = h_coarse.copy()

    # Grid search over coarse grid
    for step in range(100):
        if time() - start_time > budget_s * 0.6:
            break
            
        # Slight perturbation
        pert = np.random.normal(0, 0.03, n_coarse)
        h_test = h_coarse + pert
        h_test = project_feasible(h_test, n_coarse / 2.0)
        
        c5_test = evaluate_erdos_solution(h_test, dx_coarse)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            
        # Accept with some noise
        if c5_test < c5_coarse:
            h_coarse = h_test.copy()
            c5_coarse = c5_test
            
    # Fine grid refinement 
    h_fine = linear_resample(best_h, n_fine)
    h_fine = project_feasible(h_fine, n_fine / 2.0)
    c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    if c5_fine < best_c5:
        best_c5 = c5_fine
        best_h = h_fine.copy()

    # Local optimization in fine space
    def objective(h):
        return evaluate_erdos_solution(h, dx_fine)

    def constraint_sum(h):
        return np.sum(h) - n_fine / 2.0

    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_fine)]
    
    # Multiple restarts with refined perturbations
    for restart in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_init = h_fine.copy()
        # Warp slightly
        h_init += np.random.normal(0, 0.02, n_fine)
        h_init = project_feasible(h_init, n_fine / 2.0)
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Convert back to original resolution and final feasibility check
    h_result = linear_resample(best_h, n_points)
    h_result = project_feasible(h_result, n_points / 2.0)
    final_c5 = evaluate_erdos_solution(h_result, dx)
    
    return (list(h_result), final_c5, n_points)
