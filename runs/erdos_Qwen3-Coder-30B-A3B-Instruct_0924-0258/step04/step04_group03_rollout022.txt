import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization with refinement
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitivity = 2.0
    sociality = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    # Start with the initial solution + some perturbations
    for i in range(swarm_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Perturb the best solution
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        value = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_values.append(value)
    
    # Global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                             cognitivity * r1 * (personal_best[i] - swarm[i]) +
                             sociality * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Clamp to [0,1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Normalize to correct sum
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = value
                
                # Update global best
                if value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = value
    
    # Final refinement with convex optimization approach
    # Solve the problem via optimization with affine approximations
    try:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Estimate a good upper bound
        corr_upper_bound = global_best_value * 2
        # Add soft constraints for first few kernel values
        for k in range(1, min(12, 2 * n_points - 1), 2):
            if k < n_points:
                # Convolution for k shift
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                # This case should not happen given our indices
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            # Approximate the max constraint with a soft one
            constraints.append(corr_sum <= corr_upper_bound * dx * 0.5)
        
        # Objective: minimize the max overlap
        # Since CVX can't directly minimize max, we add a slack variable
        max_overlaps = []
        for k in range(2 * n_points - 1):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            max_overlaps.append(corr_sum)
        
        # Force one of the correlations to be below threshold
        # This is not direct optimization of maximum overlap,
        # but helps direct search to feasible solutions
        slack = cp.Variable()
        for corr_sum in max_overlaps:
            constraints.append(corr_sum <= 0.5 + slack)
        # Minimize the slack or a proxy
        prob = cp.Problem(cp.Minimize(slack), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    except Exception:
        pass
    
    # Final polish with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Start from the best particle of PSO
        try:
            res = minimize(objective, global_best, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
