import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with logistic reparameterization
    def logistic(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def logit(p):
        return np.log(p / (1.0 - p))
    
    # Initialize z parameterization
    z = logit(h_start)
    # Adjust b to satisfy sum constraint
    def compute_b(z):
        h = logistic(z)
        return np.log((n_points / 2) / np.sum(h))
    
    b = compute_b(z)
    h = logistic(z + b)
    
    # Compute overlaps and gradients
    def compute_overlap_and_grad(z, b):
        h = logistic(z + b)
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        # Compute gradient of max overlap w.r.t h
        # For shift k_max, we just take the derivative of the cross-correlation term
        # The derivative of h[i](1-h[i+k]) is (1-2h[i]) if we're differentiating w.r.t h[i] 
        # or -h[i+k] if we're differentiating w.r.t. h[i+k]
        # But this is tricky due to the convolution structure
        # For now, we estimate it using finite differences
        h_grad = np.zeros_like(h)
        for i in range(len(h)):
            h_shifted = h.copy()
            h_shifted[i] += 1e-6
            corr_plus = np.correlate(h_shifted, 1-h, mode='full')
            h_grad[i] = (corr_plus[max_k] - corr[max_k]) / 1e-6
        return corr[max_k], h_grad
    
    # Gradient step with backtracking line search
    max_iter = min(10000, int(1000000 / n_points))
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.98:
            break
            
        # Compute current overlap and gradient
        corr_max, grad_h = compute_overlap_and_grad(z, b) 
        
        # Now convert gradient to z space
        # dh/dz = sigmoid(z+b)(1-sigmoid(z+b))
        h = logistic(z + b)
        dh_dz = h * (1 - h)
        
        # Update z using gradient descent
        z_new = z - step_size * grad_h / (dh_dz + 1e-10)
        
        # Adjust b to keep sum constraint
        b_new = compute_b(z_new)
        h_new = logistic(z_new + b_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            z = z_new.copy()
            b = b_new
        else:
            z = z_new.copy()
            b = b_new
    
    # Final refinement using trust-constr
    try:
        def objective(h):
            h = np.clip(h, 0, 1)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(objective, best_h, method='trust-constr', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'gtol': 1e-8, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
