import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h, n_new):
    """Resample h to n_new points using linear interpolation"""
    h = np.array(h)
    old_indices = np.linspace(0, len(h)-1, len(h))
    new_indices = np.linspace(0, len(h)-1, n_new)
    return np.interp(new_indices, old_indices, h)

def project_to_constraints(h, n_points):
    """Project h to feasible region: [0,1] and sum==n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize from existing solution or create new
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Coarse grid setup for global search
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = linear_resample(h_start, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    
    # Fine grid setup for local refinement
    n_fine = 100
    dx_fine = 2.0 / n_fine
    h_fine = linear_resample(h_start, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    # Track best solutions
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    best_coarse = h_coarse.copy()
    best_fine = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(best_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(best_fine, dx_fine)
    
    # Coarse grid global search
    search_steps = 100
    perturb_scale = 0.03
    
    for step in range(search_steps):
        if time() - start_time > budget_s * 0.4:
            break
            
        # Perturb coarse solution
        h_perturbed = h_coarse + np.random.normal(0, perturb_scale, n_coarse)
        h_perturbed = project_to_constraints(h_perturbed, n_coarse)
        
        # Evaluate
        c5_val = evaluate_erdos_solution(h_perturbed, dx_coarse)
        
        # Accept if better
        if c5_val < best_c5_coarse:
            best_c5_coarse = c5_val
            best_coarse = h_perturbed.copy()
            
        # Transfer to fine grid for refinement
        h_fine_from_coarse = linear_resample(best_coarse, n_fine)
        h_fine_from_coarse = project_to_constraints(h_fine_from_coarse, n_fine)
        
        # Local refinement
        def objective(h_refine):
            return evaluate_erdos_solution(h_refine, dx_fine)
        
        def constraint_sum(h):
            return np.sum(h) - n_fine / 2.0
            
        bounds = [(0, 1) for _ in range(n_fine)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Refine with local optimization
        try:
            res = minimize(objective, h_fine_from_coarse, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5_fine:
                    best_c5_fine = c5_refined
                    best_fine = h_refined.copy()
        except:
            pass
            
        # Update global best
        if best_c5_fine < best_c5:
            best_c5 = best_c5_fine
            # Interpolate back to main resolution
            h_interp = linear_resample(best_fine, n_points)
            best_h = project_to_constraints(h_interp, n_points)
            
        # Adaptive scaling
        if step % 10 == 0 and perturb_scale > 0.005:
            perturb_scale *= 0.99
            
    # Fine-grid refinement using best solution so far
    def objective_final(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum_final(h):
        return np.sum(h) - n_points / 2.0
        
    bounds_final = [(0, 1) for _ in range(n_points)]
    cons_sum_final = {'type': 'eq', 'fun': constraint_sum_final}
    
    # Start refinement from best solution
    h_refine_init = best_h.copy()
    
    # Multiple restarts
    for restart in range(3):
        if time() - start_time > budget_s * 0.9:
            break
            
        if restart > 0:
            # Perturb initial solution
            h_init = h_refine_init + np.random.normal(0, 0.01, n_points)
            h_init = project_to_constraints(h_init, n_points)
        else:
            h_init = h_refine_init.copy()
            
        try:
            res = minimize(objective_final, h_init, method='SLSQP',
                           bounds=bounds_final, constraints=cons_sum_final,
                           options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective_final(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
