import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_old, n_new):
    """Linear resampling from old to new grid"""
    n_old = len(h_old)
    x_old = np.arange(n_old) * (2.0 / n_old)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def project_to_constraints(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5

    # Resampling to fine grid for optimization (100 points)
    n_fine = 100
    h_fine = linear_resample(h_init, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    dx_fine = 2.0 / n_fine

    # Best solution tracking
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx_fine)
    h_best_coarse = None
    best_c5_coarse = float('inf')
    
    # Coarse grid for global search (50 points)
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    
    # Steps for optimization
    coarse_steps = 100
    fine_steps = 200
    max_perturbation = 0.05
    perturbation_decay = 0.99

    # Coarse grid search
    h_coarse = linear_resample(h_fine, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    
    if coarse_steps > 0:
        for step in range(coarse_steps):
            if time() - start_time > budget_s * 0.7:
                break
                
            # Perturb coarse solution
            perturbation = np.random.normal(0, max_perturbation, n_coarse)
            h_trial = h_coarse + perturbation
            h_trial = project_to_constraints(h_trial, n_coarse)
            
            # Evaluate
            c5_trial = evaluate_erdos_solution(h_trial, dx_coarse)
            if c5_trial < best_c5_coarse:
                best_c5_coarse = c5_trial
                h_best_coarse = h_trial.copy()
            
            # Update perturbation scale
            max_perturbation *= perturbation_decay

    # Fine grid refinement
    if fine_steps > 0:
        # Start from coarse grid solution interpolated to fine grid
        base_h = h_best_coarse if h_best_coarse is not None else h_coarse
        h_refine = linear_resample(base_h, n_fine)
        h_refine = project_to_constraints(h_refine, n_fine)
        
        for step in range(fine_steps):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb around current solution
            perturbation = np.random.normal(0, max_perturbation/2, n_fine)
            h_trial = h_refine + perturbation
            h_trial = project_to_constraints(h_trial, n_fine)
            
            # Evaluate
            c5_trial = evaluate_erdos_solution(h_trial, dx_fine)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            
            # Update perturbation
            max_perturbation *= perturbation_decay

    # Final refinement - gradient descent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx_fine)
        
        def constraint_sum(h):
            return np.sum(h) - n_fine / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_fine)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    # Final validation - project ensuring feasibility 
    final_h = project_to_constraints(best_h, n_fine)
    final_c5 = evaluate_erdos_solution(final_h, dx_fine)
    
    # Return to original grid size
    ret_h = linear_resample(final_h, n_points)
    ret_h = project_to_constraints(ret_h, n_points)
    ret_c5 = evaluate_erdos_solution(ret_h, 2.0 / n_points)
    
    return (list(ret_h), ret_c5, n_points)
