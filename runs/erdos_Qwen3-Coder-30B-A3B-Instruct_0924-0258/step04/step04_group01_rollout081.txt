import numpy as np
from cvxpy import *
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create initial guess with piecewise constant structure
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place 2 spikes
        h_start[20:30] = 0.5
        h_start[70:80] = 0.5
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    
    # Binary search on C5
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    best_feasible_h = None
    best_feasible_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate LP feasibility problem
        h_vars = Variable(n_points)
        
        # Objective: minimize 0 (feasibility check)
        objective = Minimize(0)
        
        # Constraints: ∑ h[a](1 - h[a + k]) ≤ mid / dx for all valid k
        constraints = []
        
        # Sum constraint
        constraints.append(sum(h_vars) == n_points/2)
        
        # Bound constraints
        constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
        constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
        
        # Correlation constraints (for all lags k)
        # We have 2*n_points - 1 possible lags
        # For each lag k, we must have the correlation <= mid / dx
        full_corr = np.full(2*n_points-1, mid/dx)
        
        # Now add actual constraints based on definition of convolution
        for k in range(2*n_points-1):
            # Sum over all valid indices a such that a+k is also valid
            # Make sure that for each lag k, we compute the correct convolution term
            # The convolution is defined as sum(h[a] * (1 - h[a+k]), over all valid a)
            # for L = 2*n_points-1, we have indices a from 0 to n_points-1 and k from 0 to L-1
            
            # Only add constraint if it doesn't make sense (e.g., negative index)  
            # Actually, let's be more careful here since correlation computation is
            # \sum_{a} h[a] * (1 - h[a+k]) where the indices wrap around  
            # But this should be done in full correlation, not truncated
            
            left = max(0, k - n_points + 1)  # start index for valid convolution
            right = min(n_points - 1, k)     # end index for valid convolution
            # Direct: Convolution with 1-h
            conv_sum = 0
            valid_indices = []
            for a in range(n_points):
                shifted_a = a + k
                # We're using circular convolution effectively,  
                # but for now, we'll use correlation over finite interval
                if (0 <= shifted_a < n_points):
                    conv_sum += h_vars[a] * (1 - h_vars[shifted_a])
            if k < 2*n_points-1:
                constraints.append(conv_sum <= mid/dx)
        
        # Actually, re-approaching: treat all contraction constraints directly in a loop
        # We need to process all lags correctly
        # Let's correctly form all these constraints using full cross-product approach
        # Compute correlations manually in the LP constraints
        
        # We can start over and think of the constraints properly: correlation of h with (1-h)
        # correlation = sum over all indices a, b such that a+b = k (mod n) of h[a] * (1 - h[b])
        # But we are given lag k, so loop through all a and a+k among valid range:
        # a: 0 to n_points-1
        # then check if a+k is also valid
        
        # This should be computed correctly as: for each delta k
        # contribute sum_{a = 0}^{n_points-1} h[a] * (1 - h[a + k]) 
        # where a+k is mod n_points when not in range for symmetric convolution or 
        # we do it just as the axpy  as in one or the original problem  
        
        # Plus, let's not take full correlation, that wants wrapping and is complex
        # just do the integral like in task, follows convention
    
        # Recompute correlation moment by moment in loop:
        # compute for each k (minus circular rebounds)  
        corr_constraints = []
        
        for k in range(1, 2*n_points-1):  # total number of unique lags for convolution shape
            # For each k, compute the summation:
            total = 0
            # the original algorithms used 1-dimensional convolution corr = h * (1-h) 
            # but all the terms being considered are h[a]* (1 - h[a+k]) which means
            # a can be from 0 to n_points-1 and a+k must be from 0 to n_points-1
            # That defines valid range, valid ranges are for a in [max(0, k-np+1), min(n_points-1, k)]
            for a in range(max(0, k-n_points+1), min(n_points, k)):
                total += h_vars[a] * (1 - h_vars[a + k])
            corr_constraints.append(total <= mid / dx)
            
        constraints.extend(corr_constraints)
        
        # Solve
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=CVXOPT, verbose=False, feastol=1e-6)
            if prob.status == 'optimal':
                # Extract solution
                h_sol = np.array([float(h_vars[i].value) for i in range(n_points)])
                # Check constraints aren't violated significantly
                c5_sol = evaluate_erdos_solution(h_sol, dx)
                if c5_sol < best_feasible_c5:
                    best_feasible_h = h_sol.copy()
                    best_feasible_c5 = c5_sol
                    
                # Set to new high if feasible and near enough  
                high = mid
            else:
                # If infeasible, set new low
                low = mid
        except Exception:
            # In case of solver issue, try a different approach
            # sample a few optimized routes from best solution
            improved = False
            if iteration > 10:
                # Use refinement from best known
                h_candidate = best_h.copy()
                # Limit noisy perturbations
                noise = np.random.normal(0, 0.005, n_points)
                h_candidate += noise
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                
                try:
                    # Grid search refinement in a small ball
                    def obj(hvals):
                        return evaluate_erdos_solution(hvals, dx)
                    bounds = [(0, 1) for _ in range(n_points)]
                    res = minimize(obj, h_candidate, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-5})
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
                            improved = True
                except:
                    pass
                    
                if not improved:
                    low = mid
            else:
                low = mid
        # Early stopping if tolerance is good
        if abs(high - low) < precision:
            break
    
    if best_feasible_h is None:
        best_feasible_h = best_h.copy()
        best_feasible_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_feasible_h), best_feasible_c5, n_points)
