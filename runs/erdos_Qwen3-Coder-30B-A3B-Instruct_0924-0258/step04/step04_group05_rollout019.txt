import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    max_stagnation = 50
    max_gens = 500
    
    # Initialize population
    population = []
    fitness = []
    tabu_list = []
    
    # Initialize with perturbed versions of the best
    for i in range(30):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Add some random solutions
    for i in range(10):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Evolution loop
    stagnation_counter = 0
    current_best = min(fitness)
    
    for gen in range(max_gens):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
            if best_c5 < 0.3807:
                break
        
        # Check stagnation
        if sorted_fit[0] < current_best:
            current_best = sorted_fit[0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > max_stagnation:
            # Increase mutation, decrease crossover if stagnating
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            stagnation_counter = 0
        else:
            # Decrease mutation, increase crossover if making progress
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # New population
        new_pop = [sorted_pop[0], sorted_pop[1]]  # Elitism
        
        # Crossover and mutation
        while len(new_pop) < population_size:
            # Tournament selection
            idx1, idx2 = random.sample(range(len(sorted_pop)), 2)
            parent1 = sorted_pop[idx1] if fitness[idx1] < fitness[idx2] else sorted_pop[idx2]
            
            if random.random() < crossover_prob:
                idx3, idx4 = random.sample(range(len(sorted_pop)), 2)
                parent2 = sorted_pop[idx3] if fitness[idx3] < fitness[idx4] else sorted_pop[idx4]
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                for i in range(n_points):
                    if random.random() < 0.1:
                        child[i] += np.random.normal(0, 0.03)
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            new_pop.append(child)
        
        # Local refinement every 10 generations
        if gen % 10 == 0 and len(new_pop) >= 5:
            for i in range(min(5, len(new_pop))):
                h_init = new_pop[i]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[sorted_indices[i]]:
                            new_pop[i] = h_refined
                except:
                    pass
        
        # Update population and fitness
        population = new_pop
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
