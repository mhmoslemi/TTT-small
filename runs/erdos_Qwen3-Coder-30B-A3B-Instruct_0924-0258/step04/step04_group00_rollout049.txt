import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # We'll use projected gradient descent with adaptive step size
    max_iter = 1000
    alpha = 0.01
    patience = 0
    patience_limit = 50
    h_current = best_h.copy()
    last_improvement = 0
    
    def compute_gradient(h, c5_val):
        # Compute the gradient of the overlap with respect to h
        # We're computing the gradient of max_k integral h(x)(1-h(x+k))dx
        # For simplicity, we'll compute the gradient numerically for now
        # A more precise approach would compute it for each k separately
        # But for efficiency, we simplify:
        grad = np.zeros_like(h)
        n = len(h)
        for k in range(1, n):
            if k < n:
                for i in range(n - k):
                    grad[i] += (1 - h[i + k]) * dx
                    grad[i + k] -= h[i] * dx  # negative because of (1-h(x+k))
        # Add penalty term for sum constraint: lambda * (sum(h) - n_points/2)
        # Lagrange multiplier is implicitly handled via projected gradient descent
        grad += 2.0 * (np.sum(h) - n_points / 2.0) / n_points

        # Normalize and ensure it's a descent direction
        norm = np.linalg.norm(grad)
        if norm > 1e-12:
            grad = grad / norm
        return grad

    def project_h(h):
        # Project onto constraints: 0 <= h <= 1 and sum(h) = n_points / 2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum if h_sum > 0 else h
        return h

    iter_count = 0

    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1

        # Compute gradient
        try:
            grad = compute_gradient(h_current, best_c5)
        except Exception:
            grad = np.random.normal(0, 0.01, n_points)
        
        # Perform gradient step
        h_new = h_current - alpha * grad
        
        # Project back to feasible set
        h_new = project_h(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if improvement
        if c5_new < best_c5:
            h_current = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
            # Increase step size
            alpha = min(alpha * 1.2, 1.0)  # Cap at 1.0
            patience = 0
        else:
            patience += 1
            # Reduce step size and revert
            alpha *= 0.5
            if patience > patience_limit:
                break
                
    # Final refinement: try smooth refinement for better results
    try:
        h_refined = best_h.copy()
        # Smooth the solution using a moving average filter
        kernel = np.ones(5) / 5
        h_smoothed = np.convolve(h_refined, kernel, mode='same')
        h_smoothed = np.clip(h_smoothed, 0, 1)
        h_normalized = h_smoothed * (n_points / 2) / np.sum(h_smoothed)
        c5_refined = evaluate_erdos_solution(h_normalized, dx)
        if c5_refined < best_c5:
            best_h = h_normalized
            best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
