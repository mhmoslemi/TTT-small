import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 40
    dx = 2.0 / n_points
    
    # Initialize from existing construction or uniform
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        # Interpolate/reduce to n_points
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, n_points)
            h_start = np.interp(x_new, x_orig, h_start)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to correct integral
    h_start = h_start * (n_points / 2.0) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid search
    grid_size = 20
    search_range = np.linspace(0.36, min(0.38085, best_c5), grid_size)
    last_improvement = 0
    c5_improved = False
    
    for i, c_mid in enumerate(search_range):
        if time() - start_time > budget_s * 0.45:
            break
            
        # CVXPY feasibility check
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Check overlap constraints more efficiently
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_c5 = c5_feasible
                    best_h = h_feasible.copy()
                    last_improvement = i
                    c5_improved = True
        except:
            continue
    
    # Refinement with SLSQP and COBYLA
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multiple restarts with varying perturbations
    for restart in range(20):
        if time() - start_time > budget_s * 0.9:
            break
            
        if restart == 0:
            h_init = best_h.copy()
        else:
            # Varying perturbation strength and method
            strength = 0.01 + np.random.random() * 0.04
            h_init = best_h + np.random.normal(0, strength, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            
            # Add some structure to diversity
            if np.random.random() > 0.5:
                # Roughly increase/decrease
                step = np.random.randint(1, 5)
                h_init[::step] = np.clip(h_init[::step] + np.random.random() * 0.1, 0, 1)
        
        try:
            # Try multiple methods
            methods = ['SLSQP', 'COBYLA']
            for method in methods:
                if time() - start_time > budget_s * 0.95:
                    break
                res = minimize(objective, h_init, method=method, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 150, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
                        if best_c5 < 0.3807:
                            break
        except:
            continue
    
    # Final known good solution refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        try:
            # Fine tuning with stricter tolerances
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final projection to validate constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
