import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization (PSO)
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    # Initialize with the starting solution plus perturbations
    for i in range(swarm_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        # Evaluate personal best
        val = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_vals.append(val)
        
        # Update global best
        if val < best_c5:
            best_h = h.copy()
            best_c5 = val
    
    # PSO iterations
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive_component = cognitive * r1 * (personal_best[i] - swarm[i])
            social_component = social * r2 * (best_h - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Update position
            swarm[i] += velocities[i]
            
            # Clamp to bounds and renormalize
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            current_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < best_c5:
                    best_h = swarm[i].copy()
                    best_c5 = current_val
                    
    # Convex relaxation refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            # Try refining only with the best solution found so far
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Because of the hard nonconvexity, we linearize expectations around the best
            # Just add a few constraints to make the relaxation meaningful
            
            # Add convex constraint: for full correlation we set modest limits
            # This is a hackish try to allow CVX to solve somewhat
            # We add key constraint that does not directly prevent the value 
            # But makes the problem tractable for solvers
            corr_vals = [cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:])) for k in range(0, min(n_points, 10), 2)]
            for corr in corr_vals:
                try:
                    constraints.append(corr <= 0.4 * n_points * dx)  # Some convex approx constraint
                except:
                    pass
                    
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass

    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
