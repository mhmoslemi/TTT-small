import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    tournament_size = 2
    
    # Initialize population with feasible candidates
    population = []
    fitness_scores = []
    
    # Add initial solution
    population.append(h_start.copy())
    fitness_scores.append(evaluate_erdos_solution(h_start, dx))
    
    # Generate additional random candidates
    for _ in range(population_size - 1):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        h_random = np.clip(h_random, 0, 1)
        population.append(h_random)
        fitness_scores.append(evaluate_erdos_solution(h_random, dx))
    
    # Genetic algorithm loop
    generation = 0
    while time() - start_time < budget_s * 0.8 and generation < num_generations:
        # Tournament selection
        selected_indices = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_indices.append(winner_idx)
        
        # Create new population through crossover and mutation
        new_population = []
        new_fitness = []
        
        # Elitism: keep the best solution
        best_idx = np.argmin(fitness_scores)
        new_population.append(population[best_idx].copy())
        new_fitness.append(fitness_scores[best_idx])
        
        # Generate offspring
        for i in range(1, population_size // 2):
            parent1_idx = selected_indices[2*i-1]
            parent2_idx = selected_indices[2*i]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Uniform crossover
            crossover_point = random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Repair and mutate
            for child in [child1, child2]:
                # Clamp and normalize
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                # Mutation
                if random.random() < mutation_rate:
                    mutation_indices = random.sample(range(n_points), max(1, int(n_points * mutation_rate)))
                    for idx in mutation_indices:
                        child[idx] += np.random.normal(0, 0.01)
                    child = np.clip(child, 0, 1)
                    # Re-normalize
                    child = child * (n_points / 2) / np.sum(child)
                
                # Evaluate fitness
                fitness = evaluate_erdos_solution(child, dx)
                new_population.append(child)
                new_fitness.append(fitness)
        
        # Replace population
        population = new_population
        fitness_scores = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_c5 = fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        generation += 1
    
    # Final refinement
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
