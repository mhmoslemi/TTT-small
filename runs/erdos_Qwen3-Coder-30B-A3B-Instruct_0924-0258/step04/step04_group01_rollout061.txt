import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        # Create a structured initial guess with strategic clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Direct BOBYQA optimization with proper constraints
    def objective(h):
        # Avoid recomputing normalization
        h_scaled = h * (n_points/2) / np.sum(h)
        return evaluate_erdos_solution(h_scaled, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Use BOBYQA for derivative-free optimization
    try:
        res = minimize(objective, h_start, method='BOBYQA', 
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 500, 'rhobeg': 0.5, 'tol': 1e-6})
        if res.success:
            h_final = res.x
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # If BOBYQA didn't improve, do final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
