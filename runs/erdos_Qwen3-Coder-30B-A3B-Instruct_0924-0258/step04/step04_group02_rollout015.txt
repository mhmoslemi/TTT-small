import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with initial solution
    h_current = h_start.copy()
    
    # Gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # We'll use L-BFGS-B since it handles bounds naturally
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    max_iter = 1000
    iter_count = 0
    
    # Try multiple random restarts to escape local minima
    restarts = 5
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation of current best solution
        h_restart = best_h + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        # Normalize
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
        
        iter_count += 1
    
    # Local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 300, 'ftol': 1e-9, 'gtol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
