import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place a few spikes of height 1
        h_start[10:15] = 1.0
        h_start[20:25] = 1.0
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def solve_convex_relaxation(h_current, lambda_vals):
        h = cp.Variable(n_points)
        obj = 0
        for i in range(len(lambda_vals)):
            # Compute convolution manually
            k = i - (n_points - 1)
            if k == 0:
                conv = h_current
            elif k > 0:
                conv = np.concatenate([h_current[k:], np.zeros(k)])
            else:
                conv = np.concatenate([np.zeros(-k), h_current[:k]])
            obj += lambda_vals[i] * cp.sum(cp.multiply(h, 1 - conv))
        # Entropy penalty term
        entropy = cp.sum(cp.multiply(lambda_vals, cp.log(lambda_vals + 1e-10)))
        objective = cp.Minimize(obj - entropy)
        constraints = [
            cp.sum(h) == n_points/2,
            h >= 0,
            h <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h.value
                return new_h
        except:
            pass
        return h_current.copy()
    
    # Alternating optimization
    lambda_vals = np.ones(2*n_points - 1)
    
    for iteration in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update lambda values based on current h
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        lambda_vals = np.maximum(lambda_vals * (overlaps / (lambda_vals + 1e-10)), 1e-6)
        
        # Solve convex relaxation
        new_h = solve_convex_relaxation(best_h, lambda_vals)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new candidate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
    
    # Final refinement using scipy
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
