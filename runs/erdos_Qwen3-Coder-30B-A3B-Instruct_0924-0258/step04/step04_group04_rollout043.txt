import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place values in two clusters that avoid overlap
        h_start[5:15] = 0.5  # First cluster
        h_start[20:30] = 0.5 # Second cluster
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    # Parameters for alternating optimization
    max_iter = 100
    n_active = max(1, n_points // 10)  # How many shifts to update at a time
    dual_step = 0.1
    
    # Dual variables
    lambda_k = np.ones(2*n_points - 1)
    
    def optimize_h(h_in, lambda_k):
        """Solve the convex relaxation using CVXPY"""
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(lambda_k, 
            [cp.quad_form(h, np.eye(n_points)) - 
             cp.sum(cp.multiply(h, np.roll(1-h_in, k)) * 2.0/n_points) 
             for k in range(-n_points+1, n_points)])))
        constraints = [cp.sum(h) == n_points/2,
                       h >= 0,
                       h <= 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=500)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return h_in
    
    def update_lambdas(h, lambda_k):
        overlaps = np.correlate(h, 1 - h, mode='full') * 2.0/n_points
        lambda_k = np.maximum(lambda_k * np.exp(dual_step * (overlaps - lambda_k)), 1e-6)
        return lambda_k

    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * 2.0/n_points
        current_c5 = np.max(overlaps)
        
        # Update duals
        lambda_k = np.maximum(lambda_k * np.exp(dual_step * (overlaps - lambda_k)), 1e-6)
        
        # Update h
        h_new = optimize_h(best_h, lambda_k)
        if h_new is not None:
            best_h = h_new.copy()
            
        # Project to feasible region
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            
        # Occasionally refine with local optimization
        if iteration % 5 == 0:
            def obj(h):
                return evaluate_erdos_solution(h, 2.0/n_points)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                    refined_c5 = evaluate_erdos_solution(refined_h, 2.0/n_points)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
            except:
                pass
    
    # Final validation and polishing
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, 2.0/n_points)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # Ensure sum constraint
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    return (list(best_h), best_c5, n_points)
