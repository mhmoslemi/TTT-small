import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient w.r.t. h of overlap at shift k
        # For overlap integral at shift k: sum h[i] * (1 - h[i+k])
        # Gradient d/dh[i] = 1 - h[i+k] if i+k < n_points
        # For 1D convolution: overlap = sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k])
        grad = np.zeros(n_points)
        if k < n_points:
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
        return grad
    
    # Keep track of best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive gradient descent
    current_h = h_start.copy()
    alpha = 0.1
    max_iter = 5000
    decay_rate = 0.99
    patience = 1000
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Determine the maximum overlap
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Compute gradient for the largest overlap shift
        if max_k < len(corr):
            k = max_k - (n_points - 1)
        else:
            k = max_k - (n_points - 1) - len(corr) + 1
        
        # Compute the actual shift for correlation index
        actual_shift = max_k
        grad = compute_gradient(current_h, actual_shift)
        
        # Update h
        new_h = current_h - alpha * grad
        
        # Project to constraints
        # Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Adjust to maintain sum constraint
        current_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2) / current_sum
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update current solution
        current_h = new_h.copy()
        
        # Adaptive learning rate
        if patience_counter > patience:
            alpha *= 0.8
            patience_counter = 0
            
        alpha = max(alpha, 1e-5)
        
        # Every 50 iterations, do convex refinement
        if iteration % 50 == 0 and iteration > 0:
            try:
                # Create a linearized convex relaxation
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add overlap constraints for the top k shifts
                corr_vals = np.correlate(current_h, 1 - current_h, mode='full')
                top_k_indices = np.argsort(corr_vals)[-5:][::-1]
                
                for k_idx in top_k_indices:
                    shift = k_idx - (n_points - 1)
                    if shift >= 0:
                        # h[i] * (1 - h[i+shift]) for i=0 to n_points-shift-1
                        corr_sum = cp.sum(cp.multiply(h[:n_points-shift], 1 - h[shift:]))
                    else:
                        # h[i+|shift|] * (1 - h[i]) for i=0 to n_points+shift-1
                        corr_sum = cp.sum(cp.multiply(h[-shift:], 1 - h[:n_points+shift]))
                    # Add soft constraint
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                objective = cp.Minimize(0)  # min over 0 is trivial
                prob = cp.Problem(objective, constraints)
                try: 
                    prob.solve(solver=cp.ECOS, eps=1e-6) 
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final SLSQP polishing
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            h_restart = best_h.copy()
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
