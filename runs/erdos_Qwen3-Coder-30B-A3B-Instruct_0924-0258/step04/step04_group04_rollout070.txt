import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[10:20] = 1.0
        h_start[25:30] = 0.0
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse grid optimization (n_points=20)
    coarse_points = 20
    coarse_h = np.linspace(0, 1, coarse_points)
    coarse_h[5:15] = 1.0
    coarse_h = coarse_h * (coarse_points/2) / np.sum(coarse_h)
    
    # Fine grid optimization
    fine_points = 40
    coarse_to_fine = interp1d(np.linspace(0, 2, coarse_points), coarse_h, kind='linear')
    fine_h = coarse_to_fine(np.linspace(0, 2, fine_points))
    fine_h = fine_h * (fine_points/2) / np.sum(fine_h)
    
    # Optimization helper function
    def optimize_with_penalty(h_in, penalty_weights, penalties, max_iter=50):
        n = len(h_in)
        h_var = cp.Variable(n)
        dx = 2.0 / n
        penalties = np.array(penalties)
        penalty_weights = np.array(penalty_weights)
        
        # Constraint for sum of h
        constraints = [cp.sum(h_var) == n/2, h_var >= 0, h_var <= 1]
        
        # Compute the full correlation with penalties
        expr = 0
        for i, (pw, p) in enumerate(zip(penalty_weights, penalties)):
            if i < n:  # shifting
                exp = cp.sum(cp.multiply(h_var[:n-i], 1 - h_var[i:]))
                expr += pw * (exp + p)
        objective = cp.Minimize(expr)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                return h_var.value
        except:
            pass
        return h_in
    
    # Main optimization loop for fine grid
    h_current = fine_h.copy()
    penalty_weights = np.ones(len(h_current))  # Uniform weights for simplicity
    penalties = np.ones(len(h_current))  # Initial penalties
    
    i = 0
    while time() - start_time < budget_s * 0.95:
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h_current, 2.0/len(h_current))
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        # Get correlation to update penalties
        corr = np.correlate(h_current, 1-h_current, mode='full')
        shift_corr = corr[len(corr)//2:]
        penalty_updates = np.maximum(shift_corr * 0.1, 1e-6)
        penalties = penalty_updates + penalties * 0.5
        
        # Optimize with current penalties
        h_opt = optimize_with_penalty(h_current, penalty_weights, penalties, max_iter=25)
        if h_opt is not None:
            h_current = h_opt
        
        # Gradient smoothing
        h_current = np.convolve(h_current, np.ones(5)/5, mode='same')
        h_current = h_current * (fine_points/2) / np.sum(h_current)
        
        # Gradually shift towards SLSQP for final refinement
        if i > 100:
            # Final refinement with SLSQP
            def obj(h):
                return evaluate_erdos_solution(h, 2.0/len(h))
            
            def con_sum(h):
                return np.sum(h) - fine_points/2
                
            bounds = [(0, 1) for _ in range(len(h_current))]
            cons = {'type': 'eq', 'fun': con_sum}
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    h_current = res.x
            except:
                pass
                
        i += 1
        
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, 2.0/len(best_h))
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (list(best_h), best_c5, len(best_h))
