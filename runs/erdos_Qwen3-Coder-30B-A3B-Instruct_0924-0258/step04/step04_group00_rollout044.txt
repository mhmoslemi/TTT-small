import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Local optimization with random restarts
    max_restarts = 20
    restart_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Random perturbation and normalization
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # SLSQP local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    last_improvement = time()
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Check if it's been too long without improvement
        if time() - last_improvement > 30:
            break
    
    # Feature enhancement: Introduce local binary search
    # Transform to binary approximations and improve
    def binary_local_search(h, bounds=(0, 1)):
        h_binary = np.round(h)
        h_binary = np.clip(h_binary, bounds[0], bounds[1])
        h_binary = h_binary * (n_points/2) / np.sum(h_binary)
        return h_binary
    
    # Final refinement using carefully selected binary search
    for _ in range(5):
        h_binary = binary_local_search(best_h)
        c5_binary = evaluate_erdos_solution(h_binary, dx)
        if c5_binary < best_c5:
            best_h = h_binary
            best_c5 = c5_binary
    
    # Post-processing: ensuring exact constraint satisfaction
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
