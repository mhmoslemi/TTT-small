import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    def compute_gradient(h, k_shift):
        # Compute gradient w.r.t. the maximum overlap shift k
        grad = np.zeros_like(h)
        n = len(h)
        
        # For shift k, derivative of h[i] * (1 - h[i + k]) w.r.t. h[i]
        # = 1 - h[i + k] for i < n - k, and 0 otherwise
        # = h[i] for i >= k, and 0 otherwise
        if k_shift < n:
            # Part 1: h[i] * (1 - h[i + k]) -> derivative of 1 - h[i + k] w.r.t. h[i] = 0
            # Part 2: h[i + k] * (1 - h[i]) -> derivative of 1 - h[i] w.r.t. h[i + k] = 0
            # Actually, need overlapping correlation:
            # corr[k] = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            for i in range(n - k_shift):
                grad[i] += 1 - h[i + k_shift]
                if i + k_shift < n:
                    grad[i + k_shift] -= h[i]
        else:
            # For negative shifts, reverse indices
            k = k_shift - (n - 1)
            for i in range(n + k):
                if i + k >= 0 and i + k < n:
                    grad[i] -= h[i + k]
                if i + k < 0:
                    grad[i + k + n] -= h[i]
        
        return grad

    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    decay_rate = 0.995
    max_iterations = 5000
    
    # Track best solution found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    def project_h(h):
        # Ensure h in [0,1] bounds and sum = n/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        # Scale to exact sum constraint
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    iter_count = 0
    # Gradient descent loop
    while iter_count < max_iterations and (time() - start_time) < budget_s * 0.9:
        # Compute overlaps for all k to find maximum
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_overlap_idx = np.argmax(corr)
        k_max = max_overlap_idx - (n_points - 1)
        
        # Compute gradient for this maximum
        grad = compute_gradient(current_h, abs(k_max))
        
        # SGD update
        current_h = current_h - alpha * grad
        
        # Project to feasible region
        current_h = project_h(current_h)
        
        # Evaluate with current h
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Check if better than best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            # Reset learning rate if improvement is observed
            alpha = 0.1
            
        # Reduce learning rate
        alpha *= decay_rate
        
        iter_count += 1

    # Convex refinement every 100 iterations
    if (time() - start_time) < budget_s * 0.95:
        # Solve convex optimization problem over top overlaps
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add more active constraints to guide optimization
            corr_with_full = np.correlate(best_h, 1 - best_h, mode='full')
            max_k_indices = np.argsort(corr_with_full)[-10:]  # Top 10 overlaps
            
            for k_idx in max_k_indices:
                if k_idx < n_points:
                    # h[i] * (1 - h[i + k_idx])
                    corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k_idx], 1 - h_vars[k_idx:]))
                else:
                    # same data but shifted back, need shifting indices
                    corr_sum = cp.sum(cp.multiply(h_vars[k_idx-n_points+1:], 1 - h_vars[:n_points-(k_idx-n_points+1)]))
                # Keep constraints more strict than observed overlaps
                constraints.append(corr_sum <= 0.5 * dx * n_points) # There's a smarter way to model this
            
            # Minimize a proxy for maximum overlap
            # Don't minimize closeness to exact current overlap.
            # A better method is to add soft penalty constraints around latest solution.
            # Or refine last few trials via box constraints directly before SLSQP.
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    refined_h = h_vars.value
                    if refined_h is not None:
                        refined_c5 = evaluate_erdos_solution(refined_h, dx)
                        if refined_c5 < best_c5:
                            best_h = refined_h
                            best_c5 = refined_c5
            except Exception:
                pass
        except Exception:
            pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Try a few restarts with refined solution
        for restart in range(4):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            # Project this new vector
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
