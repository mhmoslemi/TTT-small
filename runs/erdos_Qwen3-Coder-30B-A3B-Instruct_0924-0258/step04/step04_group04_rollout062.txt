import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a simple but structured start
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Add some structure to avoid flat regions
        for i in range(n_points):
            if i > n_points/4 and i < 3*n_points/4:
                h_start[i] = 1.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Use a more focused approach: binary search on the minimum correlation
    tolerance = 1e-5
    low_bound = 0.0
    high_bound = 1.0
    best_lambda = 0.0
    refined_h = best_h.copy()
    
    def find_feasible_solution(target_lambda):
        # Start with the previous best solution
        h = refined_h.copy()
        # Ensure it meets the sum constraint
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        max_iter = 1000
        learning_rate = 0.1
        backtracking_factor = 0.5
        min_step = 1e-8
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                return h, False
                
            # Compute auto-correlations
            # Note: we need the correlation of h with (1-h) at all shifts
            corr = np.correlate(h, 1 - h, mode='full')
            corr = corr * dx
            
            # Find the minimum correlation and corresponding shift
            k_min = np.argmin(corr)
            min_val = corr[k_min]
            
            if min_val >= target_lambda:
                return h, True
                
            # Compute gradient of corr[k_min] with respect to h
            # This is complex due to overlapping computations.
            # We will approximate it with finite differences.
            eps = 1e-6
            grad = np.zeros(n_points)
            for i in range(n_points):
                temp_h = h.copy()
                temp_h[i] += eps
                temp_h = temp_h * (n_points/2) / np.sum(temp_h)
                temp_h = np.clip(temp_h, 0, 1)
                
                temp_corr = np.correlate(temp_h, 1 - temp_h, mode='full')
                temp_corr = temp_corr * dx
                temp_min_val = temp_corr[k_min]
                
                grad[i] = (temp_min_val - min_val) / eps
                
            # Take a gradient step
            step_size = learning_rate
            while step_size > min_step:
                new_h = h - step_size * grad
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                
                new_corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_corr = new_corr * dx
                new_min_val = new_corr[k_min]
                
                if new_min_val >= target_lambda:
                    h = new_h
                    break
                else:
                    step_size *= backtracking_factor
            
            if step_size <= min_step:
                # Failed to improve with gradient descent
                return h, False
                
            # Make sure to normalize again after potential increases
            h = h * (n_points/2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
        return h, False  # failed to converge

    # Binary search for the maximal feasible lambda
    while high_bound - low_bound > tolerance and (time() - start_time) < budget_s * 0.9:
        mid_lambda = (low_bound + high_bound) / 2
        candidate_h, feasible = find_feasible_solution(mid_lambda)
        
        if feasible:
            low_bound = mid_lambda
            best_lambda = mid_lambda
            refined_h = candidate_h.copy()
            
            # Check if this improves our solution
            current_c5 = evaluate_erdos_solution(refined_h, dx)
            if current_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = current_c5
        else:
            high_bound = mid_lambda

    # Final refinement with SLSQP if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass

    return (list(best_h), best_c5, n_points)
