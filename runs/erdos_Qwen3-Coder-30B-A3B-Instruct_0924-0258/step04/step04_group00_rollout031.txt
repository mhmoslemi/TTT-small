import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    num_generations = 1000
    mutation_rate = 0.1
    tournament_size = 2
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # GA main loop
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < num_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current population
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection with tournament
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[(i+1) % population_size]
            
            # Crossover: 1-point swap
            if random.random() < 0.8:  # 80% crossover rate
                crossover_point = random.randint(1, n_points - 1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
            
            # Mutation
            for off in [offspring1, offspring2]:
                for j in range(n_points):
                    if random.random() < mutation_rate:
                        off[j] += np.random.normal(0, 0.01)
                
                # Repair feasibility
                off = np.clip(off, 0, 1)
                off = off * (n_points/2) / np.sum(off)
            
            new_population[i] = offspring1
            new_population[(i+1) % population_size] = offspring2
        
        population = new_population
        generation += 1
    
    # Final refinement with local optimization
    try:
        from scipy.optimize import minimize
        
        # Try refining the best solution found
        h_refined = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
