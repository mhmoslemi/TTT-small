import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier parameters
    n_freqs = n_points // 2
    if n_freqs == 0:
        n_freqs = 1
    
    # Initialize coefficients
    a0 = 0.0
    a_coeffs = np.random.normal(0, 0.1, n_freqs)
    b_coeffs = np.random.normal(0, 0.1, n_freqs)
    
    # Adaptive learning rate
    lr = 0.01
    momentum = 0.9
    velocity_a0 = 0
    velocity_a = np.zeros(n_freqs)
    velocity_b = np.zeros(n_freqs)
    
    # Regularization
    reg_lambda = 0.001
    
    def sigmoid(x):
        return 1 / (1 + np.exp(-x))
    
    def compute_h(a0, a_coeffs, b_coeffs):
        # Build Fourier series
        x = np.linspace(0, 2, n_points)
        fourier = a0 + np.sum([a_coeffs[i] * np.cos((i+1) * np.pi * x) + 
                              b_coeffs[i] * np.sin((i+1) * np.pi * x) 
                              for i in range(n_freqs)], axis=0)
        h = sigmoid(fourier)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_overlap_gradients(a0, a_coeffs, b_coeffs, k_max):
        # Compute h
        h = compute_h(a0, a_coeffs, b_coeffs)
        
        # Compute FFT
        H = np.fft.fft(h)
        H_conj = np.conj(H)
        
        # 1-h
        H1 = np.fft.fft(1-h)
        
        # Overlap correlation
        overlap_fft = H * H1
        overlap = np.fft.ifft(overlap_fft).real
        
        # Find maximum
        max_overlap = overlap[2*n_points - 1]  # Bottom half and wrap around
        for i in range(2*n_points - 1):
            if overlap[i] > max_overlap:
                max_overlap = overlap[i]
                k_max = i
        
        # Gradients (simplified)
        # Gradient w.r.t. each coefficient
        grad_a0 = 0
        grad_a = np.zeros(n_freqs)
        grad_b = np.zeros(n_freqs)
        
        # Simple numerical gradients for demonstration
        eps = 1e-6
        for i in range(n_freqs):
            a_coeffs[i] += eps
            h_plus = compute_h(a0, a_coeffs, b_coeffs)
            a_coeffs[i] -= 2*eps
            h_minus = compute_h(a0, a_coeffs, b_coeffs)
            a_coeffs[i] += eps
            grad_a[i] = (evaluate_erdos_solution(h_plus, dx) - evaluate_erdos_solution(h_minus, dx)) / (2*eps)
        
        return grad_a0, grad_a, grad_b, k_max
    
    iteration = 0
    max_iterations = 10000
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Compute current h
        h = compute_h(a0, a_coeffs, b_coeffs)
        c5_val = evaluate_erdos_solution(h, dx)
        
        # Update best
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
        
        # Compute gradients (approximate)
        # We'll sample a few at random:
        k_max = 0
        grad_a0, grad_a, grad_b, _ = compute_overlap_gradients(a0, a_coeffs, b_coeffs, k_max)
        
        # Add regularization
        grad_a0 += reg_lambda * a0
        grad_a += reg_lambda * a_coeffs
        grad_b += reg_lambda * b_coeffs
        
        # Update with momentum
        velocity_a0 = momentum * velocity_a0 - lr * grad_a0
        velocity_a = momentum * velocity_a - lr * grad_a
        velocity_b = momentum * velocity_b - lr * grad_b
        
        a0 += velocity_a0
        a_coeffs += velocity_a
        b_coeffs += velocity_b
        
        # Check for early termination
        if c5_val < 0.3807:
            break
        
        iteration += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Local refinement with SLSQP
        def objective(params):
            a0, a_coeffs, b_coeffs = params[0], params[1:n_freqs+1], params[n_freqs+1:]
            h = compute_h(a0, a_coeffs, b_coeffs)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(params):
            a0, a_coeffs, b_coeffs = params[0], params[1:n_freqs+1], params[n_freqs+1:]
            h = compute_h(a0, a_coeffs, b_coeffs)
            return np.sum(h) - n_points / 2.0
        
        # Prepare initial guess
        initial_guess = np.concatenate([[a0], a_coeffs, b_coeffs])
        bounds = [(None, None)] + [(-10, 10)] * (2*n_freqs)  # Allow wide range
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, initial_guess, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                a0_ref, a_coeffs_ref, b_coeffs_ref = res.x[0], res.x[1:n_freqs+1], res.x[n_freqs+1:]
                h_refined = compute_h(a0_ref, a_coeffs_ref, b_coeffs_ref)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
