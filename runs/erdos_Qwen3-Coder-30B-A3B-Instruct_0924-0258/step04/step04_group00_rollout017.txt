import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Normalize the initial solution to meet the sum constraint
    h_start = h_start * (n_points / 2.0) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local search with random restarts
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
        # Random perturbation with directional bias
        h_restart = best_h + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # Local optimization using SLSQP
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(evaluate_erdos_solution, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 150, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0  # Reset counter on improvement
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Perturbation-based fast local search
        if time() - start_time < budget_s * 0.8 and restart_count % 3 == 0:
            h_local = best_h.copy()
            # Flip some values to try to escape local minima
            for _ in range(5):
                idx = np.random.randint(0, n_points)
                h_local[idx] = 1 - h_local[idx]
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points/2) / np.sum(h_local)
            c5_local = evaluate_erdos_solution(h_local, dx)
            if c5_local < best_c5:
                best_h = h_local
                best_c5 = c5_local
                
    # Final fine-tuning with pattern-aware mutation
    h_fine = best_h.copy()
    # Add some structure to the pattern: jittering around existing structure
    for i in range(n_points // 3):
        idx1 = np.random.randint(0, n_points)
        idx2 = (idx1 + 1) % n_points
        idx3 = (idx1 + 2) % n_points
        if np.random.rand() < 0.5:
            h_fine[[idx1, idx2, idx3]] = np.clip(h_fine[[idx1, idx2, idx3]] + np.random.normal(0, 0.01, 3), 0, 1)
    
    h_fine = h_fine * (n_points/2) / np.sum(h_fine)
    c5_final = evaluate_erdos_solution(h_fine, dx)
    
    if c5_final < best_c5:
        best_h = h_fine
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
