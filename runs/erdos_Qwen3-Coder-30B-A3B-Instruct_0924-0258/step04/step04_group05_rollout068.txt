import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population with diverse solutions
    def generate_population(n_points, pop_size):
        if 'initial_h_values' in globals():
            h_start = np.array(initial_h_values)
        else:
            h_start = np.ones(n_points) * 0.5
        
        population = [h_start.copy()]
        
        # Add perturbed versions of initial solution
        for _ in range(10):
            h_pert = h_start + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            population.append(h_pert)
        
        # Add completely random valid solutions
        for _ in range(5):
            h_rand = np.random.random(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            population.append(h_rand)
        
        # Fill out the population
        while len(population) < pop_size:
            h_new = np.random.random(n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            population.append(h_new)
        
        return population
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize parameters
    pop_size = 40
    mutation_rate = 0.15
    crossover_rate = 0.7
    tabu_limit = 50
    max_stagnation = 50
    stagnation_count = 0
    
    # Generate initial population
    population = generate_population(n_points, pop_size)
    fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Tabu list to store recently visited solutions
    tabu_list = set()
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    # Main genetic search loop
    for generation in range(1000):  # Safe upper bound
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best = population[0]
        current_best_c5 = fitness[0]
        if current_best_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_best_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive parameter adjustments
        if stagnation_count > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_rate = max(0.3, crossover_rate - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_rate = min(0.9, crossover_rate + 0.05)
        
        # Local gradient refinement on top solutions
        if generation % 10 == 0:
            for i in range(min(5, len(population))):
                h_refined = population[i].copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                def con(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': con}
                
                try:
                    res = minimize(obj, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_new = res.x
                        c5_new = obj(h_new)
                        # Add to population if enough better
                        if c5_new < fitness[i] * 0.999:
                            population[i] = h_new
                            fitness[i] = c5_new
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_new.copy()
                except:
                    pass
        
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        
        while len(new_population) < pop_size:
            # Crossover
            if random.random() < crossover_rate:
                parent1_idx = random.randint(0, pop_size // 2)
                parent2_idx = random.randint(0, pop_size // 2)
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Simple mask crossover
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
                
                # Mutation
                if random.random() < mutation_rate:
                    mut_idx = random.randint(0, n_points - 1)
                    child[mut_idx] = random.random()
                    
                # Repair constraints
                child_sum = np.sum(child)
                child = child * (n_points / 2) / child_sum
                child = np.clip(child, 0, 1)
                
                # Add to new population
                new_population.append(child)
            else:
                # Clone an existing individual (roulette selection)
                idx = np.random.choice(pop_size, p=np.array(range(pop_size, 0, -1)) / (pop_size * (pop_size + 1) / 2))
                child = population[idx].copy()
                
                # Mutation
                if random.random() < mutation_rate:
                    mut_idx = random.randint(0, n_points - 1)
                    child[mut_idx] = random.random()
                    
                # Repair constraints
                child_sum = np.sum(child)
                child = child * (n_points / 2) / child_sum
                child = np.clip(child, 0, 1)
                
                new_population.append(child)
        
        # Update population
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
    
    # Final local optimization
    if time() - start_time < budget_s * 0.95:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = obj(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
