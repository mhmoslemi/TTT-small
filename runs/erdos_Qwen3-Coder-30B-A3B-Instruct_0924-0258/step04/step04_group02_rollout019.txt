import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Train a simple neural-style model
    # h(x) = max(0, W * x + b) scaled to sum to 1
    # but we work with a simplified approach
    
    # Parameters to optimize: [W1, W2, ..., Wn, b, scale_factor]
    if n_points < 35:
        # Smaller number of points - direct optimization
        # Use Levy optimization approach with more aggressive search
        x0 = np.concatenate([h_start, [0.5], [1.0]])
        
        def objective(x):
            # Unpack params
            h = x[:-2]
            b = x[-2]
            scale = x[-1]
            # Ensure h is in [0,1]
            h = np.maximum(0, h)
            h = np.minimum(1, h)
            # Apply scaling to force sum to n_points/2
            h = h * scale
            h = h * (n_points / 2) / np.sum(h)
            
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(x):
            h = x[:-2]
            h = np.maximum(0, h)
            h = np.minimum(1, h)
            h = h * (n_points / 2) / np.sum(h)
            return np.sum(h) - n_points / 2.0
        
        def constraint_bounds(x):
            # h in [0,1]
            h = x[:-2]
            return np.concatenate([h, 1 - h])
        
        bounds = [(0, 1) for _ in range(n_points)] + [(-2, 2), (0.1, 10)]
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        cons_bounds = {'type': 'ineq', 'fun': constraint_bounds}
        
        result = minimize(objective, x0, method='SLSQP', 
                          bounds=bounds, constraints=[cons_sum, cons_bounds],
                          options={'maxiter': 50, 'ftol': 1e-9})
        
        if result.success:
            h_opt = result.x[:-2]
            h_opt = np.maximum(0, h_opt)
            h_opt = np.minimum(1, h_opt)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
    # Try some more heterogeneous optimizations
    n_trials = 10
    for trial in range(n_trials):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Different random initialization
        h_rand = np.random.uniform(0, 1, n_points)
        # Scale to sum = 1
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        
        # Local refinement
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, h_rand, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 30})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Try other combinations
        # Increase with local refinement
        if trial % 3 == 0:
            h_new = np.copy(h_start)
            # Add random noise and clamp
            noise = np.random.normal(0, 0.02, n_points)
            h_new += noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # One last check with mimicking the best solutions found previously
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
