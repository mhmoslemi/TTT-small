import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with adaptive step size
    max_iter = 10000
    alpha = 0.01
    stagnation_count = 0
    max_stagnation = 50
    prev_c5 = best_c5 + 1.0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient and objective
        h = best_h.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Compute gradient (per element in h)
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                if 0 <= i < n_points and 0 <= i + k < n_points:
                    if k >= 0:
                        grad[i] += (1 - h[i + k])
                    else:
                        grad[i] += (1 - h[i + k])
        
        # Update step with adaptive learning rate
        h_new = h - alpha * grad
        
        # Project to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Adaptive step size
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            stagnation_count = 0
            alpha *= 1.2  # Increase step size on improvement
        else:
            stagnation_count += 1
            alpha *= 0.5  # Decrease step size
            if stagnation_count > max_stagnation:
                break
            # Revert to previous h if stagnated
            h_new = best_h.copy()
            c5_new = best_c5
        
        # Early stopping for improvement
        if abs(prev_c5 - c5_new) < 1e-8:
            stagnation_count += 1
        else:
            stagnation_count = 0
        prev_c5 = c5_new
        
        # Occasionally refine with local optimization (fewer calls)
        if iteration % 25 == 0 and iteration > 0:
            try:
                # Simple restart if improvement stalls
                h_restart = best_h + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                c5_restart = evaluate_erdos_solution(h_restart, dx)
                if c5_restart < best_c5:
                    best_h = h_restart
                    best_c5 = c5_restart
            except Exception:
                pass
    
    # Final refinement with local search
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
