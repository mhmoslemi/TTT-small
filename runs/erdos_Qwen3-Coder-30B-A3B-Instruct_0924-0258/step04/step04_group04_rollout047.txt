import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create initial function with clusters to induce structure
        # Distribute the total mass of 1 over 3 main regions
        h_start[int(0.2/dx):int(0.6/dx)] = 0.5
        h_start[int(1.2/dx):int(1.6/dx)] = 0.5
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Split into coarse and fine optimization
    coarse_points = 20
    fine_points = 40
    dx_coarse = 2.0 / coarse_points
    dx_fine = 2.0 / fine_points
    
    # Coarse grid optimization
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / len(best_h))
    
    # Create coarse grid
    coarse_indices = np.linspace(0, 1, coarse_points, endpoint=False)
    h_coarse = np.interp(coarse_indices, np.linspace(0, 2, len(h_start)), h_start)
    h_coarse = h_coarse * (coarse_points/2) / np.sum(h_coarse)
    
    # Penalty method for constrained optimization
    penalty = 1.0
    max_iter = 50
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Compute all differences in overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        # The correlation is padding already, we need all cross-correlations
        # But for the overlap function we only care about the shifted functions
        # For 1d sequences, all the significant values are in the full correlation
        # We need to multiply each by dx_coarse
        max_corr = np.max(corr)
        if max_corr < best_c5:
            best_h = h_coarse.copy()
            best_c5 = max_corr * dx_coarse
            
        # Soft constraint: use penalty to handle max constraint
        # Use a convex relaxation approach, treating objective as:
        # minimize max_k (h * (1 - h_k)) + lambda * max_k (h * (1 - h_k) - threshold)
        # We'll take a simple projection approach
        # Try sampling a few shifts and building constraints to make
        # an LP formulation with penalty
        # Let's just optimize a convex version of the objective
        # We introduce the max term as a constraint by introducing costs
        n_pad = 2 * coarse_points - 1
        # Convolution indices from -n_points+1 to n_points-1
        shifts = np.arange(n_pad) - (coarse_points - 1)
        nearest_shifts = shifts
        # Simplified model for convex relaxation
        # We'll create a linear system with squared errors
        # Handle constraint directly in a convex formulation using CVXPY
        # Since we want a convex problem, approximate max as a large enough linear term
        # we will solve the dual
        # For convexity:
        # min_x: max_k <h, 1-h_k> 
        # etc, but let's use quadratic penalty to match approach
        
        h_var = cp.Variable(coarse_points)
        obj = cp.Minimize(cp.sum_squares(h_var - h_coarse))
        # Constraint: sum = n_points/2, on [0,1]
        constraints = [
            cp.sum(h_var) == coarse_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        try:
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Check if we improve with softer conditions
                h_new = h_new * (coarse_points/2) / np.sum(h_new)
                h_coarse = h_new
        except:
            pass
        
        # Accept based on improvement, or by randomly perturbing
        h_pert = h_coarse + np.random.normal(0, 0.01, coarse_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (coarse_points/2) / np.sum(h_pert)
        try:
            c5_pert = evaluate_erdos_solution(h_pert, dx_coarse)
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert
            h_coarse = h_pert
        except:
            pass
            
        # Perturb for diversity
        if iteration % 10 == 0 and iteration > 0:
            h_coarse += np.random.normal(0, 0.03, coarse_points)
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (coarse_points/2) / np.sum(h_coarse)
    
    # Now upsample to fine grid
    fine_indices = np.linspace(0, 1, fine_points, endpoint=False)
    h_fine = np.interp(fine_indices, np.linspace(0, 2, len(best_h)), best_h)
    h_fine = h_fine * (fine_points/2) / np.sum(h_fine)
    
    # Run fine grid optimization  
    n_points_fine = fine_points
    dx_fine = 2.0 / n_points_fine
    fine_iter = 100
    best_h_fine = h_fine.copy()
    best_c5_fine = evaluate_erdos_solution(best_h_fine, dx_fine)
    
    for iteration in range(fine_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(best_h_fine, 1 - best_h_fine, mode='full')
        max_corr = np.max(corr)
        if max_corr < best_c5_fine:
            best_c5_fine = max_corr * dx_fine
            best_h_fine = best_h_fine.copy()
            
        # Use slight L-BFGS, keep it small for stability and simplicity
        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_fine
            
        def con_sum(h):
            return np.sum(h) - fine_points/2
            
        bounds = [(0, 1) for _ in range(fine_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        # SLSQP
        try:
            res = minimize(
                obj, 
                best_h_fine, 
                method='SLSQP',
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 20, 'ftol': 1e-6}
            )
            if res.success and res.fun < best_c5_fine:
                best_h_fine = res.x
                best_c5_fine = res.fun
        except:
            pass
        
        # Add noise occasionally
        if iteration % 20 == 0:
            noise = np.random.normal(0, 0.005, fine_points)
            h_noisy = best_h_fine + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (fine_points/2) / np.sum(h_noisy)
            try:
                c5_noisy = evaluate_erdos_solution(h_noisy, dx_fine)
                if c5_noisy < best_c5_fine:
                    best_h_fine = h_noisy.copy()
                    best_c5_fine = c5_noisy
            except:
                pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx_fine
            
        def con_sum(h):
            return np.sum(h) - fine_points/2
            
        bounds = [(0, 1) for _ in range(fine_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h_fine, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx_fine)
                if final_c5 < best_c5_fine:
                    best_c5_fine = final_c5
                    best_h_fine = final_h
        except:
            pass
    
    # Final returned values:
    final_h = best_h_fine
    final_c5 = best_c5_fine
    final_n = fine_points
    
    return (list(final_h), final_c5, final_n)
