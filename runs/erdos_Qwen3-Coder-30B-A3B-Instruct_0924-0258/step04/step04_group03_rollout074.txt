import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization with global best tracking
    class PSO:
        def __init__(self, n_particles, n_dimensions, bounds, max_iter=1000):
            self.n_particles = n_particles
            self.n_dimensions = n_dimensions
            self.bounds = bounds
            self.max_iter = max_iter
            self.w = 0.7  # inertia weight
            self.c1 = 2.0  # cognitive parameter
            self.c2 = 2.0  # social parameter
            
            # Initialize particles
            self.particles = np.random.uniform(bounds[0], bounds[1], (n_particles, n_dimensions))
            self.velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_dimensions))
            self.personal_best = self.particles.copy()
            self.personal_best_scores = np.full(n_particles, np.inf)
            self.global_best = None
            self.global_best_score = np.inf
            
            # Normalize initial particles to satisfy sum constraint
            for i in range(n_particles):
                self.particles[i] *= (n_points / 2) / np.sum(self.particles[i])
            
        def evaluate(self, h):
            return evaluate_erdos_solution(h, dx)
        
        def optimize(self, objective_func):
            for iteration in range(self.max_iter):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                for i in range(self.n_particles):
                    h = self.particles[i]
                    score = self.evaluate(h)
                    
                    # Update personal best
                    if score < self.personal_best_scores[i]:
                        self.personal_best_scores[i] = score
                        self.personal_best[i] = h.copy()
                        
                    # Update global best
                    if score < self.global_best_score:
                        self.global_best_score = score
                        self.global_best = h.copy()
                
                # Update velocities and positions
                for i in range(self.n_particles):
                    r1, r2 = np.random.random(2)
                    self.velocities[i] = (self.w * self.velocities[i] + 
                                         self.c1 * r1 * (self.personal_best[i] - self.particles[i]) +
                                         self.c2 * r2 * (self.global_best - self.particles[i]))
                    
                    # Limit velocity
                    self.velocities[i] = np.clip(self.velocities[i], -0.2, 0.2)
                    
                    # Update position
                    self.particles[i] += self.velocities[i]
                    
                    # Clamp to bounds
                    self.particles[i] = np.clip(self.particles[i], self.bounds[0], self.bounds[1])
                    
                    # Normalize to correct sum
                    self.particles[i] *= (n_points / 2) / np.sum(self.particles[i])
                    
                    # Update best if improved
                    score = self.evaluate(self.particles[i])
                    if score < self.personal_best_scores[i]:
                        self.personal_best_scores[i] = score
                        self.personal_best[i] = self.particles[i].copy()
                    
                    if score < self.global_best_score:
                        self.global_best_score = score
                        self.global_best = self.particles[i].copy()
            
            return self.global_best, self.global_best_score
    
    # Run PSO
    pso = PSO(n_particles=50, n_dimensions=n_points, bounds=(0, 1), max_iter=1000)
    best_h, best_c5 = pso.optimize(evaluate_erdos_solution)
    
    # Convex Refinement Phase
    for refine_iter in range(10):
        if time() - start_time > budget_s * 0.95:
            break
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add a proxy of overlap constraint
            for k in range(0, min(10, 2 * n_points - 1), 2):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * dx * n_points * 1.1)  # some slack
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final SLSQP Polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass

    return (list(best_h), best_c5, n_points)
