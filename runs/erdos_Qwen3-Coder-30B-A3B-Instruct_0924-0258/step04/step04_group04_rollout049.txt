import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.zeros(n_points)
        h_start[5:15] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def optimize_single_h(h_in, penalties=None):
        if penalties is None:
            penalties = np.zeros(2 * n_points - 1)
        
        # CVXPY formulation
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.pos(cp.convolve(h_var, 1 - h_var, mode='full') + penalties)))
        
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                return h_var.value
        except:
            pass
        return h_in
    
    # Coarse optimization with penalty method
    penalties = np.ones(2 * n_points - 1) * 0.1
    
    for iteration in range(100):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps and update penalties
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        penalties = np.maximum(penalties * (corr / (penalties + 1e-8)), 1e-6)
        
        # Optimize h with current penalties
        h_new = optimize_single_h(best_h, penalties)
        
        if h_new is not None:
            h_new = project(h_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
    
    # Upsample to fine grid
    h_fine = np.linspace(0, 2, 2*n_points)
    h_coarse = np.linspace(0, 2, n_points)
    interpolator = interp1d(h_coarse, best_h, kind='linear', fill_value='extrapolate')
    h_fine = interpolator(h_fine)
    n_points = 2 * n_points
    dx = 2.0 / n_points
    h_fine = project(h_fine)
    
    # Fine optimization with smoother penalty update
    penalties = np.ones(2 * n_points - 1) * 0.1
    for iteration in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        penalties = np.maximum(penalties * (corr / (penalties + 1e-8)), 1e-6)
        h_new = optimize_single_h(h_fine, penalties)
        
        if h_new is not None:
            h_new = project(h_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
    
    # Final refinement with more precise optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def con_sum(h):
        return np.sum(h) - n_points/2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
