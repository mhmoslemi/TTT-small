import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    n_points = 35
    dx = 2.0 / n_points
    population_size = 40
    tabu_limit = 50
    max_stagnation = 50
    
    # Initialize population with diverse solutions
    population = []
    tabu_list = []

    # Generate population
    for i in range(10):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Add random solutions
    for i in range(5):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Warm-up: initial fitness evaluation
    fitness = []
    for i, h in enumerate(population):
        corrected_h = np.clip(h, 0, 1)
        sum_h = np.sum(corrected_h)
        if abs(sum_h - n_points/2) > 1e-6:
            corrected_h = corrected_h * (n_points/2) / sum_h
        fit = evaluate_erdos_solution(corrected_h, dx)
        fitness.append(fit)
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_counter = 0
    
    # Main genetic loop for time budget
    generation = 0
    while time() - start_time < budget_s * 0.95:
        generation += 1
        new_population = []
        new_fitness = []
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        
        # Keep top 20% (elitism)
        top_indices = sorted_indices[:population_size // 5]
        for idx in top_indices:
            new_population.append(population[idx].copy())
            new_fitness.append(fitness[idx])
        
        # Generate offspring
        while len(new_population) < population_size:
            if random.random() < crossover_prob and len(new_population) > 1:
                # Tournament selection
                parent1_idx = random.choice(sorted_indices[:population_size // 2])
                parent2_idx = random.choice(sorted_indices[:population_size // 2])
                
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([population[parent1_idx][:crossover_point],
                                       population[parent2_idx][crossover_point:]])
            else:
                # Clone a random parent and mutate
                parent_idx = random.choice(sorted_indices[:population_size // 2])
                child = population[parent_idx].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutate_idx = random.randint(0, n_points - 1)
                child[mutate_idx] += np.random.normal(0, 0.03)
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if abs(sum_child - n_points/2) > 1e-6:
                child = child * (n_points/2) / sum_child
            
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))
        
        # Update population and fitness
        population = new_population
        fitness = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_counter = 0
            
            # Early stopping if target achieved
            if best_c5 < 0.3807:
                break
        else:
            stagnation_counter += 1
        
        # Adaptive parameter adjustment
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        else:
            if mutation_rate > 0.05:
                mutation_rate = max(0.05, mutation_rate - 0.05)
            if crossover_prob < 0.9:
                crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Occasionally refine top individuals with local search
        if generation % 10 == 0 and time() - start_time < budget_s * 0.9:
            top_indices = np.argsort(fitness)[:5]
            for idx in top_indices:
                h_init = population[idx].copy()
                
                # Local optimization
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[idx]:
                            population[idx] = h_refined
                            fitness[idx] = c5_refined
                            if c5_refined < best_c5:
                                best_c5 = c5_refined
                                best_h = h_refined.copy()
                except:
                    pass
    
    # Final local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
