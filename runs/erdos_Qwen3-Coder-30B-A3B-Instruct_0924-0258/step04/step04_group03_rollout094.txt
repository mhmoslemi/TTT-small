import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach with constraint projection
    max_iterations = 5000
    alpha = 0.1
    alpha_decay = 0.99
    max_k = min(50, 2 * n_points - 1)
    
    # Precompute indices for convolution
    conv_indices = []
    for k in range(2 * n_points - 1):
        if k < n_points:
            indices = (np.arange(n_points - k), np.arange(k, n_points))
        else:
            indices = (np.arange(k - n_points + 1, n_points), np.arange(n_points - (k - n_points + 1)))
        conv_indices.append(indices)
    
    def compute_gradient(h, k):
        # Compute gradient of convolution with 1-h at shift k
        i1, i2 = conv_indices[k]
        grad = np.zeros_like(h)
        grad[i1] += 1 - h[i2]
        return grad
    
    current_h = h_start.copy()
    current_c5 = best_c5
    step_count = 0
    
    for iteration in range(max_iterations):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Find maximum overlap
        overlaps = np.correlate(current_h, 1 - current_h, mode='full') * dx
        max_k_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_k_idx]
        
        # Compute gradient of this maximal overlap
        grad = compute_gradient(current_h, max_k_idx)
        
        # Update with gradient descent
        new_h = current_h - alpha * grad
        
        # Project to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            alpha *= alpha_decay
        
        step_count += 1
    
    # Convex relaxation refinement (sparse constraints)
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add key overlap constraints (top shifts)
            indices_to_use = list(range(0, min(20, 2 * n_points - 1), 2))
            
            for k in indices_to_use:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                # Add a soft upper bound constraint
                constraints.append(corr_sum <= best_c5 * 1.1)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40 and step_count > 100:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Try few restarts
            for restart in range(4):
                # Slightly perturb the best solution
                h_restart = best_h + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
