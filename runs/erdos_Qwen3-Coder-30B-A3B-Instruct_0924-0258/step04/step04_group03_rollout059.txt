import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            # Add small random perturbation
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.rand(n_points) * 0.2 + 0.4
            
        # Normalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1 - 0.05)
        val = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_vals.append(val)
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    w = w_start
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = val
                if val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = val
                    
    # Convex optimization refinement of global best
    try:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        # Add overlap constraint with current objective
        current_c5 = global_best_val
        # The exact constraint would be to minimize the overlap
        # But we will approximate with a tightened feasible constraint:
        corr = np.correlate(global_best, 1 - global_best, mode='full')
        max_k = np.max(corr)
        # We expect max_k to match our objective
        # Add soft constraints that force low overlap
        # For simpler form, assume each convolution <= 0.5 * (n_points/2) * dx
        # This is a relax constraint to encourage smaller overlap
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    except Exception:
        pass
    
    # Final SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Use best from PSO
        h_initial = global_best
        try:
            res = minimize(objective, h_initial, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success or not np.isnan(res.fun):
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    # Final validation
    final_sum = np.sum(best_h)
    final_h = best_h * (n_points / 2) / final_sum
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
