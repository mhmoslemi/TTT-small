import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization
    pop_size = 20
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    # Population of solutions
    population = []
    for _ in range(pop_size):
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Optimization loop
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate and update best
        for _, h in enumerate(population):
            curr_c5 = evaluate_erdos_solution(h, dx)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
        
        # Refine with SLSQP
        refined_population = []
        for h in population:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                else:
                    refined_h = h
            except:
                refined_h = h
            
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = refined_c5
            refined_population.append(refined_h)
        
        population = refined_population
        
        # Generate new population
        new_population = []
        for h in population:
            h_pert = h + np.random.normal(0, perturb_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            old_c5 = evaluate_erdos_solution(h, dx)
            new_c5 = evaluate_erdos_solution(h_pert, dx)
            if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
                new_population.append(h_pert)
            else:
                new_population.append(h)
        
        population = new_population
        temp *= cool_rate
    
    # Up-sample to refine
    n_points_up = 40
    dx_up = 2.0 / n_points_up
    
    # Interpolate
    h_up = np.interp(np.linspace(0, 2, n_points_up), 
                     np.linspace(0, 2, n_points), best_h)
    h_up = h_up * (n_points_up/2) / np.sum(h_up)
    
    # Final optimization at higher resolution
    def obj(h):
        return evaluate_erdos_solution(h, dx_up)
        
    def con_sum(h):
        return np.sum(h) - n_points_up/2
        
    bounds = [(0, 1) for _ in range(n_points_up)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, h_up, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            h_final = res.x
        else:
            h_final = h_up
    except:
        h_final = h_up
    
    c5_final = evaluate_erdos_solution(h_final, dx_up)
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points_up)
