import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Simple piecewise constant construction
        h_start[10:30] = 0.5
        h_start[70:90] = 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Set up binary search bounds
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Binary search on C5
    for _ in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate and solve LP feasibility problem
        h = Variable(n_points)
        
        # Constraints: sum = n/2, bounds [0,1]
        constraints = [sum(h) == n_points / 2.0, h >= 0, h <= 1]
        
        # Add overlap constraints for all lags
        dx = 2.0 / n_points
        for k in range(2 * n_points - 1):
            # Compute indices for correlation
            if k < n_points:
                # h[i] * (1 - h[i+k]) for i in 0 to n_points-1-k
                # This is the correlation term at lag k
                corr_term = 0
                for i in range(n_points - k):
                    corr_term += h[i] * (1 - h[i + k])
            else:
                # h[i] * (1 - h[i+k]) for i in 0 to k-n_points
                corr_term = 0
                for i in range(k - n_points + 1, n_points):
                    corr_term += h[i] * (1 - h[i + k - n_points + 1])
            
            constraints.append(corr_term <= mid / dx)
            
        # We're just checking feasibility, so objective doesn't matter
        prob = Problem(Minimize(0), constraints)
        
        try:
            prob.solve(solver=GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Feasible solution found
                h_sol = h.value
                c5_sol = evaluate_erdos_solution(h_sol, dx)
                if c5_sol < best_c5:
                    best_h = h_sol
                    best_c5 = c5_sol
                high = mid
            else:
                low = mid
        except:
            low = mid
            
    # Final validation of best solution
    final_dx = 2.0 / n_points
    final_c5 = evaluate_erdos_solution(best_h, final_dx)
    
    return (list(best_h), final_c5, n_points)
