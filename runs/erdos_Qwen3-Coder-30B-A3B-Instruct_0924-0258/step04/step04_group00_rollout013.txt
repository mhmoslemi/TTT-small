import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhanced evolutionary approach with local refinement
    population_size = 50
    mutation_strength = 0.05
    elite_size = 5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    population.append(best_h)
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection and reproduction
        new_population = population[:elite_size].copy()
        
        # Create offspring through mutation
        for _ in range(population_size - elite_size):
            parent = population[np.random.randint(0, elite_size)]
            child = parent + np.random.normal(0, mutation_strength, n_points)
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Adaptive mutation strength
        if generation % 20 == 0:
            diversity = np.std([np.std(h) for h in population[:10]])
            mutation_strength = max(0.005, min(0.2, mutation_strength * (1 + 0.05 * (diversity - 0.1))))
    
    # Final local optimization
    try:
        h_local = best_h.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points/2) / np.sum(h_local)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_local, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 50, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
