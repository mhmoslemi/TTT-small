import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a more structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Three clusters
        clusters = [(0.4, 0.2), (1.0, 0.3), (1.6, 0.2)]
        for center, height in clusters:
            start = max(0, int((center - height/2)/dx))
            end = min(n_points, int((center + height/2)/dx))
            h_start[start:end] = 1.0
        # Normalize to sum to n/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    elite_size = 4
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h_candidate = h_start + np.random.normal(0, 0.01, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Inherit best from starting point
    population[0] = h_start.copy()
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(c5_val)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new generation
        new_population = population[:elite_size]  # Keep best
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, population_size//2)]
            parent2_idx = sorted_indices[np.random.randint(0, population_size//2)]
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Gaussian mutation
                    child += np.random.normal(0, 0.02, n_points)
                    child = np.clip(child, 0, 1)
                    # Normalize
                    child = child * (n_points/2) / np.sum(child)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final refinement with SQP for best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    if np.sum(best_h) > 0:
        best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
