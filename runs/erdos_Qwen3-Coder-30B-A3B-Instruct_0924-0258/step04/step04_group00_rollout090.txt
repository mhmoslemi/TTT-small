import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Integer relaxation + refinement approach
    def binary_relaxation(h):
        # Project to binary values
        binary_h = (h > 0.5).astype(float)
        # Ensure sum constraint
        current_sum = np.sum(binary_h)
        target_sum = n_points / 2.0
        if current_sum != target_sum:
            # Adjust binary values
            diff = target_sum - current_sum
            if diff > 0:
                # Need to add more ones
                indices = np.argsort(-h)
                for i in range(int(diff)):
                    if indices[i] < n_points:
                        binary_h[indices[i]] = 1.0
            else:
                # Need to remove ones
                indices = np.argsort(h)
                for i in range(int(-diff)):
                    if indices[i] < n_points:
                        binary_h[indices[i]] = 0.0
        return binary_h
    
    # Local optimization with projected gradient
    def project_h(h):
        # Ensure h is in [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Scale to sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2.0) / current_sum
        return h
    
    max_restarts = 100
    restart_count = 0
    proj_iter = 0
    max_proj_iter = 200
    
    while time() - start_time < budget_s * 0.95 and restart_count < max_restarts and proj_iter < max_proj_iter:
        # Random perturbation with small noise
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points) 
        h_perturbed = project_h(h_perturbed)
        
        # Local search with projected gradient
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Wrapper for scipy optimizer
        def fun(h):
            return objective(h)
        
        def jac(h):
            # Approximate gradient with finite differences
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project_h(h_plus)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = project_h(h_minus)
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
            return grad
        
        try:
            res = minimize(fun, h_perturbed, method='L-BFGS-B', 
                           bounds=[(0, 1) for _ in range(n_points)],
                           options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                h_refined = project_h(h_refined)
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0  # Reset counter on improvement
                else:
                    restart_count += 1
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Periodic binary relaxation and refinement
        if proj_iter % 10 == 0:
            h_binary = binary_relaxation(best_h)
            h_binary = project_h(h_binary)
            c5_binary = objective(h_binary)
            if c5_binary < best_c5:
                best_h = h_binary
                best_c5 = c5_binary
        
        # Additional convex relaxation check
        if proj_iter % 20 == 0 and time() - start_time < budget_s * 0.8:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add a few key overlap constraints
                for k in range(min(10, n_points // 2)):
                    if k > 0:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(h_opt * (1 - h_opt))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    h_feasible = project_h(h_feasible)
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        proj_iter += 1
    
    # Final check and refinement
    try:
        best_h = project_h(best_h)
        final_c5 = evaluate_erdos_solution(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
