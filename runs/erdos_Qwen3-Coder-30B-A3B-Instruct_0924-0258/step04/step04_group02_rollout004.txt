import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize z values (preimage for logistic reparameterization)
    z_start = np.log(h_start / (1 - h_start + 1e-12))  # Avoid log(0)
    z_current = z_start.copy()
    
    def h_from_z(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def adjust_z_to_sum(z):
        # Compute h from z
        h = h_from_z(z)
        # Adjust to make sum equal to n_points / 2
        current_sum = np.sum(h)
        adjustment = (n_points / 2 - current_sum) / n_points
        return z + adjustment
    
    # Optimize using gradient-based method with reparameterization
    max_iter = 1000
    for iteration in range(max_iter):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Ensure we maintain sum constraint
        z_current = adjust_z_to_sum(z_current)
        h_current = h_from_z(z_current)
        
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        # Simple gradient approximation via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            z_test = z_current.copy()
            z_test[i] += eps
            h_test = h_from_z(z_test)
            c5_test = evaluate_erdos_solution(h_test, dx)
            grad[i] = (c5_test - current_c5) / eps
            
        # Simple gradient descent with adaptive step size
        step_size = 0.1 * (1.0 + iteration * 0.01)  # Adaptive
        z_new = z_current - step_size * grad
        
        # Boundary checks after update
        z_current = z_new
        
    # Final refinement with local optimizer
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Try a few restarts with small disturbances
        for restart in range(5):
            if (time() - start_time) > budget_s * 0.95:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
                
    return (list(best_h), best_c5, n_points)
