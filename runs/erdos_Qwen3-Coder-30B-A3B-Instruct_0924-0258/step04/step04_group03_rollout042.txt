import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient of overlap with shift k
        grad = np.zeros_like(h)
        # h[i] contributes to overlap with (1-h[i+k]) and h[i-k]
        # extend h for boundary conditions
        h_ext = np.concatenate([h, [0] * (n_points - 1)])
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += -h_ext[i + k]
            if i - k >= 0:
                grad[i] += -h_ext[i - k]
        return grad
    
    def project_to_feasible(h):
        # Project to [0, 1] and sum = n_points / 2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Gradient descent
    h = h_start.copy()
    c5 = best_c5
    alpha = 0.1
    max_iter = 5000
    
    # Track best
    best_h = h.copy()
    best_c5 = c5
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find current maximum overlap and corresponding shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k] * dx
        
        # Get gradient for that shift
        if max_k < n_points:
            # forward shift
            grad = np.zeros(n_points)
            for j in range(n_points - max_k):
                grad[j] += -h[j + max_k]
        else:
            # backward shift
            shift = max_k - n_points + 1
            grad = np.zeros(n_points)
            for j in range(shift, n_points):
                grad[j] += -h[j - shift]
        
        # Update step
        h = h - alpha * grad
        h = project_to_feasible(h)
        
        # Re-evaluate
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Adaptive learning rate
        if i % 100 == 0:
            alpha *= 0.99
    
    # Convex refinement every 50 iterations
    h = best_h.copy()
    for i in range(50):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Solve relaxed convex problem
        h_var = cp.Variable(n_points)
        constraints = [cp.sum(h_var) == n_points / 2.0, h_var >= 0, h_var <= 1]

        # Add overlap constraints - top 5 most binding
        corr = np.correlate(h, 1 - h, mode='full')
        top_k_indices = np.argsort(corr)[-5:][::-1]
        for k in top_k_indices:
            # Add constraint based on current values
            max_index = len(corr) - 1
            if k < n_points:
                # Positive shift
                corr_sum = cp.sum(cp.multiply(h_var[:n_points - k], 1 - h_var[k:]))
            else:
                # Negative shift
                shift = k - max_index + 1
                corr_sum = cp.sum(cp.multiply(h_var[shift:], 1 - h_var[:n_points - shift]))
            constraints.append(corr_sum <= 0.5 * dx * n_points)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h = np.array(h_var.value).flatten()
                c5 = evaluate_erdos_solution(h, dx)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
        except:
            pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Local optimization
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    return (list(best_h), best_c5, n_points)
