import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    T = T_initial
    delta_max = 0.2
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Ensure new values stay in [0,1]
        new_h = h_current.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Clamp to bounds
        new_h[i] = np.clip(new_h[i], 0, 1)
        new_h[j] = np.clip(new_h[j], 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(new_h) - (n_points / 2)
        if abs(sum_diff) > 1e-10:
            # Adjust the first point to maintain sum
            new_h[0] -= sum_diff
            new_h[0] = np.clip(new_h[0], 0, 1)
        
        # Compute new objective
        c5_candidate = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on simulated annealing
        if c5_candidate < best_c5 or np.random.uniform() < np.exp(-(c5_candidate - best_c5) / T):
            h_current = new_h
            if c5_candidate < best_c5:
                best_h = new_h.copy()
                best_c5 = c5_candidate
        
        # Cooling schedule
        T *= alpha
        
        # Dynamic delta adjustment
        if iteration % 1000 == 0 and delta_max > 0.01:
            delta_max *= 0.99
    
    return (list(best_h), best_c5, n_points)
