import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 100
    num_generations = 1000
    mutation_rate = 0.1
    tournament_size = 2
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def tournament_selection(pop, k=tournament_size):
        selected = random.sample(pop, k)
        return min(selected, key=fitness)
    
    def crossover(parent1, parent2):
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return repair_h(child1), repair_h(child2)
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] += np.random.normal(0, 0.01)
        return repair_h(h)
    
    # Evolution loop
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < num_generations:
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best individual
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness_scores[best_idx]
        
        # Create new population
        new_pop = [population[best_idx].copy()]  # Elitism
        
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_pop.extend([child1, child2])
        
        population = new_pop[:pop_size]
        generation += 1
    
    # Final refinement
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
