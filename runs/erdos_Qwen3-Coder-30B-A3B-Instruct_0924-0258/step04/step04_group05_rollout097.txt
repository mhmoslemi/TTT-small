import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population for genetic search
    population_size = 40
    population = []
    if 'initial_h_values' in globals():
        # Generate diverse initial population from initial_h
        for _ in range(10):
            perturbed = h_start + np.random.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            # Re-normalize
            perturbed = perturbed * (n_points/2) / np.sum(perturbed)
            population.append(perturbed)
        
        # Add random valid solutions
        for _ in range(5):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            population.append(h_rand)
    else:
        # Generate starting population randomly
        for _ in range(15):
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
    
    # Add some variations of the initial solution
    if 'initial_h_values' in globals():
        for _ in range(10):
            h = h_start + np.random.normal(0, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
    
    # Fill up to population_size
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Initialize tabu list
    tabu_list = []
    tabu_limit = 50
    stagnation_count = 0
    
    # Genetic algorithm parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    max_stagnation = 50
    
    # Main GA loop
    for gen in range(20000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness for each
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Find best and update global
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_h = population[min_idx].copy()
            best_c5 = fitness[min_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adapt parameters if stagnation occurs
        if stagnation_count > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        elif stagnation_count > 10:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        
        # Create new population
        new_population = [population[i] for i in top_indices]
        
        # Elitism: keep the best
        new_population.append(population[min_idx])
        
        # Generate new solutions
        while len(new_population) < population_size:
            # Selection
            if random.random() < crossover_prob and len(top_indices) >= 2:
                i1, i2 = random.sample(top_indices, 2)
                parent1, parent2 = population[i1], population[i2]
                
                # Crossover: uniform mask
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Mutation: random
                parent_idx = random.choice(top_indices)
                child = population[parent_idx].copy()
                
                # Add noise to a few indices
                indices = random.sample(range(n_points), max(1, n_points // 10))
                for i in indices:
                    child[i] += np.random.normal(0, mutation_rate)
                child = np.clip(child, 0, 1)
            
            # Normalize
            child = child * (n_points/2) / np.sum(child)
            
            # Add to tabu list if not present
            # Normalize to integer for comparison
            child_hash = tuple((child * 1000).astype(int))
            if child_hash not in tabu_list:
                new_population.append(child)
                tabu_list.append(child_hash)
                if len(tabu_list) > tabu_limit:
                    tabu_list.pop(0)
            else:
                # If in tabu, make a different random solution
                new_child = np.random.rand(n_points)
                new_child = new_child * (n_points/2) / np.sum(new_child)
                new_population.append(new_child)
        
        population = new_population[:population_size]
        
        # Sometimes refine with gradient descent
        if gen % 10 == 0:
            # Take top candidates
            top_fitness = []
            top_candidates = []
            for i in range(min(5, len(population))):
                top_fitness.append(fitness[i])
                top_candidates.append(population[i])
                
            # Refine each with local search
            for h in top_candidates:
                def obj(h_ref):
                    return evaluate_erdos_solution(h_ref, dx)
                    
                def con(h_ref):
                    return np.sum(h_ref) - n_points / 2.0
                    
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': con}
                try:
                    res = minimize(obj, h, method='SLSQP', bounds=bounds, 
                                   constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_new = res.x
                        c5_new = obj(h_new)
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
                except:
                    pass
                
        # Early stopping if target met
        if best_c5 < 0.3807:
            break
    
    return (list(best_h), best_c5, n_points)
