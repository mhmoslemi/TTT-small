import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Algorithm
    n_particles = 50
    max_iter = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    for i in range(n_particles):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal bests
    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_erdos_solution(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iterations
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to maintain constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate fitness
            c5 = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
        
        # Periodically refine with convex programming
        if iter % 50 == 0 and iter > 0:
            try:
                # Try to refine the global best with convex optimization
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Ensure we don't have infeasible constraints
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < gbest_c5:
                            gbest = h_refined
                            gbest_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Run several random restarts with SLSQP
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = gbest + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < gbest_c5:
                        gbest = h_refined
                        gbest_c5 = c5_refined
            except:
                pass
    
    # Ensure final solution meets constraints
    final_h = np.clip(gbest, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
