import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 1000
    tournament_size = 2
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def repair_individual(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    population = [repair_individual(create_individual()) for _ in range(population_size)]
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population[0] = h_start.copy()
    
    # Fitness evaluation for whole population
    def evaluate_population(pop):
        return [evaluate_erdos_solution(ind, dx) for ind in pop]
    
    # Tournament selection
    def tournament_selection(pop, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(pop[winner_idx].copy())
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.01)
        return repair_individual(mutated)
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        fitnesses = evaluate_population(population)
        best_in_generation = min(fitnesses)
        if best_in_generation < best_c5:
            best_c5 = best_in_generation
            best_h = population[np.argmin(fitnesses)].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        new_pop = sorted_pop[:elite_size]
        
        # Selection, crossover and mutation
        selected = tournament_selection(sorted_pop, sorted_fitnesses)
        for i in range(0, population_size - elite_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
        
        # Trim population to exact size
        population = new_pop[:population_size]
        
        # Occasionally add some new individuals for diversity
        if gen % 10 == 0:
            for i in range(5):
                if len(population) < population_size:
                    population.append(repair_individual(create_individual()))
    
    # Final refinement
    try:
        final_h = repair_individual(best_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
