import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, norm, sum_entries

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create structured initial guess with three clusters
        cluster_size = 5
        cluster_centers = [0.5, 1.0, 1.5]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Binary search parameters  
    low = 0.3
    high = 0.38085
    precision = 1e-6
    
    # Initialize with uniform proposal between 0.1 and 0.9
    def init_h_estimate():
        h = np.random.rand(n_points) * 0.8 + 0.1
        h = h * (n_points/2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Convert a vector to a structured form close to init_h_values
    def project_to_structured(h):
        # Keep a small random perturbation
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def try_feasible(h, C_val):
        """Try to validate feasibility using convex relaxation"""
        try:
            # Set up the convex optimization problem
            h_var = Variable(n_points)
            
            # Objective (should be zero)
            obj = Minimize(0)
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, sum_entries(h_var) == n_points / 2]
            
            # Generate all overlap constraints: sum h[i] * h[i+k] >= 1 - C*dx
            dx = 2.0 / n_points
            for shift in range(n_points):
                if shift == 0:
                    continue  # skip zero-shift, not needed for the max
            
                # Use cross-correlation approach to ensure all shifts
                corr_term = 0
                for i in range(n_points):
                    j = (i + shift) % n_points
                    corr_term += h_var[i] * h_var[j]
                
                constraints.append(corr_term >= 1 - C_val * dx)
            
            # Solve the convex program
            prob = Problem(obj, constraints)
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=500)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h_var.value
            else:
                return None
                
        except Exception:
            return None
            
    # Binary search for the tightest C5 bound
    best_feasible_C = 0.38085
    max_iter = 50 
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        C = (low + high) / 2.0
        if abs(high - low) < precision:
            break
            
        # Try to find a feasible solution for this C
        h_candidate = best_h.copy()
        h_feasible = try_feasible(h_candidate, C)
        
        if h_feasible is not None:
            # Valid solution found for C, try to improve it
            curr_c5 = evaluate_erdos_solution(h_feasible, 2.0/n_points)
            if curr_c5 < best_c5:
                best_h = h_feasible.copy()
                best_c5 = curr_c5
                
            # Try more refined optimizations, local search
            def obj(h):
                return evaluate_erdos_solution(h, 2.0/n_points)
            
            # SLSQP refinement to improve local behavior
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                # Refine with a small local optimization
                res = minimize(obj, h_feasible, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-6})
                if res.success and res.fun < best_c5:
                    best_h = res.x.copy()
                    best_c5 = res.fun
            except:
                pass
                
            high = C
            if C < best_feasible_C:
                best_feasible_C = C
        else:
            low = C
            # Try a new random restart to escape local minima
            h_try = init_h_estimate()
            c_try = evaluate_erdos_solution(h_try, 2.0/n_points)
            if c_try < best_c5:
                best_h = h_try.copy()
                best_c5 = c_try
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
 
    return (list(best_h), best_c5, n_points)
