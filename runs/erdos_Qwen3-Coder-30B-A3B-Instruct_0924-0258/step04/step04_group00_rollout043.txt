import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local optimization with restarts and better constraints
    max_restarts = 50
    restart_count = 0
    max_iter = 100
    
    while time() - start_time < budget_s * 0.95:
        # Random perturbation with better initialization
        h_restart = best_h + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # More robust optimization
        try:
            # Objective function
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            # Constraints
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            # Bounds
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Try multiple optimizers
            optimizers = ['SLSQP', 'L-BFGS-B']
            best_local = float('inf')
            best_local_h = h_restart
            
            for opt in optimizers:
                try:
                    res = minimize(objective, h_restart, method=opt,
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': max_iter, 'ftol': 1e-8})
                    if res.success:
                        c5_local = objective(res.x)
                        if c5_local < best_local:
                            best_local = c5_local
                            best_local_h = res.x
                except:
                    continue
            
            if best_local < best_c5:
                best_h = best_local_h
                best_c5 = best_local
                restart_count = 0
            else:
                restart_count += 1
                
        except Exception as e:
            restart_count += 1
        
        # Occasionally use convex relaxation to improve bounds
        if time() - start_time < budget_s * 0.8 and restart_count % 7 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add key constraints only (not all possible shifts for speed)
                # Use only small shift values
                max_shift = min(10, n_points // 2)
                for k in range(1, max_shift):
                    if k < n_points // 2:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        # Early stopping condition
        if restart_count > max_restarts:
            break
    
    # Final improvement with local search
    try:
        # Try to improve current solution by local perturbations
        h_final = best_h.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.99:
                break
            idx = np.random.randint(0, n_points)
            step = np.random.normal(0, 0.02)
            h_trial = h_final.copy()
            h_trial[idx] += step
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial, dx)
            if c5_trial < best_c5:
                h_final = h_trial
                best_c5 = c5_trial
        best_h = h_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
