import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Re-parameterize using logistic function
    # h = sigmoid(z + b) where sum(h) = n_points/2
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def compute_b(z):
        h = sigmoid(z)
        return np.log((n_points/2) / np.sum(h) - 1)
    
    def reparameterize(z):
        b = compute_b(z)
        return sigmoid(z + b)
    
    # Initialize z from h_start
    h_current = h_start.copy()
    z_current = np.log(h_current / (1 - h_current))
    b_current = compute_b(z_current)
    h_current = sigmoid(z_current + b_current)
    
    def objective(z):
        h = reparameterize(z)
        return evaluate_erdos_solution(h, dx)
    
    # Gradient based optimization with line search
    max_iter = 3000
    learning_rate = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        h = reparameterize(z_current)
        c5_val = evaluate_erdos_solution(h, dx)
        
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
        
        # Evaluate all correlations to find worst case
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Gradient of overlap w.r.t h
        # For shift k_max:
        # C_k = sum_i h[i] * (1 - h[i + k_max])
        # dC_k/dh[i] = 1 - h[i + k_max] - h[i] * (1 - h[i + k_max]) * (1 - h[i + k_max])?
        # Actually let's compute dC_k/dh directly:
        # dC_k/dh[i] = 1 - h[i + k_max]
        
        # More accurate gradient computation
        grad = np.zeros_like(h)
        for i in range(len(h)):
            k = k_max - i + len(h) - 1  # The shift from h[i] to h[i + k_max]
            if 0 <= k < len(h):
                grad[i] = 1 - h[k]
        
        # Our parametrization: h = sigmoid(z + b), and b depends on z
        # We need to use chain rule to differentiate w.r.t. z
        h_new = sigmoid(z_current + b_current)
        grad_h_z = h_new * (1 - h_new)  # d(h)/d(z) = h*(1-h)
        
        # chain rule to get dC/dz
        grad_z = grad * grad_h_z
        
        # Update z
        z_current -= learning_rate * grad_z
        b_current = compute_b(z_current)  # update b to maintain sum constraint
        
        # Basic line search
        h_test = reparameterize(z_current)
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test >= c5_val:
            # Backtrack if no improvement
            z_current += learning_rate * grad_z  # Undo step
            learning_rate *= 0.5  # Reduce learning rate
        else:
            learning_rate *= 1.05  # Increase learning rate
        
        # Keep learning rate bounded
        learning_rate = max(1e-6, min(0.5, learning_rate))
    
    # Final evaluation at best solution
    final_h = reparameterize(z_current)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
        
    # Ensure solution is valid
    best_h = np.clip(best_h, 0, 1)
    
    return (list(best_h), best_c5, n_points)
