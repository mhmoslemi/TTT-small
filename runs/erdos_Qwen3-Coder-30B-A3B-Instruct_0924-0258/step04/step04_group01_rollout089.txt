import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create clusters at strategic positions
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define objective function
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Bounds: [0, 1] for all elements
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use BOBYQA optimization with constraints
    # BOBYQA is suitable for bound-constrained problems with expensive function evaluations
    options = {'maxiter': 500, 'disp': False}
    
    # Try multiple random restarts to avoid local minima
    num_restarts = 10
    for restart in range(num_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start from a slightly perturbed version of current best
        h_init = best_h + np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
        try:
            # Use trust-constr which handles both bounds and constraints in a robust way
            res = minimize(objective, h_init, method='trust-constr', 
                          bounds=bounds, 
                          constraints={'type': 'eq', 'fun': constraint_sum},
                          options=options)
            
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        except:
            continue
    
    # Final cleanup to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
