    max_overlap = np.max(corr) * dx
    return max_overlap

        current_mutation_step = mutation_step_start - (mutation_step_start - mutation_step_end) * (gen_count / max_generations)

            except:
                pass

    return (list(best_h), actual_c5, n_points)
