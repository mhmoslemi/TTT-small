import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary approach
    population_size = 50
    mutation_step = 0.05
    selection_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    num_generations = 0
    while (time() - start_time) < budget_s * 0.95 and num_generations < max_generations:
        # Evaluate fitness of each individual
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution found so far
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Create new generation
        new_population = population[:int(population_size * selection_rate)]
        num_to_create = population_size - len(new_population)
        
        # Mutation and crossover
        for _ in range(num_to_create):
            # Selection of parents
            parent1 = population[np.random.randint(0, int(population_size * selection_rate))]
            parent2 = population[np.random.randint(0, int(population_size * selection_rate))]
            
            # Crossover
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            noise = np.random.normal(0, mutation_step, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            
            # Renormalize
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        num_generations += 1
        
        # Decrease mutation step
        mutation_step = max(0.001, mutation_step * 0.995)
    
    # Refine top solutions with gradient-based methods
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        top_individuals = population[:5]
        for h in top_individuals:
            if time() - start_time > budget_s * 0.95:
                break
                
            # SLSQP refinement
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h.copy(), method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
