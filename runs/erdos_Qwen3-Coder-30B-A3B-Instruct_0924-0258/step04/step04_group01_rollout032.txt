import numpy as np
from cvxpy import *
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a more structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Add two peaks
        peak1 = 0.5
        peak2 = 1.5
        width = 0.2
        for i in range(n_points):
            x = i * dx
            h_start[i] = max(0, 1.0 - abs(x - peak1) / width) * max(0, 1.0 - abs(x - peak2) / width)
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iterations = 60
    best_h = h_start.copy()
    best_c5 = float('inf')
    
    # Discretization
    dx = 2.0 / n_points
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate LP for this candidate C₅
        h = Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(sum(h) == n_points / 2.0)
        
        # Overlap constraints for all lags
        for k in range(2 * n_points - 1):
            # Create indices for valid shifts and compute overlap
            if k < n_points:
                # Left shift case
                indices = list(range(n_points - k))
                term = sum(h[i] * (1 - h[i + k]) for i in indices)
            else:
                # Right shift case
                indices = list(range(k - n_points + 1, n_points))
                term = sum(h[i] * (1 - h[i - k + n_points]) for i in indices)
            
            constraints.append(term <= mid / dx)
        
        # Solve feasibility problem (minimize 0)
        prob = Problem(Minimize(0), constraints)
        
        try:
            prob.solve(solver=GLPK, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                # Solution is feasible, try to improve C₅
                h_solution = h.value
                current_c5 = max(np.correlate(h_solution, 1 - h_solution, mode='full')) * dx
                if current_c5 < best_c5:
                    best_h = h_solution.copy()
                    best_c5 = current_c5
                high = mid
            else:
                # Not feasible, increase C₅
                low = mid
        except:
            # If solving fails, try another approach or fallback
            low = mid
            
    # Final check and refinement
    final_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    # Ensure constraints are met
    if abs(np.sum(best_h) - n_points / 2.0) > 1e-6:
        best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (list(best_h), final_c5, n_points)
