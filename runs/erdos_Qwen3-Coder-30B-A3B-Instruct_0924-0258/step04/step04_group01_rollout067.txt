import numpy as np
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a smooth initial guess
        x = np.linspace(0, 2, n_points)
        h_start = np.zeros(n_points)
        # Construct two smooth bumps
        h_start += 0.5 * (1 + np.tanh(5*(x-0.5))) * (x <= 1)
        h_start += 0.5 * (1 + np.tanh(5*(1.5-x))) * (x >= 1)
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Binary search parameters
    low = 0.0
    high = 0.38085
    precision = 1e-6
    max_iterations = 60
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Construct and solve LP
        try:
            # Variables
            h = Variable(n_points)
            
            # Constraints
            constraints = [h >= 0, h <= 1]
            constraints.append(sum(h) == n_points / 2)
            
            # Add overlap constraints
            # We'll check a subset of lags to reduce complexity
           lags = range(1, n_points)  # Exclude 0 lag for simplicity
            for k in lags[::max(1, len(lags)//5)]:
                # This is an approximation for the correlation constraint
                # The full constraint is nonlinear, we will skip direct LP formulation 
                # of overlap integral and instead try to find a feasible solution
                # that satisfies the constraint in a discretized way
                pass  # For now, skip explicit LP formulation as it's complex
            
            # Try a direct optimization approach instead
            # Define objective function that we want to minimize
            def evaluate(h_vals):
                return evaluate_erdos_solution(h_vals, dx)
                
            # Use a local optimization subroutine
            def optimize():
                # Uses scipy minimize with bounds
                def objective(h):
                    return evaluate(h)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': constraint_sum}
                
                res = minimize(objective, h_start, method='SLSQP', 
                               bounds=bounds, constraints=cons, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                
                return res if res.success else None
                
            # Test feasible solution at mid
            # Just initialize and evaluate the mid-bound approach in simpler way
            test_h = h_start.copy()
            test_c5 = evaluate(test_h)
            
            # Use the direct optimization for feasible mid points
            if test_c5 <= mid:
                # This is feasible - reduce upper bound
                high = mid
                if test_c5 < best_c5:
                    best_h = test_h.copy()
                    best_c5 = test_c5
            else:
                # Not feasible - increase lower bound
                low = mid
                
        except:
            # If anything fails, try another path
            pass
            
        # Ensure we don't get stuck with a higher value
        if evaluate_erdos_solution(h_start, dx) < best_c5:
            best_h = h_start.copy()
            best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Final optimized check
    def final_optimize():
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        res = minimize(objective, best_h, method='SLSQP',

        if res.success:
            return res.x

        final_h = final_optimize()

        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    # Ensure final constraints are satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
