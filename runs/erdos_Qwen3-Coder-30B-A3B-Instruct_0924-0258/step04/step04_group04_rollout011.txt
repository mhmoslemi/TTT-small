```python
import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.0
        h_start[-1] = 0.0
        h_start = h_start * (n_points/2) / np.sum(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search on lambda (minimum auto-correlation)
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iter = 60

    def compute_auto_correlations(h, dx):
        # Compute all auto-correlations
        corr = np.correlate(h, 1-h, mode='full')
        # The folding happens automatically in np.correlate with mode='full'
        # Here we compute for all shifts k in [0, 2]
        # The correlation at shift k is corr[n_points-1+k] (we shift by n_points-1)
        # For our domain [0,2], the shifts are from 0 to 2,
        # and the indices in full correlation array that correspond to these:
        # k=0, index = n_points-1
        # k=1, index = n_points
        # k=2, index = n_points+1  
        # etc., but shifted from 0 to 2. We thus map the full correlation array
        # properly.
        # Actually, to align shifts with index in full, we identify:
        # shift k = 0 maps to index n_points-1 in the full corr array
        # shift k = 1 maps to index n_points in the full corr array
        # ...
        # General rule: shift k maps to index n_points-1 + k
        # But we need to ensure we handle the full length constraint properly.
        # We compute shift values [0, dx, 2*dx, ..., 2*dx]
        # But since dx = 2 / n_points, the shift values are: 0, 2/n_points, 4/n_points, ..., 2
        # So k values: 0, 1, 2, ..., n_points (n_points+1 steps)
        # For time-shift k, corresponding correlation value is corr[n_points-1+k]
        ac_shifts = []
        for i in range(0, n_points+1):
            ac_shifts.append(corr[n_points-1+i] * dx)
        return np.array(ac_shifts)

    def attempt_feasible(h_start, lambda_val):
        # Computes gradient of min AC and tries to increase it toward lambda_val
        # Given h_start, test if we can meet lambda_val constraint
        ac_vals = compute_auto_correlations(h_start, dx)
        min_k = np.argmin(ac_vals)
        min_ac = ac_vals[min_k]
        
        if min_ac >= lambda_val:
            return h_start.copy(), True
        
        # Now we want to increase the minimum autocorrelation
        # Formulate this as an optimization problem:
        # max sum over all x, x+k of h(x)(1-h(x+k)) such that sum h = 1  
        # For the specific minimum point k_min, we have:
        # min_ac = sum(h(i) * (1 - h(i + k_min))) = sum(h(i) - h(i)*h(i+k_min))  
        # This is related to the gradient of min_ac w.r.t. h
        
        # For now, we will try to increase min_ac by using gradient ascent on a
        # projected greedy move, without full blown optimization:
        # direct gradient step move
        h = h_start.copy()
        learning_rate = 0.1
        max_iter = 100
        for iter in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            ac_vals = compute_auto_correlations(h, dx)
            min_k = np.argmin(ac_vals)
            min_ac = ac_vals[min_k]
            if min_ac >= lambda_val:
                return h, True
            # Simple gradient-based approach:
            # Approximate gradient of min_ac w.r.t h at each point
            # Gradients: for item h[i] contributing to ac with (x+k) where x = i, k = k_min
            # if k_min is positive, it corresponds to h[i+k_min] in the original h
            # The gradient of h[i] * (1 - h[i+k_min]) w.r.t. h[i] is (1 - h[i+k_min])
            # Gradient of h[i] * (1 - h[i-k_min]) w.r.t. h[i] is (1 - h[i-k_min])
            # But since we are sampling and h values are computed for all x, 
            # h[i-k_min] only exists for i >= k_min. For i < k_min, it's 0 or off the sampling.
            # The full gradient is: 
            # d/dh[i] (sum of h[j] (1-h[j+k]) over j for k = k_min) 
            # = (1 - h[i+k_min]) for case j=i
            
            # So the gradient is: 1 - h[i+k_min] at the current point i
            # For each i in [0, n_points-1]
            # But we take averaged gradient
            grad = np.zeros_like(h)
            # add gradient term where value of i was used in computing AC at shift k_min
            # i corresponds to h[i], part of product h[i] * (1 - h[i+k_min])
            # i --> apply 1-h[i+k_min] 
            # So we use for positive k_min: slope at h[i] = 1 - h[i+k_min]
            # But to calculate decrease, we can add negative gradient direction:
            # All the points contributing to this sum are the ones at which
            # shift k_min is effectively taken, i.e., 
            # the i-th element of h contributes to (i + k_min)-th element of h
            # so each index contributes 1-h[i+k_min] to gradient of h[i]
            # OR simply, h[i+k_min] contributes to gradient of h[i] as a factor of -1
            
            # Let's take conservative Scala approach for good gradient estimate:
            # We perform gradient ascent on current min_AC
            # Approximate gradient of h[i] w.r.t min_ac (over look k_min shift)
            for i in range(n_points):
                if i + min_k < n_points:
                    grad[i] += -1.0  # Gradient in direction that increases h[i] * (1 - h[i + min_k])
                else:
                    grad[i] += 0.0  # Padding equivalent, do nothing
            
            # Apply gradient step
            h_new = h - learning_rate * grad
            
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            if np.sum(h_new) > n_points/2 + 1e-8:
                # Adjust to be at exact sum
                h_new = h_new * (n_points/2) / np.sum(h_new)
                
            h = h_new
            
        # After optimization, return with possibly updated h
        return h, (np.min(compute_auto_correlations(h, dx)) >= lambda_val)

    # Binary search loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        feasible_h, feasible = attempt_feasible(best_h, lambda_mid)
        
        if feasible:
            lambda_low = lambda_mid
            if evaluate_erdos_solution(feasible_h, dx) < best_c5:
                best_h = feasible_h
                best_c5 = evaluate_erdos_solution(feasible_h, dx)
        else:
            lambda_high = lambda_mid
    
    # Final refinement of best solution via SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass

    return (list(best_h), best_c5, n_points)
```
