import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create structured initial guess with 3 clusters
        cluster_width = 3
        centers = [0.5, 1.0, 1.5]
        for c in centers:
            start = max(0, int((c - cluster_width/2)/dx))
            end = min(n_points, int((c + cluster_width/2)/dx))
            h_start[start:end] = 1.0 / cluster_width
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    elite_size = 5
    
    # Create initial population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            # Start with the best solution and add noise
            h = best_h + np.random.normal(0, 0.02, n_points)
        else:
            # Random initialization
            h = np.random.rand(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # GA main loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(c5_val)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = population[:elite_size]  # Keep best
        
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                parent1 = population[np.random.randint(0, population_size//2)]
                parent2 = population[np.random.randint(0, population_size//2)]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Apply mutations
                if np.random.rand() < mutation_rate:
                    mutate_idx = np.random.randint(0, n_points)
                    child1[mutate_idx] = np.random.rand() * 0.5
                if np.random.rand() < mutation_rate:
                    mutate_idx = np.random.randint(0, n_points)
                    child2[mutate_idx] = np.random.rand() * 0.5
                    
                # Ensure constraints
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.extend([child1, child2])
            else:
                # Mutation only
                parent = population[np.random.randint(0, population_size//2)]
                child = parent + np.random.normal(0, 0.02, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final SQP refinement of best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-6})
        if res.success:
            refined_h = res.x
            refined_c5 = obj(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
