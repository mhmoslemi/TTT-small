import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 40
    population = []
    fitness = []
    
    # Create diverse initial population
    for i in range(10):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Add some random valid solutions
    for i in range(5):
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Fill up population
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # Tabu search parameters
    tabu_list = []
    tabu_limit = 50
    stagnation_counter = 0
    max_stagnation = 50
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Main evolutionary loop
    for gen in range(2000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Adaptive parameters
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Generate new population
        new_population = [population[0]]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if random.random() < crossover_prob and len(population) >= 2:
                parent1, parent2 = random.sample(population[:10], 2)
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Select parent
                parent = random.choice(population[:10])
                child = parent.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                mutate_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for idx in mutate_indices:
                    child[idx] = random.uniform(0, 1)
                # Repair
                child = child * (n_points/2) / np.sum(child)
                
            # Check tabu
            child_tuple = tuple(child)
            if child_tuple in tabu_list:
                child = population[random.randint(0, len(population)-1)].copy()
                child = child * (n_points/2) / np.sum(child)
                
            new_population.append(child)
            
            # Add to tabu list
            tabu_list.append(child_tuple)
            if len(tabu_list) > tabu_limit:
                tabu_list.pop(0)
        
        # Local refinement
        if gen % 10 == 0:
            for i in range(min(5, len(new_population))):
                h_refined = new_population[i]
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_refined, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        new_population[i] = h_refined
                except:
                    pass
        
        # Update population
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
