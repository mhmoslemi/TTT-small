import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient of overlap integral for shift k
        grad = np.zeros_like(h)
        # For overlap at shift k: integral of h(x) * (1 - h(x+k))
        # Gradient w.r.t. h[i] is: 1 - h[i+k] if i+k < n_points, 
        # and 1 - h[i-k] if i-k >= 0
        
        # Ensure k is in valid range
        if k >= n_points:
            k = n_points - 1
        if k < 0:
            k = 0
            
        if k == 0:
            # No shift, gradient is -1 for all positions
            grad.fill(-1.0)
        else:
            # Compare h[i] with h[i+k] values
            # We're computing d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} if k>0
            # and -h_{i-k} if k<0 (but that's inverse)
            # actually it's more involved as we want to consider where terms overlap
            # So we compute convolution gradient specially
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] = 1 - h[i + k]
                else:
                    grad[i] = 0  # Contribution to cross term doesn't exist
            if k > 0:
                for i in range(n_points):
                    if i >= k:
                        grad[i] += -h[i - k]  # negative contribution due to h_i * (1-h_{i+k})
                    else:
                        grad[i] += 0  # no corresponding term from h_{i-k}
        return grad

    # Gradient descent with adaptive step size
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    decay = 0.99
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.8:
        # Compute all overlaps and find the worst case
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (len(current_h) - 1)  # shift value
        
        grad = compute_gradient(current_h, k)
        
        # Update with gradient
        new_h = current_h - alpha * grad
        
        # Clip the values to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Project to satisfy constraint sum(h) * dx = 1
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate the new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / 0.01):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        iter_count += 1
        
        # Adjust learning rate
        if iter_count % 100 == 0:
            alpha *= decay
    
    # Periodic convex refinement
    iter_count = 0
    while (time() - start_time) < budget_s * 0.9 and iter_count < 20:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add key overlap constraints
            # Find top 5 worst overlaps to be treated in convex optimization
            corr_orig = np.correlate(best_h, 1 - best_h, mode='full')
            worst_indices = np.argpartition(corr_orig, -5)[-5:]
            for idx in worst_indices:
                k = idx - (len(best_h) - 1)
                # Sum of products of overlapping elements
                if k >= 0:
                    if k < n_points:
                        expr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        expr = cp.sum(cp.multiply(h[:0], 1 - h[:0]))
                else:
                    if -k < n_points:
                        expr = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
                    else:
                        expr = cp.sum(cp.multiply(h[:0], 1 - h[:0]))
                constraints.append(expr <= 0.5 * dx * n_points)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
        
        iter_count += 1
    
    # Final SLSQP polishing
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
