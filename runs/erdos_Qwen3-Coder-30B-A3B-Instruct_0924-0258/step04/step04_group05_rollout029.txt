import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Ensure normalization
    h_start = h_start * (n_points / 2.0) / np.sum(h_start)
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refinement with increased resolution and better exploration
    n_points = 40  # Higher resolution
    dx = 2.0 / n_points
    
    # Multi-start optimization with diverse perturbations
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Convex feasibility check function
    def check_feasibility(h_guess, c_max):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Compute correlation constraints for all shifts
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_max * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            return prob.status == cp.OPTIMAL
        except:
            return False

    # Use a more aggressive grid scan
    max_grid = 0.38085
    min_grid = 0.37
    step = (max_grid - min_grid) / 50
    
    current_best = best_c5
    best_h_final = best_h.copy()
    
    # Phase 1: Grid search with convex feasibility checks
    for c_test in np.arange(min_grid, max_grid, step):
        if time() - start_time > budget_s * 0.6:
            break
            
        # Quick feasibility check
        if check_feasibility(best_h_final, c_test):
            # Refine around this target
            h_init = best_h_final.copy()  # Random perturbation to diversity
            for _ in range(5):
                if time() - start_time > budget_s * 0.9:
                    break
                h_perturb = h_init + np.random.normal(0, 0.02, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
                
                try:
                    res = minimize(objective, h_perturb, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < current_best:
                            current_best = c5_refined
                            best_h_final = h_refined

                    continue

    # Phase 2: Final refinement

                if time() - start_time > budget_s - 10:
                    break
                h_perturb = h_final + np.random.normal(0, 0.01, n_points)
                h_perturb = np.clip(h_perturb, 0, 1)
                h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
                
                res = minimize(objective, h_perturb, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 150, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < current_best:
                        current_best = c5_refined
                        best_h_final = h_refined
        except:
            pass
    
    return (list(best_h_final), current_best, n_points)
