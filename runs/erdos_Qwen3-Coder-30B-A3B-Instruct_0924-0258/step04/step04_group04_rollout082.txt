import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient descent with line search
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    
    # For tracking progress
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time() - start_time < budget_s * 0.95:
        # Compute all overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient of overlap w.r.t. h for shift k_max
        # For a shift k, the overlap with h is sum(h[i] * (1 - h[i+k]))
        # So d/dh[i] overlap = (1 - h[i+k]) - h[i] for k != 0
        # But here we consider h[i] to the left, so for shift k_max:
        # For i = 0 to n_points-1, we need to look at position i+k_max
        # and h[i] gets an influence from 1-h[i+k_max] if k_max > 0
        # or from h[i-k_max] if k_max < 0
        # Adjusted: overlap = sum_{i=0}^{n_points-1} h[i] * (1 - h[i + shift])
        # So d/dh[i] overlap = 1 - h[i + shift] (if i+shift is valid)
        #                = 1 - h[i + shift] - h[i] (if i+shift is valid)
        # Actually, we need to be more accurate. The corr has shifted indices.
        # If shift k_max = k, then position i contributes to the overlap
        # because h[i] * (1 - h[i + k]) 
        # So d/dh[i] = (1 - h[i+k]) + (when i+k is valid: -1) 
        # Actually, more carefully:
        # When computing sum_{i=0}^{n_points-1} h[i] * (1 - h[i + k]), 
        # for a fixed i, the derivative wrt h[i] is 1 - h[i+k]
        # For i+k, the derivative wrt h[i+k] is -h[i]
        # This is not a good approach; use finite differences or explicit computation.
        # Let's build the Jacobian or implement a correct gradient approach
        
        # Alternatively, just do direct gradient descent on the most problematic shift
        # The overlap at shift k_max is: sum(h[i] * (1 - h[i+k_max]))
        # d/dh[i] = 1 - h[i+k_max] when i+k_max is valid, and we're differentiating 
        # wrt h[i]. (There are derivatives which depend on h[i+k] corresponding to 
        # positions where we are evaluating changes in h[i] and also are in the sum)
        # A simpler accurate approach: treat as a vector function
        # We use direct approx of gradient: finite differences or with correlate
        # i-offer a direct computation for the derivative at i for k_max
        # Let g = corr[k_max], and h = [h[i]], we consider effect of changing h[i]
        # contribution to overlap = h[i] * (1 - h[i+k_max])
        # d/dh[i] = (1 - h[i+k_max]) - h[i] * 0 (when k_max > 0 and i + k_max < n_points) 
        #          = (1 - h[i+k_max]) 
        # except when i = n_points-k_max, it should also include h[i]
        # Because the shift k_max from index i refers to input h[i+k_max], 
        # we compute derivative of h[i] with respect to overlap:
        # The loss = h[i] * (1 - h[i+k]) for i+k >= 0, i+k < n_points
        # d/dh[i] = (1 - h[i+k]) (we include a negative sign as we're computing d/dh[i] for effect on sum)
        # Actually given the overlap = sum h[i] * (1 - h[i+k])
        # This is the convolution h * (1 - h) with mode='full'.   
        
        # Compute gradient for k_max
        grad = np.zeros(n_points)
        # We need to compute d/dh[i] overlap at shift k_max
        # For all valid i, delta[shift + i] contributes like 1 - h[shift + i] 
        # So for i fixed, gradient w.r.t h[i] = dy/dh[i] 
        # If we look at the shift k = k_max, contribution to overlap = h[i] * (1 - h[i + k_max])
        # d/dh[i] (h[i] * (1 - h[i+k])) = (1 - h[i+k])
        # For shift k_max, i iterates from 0 to n_points - k_max - 1 (k_max is ij or k if blocked from left)
        # But ensure bounds
        if k_max <= n_points - 1: # valid shift; index to overlap map could start at 0
            # actually in scipy corr, it's -k_max to n_points-1; indices are at 0..2*n-2
            # e.g. if shift is 0, we look at h[i] * (1 - h[i]), 
            #       if shift 1, we look at h[i] * (1 - h[i+1]), etc.
            # corr[i] = sum_{j=0}^{n-1} h[j] * (1 - h[j + i-n+1]) for i = 0..2*n-2
            # Be careful with indexing...
            # i in corr corresponds to shift = i - (n-1) when going from k = -n+1 to n-1
            # our k_max = i - (n-1)
            # hence i = k_max + (n-1)
            corr_idx = k_max + (n_points - 1)  # Position in full correlation
            # Now we must get the h's that participate in this correlation
            for i in range(n_points):
                # h[i] contributes to correlation as h[i] * (1 - h[i + k_max - n+1])
                j = i + k_max - (n_points - 1)
                if 0 <= j < n_points:
                    # Now 1 - h[j] contributes to d/dh[i] = 1 - h[j]
                    grad[i] += (1 - best_h[j])
                    
        # Normalize gradient to prevent instability
        grad_total = np.sum(np.abs(grad))
        if grad_total > 1e-12:
            grad = grad / grad_total
        
        # Update with gradient descent
        h_new = best_h - step_size * grad
        
        # Project to valid region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute overlap with new h
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Check if improvement, but also implement backtracking line search
        if c5_new < best_c5 - tolerance:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
        else:
            # Reduce step size if no improvement
            step_size *= 0.99
            
        iter_count += 1
    
    # Final fit
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Try more optimization on the best solution
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        grad = np.zeros(n_points)
        corr_idx = k_max + (n_points - 1)
        for i in range(n_points):
            j = i + k_max - (n_points - 1)
            if 0 <= j < n_points:
                grad[i] += (1 - best_h[j])
        
        grad_total = np.sum(np.abs(grad))
        if grad_total > 1e-12:
            grad = grad / grad_total
        
        h_new = best_h - 0.001 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        c5_final = evaluate_erdos_solution(h_new, dx)
        if c5_final < best_c5:
            best_h = h_new
            best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
