import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    max_iter = 1000
    step_size = 0.01
    patience = 100
    patience_counter = 0
    
    h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Identify the shift that causes max overlap
        shift = max_idx - (len(h) - 1)
        
        # Compute gradient for this shift
        grad = np.zeros_like(h)
        if shift == 0:
            for i in range(len(h)):
                grad[i] = -1
        elif shift > 0:
            # h[i] contributes to overlap with h[i-shift] 
            for i in range(shift, len(h)):
                grad[i] = -1
            for i in range(shift):
                grad[i] = 0
        else:
            # shift < 0
            shift_abs = -shift
            for i in range(shift_abs, len(h)):
                grad[i-shift_abs] = -1
            for i in range(shift_abs):
                grad[i] = 0
        
        # Add a small term to prevent numerical issues at boundaries
        # For now just use the pattern of the original gradient
        # Simplified approximation: gradient for overlap w.r.t. h[i] 
        # is (1 - h[i+shift]) - h[i] * delta(shift != 0)
        grad = np.zeros_like(h)
        if shift != 0:
            # For shift > 0, every h[i] contributes -1 to the derivative if there's a valid corresponding h[i-shift]
            for i in range(len(h)-abs(shift)):
                grad[i + abs(shift)] += -1
                grad[i] += -1
        else:
            # For shift = 0, every h[i] contributes -1 to its own derivative
            grad = np.full_like(h, -1)
            
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Perform gradient descent step
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Check if the new solution is better
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            step_size *= 0.9
            patience_counter = 0
            
        h = h_new.copy()
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Fine tune the final result
        initial_h = best_h.copy()
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            from scipy.optimize import minimize
            res = minimize(obj, initial_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
