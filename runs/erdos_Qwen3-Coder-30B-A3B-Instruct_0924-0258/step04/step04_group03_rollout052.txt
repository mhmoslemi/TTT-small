import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhanced Simulated Annealing with pairwise perturbations
    def neighbor(h, step=0.05):
        # Pairwise perturbation
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.normal(0, step)
        h_new[i] += delta
        h_new[j] -= delta
        # Clamp values
        h_new = np.clip(h_new, 0, 1)
        # Adjust to keep sum constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        return h_new
    
    # SA parameters
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.9995
    max_steps = 10000
    
    # Track best
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Generate neighbor with adaptive step size
        step_size = 0.01 + (0.05 - 0.01) * (1 - temperature)
        new_h = neighbor(current_h, step_size)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
    
    # Convex refinement using CVXPY
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add optimization-based constraints directly targeting overlap
            # We'll use a proxy constraint: bound correlation at several points
            for k in range(0, min(5, 2 * n_points - 1)):
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                constraints.append(corr_sum <= best_c5 * dx * n_points)
            
            # Minimize a relaxed version of the objective
            objective = cp.Minimize(cp.sum(cp.multiply(h, 1 - h)))
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final gradient-based refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Multiple random restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb best found so far
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                            bounds=bounds, constraints=cons_sum, 
                            options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
