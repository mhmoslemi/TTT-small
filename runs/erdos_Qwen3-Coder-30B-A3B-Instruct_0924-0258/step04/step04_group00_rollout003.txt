import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 40, 50]  # Start with moderate resolution
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_current = best_h.copy()
        if res > n_points:
            # Interpolate to higher resolution
            h_current = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h_current)
        elif res < n_points:
            # Downsample
            h_current = np.array([np.mean(h_current[i:i+(n_points//res)]) for i in range(0, n_points, n_points//res)])
        
        n_points_curr = len(h_current)
        dx_curr = 2.0 / n_points_curr
        
        # Local optimization with more iterations at higher resolution
        max_iter = 150 if res > 35 else 100
        
        # Try multiple random restarts
        restarts = 15 if res > 35 else 10
        for i in range(restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Random restart
            h_restart = h_current + np.random.normal(0, 0.02, n_points_curr)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points_curr/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_curr / 2.0}
                bounds = [(0, 1) for _ in range(n_points_curr)]
                res_min = minimize(
                    lambda x: evaluate_erdos_solution(x, dx_curr),
                    h_restart,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'maxiter': max_iter, 'ftol': 1e-8}
                )
                if res_min.success:
                    h_refined = res_min.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx_curr)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                continue
        
        # Convex relaxation to tighten bounds
        try:
            # Only add constraints for a sample of shifts to keep it efficient
            num_constraints = min(10, n_points_curr // 2)
            shifts = np.random.choice(n_points_curr // 2, size=num_constraints, replace=False)
            shifts = np.concatenate([shifts, n_points_curr - shifts - 1])
            
            h_opt = cp.Variable(n_points_curr)
            constraints = [
                cp.sum(h_opt) == n_points_curr / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            for k in shifts:
                if k < n_points_curr // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points_curr-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points_curr+1:], 1 - h_opt[:n_points_curr-(k-n_points_curr+1)]))
                constraints.append(corr_sum <= best_c5 * dx_curr + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx_curr)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final refinement with full resolution
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (len(h_final)/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, 2.0 / len(h_final))
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, len(best_h))
