import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with projection
    def project(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = h * target_sum / current_sum
        else:
            h.fill(target_sum / len(h))
        return h
    
    # Compute gradient for a specific shift k
    def compute_gradient(h, k):
        grad = np.zeros_like(h)
        n = len(h)
        if k < n:
            # Convolution: h[i] contributes to overlap with (1-h[i+k]) 
            # For i = 0 to n-k-1, grad[i] += -1 * (1-h[i+k])
            for i in range(n - k):
                grad[i] -= 1 - h[i + k]
        else:
            # Reverse case: k >= n
            # h[i] contributes to overlap with (1-h[i-k]) for i = k-n+1 to n-1
            for i in range(k - n + 1, n):
                grad[i] -= 1 - h[i - k + n - 1]
        return grad
    
    # Optimize using gradient descent
    max_iter = 5000
    alpha = 0.1
    alpha_decay = 0.99
    h_current = h_start.copy()
    h_current = project(h_current, n_points/2)
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find worst shift k
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_best = np.argmax(corr)
        max_overlap = corr[k_best] * dx
        
        if max_overlap < best_c5:
            best_h = h_current.copy()
            best_c5 = max_overlap
        
        # Compute gradient for k_best
        grad = compute_gradient(h_current, k_best)
        
        # Update step
        h_new = h_current - alpha * grad
        h_new = project(h_new, n_points/2)
        
        # Check if improvement
        new_overlap = evaluate_erdos_solution(h_new, dx)
        if new_overlap < max_overlap:
            h_current = h_new
        else:
            # Reduce learning rate
            alpha *= alpha_decay
            # Perturb if stuck
            if np.random.rand() < 0.01:
                h_current += np.random.normal(0, 0.005, n_points)
                h_current = project(h_current, n_points/2)
        
        # Periodic convex refinement
        if it % 50 == 0 and it > 0:
            try:
                # Linear relaxation approach
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                # Add constraints for top 10 critical shifts from current h
                corr_current = np.correlate(h_current, 1 - h_current, mode='full')
                for k_idx in np.argsort(corr_current)[-10:]:
                    if k_idx < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k_idx], 1 - h[k_idx:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k_idx-n_points+1:], 1 - h[:n_points-(k_idx-n_points+1)]))
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polish with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
