import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h = 1 / (1 + np.exp(-z))
        return np.log(n_points / 2 / np.sum(h))
    
    b = compute_b(z)

    max_iterations = 5000
    lr = 0.1

        # Update h from z + b
        h = 1 / (1 + np.exp(-(z + b)))

        h = np.clip(h, 1e-6, 1 - 1e-6)
        
        # Compute b again to satisfy constraint
        b = compute_b(z)

        corr = np.correlate(h, 1 - h, mode='full')

        gradient = np.zeros(n_points)

        for i in range(n_points):
