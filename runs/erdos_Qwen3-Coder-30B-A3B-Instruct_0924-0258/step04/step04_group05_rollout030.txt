import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_old, n_new):
    """Linearly resample h_old to n_new points"""
    if len(h_old) == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def project_to_constraints(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup grid sizes
    n_coarse = 50
    n_fine = 100
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init_coarse = linear_resample(initial_h_values, n_coarse)
        h_init_fine = linear_resample(initial_h_values, n_fine)
    else:
        h_init_coarse = np.ones(n_coarse) * 0.5
        h_init_fine = np.ones(n_fine) * 0.5
        
    h_coarse = project_to_constraints(h_init_coarse, n_coarse)
    h_fine = project_to_constraints(h_init_fine, n_fine)
    
    best_h_coarse = h_coarse.copy()
    best_h_fine = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse grid optimization
    perturbation_scale = 0.03
    for step in range(100):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Perturb
        h_perturbed = h_coarse + np.random.normal(0, perturbation_scale, n_coarse)
        h_repaired = project_to_constraints(h_perturbed, n_coarse)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_repaired, dx_coarse)
        if c5_new < best_c5_coarse:
            best_c5_coarse = c5_new
            best_h_coarse = h_repaired.copy()
            
        # Update perturbation
        perturbation_scale *= 0.99
    
    # Fine grid refinement around best coarse solution
    h_fine = linear_resample(best_h_coarse, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    best_h_fine = h_fine.copy()
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    perturbation_scale = 0.02
    
    # Local optimization on fine grid
    for step in range(200):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_perturbed = h_fine + np.random.normal(0, perturbation_scale, n_fine)
        h_repaired = project_to_constraints(h_perturbed, n_fine)
        
        c5_new = evaluate_erdos_solution(h_repaired, dx_fine)
        if c5_new < best_c5_fine:
            best_c5_fine = c5_new
            best_h_fine = h_repaired.copy()
            
        # Update perturbation
        perturbation_scale *= 0.995
    
    # Final optimization using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        def objective(h):
            return evaluate_erdos_solution(h, dx_fine)
        
        def constraint_sum(h):
            return np.sum(h) - n_fine / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_fine)]
        
        res = minimize(objective, best_h_fine, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        
        if res.success:
            h_final = project_to_constraints(res.x, n_fine)
            c5_final = objective(h_final)
            if c5_final < best_c5_fine:
                best_h_fine = h_final
                best_c5_fine = c5_final
    
    # Return the best solution found
    return (list(best_h_fine), best_c5_fine, n_fine)
