import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary relaxation approach
    def binary_relaxation_search():
        nonlocal best_h, best_c5
        
        # Initialize binary decision variables
        h_bin = (h_start > 0.5).astype(float)
        h_bin = h_bin * (n_points / 2) / np.sum(h_bin)
        
        # MIP-based constraint tightening
        C_max = 0.4
        max_iters = 50
        iter_count = 0
        
        while iter_count < max_iters and time() - start_time < budget_s * 0.6:
            try:
                # Constraint generation
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add overlap constraints with current bound
                for k in range(1, n_points):
                    if k <= n_points // 2:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= C_max * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
                        C_max = best_c5
                else:
                    break
                    
            except Exception:
                break
                
            iter_count += 1
    
    # Local search refinement with binary mutation
    def refine_solution():
        nonlocal best_h, best_c5
        
        h_current = best_h.copy()
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        max_refinements = 200
        
        for i in range(max_refinements):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Binary mutation: toggle 10% of entries
            mutation_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            h_new = h_current.copy()
            
            # Mutate selected entries between 0, 0.5, 1 
            for idx in mutation_indices:
                old_val = h_new[idx]
                if old_val < 0.25:
                    h_new[idx] = 1.0
                elif old_val > 0.75:
                    h_new[idx] = 0.0
                else:
                    h_new[idx] = 0.5
            
            # Normalize
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
            # Occasionally refine even if no improvement
            if i % 10 == 0:
                h_current = best_h.copy()
            else:
                h_current = h_new
    
    # Execute algorithms
    binary_relaxation_search()
    refine_solution()
    
    return (list(best_h), best_c5, n_points)
