import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 35
    dx = 2.0 / n_points
    population_size = 40
    max_generations = 500
    tabu_limit = 50
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_limit = 50
    
    # Initialize population
    population = []
    fitness_history = []
    
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Add perturbed versions of initial solution
        for _ in range(10):
            h_perturbed = initial_h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            population.append(h_perturbed.copy())
        
        # Add random valid solutions
        for _ in range(5):
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            population.append(h_random.copy())
    else:
        # Start with uniform distribution
        h_uniform = np.ones(n_points) * 0.5
        population.append(h_uniform)
        
        # Add random valid solutions
        for _ in range(15):
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points/2) / np.sum(h_random)
            population.append(h_random.copy())
    
    # Fill up population
    while len(population) < population_size:
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random.copy())
    
    # Tabu list
    tabu_list = set()
    best_h = None
    best_c5 = float('inf')
    stagnation_counter = 0
    
    # Evaluation function
    def evaluate_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Repair function to ensure constraints
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-8:
            h = h * (n_points/2) / h_sum
        return h
    
    # Crossover operation
    def crossover(h1, h2):
        if random.random() > crossover_prob:
            return h1.copy(), h2.copy()
            
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child1 = np.where(mask, h1, h2)
        child2 = np.where(mask, h2, h1)
        
        return repair_h(child1), repair_h(child2)
    
    # Mutation operation
    def mutate(h, rate):
        h = h.copy()
        for i in range(n_points):
            if random.random() < rate:
                # Add Gaussian noise with adaptive std
                noise = np.random.normal(0, 0.02)
                h[i] += noise
        return repair_h(h)
    
    # Fitness evaluation with repair
    def evaluate_population(pop):
        fitnesses = []
        for i, h in enumerate(pop):
            h_repaired = repair_h(h)
            fitness = evaluate_fitness(h_repaired)
            pop[i] = h_repaired  # Update with repaired version
            fitnesses.append(fitness)
        return fitnesses
    
    # Evaluate initial population
    fitnesses = evaluate_population(population)
    
    # Update best solution
    for h, f in zip(population, fitnesses):
        if f < best_c5:
            best_c5 = f
            best_h = h.copy()
    
    # Main genetic algorithm loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitnesses[0] < best_c5:
            best_c5 = sorted_fitnesses[0]
            best_h = sorted_population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Adaptive parameters
        if stagnation_counter > stagnation_limit:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Create new population
        new_population = []
        
        # Elitism
        new_population.extend(sorted_population[:5])
        
        # Selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = sorted_population[random.randint(0, 15)]
            parent2 = sorted_population[random.randint(0, 15)]
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        fitnesses = evaluate_population(population)
        
        # Gradient-based refinement every 10 generations
        if generation % 10 == 0 and generation > 0:
            # Refine top 5 solutions
            for i in range(min(5, len(population))):
                h = population[i].copy()
                
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        h_refined = repair_h(h_refined)
                        population[i] = h_refined
                        fitnesses[i] = evaluate_erdos_solution(h_refined, dx)
                except:
                    pass
    
    # Final local optimization of best solution
    if best_h is not None:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_h = repair_h(final_h)
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
