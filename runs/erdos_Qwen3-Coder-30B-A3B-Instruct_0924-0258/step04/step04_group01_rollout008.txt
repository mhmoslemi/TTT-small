        # Create constraints for the feasibility problem
        constraints = [h >= 0, h <= 1]

        # For 50 points, it's manageable to create all

        # Since cvxpy does not support general overlap constraints directly,

        # Generate the constraints more carefully

    # Ensure that best_c5 is actual C5 value from final computed h
    best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
