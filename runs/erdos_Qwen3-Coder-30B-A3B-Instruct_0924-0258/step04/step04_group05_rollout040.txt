import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population and solution tracking
    n_points = 35
    dx = 2.0 / n_points
    population_size = 40
    tabu_limit = 50
    max_stagnation = 50
    
    # Tabu memory to store recent solutions
    tabu_memory = set()
    
    # Initialize population
    population = []
    fitness_scores = []
    
    # If initial solution exists, use it as base
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points) * 0.5
    
    # Generate diverse initial population
    for i in range(10):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        
    for i in range(5):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        
    # Fill up with random valid solutions
    for i in range(population_size - 15):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial fitness
    for h in population:
        f = evaluate_erdos_solution(h, dx)
        fitness_scores.append(f)
    
    best_idx = np.argmin(fitness_scores)
    best_h = population[best_idx].copy()
    best_c5 = fitness_scores[best_idx]
    
    # Parameters for evolution
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    stagnation_count = 0
    best_improvement = 0
    
    # Main search loop
    while time() - start_time < budget_s * 0.98:
        # Evaluate current best
        current_best = min(fitness_scores)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitness_scores)
            best_h = population[best_idx].copy()
            best_improvement = 0
        else:
            best_improvement += 1
            if best_improvement > max_stagnation:
                # Increase mutation rate and decrease crossover rate if stuck
                mutation_rate = min(0.5, mutation_rate + 0.1)
                crossover_prob = max(0.3, crossover_prob - 0.1)
                best_improvement = 0
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # For each generation, do selection-crossover-mutation
        new_population = []
        new_fitness = []
        
        # Tournament selection + crossover + mutation
        for _ in range(population_size):
            # Select two parents (tournament selection)
            t_size = 3
            idx1 = random.randint(0, population_size-1)
            idx2 = random.randint(0, population_size-1)
            
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Perform crossover
            if random.random() < crossover_prob:
                # Uniform crossover
                mask = np.random.randint(0, 2, n_points)
                child = np.where(mask == 1, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Apply mutation
            if random.random() < mutation_rate:
                # Add Gaussian noise
                mut_strength = 0.05 * (1 - mutation_rate) + 0.02
                noise = np.random.normal(0, mut_strength, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            
            # Add to new population
            new_population.append(child)
            
        # Evaluate new population
        for h in new_population:
            f = evaluate_erdos_solution(h, dx)
            new_fitness.append(f)
        
        # Local gradient refinement for top candidates (every 10 generations)
        for _ in range(0, population_size, 4):
            idx = np.argsort(new_fitness)[:4]  # Top 4 by fitness
            for i in range(len(idx)):
                h_init = new_population[idx[i]]  # Get top fitness candidates
                bounds = [(0, 1) for _ in range(n_points)]
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        f_refined = objective(h_refined)
                        if f_refined < new_fitness[idx[i]]:
                            new_population[idx[i]] = h_refined
                            new_fitness[idx[i]] = f_refined
                except:
                    pass
        
        # Update population
        population = new_population
        fitness_scores = new_fitness
        
        # Update tabu memory: remove oldest if at capacity
        if len(tabu_memory) >= tabu_limit:
            # For simplicity, we'll remove a random entry
            if tabu_memory:
                tabu_memory.pop()
                
        # Stop early if target met
        if best_c5 < 0.3807:
            break
    
    # Final validation and return
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
