import numpy as np
from cvxpy import *
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a smooth initial guess with a spike and a plateau
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Spike at left center
        spike_start = int(0.3/dx)
        spike_end = int(0.5/dx)
        h_start[spike_start:spike_end] = 1.0
        # Plateau in middle
        plate_start = int(0.6/dx)
        plate_end = int(0.8/dx)
        h_start[plate_start:plate_end] = 0.7
        # Normalize to sum to n_points / 2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    # Binary search bounds
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iterations = 60
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # LP feasibility check
        h_vars = Variable(n_points)
        
        # Objective: minimize 0 (feasibility check)
        objective = Minimize(0)
        
        # Constraints
        constraints = []
        
        # Bounds on h
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Sum constraint
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Overlap constraints
        for k in range(2 * n_points - 1):
            # Compute the dot product sum over the valid indices
            overlap_sum = 0
            # For indices that can be valid
            if k < n_points:
                # h[a] * (1 - h[a + k])
                for a in range(n_points - k):
                    overlap_sum += h_vars[a] * (1 - h_vars[a + k])
            else:
                # h[a] * (1 - h[a + k]) for shifted indices
                for a in range(k - n_points + 1, n_points):
                    overlap_sum += h_vars[a] * (1 - h_vars[a + k - n_points + 1])
            
            constraints.append(overlap_sum <= mid / dx)
        
        # Create and solve the problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=GLPK, verbose=False, feastol=1e-6, reltol=1e-6)
            if prob.status == 'optimal':
                # Feasible: try to find a better solution using this construction
                h_feasible = np.array([v.value for v in h_vars])
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible.copy()
                    best_c5 = c5_feasible
                high = mid
            else:
                # Infeasible: increase the bound
                low = mid
        except:
            # Failed to solve, try a different approach or increase bound
            low = mid

    # Ensure final answer is feasible
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
