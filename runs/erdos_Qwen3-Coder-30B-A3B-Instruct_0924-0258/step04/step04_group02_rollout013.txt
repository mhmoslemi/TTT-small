import numpy as np
from scipy.optimize import minimize
from time import time
from math import pi

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier parameterization (adaptive)
    def h_from_coeffs(coeffs, n_points):
        # Sigmoid of Fourier series
        x_vals = np.linspace(0, 2, n_points)
        fourier_part = coeffs[0] + np.sum([
            coeffs[2*i+1] * np.cos(pi * (i+1) * x_vals) + 
            coeffs[2*i+2] * np.sin(pi * (i+1) * x_vals)
            for i in range((len(coeffs)-1)//2)
        ], axis=0)
        h_vals = 1.0 / (1.0 + np.exp(-2.0 * fourier_part))
        return h_vals
    
    # Optimize with transformed parameters
    # Initial coefficients: just the DC term
    init_coeffs = np.random.normal(0, 0.1, 2*(n_points//4) + 1)
    init_coeffs[0] = 0.0  # Center around 0.5
    
    def objective(coeffs):
        h_vals = h_from_coeffs(coeffs, n_points)
        # Normalize so sum is exactly n_points/2
        h_vals = h_vals * (n_points/2) / np.sum(h_vals)
        return evaluate_erdos_solution(h_vals, dx)
    
    # Use scipy optimization with trust-constr
    try:
        res = minimize(objective, init_coeffs, method='trust-constr',
                       options={'maxiter': 500, 'verbose': 0})
        if res.success:
            h_final = h_from_coeffs(res.x, n_points)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Local refinement for better convergence
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 60:
        # Try multiple random restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            coeffs_restart = np.random.normal(0, 0.1, 2*(n_points//4) + 1)
            coeffs_restart[0] = 0.0
            
            try:
                res = minimize(objective, coeffs_restart, method='trust-constr',
                               options={'maxiter': 100, 'verbose': 0})
                if res.success:
                    h_refined = h_from_coeffs(res.x, n_points)
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
