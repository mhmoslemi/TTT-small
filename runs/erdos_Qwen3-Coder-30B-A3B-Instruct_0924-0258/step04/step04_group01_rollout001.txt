import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # 3-cluster construction
        cluster_width = 0.3
        centers = [0.4, 1.0, 1.6]
        for c in centers:
            start = max(0, int((c - cluster_width/2)/dx))
            end = min(n_points, int((c + cluster_width/2)/dx))
            h_start[start:end] = 1.0 / (2 * cluster_width)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Population-based GA with SQP refinement
    population_size = 20
    population = []
    
    # Initialize population
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main evolution loop
    max_iter = 300
    mutation_scale = 0.02
    crossover_rate = 0.5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all population members
        fitness = [evaluate_erdos_solution(h, 2.0/n_points) for h in population]
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # New population
        new_population = [population[0]]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection proportional to rank
            ranks = np.arange(len(population), 0, -1)
            probs = ranks / np.sum(ranks)
            
            # Select two parents
            parent_indices = np.random.choice(len(population), size=2, p=probs)
            h1, h2 = population[parent_indices[0]], population[parent_indices[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
            else:
                child1, child2 = h1.copy(), h2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < 0.3:  # 30% chance of mutation
                    # Gaussian mutation
                    child += np.random.normal(0, mutation_scale, n_points)
                    # Clip to bounds
                    child = np.clip(child, 0, 1)
                    # Normalize sum
                    child = child * (n_points/2) / np.sum(child)
                
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
        
        # Replace population
        population = new_population[:population_size]
        
        # Refine some individuals with SQP
        if iteration % 2 == 0:  # Every other iteration
            refine_indices = np.random.choice(len(population), size=5, replace=False)
            for idx in refine_indices:
                h_refine = population[idx]
                
                def obj(h):
                    return evaluate_erdos_solution(h, 2.0/n_points)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        h_refined = res.x
                        population[idx] = h_refined
                except:
                    pass

    # Final cleanup and validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
