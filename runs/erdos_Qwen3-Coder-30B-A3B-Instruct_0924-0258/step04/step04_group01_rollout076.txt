import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a 3-cluster structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster1 = (0.3, 0.5, 0.7)  # positions
        cluster2 = (1.3, 1.5, 1.7)
        cluster3 = (0.8, 1.2)  # positions
        for i, pos in enumerate(cluster1):
            start = max(0, int((pos - 0.2)/dx))
            end = min(n_points, int((pos + 0.2)/dx))
            h_start[start:end] = 0.6  # height
        for i, pos in enumerate(cluster2):
            start = max(0, int((pos - 0.2)/dx))
            end = min(n_points, int((pos + 0.2)/dx))
            h_start[start:end] = 0.6
        for i, pos in enumerate(cluster3):
            start = max(0, int((pos - 0.2)/dx))
            end = min(n_points, int((pos + 0.2)/dx))
            h_start[start:end] = 0.4
            
        # Normalize to ensure sum is n/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for genetic algorithm
    population_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Create initial population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            h = h_start + np.random.normal(0, 0.02, n_points)
        else:
            h = np.random.rand(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Add initial solution
    population[0] = h_start.copy()
    
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(c5_val)
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update global best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select top 50%
        top = population[:population_size//2]
        
        # Crossover and mutation
        new_population = []
        new_population.extend(top)
        
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(top, 2, replace=False)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Gaussian mutation
                    noise = np.random.normal(0, 0.02, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    # Renormalize
                    child = child * (n_points/2) / np.sum(child)
                    
                    # SQP refinement
                    def obj(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    
                    try:
                        res = minimize(obj, child, method='SLSQP', 
                                       bounds=bounds, constraints=cons_sum, 
                                       options={'maxiter': 50, 'ftol': 1e-6})
                        if res.success:
                            child = res.x
                    except:
                        pass
                
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
        
        population = new_population[:population_size]
    
    # Final refinement
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_final, n_points)
