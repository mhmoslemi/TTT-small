import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local refinement using SLSQP first
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Refine initial solution
    try:
        res = minimize(objective, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Multi-perturbation with adaptive scaling
    max_iter = 500
    sigma = 0.02
    retry_count = 0
    max_retries = 3
    decay_factor = 0.95
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate perturbation
        perturbation = np.random.normal(0, sigma, n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Renormalize
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Local refinement
        try:
            res = minimize(objective, h_perturbed, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_local = res.x
                c5_local = objective(h_local)
                
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
                    retry_count = 0  # Reset retry count on improvement
                else:
                    retry_count += 1
            else:
                retry_count += 1
        except:
            retry_count += 1
        
        # Adaptive perturbation
        if retry_count >= max_retries:
            # Increase perturbation scale and reset retry count
            sigma *= 1.1
            retry_count = 0
        else:
            # Decrease perturbation scale
            sigma *= decay_factor
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final validation
    final_c5_check = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5_check, n_points)
