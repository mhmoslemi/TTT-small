import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create initial construction with 35 points
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Three clusters
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_width = 0.25
        for c in cluster_centers:
            start_idx = max(0, int((c - cluster_width)/dx))
            end_idx = min(n_points, int((c + cluster_width)/dx))
            h_start[start_idx:end_idx] = 1.0
        # Normalize sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    low = 0.3
    high = 0.38085
    tolerance = 1e-6
    max_iterations = 50
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2.0
        
        # Try to find a feasible solution for mid
        h_feasible = h_start.copy()
        
        # Sequential convex programming approach
        # Each iteration linearizes the overlap constraint
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            # Create cvxpy problem
            h_vars = cp.Variable(n_points)
            
            # Objective (minimize C5 via feasibility check)
            constraints = []
            
            # Add bound constraints
            constraints += [h_vars[i] >= 0 for i in range(n_points)]
            constraints += [h_vars[i] <= 1 for i in range(n_points)]
            constraints += [cp.sum(h_vars) == n_points / 2]
            
            # Add overlap constraints
            for k in range(n_points):
                # Linearize overlap integral
                if k == 0:
                    # For k=0, we want h[i]*h[i] = h[i]^2
                    # Using: h[i]*h[i] >= 1 - mid*dx
                    # This is not convex, so we approx. with cross terms
                    pass
                else:
                    # Approximate sum h[i]*h[i+k] with linearization
                    linear_terms = 0
                    for i in range(n_points - k):
                        linear_terms += h_feasible[i] * h_vars[i + k] + h_feasible[i + k] * h_vars[i] - h_feasible[i] * h_feasible[i + k]
                    # The actual constraint: sum(h[i]*h[i+k]) >= 1 - mid*dx
                    # In linearized form: linear_terms >= 1 - mid*dx
                    constraints.append(linear_terms >= 1 - mid * dx)
            
            # Create the problem
            prob = cp.Problem(cp.Minimize(0), constraints)
            
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_feasible = h_vars.value
                    h_feasible = np.array(h_feasible)
                    # Normalize
                    h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                    h_feasible = np.clip(h_feasible, 0, 1)
            except:
                pass
        
        # Evaluate this solution
        c5_val = evaluate_erdos_solution(h_feasible, dx)
        
        if c5_val < best_c5:
            best_h = h_feasible.copy()
            best_c5 = c5_val
        
        # Update binary search bounds
        if c5_val <= mid:
            # Feasible
            high = mid
        else:
            # Infeasible
            low = mid
            
        if abs(high - low) < tolerance:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
