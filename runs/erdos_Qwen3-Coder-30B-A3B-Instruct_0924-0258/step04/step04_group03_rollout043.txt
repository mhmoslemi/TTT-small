import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient for overlap at shift k
        n = len(h)
        grad = np.zeros(n)
        # Manually compute derivative of correlation with respect to h_i
        for i in range(n):
            if i + k < n:
                grad[i] += (1 - h[i+k]) * dx
            if i - k >= 0:
                grad[i] += h[i-k] * dx
        return grad
    
    # Project to feasible set
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Gradient descent with adaptive step size
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    decay_rate = 0.99
    min_alpha = 1e-6
    max_iter = 5000
    
    for iter_count in range(max_iter):
        # Check time budget
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Compute all overlaps and find the max
        overlaps = np.correlate(current_h, 1 - current_h, mode='full')
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k] * dx
        
        # Compute gradient for the most binding constraint
        grad = compute_gradient(current_h, max_k)
        
        # Update
        new_h = current_h - alpha * grad
        new_h = project_h(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            # Increase step size if successful
            alpha = min(1.0, alpha * 1.01)
        else:
            # Decrease step size if failed
            alpha = max(min_alpha, alpha * decay_rate)
            
        # Occasionally refine with a convex problem
        if iter_count % 50 == 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add constraints based on top 10 shifts
                top_k = np.argsort(overlaps)[-10:][::-1]
                for k in top_k:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    # Set high constraint to avoid local minima
                    constraints.append(corr_sum <= (overlaps[k] * dx + 0.01))
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_guess = best_h + np.random.normal(0, 0.02, n_points)
            h_guess = np.clip(h_guess, 0, 1)
            h_guess = h_guess * (n_points/2) / np.sum(h_guess)
            
            try:
                res = minimize(objective, h_guess, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
