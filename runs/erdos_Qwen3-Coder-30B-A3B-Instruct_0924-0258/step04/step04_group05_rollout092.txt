import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    # Initialize population
    population_size = 40
    population = []
    
    # Add perturbed initial solution
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random valid solutions  
    for _ in range(30):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Tabu search memory
    tabu_list = set()
    
    # Genetic algorithm parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_count = 0
    max_stagnation = 50
    tabu_limit = 50
    
    def encode_h(h):
        return tuple(np.round(h * 1000).astype(int))
    
    def decode_h(h_tuple):
        return np.array(h_tuple) / 1000.0
    
    def crossover(parent1, parent2):
        if random.random() > crossover_prob:
            return parent1.copy()
        mask = np.random.randint(0, 2, n_points)
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h):
        if random.random() > mutation_rate:
            return h.copy()
        h_mut = h.copy()
        idx = np.random.randint(0, n_points)
        h_mut[idx] += random.gauss(0, 0.03)
        h_mut[idx] = np.clip(h_mut[idx], 0, 1)
        # Renormalize
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        return h_mut
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Main GA loop
    for gen in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        population_fitness = []
        for h in population:
            # Ensure validity
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            fit = fitness(h)
            population_fitness.append((fit, h))
        
        # Sort by fitness
        population_fitness.sort(key=lambda x: x[0])
        best_in_gen = population_fitness[0][0]
        
        if best_in_gen < best_c5:
            best_c5 = best_in_gen

            stagnation_count = 0

        if gen % 10 == 0 and gen > 0:

                def objective(h_local):
                    return evaluate_erdos_solution(h_local, dx)

                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h, method='SLSQP',

            child = crossover(selected[0], selected[1])
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Adaptive parameters
        if stagnation_count > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]

            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
