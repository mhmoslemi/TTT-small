import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better initial guess with two peaks
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two clusters: one at 0.5, one at 1.5
        cluster_size = 4
        for center in [0.5, 1.5]:
            start = max(0, int((center - cluster_size/2)/dx))
            end = min(n_points, int((center + cluster_size/2)/dx))
            h_start[start:end] = 1.0
        # Scale to meet constraint
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search on the minimum correlation
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    
    # Feasibility checking with gradient descent
    def check_feasibility(h_guess, lambda_target):
        h = h_guess.copy()
        # Project to valid range and sum
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        max_iter = 1000
        lr = 0.1
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute all auto-correlations
            corr = np.correlate(h, 1 - h, mode='full')
            # Extract full correlation, same length as h
            corr_full = corr[n_points-1:]
            
            # Find the minimum correlation (most constraining)
            min_idx = np.argmin(corr_full)
            min_corr = corr_full[min_idx]
            
            if min_corr >= lambda_target:
                return h, True
                
            # Gradient of correlation w.r.t. h (approximate)
            # Since we're looking at correlation at shift min_idx
            # Just use simple finite differences for gradient
            # Both h and (1-h) are being convolved
            # Gradient w.r.t. one element: 
            # we need to measure how much h[pos] contributes to the correlation at shift min_idx
            
            # Forward difference approximation for gradient
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(n_points):
                h_pert = h.copy()
                h_pert[i] += eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                corr_pert = np.correlate(h_pert, 1-h_pert, mode='full')
                corr_pert = corr_pert[n_points-1:]
                grad[i] = (corr_pert[min_idx] - min_corr) / eps
            
            # Update h
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            h = h_new
            
        return h, False
    
    # Binary search over lambda
    while lambda_high - lambda_low > tolerance:
        if time() - start_time > budget_s * 0.95:
            break
        
        lambda_mid = (lambda_low + lambda_high) / 2
        
        # Check feasibility of lambda_mid
        h_feasible, feasible = check_feasibility(best_h, lambda_mid)
        
        if feasible:
            lambda_low = lambda_mid
            best_h = h_feasible
            best_c5 = evaluate_erdos_solution(best_h, dx)
        else:
            lambda_high = lambda_mid
            
        # Small step to make sure we progress
        if lambda_high - lambda_low < 1e-6:
            lambda_high = lambda_low + 1e-6
    
    # Final optimization with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass

    return (list(best_h), best_c5, n_points)
