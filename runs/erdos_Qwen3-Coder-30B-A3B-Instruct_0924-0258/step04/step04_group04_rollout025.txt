import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization with annealing
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    max_iter_coarse = 100
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.4:
            break
            
        h_pert = best_h + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        old_c5 = evaluate_erdos_solution(best_h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            best_h = h_pert
            best_c5 = new_c5
        
        temp *= cool_rate
    
    # Upsample to n_points = 40
    if n_points < 40:
        n_points_new = 40
        h_upsampled = np.interp(np.linspace(0, 2, n_points_new), 
                               np.linspace(0, 2, n_points), 
                               best_h)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_points_new/2) / np.sum(h_upsampled)
        best_h = h_upsampled
        n_points = n_points_new
        dx = 2.0 / n_points
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine optimization with SLSQP
    max_iter_fine = 100
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    for iteration in range(max_iter_fine):
        if time() - start_time > budget_s * 0.8:
            break
            
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 5, 'ftol': 1e-6})
            if res.success:
                refined_h = res.x
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass
    
    # Final refinement with gradient descent
    max_iter_final = 100
    learning_rate = 0.01
    
    for iteration in range(max_iter_final):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient numerically (simple finite differences)
        grad = np.zeros(n_points)
        eps = 1e-4
        base_c5 = evaluate_erdos_solution(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update
        new_h = best_h - learning_rate * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
    
    return (list(best_h), best_c5, n_points)
