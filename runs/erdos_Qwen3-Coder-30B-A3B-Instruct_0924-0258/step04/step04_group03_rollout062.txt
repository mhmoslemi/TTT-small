from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    population_size = 50
    mutation_rate = 0.05
    selection_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())

            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evolutionary loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break

        fitness = []
        for h in population:
            c = evaluate_erdos_solution(h, dx)
            fitness.append(c)

        sorted_indices = np.argsort(fitness)
        elite_count = int(population_size * selection_rate)
        elite = [population[i] for i in sorted_indices[:elite_count]]

        new_population = elite.copy()

        while len(new_population) < population_size:

            new_population.append(child)

        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:

    top_fitness = []

    return (list(best_h), final_c5, n_points)
