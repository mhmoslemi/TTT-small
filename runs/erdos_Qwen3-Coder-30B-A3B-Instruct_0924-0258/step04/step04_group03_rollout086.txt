from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy with gradient refinement
    population_size = 50
    num_generations = 1000
    elite_size = population_size // 5  # Top 20%
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Use initial as seed
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            pert = h_start + np.random.normal(0, 0.05, n_points)
            pert = np.clip(pert, 0, 1)
            pert = pert * (n_points / 2) / np.sum(pert)
            population.append(pert)
    else:
        # Initialize randomly
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Main evolutionary loop
    for gen in range(num_generations):  
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: keep top elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)] 
            
            # Crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            mutation_step = 0.05 * (1 - gen / num_generations) + 0.001
            noise = np.random.normal(0, mutation_step, n_points)
            child = child + noise
            
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Occasionally refine top solutions with gradient
        if gen % 5 == 0:
            # Pick top performers and refine with SLSQP
            for i in range(min(5, len(population))):

                    break

                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:

                            best_h = h_refined

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points / 2.0

        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                best_h = res.x
                best_c5 = objective(best_h)
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
