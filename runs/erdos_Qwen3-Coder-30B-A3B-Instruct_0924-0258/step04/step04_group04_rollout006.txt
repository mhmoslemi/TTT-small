import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        # Create a smooth initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place a bump around center
        center_idx = n_points // 2
        width = n_points // 8
        for i in range(max(0, center_idx - width//2), min(n_points, center_idx + width//2)):
            h_start[i] = 1.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Adaptive gradient descent with line search
    max_iter = 1000
    step_size = 0.01
    patience = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the shift that maximizes overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_shift_idx = np.argmax(corr)
        max_overlap = corr[max_shift_idx] * dx
        
        # Compute gradient for this shift
        # Need to compute d/dh_i of sum h(x) (1 - h(x+k))
        # = sum_{x} h(x) * (-delta(x, i+k)) + (1-h(x)) * delta(x-i, k))
        # = -h(i+k) + (1-h(i)) = (1-h(i)) - h(i+k)
        # More carefully: if k >= 0, then we sum h(x) (1-h(x+k))
        # d/dh_i [h(x) (1 - h(x+k))] = (1-h(x+k)) * delta_{x,i} - h(x) * delta_{x+k,i} 
        # So for fixed i:
        # For x = i: + (1-h(i+k))  
        # For x = i-k: - h(i-k) (if i-k >= 0)
        # For all other x: 0 
        # So derivative wrt h_i is: (1-h(i+k)) - (i-k >= 0)*h(i-k)
        # Special case: if k=0: derivative is (1-h(i)) - h(i) = 1 - 2*h(i)
        # So if k >= 0, we compute:
        # grad[i] = (1-h[i+k]) - (i-k >= 0)*h[i-k]  if k != 0
        # grad[i] = 1 - 2*h[i]                     if k == 0
        
        # Let's do this properly using array manipulation:
        gradient = np.zeros(n_points)
        if max_shift_idx >= n_points:
            # We have a positive shift k = max_shift_idx - n_points
            k = max_shift_idx - n_points
            if k > 0:
                # we look at values where h(x) (1 - h(x+k)) 
                # The gradient w.r.to h(i) is (1 - h(i+k)) - h(i-k) if k > 0, 
                # and we also have special case where i-k is negative.
                # Indices to consider for gradient of i-th element:
                # -         h(i+k) (would be at index i+k)
                # - h(i-k) (if i-k >= 0)
                for i in range(n_points):
                    if i + k < n_points:
                        gradient[i] += -best_h[i + k]
                    if i - k >= 0:
                        gradient[i] += -best_h[i - k]
                    # Occasionally we add 1 when making up for our double counting
                    gradient[i] += 1
            else:  # k == 0
                # Standard case: derivative of h(x)(1 - h(x+k)) = h(x) (1 - h(x))
                # which has derivative wrt h(x) as (1 - 2*h(x))  
                gradient = 1 - 2 * best_h
        else:
            # k is negative (max_shift_idx < n_points)
            k = n_points - max_shift_idx  # Makes it positive
            # We compute for i:
            # Derivative = (1-h(i-k)) - h(i+k)  
            for i in range(n_points):
                if i - k >= 0:
                    gradient[i] += -best_h[i - k]
                if i + k < n_points:
                    gradient[i] += -best_h[i + k]
                gradient[i] += 1

        # Clip/normalize the gradient for numerical stability (not really needed, but helps)
        gradient = np.clip(gradient, -1, 1)
        
        # Line search: find step size that reduces overlap
        trial_step = step_size * gradient
        trial_h = best_h - trial_step
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = trial_h * (n_points/2) / np.sum(trial_h)
        
        trial_c5 = evaluate_erdos_solution(trial_h, dx)
        iteration_improvement = max_overlap - trial_c5
        
        # If step improves overlap, accept it for next iteration
        if iteration_improvement > 1e-8:
            best_h = trial_h
            best_c5 = trial_c5
            if iteration_improvement > 1e-4:
                step_size *= 1.1  # Adaptive: if improvements are big, do larger steps
            if best_c5 < 0.38085:
                break
        else:
            step_size *= 0.9  # Reduce step size when no improvement
        
        # Early termination if no improvement in a while
        if iteration > patience and iteration_improvement < 1e-6:
            break
        
    # Final validation and refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            from scipy.optimize import minimize
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
