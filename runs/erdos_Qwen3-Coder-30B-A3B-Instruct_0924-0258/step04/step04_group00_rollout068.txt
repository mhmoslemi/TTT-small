def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    def binary_relaxation(h_continuous, target_sum):
        """Convert continuous solution to binary relaxation"""

        h_binary = (h_continuous > 0.5).astype(float)

        current_sum = np.sum(h_binary)
        if current_sum > 0:
            scale = target_sum / current_sum
            h_binary = np.clip(h_binary * scale, 0, 1)
            # Adjust to exact sum
            diff = target_sum - np.sum(h_binary)
            if diff != 0:
                # Add to smallest values
                indices = np.argsort(h_binary)
                for i in range(int(np.ceil(abs(diff)))):
                    if diff > 0 and h_binary[indices[i]] < 1:
                        h_binary[indices[i]] += 1
                    elif diff < 0 and h_binary[indices[-(i+1)]] > 0:
                        h_binary[indices[-(i+1)]] -= 1
