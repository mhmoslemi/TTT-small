import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with Adam-like updates
    # Parameterize as linear function with bias and scale to control sum
    theta = np.random.randn(n_points) * 0.1
    beta = 0.0
    scale = 1.0
    
    def get_h(theta, beta, scale):
        h_unscaled = np.maximum(0, theta + beta)
        h_scaled = np.minimum(1, scale * h_unscaled)
        return h_scaled
    
    # Gradient computation via finite differences for speed
    def compute_gradients(theta, beta, scale):
        h = get_h(theta, beta, scale)
        target_val = evaluate_erdos_solution(h, dx)
        eps = 1e-4
        
        grad_theta = np.zeros_like(theta)
        grad_beta = 0.0
        grad_scale = 0.0
        
        # Compute gradients with respect to theta
        for i in range(n_points):
            theta_plus = theta.copy()
            theta_plus[i] += eps
            h_plus = get_h(theta_plus, beta, scale)
            val_plus = evaluate_erdos_solution(h_plus, dx)
            
            theta_minus = theta.copy()
            theta_minus[i] -= eps
            h_minus = get_h(theta_minus, beta, scale)
            val_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad_theta[i] = (val_plus - val_minus) / (2 * eps)
        
        # Compute gradient with respect to beta
        beta_plus = beta + eps
        h_plus = get_h(theta, beta_plus, scale)
        val_plus = evaluate_erdos_solution(h_plus, dx)
        
        beta_minus = beta - eps
        h_minus = get_h(theta, beta_minus, scale)
        val_minus = evaluate_erdos_solution(h_minus, dx)
        
        grad_beta = (val_plus - val_minus) / (2 * eps)
        
        # Compute gradient with respect to scale
        scale_plus = scale + eps
        h_plus = get_h(theta, beta, scale_plus)
        val_plus = evaluate_erdos_solution(h_plus, dx)
        
        scale_minus = scale - eps
        h_minus = get_h(theta, beta, scale_minus)
        val_minus = evaluate_erdos_solution(h_minus, dx)
        
        grad_scale = (val_plus - val_minus) / (2 * eps)
        
        return grad_theta, grad_beta, grad_scale
    
    # Optimization loop
    lr = 5e-3
    max_iter = int(1e6)
    patience = 0
    patience_limit = 200
    recent_improvements = []
    
    # Optimization target for constraints
    def target_sum_constraint(h):
        return np.sum(h) - n_points / 2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current solution
        h_current = get_h(theta, beta, scale)
        current_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            recent_improvements = [current_c5]
        else:
            recent_improvements.append(current_c5)
            if len(recent_improvements) > 50:
                recent_improvements.pop(0)

        if len(recent_improvements) >= 50:
            if (np.mean(recent_improvements[-10:]) >=

                patience += 1
                if patience >= patience_limit:
                    break
            else:
                patience = 0
        
        # Compute gradients
        try:
            grad_theta, grad_beta, grad_scale = compute_gradients(theta, beta, scale)
            
            # Update parameters
            theta -= lr * grad_theta
            beta -= lr * grad_beta
            
            # Ensure positive scale via safe update
            if grad_scale < 0:
                scale += lr * grad_scale

                scale *= 0.999  # Small decay when increasing scale is not beneficial

            scale = max(0.1, min(100, scale))

        h_proj = get_h(theta, beta, scale)

            h_proj = h_proj * mu
        
        # Reset parameters to match projected solution
        h_current = h_proj
        current_sum = np.sum(h_current)
        scale = current_sum / np.sum(np.clip(np.maximum(0, theta + beta), 0, 1))
        
        # Add some noise occasionally for escape
        if iteration % 100 == 0 and iteration > 0:
            theta += np.random.normal(0, 0.01, n_points)
            beta += np.random.randn() * 0.01
            scale += np.random.randn() * 0.01

    def objective(h_vec):
        # Reshape into proper h
        h = h_vec.reshape(-1)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_func(h_vec):
        h = h_vec.reshape(-1)
        return np.sum(h) - n_points / 2
    
    # Unpack optimized parameters to h
    h_last = get_h(theta, beta, scale)
    h_last = np.clip(h_last, 0, 1)
    
    # Ensure sum equals exactly n_points/2
    h_last = h_last * (n_points / 2) / np.sum(h_last)
    
    # Refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            # Random restarts to search for better local minima
            for _ in range(3):
                h_start = h_last + np.random.normal(0, 0.01, n_points)
                h_start = np.clip(h_start, 0, 1)
                h_start = h_start * (n_points / 2) / np.sum(h_start)
                
                res = minimize(objective, h_start, method='SLSQP',
                               bounds=[(0, 1) for _ in range(n_points)],
                               constraints={'type': 'eq', 'fun': constraint_func},
                               options={'maxiter': 50, 'ftol': 1e-10})
                if res.success:
                    h_candidate = res.x.reshape(-1)
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate
                        best_c5 = c5_candidate
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
