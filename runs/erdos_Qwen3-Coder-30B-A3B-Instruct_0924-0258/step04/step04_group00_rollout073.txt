import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with adaptive step size
    h = best_h.copy()
    alpha = 0.01
    stagnation_count = 0
    max_stagnation = 50
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def compute_gradient(h):
        # Compute overlapped values for all shifts
        n = len(h)
        grad = np.zeros(n)
        for k in range(1, n):
            if k <= n - 1:
                # Forward shift
                corr = h[:n-k] * (1 - h[k:])
                grad[:n-k] += corr
                grad[k:] -= corr
            else:
                # Backward shift
                corr = h[k-n+1:] * (1 - h[:n-(k-n+1)])
                grad[k-n+1:] += corr
                grad[:n-(k-n+1)] -= corr
        return grad
    
    iterations = 0
    max_iterations = int(1000 * (budget_s / 600))
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iterations:
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update with gradient step
        h_new = h - alpha * grad
        
        # Project
        h_new = project(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject step
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            stagnation_count = 0
            alpha = min(alpha * 1.2, 1.0)  # Increase step size
        else:
            h = project(h)  # Project to feasible set
            stagnation_count += 1
            alpha *= 0.5  # Decrease step size
            
        if stagnation_count >= max_stagnation:
            break
            
        iterations += 1
    
    # Final refinement with local averaging
    final_h = best_h.copy()
    # Apply smoothing
    smoothed = np.zeros_like(final_h)
    for i in range(len(final_h)):
        neighbors = []
        for j in range(max(0, i-2), min(len(final_h), i+3)):
            neighbors.append(final_h[j])
        smoothed[i] = np.mean(neighbors)
    
    final_h = project(smoothed)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
