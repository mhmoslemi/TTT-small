import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        h_start = np.zeros(n_points)

        # Place three clusters: 1st at 0.4, 2nd at 1.0, 3rd at 1.6
        cluster_centers = [0.4, 1.0, 1.6]
        cluster_size = 4
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    low, high = 0.3, 0.38085
    precision = 1e-6
    max_iterations = 50
    
    for _ in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2
        
        # Use sequential convex programming to check feasibility
        h_current = h_start.copy()
        max_scp_iter = 5
        
        # SCP iterations to satisfy overlap constraints
        for scp_iter in range(max_scp_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Formulate convex constraints for current h
            h_var = cp.Variable(n_points)
            
            # Define constraints for each shift k
            constraints = []
            for k in range(n_points):
                # Overlap constraint for shift k: sum(h[i]*h[i+k]) >= 1 - mid*dx
                if k == 0:
                    # Direct overlap
                    overlap_expr = cp.sum_squares(h_var)
                else:
                    # Shifted overlap
                    overlap_expr = 0
                    for i in range(n_points - k):
                        overlap_expr += h_var[i] * h_var[i+k]
                
                # This is an approximation constraint
                # Linearization around current h_current:
                h_current_shift = np.roll(h_current, k)
                if k == 0:
                    linearized = cp.sum_squares(h_var)
                else:
                    linearized = 0
                    for i in range(n_points - k):
                        linearized += 2 * h_current[i] * h_var[i+k] - h_current[i] * h_current[i+k]
                
                # Alternative: use sum of products approximation
                if k > 0:  # Avoid double counting
                    circ_overlap = cp.sum(cp.multiply(h_var, cp.roll(h_var, k)))
                    constraints.append(circ_overlap >= 1 - mid * dx)
                
            # Constraints: bounds, sum equal to n_points/2
            constraints.extend([
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ])
            
            try:
                # Solve this problem
                objective = cp.Minimize(0)
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.MOSEK, verbose=False)

                    h_current = h_var.value

                    h_current = h_current * (n_points/2) / np.sum(h_current)

                    # Fallback to simple approach
                    break

                break

        current_c5 = evaluate_erdos_solution(h_current, dx)

            best_h = h_current.copy()

        if current_c5 <= mid * dx:
            high = mid
        else:
            low = mid
            
        if abs(high - low) < precision:
            break
            
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
