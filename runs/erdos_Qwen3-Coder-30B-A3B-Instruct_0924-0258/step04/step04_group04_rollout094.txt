import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, multiply
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Initialize dual variables
    dual_vars = np.ones(2 * n_points - 1)
    
    # Main loop with alternating optimization
    max_iter = 50
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        C_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
        
        # Update dual variables
        max_C = np.max(C_vals)
        dual_vars = np.maximum(dual_vars * (C_vals / (dual_vars + 1e-8)), 1e-6)
        
        # Solve convex relaxation
        h = Variable(n_points)
        objective = cp.Minimize(cp_sum(multiply(dual_vars, C_vals)))
        
        constraints = [
            cp_sum(h) == n_points/2,
            h >= 0,
            h <= 1
        ]
        
        prob = Problem(objective, constraints)
        try:
            prob.solve(warm_start=True, solver=cp.SCS, max_iters=1000)
            new_h = np.array(h.value).flatten()
            # Project to feasible set
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            
            # Check if new solution improves
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        except:
            # If CVXPY fails, keep current best
            pass
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
