import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Main annealing loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate new candidate by perturbing two random points
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Adjust both points while maintaining constraint
        delta = np.random.uniform(-0.1, 0.1)
        
        # Compute how much we can change
        delta_i = delta
        delta_j = -delta_i
        
        # Ensure bounds for both points
        if current_h[i] + delta_i > 1:
            delta_i = 1 - current_h[i]
            delta_j = -delta_i
        elif current_h[i] + delta_i < 0:
            delta_i = -current_h[i]
            delta_j = -delta_i
            
        if current_h[j] + delta_j > 1:
            delta_j = 1 - current_h[j]
            delta_i = -delta_j
        elif current_h[j] + delta_j < 0:
            delta_j = -current_h[j]
            delta_i = -delta_j
            
        # Apply changes
        h_new = current_h.copy()
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Renormalize to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down
        T *= alpha
    
    return (list(best_h), best_c5, n_points)
