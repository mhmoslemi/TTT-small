import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        # Structured initialization with 3 clusters
        cluster_centers = [0.3, 1.0, 1.7]
        cluster_size = 4
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to sum to n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for C5
    low = 0.3
    high = 0.38085
    precision = 1e-6
    max_iterations = 50
    iteration = 0
    
    # Helper to check feasibility given C
    def is_feasible(h_guess, C):
        # Simple constrained optimization - 5 iterations of refinement
        for _ in range(5):
            # Too expensive to do full CVX - reuse direct optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_guess, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_guess = res.x
                    if obj(h_guess) < C:
                        return True
                else:
                    # Restore h_guess and try local optimization
                    h_guess = np.clip(h_guess, 0, 1)
                    h_guess = h_guess * (n_points/2) / np.sum(h_guess)

        h_guess = np.clip(h_guess, 0, 1)
        h_guess = h_guess * (n_points/2) / np.sum(h_guess)
        return evaluate_erdos_solution(h_guess, dx) < C
    
    # Start with best known solution
    h_current = h_start.copy()
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        mid = (low + high) / 2
        
        # Check feasibility using refinement search
        if is_feasible(h_current, mid):
            high = mid
            if mid < best_c5:
                best_c5 = mid
                best_h = h_current.copy()
        else:
            low = mid
            
        iteration += 1
    
    # Final optimization to improve best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}

            best_c5 = obj(best_h)
    except:
        pass
    
    # Ensuring final constraint satisfaction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
