import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Warm-start approach with binary search
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search on the minimum overlap value
    low = 0.0
    high = 1.0
    tol = 1e-5
    max_iter = 100
    
    def get_ac(h, dx):
        ac = np.correlate(h, 1 - h, mode='full')
        return ac * dx
    
    def project(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = h * target_sum / current_sum
        return h
    
    def check_feasibility(h, target_ac, dx):
        ac = get_ac(h, dx)
        min_ac = np.min(ac)
        return min_ac >= target_ac
    
    # Gradient-based refinement
    def refine_h(h, target_ac, dx, max_iter=200):
        current_h = h.copy()
        learning_rate = 0.1
        
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            ac = get_ac(current_h, dx)
            min_idx = np.argmin(ac)
            min_ac = ac[min_idx]
            
            if min_ac >= target_ac:
                return current_h
            
            # Compute gradient for the min_AC position
            # This is approximate but works reasonably well
            grad = np.zeros_like(current_h)
            # Even though it's not exact, this is a reasonable descent direction
            for i in range(len(current_h)):
                # Gradient for h[i] WRT min_ac...
                # The simple heuristic is to push h[i] up where needed
                # Calculate shift index: idx = i + min_idx - (len(h)-1)
                # We'll approximate with a simpler gradient
                if i == min_idx:
                    grad[i] = 1
                else:
                    # For the other terms, simple consideration: we want the minimum AC to rise
                    grad[i] = 0
                    
            # Slightly different approach: just increase min AC term
            current_h = current_h + learning_rate * grad
            current_h = project(current_h, n_points/2)
            
            # Early termination attempt
            ac = get_ac(current_h, dx)
            if np.min(ac) >= target_ac:
                return current_h                
        
        return current_h
    
    iter_count = 0
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        iter_count += 1
        mid = (low + high) / 2
        
        # Try to find feasible solution with this minimum AC value
        # Start with the best known solution
        try:
            test_h = best_h.copy()
            test_h = refine_h(test_h, mid, dx, max_iter=100)
            if check_feasibility(test_h, mid, dx):
                low = mid
                if evaluate_erdos_solution(test_h, dx) < best_c5:
                    best_h = test_h
                    best_c5 = evaluate_erdos_solution(test_h, dx)
            else:
                high = mid
        except:
            high = mid

    # Final refinement of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
