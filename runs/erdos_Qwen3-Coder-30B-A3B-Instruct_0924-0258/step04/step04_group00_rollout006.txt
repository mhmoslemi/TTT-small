from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def tournament_select(population, fitness, tournament_size=2):
    """Tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Single-point crossover"""
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Normalize
    child1 = child1 * (len(parent1)/2) / np.sum(child1)
    child2 = child2 * (len(parent2)/2) / np.sum(child2)
    
    return child1, child2

def mutate(individual, mutation_rate):
    """Gaussian mutation"""
    indices = np.random.choice(len(individual), 
                              size=int(len(individual) * mutation_rate), 
                              replace=False)
    for idx in indices:
        individual[idx] += np.random.normal(0, 0.01)
    # Clip and re-normalize
    individual = np.clip(individual, 0, 1)
    individual = individual * (len(individual)/2) / np.sum(individual)
