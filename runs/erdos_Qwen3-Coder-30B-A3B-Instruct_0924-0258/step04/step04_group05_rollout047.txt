import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Genetic algorithm parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    max_stagnation = 50
    
    # Initialize population with variations of initial solution
    population = []
    for _ in range(10):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    # Add random feasible points
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fill remaining population
    while len(population) < population_size:
        h_new = np.random.rand(n_points)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        population.append(h_new)
    
    # Tabu search memory
    tabu_list = set()
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    # Fitness function with constraint handling
    def fitness(h):
        return evaluate_erdos_solution(h, 2.0/n_points)
        
    # Initialize population fitness
    population_fitness = []
    for h in population:
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        f = fitness(h)
        population_fitness.append(f)
        # Store in tabu list
        h_tuple = tuple(np.round(h, decimals=6))
        tabu_list.add(h_tuple)
        
        if f < best_c5:
            best_c5 = f
            best_h = h.copy()
    
    # Genetic algorithm iterations
    generation = 0
    while time() - start_time < budget_s * 0.95 and stagnation_count < max_stagnation:
        generation += 1
        
        # Sort population by fitness
        sorted_idx = np.argsort(population_fitness)
        sorted_pop = [population[i] for i in sorted_idx]
        sorted_fit = [population_fitness[i] for i in sorted_idx]
        
        # Select top performers for breeding
        elite = sorted_pop[:population_size//2]
        elite_fitness = sorted_fit[:population_size//2]
        
        # Create new population
        new_pop = elite[:]
        new_pop_fitness = elite_fitness[:]
        
        # Generate offspring
        while len(new_pop) < population_size:
            if random.random() < crossover_prob and len(elite) >= 2:
                # Select two parents
                p1, p2 = random.sample(elite, 2)
                
                # Crossover (uniform)
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, p1, p2)
                
                # Mutation
                if random.random() < mutation_rate:
                    mutate_idx = random.randint(0, n_points-1)
                    child[mutate_idx] = random.random()
                    
                # Repair constraints
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
                # Check tabu list
                child_tuple = tuple(np.round(child, decimals=6))
                if child_tuple in tabu_list:
                    # Random perturb if in tabu
                    child = child + np.random.normal(0, 0.01, n_points)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points/2) / np.sum(child)
                    
                # Add to population (ensure unique)
                child_tuple = tuple(np.round(child, decimals=6))
                if child_tuple not in tabu_list:
                    new_pop.append(child)
                    new_pop_fitness.append(fitness(child))
                    tabu_list.add(child_tuple)
                    if len(tabu_list) > tabu_limit:
                        tabu_list.pop()
                    
                    if len(new_pop) >= population_size:
                        break
        
        # Update population
        population = new_pop
        population_fitness = new_pop_fitness
        
        # Update best solution
        for i, h in enumerate(population):
            if population_fitness[i] < best_c5:
                best_c5 = population_fitness[i]
                best_h = h.copy()
                stagnation_count = 0
            elif population_fitness[i] < best_c5 + 1e-6 and generation % 10 == 0:  # Smooth handling
                stagnation_count += 1

        # Adaptive parameter updates every 10 generations
        if generation % 10 == 0:
            if stagnation_count > 0:
                mutation_rate = min(0.5, mutation_rate + 0.1)
                crossover_prob = max(0.3, crossover_prob - 0.1)
                tabu_limit = max(10, tabu_limit // 2)
            else:
                mutation_rate = max(0.05, mutation_rate - 0.05)
                crossover_prob = min(0.9, crossover_prob + 0.05)
                tabu_limit = min(200, tabu_limit * 2)
    
    # Local refinement of best solution using SLSQP
    if time() - start_time < budget_s * 0.98:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0/n_points)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                refined_h = res.x
                refined_c5 = objective(refined_h)
                if refined_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = refined_c5
        except:
            pass
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points)
    
    return (list(final_h), float(final_c5), n_points)
