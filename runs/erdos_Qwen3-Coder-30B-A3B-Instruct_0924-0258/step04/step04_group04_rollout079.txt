import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization (n_points = 20)
    max_iter_coarse = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.4:
            break
        h_pert = best_h + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        old_c5 = evaluate_erdos_solution(best_h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            best_h = h_pert
            best_c5 = new_c5
        temp *= cool_rate
    
    # Upsample to n_points = 40
    n_points = 40
    dx = 2.0 / n_points
    h_upsampled = np.zeros(n_points)
    for i in range(n_points//2):
        h_upsampled[2*i] = best_h[i]
        h_upsampled[2*i+1] = best_h[i]
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    best_h = h_upsampled.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine optimization (n_points = 40)
    max_iter_fine = 100
    perturb_mag_fine = 0.02
    for iteration in range(max_iter_fine):
        if time() - start_time > budget_s * 0.8:
            break
        h_pert = best_h + np.random.normal(0, perturb_mag_fine, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        old_c5 = evaluate_erdos_solution(best_h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        if new_c5 < old_c5:
            best_h = h_pert
            best_c5 = new_c5
    
    # Final refinement with SLSQP
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final check to avoid invalid solutions
    if not np.all((best_h >= 0) & (best_h <= 1)):
        best_h = np.clip(best_h, 0, 1)
        
    if np.abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
