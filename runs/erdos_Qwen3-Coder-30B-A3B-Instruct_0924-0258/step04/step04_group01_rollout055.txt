import numpy as np
from cvxpy import *
import math
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create an initial construction
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two spikes near 0.5 and 1.5
        spike_width = 0.2
        spike1_center = 0.5
        spike2_center = 1.5
        for i in range(n_points):
            x = i * dx
            if abs(x - spike1_center) < spike_width/2:
                h_start[i] = 1.0
            elif abs(x - spike2_center) < spike_width/2:
                h_start[i] = 1.0
        
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = float('inf')
    
    # Binary search bounds
    low = 0.3
    high = 0.38085
    precision_threshold = 1e-5
    max_iterations = 60
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def solve_feasibility(h, C):
        # Variables
        h_vars = Variable(n_points)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Bound constraints
        constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
        constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
        
        # Overlap constraints
        # For each lag k, we want sum h[i] * (1 - h[i + k]) <= C / dx
        # Let's compute all valid shifts
        for k in range(2 * n_points - 1):
            # Create constraint for lag k
            # This is a sum over valid indices
            expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    expr += h_vars[i] * (1 - h_vars[i + k])
                else:
                    break
            
            if expr != 0:
                constraints.append(expr <= C / dx)
        
        # Objective is to solve feasibility
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=GLPK, verbose=False)
            return prob.status == "optimal"
        except:
            return False
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2
        
        # Try to find a feasible solution for this mid value
        if solve_feasibility(h_start, mid):
            # This is feasible, so we can try to improve
            high = mid
            h_candidate = h_start.copy()
            c5_candidate = evaluate_c5(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        else:
            # This is not feasible, so we need to increase the bound
            low = mid
            
    # Final small refinement using gradient-based local optimization if allowed
    if best_c5 == float('inf'):
        # Fallback to simpler approach
        # Use an even simpler construction
        h_simple = np.ones(n_points) * 0.5
        scale = (n_points/2) / np.sum(h_simple)
        h_simple *= scale
        best_h = h_simple
        best_c5 = evaluate_c5(h_simple)
    
    # Ensure final feasible solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
