import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Start with initial solution
    def optimize_from_h0(h0):
        h = h0.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        def objective(h_vec):
            h_local = h_vec.copy()
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            return evaluate_erdos_solution(h_local, dx)

            h_local = h_vec.copy()
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            # This is a placeholder gradient - in practice would compute more carefully
            return np.random.normal(0, 0.01, n_points)
            
        # Use L-BFGS-B which is more robust for bound-constrained problems
        try:
            res = minimize(objective, h_local, method='L-BFGS-B',

                           options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-6})
            if res.success:
                h_result = res.x

        h_pert = h_start + np.random.normal(0, 0.05, n_points)
