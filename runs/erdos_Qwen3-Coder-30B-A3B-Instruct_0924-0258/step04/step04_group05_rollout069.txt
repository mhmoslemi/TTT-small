import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based search with tabu and local refinement
    population_size = 40
    max_generations = 300
    tabu_limit = 50
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_limit = 50
    tabu_memory = set()
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Add variations of initial solution
    for i in range(10):
        h_var = h_start + np.random.normal(0, 0.02, n_points)
        h_var = np.clip(h_var, 0, 1)
        h_var = h_var * (n_points/2) / np.sum(h_var)
        population.append(h_var)
        
    # Add random solutions
    for i in range(30):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
        
    # Evaluate initial population
    for h in population:
        fitness = evaluate_erdos_solution(h, dx)
        fitnesses.append(fitness)
        
    # Track best
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_h = population[best_idx].copy()
        best_c5 = fitnesses[best_idx]
    
    # Main genetic search loop
    stagnation_count = 0
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitnesses[i] for i in sorted_indices]
        
        # Check for stagnation and adjust parameters
        if generation > 0 and sorted_fit[0] >= best_c5:
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                # Increase mutation and decrease crossover
                mutation_rate = min(0.5, mutation_rate + 0.1)
                crossover_prob = max(0.3, crossover_prob - 0.1)
                tabu_limit = max(10, tabu_limit // 2)
                stagnation_count = 0
        else:
            stagnation_count = 0
            # Decrease mutation and increase crossover
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
            
        # Select top 20% for breeding
        elite_size = population_size // 5
        elite = sorted_pop[:elite_size]
        
        # Create next generation
        new_population = elite.copy()
        new_fitnesses = sorted_fit[:elite_size]
        
        # Crossover and mutation for the rest
        while len(new_population) < population_size:
            if random.random() < crossover_prob:
                # Crossover two parents
                parent1 = random.choice(elite)
                parent2 = random.choice(elite)
                
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                # Select one parent and mutate
                parent = random.choice(elite)
                child = parent.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                mut_point = random.randint(0, n_points - 1)
                child[mut_point] = random.random()
                
            # Repair to constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            # Tabu check
            child_tuple = tuple(child.round(6))
            if child_tuple in tabu_memory:
                continue
                
            tabu_memory.add(child_tuple)
            if len(tabu_memory) > tabu_limit * 2:
                # Remove oldest entries
                tabu_memory.pop()
                
            new_population.append(child)
            new_fitnesses.append(evaluate_erdos_solution(child, dx))
            
        # Replace population
        population = new_population
        fitnesses = new_fitnesses
        
        # Update best
        best_new_idx = np.argmin(new_fitnesses)
        if new_fitnesses[best_new_idx] < best_c5:
            best_c5 = new_fitnesses[best_new_idx]
            best_h = new_population[best_new_idx].copy()
            
        # Local refinement of top solutions every 10 generations
        if generation % 10 == 0:
            for i in range(min(5, len(new_population))):
                h_candidate = new_population[i]
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_candidate, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < new_fitnesses[i]:
                            new_population[i] = h_refined
                            new_fitnesses[i] = c5_refined
                            if c5_refined < best_c5:
                                best_c5 = c5_refined
                                best_h = h_refined.copy()
                except:
                    pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Very final optimization of the best solution
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
