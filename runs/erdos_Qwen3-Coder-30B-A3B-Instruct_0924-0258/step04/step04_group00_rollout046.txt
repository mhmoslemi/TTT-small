import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with adaptive step size
    max_iter = 10000
    alpha = 0.01
    patience = 0
    max_patience = 50
    
    # Gradient computation helper
    def compute_gradient(h, c5_current):
        # Create correlation matrix approach to compute gradient efficiently
        grad = np.zeros_like(h)
        corr = np.correlate(h, 1 - h, mode='full')
        shift_indices = np.arange(-len(h) + 1, len(h))
        
        # For each element, compute influence on overlaps
        for i in range(len(h)):
            for k in range(len(shift_indices)):
                shift = shift_indices[k]
                if 0 <= i + shift < len(h) and shift != 0:
                    # Gradient contribution from this shift
                    if shift > 0:
                        grad[i] += (1 - h[i + shift])
                    else:
                        grad[i] += (1 - h[i - shift])
        
        return grad

    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current gradient
        grad = compute_gradient(best_h, best_c5)
        
        # Gradient descent step
        h_new = best_h - alpha * grad
        
        # Project onto [0,1] box constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Adaptive step size
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            alpha *= 1.2  # Increase step size
            patience = 0
        else:
            alpha *= 0.5  # Decrease step size
            patience += 1
            # Restore past solution if no improvement
            h_new = best_h
            c5_new = best_c5
            
        # Early stopping if no improvement
        if patience > max_patience:
            break
    
    # Final check of feasibility
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points / 2) < 1e-6
    
    # Additional smoothing
    h_smooth = np.copy(best_h)
    for _ in range(5):
        h_smooth = np.convolve(h_smooth, [0.25, 0.5, 0.25], mode='same')
        h_smooth = np.clip(h_smooth, 0, 1)
        h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
    
    c5_smooth = evaluate_erdos_solution(h_smooth, dx)
    if c5_smooth < best_c5:
        best_h = h_smooth
        best_c5 = c5_smooth
    
    return (list(best_h), best_c5, n_points)
