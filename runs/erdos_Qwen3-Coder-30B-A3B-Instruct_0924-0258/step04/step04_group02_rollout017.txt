import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Number of iterations to allow for optimization
    max_iter = 10000
    iter_count = 0
    
    # Gradient descent with adaptive learning
    lr = 1e-3
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    # Adam variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # To prevent oscillation, add small random perturbations occasionally
    perturb_freq = 100
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Current h values (copy to avoid modifications)
        h_current = best_h.copy()
        
        # Reconstruct the function in a way that facilitate gradient computation
        # The gradient will be estimated numerically for the objective
        def objective(h_vec):
            return evaluate_erdos_solution(h_vec, dx)
        
        # Get gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h_current)
        
        # Estimate gradient numerically, loop over each point
        for i in range(n_points):
            h_plus = h_current.copy()
            h_minus = h_current.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            # Re-normalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * epsilon)
        
        # Adam update for h
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        
        # Update the weights
        h_update = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h_current - h_update
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new objective
        c5_new = objective(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Occasionally add noise to escape local minima

            noise = np.random.normal(0, 0.005, n_points)

            return np.sum(h) - n_points / 2.0

                if c5_refined < best_c5:

                    best_c5 = c5_refined
