import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 100
    num_generations = 1000
    mutation_rate = 0.1
    tournament_size = 2
    elite_size = 5
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def create_population():
        population = []
        if 'initial_h_values' in globals():
            population.append(h_start.copy())
        for _ in range(population_size - len(population)):
            population.append(create_individual())
        return population
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def select_tournament(population, fitnesses):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        return population[winner_index].copy()
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Repair and normalize
        child1 = np.clip(child1, 0, 1)
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = np.clip(child2, 0, 1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        indices = np.random.choice(n_points, int(n_points * mutation_rate), replace=False)
        for idx in indices:
            mutated[idx] += np.random.normal(0, 0.02)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main genetic algorithm loop
    population = create_population()
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [fitness(ind) for ind in population]
        
        # Find best individual in current population
        best_in_generation = min(fitnesses)
        if best_in_generation < best_c5:
            best_c5 = best_in_generation
            best_h = population[np.argmin(fitnesses)].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = select_tournament(population, fitnesses)
            parent2 = select_tournament(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final refinement
    try:
        final_h = best_h.copy()
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
