import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    swarm = np.random.rand(swarm_size, n_points) * 0.5
    for i in range(swarm_size):
        swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
    
    velocities = np.zeros_like(swarm)
    personal_best = swarm.copy()
    personal_best_values = np.array([evaluate_erdos_solution(p, dx) for p in personal_best])
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Apply bounds and constraint
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate and update personal best
            current_value = evaluate_erdos_solution(swarm[i], dx)
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
                
                # Update global best if needed
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
    
    # Store the best from PSO
    best_h = global_best.copy()
    best_c5 = global_best_value
    
    # Convex optimization refinement
    try:
        h = cp.Variable(n_points)
        # Constraint that sum equals n/2
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        # Estimate a reasonable upper bound for C5
        upper_bound = best_c5 * 1.1
        
        # Add a constraint for the overlap that is less than or equal to estimated upper bound
        for k in range(n_points):
            if k < n_points:
                # We compute correlation manually to make constraints
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            
            # Set the constraint that each correlation should not exceed upper_bound
            constraints.append(corr_sum <= upper_bound * dx * n_points)
        
        # Minimize a dummy objective to satisfy cvxpy interface
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    except Exception:
        pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
