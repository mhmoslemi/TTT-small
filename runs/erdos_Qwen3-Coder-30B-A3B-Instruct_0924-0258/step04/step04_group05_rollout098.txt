```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    pop_size = 40
    tabu_limit = 50
    max_stagnation = 50
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Initialize population with diverse solutions
    population = []
    fitness = []
    
    # Start with initial solution and perturbations
    for i in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
        f = evaluate_erdos_solution(h_pert, dx)
        fitness.append(f)
    
    # Add random valid solutions
    for i in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
        f = evaluate_erdos_solution(h_rand, dx)
        fitness.append(f)
    
    # Tabu list to store recently visited solutions
    tabu_list = []
    
    # Track best and stagnation
    best_fitness = min(fitness)
    stagnation_count = 0
    
    # Main GA loop
    for gen in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best if improvement
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Check for early termination
        if best_fitness < 0.3807:
            break
            
        # Check for stagnation and adapt parameters
        if stagnation_count > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
            
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        new_fitness = [fitness[0]]
        
        # Select parents and create offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = random.randint(0, pop_size//2 - 1)
            parent2_idx = random.randint(0, pop_size//2 - 1)
            
            # Crossover with probability
            if random.random() < crossover_prob:
                # Mask crossover
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, population[parent1_idx], population[parent2_idx])
            else:
                child = population[parent1_idx].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, size=max(1, int(n_points * 0.1)), replace=False)
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, 0.03)
                    
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            # Avoid tabu solutions
            if len(tabu_list) > 0:
                # Check if child is too similar to tabu solutions
                similar = False
                for tabu_sol in tabu_list:
                    if np.allclose(child, tabu_sol, atol=1e-3):
                        similar = True
                        break
                if similar:
                    # Add random perturbation
                    child = child + np.random.normal(0, 0.01, n_points)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
            
            # Add to tabu list
            tabu_list.append(child)
            if len(tabu_list) > tabu_limit:
                tabu_list.pop(0)
                
            new_population.append(child)
            f = evaluate_erdos_solution(child, dx)
            new_fitness.append(f)
            
        population = new_population
        fitness = new_fitness
    
    # Final gradient-based refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Refine the best solution
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
```
