import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier-based optimization
    # Represent h(x) = sigmoid(a0 + sum_k [ak*cos(k*pi*x) + bk*sin(k*pi*x)])
    # We'll use a subset of frequencies for smoother function
    max_freq = 8
    num_coeffs = 1 + 2 * max_freq  # a0 + ak + bk for k=1..max_freq
    
    # Initial coefficients - if we have a good start, use it
    if 'initial_h_values' in globals():
        # Try to fit Fourier coefficients to the initial solution
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = h_start
        
        # We'll refit using FFT approach
        coeffs = np.zeros(num_coeffs)
        # Start with simple fitting approach
        coeffs[0] = np.mean(h)  # a0 as average
        # Use more coefficients for fine-tuning
        for i in range(1, max_freq+1):
            coeffs[2*i-1] = 2 * np.mean(h * np.cos(i * np.pi * x))  # ak
            coeffs[2*i] = 2 * np.mean(h * np.sin(i * np.pi * x))    # bk
    else:
        # Initialize coefficients
        coeffs = np.random.normal(0, 0.1, num_coeffs)
    
    def sigmoid(x):
        return 1 / (1 + np.exp(-x))
    
    def compute_h(coeffs):
        x = np.linspace(0, 2, n_points, endpoint=False)
        # Fourier series
        h = coeffs[0]  # a0
        for k in range(1, max_freq+1):
            h += coeffs[2*k-1] * np.cos(k * np.pi * x)  # ak*cos(k*pi*x)
            h += coeffs[2*k] * np.sin(k * np.pi * x)    # bk*sin(k*pi*x)
        # Apply sigmoid and normalize
        h = sigmoid(h)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def compute_overlap(coeffs):
        h = compute_h(coeffs)
        return evaluate_erdos_solution(h, dx)
    
    def gradient_descent_step(coeffs, lr=0.01):
        h_old = compute_h(coeffs)
        loss_old = evaluate_erdos_solution(h_old, dx)
        
        # Numerical gradient estimation
        eps = 1e-5
        grad = np.zeros_like(coeffs)
        
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = compute_h(coeffs_plus)
            loss_plus = evaluate_erdos_solution(h_plus, dx)
            
            grad[i] = (loss_plus - loss_old) / eps
            
        # Step
        coeffs_new = coeffs - lr * grad
        return coeffs_new
    
    iteration = 0
    max_iterations = 500
    lr = 0.1
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        coeffs = gradient_descent_step(coeffs, lr)
        c5 = compute_overlap(coeffs)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = compute_h(coeffs)
        
        # Decrease learning rate over time
        lr *= 0.995
        iteration += 1
    
    # Refine with scipy
    if (time() - start_time) < budget_s * 0.9:
        # Convert to flat vector
        def obj_func(c):
            coeffs_flat = c.copy()
            h = compute_h(coeffs_flat)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(c):
            coeffs_flat = c.copy()
            h = compute_h(coeffs_flat)
            return np.sum(h) - n_points / 2.0
        
        bounds = [(-5, 5)] * num_coeffs
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj_func, coeffs, method='SLSQP', bounds=bounds, 
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                coeffs_final = res.x
                h_refined = compute_h(coeffs_final)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
