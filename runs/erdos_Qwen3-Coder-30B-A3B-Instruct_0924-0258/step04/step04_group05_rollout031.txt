import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Linearly resample h_in to n_out points"""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_feasible(h):
    """Project h to feasible region: [0,1] and sum=1"""
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize the finest resolution
    n_fine = 100
    n_coarse = 50
    dx_fine = 2.0 / n_fine
    dx_coarse = 2.0 / n_coarse
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_fine) * 0.5
    
    # Resample to both grids
    h_coarse = linear_resample(h_init, n_coarse)
    h_fine = linear_resample(h_init, n_fine)
    
    # Project to feasible region
    h_coarse = project_to_feasible(h_coarse)
    h_fine = project_to_feasible(h_fine)
    
    best_h = h_fine.copy()
    best_c5 = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse grid search (varied from original)
    perturb_scale = 0.05
    for step in range(200):
        if time() - start_time > budget_s * 0.5:
            break
            
        # Perturb with adaptive scale
        h_pert = h_coarse + np.random.normal(0, perturb_scale, n_coarse)
        h_proj = project_to_feasible(h_pert)
        
        c5_val = evaluate_erdos_solution(h_proj, dx_coarse)
        if c5_val < best_c5:
            best_c5 = c5_val
            # Interpolate back to fine grid for better resolution
            best_h = linear_resample(h_proj, n_fine)
            best_h = project_to_feasible(best_h)
            
        # Decaying perturbation scale
        perturb_scale *= 0.995
    
    # Local refinement with fine grid
    def objective(h):
        return evaluate_erdos_solution(h, dx_fine)
    
    def constraint_sum(h):
        return np.sum(h) - n_fine / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_fine)]
    
    # Multi-start local optimization
    for restart in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Start from best found so far
        h_init = best_h.copy()
        # Add small random noise
        h_init += np.random.normal(0, 0.02, n_fine)
        h_init = project_to_feasible(h_init)
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            continue
    
    # Final fine-tuning with stricter tolerances
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_fine)
