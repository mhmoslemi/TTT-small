import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based search with adaptive parameters and tabu search
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 40
    population = []
    tabu_memory = set()
    tabu_limit = 50
    
    # Generate diverse initial population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Add 10 perturbed versions of the initial solution
        for _ in range(10):
            h_pert = initial_h + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            population.append(h_pert)
        
        # Add 5 random valid solutions
        for _ in range(5):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            population.append(h_rand)
    else:
        # Start with uniform distribution
        h_uniform = np.ones(n_points) * 0.5
        population.append(h_uniform)
        # Add 10 random valid solutions
        for _ in range(10):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points/2) / np.sum(h_rand)
            population.append(h_rand)
    
    # Fill the rest with random valid solutions
    while len(population) < population_size:
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fitness evaluation and tabu management
    def compute_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def is_tabu(h_tuple):
        return h_tuple in tabu_memory
    
    def add_to_tabu(h_tuple):
        if len(tabu_memory) >= tabu_limit:
            tabu_memory.pop()
        tabu_memory.add(h_tuple)
    
    def clip_and_normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h

    best_solution = None
    best_fitness = float('inf')
    stagnation_counter = 0
    max_stagnation = 50
    
    # Adaptive parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    gradient_refinement_freq = 10
    
    # Main evolutionary loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        current_fitnesses = [compute_fitness(h) for h in population]
        
        # Update best
        min_fitness = min(current_fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_solution = population[current_fitnesses.index(min_fitness)].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Adapt parameters if stagnation
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
            
        # Local gradient descent on top candidates
        if gen % gradient_refinement_freq == 0:
            top_indices = np.argsort(current_fitnesses)[:5]
            for idx in top_indices:
                h = population[idx].copy()
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(objective, h, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        h_refined = clip_and_normalize(h_refined)
                        fitness_refined = compute_fitness(h_refined)
                        if fitness_refined < current_fitnesses[idx]:
                            population[idx] = h_refined
                            current_fitnesses[idx] = fitness_refined
                            if fitness_refined < best_fitness:
                                best_fitness = fitness_refined
                                best_solution = h_refined.copy()
                except:
                    pass
        
        # Selection, crossover, and mutation
        selected_indices = np.random.choice(len(population), size=20, replace=False)
        new_population = [population[i].copy() for i in selected_indices]
        
        # Crossover
        for _ in range(10):
            if time() - start_time > budget_s * 0.95:
                break
            if random.random() < crossover_prob:
                parent1, parent2 = random.sample(new_population[:20], 2)
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
                child = clip_and_normalize(child)
                new_population.append(child)
        
        # Mutation
        for i in range(len(new_population)):
            if time() - start_time > budget_s * 0.95:
                break
            if random.random() < mutation_rate:
                child = new_population[i].copy()
                mutation_indices = np.random.choice(n_points, size=max(1, n_points // 10))
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, 0.03)
                child = clip_and_normalize(child)
                new_population[i] = child
        
        # Evict poor individuals and introduce new random solutions if needed
        if len(new_population) < population_size:
            while len(new_population) < population_size:
                h_rand = np.random.rand(n_points)
                h_rand = h_rand * (n_points/2) / np.sum(h_rand)
                new_population.append(h_rand)
        
        population = new_population[:population_size]
        
        # Ensure tabu memory and constraint maintenance
        for h in population:
            h_tuple = tuple(np.round(h, decimals=8).tolist())
            add_to_tabu(h_tuple)
    
    # Final refinement using more aggressive optimization
    if best_solution is not None:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_solution, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                best_solution = res.x
                best_fitness = objective(best_solution)
        except:
            pass
    
    # Final checks
    if best_solution is not None:
        final_sum = np.sum(best_solution)
        if abs(final_sum - n_points / 2) > 1e-6:
            best_solution = best_solution * (n_points/2) / final_sum
        final_fitness = evaluate_erdos_solution(best_solution, dx)
        return (list(best_solution), final_fitness, n_points)
    
    # Return the best found if avaliable
    if 'best_solution' in locals():
        final_fitness = evaluate_erdos_solution(best_solution, dx)
        return (list(best_solution), final_fitness, n_points)
    
    # Fallback with random solution constrained to sum = n_points/2
    fallback = np.random.rand(n_points)
    fallback = fallback * (n_points/2) / np.sum(fallback)
    fallback_fitness = evaluate_erdos_solution(fallback, dx)
    return (list(fallback), fallback_fitness, n_points)
