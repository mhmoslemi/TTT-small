import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with Augmented Lagrangian
    max_iter = 1000
    alpha = 0.01
    lambda_ = 0.0
    mu = 1.0
    patience = 0
    max_patience = 50
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlap values for all shifts
        overlap_vals = []
        for k in range(-n_points+1, n_points):
            if k < 0:
                # Left shift
                corr = np.sum(h_start[:n_points+k] * (1 - h_start[-k:]))
            elif k > 0:
                # Right shift
                corr = np.sum(h_start[k:] * (1 - h_start[:-k]))
            else:
                # Zero shift (should be ignored for max)
                corr = 0
            overlap_vals.append(corr)
        
        # Find maximum overlap
        max_k = np.argmax(overlap_vals)
        max_overlap = overlap_vals[max_k] * dx
        
        if max_overlap < best_c5:
            best_h = h_start.copy()
            best_c5 = max_overlap
            
        # Compute gradient (simplified approach)
        grad = np.zeros(n_points)
        k_max = max_k - (n_points - 1)
        if k_max < 0:
            # Left shift case
            grad[:n_points+k_max] += (1 - h_start[-k_max:])
            grad[-k_max:] -= h_start[:n_points+k_max]
        elif k_max > 0:
            # Right shift case
            grad[k_max:] += (1 - h_start[:-k_max])
            grad[:n_points-k_max] -= h_start[k_max:]
        else:
            # Zero shift case - ignored for now
            
        # Add penalty term from augmented Lagrangian
        penalty = lambda_ + 2 * mu * (np.sum(h_start) - n_points/2)
        grad += penalty
        
        # Update h
        h_new = h_start - alpha * grad
        
        # Project to [0,1]
        h_proj = np.clip(h_new, 0, 1)
        
        # Normalize to sum to n_points / 2
        h_proj = h_proj * (n_points/2) / np.sum(h_proj)
        
        # Update Lagrange multipliers
        constraint_violation = np.sum(h_proj) - n_points/2
        lambda_ += mu * constraint_violation
        if abs(constraint_violation) < 1e-6:
            mu *= 0.99
        
        # Adapt step size
        h_start = h_proj
        
        # Check for improvement
        if max_overlap < best_c5:
            patience = 0
            alpha *= 1.2  # Increase step size for faster convergence
        else:
            patience += 1
            alpha *= 0.5  # Reduce step size
            # Keep the better solution
            
        # Exit if no improvement in patience
        if patience > max_patience:
            break
    
    # Final refinement step with smoothing
    h_final = best_h.copy()
    h_smooth = np.zeros(n_points)
    for i in range(n_points):
        start_idx = max(0, i-1)
        end_idx = min(n_points, i+2)
        h_smooth[i] = np.mean(h_final[start_idx:end_idx])
    
    h_smooth = h_smooth * (n_points/2) / np.sum(h_smooth)
    h_smooth = np.clip(h_smooth, 0, 1)
    
    final_c5 = evaluate_erdos_solution(h_smooth, dx)
    
    if final_c5 < best_c5:
        best_h = h_smooth
        best_c5 = final_c5
        
    return (list(best_h), best_c5, n_points)
