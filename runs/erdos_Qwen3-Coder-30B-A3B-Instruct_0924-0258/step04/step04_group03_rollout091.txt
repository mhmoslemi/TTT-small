import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent approach
    max_iterations = 5000
    alpha = 0.1
    decay = 0.99
    tolerance = 1e-7
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_gradients(h, k):
        # Compute gradient of overlap with respect to h for shift k
        grad = np.zeros_like(h)
        if k < len(h):
            grad[k:] += (1 - h[:-k]) * (-1)
            grad[:-k] += (1 - h[k:]) * (-1)
        else:
            idx = k - len(h) + 1
            grad[:idx] += (1 - h[-(len(h)-idx):]) * (-1)
            grad[-(len(h)-idx):] += (1 - h[:idx]) * (-1)
        return grad

    current_h = h_start.copy()
    current_c5 = best_c5
    iteration = 0
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Find maximum overlap and corresponding shift
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (len(current_h) - 1)
        max_c5 = corr[max_idx] * dx
        
        if abs(max_c5 - current_c5) < tolerance:
            alpha *= 0.9
            if alpha < 1e-6:
                break
                
        # Compute gradient
        gradient = compute_gradients(current_h, k)
        
        # Update h
        current_h -= alpha * gradient
        current_h = project_h(current_h)
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        iteration += 1
        alpha *= decay
        
        # Occasionally refine with convex optimization
        if iteration % 50 == 0:
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                # Add top overlaps as constraints
                corr_full = np.correlate(current_h, 1 - current_h, mode='full')
                top_indices = np.argsort(corr_full)[-10:][::-1]
                top_shifts = [i - (n_points - 1) for i in top_indices]
                
                for k_idx in top_shifts:
                    if k_idx < 0:
                        corr_term = cp.sum(cp.multiply(h_var[:n_points + k_idx], 1 - h_var[-k_idx:]))
                    else:
                        corr_term = cp.sum(cp.multiply(h_var[k_idx:], 1 - h_var[:n_points - k_idx]))
                    constraints.append(corr_term <= (np.max(corr_full) + 0.1) * dx)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h_var.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
