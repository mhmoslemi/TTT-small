import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Problem, Minimize, Variable, norm, sum, multiply
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a stepped initial guess
        for i in range(n_points):
            x = i * dx
            if 0.2 <= x <= 0.8:
                h_start[i] = 1.0
            elif 1.2 <= x <= 1.8:
                h_start[i] = 1.0
        # Normalize to sum = n_points / 2
        scale = (n_points / 2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)

    # Binary search on C5
    low = 0.0
    high = 0.38085
    tolerance = 1e-6
    max_iters = 60
    iter_count = 0

    # For numerical stability, we'll use a small epsilon to ensure
    # compliance with constraints
    epsilon = 1e-8

    while iter_count < max_iters and time() - start_time < budget_s * 0.95:
        mid = (low + high) / 2.0
        if high - low < tolerance:
            break

        # LP formulation for feasibility check
        h_vars = Variable(n_points)
        # Add a small slack to stay away from bounds
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2,
        ]

        # Overlap constraints
        # For each possible lag k, form the overlap integral as a linear
        # constraint on h_vars using a piecewise linear approximation
        k_max = n_points - 1
        for k in range(k_max):
            # We can approximate the integral with a discrete sum
            # This uses a flipped version of h shifted by k
            integral_approx = 0.0
            # The correlate operation with mode='full' is equivalent to:
            # for each lag k:
            # integral = sum(h[i] * (1 - h[i+k])) if valid
            for i in range(n_points):
                j = i + k
                if j < n_points:
                    integral_approx += h_vars[i] * (1.0 - h_vars[j])
            constraints.append(integral_approx <= mid)

        # Create and solve the problem
        objective = Minimize(0.0)  # dummy objective
        prob = Problem(objective, constraints)

        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, feastol=1e-8, reltol=1e-8)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # Solution is feasible, so C5 is feasible
                high = mid
                # Keep the solution found for potential earlier use in refining
                h_sol = h_vars.value
                if h_sol is not None:
                    # Normalize
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    current_c5 = evaluate_erdos_solution(h_sol, dx)
                    if current_c5 < best_c5:
                        best_h = h_sol
                        best_c5 = current_c5
            else:
                # Infeasible - increase lower bound
                low = mid
        except Exception:
            # If solver fails, assume it's infeasible
            low = mid

        iter_count += 1

    # Final refinement step with local optimization from best_h
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}

    # Try to improve upon our best solution
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = obj(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass

    # Ensure final solution is within bounds and constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)

    return (list(best_h), final_c5, n_points)
