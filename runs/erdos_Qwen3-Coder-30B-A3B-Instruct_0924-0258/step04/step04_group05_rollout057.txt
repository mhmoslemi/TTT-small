import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Resample h_in to n_out points using linear interpolation"""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    # Coarse grid setup
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = linear_resample(h_init, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    
    # Fine grid setup
    n_fine = 100
    dx_fine = 2.0 / n_fine
    h_fine = linear_resample(h_init, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    # Best solutions tracking
    best_h_coarse = h_coarse.copy()
    best_h_fine = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse grid search
    perturbation_scale = 0.03
    for step in range(100):
        if time() - start_time > budget_s * 0.4:
            break
            
        # Perturb
        h_pert = h_coarse + np.random.normal(0, perturbation_scale, n_coarse) * 0.5
        h_pert = project_to_constraints(h_pert, n_coarse)
        c5 = evaluate_erdos_solution(h_pert, dx_coarse)
        
        if c5 < best_c5_coarse:
            best_c5_coarse = c5
            best_h_coarse = h_pert.copy()
            # Interpolate to fine grid
            h_fine_new = linear_resample(best_h_coarse, n_fine)
            h_fine_new = project_to_constraints(h_fine_new, n_fine)
            c5_fine_new = evaluate_erdos_solution(h_fine_new, dx_fine)
            if c5_fine_new < best_c5_fine:
                best_c5_fine = c5_fine_new
                best_h_fine = h_fine_new.copy()
        
        # Reduce perturbation
        perturbation_scale *= 0.99
    
    # Fine grid refinement
    perturbation_scale = 0.02
    for step in range(200):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Perturb
        h_pert = best_h_fine + np.random.normal(0, perturbation_scale, n_fine)
        h_pert = project_to_constraints(h_pert, n_fine)
        c5 = evaluate_erdos_solution(h_pert, dx_fine)
        
        if c5 < best_c5_fine:
            best_c5_fine = c5
            best_h_fine = h_pert.copy()
        
        # Reduce perturbation
        perturbation_scale *= 0.985
    
    # Re-optimize on best found, maybe improve further with local method
    def objective(h):
        return evaluate_erdos_solution(h, dx_fine)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_fine)]

    # Local optimization with SLSQP
    try:
        res = minimize(objective, h_final, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})

            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5_fine:
                best_c5_fine = c5_final
    except:
        pass
    
    # Final evaluation 
    final_c5 = evaluate_erdos_solution(best_h_fine, dx_fine)
    
    return (list(best_h_fine), final_c5, n_fine)
