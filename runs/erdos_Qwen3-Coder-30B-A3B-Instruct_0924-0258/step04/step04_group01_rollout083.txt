import numpy as np
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Simple step function
        h_start[:n_points//4] = 1.0
        h_start[n_points//4:3*n_points//4] = 0.0
        h_start[3*n_points//4:] = 1.0
        # Normalize to ensure sum is n/2
        h_start = h_start * (n_points/2) / np.sum(h_start)

    # Initialize bounds for binary search
    low = 0.3
    high = 0.38085
    precision_threshold = 1e-5
    max_iterations = 60
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Formulate and solve LP for feasibility
        h = Variable(n_points)
        
        # Objective is to minimize 0 (feasibility check)
        objective = Minimize(0)
        
        # Sum constraint
        constraints = [sum(h) == n_points / 2.0]
        
        # Bounds on h
        constraints.extend([h[i] >= 0 for i in range(n_points)])
        constraints.extend([h[i] <= 1 for i in range(n_points)])
        
        # Overlap constraints for all lags
        dx = 2.0 / n_points
        for k in range(2 * n_points - 1):
            # Sum over valid shifts
            expr = 0
            for a in range(n_points):
                if a + k < n_points:
                    expr += h[a] * (1 - h[a + k])
                else:
                    break
            constraints.append(expr <= mid / dx)
        
        # Create and solve the problem
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=GLPK, tol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Feasible solution found
                h_solution = np.array([float(h[i].value) for i in range(n_points)])
                current_c5 = evaluate_erdos_solution(h_solution, dx)
                if current_c5 < best_c5:
                    best_h = h_solution.copy()
                    best_c5 = current_c5
                high = mid
            else:
                low = mid
        except Exception:
            low = mid
            
    # Final validation to ensure feasibility
    # Ensure final evaluation doesn't fail
    current_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    best_c5 = min(best_c5, current_c5)  # Might have slightly improved
    
    return (list(best_h), best_c5, n_points)
