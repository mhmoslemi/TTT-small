import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    n_particles = 50
    max_iters = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best_idx = 0
    global_best_val = best_c5
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h *= (n_points / 2) / np.sum(h)  # Ensure sum constraint
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        val = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_vals.append(val)
        if val < global_best_val:
            global_best_val = val
            global_best_idx = i
    
    # PSO Main Loop
    for iter_num in range(max_iters):
        if (time() - start_time) > budget_s * 0.9:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iters
        
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Enforce sum constraint
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = val
                
                # Update global best
                if val < global_best_val:
                    global_best_val = val
                    global_best_idx = i
        
        # Update best solution found
        if global_best_val < best_c5:
            best_h = particles[global_best_idx].copy()
            best_c5 = global_best_val
    
    # Convex refinement using the best found solution
    if time() - start_time < budget_s * 0.95:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Create overlap constraints  
            max_k = min(10, 2 * n_points - 1)
            for k in range(max_k):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= best_c5 * n_points / 2)
            
            # Solve for tightening
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
