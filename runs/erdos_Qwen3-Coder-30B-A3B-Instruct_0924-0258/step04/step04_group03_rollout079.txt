import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Modified Simulated Annealing with pairwise perturbations
    def neighbor(h, step=0.05):
        h_new = h.copy()
        # Select 2 indices to perturb
        idx1, idx2 = np.random.randint(0, len(h), 2)
        delta = np.random.uniform(-step, step)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        # Ensure it stays within bounds
        h_new = np.clip(h_new, 0, 1)
        # Maintain sum constraint
        total = np.sum(h_new)
        h_new = h_new * (n_points / 2) / total
        return h_new
    
    current_h = h_start.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # SA parameters
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.9995
    step_count = 0
    max_steps = int(0.6 * budget_s * 1000)  # SA time
    
    while temperature > min_temp and step_count < max_steps and (time() - start_time) < budget_s * 0.95:
        # Generate neighbor
        new_h = neighbor(current_h, 0.03 * temperature)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            # Update best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        temperature *= alpha
        step_count += 1
    
    # Convex refinement with tighter constraints
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        try:
            h = cp.Variable(n_points)
            # Main constraints
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add key overlap constraints
            max_overlap = cp.Parameter(nonneg=True)
            max_overlap.value = best_c5
            
            # Constrain all valid overlaps
            for k in range(2 * n_points - 1):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= max_overlap * dx * 1.1)  # slack factor
            
            # Solve minimizing the shift constraint
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-7, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final fine-tuning with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Multiple restarts
            for restart in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                h_restart = best_h + np.random.normal(0, 0.01, n_points)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (n_points/2) / np.sum(h_restart)
                
                try:
                    res = minimize(objective, h_restart, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 200, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
