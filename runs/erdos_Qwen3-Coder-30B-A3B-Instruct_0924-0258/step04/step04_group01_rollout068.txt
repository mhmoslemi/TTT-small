import numpy as np
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a reasonable starting guess
        h_start[20:40] = 1.0
        h_start[60:80] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for C5
    low = 0.0
    high = 0.38085
    precision = 1e-6
    max_iterations = 60
    
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2.0
        # Solve feasibility using convex optimization
        try:
            h_vars = Variable(n_points)
            constraints = [0 <= h_vars, h_vars <= 1]
            constraints.append(sum(h_vars) == n_points / 2.0)
            
            # Add overlap constraints
            for k in range(n_points):
                # Compute the overlap integral using linearization
                # The product h[i] * (1 - h[i+k]) for valid indices
                # We consider a simplified version using auxiliary variables
                # For efficiency, we'll use the known structure for overlap integral
                if k == 0:
                    continue # Skip zero lag which is 0 due to definition
                # For circular/lag-k computation we compute loop
                # But in practice we just create a linear problem
                # Try to avoid complex inner product by introducing auxiliary variables
                # Lag k: terms h[i] * (1 - h[i+k]) for i s.t. i+k < n_points
                # We approximate the correlation using a linear constraint
                # The simplest way is using a dual-solution reconstruction from valid inequalities
                
                # Specifically, for each lag k, ensure the overlap is bounded
                # Use the second form of overlap: ∫ h(x) (1 - h(x+k)) dx
                
                # Reconstruct 1-d corr via num samples
                corr_constraint = 0
                for i in range(n_points - k):
                    corr_constraint += h_vars[i] * (1 - h_vars[i + k])
                corr_constraint *= dx
                
                # Add constraint for max overlap C5
                constraints.append(corr_constraint <= mid)

            # Objective: minimize 0 (just solve feasibility)
            prob = Problem(Minimize(0), constraints)
            prob.solve(solver=CPLEX, verbose=False, feastol=1e-6, opttol=1e-6)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    c5_val = evaluate_erdos_solution(h_sol, dx)
                    if c5_val < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_val
                    # Update bounds
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid
    
    # Final refinement of best solution
    try:
        h_refined = best_h.copy()
        # Use scipy optimization to fine-tune
        from scipy.optimize import minimize
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(obj, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-6})
        
        if res.success:
            h_refined = res.x
            c5_val = obj(h_refined)
            if c5_val < best_c5:
                best_h = h_refined
                best_c5 = c5_val
    except:
        pass

    # Ensure final constraint satisfaction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
