import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with 3 clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.4, 1.0, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 20
    generations = 150
    mutation_rate = 0.3
    crossover_rate = 0.5
    elite_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            # Random perturbation of the initial solution
            h_candidate = h_start + np.random.normal(0, 0.02, n_points)
        else:
            # Different random initialization
            h_candidate = np.random.rand(n_points) * 0.5
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Add the initial solution to population
    population.append(h_start.copy())
    
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(c5_val)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: keep top elite_size
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                child1 = mutate(child1, dx, n_points)
            if np.random.rand() < mutation_rate:
                child2 = mutate(child2, dx, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
    
    # Final SQP refinement of the best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-7})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final boundary check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitness, k=3):
    selected_indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = selected_indices[np.argmin([fitness[i] for i in selected_indices])]
    return population[best_idx]

def crossover(parent1, parent2):
    # Uniform crossover
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, dx, n_points):
    # Gaussian mutation
    mutation_strength = 0.03
    h_mutated = h + np.random.normal(0, mutation_strength, n_points)
    h_mutated = np.clip(h_mutated, 0, 1)
    
    # Normalize sum
    h_mutated = h_mutated * (n_points/2) / np.sum(h_mutated)
    return h_mutated
