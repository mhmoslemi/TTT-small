import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_limit = 50
    stagnation_count = 0
    max_stagnation = 50
    
    # Genetic algorithm parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Initialize population
    population = []
    fitness = []
    
    # Add perturbed versions of initial solution
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random valid solutions
    for _ in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Initialize with remaining population
    while len(population) < population_size:
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Main optimization loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Find best in current population
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check termination conditions
        if best_c5 < 0.3807:
            break
            
        if stagnation_count > max_stagnation:
            # Increase mutation rate and decrease crossover
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            stagnation_count = 0
            tabu_list = []  # Clear tabu list to encourage exploration
        
        # Create new population via selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best individual
        best_individual = population[best_idx]
        new_population.append(best_individual)
        
        # Generate rest of population
        while len(new_population) < population_size:
            # Selection (tournament)
            tournament_size = 3
            idx1 = np.random.choice(len(population), tournament_size)
            idx2 = np.random.choice(len(population), tournament_size)
            best1 = np.argmin([fitness[i] for i in idx1])
            best2 = np.argmin([fitness[i] for i in idx2])
            
            parent1 = population[idx1[best1]]
            parent2 = population[idx2[best2]]
            
            # Crossover
            if np.random.random() < crossover_prob:
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Add small Gaussian noise
                child += np.random.normal(0, 0.01, n_points)
                child = np.clip(child, 0, 1)
                # Renormalize
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        # Local refinement for top solutions
        if gen % 10 == 0:
            # Refine top 5 individuals
            top_indices = np.argsort(fitness)[:5]
            for idx in top_indices:
                h_refine = population[idx].copy()
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = obj(refined_h)
                        if refined_c5 < fitness[idx]:
                            population[idx] = refined_h
                            fitness[idx] = refined_c5
                except:
                    pass
        
        population = new_population[:population_size]
    
    # Final gradient refinement on best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
