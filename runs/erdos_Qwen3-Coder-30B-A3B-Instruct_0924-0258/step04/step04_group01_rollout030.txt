import numpy as np
from cvxpy import *
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Simple step function
        h_start[:n_points//2] = 1
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iterations = 60
    
    # Solver settings
    solver = 'GLPK'
    
    best_feasible_h = best_h.copy()
    best_feasible_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Setup LP for feasibility check
        h_vars = Variable(n_points)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(sum(h_vars) == n_points / 2)
        
        # Bounds
        for i in range(n_points):
            constraints.append(h_vars[i] >= 0)
            constraints.append(h_vars[i] <= 1)
        
        # Overlap constraints for all valid lags
        # For each shift k (from -n_points+1 to n_points-1)
        for k in range(-n_points+1, n_points):
            # This represents sum over valid a of h[a] * (1 - h[a+k])
            # Need to build up this constraint manually
            # We'll use idx for a from 0 to n_points-1
            # For h[a+k], if a+k < 0 or a+k >= n_points, we set to 0 in the sum
            # But cvxpy variables stay in [0,1], so we'll handle bounds implicitly
            # Through indexing in Python in a loop
            lag_sum = 0
            # Manually construct this term: sum over a of h[a] * (1 - h[a+k])
            # This requires indexing:
            # When k >= 0: h[a+k] exists when a < n_points-k
            # When k < 0: h[a+k] exists when a >= -k
            if k >= 0:
                # a from 0 to n_points-k-1
                # kth element shifts have as indices k to n_points-1
                for a in range(n_points - k):
                    lag_sum += h_vars[a] * (1 - h_vars[a + k])
            else:
                # k < 0, so a+k ranges from k to n_points-1
                # which means a from -k to n_points-1
                for a in range(-k, n_points):
                    lag_sum += h_vars[a] * (1 - h_vars[a + k])
            
            constraints.append(lag_sum <= mid / dx)
        
        # Try solving the feasibility LP
        try:
            prob = Problem(Minimize(0), constraints)
            prob.solve(solver=solver, verbose=False)
            
            if prob.status == 'optimal':
                # Found a feasible solution
                h_sol = np.array([float(h_vars[i].value) for i in range(n_points)])
                # Check if this is better than our global best
                c5_sol = evaluate_erdos_solution(h_sol, dx)
                if c5_sol < best_feasible_c5:
                    best_feasible_h = h_sol.copy()
                    best_feasible_c5 = c5_sol
                # Update bound
                high = mid
            else:
                # Check if infeasible, tighten lower bound
                low = mid
                
        except:
            # Fallback to next iteration
            low = mid
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_feasible_h, dx)
    final_h = best_feasible_h.copy()
    
    # Ensure it meets constraints
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
