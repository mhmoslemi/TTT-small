import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Gradient descent with adaptive learning rate
    h_current = best_h.copy()
    lr = 1e-3
    max_iter = 10000
    patience = 50
    best_obj = objective(h_current)
    patience_counter = 0
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient approximation (finite differences)
        eps = 1e-5
        grad = np.zeros(n_points)
        base_obj = objective(h_current)
        
        for j in range(n_points):
            h_plus = h_current.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus)
            obj_plus = objective(h_plus)
            grad[j] = (obj_plus - base_obj) / eps
        
        # Update step
        h_new = h_current - lr * grad
        h_new = project_to_feasible(h_new)
        
        # Evaluate new objective
        new_obj = objective(h_new)
        
        if new_obj < best_obj:
            h_current = h_new
            best_obj = new_obj
            patience_counter = 0
            if best_obj < best_c5:
                best_h = h_current.copy()
                best_c5 = best_obj
        else:
            patience_counter += 1
            if patience_counter > patience:
                lr *= 0.9
                patience_counter = 0
                
        # Occasionally inject noise to escape local minima
        if i % 100 == 0 and i > 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_current += noise
            h_current = project_to_feasible(h_current)
            
    # Final local refinement
    def ref_objective(h):
        return evaluate_erdos_solution(h, dx)
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    for restart in range(3):
        if time() - start_time > budget_s * 0.95:
            break
        h_restart = h_current + np.random.normal(0, 0.01, n_points)
        h_restart = project_to_feasible(h_restart)
        try:
            res = minimize(ref_objective, h_restart, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = ref_objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
