import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy with adaptive mutation
    population_size = 50
    mutation_rate = 0.05
    elite_size = population_size // 5
    max_generations = 1000
    
    # Initialize population around the starting point
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_new = h_start + np.random.normal(0, mutation_rate, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        population.append(h_new)
    
    generation = 0
    while (time() - start_time) < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness of population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: keep elite
        elite = population[:elite_size]
        
        # Create new population through mutation and crossover
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Crossover (arithmetic)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            mut_strength = max(0.001, mutation_rate * (1 - generation / max_generations))
            noise = np.random.normal(0, mut_strength, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
        generation += 1
        
        # Occasionally refine top solutions using SLSQP
        if generation % 10 == 0:
            for i in range(min(5, len(population))):
                h_refine = population[i].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
    
    return (list(best_h), best_c5, n_points)
