import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based search with adaptive parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    stagnation_limit = 50
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Start with best known or random solutions
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add diverse random solutions
    for _ in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fill rest with variations
    while len(population) < population_size:
        h1 = population[np.random.randint(0, len(population))]
        h2 = population[np.random.randint(0, len(population))]
        if np.random.random() < crossover_prob:
            mask = np.random.random(n_points) > 0.5
            h_child = np.where(mask, h1, h2)
        else:
            h_child = h1.copy()
        if np.random.random() < mutation_rate:
            mut_idx = np.random.randint(0, n_points)
            h_child[mut_idx] = np.random.random()
        h_child = np.clip(h_child, 0, 1)
        h_child = h_child * (n_points/2) / np.sum(h_child)
        population.append(h_child)
    
    # Tabu list for recent solutions
    tabu_list = []
    
    # Adaptive parameters
    stagnation_count = 0
    last_improvement = 0
    
    # Evaluate initial population
    for i in range(len(population)):
        h = population[i]
        fitness = evaluate_erdos_solution(h, dx)
        fitnesses.append(fitness)
        if fitness < best_c5:
            best_c5 = fitness
            best_h = h.copy()
    
    # Main genetic search loop
    iteration = 0
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        if iteration % 10 == 0:
            # Local refine top 5 candidates
            sorted_indices = np.argsort(fitnesses)
            for i in range(min(5, len(sorted_indices))):
                idx = sorted_indices[i]
                h_refine = population[idx].copy()
                def obj_func(h):
                    return evaluate_erdos_solution(h, dx)
                res = minimize(obj_func, h_refine, method='SLSQP', 
                               bounds=[(0,1) for _ in range(n_points)], 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_new = res.x
                    fitness_new = evaluate_erdos_solution(h_new, dx)
                    if fitness_new < best_c5:
                        best_c5 = fitness_new
                        best_h = h_new.copy()
                    # Update in population if better
                    if fitness_new < fitnesses[idx]:
                        fitnesses[idx] = fitness_new
                        population[idx] = h_new.copy()
                if best_c5 < 0.3807:
                    break
        
        # Selection: tournament
        new_population = []
        new_fitnesses = []
        
        # Keep good solutions
        top_indices = np.argsort(fitnesses)[:population_size//4]
        for idx in top_indices:
            new_population.append(population[idx].copy())
            new_fitnesses.append(fitnesses[idx])
        
        # Generate new population
        while len(new_population) < population_size:
            # Tournament selection
            tourn_size = 3
            idx1 = np.random.choice(len(population), tourn_size, replace=False)
            idx2 = np.random.choice(len(population), tourn_size, replace=False)
            best1 = idx1[np.argmin([fitnesses[i] for i in idx1])]
            best2 = idx2[np.argmin([fitnesses[i] for i in idx2])]
            
            # Crossover
            if np.random.random() < crossover_prob:
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, population[best1], population[best2])
            else:
                child = population[best1].copy()
            
            # Mutation
            if np.random.random() < mutation_rate:
                mut_pos = np.random.randint(0, n_points)
                child[mut_pos] = np.random.random()
            
            # Repair and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Avoid tabu
            if len(tabu_list) > 0:
                tabu_match = False
                for tabu_h in tabu_list:
                    if np.allclose(child, tabu_h, atol=1e-5):
                        tabu_match = True
                        break
                if tabu_match:
                    # Slight random perturb
                    child += np.random.normal(0, 0.005, n_points)
                    child = np.clip(child, 0, 1)
                    child = child * (n_points/2) / np.sum(child)
            
            # Add to new population
            new_population.append(child)
            fitness = evaluate_erdos_solution(child, dx)
            new_fitnesses.append(fitness)
            
            # Update tabu list
            if len(tabu_list) >= tabu_limit:
                tabu_list.pop(0)
            tabu_list.append(child.copy())
        
        population = new_population
        fitnesses = new_fitnesses
        
        # Adaptive parameters
        current_best = min(fitnesses)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[fitnesses.index(current_best)].copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                # Increase mutation rate
                mutation_rate = min(0.5, mutation_rate + 0.1)
                crossover_prob = max(0.3, crossover_prob - 0.1)
                tabu_limit = max(10, tabu_limit // 2)
                stagnation_count = 0
            elif iteration - last_improvement > 20:
                # Slightly increase mutation and crossover
                if mutation_rate > 0.05:
                    mutation_rate -= 0.02
                if crossover_prob < 0.9:
                    crossover_prob += 0.02
        
        if best_c5 < 0.3807:
            break
    
    # Final gradient refinement of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj_func(h):
            return evaluate_erdos_solution(h, dx)
        
        res = minimize(obj_func, best_h, method='SLSQP', 
                       bounds=[(0,1) for _ in range(n_points)], 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
