import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 40, 50]
    if n_points not in resolutions:
        resolutions = [n_points] + resolutions
    
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        current_n = res
        dx_current = 2.0 / current_n
        
        # Interpolate if needed
        if current_n == n_points:
            h_current = best_h.copy()
        else:
            h_current = np.interp(np.linspace(0, 2, current_n), 
                                 np.linspace(0, 2, n_points), 
                                 best_h)
        
        # Ensure constraints
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (current_n/2) / np.sum(h_current)
        
        # Initialize optimization variables
        h_opt = h_current.copy()
        c5_opt = evaluate_erdos_solution(h_opt, dx_current)
        
        max_restarts = 30
        restart_count = 0
        improvement_wait = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Local search with SLSQP
            def objective(h):
                return evaluate_erdos_solution(h, dx_current)
            
            def constraint_sum(h):
                return np.sum(h) - current_n / 2.0
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(current_n)]
                
                # Random restart
                h_restart = h_opt + np.random.normal(0, 0.02, current_n)
                h_restart = np.clip(h_restart, 0, 1)
                h_restart = h_restart * (current_n/2) / np.sum(h_restart)
                
                res_opt = minimize(objective, h_restart, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 200, 'ftol': 1e-8})
                
                if res_opt.success:
                    h_refined = res_opt.x
                    c5_refined = objective(h_refined)
                    if c5_refined < c5_opt:
                        h_opt = h_refined
                        c5_opt = c5_refined
                        improvement_wait = 0
                    else:
                        improvement_wait += 1
                else:
                    improvement_wait += 1
            except Exception:
                improvement_wait += 1
            
            # Occasionally try convex relaxation
            if time() - start_time < budget_s * 0.8 and improvement_wait % 4 == 0 and current_n > 30:
                try:
                    h_cvx = cp.Variable(current_n)
                    constraints = [
                        cp.sum(h_cvx) == current_n / 2.0,
                        h_cvx >= 0,
                        h_cvx <= 1
                    ]
                    
                    # Add overlap constraints for a subset of shifts
                    # We only use a few representative shifts to avoid large constraints
                    shifts_to_check = [1, current_n//4, current_n//2, 3*current_n//4, current_n-1]
                    for k in shifts_to_check:
                        if k < current_n:
                            corr_sum = cp.sum(cp.multiply(h_cvx[:current_n-k], 1 - h_cvx[k:]))
                        else:
                            corr_sum = cp.sum(cp.multiply(h_cvx[k-current_n+1:], 1 - h_cvx[:current_n-(k-current_n+1)]))
                        constraints.append(corr_sum <= c5_opt * dx_current + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-7)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_cvx.value
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx_current)
                        if c5_feasible < c5_opt:
                            h_opt = h_feasible
                            c5_opt = c5_feasible
                except Exception:
                    pass
            
            # Update best solution
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                restart_count = 0
            else:
                restart_count += 1
            
            if improvement_wait > 10:
                break
        
        # Final adjustment
        try:
            h_final = h_opt.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (current_n/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx_current)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, len(best_h))
