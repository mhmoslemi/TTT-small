import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 500
    mutation_rate = 0.1
    tournament_size = 2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        population[0] = population[0] * (n_points/2) / np.sum(population[0])
    
    # Fill the rest with random feasible individuals
    for i in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Optimization loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Find best in current population
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
        
        # Selection and reproduction
        new_population = []
        
        # Keep the best individual
        new_population.append(population[best_idx].copy())
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = random.randint(0, population_size - 1)
            parent2_idx = random.randint(0, population_size - 1)
            
            if fitness[parent1_idx] < fitness[parent2_idx]:
                parent1 = population[parent1_idx]
            else:
                parent1 = population[parent2_idx]
            
            parent2_idx = random.randint(0, population_size - 1)
            if fitness[parent1_idx] < fitness[parent2_idx]:
                parent2 = population[parent1_idx]
            else:
                parent2 = population[parent2_idx]
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.01)
            
            # Repair feasibility
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
    # Final refinement with local search on best
    try:
        h_local = best_h.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points/2) / np.sum(h_local)
        c5_local = evaluate_erdos_solution(h_local, dx)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
