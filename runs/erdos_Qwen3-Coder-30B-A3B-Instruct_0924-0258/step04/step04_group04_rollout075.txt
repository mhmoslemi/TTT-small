import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with parent construction or uniform distribution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization
    max_iter = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    # Population of solutions
    pop_size = 15
    population = []
    for _ in range(pop_size):
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    def optimize_single(h_in):
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_in, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                return res.x
        except:
            pass
        return h_in
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        for i, h in enumerate(population):
            curr_c5 = evaluate_erdos_solution(h, dx)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
                
            # Refine with SQP
            refined = optimize_single(h)
            refined_c5 = evaluate_erdos_solution(refined, dx)
            if refined_c5 < curr_c5:
                population[i] = refined
                if refined_c5 < best_c5:
                    best_h = refined.copy()
                    best_c5 = refined_c5
        
        # Generate new population with perturbations and simulated annealing
        new_population = []
        for h in population:
            # Perturb
            h_pert = h + np.random.normal(0, perturb_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            # Accept with probability based on temperature
            old_c5 = evaluate_erdos_solution(h, dx)
            new_c5 = evaluate_erdos_solution(h_pert, dx)
            if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
                new_population.append(h_pert)
            else:
                new_population.append(h)
        
        population = new_population
        temp *= cool_rate
    
    # Upsample to finer grid
    n_points_fine = 40
    dx_fine = 2.0 / n_points_fine
    h_fine = np.interp(np.linspace(0, 2, n_points_fine), 
                       np.linspace(0, 2, n_points), best_h)
    h_fine = h_fine * (n_points_fine/2) / np.sum(h_fine)
    
    # Fine grid optimization
    def obj_fine(h):
        return evaluate_erdos_solution(h, dx_fine)
        
    def con_sum_fine(h):
        return np.sum(h) - n_points_fine/2
        
    bounds_fine = [(0, 1) for _ in range(n_points_fine)]
    cons_fine = {'type': 'eq', 'fun': con_sum_fine}
    
    try:
        res = minimize(obj_fine, h_fine, method='SLSQP', 
                      bounds=bounds_fine, constraints=cons_fine, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            h_fine = res.x
            c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
            if c5_fine < best_c5:
                best_h = h_fine
                best_c5 = c5_fine
    except:
        pass
    
    # Final refinement using gradient descent
    if time() - start_time < budget_s - 10:
        h_refine = best_h.copy()
        learning_rate = 0.01
        for _ in range(50):
            if time() - start_time > budget_s - 5:
                break
            grad = np.zeros_like(h_refine)
            # Compute approximate gradient using finite differences
            epsilon = 1e-4
            for i in range(len(h_refine)):
                h_plus = h_refine.copy()
                h_plus[i] += epsilon
                h_plus = h_plus * (n_points_fine/2) / np.sum(h_plus)
                val_plus = evaluate_erdos_solution(h_plus, dx_fine)
                
                h_minus = h_refine.copy()
                h_minus[i] -= epsilon
                h_minus = h_minus * (n_points_fine/2) / np.sum(h_minus)
                val_minus = evaluate_erdos_solution(h_minus, dx_fine)
                
                grad[i] = (val_plus - val_minus) / (2 * epsilon)
            
            # Update
            h_refine -= learning_rate * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points_fine/2) / np.sum(h_refine)
            
            curr_c5 = evaluate_erdos_solution(h_refine, dx_fine)
            if curr_c5 < best_c5:
                best_h = h_refine.copy()
                best_c5 = curr_c5
    
    final_c5 = evaluate_erdos_solution(best_h, dx_fine)
    
    return (list(best_h), final_c5, n_points_fine)
