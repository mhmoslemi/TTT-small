import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient-based optimization with shift-specific gradient
    max_iter = 1000
    step_size = 0.01
    patience = 100
    patience_counter = 0
    
    h = best_h.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current overlap for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        # Compute gradient w.r.t h for the shift k_max
        # The gradient for h[i] is (1 - h[i+k_max]) - h[i] if k_max != 0
        # Else it's (1 - h[i]) - h[i] = 1 - 2*h[i]
        gradient = np.zeros(n_points)
        shift = k_max - (n_points - 1)  # k_max is indexed from 0 to 2*n_points-2
        
        if shift == 0:
            for i in range(n_points):
                gradient[i] = 1 - 2 * h[i]
        else:
            # Use circular shift for gradient computation
            # This requires careful indexing 
            for i in range(n_points):
                j = (i + shift) % n_points
                gradient[i] = 1 - h[j] - h[i] if shift != 0 else 1 - h[i] * 2
            
        # Update with gradient descent
        h_new = h - step_size * gradient
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
        
        h = h_new.copy()
        
        # Reduce step size if no improvement in some iterations
        if patience_counter > 20:
            step_size *= 0.9
        
        if patience_counter > patience:
            break
    
    # Final refinement using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
