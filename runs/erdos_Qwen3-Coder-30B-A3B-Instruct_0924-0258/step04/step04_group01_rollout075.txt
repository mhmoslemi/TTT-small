import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # 3-cluster construction
        cluster_width = 1.0
        centers = [0.3, 1.0, 1.7]
        for c in centers:
            start = max(0, int((c - cluster_width/2)/dx))
            end = min(n_points, int((c + cluster_width/2)/dx))
            h_start[start:end] = 1.0 / cluster_width
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 20
    generations = 300
    mutation_scale = 0.02
    crossover_rate = 0.5
    elite_size = 4
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h_candidate = h_start + np.random.normal(0, mutation_scale, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Track best in population
    for h_candidate in population:
        c5_val = evaluate_erdos_solution(h_candidate, dx)
        if c5_val < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_val

    # Main GA loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(c5_val)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Create new generation
        new_population = population[:elite_size]  # Keep best
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection (tournament)
            idx1 = np.random.randint(0, population_size//2)
            idx2 = np.random.randint(0, population_size//2)
            h1 = population[idx1]
            h2 = population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                offspring1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                offspring2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
            else:
                offspring1 = h1.copy()
                offspring2 = h2.copy()
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% mutation rate
                    offspring1[i] += np.random.normal(0, mutation_scale)
                    offspring2[i] += np.random.normal(0, mutation_scale)
                    
            # Clip and renormalize
            offspring1 = np.clip(offspring1, 0, 1)
            offspring1 = offspring1 * (n_points/2) / np.sum(offspring1)
            
            offspring2 = np.clip(offspring2, 0, 1)
            offspring2 = offspring2 * (n_points/2) / np.sum(offspring2)
            
            new_population.append(offspring1)
            if len(new_population) < population_size:
                new_population.append(offspring2)
        
        population = new_population
        
        # Refinement step
        for i in range(len(population)):
            h_current = population[i]
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < fitness[i]:
                        population[i] = h_refined
                        if c5_refined < best_c5:
                            best_h = h_refined.copy()
                            best_c5 = c5_refined
            except:
                pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
