import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Always use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Initial guess: piecewise constant with structure
        n_points = 40
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a structure that balances concentration and spread
        h_start[8:12] = 0.8  # Higher concentration in middle
        h_start[28:32] = 0.8  # Another high concentration near end
        h_start[10:25] = 0.4  # Lower spread in between
        # Normalize to sum = n_points / 2
        h_start *= (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    # Phase 1: Coarse grid optimization
    coarse_n = 20
    coarse_dx = 2.0 / coarse_n
    
    # Initialize coarse h
    coarse_h = np.zeros(coarse_n)
    for i in range(coarse_n):
        start = int(i * 2.0 / coarse_n * n_points)
        end = int((i + 1) * 2.0 / coarse_n * n_points)
        if end > start:
            coarse_h[i] = np.mean(best_h[start:end])
        else:
            coarse_h[i] = 0.0
    coarse_h = coarse_h * (coarse_n/2) / np.sum(coarse_h)
    
    # Penalty-based optimization for coarse grid
    penalty = np.ones(2*coarse_n - 1)
    learning_rate = 0.1
    max_iter = 100
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps
        corr = np.correlate(coarse_h, 1 - coarse_h, mode='full')
        
        # Update penalties
        for i in range(len(penalty)):
            penalty[i] = max(penalty[i] * (1 + learning_rate * (corr[i] - penalty[i])), 1e-6)
            
        # For CVX optimization
        h_var = cp.Variable(coarse_n)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == coarse_n / 2
        ]
        obj = cp.Minimize(cp.sum(cp.maximum(corr + penalty, 0)))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, warm_start=True, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = np.array(h_var.value).flatten()
                new_h = new_h * (coarse_n/2) / np.sum(new_h)
                coarse_h = new_h
        except:
            pass
        
        # Evaluate
        curr_c5 = evaluate_erdos_solution(coarse_h, coarse_dx)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            # Interpolate back to fine grid
            fine_interp = interp1d(np.linspace(0, 2, coarse_n), coarse_h, 
                                 kind='linear', fill_value='extrapolate')
            temp_h = fine_interp(np.linspace(0, 2, n_points))
            temp_h = np.clip(temp_h, 0, 1)
            temp_h = temp_h * (n_points/2) / np.sum(temp_h)
            if np.allclose(np.sum(temp_h), n_points/2):
                best_h = temp_h.copy()
    
    # Phase 2: Upsample and fine optimization
    fine_h = np.zeros(n_points)
    fine_interp = interp1d(np.linspace(0, 2, coarse_n), coarse_h, 
                         kind='linear', fill_value='extrapolate')
    fine_h = fine_interp(np.linspace(0, 2, n_points))
    fine_h = np.clip(fine_h, 0, 1)
    fine_h = fine_h * (n_points/2) / np.sum(fine_h)
    coarse_h = fine_h.copy()
    
    # Use more aggressive learning rate
    penalty = np.ones(2*n_points - 1)
    learning_rate = 0.2
    
    max_iter = 200
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(coarse_h, 1 - coarse_h, mode='full')
        
        # Update penalties
        for i in range(len(penalty)):
            if penalty[i] > 0:
                penalty[i] = max(penalty[i] * (1 + learning_rate * (corr[i] - penalty[i])), 1e-6)
            
        # Convex optimization
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        obj = cp.Minimize(cp.sum(cp.maximum(corr + penalty, 0)))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, warm_start=True, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = np.array(h_var.value).flatten()
                new_h = new_h * (n_points/2) / np.sum(new_h)
                coarse_h = new_h
        except:
            pass
            
        # Evaluate
        curr_c5 = evaluate_erdos_solution(coarse_h, 2.0/n_points)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = coarse_h.copy()
    
    # Final refinement with NLP solver
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, 2.0/n_points)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        # Start from best solution
        refined = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-8})
        if refined.success:
            final_h = refined.x
            final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
