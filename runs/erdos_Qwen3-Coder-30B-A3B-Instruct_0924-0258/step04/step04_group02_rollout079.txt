import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    # Use initial construction if available

    def reparameterize(y, b):
        # Convert y to h using sigmoid
        return 1.0 / (1.0 + np.exp(-y - b))
    
    def compute_b(y):
        # Compute b such that integral of σ(y + b) equals 1
        h = reparameterize(y, 0)  # Just to know shape
        integral = np.sum(h) * dx
        if integral > 0:
            # Compute b to make integral = 1
            # integral = sum(1/(1+exp(-y-b)))*dx = 1
            # approximate using logit
            # logit(h) = -y - b = log(h/(1-h))
            # sum(log(h/(1-h)))*dx = sum(log(h) - log(1-h))*dx
            # 
            # We want sum(h)*dx = 1, use b such that this holds
            # With Newton's method on this constraint
            # No simple form - we solve using linsearch
            # Try scaling y so sum = 1
            h = reparameterize(y, 0)
            integral = np.sum(h) * dx
            b = -np.log(2 * integral / (2 - integral)) if integral > 0 else 0
            return b
        return 0

        b = compute_b(y)
        h = reparameterize(y, b)
        return evaluate_erdos_solution(h, dx)

        b = compute_b(y)
        h = reparameterize(y, b)
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(y)

            y_minus = y.copy()

            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')

        except:
            pass
