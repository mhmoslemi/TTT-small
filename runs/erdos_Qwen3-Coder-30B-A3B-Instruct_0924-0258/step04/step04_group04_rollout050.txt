import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_coarse = 20
    n_fine = 40
    max_iter_coarse = 100
    max_iter_fine = 100
    temp_start = 0.1
    cool_rate = 0.98
    perturb_mag_coarse = 0.03
    perturb_mag_fine = 0.02
    
    # Initialize from parent or create starting point
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 20
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_init.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Coarse grid optimization
    temp = temp_start
    for iter_coarse in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Generate perturbation
        h_pert = h_current + np.random.normal(0, perturb_mag_coarse, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Evaluation
        old_c5 = evaluate_erdos_solution(h_current, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        # Accept or reject
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            h_current = h_pert.copy()
            if new_c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = new_c5
        
        temp *= cool_rate
    
    # Upsample to fine grid
    if n_points == n_coarse:
        # Interpolate to fine grid
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), h_current)
        h_fine = h_fine * (n_fine/2) / np.sum(h_fine)
        n_points = n_fine
        dx = 2.0 / n_points
        h_current = h_fine.copy()
        
        # Update best
        current_c5 = evaluate_erdos_solution(h_current, dx)
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
    
    # Fine grid optimization with SLSQP
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 100:
        try:
            res = minimize(obj, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': max_iter_fine, 'ftol': 1e-6})
            if res.success:
                h_opt = res.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
    
    # Final refinement with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        h_refine = best_h.copy()
        max_iter_refine = 50
        lr = 1e-3
        for iter_refine in range(max_iter_refine):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient approximation
            current_c5 = evaluate_erdos_solution(h_refine, dx)
            eps = 1e-5
            grad = np.zeros_like(h_refine)
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (c_plus - current_c5) / eps
            
            # Update
            h_new = h_refine - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < current_c5:
                h_refine = h_new.copy()
                if new_c5 < best_c5:
                    best_h = h_refine.copy()
                    best_c5 = new_c5
            else:
                lr *= 0.95
    
    return (list(best_h), best_c5, n_points)
