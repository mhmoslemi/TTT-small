import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with coarse resolution
    n_points = 20
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Adjust to current resolution
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, len(h)), np.linspace(0, 2, len(h)), h)
            h = np.clip(h, 0, 1)
        # Normalize to sum constraint
        h = h * (n_points/2) / np.sum(h)
    else:
        # Start with a piecewise constant approximation
        h = np.ones(n_points) * 0.25
        # Ensure sum constraint
        h = h * (n_points/2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization (n_points=20)
    max_iter_coarse = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.85:
            break
            
        # Perturb
        h_pert = h + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Acceptance probability
        old_c5 = evaluate_erdos_solution(h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            h = h_pert
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        
        temp *= cool_rate

    # Upsample to n_points = 40
    n_points = 40
    h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    h_upsampled = np.clip(h_upsampled, 0, 1)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    dx = 2.0 / n_points
    h = h_upsampled
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()

    # Fine optimization (n_points=40)
    max_iter_fine = 100
    temp = 0.05
    cool_rate = 0.99
    
    for iteration in range(max_iter_fine):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        h_pert = h + np.random.normal(0, 0.015, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # Acceptance probability
        old_c5 = evaluate_erdos_solution(h, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            h = h_pert
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        
        temp *= cool_rate
    
    # Final refinement with constrained optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    return (list(best_h), best_c5, n_points)
