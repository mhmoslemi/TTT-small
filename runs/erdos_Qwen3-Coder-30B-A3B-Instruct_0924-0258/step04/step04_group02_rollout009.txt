import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with reparameterization
    max_iter = 10000
    lr = 0.5
    tolerance = 1e-8
    
    # Reparameterize using logistic function, h = sigmoid(z + b)
    z = np.log(h_start / (1 - h_start))  # Initial z values (inverse sigmoid)
    
    # Compute b such that sum(h) = n_points / 2 by binary search
    def compute_b(z, target_sum):
        def sum_h(b):
            return np.sum(1 / (1 + np.exp(-(z + b))))
        # Binary search for b
        low, high = -10, 10
        while high - low > 1e-10:
            mid = (low + high) / 2
            if sum_h(mid) < target_sum:
                low = mid
            else:
                high = mid
        return (low + high) / 2
    
    b = compute_b(z, n_points / 2)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Convert z and b to h
        h = 1 / (1 + np.exp(-(z + b)))
        
        # Find k that maximizes overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Compute gradient of max overlap w.r.t h
        # This is only the gradient for the specific shift k_max
        # The overlap is sum over i of h[i] * (1 - h[i + k])
        # dC_k/dh_i = (1 - h[i + k]) if i + k < n_points and 0 otherwise
        gradient = np.zeros_like(h)
        for i in range(n_points):
            j = i + max_k - (n_points - 1)  # shift in terms of correlation indexing
            if 0 <= j < n_points:
                gradient[i] = 1 - h[j]
        
        # Convert gradient to z space using chain rule
        # dh/dz = h(1-h) = h * (1-h)
        dh_dz = h * (1 - h)
        if np.any(dh_dz == 0):
            # Avoid division by zero
            dz = np.where(dh_dz > 0, -gradient / dh_dz, 0)
        else:
            dz = -gradient / dh_dz
        
        # Line search for optimal step size
        step = lr * 0.5
        while step > 1e-10:
            z_new = z + step * dz
            h_new = 1 / (1 + np.exp(-z_new))
            b_new = compute_b(z_new, n_points / 2)
            h_new = 1 / (1 + np.exp(-(z_new + b_new)))
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5 or step < 1e-8:
                z = z_new
                b = b_new
                c5_old = best_c5
                best_c5 = c5_new
                if c5_new < c5_old:
                    best_h = h_new.copy()
                break
            step *= 0.5
        
        # Adapt learning rate
        if iteration % 100 == 0:
            lr *= 0.998
    
    return (list(best_h), best_c5, n_points)
