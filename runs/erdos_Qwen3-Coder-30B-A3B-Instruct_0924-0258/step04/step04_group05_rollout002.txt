import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Ensure the initial solution meets the constraint sum(h) = n_points/2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid search with adaptive steps
    search_range = np.linspace(0.36, min(0.38085, best_c5), 50)
    patience = 0
    best_c5_grid = best_c5
    
    for c_mid in search_range:
        if time() - start_time > budget_s * 0.5:
            break
            
        # Check feasibility with convex optimization
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=500)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5_grid:
                    best_c5_grid = c5_feasible
                    best_h = h_feasible.copy()
                    patience = 0
                else:
                    patience += 1
                    if patience > 5:  # early stopping if no improvement
                        break
        except:
            continue
    
    # Gradient-based refinement with restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    max_restarts = 20
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbations
        if restart == 0:
            h_init = best_h.copy()
        else:
            # Apply noise and rescale to meet constraint
            noise = np.random.normal(0, 0.025, n_points)
            h_init = best_h + noise
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            
        try:
            # Try multiple methods for robustness
            methods = ['SLSQP', 'COBYLA']
            for method in methods:
                if time() - start_time > budget_s * 0.95:
                    break
                res = minimize(objective, h_init, method=method, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        if best_c5 < 0.3807:
                            max_restarts = restart + 1  # early stopping
                        break
        except:
            continue

    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            # Try with stronger tolerance
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
