import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with gradient refinement
    population_size = 50
    population = []
    population_fitness = []
    
    # Generate initial diverse population
    # Start with perturbed initial solution
    for i in range(10):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        population.append(h_perturb)
        population_fitness.append(evaluate_erdos_solution(h_perturb, dx))
    
    # Add some random solutions
    for i in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
        population_fitness.append(evaluate_erdos_solution(h_rand, dx))
    
    # Fill up rest randomly
    for i in range(population_size - 15):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
        population_fitness.append(evaluate_erdos_solution(h_rand, dx))
    
    # Best solution tracking
    best_idx = np.argmin(population_fitness)
    best_h = population[best_idx].copy()
    best_c5 = population_fitness[best_idx]
    
    # Genetic algorithm with local search
    # Parameters
    num_generations = 500
    mutation_rate = 0.15
    crossover_rate = 0.7
    elite_size = 5

    tabu_list = []

        if time() - start_time > budget_s * 0.95:

        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [population_fitness[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fit[0] < best_c5:
            best_h = sorted_pop[0].copy()
            best_c5 = sorted_fit[0]
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(sorted_pop[i].copy())
            new_fitness.append(sorted_fit[i])
            
        # Generate offspring
        while len(new_population) < population_size:
            # Selection (tournament)
            tournament_size = 3
            indices = np.random.choice(population_size, tournament_size)
            winner_idx = indices[np.argmin([population_fitness[i] for i in indices])]
            
            parent1 = population[winner_idx]

            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:

                winner_idx2 = indices[np.argmin([population_fitness[i] for i in indices])]
                parent2 = population[winner_idx2]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])

                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                new_fitness.append(evaluate_erdos_solution(child1, dx))
                
                if len(new_population) < population_size:
                    new_population.append(child2)
                    new_fitness.append(evaluate_erdos_solution(child2, dx))

                # Clone and mutate
                child = parent1.copy()

                    child[mutate_idx] = np.clip(child[mutate_idx] + np.random.normal(0, 0.05), 0, 1)
                    
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
                new_fitness.append(evaluate_erdos_solution(child, dx))

        population_fitness = new_fitness

        if gen % 10 == 0:

                    return evaluate_erdos_solution(h, dx)

                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_candidate, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = obj(h_refined)
                        if c5_refined < population_fitness[i]:
                            population[i] = h_refined
                            population_fitness[i] = c5_refined
                except:
                    pass
    
    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30 and best_c5 > 0.3807:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]

            res = minimize(obj, best_h, method='SLSQP',

                          options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = obj(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
