import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization with structured refinement
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    # Start with the best known solution
    swarm.append(h_start.copy())
    velocities.append(np.zeros(n_points))
    personal_best.append(h_start.copy())
    personal_best_vals.append(best_c5)
    
    # Generate random swarm
    for i in range(1, swarm_size):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(h.copy())
        personal_best_vals.append(evaluate_erdos_solution(h, dx))
    
    # Track global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (inertia * velocities[i] + 
                         cognitive * r1 * (personal_best[i] - swarm[i]) + 
                         social * r2 * (global_best - swarm[i]))
            velocities[i] = vel_update
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply bounds and normalization
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points/2) / np.sum(swarm[i])
            
            # Evaluate new position
            new_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if new_val < personal_best_vals[i]:
                personal_best_vals[i] = new_val
                personal_best[i] = swarm[i].copy()
                
                # Check if personal best is global best
                if new_val < global_best_val:
                    global_best_val = new_val
                    global_best = swarm[i].copy()
                    global_best_idx = i
        
        # Update best solution found
        if global_best_val < best_c5:
            best_c5 = global_best_val
            best_h = global_best.copy()
    
    # Convex relaxation refinement (subset of particles)
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Select a random particle from swarm
        idx = np.random.randint(0, len(swarm))
        h_ref = swarm[idx].copy()
        
        # Create a convex optimization problem
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2.0,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add constraints for multiple k values
        for k in range(0, min(10, 2 * n_points - 1), 2):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_vars[k-n_points+1:], 1 - h_vars[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= global_best_val * dx * 1.1)  # Soft constraint
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h_vars.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Multiple restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
