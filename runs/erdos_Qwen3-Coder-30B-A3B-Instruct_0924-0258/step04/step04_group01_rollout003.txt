import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create 3-cluster structure with different heights
        cluster_centers = [0.4, 1.0, 1.6]
        cluster_sizes = [3, 4, 3]
        cluster_heights = [0.3, 0.7, 0.4]
        for i, (c, size, h) in enumerate(zip(cluster_centers, cluster_sizes, cluster_heights)):
            start = max(0, int((c - size/2)/dx))
            end = min(n_points, int((c + size/2)/dx))
            h_start[start:end] = h
        # Normalize to sum to n_points / 2
        h_start *= (n_points / 2) / np.sum(h_start)
    
    # GA parameters
    pop_size = 20
    num_generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    elite_size = 2
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start.copy()
        if np.random.rand() < 0.7:  # Randomize some
            perturb = np.random.normal(0, 0.02, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
        # Normalize to sum constraint
        h *= (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Evolution loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0/n_points)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            
        # Create new population
        new_population = []
        
        # Elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection with tournament
            idx1 = np.random.randint(0, pop_size//2)
            idx2 = np.random.randint(0, pop_size//2)
            if fitness[idx1] < fitness[idx2]:
                parent1 = population[idx1]
            else:
                parent1 = population[idx2]
                
            idx3 = np.random.randint(0, pop_size//2)
            idx4 = np.random.randint(0, pop_size//2)
            if fitness[idx3] < fitness[idx4]:
                parent2 = population[idx3]
            else:
                parent2 = population[idx4]
                
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                
            # Normalize
            child *= (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:pop_size]
        
        # SQP refinement for best individuals
        for i in range(min(elite_size, len(population))):
            h = population[i].copy()
            
            def obj(h):
                return evaluate_erdos_solution(h, 2.0/n_points)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                    c5_refined = obj(refined_h)
                    if c5_refined < fitness[i]:
                        population[i] = refined_h
                        fitness[i] = c5_refined
                        if c5_refined < best_c5:
                            best_h = refined_h.copy()
                            best_c5 = c5_refined
            except:
                pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
