import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert to logit space for easier optimization
    def h_from_logits(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def project_to_sum_one(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Frank-Wolfe style optimization in logit space
    z = np.log(h_start / (1 - h_start))  # logit transform
    z = np.clip(z, -20, 20)  # clip to avoid overflow
    
    # Optimize using gradient descent with adaptive step size
    max_iter = 500
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current h and value
        h = h_from_logits(z)
        h = project_to_sum_one(h)
        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Compute gradient of overlap w.r.t. logits
        # We approximate it: ∂C/∂z ≈ ∂C/∂h * ∂h/∂z
        # ∂h/∂z = h * (1 - h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        # Compute approximate gradient based on the correlation structure
        grad_approx = np.zeros(n_points)
        for i in range(n_points):
            for j in range(n_points):
                k = j - i
                if 0 <= k < n_points:
                    grad_approx[i] += (1 - h[j]) * h[i + k] if i + k < n_points else 0
                    
        # Apply sigmoid derivative
        grad = grad_approx * h * (1 - h)
        
        # Update step in logit space
        lr = 0.1 * (1.0 - iter_num / max_iter)  # decay learning rate
        z_new = z - lr * grad
        z_new = np.clip(z_new, -20, 20)
        
        # Perturb occasionally to escape local minima
        if iter_num % 10 == 0 and iter_num > 0:
            z_new += np.random.normal(0, 0.02, n_points)
            z_new = np.clip(z_new, -20, 20)
        
        z = z_new
    
    # Final refinement with SLSQP if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        h_refine = best_h.copy()
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
