import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Strategy with Augmented Lagrangian
    pop_size = 50
    mutation_strength = 0.05
    elite_size = 10
    num_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Lagrangian multipliers
    lambda_sum = 0.0
    lambda_bounds = np.zeros(n_points)
    
    # Main optimization loop
    gen = 0
    last_improvement = 0
    while time() - start_time < budget_s * 0.95 and gen < num_generations:
        gen += 1
        
        # Evaluate population
        fitness = []
        for i, h in enumerate(population):
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append((c5, i))
        
        # Sort by fitness (lower is better)
        fitness.sort(key=lambda x: x[0])
        best_idx = fitness[0][1]
        
        if fitness[0][0] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[0][0]
            last_improvement = gen
        
        # Selection: keep top elite
        elite_indices = [idx for _, idx in fitness[:elite_size]]
        elite = [population[i] for i in elite_indices]
        
        # Mutation
        new_population = elite.copy()
        while len(new_population) < pop_size:
            parent = elite[np.random.randint(0, len(elite))]
            child = parent + np.random.normal(0, mutation_strength, n_points)
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Adjust mutation strength based on diversity and improvement
        if gen - last_improvement > 20:
            mutation_strength *= 1.1
        elif gen - last_improvement < 5 and mutation_strength > 0.001:
            mutation_strength *= 0.9
        
        # Early termination if no improvement
        if gen - last_improvement > 50:
            break
    
    # Final refinement: local optimization of best solution
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
