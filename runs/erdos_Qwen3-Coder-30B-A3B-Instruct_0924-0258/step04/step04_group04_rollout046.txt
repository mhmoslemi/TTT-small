import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.ndimage import gaussian_filter1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = h_start * (n_points/2) / np.sum(h_start)

    best_c5 = evaluate_erdos_solution(best_h, dx)

    max_iter = 100
    penalty_lr = 0.2
    smooth_sigma = 0.1

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points/2
