import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive Simulated Annealing with pairwise perturbations
    def neighbor(h, temperature):
        h_new = h.copy()
        # Select 2 different indices
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        
        # Compute adaptive step size
        step = 0.05 * temperature
        
        # Perturb in opposite directions
        delta = np.random.uniform(-step, step)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure bounds
        h_new[idx1] = np.clip(h_new[idx1], 0, 1)
        h_new[idx2] = np.clip(h_new[idx2], 0, 1)
        
        # Enforce sum constraint
        current_sum = np.sum(h_new)
        correction = (n_points / 2.0 - current_sum) / 2.0
        h_new[idx1] += correction
        h_new[idx2] += correction
        
        # Clamp back to bounds
        h_new[idx1] = np.clip(h_new[idx1], 0, 1)
        h_new[idx2] = np.clip(h_new[idx2], 0, 1)
        
        return h_new
    
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.999
    max_steps = 10000
    step_count = 0
    last_refinement = 0
    
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Generate neighbor
        new_h = neighbor(current_h, temperature)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            # Check if it's better than best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
        
        # Periodically refine with convex optimization
        if step_count - last_refinement > 500 and (time() - start_time) < budget_s * 0.9:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add constraints based on latest best solution
                # Only add stricter constraints if we have better solution
                local_c5 = evaluate_erdos_solution(current_h, dx)
                # Add some overlap constraints to make problem more constrained
                # Clip to safe values for constraints
                safe_c5 = min(local_c5, 0.5)
                
                # Add convex overlap constraints for k in reasonable range
                for k in range(0, min(15, 2 * n_points - 1)):
                    # This is for a subset, we'll refine just a few key indices
                    if k < n_points:
                        indices = np.arange(n_points - k)
                        corr_sum = cp.sum(cp.multiply(h[indices], 1 - h[indices + k]))
                    else:
                        indices = np.arange(2 * n_points - k - 1)
                        corr_sum = cp.sum(cp.multiply(h[indices], 1 - h[k - n_points + 1 + indices]))
                    # Add constraint: corr_sum <= safe_c5 * dx * n_points / 10
                    constraints.append(corr_sum <= safe_c5 * dx * 2)
                
                # Minimize the objective (which should be overridden by constraints)
                # Using a surplus objective of squared error to bring solution in line
                objective = cp.Minimize(cp.sum_squares(h - current_h) * 1e-3)
                prob = cp.Problem(objective, constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = np.array(h.value).flatten()
                        # Actually evaluate to check
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
                
            last_refinement = step_count
    
    # Final refinement using gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts
        for restart in range(6):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb the best solution a bit
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            # Re-adjust to sum constraint
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
