import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with constraint projection
    max_iterations = 5000
    alpha = 0.1
    alpha_decay = 0.99
    decay_interval = 100
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2.0) / h_sum
        return h
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Gradient descent loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps and find maximum
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient with respect to the most dominant overlap
        # The overlap at index k corresponds to shift k
        k = max_idx - (n_points - 1)  # k is in range [-n+1, n-1]
        if k >= 0:
            grad = np.zeros(n_points)
            grad[:n_points-k] += 1 - current_h[k:]
            grad[k:] += current_h[:n_points-k]
        else:
            grad = np.zeros(n_points)
            grad[-k:] += 1 - current_h[:n_points+k]
            grad[:n_points+k] += current_h[-k:]
        
        # Update with gradient descent
        new_h = current_h - alpha * grad
        new_h = project_h(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Decay learning rate
        if iteration % decay_interval == 0:
            alpha *= alpha_decay
            
        # Occasionally refine soln with convex optimization
        if iteration % 50 == 0 and (time() - start_time) < budget_s * 0.9:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                # Add constraints for top overlap shifts
                corr_current = np.correlate(current_h, 1 - current_h, mode='full')
                top_indices = np.argsort(corr_current)[-10:]
                
                for idx in top_indices:
                    k = idx - (n_points - 1)
                    if k >= 0:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
                    constraints.append(corr_sum <= max(corr_current[idx] * dx, 0.001))
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        if np.all(h_refined >= 0) and np.all(h_refined <= 1):
                            c5_refined = evaluate_erdos_solution(h_refined, dx)
                            if c5_refined < best_c5:
                                best_h = h_refined
                                best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        for restart in range(4):
            if time() - start_time > budget_s * 0.95:
                break
            
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
