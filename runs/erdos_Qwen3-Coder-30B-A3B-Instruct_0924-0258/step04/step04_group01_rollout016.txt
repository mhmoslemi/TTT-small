import numpy as np
from scipy.optimize import linprog

import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a new initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Set 2 peaks
        peak1_start = int(0.4 / dx)
        peak1_end = int(0.6 / dx)
        peak2_start = int(1.4 / dx)
        peak2_end = int(1.6 / dx)
        h_start[peak1_start:peak1_end] = 1.0
        h_start[peak2_start:peak2_end] = 1.0
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = float('inf')
    
    # Binary search parameters
    low = 0.0
    high = 0.38085
    precision = 1e-6
    max_iter = 60
    iter_count = 0
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2.0
        # Attempt to find feasible solution for this C
        feasible_h = solve_feasibility(h_start, mid, n_points, dx, start_time, budget_s)
        if feasible_h is not None:
            if best_c5 > mid:
                best_h = feasible_h
                best_c5 = mid
            high = mid
        else:
            low = mid
        iter_count += 1
        
    # Final validate the solution
    if best_c5 == float('inf'):
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)

def solve_feasibility(h_start, c5_target, n_points, dx, start_time, budget_s):
    # Use a simpler approach: build a convex relaxation for the overlap
    # We'll work with a simplified approximate approach using CVXPY
    
    h = cp.Variable(n_points)
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == n_points/2
    ]
    
    # For each k, form an approximation of the integral:
    # With a fixed small angle k in discrete space
    # Approximate the overlap as a linear program
    for shift in range(1, n_points):
        # For simplicity, we're using piece-wise linear constraints
        # and binary search, so we proxy the integral calculation
        if time() - start_time > budget_s * 0.9:
            return None
        
        # Solve a relaxed problem to find a candidate h that could satisfy this constraint
        # Note: this is a linearization. The actual overlap is non-linear.
        # But for this test we solve with constraints that give a feasible h
        try:
            # More detailed relaxation - for some selected shifts
            # Main objective is to create structure that bounds the max
            obj = cp.Minimize(cp.sum(h))  # We're trying to find a feasible h, not minimize
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                h_sol = np.maximum(0, np.minimum(1, h_sol))
                h_sol = h_sol * (n_points/2) / np.sum(h_sol)
                # Evaluate the result
                eval_result = evaluate_erdos_solution(h_sol, dx)
                if eval_result <= c5_target:
                    return h_sol
        except:
            continue
            
    # If nothing works, return initial guess if still feasible
    eval_result = evaluate_erdos_solution(h_start, dx)
    if eval_result <= c5_target:
        return h_start
        
    return None

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap
