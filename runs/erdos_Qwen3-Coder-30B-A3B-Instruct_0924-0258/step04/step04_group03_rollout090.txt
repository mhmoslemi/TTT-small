import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with projection
    def project_h(h):
        """Project h to satisfy sum=1 and bounds [0,1]"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_gradient(h, k):
        """Compute gradient of overlap for shift k"""
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k < len(h):
                grad[i] += (1 - h[i + k])
            if i >= k:
                grad[i] += h[i - k]
        return grad * dx
    
    # Main optimization loop
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    max_iter = 5000
    proj_count = 0
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find current maximum overlap and shift k
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_h = current_h.copy()
            best_c5 = max_overlap
            
        # Compute gradient for maximum overlap shift
        grad = compute_gradient(current_h, k_max)
        
        # Update with gradient descent
        current_h -= alpha * grad
        current_h = project_h(current_h)
        
        # Adaptive learning rate
        if iter_num % 100 == 0:
            alpha *= 0.99
        
        # Occasionally refine with convex optimization
        if iter_num % 50 == 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add constraint for top 10 overlaps to prevent overfitting
                corr_indices = np.argsort(corr)[-10:]
                for k in corr_indices:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= max_overlap * 1.1)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    h_refined = project_h(h_refined)
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        current_h = h_refined
                        current_c5 = c5_refined
            except:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try random restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = project_h(h_restart)
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
