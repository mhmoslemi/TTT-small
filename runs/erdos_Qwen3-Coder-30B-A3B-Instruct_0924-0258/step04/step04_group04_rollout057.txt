import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient-based optimization with line search
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps and find maximum
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient for the maximum overlap shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max >= 0 and i + k_max < n_points:
                grad[i] = (1 - best_h[i + k_max]) - best_h[i]
            elif i + k_max < 0:
                grad[i] = 1 - best_h[i]
            else:
                grad[i] = -best_h[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Line search with backtracking
        current_overlap = evaluate_erdos_solution(best_h, dx)
        alpha = step_size
        
        for _ in range(10):
            h_trial = best_h - alpha * grad
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (n_points/2) / np.sum(h_trial)
            trial_overlap = evaluate_erdos_solution(h_trial, dx)

        # Update
        best_h = best_h - alpha * grad
        best_h = np.clip(best_h, 0, 1)

            best_c5 = new_c5
            last_improvement = 0
        else:
            last_improvement += 1
            
        if last_improvement > 100:
            break

    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:

            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2

        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-6})

                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
