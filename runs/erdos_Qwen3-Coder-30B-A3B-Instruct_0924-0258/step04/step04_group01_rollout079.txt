import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a 3-cluster distribution
        cluster_width = 3
        centers = [0.5, 1.0, 1.5]
        for c in centers:
            start = max(0, int((c - cluster_width/2)/dx))
            end = min(n_points, int((c + cluster_width/2)/dx))
            h_start[start:end] = 0.3
        # Normalize sum
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 150
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Initial population
    population = [h_start.copy()]
    for i in range(population_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0/n_points)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select top half
        selected = population[:population_size//2]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                # Crossover
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
            else:
                # Clone or mutate
                parent = np.random.choice(selected)
                child = parent.copy()
                
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for i in mutate_indices:
                    child[i] = max(0, min(1, child[i] + np.random.normal(0, 0.02)))
                child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final refinement using SQP on best solution
    def obj(h):
        return evaluate_erdos_solution(h, 2.0/n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
