import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 1000
    mutation_rate = 0.1
    tournament_size = 2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(best_h.copy())
    
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Precompute indices for speed
    indices = list(range(n_points))
    
    # Main GA loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Find best in current generation
        best_in_gen = min(fitness)
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[fitness.index(best_in_gen)].copy()
        
        # Selection and reproduction
        new_population = []
        
        # Keep best individual (elitism)
        best_idx = fitness.index(min(fitness))
        new_population.append(population[best_idx].copy())
        
        # Tournament selection and reproduction
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = random.sample(range(population_size), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent1 = population[winner_idx]

            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent2 = population[winner_idx]
            
            # Crossover
            crossover_point = random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.01)
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final refinement with local search
    if time() - start_time < budget_s * 0.95:

            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            c5_refined = evaluate_erdos_solution(h_restart, dx)
            if c5_refined < best_c5:
                best_h = h_restart
                best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
