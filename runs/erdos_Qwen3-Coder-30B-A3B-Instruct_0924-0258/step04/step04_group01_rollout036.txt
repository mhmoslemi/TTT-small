import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # BOBYQA-like approach with trust region
    max_iter = 500
    trust_radius = 0.5
    min_trust_radius = 1e-5
    
    def objective(h):
        return evaluate_erdos_solution(h, 2.0/n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create a perturbed version
        perturbation = np.random.normal(0, trust_radius, n_points)
        h_candidate = current_h + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Local refinement using SLSQP
        try:
            res = minimize(objective, h_candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                if c5_refined < current_c5:
                    current_h = h_refined.copy()
                    current_c5 = c5_refined
            else:
                h_refined = h_candidate
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                if c5_refined < current_c5:
                    current_h = h_refined.copy()
                    current_c5 = c5_refined
        except:
            h_refined = h_candidate
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
            if c5_refined < current_c5:
                current_h = h_refined.copy()
                current_c5 = c5_refined
                
        # Update trust region
        if current_c5 < best_c5:
            trust_radius = min(1.0, trust_radius * 1.2)
        else:
            trust_radius *= 0.5
            if trust_radius < min_trust_radius:
                trust_radius = min_trust_radius
            
    # Final cleanup
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points)
    
    return (list(final_h), final_c5, n_points)
