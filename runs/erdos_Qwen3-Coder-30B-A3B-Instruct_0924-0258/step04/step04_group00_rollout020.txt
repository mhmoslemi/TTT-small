import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with Augmented Lagrangian
    max_iter = 1000
    alpha = 0.01  # Step size
    lambda_lag = 0.0  # Lagrange multiplier for sum constraint
    mu = 1.0  # Penalty parameter for sum constraint
    stagnation_count = 0
    max_stagnation = 50
    
    h = best_h.copy()
    h_prev = h.copy()
    c5_prev = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap values and gradients
        overlaps = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for i in range(n_points):
            # Compute gradient contribution from all shifts where index i appears in the first term
            contrib = 0
            # Loop through possible j (shifts)
            for k in range(1, n_points):
                if i + k < n_points:
                    contrib += (1 - h[i+k])  # Count how many times h[i] multiplies (1-h[i+k])
            # And also the opposite terms
            for k in range(1, n_points):
                if i - k >= 0:
                    contrib -= h[i-k]  # Count how many times h[i] multiplies (1-h[i-k])
            gradient[i] = contrib
        
        # Update with projected gradient descent
        h_new = h - alpha * gradient
        
        # Project onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            # Reset stagnation counter
            stagnation_count = 0
        else:
            # Reduce step size if no improvement
            alpha *= 0.5
            stagnation_count += 1
            h_new = h_prev.copy()  # Revert to previous step
            c5_new = c5_prev
            
        # Update variables for next iteration
        h = h_new
        h_prev = h_new.copy()
        c5_prev = c5_new
        
        # Early termination if no improvement
        if stagnation_count >= max_stagnation:
            break
        
        # Adaptive step size
        if c5_new < c5_prev:
            alpha *= 1.2
        else:
            alpha *= 0.5
    
    # Final smoothing
    h_smooth = np.convolve(h, np.ones(3)/3, mode='same')
    h_smooth = np.clip(h_smooth, 0, 1)
    h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
    c5_smooth = evaluate_erdos_solution(h_smooth, dx)
    
    if c5_smooth < best_c5:
        best_h = h_smooth
        best_c5 = c5_smooth
    
    return (list(best_h), best_c5, n_points)
