import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Adjust to same resolution and normalize
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Grid search in promising region
    grid_size = 100
    search_range = np.linspace(0.36, best_c5, grid_size)
    best_c5_grid = best_c5
    
    # Use a more adaptive grid search with CVXPY feasibility checks
    for i, c_mid in enumerate(search_range):
        if time() - start_time > budget_s * 0.45:
            break
            
        # Use a robust feasibility check via CVXPY
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraint for a maximum number of shifts
        max_shifts = 2 * n_points
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=3000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5_grid:
                    best_c5_grid = c5_feasible
                    best_h = h_feasible.copy()
        except:
            continue
    
    # Refinement using multiple restarts of gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multiple restarts
    n_restarts = 20
    for restart in range(n_restarts):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Start from better solutions or random perturbations
        if restart == 0:
            h_init = best_h.copy()
        else:
            # Random perturbation with different scales
            perturb_scale = np.random.uniform(0.01, 0.05)
            h_init = best_h + np.random.normal(0, perturb_scale, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    # Early stopping if we're close to target
                    if best_c5 < 0.3807:
                        break
        except:
            continue
    
    # Final fine-tuning
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
