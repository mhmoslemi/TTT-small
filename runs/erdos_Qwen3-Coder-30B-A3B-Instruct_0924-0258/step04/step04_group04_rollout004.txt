import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization
    max_iter = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbor solution
        h_new = best_h + np.random.normal(0, perturb_mag, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        current_c5 = evaluate_erdos_solution(best_h, dx)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept with probability based on temperature
        if new_c5 < current_c5 or np.random.rand() < math.exp(-(new_c5 - current_c5) / temp):
            best_h = h_new
            best_c5 = new_c5
            
        temp *= cool_rate
    
    # Upsample to 40 points
    n_points = 40
    dx = 2.0 / n_points
    h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    best_h = h_upsampled.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine grid optimization with SLSQP
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 10, 'ftol': 1e-6})
            if res.success:
                best_h = res.x
                best_c5 = evaluate_erdos_solution(best_h, dx)
        except:
            pass
    
    # Final refinement with gradient descent
    learning_rate = 0.01
    for _ in range(200):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        grad = np.zeros_like(best_h)
        
        # Compute gradient numerically
        for i in range(len(best_h)):
            h_plus = best_h.copy()
            h_plus[i] += 1e-4
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr)) / 1e-4
        
        # Update with gradient step and project
        best_h = best_h - learning_rate * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points/2) / np.sum(best_h)
        
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    # Final check
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
