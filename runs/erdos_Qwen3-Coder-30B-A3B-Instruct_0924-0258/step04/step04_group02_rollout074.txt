import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = 100000
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.2, 0.2)
        
        # Ensure we don't go out of bounds
        h_new = current_h.copy()
        if h_new[i] + delta > 1:
            delta = 1 - h_new[i]
        elif h_new[i] + delta < 0:
            delta = -h_new[i]
            
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure feasibility
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
            
        # Maintain the constraint sum(h) = n_points/2
        sum_diff = np.sum(h_new) - n_points / 2
        if abs(sum_diff) > 1e-10:  # Only adjust if needed
            # Adjust first point to maintain sum
            h_new[0] += sum_diff
            
            # Ensure bounds
            if h_new[0] < 0:
                h_new[0] = 0
            elif h_new[0] > 1:
                h_new[0] = 1
                
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = h_new
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool down
        T *= alpha

    return (list(best_h), best_c5, n_points)
