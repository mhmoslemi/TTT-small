import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with Augmented Lagrangian
    max_iter = 10000
    alpha = 0.01
    beta = 0.5
    gamma = 1.2
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 50
    
    # Keep track of last feasible solution
    last_feasible_h = best_h.copy()
    last_feasible_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient of overlap w.r.t h
        # We compute for all shifts and sum contributions
        grad = np.zeros(n_points)
        h = best_h.copy()
        
        # Compute correlation manually to get gradient
        # For each shift k, compute derivative of sum h[i]*(1-h[i+k]) w.r.t h[i]
        # This is 1 - h[i+k] - h[i+k] = 1 - 2*h[i+k] for the terms we're differentiating
        for k in range(-n_points+1, n_points):  # shifts from -n+1 to n-1
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i=0..n_points-k-1
                for i in range(n_points - k):
                    grad[i] += (1 - 2 * h[i + k])
            else:
                # h[i] * (1 - h[i+k]) for i=0..n_points+k-1 where k is negative
                for i in range(n_points + k):
                    grad[i + k] += (1 - 2 * h[i])
        
        # Apply gradient step: h_new = h - alpha * grad
        h_new = h - alpha * grad
        
        # Project onto [0,1] box
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            last_feasible_h = h_new.copy()
            last_feasible_c5 = c5_new
            alpha *= gamma
            stagnation_count = 0
        else:
            # Check if we should decrease step size or terminate
            alpha *= beta
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                break
    
    # Final validation/optimization step
    try:
        # Try to improve using local optimizer
        h_local = best_h.copy()
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points / 2) / np.sum(h_local)
        c5_local = evaluate_erdos_solution(h_local, dx)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
