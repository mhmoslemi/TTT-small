from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)
    else:

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with evolution and refinement
    population_size = 50
    mutation_rate = 0.05
    selection_rate = 0.2
    max_generations = 1000
    generations = 0
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        # Add random variations around the initial solution
        for _ in range(population_size - 1):
            perturbed = h_start + np.random.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    while (time() - start_time) < budget_s * 0.95 and generations < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c = evaluate_erdos_solution(h, dx)
            fitness.append(c)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: keep top 20%
        num_keep = int(population_size * selection_rate)
        selected = population[:num_keep]
        
        # Create new population
        new_population = selected.copy()
        
        # Mutate and crossover
        while len(new_population) < population_size:
            # Selection for crossover
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Crossover - arithmetic mean
            child = (parent1 + parent2) / 2.0
            
            # Mutation - add Gaussian noise
            noise = np.random.normal(0, mutation_rate, n_points)
            child = child + noise
            
            # Clamp and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        generations += 1
    
    # Refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        # Take top solutions and refine them
        top_candidates = population[:5]
        for candidate in top_candidates:
            if (time() - start_time) > budget_s * 0.95:
                break
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
