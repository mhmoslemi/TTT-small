import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization
    n_particles = 50
    n_iter = 1000
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Add the best known solution as one particle
    particles.append(h_start.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random perturbations
    for i in range(n_particles - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best positions
    p_best = [p.copy() for p in particles]
    p_best_values = [evaluate_erdos_solution(p, dx) for p in particles]
    
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    # PSO iterations
    for iteration in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Keep in bounds [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to satisfy sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_value
                
                # Update global best
                if current_value < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_value
                    
                    if g_best_value < best_c5:
                        best_h = g_best.copy()
                        best_c5 = g_best_value

    # Convex relaxation refinement
    for i in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            n = n_points
            h = cp.Variable(n)
            
            # Linearized overlap constraints
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add a few key overlap constraints for convexity
            # Interesting k values to avoid symmetry
            k_values = [1, 2, 5, 10]
            for k in k_values:
                if k < n:
                    # Add constraint that overlap at k is bounded
                    corr_sum = cp.sum(cp.multiply(h[:n-k], 1 - h[k:]))
                    constraints.append(corr_sum <= best_c5 * n_points * dx * 1.1)
                else:
                    # Handle symmetric case
                    corr_sum = cp.sum(cp.multiply(h[k-n+1:], 1 - h[:n-(k-n+1)]))
                    constraints.append(corr_sum <= best_c5 * n_points * dx * 1.1)
            
            # Objective: minimize maximum overlap (over all k)
            # Use big-M approach to avoid max constraints
            M = 1.0
            max_overlap = cp.Variable()
            constraints.append(max_overlap >= 0)
            
            # Add actual overlap computation
            for k in range(0, min(20, 2 * n - 1)):  # reasonable range
                if k < n:
                    corr_sum = cp.sum(cp.multiply(h[:n-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n+1:], 1 - h[:n-(k-n+1)]))
                constraints.append(corr_sum <= max_overlap)
            
            prob = cp.Problem(cp.Minimize(max_overlap * dx), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass

    # Final polish with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts with better starting point
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Start from best so far
            h_start = best_h + np.random.normal(0, 0.02, n_points)
            h_start = np.clip(h_start, 0, 1)
            h_start = h_start * (n_points/2) / np.sum(h_start)
            
            try:
                res = minimize(objective, h_start, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass

    return (list(best_h), best_c5, n_points)
