import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with Augmented Lagrangian
    max_iter = 1000
    alpha = 0.01
    lambda_ = 0.0
    mu = 1.0
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 50
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Gradient computation function
    def compute_gradient(h):
        n = len(h)
        grad = np.zeros(n)
        # For each shift k, compute contribution to gradient
        for k in range(1, n):
            if k <= n//2:
                # Forward shift
                for i in range(n-k):
                    grad[i] += (1 - h[i+k]) - h[i+k]
            else:
                # Backward shift
                for i in range(n-(n-k)):
                    grad[i+(n-k)] += (1 - h[i]) - h[i]
        
        # Add Lagrangian penalty for sum constraint
        sum_h = np.sum(h)
        grad += lambda_ + 2 * mu * (sum_h - n_points/2)
        return grad
    
    # Project function
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current gradient and update
        grad = compute_gradient(h_current)
        h_new = h_current - alpha * grad
        h_projected = project_h(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_projected, dx)
        
        # Line search: double alpha if successful, half if not
        if c5_new < c5_current:
            h_current = h_projected
            c5_current = c5_new
            alpha *= 1.2
            stagnation_count = 0
        else:
            alpha *= 0.5
            stagnation_count += 1
            
        # Update Lagrange multipliers based on constraint violation
        sum_h = np.sum(h_current)
        lambda_ += mu * (sum_h - n_points/2)
        
        # Update best solution so far
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
        
        if stagnation_count >= max_stagnation:
            break
    
    # Final refinement by smoothing with local averaging
    h_final = best_h.copy()
    smooth_factor = 0.1
    
    # Simple smooth and project
    h_smooth = np.convolve(h_final, np.ones(5)/5, mode='same')
    h_final = project_h(h_smooth * (1 - smooth_factor) + h_final * smooth_factor)
    
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
