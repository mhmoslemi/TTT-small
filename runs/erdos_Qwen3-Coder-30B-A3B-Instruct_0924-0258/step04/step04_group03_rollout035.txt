import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based evolutionary optimization
    population_size = 50
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_seed = np.array(initial_h_values)
    else:
        h_seed = np.ones(n_points) * 0.5
    
    # Initialize population
    population = []
    population.append(h_seed.copy())
    
    # Add diverse initial population
    for i in range(population_size - 1):
        pert = np.random.normal(0, 0.1, n_points)
        h = h_seed + pert
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = h_seed.copy()
    best_c5 = evaluate_erdos_solution(h_seed, dx)
    
    # Parameters for evolution
    mutation_step = 0.05
    min_mutation = 0.001
    decay_rate = 0.999
    
    max_generations = 2000
    selection_rate = 0.2
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all members
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection - keep top 20%
        keep_count = int(population_size * selection_rate)
        selected = population[:keep_count]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover two random members
            idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
            parent1 = selected[idx1]
            parent2 = selected[idx2]
            
            # Arithmetic crossover
            alpha = np.random.random()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            if np.random.random() < 0.8:  # 80% chance to mutate
                noise = np.random.normal(0, mutation_step, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Adaptive mutation step
        mutation_step = max(min_mutation, mutation_step * decay_rate)
    
    # Final gradient refinement of best solutions
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        top_candidates = []
        for _ in range(5):
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            top_candidates.append(h)
        
        for h_candidate in top_candidates:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
