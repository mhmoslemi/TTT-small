import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    max_time = budget_s
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    max_stagnation = 50
    stagnation_count = 0
    
    # Tabu list to store recent solutions (hash-based)
    tabu_list = set()
    tabu_size = tabu_limit
    
    # Initialize population
    population = []
    fitness = []
    
    # Create population by perturbing the initial solution and adding random ones
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
        
    # Add some random solutions
    for _ in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fill to population size
    while len(population) < population_size:
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Evaluate initial population
    for h in population:
        fit = evaluate_erdos_solution(h, dx)
        fitness.append(fit)
    
    # Main GA loop
    gen = 0
    while time() - start_time < max_time * 0.95 and stagnation_count < max_stagnation:
        gen += 1
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_current = fitness[sorted_indices[0]]
        
        if best_current < best_c5:
            best_c5 = best_current
            best_h = population[sorted_indices[0]].copy()
            
        # Check for stagnation
        if gen > 1 and best_current >= np.min(fitness[:5]) * 0.9999:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        # Update parameters
        if stagnation_count > 30:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_size = max(10, tabu_size // 2)
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Selection (tournament)
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_size = 3
            selected_indices = np.random.choice(population_size, tournament_size)
            best_in_tourn = min(selected_indices, key=lambda i: fitness[i])
            parent = population[best_in_tourn].copy()
            
            # Crossover and mutation
            if random.random() < crossover_prob and len(new_population) < population_size - 1:
                parent2_idx = random.choice(selected_indices)
                parent2 = population[parent2_idx]
                
                # Single point crossover
                crossover_point = random.randint(1, n_points-1)
                child = np.concatenate([parent[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent.copy()
                
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.02)
            # Repair and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            # Check tabu
            child_tuple = tuple(child.round(8))
            if child_tuple in tabu_list:
                # Modify slightly to avoid tabu
                child += np.random.normal(0, 0.001, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                child_tuple = tuple(child.round(8))
                
            new_population.append(child)
            tabu_list.add(child_tuple)
            if len(tabu_list) > tabu_size:
                # Remove oldest entries
                tabu_list.pop()
        
        # Local refinement every 10 generations with SLSQP
        if gen % 10 == 0:
            for i in range(min(5, len(new_population))):
                h_refined = new_population[i]
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                def con(h):
                    return np.sum(h) - n_points / 2.0
                
                cons = {'type': 'eq', 'fun': con}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(obj, h_refined, method='SLSQP', bounds=bounds, 
                                   constraints=cons, options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_new = res.x
                        f_new = obj(h_new)
                        if f_new < fitness[i]:
                            new_population[i] = h_new
                            fitness[i] = f_new
                except:
                    pass
                    
        # Update poplulation
        population = new_population
        # Recalculate fitness
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Early stopping if target is hit
        if best_c5 < 0.3807:
            break
    
    # Final local refinement of best solution using tight tolerance
    remaining_time = max_time - (time() - start_time)
    
    if remaining_time > 30:
        h_init = best_h.copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
