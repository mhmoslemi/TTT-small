import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Strategy with gradient refinement
    population_size = 50
    mutation_step = 0.05
    selection_rate = 0.2
    n_selected = int(population_size * selection_rate)
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Start with the initial solution and perturbations
        population.append(h_start)
        for _ in range(population_size - 1):
            pert = h_start + np.random.normal(0, mutation_step, n_points) * 0.5
            pert = np.clip(pert, 0, 1)
            pert *= (n_points / 2) / np.sum(pert)
            population.append(pert)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / np.sum(h)
            population.append(h)
    
    iteration = 0
    while (time() - start_time) < budget_s * 0.95:
        # Evaluate the whole population
        fitness = []
        for h in population:
            c = evaluate_erdos_solution(h, dx)
            fitness.append(c)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_fitness = fitness[sorted_indices[0]]
        
        if best_fitness < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = best_fitness
        
        # Selection
        selected = [population[i] for i in sorted_indices[:n_selected]]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            p1 = selected[np.random.randint(0, len(selected))]
            p2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover: blend
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            
            # Mutation
            mutation = np.random.normal(0, mutation_step, n_points)
            child += mutation
            child = np.clip(child, 0, 1)
            child *= (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        # Update population
        population = new_population
        
        # Adaptive mutation step
        if iteration % 10 == 0 and mutation_step > 0.001:
            mutation_step *= 0.99
        
        iteration += 1
    
    # Final gradient refinement on top performers
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        top_performers = []
        top_fitness = []
        
        for h in population:
            c = evaluate_erdos_solution(h, dx)
            top_performers.append(h)
            top_fitness.append(c)
        
        # Sort top performers
        sorted_indices = np.argsort(top_fitness)
        top_performers = [top_performers[i] for i in sorted_indices[:5]]
        
        for h in top_performers:
            try:
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                res = minimize(objective, h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    return (list(best_h), best_c5, n_points)
