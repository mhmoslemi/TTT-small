import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a balanced initial guess
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Constraint: sum(h) = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points/2
    
    # Optimization using CVXPY with alternating dual updates
    # Initialize dual variables
    n_shifts = 2 * n_points - 1
    dual_vars = np.ones(n_shifts)
    
    # Inner loop iterations
    max_iter = 100
    dual_update_prob = 0.8
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # 1. Compute current overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_overlaps = corr * dx
        
        # 2. Update dual variables
        if np.random.rand() < dual_update_prob:
            # Use a projection-based update
            dual_vars = np.maximum(dual_vars * (current_overlaps / (dual_vars + 1e-12)), 1e-6)
        
        # 3. Convex relaxation: minimize weighted sum of overlaps
        # Define optimization variables
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Use duals to weight the sum
        cost = 0
        for i, (k, dual_val) in enumerate(zip(range(-n_points+1, n_points), dual_vars)):
            if i < n_shifts:
                # Approximate the overlapping term
                cost += dual_val * cp.sum(cp.multiply(h_var, 1 - h_var))
        
        # Create and solve the problem
        prob = cp.Problem(cp.Minimize(cost), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.maximum(np.minimum(h_new, 1.0), 0.0)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    new_c5 = evaluate_erdos_solution(h_new, dx)
                    if new_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = new_c5
        except:
            pass
        
        # Optional: local refinement with scipy
        if np.random.rand() < 0.3:  # Small chance of local refinement
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 20, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                    refined_h = np.clip(refined_h, 0, 1)
                    refined_h = refined_h * (n_points/2) / np.sum(refined_h)
                    refined_c5 = evaluate_erdos_solution(refined_h, dx)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
            except:
                pass
    
    # Final local refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_h = np.clip(final_h, 0, 1)
                final_h = final_h * (n_points/2) / np.sum(final_h)
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
