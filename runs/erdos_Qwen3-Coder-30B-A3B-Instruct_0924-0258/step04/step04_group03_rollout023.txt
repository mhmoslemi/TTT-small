import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization with Convex Refinement
    def pso_optimization():
        n_particles = 50
        n_iter = 1000
        omega = 0.7
        c1 = 2.0
        c2 = 2.0
        v_max = 0.1
        
        # Initialize particles
        particles = []
        velocities = []
        for _ in range(n_particles):
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
        
        # Personal bests
        pbest = [p.copy() for p in particles]
        pbest_scores = [evaluate_erdos_solution(p, dx) for p in particles]
        
        # Global best
        gbest_idx = np.argmin(pbest_scores)
        gbest = pbest[gbest_idx].copy()
        gbest_score = pbest_scores[gbest_idx]
        
        for _ in range(n_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            for i in range(n_particles):
                # Update velocity
                r1, r2 = np.random.rand(2)
                velocities[i] = (omega * velocities[i] + 
                                c1 * r1 * (pbest[i] - particles[i]) + 
                                c2 * r2 * (gbest - particles[i]))
                
                # Clip velocity
                velocities[i] = np.clip(velocities[i], -v_max, v_max)
                
                # Update position
                particles[i] += velocities[i]
                
                # Ensure bounds and normalization
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                
                # Evaluate and update personal best
                score = evaluate_erdos_solution(particles[i], dx)
                if score < pbest_scores[i]:
                    pbest[i] = particles[i].copy()
                    pbest_scores[i] = score
                    if score < gbest_score:
                        gbest = particles[i].copy()
                        gbest_score = score
            
            # Early stopping based on time limit
            if time() - start_time > budget_s * 0.95:
                break
                
        return gbest, gbest_score
    
    # Run PSO
    try:
        pso_h, pso_c5 = pso_optimization()
        if pso_c5 < best_c5:
            best_h = pso_h
            best_c5 = pso_c5
    except Exception:
        pass
    
    # Convex refinement (only if better solution found)
    if best_c5 < 0.4:  # Only refine if we have room for improvement
        try:
            # Create a convex approximation
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            # Add linear constraints based on current best solution
            # We approximate the correlation as a linear function around the current solution
            # This is a coarse linearization of the problem to reduce complexity
            
            # Simplified convex relaxation
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            max_idx = np.argmax(corr)
            max_corr = corr[max_idx]
            
            # Add a linear constraint at the maximum overlap point
            # This is a heuristic approach to create a better convex relaxation
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final polishing with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        # Start from the best solution and do fine-tuning
        h_init = best_h.copy()
        
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Run multiple restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Perturb a bit
            h_perturbed = h_init + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            
            try:
                res = minimize(objective, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
