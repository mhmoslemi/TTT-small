import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert to logit space for optimization
    # h = sigmoid(z + b) where b ensures sum(h) = n_points/2
    def h_from_z(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def compute_b(z):
        h = h_from_z(z)
        return np.log((n_points / 2) / np.sum(h) - 1.0) if np.sum(h) > 0 else 0.0
    
    # Initialize
    z = np.log(h_start / (1.0 - h_start)) if np.all(h_start > 0) and np.all(h_start < 1) else np.zeros(n_points)
    b = compute_b(z)
    h = h_from_z(z + b)
    
    # Optimize using gradient descent
    max_iter = 1000
    lr = 0.05
    tolerance = 1e-8
    decay_rate = 0.95
    
    for iteration in range(max_iter):
        # Check time limit
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current h
        h = h_from_z(z + b)
        
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
            
        # Compute gradient of overlap w.r.t. h for the worst shift k_max
        grad = np.zeros(n_points)
        
        # For each position in the correlation
        for i in range(n_points):
            # The term at i (in context of overlap with shifted version)
            # For a shift k_max, derivatives are an alternating pattern
            # Index in correlation is k_max + i (mod n_points) for linearity
            if i + k_max < n_points:
                grad[i] = 1.0 - h[i + k_max]
            else:
                # wrap around case for full correlation - kinetic formulation
                grad[i] = 1.0 - h[i + k_max - n_points]
                
        # Compute gradient w.r.t. z (chain rule)
        # h' = h * (1 - h)
        h_prime = h * (1.0 - h)
        dh_dz = h_prime
        
        # Gradient in z space
        grad_z = np.zeros(n_points)
        for i in range(n_points):
            # Compute derivative of current overlap term w.r.t. h[i]
            # It is proportional to (1 - h[i+kmax]) - if computation systemic...
            # More carefully:
            # C_k = sum_i h[i] (1 - h[i + k])
            # dC_k / dh_i = (1 - h[i + k])
            k = k_max
            j = (i + k) % n_points
            grad_z[i] = dh_dz[i] * (1.0 - h[j])  # Chain rule
            
        # Apply gradient step 
        z = z - lr * grad_z
        b = compute_b(z)
        
        # Reduce learning rate if we don't make progress
        if iteration % 100 == 0:
            lr *= decay_rate
            
        # Apply bounds for stability
        z = np.clip(z, -10, 10)
        
    # Final refinement with SLSQP
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
        
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    # Refinement
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    for restart in range(3):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            result = minimize(objective, h_restart, method='SLSQP', 
                              bounds=bounds, constraints=cons_sum, 
                              options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                h_refined = result.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
