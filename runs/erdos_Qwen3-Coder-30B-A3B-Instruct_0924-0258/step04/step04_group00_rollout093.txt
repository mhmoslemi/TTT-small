import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use direct optimization approach with different strategies
    max_restarts = 50
    restart_count = 0
    local_search_steps = 200
    
    # Binary relaxation approach
    def binary_relaxation(h):
        return np.round(h).astype(int)
    
    def project_to_feasible(h):
        # Ensure sum = n_points/2 and all values in [0,1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Optimization using direct search with refinement
    while time() - start_time < budget_s * 0.8:
        # Local perturbation around best solution
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = project_to_feasible(h_perturbed)
        
        # SLSQP local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                evaluate_erdos_solution,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            if res.success:
                h_refined = project_to_feasible(res.x)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                else:
                    restart_count += 1
            else:
                restart_count += 1

            restart_count += 1
        
        if restart_count > max_restarts:
            break
    
    # Additional refinement with projected gradient steps
    h_refined = best_h.copy()
    for step in range(local_search_steps):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Apply small random perturbations to improve
        perturbation = np.random.normal(0, 0.005, n_points)
        h_test = h_refined + perturbation
        h_test = project_to_feasible(h_test)
        
        c5_test = evaluate_erdos_solution(h_test, dx)
        if c5_test < best_c5:
            h_refined = h_test.copy()
            best_c5 = c5_test
    
    # Final validation
    h_final = project_to_feasible(h_refined)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5

    return (list(best_h), best_c5, n_points)
