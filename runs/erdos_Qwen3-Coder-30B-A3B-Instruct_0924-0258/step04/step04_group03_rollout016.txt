import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient descent with adaptive learning rate
    def compute_gradient(h, k):
        # compute gradient of overlap integral for shift k
        n = len(h)
        grad = np.zeros(n)
        # for convolution with 1-h: overlap with shifted version of 1-h
        # this means at index i of h, we look at (1-h)[i+k] or (1-h)[i-k]  
        # Since we are differentiating h[i], we consider where h[i] appears in the correlation
        # that is where 1-h[i+k] is expected to be (for overlap at shift k)
        # gradient w.r.t h[i] is contribution of that 1-h[i+k] and 1-h[i-k] terms
        if k < n:
            grad[:n-k] += 1 - h[k:]
        else:
            # right side, compute shift from right
            grad[k-n+1:] += 1 - h[:n-(k-n+1)]
        return grad
    
    # Initialize
    h_current = h_start.copy()
    h_current *= (n_points / 2) / np.sum(h_current)  # Ensure sum constraint
    
    # ADAM optimizer setup
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    learning_rate = 0.1
    t = 0
    
    # Track best solution
    best_c5 = evaluate_erdos_solution(h_current, dx)
    best_h = h_current.copy()
    
    max_iter = 5000
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find the worst shift
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_best = np.argmax(corr)
        max_overlap = corr[k_best] * dx
        
        # Compute gradient for this shift
        gradient = compute_gradient(h_current, k_best)
        
        # ADAM updates
        t += 1
        m = beta1 * m + (1 - beta1) * gradient
        v = beta2 * v + (1 - beta2) * (gradient ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Update h
        h_new = h_current - learning_rate * m_hat / (np.sqrt(v_hat) + eps)
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to sum constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        h_current = h_new
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_current, dx)
        
        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_current.copy()
            
        # Adaptive learning rate
        if iter % 100 == 0:
            learning_rate *= 0.99
            if learning_rate < 1e-5:
                learning_rate = 1e-5
                
        # Occasionally add small noise to escape local minima
        if iter % 200 == 0 and iter > 0:
            h_current += np.random.normal(0, 0.005, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current *= (n_points / 2) / np.sum(h_current)
    
    # Convex Optimization Refinement
    if time() - start_time < budget_s * 0.9:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Get top constraints (based on observed worst overlap)
            corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
            top_k_indices = np.argsort(-corr_vals)[:min(10, len(corr_vals))]
            for k_idx in top_k_indices:
                k = k_idx - (n_points - 1)
                corr_sum = 0
                if k >= 0:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
                constraints.append(corr_sum <= best_c5 * n_points / 2)  # Approximate constraint
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20 and time() - start_time < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Perturb h slightly for better convergence
        perturbed_h = best_h + np.random.normal(0, 0.01, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points/2) / np.sum(perturbed_h)
        
        try:
            res = minimize(objective, perturbed_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
