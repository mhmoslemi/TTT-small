import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Logistic parameterization: h = sigmoid(z + b)
    # To maintain sum(h) = n_points/2, we compute b
    def compute_b(z):
        h = 1.0 / (1.0 + np.exp(-z))
        return np.log((n_points/2) / np.sum(h) * (1 - np.sum(h) / n_points))
    
    def sigmoid(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    # Initialize z and b
    z = np.log(h_start / (1 - h_start))
    b = compute_b(z)
    
    # Gradient-based optimization
    max_iterations = 500
    lr = 0.001
    tolerance = 1e-6
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Propagate z to h and compute b
        h = sigmoid(z + b)
        h = np.clip(h, 1e-6, 1 - 1e-6)  # Safety clamps
        
        # Update b to satisfy sum(h) = n_points/2
        b = compute_b(z)
        
        # Compute all overlaps and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Compute gradient of max_overlap w.r.t. h
        # For the k-th shift, derivative of int h(x)(1-h(x+k)) dx
        # is 1 - 2*h(x) at x, and -h(x-k) at x-k
        # So gradient is 1 - 2*h - shift(0, h)
        # But we're looking for the shift that maximizes overlap
        # For the specific k_max@max_idx:
        gradient = np.zeros_like(h)
        k_max = max_idx - (len(h) - 1)
        # For the x-k case
        if k_max <= 0:
            gradient[-k_max:] = -(1 - h[:-k_max])
        else:
            gradient[:-k_max] = -(1 - h[k_max:])
        # Add local term  
        gradient = 1 - 2 * h - gradient
        
        # Convert gradient to z-space
        # dh/dz = sigmoid'(z+b) = h * (1 - h)
        dh_dz = h * (1 - h)
        dz = gradient / dh_dz
        
        # Update parameters
        z = z - lr * dz
        
        # Refine b after update
        b = compute_b(z)
        
        # Evaluate new h and update best
        h_new = sigmoid(z + b)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # Backtracking line search
        # Try smaller steps if necessary
        if iteration % 20 == 0 and lr > 1e-6:
            lr *= 0.9
    
    # Final refinement with scipy
    if time() - start_time < budget_s * 0.9:
        def objective(h_vec):
            h = np.array(h_vec)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try with best result
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
