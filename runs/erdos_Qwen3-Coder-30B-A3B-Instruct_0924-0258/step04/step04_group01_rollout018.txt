from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        # Centered spike at 0.5 with width 0.5
        start = int(0.25/dx)
        end = int(0.75/dx)
        h_start[start:end] = 1.0 / (0.5/dx)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Binary search approach for tighter bound
    low, high = 0.0, 0.38085
    precision = 1e-6
    iterations = 0
    max_iterations = 60
    
    while time() - start_time < budget_s * 0.95 and iterations < max_iterations and high - low > precision:
        mid = (low + high) / 2.0
        
        # Try to find a feasible solution with this C5 bound
        # We'll do a simple gradient-based optimization starting from current best
        def objective(h):
            return evaluate_erdos_solution(h, 2.0/n_points)
        
        # Simple local optimization around current solution
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Vary perturbation scale and number of attempts to improve robustness
        attempts = 3
        for attempt in range(attempts):
            if time() - start_time > budget_s * 0.95:
                break
            # Start from best solution or a small random perturbation
            h_init = best_h + np.random.normal(0, 0.01, n_points) if attempt > 0 else best_h
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            
            try:
                res = minimize(objective, h_init, method='SLSQP',

                               options={'maxiter': 200, 'ftol': 1e-6})

                    h_candidate = res.x
                    c5_candidate = objective(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
        
        # Use current best to check feasibility for mid
        current_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
        
        if current_c5 <= mid:
            high = mid
        else:
            low = mid
            
        iterations += 1
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
