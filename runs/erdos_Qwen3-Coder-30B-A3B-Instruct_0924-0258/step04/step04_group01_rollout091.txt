import numpy as np
from cvxpy import *
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a piecewise constant construction
        h_start[20:40] = 0.5
        h_start[60:80] = 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    # Binary search on C5
    low = 0.0
    high = 0.38085
    precision = 1e-6
    max_iterations = 60
    best_feasible_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2.0
        
        # Check if mid is feasible via LP
        feasible, h_feasible = check_feasibility(h_start, mid, n_points, 2.0/n_points)
        
        if feasible:
            high = mid
            if evaluate_erdos_solution(h_feasible, 2.0/n_points) < best_c5:
                best_c5 = evaluate_erdos_solution(h_feasible, 2.0/n_points)
                best_feasible_h = h_feasible.copy()
        else:
            low = mid
            
        if abs(high - low) < precision:
            break
    
    # Final check
    if not check_feasibility(best_feasible_h, high, n_points, 2.0/n_points)[0]:
        # Re-evaluate with more precise value
        final_h = best_feasible_h.copy()
        final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points)
    else:
        final_h = best_feasible_h.copy()
        final_c5 = high
        
    # Final normalization check
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, 2.0/n_points)
    
    return (list(final_h), final_c5, n_points)

def check_feasibility(h_initial, c5_bound, n_points, dx):
    # Variable to be optimized
    h = Variable(n_points)
    
    # Sum constraint
    constraints = [sum(h) == n_points/2.0]
    
    # Bound constraints
    constraints += [h[i] >= 0 for i in range(n_points)]
    constraints += [h[i] <= 1 for i in range(n_points)]
    
    # Overlap constraints - simplified by linearizing part of overlap
    # Quick approximate: only consider a few key lags
    for lag in [0, 1, 2, 3, 4, 5]:
        # Create correlation approximation
        if lag == 0:
            # h(x) * (1 - h(x)) 
            corr = 0
            for i in range(n_points):
                corr += h[i] * (1 - h[i]) * dx
            constraints += [corr <= c5_bound]
        else:
            # Approximate correlation for the sample pick
            if lag < n_points:
                corr = 0
                for i in range(n_points - lag):
                    corr += h[i] * (1 - h[i + lag]) * dx
                constraints += [corr <= c5_bound]
    
    # Solve the optimization problem
    objective = Minimize(0)  # Just checking feasibility
    problem = Problem(objective, constraints)
    
    try:
        problem.solve(solver=SCIPY, verbose=False)
        if problem.status in ["optimal", "optimal_inaccurate"]:
            return True, np.array([h.value[i] for i in range(n_points)])
    except:
        pass
        
    return False, h_initial
