import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_tenure = 50
    tabu_list = []
    stagnation_count = 0
    max_stagnation = 50
    
    # GA parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Initialize population with diverse solutions
    population = []
    fitness = []
    
    # Start with initial solution and perturbations
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random solutions
    for _ in range(30):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
    
    # Evaluate initial population
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # Main GA loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        if gen % 10 == 0:
            # Local refinement with SLSQP
            top_indices = np.argsort(fitness)[:5]
            for idx in top_indices:
                h_refine = population[idx].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                cons = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                try:
                    res = minimize(objective, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < fitness[idx]:
                            population[idx] = h_refined
                            fitness[idx] = c5_refined
                except:
                    pass
        
        # Selection - tournament selection
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            if i + 1 < population_size and np.random.rand() < crossover_prob:
                parent1 = selected[i]
                parent2 = selected[i+1]
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
            else:
                child1 = selected[i].copy()
                child2 = selected[i+1].copy() if i + 1 < population_size else selected[i].copy()
            
            # Apply mutation
            mutation_mask = np.random.rand(n_points) < mutation_rate
            if np.any(mutation_mask):
                child1[mutation_mask] = np.clip(child1[mutation_mask] + 
                                               np.random.normal(0, 0.03, np.sum(mutation_mask)), 
                                               0, 1)
                child1 = child1 * (n_points/2) / np.sum(child1)
            
            if np.any(mutation_mask) and i + 1 < population_size:
                child2[mutation_mask] = np.clip(child2[mutation_mask] + 
                                               np.random.normal(0, 0.03, np.sum(mutation_mask)), 
                                               0, 1)
                child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Trim population if too large
        population = new_population[:population_size]
        
        # Evaluate fitness of new population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness[best_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check for improvement and update adaptive parameters
        if stagnation_count > 0 and stagnation_count % 20 == 0:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_tenure = max(10, tabu_tenure // 2)
        
        if stagnation_count > max_stagnation:
            break
    
    # Final local refinement on best solution
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, best_h.copy(), method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except:
        pass
    
    # Validation
    if not (all(0 <= val <= 1 for val in best_h) and 
            abs(np.sum(best_h) - n_points/2) < 1e-6):
        # Revert to safe solution if invalid
        best_h = h_start.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
