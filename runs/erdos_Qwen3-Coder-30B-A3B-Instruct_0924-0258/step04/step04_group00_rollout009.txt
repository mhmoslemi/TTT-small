import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def make_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = make_feasible(h_start, n_points)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 5000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 2
    
    # Initialize population
    population = [make_feasible(np.random.rand(n_points), n_points) for _ in range(population_size - 1)]
    population.append(best_h)
    
    # Evaluate initial population
    fitness = [evaluate_erdos_solution(individual, dx) for individual in population]
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_index])
        
        # Create offspring
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
            
            # Mutation
            for offspring in [offspring1, offspring2]:
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[j] += np.random.normal(0, 0.01)
                offspring = make_feasible(offspring, n_points)
                new_population.append(offspring)
        
        # Replace worst individuals
        new_fitness = [evaluate_erdos_solution(ind, dx) for ind in new_population]
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Sort by fitness and keep best
        sorted_indices = np.argsort(combined_fitness)
        population = [combined_population[i] for i in sorted_indices[:population_size]]
        fitness = [combined_fitness[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
    
    return (list(best_h), best_c5, n_points)
