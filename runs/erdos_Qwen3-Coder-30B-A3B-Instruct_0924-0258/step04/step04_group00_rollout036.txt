import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ES parameters
    pop_size = 50
    num_generations = 1000
    mutation_strength = 0.1
    elite_size = 5
    tournament_size = 5
    
    # Population initialization
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # ES loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            best_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[best_idx].copy())
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        elite = [population[i] for i in elite_indices]
        selected[:elite_size] = elite
        
        # Mutation and recombination
        new_population = []
        for i in range(pop_size):
            parent = selected[i]
            # Simple mutation with Gaussian noise
            mutated = parent + np.random.normal(0, mutation_strength, n_points)
            mutated = np.clip(mutated, 0, 1)
            # Normalize to correct sum
            mutated = mutated * (n_points/2) / np.sum(mutated)
            new_population.append(mutated)
        
        population = new_population
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Adapt mutation strength
        if gen % 10 == 0:
            diversity = np.std([np.sum((p - best_h)**2) for p in population])
            if diversity < 0.01:
                mutation_strength *= 1.1
            else:
                mutation_strength *= 0.95
    
    # Final refinement with optimization
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
