import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Transform to Fourier coefficients
    def h_from_coeffs(coeffs, n_points):
        # Reconstruct h from coefficients using sigmoid activation
        k = np.arange(1, len(coeffs)//2 + 1)
        x = np.linspace(0, 2, n_points, endpoint=False)
        # Fourier reconstruction with DC term
        h_reconstructed = coeffs[0] + np.sum(
            coeffs[1::2][:len(k)] * np.cos(k * np.pi * x[:, None]) +
            coeffs[2::2][:len(k)] * np.sin(k * np.pi * x[:, None]),
            axis=1
        )
        # Apply sigmoid to ensure range [0,1]
        h = 1.0 / (1.0 + np.exp(-10 * h_reconstructed))  # slope = 10
        return h
    
    def normalize_h(h, n_points):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Initial coefficients
    coeffs_start = np.random.normal(0, 0.1, 2 * n_points - 2)  # 2*n-2 coefficients for n points
    coeffs_best = coeffs_start.copy()
    coeffs = coeffs_start.copy()
    
    # Objective function for optimization
    def objective(coeffs):
        h = h_from_coeffs(coeffs, n_points)
        h_scaled = normalize_h(h, n_points)
        c5 = evaluate_erdos_solution(h_scaled, dx)
        return c5
    
    # Optimization loop with trust-region method
    max_iter = 300
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        try:
            # Evaluate current point
            c5_current = objective(coeffs)
            if c5_current < best_c5:
                best_c5 = c5_current
                coeffs_best = coeffs.copy()
            
            # Compute approximate gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(coeffs)
            for i in range(len(coeffs)):
                coeffs_plus = coeffs.copy()
                coeffs_minus = coeffs.copy()
                coeffs_plus[i] += eps
                coeffs_minus[i] -= eps
                c5_plus = objective(coeffs_plus)
                c5_minus = objective(coeffs_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with step size
            lr = 0.5
            coeffs_new = coeffs - lr * grad
            c5_new = objective(coeffs_new)
            
            # Accept new point if improvement
            if c5_new < c5_current:
                coeffs = coeffs_new
            else:
                # Try smaller step if no improvement
                lr /= 2
                coeffs_new = coeffs - lr * grad
                c5_new = objective(coeffs_new)
                if c5_new < c5_current:
                    coeffs = coeffs_new
        except Exception as e:
            # Fall back to restart with random perturbation
            coeffs = coeffs_best + np.random.normal(0, 0.01, len(coeffs))
            if time() - start_time > budget_s * 0.95:
                break
    
    # Final refinement
    h_final = h_from_coeffs(coeffs_best, n_points)
    h_final = normalize_h(h_final, n_points)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final
    
    return (list(best_h), best_c5, n_points)
