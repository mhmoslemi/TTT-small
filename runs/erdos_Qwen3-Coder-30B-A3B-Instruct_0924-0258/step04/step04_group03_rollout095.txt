import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Start with the initial solution
    h_initial = h_start.copy()
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    swarm.append(h_initial.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h_initial.copy())
    pbest_values.append(evaluate_erdos_solution(h_initial, dx))
    
    # Add random perturbations
    for _ in range(swarm_size - 1):
        h = h_initial + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for j in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[j] = (inertia * velocities[j] + 
                            c1 * r1 * (pbest[j] - swarm[j]) + 
                            c2 * r2 * (gbest - swarm[j]))
            
            # Update position
            swarm[j] += velocities[j]
            
            # Clamp to [0,1]
            swarm[j] = np.clip(swarm[j], 0, 1)
            
            # Normalize to maintain constraint
            swarm[j] = swarm[j] * (n_points / 2) / np.sum(swarm[j])
            
            # Evaluate
            curr_value = evaluate_erdos_solution(swarm[j], dx)
            
            # Update personal best
            if curr_value < pbest_values[j]:
                pbest[j] = swarm[j].copy()
                pbest_values[j] = curr_value
                
                # Update global best
                if curr_value < gbest_value:
                    gbest = swarm[j].copy()
                    gbest_value = curr_value
        
        # Check for early stopping
        if i % 50 == 0:
            if gbest_value < best_c5:
                best_h = gbest.copy()
                best_c5 = gbest_value
    
    # Convex refinement step
    try:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, eps=1e-6)
        if prob.status == cp.OPTIMAL:
            h_refined = h.value
            c5_refined = evaluate_erdos_solution(h_refined, dx)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Take the best from PSO and refine further
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
