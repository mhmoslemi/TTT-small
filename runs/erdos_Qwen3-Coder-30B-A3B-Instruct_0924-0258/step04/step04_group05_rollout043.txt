import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 50
    max_stagnation = 50
    stagnation_counter = 0
    
    # Genetic algorithm parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Initialize population
    population = []
    fitness = []
    
    # Add perturbed versions of initial solution
    for i in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
        fit = evaluate_erdos_solution(h_pert, dx)
        fitness.append(fit)
    
    # Add random solutions
    for i in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
        fit = evaluate_erdos_solution(h_rand, dx)
        fitness.append(fit)
    
    # Local search parameters
    local_search_freq = 10
    
    # Main GA loop
    generation = 0
    while time() - start_time < budget_s * 0.95:
        if len(population) == 0:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best if improved
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Check if we've met target
        if best_c5 < 0.3807:
            break
            
        # Check for stagnation to adapt parameters
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_tenure = max(10, tabu_tenure // 2)
            stagnation_counter = 0
        
        # Selection - top 20%
        elite_size = max(5, population_size // 5)
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite[:]
        
        while len(new_population) < population_size:
            if random.random() < crossover_prob:
                # Crossover
                parent1_idx = random.randint(0, len(elite)-1)
                parent2_idx = random.randint(0, len(elite)-1)
                parent1 = elite[parent1_idx]
                parent2 = elite[parent2_idx]
                
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Select parent for mutation
                parent_idx = random.randint(0, len(elite)-1)
                child = elite[parent_idx].copy()
                
            # Mutation
            if random.random() < mutation_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.01, n_points)
                child += noise
                # Repair bounds
                child = np.clip(child, 0, 1)
                # Repair normalization
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        # Local refinement every few generations
        if generation % local_search_freq == 0 and len(new_population) > 0:
            # Refine top solutions
            top_candidates = new_population[:5]
            refined_population = []
            
            for h_candidate in top_candidates:
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                        
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    res = minimize(objective, h_candidate, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        refined_h = res.x
                        refined_fitness = objective(refined_h)
                        refined_population.append(refined_h)
                        if refined_fitness < best_c5:
                            best_c5 = refined_fitness
                            best_h = refined_h.copy()
                    else:
                        refined_population.append(h_candidate)
                except Exception:
                    refined_population.append(h_candidate)
            
            # Replace original population with refined ones
            for i in range(min(len(refined_population), len(new_population))):
                new_population[i] = refined_population[i]
        
        population = new_population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        generation += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
