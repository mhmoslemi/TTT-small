import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize Fourier coefficients
    # We use a sigmoid activated Fourier series
    # h(x) = sigmoid(a0 + sum_k a_k*cos(k*pi*x) + b_k*sin(k*pi*x))
    n_freq = n_points // 2 + 1  # Number of frequency components
    a_coeffs = np.random.normal(0, 0.1, n_freq)
    b_coeffs = np.random.normal(0, 0.1, n_freq-1)  # Skip k=0 for b
    
    # Optimization loop
    max_iter = 1000
    learning_rate = 0.01
    reg_factor = 1e-6
    best_c5 = float('inf')
    best_h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute h from Fourier coefficients with sigmoid activation
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        
        # Add DC component separately
        h += a_coeffs[0]
        
        # Add frequency components  
        for k in range(1, n_freq):
            h += a_coeffs[k] * np.cos(k * np.pi * x) + b_coeffs[k-1] * np.sin(k * np.pi * x)
        
        # Apply sigmoid
        h = 1.0 / (1.0 + np.exp(-h))
        
        # Normalize to sum to 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute objective value
        c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradients using finite differences for simplicity
        # This approximates the gradient of C5 w.r.t. coefficients
        eps = 1e-5
        grad_a = np.zeros_like(a_coeffs)
        grad_b = np.zeros_like(b_coeffs)
        
        for i in range(n_freq):
            if i == 0:
                a_coeffs[i] += eps
                h_plus = 1.0 / (1.0 + np.exp(-np.sum([a_coeffs[k] * np.cos(k * np.pi * x) + 
                                                      (b_coeffs[k-1] if k > 0 else 0) * np.sin(k * np.pi * x) 
                                                      for k in range(n_freq)], axis=0)))
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad_a[i] = (c5_plus - c5) / eps
                a_coeffs[i] -= eps
            else:
                a_coeffs[i] += eps
                # Recompute full h
                h_plus = np.zeros(n_points)
                h_plus += a_coeffs[0]
                for k in range(1, n_freq):
                    h_plus += a_coeffs[k] * np.cos(k * np.pi * x) + b_coeffs[k-1] * np.sin(k * np.pi * x)
                h_plus = 1.0 / (1.0 + np.exp(-h_plus))
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad_a[i] = (c5_plus - c5) / eps
                a_coeffs[i] -= eps

        # Update coefficients using gradient descent
        a_coeffs -= learning_rate * grad_a + reg_factor * a_coeffs
        b_coeffs -= learning_rate * grad_b + reg_factor * b_coeffs
        
        # Update learning rate and regularization
        learning_rate *= 0.9995
        reg_factor *= 1.0001

    # Final refinement with scipy optimization
    def objective(coeffs):
        a = coeffs[:n_freq]
        b = coeffs[n_freq:]
        h = np.zeros(n_points)
        
        # Recompute h
        x = np.linspace(0, 2, n_points)
        h += a[0]
        for k in range(1, n_freq):
            h += a[k] * np.cos(k * np.pi * x) + b[k-1] * np.sin(k * np.pi * x)
        
        h = 1.0 / (1.0 + np.exp(-h))
        h = h * (n_points / 2) / np.sum(h)
        
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(coeffs):
        a = coeffs[:n_freq]
        b = coeffs[n_freq:]
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        h += a[0]
        for k in range(1, n_freq):
            h += a[k] * np.cos(k * np.pi * x) + b[k-1] * np.sin(k * np.pi * x)
        h = 1.0 / (1.0 + np.exp(-h))
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    # Flatten coefficients for scipy
    coeffs_0 = np.concatenate([a_coeffs, b_coeffs])
    
    # Use COBYLA for robust constraint handling
    con = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(-10, 10)] * len(coeffs_0)
    
    try:
        res = minimize(objective, coeffs_0, method='COBYLA', constraints=con, 
                       bounds=bounds, options={'maxiter': 50, 'tol': 1e-8})
        if res.success:
            a_best = res.x[:n_freq]
            b_best = res.x[n_freq:]
            h_final = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            h_final += a_best[0]
            for k in range(1, n_freq):
                h_final += a_best[k] * np.cos(k * np.pi * x) + b_best[k-1] * np.sin(k * np.pi * x)
            h_final = 1.0 / (1.0 + np.exp(-h_final))
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception as e:
        pass
        
    # Ensure we return a valid solution
    if np.any(best_h < 0) or np.any(best_h > 1) or abs(np.sum(best_h) - n_points / 2) > 1e-6:
        h_final = h_start.copy()
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        best_c5 = evaluate_erdos_solution(h_final, dx)
        best_h = h_final
    
    return (list(best_h), best_c5, n_points)
