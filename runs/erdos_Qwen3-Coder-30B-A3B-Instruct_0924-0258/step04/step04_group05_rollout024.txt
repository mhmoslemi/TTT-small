import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 40
    population = []
    fitness = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        h_start = np.ones(n_points) * 0.5
    
    # Generate diverse initial population
    # 1. Perturb the initial solution
    for i in range(10):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # 2. Add random valid solutions
    for i in range(5):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
    
    # 3. Add variants of initial solution
    for i in range(15):
        # Add noise and normalize
        h = h_start * (1 + np.random.normal(0, 0.05))
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        
    # Initialize Tabu list
    tabu_list = set()
    tabu_limit = 50
    stagnation_count = 0
    best_h = None
    best_c5 = float('inf')
    
    # GA parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Evolutionary loop
    max_generations = 500
    generation = 0
    
    while generation < max_generations and (time() - start_time) < budget_s - 30:
        # Evaluate population fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                # Early exit if target reached
                if best_c5 <= 0.3807:
                    break
        
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for stagnation
        if generation > 0 and fitness[0] == fitness[0]:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count > 50:
            # Adapt parameters
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
            
        # Create new population
        new_population = []
        
        # Elitism - keep top 5
        for i in range(5):
            new_population.append(population[i].copy())
        
        # Crossover and mutation for rest
        while len(new_population) < population_size:
            if random.random() < crossover_prob and len(new_population) + 1 < population_size:
                # Select two parents (tournament selection)
                parent1 = population[random.randint(0, 10)]
                parent2 = population[random.randint(0, 10)]
                
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Mutation
                if random.random() < mutation_rate:
                    idx = random.randint(0, n_points-1)
                    child1[idx] = random.uniform(0, 1)
                if random.random() < mutation_rate:
                    idx = random.randint(0, n_points-1)
                    child2[idx] = random.uniform(0, 1)
                
                # Repair constraints
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate a random individual
                parent = population[random.randint(0, 10)]
                child = parent.copy()
                if random.random() < mutation_rate:
                    idx = random.randint(0, n_points-1)
                    child[idx] = random.uniform(0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        # Update population
        population = new_population[:population_size]
        generation += 1
        
        # Occasionally refine best candidates with local search
        if generation % 10 == 0:
            for i in range(min(5, len(population))):
                h = population[i]
                def objective(h_test):
                    return evaluate_erdos_solution(h_test, dx)
                
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-6})
                    if res.success:
                        refined_h = res.x
                        try:
                            refined_c5 = evaluate_erdos_solution(refined_h, dx)
                            if refined_c5 < fitness[i]:
                                population[i] = refined_h
                                fitness[i] = refined_c5
                        except:
                            pass
                except:
                    pass
    
    if best_h is None:
        # Return best from final population
        final_fitness = [evaluate_erdos_solution(h, dx) for h in population]
        best_idx = np.argmin(final_fitness)
        best_h = population[best_idx]
        best_c5 = final_fitness[best_idx]
    
    # Final validity check
    if best_h is not None:
        # Ensure sum constraint
        best_h = best_h * (n_points/2) / np.sum(best_h)
        # Ensure bounds
        best_h = np.clip(best_h, 0, 1)
        # Final evaluation
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
