import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a better structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place concentrated regions that are likely to minimize overlap
        # Cluster around 0.5 and 1.5
        for i in range(n_points):
            x = i * dx
            if 0.3 <= x <= 0.7:
                h_start[i] = 0.8
            elif 1.3 <= x <= 1.7:
                h_start[i] = 0.8
            else:
                h_start[i] = 0.0
        # Normalize
        h_start *= (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iter = 100
    
    # Gradient descent parameters per feasibility check
    max_grad_steps = 500
    learning_rate = 0.1
    backtrack_factor = 0.5
    
    # Main binary search loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        
        # Attempt to find a feasible solution with this lambda
        h_feasible = best_h.copy()
        improved = True
        grad_steps = 0
        
        while improved and grad_steps < max_grad_steps and time() - start_time < budget_s * 0.95:
            improved = False
            # Compute auto-correlations
            corr = np.correlate(h_feasible, 1 - h_feasible, mode='full')
            # Get the minimum overlap shift
            min_idx = np.argmin(corr)
            # The actual shifts correspond to distance from zero
            shift_distances = np.arange(-n_points + 1, n_points)
            k_min = shift_distances[min_idx]
            
            actual_overlap = corr[min_idx] * dx
            
            if actual_overlap >= lambda_mid:
                # Already feasible
                break
                
            # Calculate gradient of the minimum overlap with respect to h
            # The overlap at shift k is: sum_i h[i] * (1 - h[i+k])
            # d/dh[i] overlap at shift k = (1 - h[i+k]) - h[i+k] * 1 (assuming valid indices)
            # This leads to: (1 - h[i+k]) - h[i+k] = 1 - h[i+k] - h[i+k] = 1 - 2*h[i+k]
            # But we need to be more careful.
            # Simple approach using finite differences of the full correlation
            # To be more precise, consider only the elements involved in this correlation term
            # For shift k, the i-th element of h in the correlation weighs (1 - h[i+k])
            # We have to compute the gradient of overlap at k wrt h[i]
            
            # Simplified gradient approximation
            # Adjust h to increase overlap at this shift
            grad = np.zeros(n_points)
            for i in range(n_points):
                i_plus_k = i + k_min
                if 0 <= i_plus_k < n_points:
                    grad[i] = 1.0 - 2.0 * h_feasible[i_plus_k]
            
            # Normalize grad
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 0:
                grad /= grad_norm
            
            old_h = h_feasible.copy()
            
            # Perform line search - decrease learning rate until improvement
            lr = learning_rate
            for _ in range(10):  # max backtrack steps
                h_try = h_feasible - lr * grad
                h_try = np.clip(h_try, 0, 1)
                h_try = h_try * (n_points/2) / np.sum(h_try)
                
                test_corr = np.correlate(h_try, 1 - h_try, mode='full')
                test_overlap = test_corr[min_idx] * dx
                
                if test_overlap >= lambda_mid:
                    h_feasible = h_try
                    improved = True
                    break
                else:
                    lr *= backtrack_factor
                    
            if not improved:
                # Even after backtrack, failed to improve
                # This can happen with small gradients
                break
              
            grad_steps += 1
            
        # Check final feasibility
        corr = np.correlate(h_feasible, 1 - h_feasible, mode='full')
        min_overlap = np.min(corr) * dx
        
        if min_overlap >= lambda_mid:
            lambda_low = lambda_mid
            if min_overlap < best_c5:
                best_h = h_feasible.copy()
                best_c5 = min_overlap
        else:
            lambda_high = lambda_mid
            
        # Periodically verify current best solution is valid
        if iteration % 5 == 0:
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            test_c5 = np.max(corr) * dx
            if test_c5 < best_c5:
                best_c5 = test_c5
    
    # Final clean-up optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
