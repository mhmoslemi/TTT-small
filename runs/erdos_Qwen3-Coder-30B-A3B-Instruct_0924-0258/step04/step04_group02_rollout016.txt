import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Adam optimizer with momentum for smooth convergence
    def adam_optimizer(objective, h_init, max_iter=2000, lr=1e-3, beta1=0.9, beta2=0.999, eps=1e-8):
        h = h_init.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        best_h = h.copy()
        best_f = objective(h)
        
        for t in range(1, max_iter + 1):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Compute gradient using finite differences
            eps_val = 1e-6
            grad = np.zeros_like(h)
            
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps_val
                h_plus = project_h(h_plus)
                obj_plus = objective(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= eps_val
                h_minus = project_h(h_minus)
                obj_minus = objective(h_minus)
                
                grad[i] = (obj_plus - obj_minus) / (2 * eps_val)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            # Update parameters
            h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_h(h)
            
            # Check for improvement
            current_f = objective(h)
            if current_f < best_f:
                best_f = current_f
                best_h = h.copy()
                
            # Occasionally inject noise to escape local minima
            if t % 100 == 0:
                noise = np.random.normal(0, 0.01, n_points)
                h = h + noise
                h = project_h(h)
        
        return best_h, best_f
    
    # Run multiple initializations to avoid local minima
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use both starting point and random perturbations
    initial_solutions = [h_start]
    
    for _ in range(5):
        h_pert = h_start + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initial_solutions.append(h_pert)
    
    for i, h_start_cycle in enumerate(initial_solutions):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_refined, c5_refined = adam_optimizer(objective, h_start_cycle, max_iter=800)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    
    # Final refinement with scipy
    if time() - start_time < budget_s * 0.95:
        def objective_vec(h_vec):
            h = h_vec.reshape(-1)
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
        
        for restart in range(2):
            if time() - start_time > budget_s * 0.95:
                break
            
            try:
                # Slightly perturb best solution
                h_init_pert = best_h + np.random.normal(0, 0.01, n_points)
                h_init_pert = np.clip(h_init_pert, 0, 1)
                h_init_pert = h_init_pert * (n_points/2) / np.sum(h_init_pert)
                
                res = minimize(objective_vec, h_init_pert, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 200, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    h_refined = h_refined.reshape(-1)
                    h_refined = project_h(h_refined)
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
