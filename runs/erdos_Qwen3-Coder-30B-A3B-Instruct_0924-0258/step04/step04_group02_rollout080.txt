import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert h to z space where h = sigmoid(z + b)
    def h_from_z(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def logit(p):
        return np.log(p / (1.0 - p + 1e-10))
    
    # Initialize z and b
    z = logit(h_start)
    # Adjust b to ensure sum(h) = n_points/2
    def adjust_b(z, target_sum):
        h = h_from_z(z)
        actual_sum = np.sum(h)
        adjustment = (target_sum - actual_sum) / n_points
        return adjustment
    
    b = adjust_b(z, n_points / 2)
    
    # Precompute indices for overlap computation
    shifts = np.arange(n_points)
    
    # Gradient-based optimization
    max_iter = 5000
    lr = 0.1
    patience = 20
    patience_counter = 0
    recent_improvements = []
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update h
        h = h_from_z(z + b)
        
        # Evaluate overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # For gradient computation we need to simulate the gradient
        # of the maximum overlap with respect to the current h values
        # This is an approximation - compute full gradient matrix and find
        # the most sensitive element in convolution
        # However, for this specific construction, we'll compute gradient
        # directly for the worst-case shift
        if k_max < n_points:
            # k_max in [0, n_points-1]
            shift = k_max
        else:
            # k_max in [n_points, 2*n_points-2]
            # We need to map back to the actual array indices
            shift = k_max - n_points + 1
        
        # Gradient approximation using finite difference
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_eps = h.copy()
            h_eps[j] += eps
            if h_eps[j] > 1:
                h_eps[j] = 1
            elif h_eps[j] < 0:
                h_eps[j] = 0
                
            # Re-normalize
            h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
            
            corr_eps = np.correlate(h_eps, 1 - h_eps, mode='full')
            overlap_eps = corr_eps[k_max] * dx
            grad[j] = (overlap_eps - max_overlap) / eps
        
        # Update z in direction of negative gradient
        z_step = -lr * grad
        
        # Compute new z
        z_new = z + z_step
        
        # Compute new b
        b_new = adjust_b(z_new, n_points / 2)
        
        # Test step
        h_new = h_from_z(z_new + b_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0  # reset patience
        else:
            patience_counter += 1
            
        z = z_new
        b = b_new
        
        # Reduce learning rate if stuck
        if patience_counter > patience:
            lr *= 0.8
            patience_counter = 0
            
        recent_improvements.append(max_overlap)
        if len(recent_improvements) > 30:
            # Early stopping check
            recent_improvements.pop(0)
            if len(recent_improvements) >= 5:
                improvement = recent_improvements[-5] - recent_improvements[-1]
                if improvement < 1e-6:
                    break
    
    # Final local refinement 
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try optimization from best solution found
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
