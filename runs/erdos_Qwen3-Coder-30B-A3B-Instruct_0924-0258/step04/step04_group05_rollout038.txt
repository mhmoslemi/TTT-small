import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # SLSQP local optimization
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Multi-perturbation search with adaptive strategy
    max_iterations = 500
    perturbation_scale = 0.02
    retry_count = 0
    max_retries = 3
    decay_rate = 0.95
    adaptive_scale_factor = 1.1
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create perturbed version
        h_perturbed = best_h + np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize to correct sum
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Local refinement of perturbed solution
        try:
            res = minimize(objective, h_perturbed, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    retry_count = 0
                else:
                    retry_count += 1
            else:
                retry_count += 1
        except:
            retry_count += 1
        
        # Adjust perturbation scale based on performance
        if retry_count >= max_retries:
            # Increase scale if we've been failing
            perturbation_scale *= adaptive_scale_factor
            retry_count = 0
        else:
            # Decay scale if we're making progress
            perturbation_scale *= decay_rate
            
        # Ensure scale stays within bounds
        if perturbation_scale > 0.2:
            perturbation_scale = 0.2
        elif perturbation_scale < 0.005:
            perturbation_scale = 0.005
            
        # Early stopping if we reach target
        if best_c5 < 0.3807:
            break

    remaining_time = budget_s - (time() - start_time)

                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final validation
    if not (np.all(best_h >= 0) and np.all(best_h <= 1)):
        raise ValueError("Final solution out of bounds")
    if abs(np.sum(best_h) - n_points / 2.0) > 1e-6:
        # Re-normalize if needed
        best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
