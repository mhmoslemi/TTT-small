import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create 3-cluster structure
        cluster_positions = [0.5, 1.0, 1.5]
        cluster_width = 0.3
        for pos in cluster_positions:
            start_idx = max(0, int((pos - cluster_width) / dx))
            end_idx = min(n_points, int((pos + cluster_width) / dx))
            h_start[start_idx:end_idx] = 1.0
        # Normalize sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    elite_size = 4
    
    # Create initial population
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolution loop
    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Select elite
        new_population = population[:elite_size]
        
        # Create offspring
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                # Crossover two parents
                parent1_idx = np.random.randint(0, elite_size)
                parent2_idx = np.random.randint(0, elite_size)
                while parent2_idx == parent1_idx:
                    parent2_idx = np.random.randint(0, elite_size)
                
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Mutation
                for child in [child1, child2]:
                    if np.random.rand() < mutation_rate:
                        mutation_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                        for i in mutation_indices:
                            child[i] += np.random.normal(0, 0.02)
                        child = np.clip(child, 0, 1)
                        child = child * (n_points/2) / np.sum(child)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate random individual
                idx = np.random.randint(0, elite_size)
                h = population[idx].copy()
                if np.random.rand() < mutation_rate:
                    mutation_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
                    for i in mutation_indices:
                        h[i] += np.random.normal(0, 0.02)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                new_population.append(h)
        
        population = new_population
    
    # Final SQP refinement on best solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-6})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
