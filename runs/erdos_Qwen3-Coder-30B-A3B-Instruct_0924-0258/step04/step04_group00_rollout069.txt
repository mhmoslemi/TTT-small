import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Modification: Use more sophisticated optimization
    # Employ alternating optimization between binary and continuous relaxation
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Alternate relaxation with binary and continuous refinement
    # Binary approximation: round to 0/1 and scale
    def project_to_binary(h):
        h_bin = np.zeros_like(h)
        sorted_indices = np.argsort(h)[::-1]  # Sort indices by descending values
        ones_count = int(np.round(n_points / 2.0))
        h_bin[sorted_indices[:ones_count]] = 1.0
        return h_bin
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        return h
    
    # Main optimization loop
    max_iter = 1000
    iter_count = 0
    last_improvement = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Try binary relaxation
        h_binary = project_to_binary(best_h)
        h_binary = project_to_feasible(h_binary)
        c5_binary = objective(h_binary)
        if c5_binary < best_c5:
            best_h = h_binary
            best_c5 = c5_binary
            last_improvement = iter_count
        
        # Continuous refinement using L-BFGS-B
        h_init = best_h.copy()
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_init, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    last_improvement = iter_count
        except Exception:
            pass

        # Local perturbation to escape local minima
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = project_to_feasible(h_pert)
        c5_pert = objective(h_pert)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert
            last_improvement = iter_count
        
        # Add small random changes
        random_mask = np.random.rand(n_points) < 0.1
        best_h[random_mask] = np.random.rand(np.sum(random_mask))
        best_h = project_to_feasible(best_h)
        c5_test = objective(best_h)
        if c5_test < best_c5:
            best_c5 = c5_test
        
        iter_count += 1
        
        # Early stopping check
        if iter_count - last_improvement > 50 and iter_count > 100:
            break
    
    # Final refinement step
    try:
        h_final = project_to_feasible(best_h)
        c5_final = objective(h_final)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
