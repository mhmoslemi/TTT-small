import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with Augmented Lagrangian
    max_iter = 2000
    alpha = 0.01
    beta = 0.01
    lambda_ = 0.0
    mu = 1.0
    
    h_current = best_h.copy()
    h_prev = h_current.copy()
    c5_prev = best_c5
    
    improvement_count = 0
    no_improvement_count = 0
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradients
        grad = np.zeros(n_points)
        corr_full = np.correlate(h_current, 1 - h_current, mode='full')
        shift_indices = np.arange(-n_points + 1, n_points)
        max_shift_idx = np.argmax(corr_full)
        max_shift = shift_indices[max_shift_idx]
        
        # Gradient for the max overlap
        if max_shift >= 0:
            for i in range(n_points - max_shift):
                grad[i] += (1 - h_current[i + max_shift]) 
                grad[i + max_shift] -= h_current[i]
        else:
            for i in range(-max_shift, n_points):
                grad[i] += (1 - h_current[i + max_shift])
                grad[i + max_shift] -= h_current[i]
        
        # Augmented Lagrangian gradient
        lagrangian_grad = grad + lambda_ + 2 * mu * (np.sum(h_current) - n_points / 2.0)
        
        # Update h
        h_new = h_current - alpha * lagrangian_grad
        
        # Project h_new onto [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to satisfy sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update Lagrange multipliers
        sum_violation = np.sum(h_new) - n_points / 2.0
        lambda_ += beta * sum_violation
        mu = max(1.0, mu * 1.01)
        
        # Accept or reject
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            improvement_count += 1
            no_improvement_count = 0
            alpha *= 1.2
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                alpha *= 0.5
                h_new = h_prev.copy()
                c5_new = c5_prev
        
        h_prev = h_current.copy()
        c5_prev = c5_new
        h_current = h_new.copy()
        
        # Early stopping if no improvement for a while
        if no_improvement_count > 50:
            break
    
    # Final refinement
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
