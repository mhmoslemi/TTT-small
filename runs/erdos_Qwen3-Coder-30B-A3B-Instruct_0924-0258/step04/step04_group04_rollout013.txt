import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5  # Simple uniform start
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_binary_iter = 60
    
    def compute_auto_correlation(h, dx):
        # Compute auto-correlation of h with (1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        # Only consider valid shifts (half the length of full correlation)
        mid = len(corr) // 2
        valid_corr = corr[mid - len(h) + 1 : mid + len(h)]
        return valid_corr * dx
    
    def is_feasible(h, lambda_val, dx):
        ac = compute_auto_correlation(h, dx)
        return np.min(ac) >= lambda_val
    
    def gradient_step(h, lambda_val, dx, learning_rate=0.1):
        ac = compute_auto_correlation(h, dx)
        min_idx = np.argmin(ac)
        
        # Gradient w.r.t. h for the minimum auto-correlation term
        # This is crude but computationally cheap proxy
        h_grad = np.zeros_like(h)
        # Increment one by one at positions
        for i in range(len(h)):
            if i + min_idx < len(h):
                h_grad[i] += 1 - h[i + min_idx]
            if i - min_idx >= 0:
                h_grad[i] += h[i - min_idx]
        
        h_new = h + learning_rate * h_grad
        h_new = np.clip(h_new, 0, 1)
        # Project to correct sum
        h_new = h_new * (n_points/2) / np.sum(h_new)
        return h_new
    
    # Warm start from best known solution
    current_h = best_h.copy()
    
    # Binary search loop
    for iter_count in range(max_binary_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        
        # Try to find feasible h for this lambda mid value
        h_candidate = current_h.copy()
        max_iter = 1000
        for step in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            if is_feasible(h_candidate, lambda_mid, dx):
                lambda_low = lambda_mid
                current_h = h_candidate.copy()
                c5_candidate = evaluate_erdos_solution(current_h, dx)
                if c5_candidate < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5_candidate
                break
            else:
                # Improve by gradient step
                h_candidate = gradient_step(h_candidate, lambda_mid, dx)
                # Ensure feasibility
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        else:
            # Not converged, lower the lambda
            lambda_high = lambda_mid
            
    # Refine on best solution found
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
