import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization using successive refinement
    max_iter = 500
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Local search with random perturbations
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        # SLSQP optimization from perturbed point
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx),
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=cons_sum,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
            
        # Occasionally use a more aggressive optimization
        if i % 10 == 0 and time() - start_time < budget_s * 0.8:
            try:
                # Nonlinear optimization with more iterations
                h_trial = best_h + np.random.normal(0, 0.05, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points/2) / np.sum(h_trial)
                
                cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(
                    lambda h: evaluate_erdos_solution(h, dx),
                    h_trial,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons_sum,
                    options={'maxiter': 100, 'ftol': 1e-9}
                )
                
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement - project to ensure sum constraint and clip
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
