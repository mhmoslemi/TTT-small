import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Enhanced Evolution Strategy with Augmented Lagrangian
    def augmented_lagrangian(h, lambda_sum, lambda_bounds, penalty_sum, penalty_bounds):
        c5 = evaluate_erdos_solution(h, dx)
        constraint_violation_sum = abs(np.sum(h) - n_points / 2.0)
        constraint_violation_bounds = np.sum(np.maximum(0, h) + np.maximum(0, 1 - h))
        penalty = penalty_sum * constraint_violation_sum + penalty_bounds * constraint_violation_bounds
        return c5 + penalty
    
    def mutate(h, sigma):
        h_mut = h + np.random.normal(0, sigma, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        # Normalize to satisfy sum constraint
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        return h_mut
    
    # ES parameters
    pop_size = 50
    sigma = 0.1
    learning_rate = 0.1
    max_generations = 1000
    stagnation_limit = 50
    stagnation_count = 0
    last_best = best_c5
    
    # Augmented Lagrangian parameters
    lambda_sum = 0.0
    lambda_bounds = np.zeros(n_points)
    penalty_sum = 1.0
    penalty_bounds = 1.0
    
    # Initialize population
    population = []
    fitness = []
    for _ in range(pop_size):
        h = mutate(best_h, sigma)
        population.append(h)
        fitness.append(evaluate_erdos_solution(h, dx))
    
    # ES iterations
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
            last_best = best_c5
        else:
            stagnation_count += 1
            
        # Check stagnation
        if stagnation_count > stagnation_limit:
            sigma = min(2.0, sigma * 1.2)
            stagnation_count = 0
            
        # Tournament selection
        selected = []
        for _ in range(pop_size // 2):
            tournament_size = 5
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Create new population
        new_population = []
        new_fitness = []
        
        # Elitism - keep best
        new_population.append(population[0].copy())
        new_fitness.append(fitness[0])
        
        # Generate offspring
        for i in range(len(selected)):
            h1 = selected[i]
            h2 = selected[np.random.randint(len(selected))]
            # Crossover
            alpha = 0.5
            h_child = alpha * h1 + (1 - alpha) * h2
            # Mutation
            h_child = mutate(h_child, sigma)
            new_population.append(h_child)
            new_fitness.append(evaluate_erdos_solution(h_child, dx))
            
        # Add some fresh random solutions
        for _ in range(pop_size // 8):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            new_population.append(h)
            new_fitness.append(evaluate_erdos_solution(h, dx))
            
        population = new_population
        fitness = new_fitness
        
        # Update penalties/adaptation
        if generation % 10 == 0:
            # Update penalty parameters
            penalty_sum *= 1.01
            penalty_bounds *= 1.01
            
            # Update lagrange multipliers based on constraint violations
            for h in population:
                constraint_violation_sum = abs(np.sum(h) - n_points / 2.0)
                constraint_violation_bounds = np.sum(np.maximum(0, h) + np.maximum(0, 1 - h))
                lambda_sum += penalty_sum * constraint_violation_sum * 0.01
                lambda_bounds += penalty_bounds * constraint_violation_bounds * 0.01
    
    # Final projection to ensure feasibility
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
