import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Initialize parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    max_stagnation = 50
    elite_size = 5
    
    # Generate initial population
    population = []
    tabu_list = []
    
    # Add variations of initial solution
    for i in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random solutions
    for i in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fill remaining slots
    while len(population) < population_size:
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Evaluate initial fitness
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Main GA loop
    stagnation_count = 0
    generation = 0
    
    while time() - start_time < budget_s * 0.95:
        generation += 1
        if stagnation_count > max_stagnation:
            # Increase mutation rate if stagnation occurs
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Select elite for next generation
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection (tournament)
            tournament_size = 3
            idx1 = random.randint(0, tournament_size-1)
            idx2 = random.randint(0, tournament_size-1)
            
            # Crossover
            if random.random() < crossover_prob:
                h1 = population[idx1]
                h2 = population[idx2]
                
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, h1, h2)
            else:
                # Clone
                child = population[random.randint(0, len(population)-1)].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child[mutate_idx] = np.random.rand()
            
            # Repair to satisfy constraint
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            # Check if this solution is tabu
            tabu_check = True
            for tabu_h in tabu_list:
                if np.allclose(child, tabu_h, atol=1e-5):
                    tabu_check = False
                    break
            if tabu_check and not np.any(np.allclose(new_population, child, atol=1e-5)):
                new_population.append(child)
        
        # Update population
        population = new_population
        
        # Evaluate fitness of new population (efficient only for new ones)
        new_fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, 2.0 / n_points)
            new_fitness.append(fit)
        
        fitness = new_fitness
        
        # Update tabu list
        if len(tabu_list) > tabu_limit:
            tabu_list = tabu_list[1:]
        current_best = population[0]
        tabu_list.append(current_best.copy())
    
    # Final refinement of best solution
    if time() - start_time < budget_s - 30:
        def objective(h):
            return evaluate_erdos_solution(h, 2.0 / n_points)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                best_h = res.x
                best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
