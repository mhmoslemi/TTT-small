import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    pop_size = 100
    num_generations = 500
    mutation_rate = 0.1
    elite_size = 10
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
    
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    def ensure_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        point = random.randint(1, n_points-1)
        child = np.concatenate([parent1[:point], parent2[point:]])
        return ensure_feasible(child)
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                h[i] += random.gauss(0, 0.01)
        return ensure_feasible(h)
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[random.randint(0, pop_size//2)]
            parent2 = population[random.randint(0, pop_size//2)]
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final check
    try:
        final_h = ensure_feasible(best_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
