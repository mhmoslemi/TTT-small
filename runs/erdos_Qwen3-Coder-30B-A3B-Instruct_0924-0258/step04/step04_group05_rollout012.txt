import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local refinement using SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass

    # Multi-perturbation global search with adaptive strategy
    h_current = best_h.copy()
    c5_current = best_c5
    sigma = 0.02
    retries = 0
    max_retries = 3
    gamma = 0.95
    max_iterations = 500
    
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iterations:
        iter_count += 1
        
        # Perturb
        h_perturbed = h_current + np.random.normal(0, sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize sum
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Local refinement
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                
                if c5_candidate < c5_current:
                    h_current = h_candidate
                    c5_current = c5_candidate
                    retries = 0
                    sigma = max(0.005, sigma * gamma)  # Reset sigma on improvement
                else:
                    retries += 1
                    if retries >= max_retries:
                        sigma *= 1.1  # Increase perturbation
                        retries = 0
                    else:
                        # Revert to previous solution for next iteration
                        pass
            else:
                retries += 1
                if retries >= max_retries:
                    sigma *= 1.1
                    retries = 0
        except:
            retries += 1
            if retries >= max_retries:
                sigma *= 1.1
                retries = 0
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, h_current, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < c5_current:
                    h_current = h_final
                    c5_current = c5_final
        except:
            pass
    
    # Ensure final feasibility
    h_final = np.clip(h_current, 0, 1)
    h_final = h_final * (n_points / 2.0) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
