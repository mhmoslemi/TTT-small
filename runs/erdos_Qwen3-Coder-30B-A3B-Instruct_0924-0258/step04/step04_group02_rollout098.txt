import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5

    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    T = T_initial
    max_iterations = int(1e5)
    
    # Iteration counter
    iteration = 0
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Create a candidate solution by perturbing two random points
        h_candidate = best_h.copy()
        
        # Select two random indices
        i, j = np.random.choice(n_points, size=2, replace=False)
        
        # Determine maximum delta that keeps both points in [0,1]
        delta_max = min(h_candidate[i], 1 - h_candidate[j])
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Apply changes
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Ensure bounds
        h_candidate[i] = np.clip(h_candidate[i], 0, 1)
        h_candidate[j] = np.clip(h_candidate[j], 0, 1)
        
        # Adjust the candidate to maintain sum constraint
        current_sum = np.sum(h_candidate)
        adjustment = (n_points / 2 - current_sum) / n_points
        
        # Apply uniform adjustment to all points
        h_candidate += adjustment
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Recompute C5 for candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject the candidate
        if candidate_c5 < best_c5 or np.random.rand() < np.exp(-(candidate_c5 - best_c5) / T):
            best_h = h_candidate.copy()
            best_c5 = candidate_c5
        
        # Cool down
        T *= alpha
        iteration += 1
        
        if iteration % 1000 == 0:
            # Periodic check on time
            if time() - start_time > budget_s * 0.95:
                break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
