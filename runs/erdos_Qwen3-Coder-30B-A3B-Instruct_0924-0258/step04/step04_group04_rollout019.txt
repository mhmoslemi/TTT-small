import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Dual variables and initial setup
    n_shifts = 2 * n_points - 1
    lambda_dual = np.ones(n_shifts)
    updates = 0
    
    def solve_convex_relaxation(h_current, lambda_current):
        h = cp.Variable(n_points)
        obj = 0
        # Entropy penalty term
        entropy = cp.sum(cp.entr(lambda_current) + lambda_current)
        
        for k in range(n_shifts):
            # Create appropriate shift matrix
            shift = k - n_points + 1
            # Create shifted version of h
            if shift >= 0:
                shifted_h = cp.hstack([cp.Variable(shift), h[:n_points-shift]])
            else:
                shifted_h = cp.hstack([h[-shift:], cp.Variable(-shift)])
            
            # Form convolution term
            conv_term = cp.sum(cp.multiply(h, 1 - shifted_h))
            obj += lambda_current[k] * conv_term
        
        obj -= entropy
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=500)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                return h_sol
        except:
            pass
        return h_current
    
    # Main algorithm loop
    max_iter = 100
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap = np.max(overlaps)
        
        if max_overlap < best_c5:
            best_h = best_h.copy()
            best_c5 = max_overlap
            
        # Update dual variables
        for i in range(n_shifts):
            lambda_dual[i] = max(lambda_dual[i] * (overlaps[i] / lambda_dual[i]), 1e-6)
        
        # Solve convex relaxation
        h_new = solve_convex_relaxation(best_h, lambda_dual)
        if h_new is not None:
            h_new = np.clip(h_new, 0, 1)
            scale = (n_points/2) / np.sum(h_new)
            h_new *= scale
            best_h = h_new.copy()
            
        # Occasionally refine with local optimization
        if iteration % 5 == 0:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                best_h = res.x
                best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
