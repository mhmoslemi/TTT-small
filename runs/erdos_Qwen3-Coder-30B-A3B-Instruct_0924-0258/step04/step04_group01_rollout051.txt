import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a more structured initial guess with 3 clusters
        cluster_centers = [0.4, 1.0, 1.6]
        cluster_sizes = [3, 4, 3]
        for i, (c, size) in enumerate(zip(cluster_centers, cluster_sizes)):
            start_idx = max(0, int((c - size/2)/dx))
            end_idx = min(n_points, int((c + size/2)/dx))
            # Make each cluster have a different height for diversity
            h_start[start_idx:end_idx] = 1.0 + 0.3 * (i - 1)  # Vary heights
        # Normalize to ensure sum is n/2
        h_start = np.clip(h_start, 0, 1)
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    population_size = 20
    max_iter = 300
    mutation_scale = 0.02
    crossover_rate = 0.5
    
    # Create initial diverse population
    population = [h_start.copy()]
    
    for _ in range(population_size - 1):
        # Create mutated version of h_start
        h_mut = h_start + np.random.normal(0, mutation_scale, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        # Normalize sum to target
        h_mut = h_mut * (n_points/2) / np.sum(h_mut)
        population.append(h_mut)
    
    # Track best solution in population
    for h in population:
        c5_val = evaluate_erdos_solution(h, dx)
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        population_scores = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(population_scores)
        top_half = sorted_indices[:population_size//2]
        
        # Selection: Keep top performers
        new_population = [population[i].copy() for i in top_half]
        
        # Crossover and mutation to generate offspring
        while len(new_population) < population_size:
            # Select two parents
            p1_idx = np.random.choice(top_half)
            p2_idx = np.random.choice(top_half)
            
            if np.random.rand() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([population[p1_idx][:crossover_point],
                                       population[p2_idx][crossover_point:]])
            else:
                # Clone parent
                child = population[p1_idx].copy()
                
            # Mutation
            if np.random.rand() < 0.7:  # 70% chance of mutation
                mutation_indices = np.random.choice(n_points, 
                                                   size=max(1, int(0.1 * n_points)), 
                                                   replace=False)
                for idx in mutation_indices:
                    child[idx] += np.random.normal(0, mutation_scale)
                    
            # Boundary clip and renormalize
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
        
        # Refinement step with SQP for each population member
        for i, h_candidate in enumerate(population):
            if time() - start_time > budget_s * 0.95:
                break
                
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                    refined_c5 = obj(refined_h)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
                    population[i] = refined_h
            except:
                pass
    
    # Final normalization to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
