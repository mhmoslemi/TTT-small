import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization (PSO) approach
    swarm_size = 50
    max_iters = 1000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Start with initial solution
    swarm.append(h_start.copy())
    velocities.append(np.zeros_like(h_start))
    
    # Generate random swarm
    for _ in range(swarm_size - 1):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal and global bests
    personal_best = [h.copy() for h in swarm]
    personal_best_scores = [evaluate_erdos_solution(h, dx) for h in swarm]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    for iter_idx in range(max_iters):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive_component = cognitive * r1 * (personal_best[i] - swarm[i])
            social_component = social * r2 * (global_best - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Update position
            swarm[i] += velocities[i]
            
            # Ensure constraints
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = score
                
                # Update global best if needed
                if score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = score
        
        # Early termination if improvement is minimal
        if iter_idx > 10 and abs(global_best_score - best_c5) < 1e-6:
            break
    
    # Use the best found during PSO
    best_h = global_best.copy()
    best_c5 = global_best_score
    
    # Convex relaxation refinement with CVXPY
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 60:
        try:
            h_vars = cp.Variable(n_points)
            constraints = [
                cp.sum(h_vars) == n_points / 2.0,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add constraints to reduce the search space
            # Use a simpler version of constraints for better performance
            max_corr = max([evaluate_erdos_solution(build_h(n_points, i), dx) for i in range(20)])
            
            # Small relaxation on some overlap constraint
            for k in range(0, min(10, 2 * n_points - 1), 2):
                if k < n_points:
                    corr = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
                else:
                    corr = cp.sum(cp.multiply(h_vars[k-n_points+1:], 1 - h_vars[:n_points-(k-n_points+1)]))
                constraints.append(corr <= max_corr * 0.9)
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_vars)), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_refined = h_vars.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final polish with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        best_final_h = best_h.copy()
        best_final_c5 = best_c5
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.005, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_final_c5:
                        best_final_h = h_refined
                        best_final_c5 = c5_refined
            except:
                pass
        
        best_h = best_final_h
        best_c5 = best_final_c5
    
    return (list(best_h), best_c5, n_points)

def build_h(n_points, variation):
    """Build a sample h vector for CVXPY constraint testing"""
    h = np.ones(n_points) * 0.5
    if variation == 0:
        # Uniform
        pass
    elif variation == 1:
        # Single peak
        idx = n_points // 2
        h[idx] = 1.0
    elif variation == 2:
        # Double peak
        h[0] = 0.5
        h[n_points//2] = 1.0
    elif variation == 3:
        # Block
        h[:n_points//4] = 1.0
    elif variation == 4:
        # Gaussian-like
        x = np.arange(n_points)
        h = np.exp(-((x - n_points/2) / (n_points/6))**2)
        h = h * (n_points/2) / np.sum(h)
    else:
        # Random
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
    return h
