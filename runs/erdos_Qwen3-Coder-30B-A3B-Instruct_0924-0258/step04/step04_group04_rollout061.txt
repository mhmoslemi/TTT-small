import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Use a two-peak structure
        peak1_start = int(0.3 / dx)
        peak1_end = int(0.7 / dx)
        peak2_start = int(1.3 / dx)
        peak2_end = int(1.7 / dx)
        h_start[peak1_start:peak1_end] = 1.0
        h_start[peak2_start:peak2_end] = 1.0
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iterations = 100
    
    def compute_auto_correlation(h):
        # Compute correlation between h and (1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        # Take only relevant part (shifts in [0, 2])
        half_len = len(corr) // 2
        return corr[half_len:]
    
    def project_to_feasible(h):
        # Clamp values to [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to meet sum constraint
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Optimization parameters
    learning_rate = 0.1
    max_inner_iter = 1000
    line_search_decay = 0.5
    min_step_size = 1e-6
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        current_lambda = (lambda_low + lambda_high) / 2.0
        
        # Try to find feasible h with this lambda
        h_current = best_h.copy()
        h_current = project_to_feasible(h_current)
        was_feasible = False
        
        # Gradient descent to satisfy constraint
        for inner_iter in range(max_inner_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            corr = compute_auto_correlation(h_current)
            # Find the shift with minimum correlation
            k_min_idx = np.argmin(corr)
            min_correlation = corr[k_min_idx]
            
            if min_correlation >= current_lambda:
                was_feasible = True
                if evaluate_erdos_solution(h_current, dx) < best_c5:
                    best_h = h_current.copy()
                    best_c5 = evaluate_erdos_solution(h_current, dx)
                break
            
            # Compute gradient of correlation with respect to h
            # This is a simplified first-order gradient approximation
            # We compute approximate gradient using finite differences
            epsilon = 1e-5
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_test = h_current.copy()
                h_test[i] += epsilon
                h_test = project_to_feasible(h_test)
                corr_plus = compute_auto_correlation(h_test)[k_min_idx]
                
                h_test[i] -= 2*epsilon
                h_test = project_to_feasible(h_test)
                corr_minus = compute_auto_correlation(h_test)[k_min_idx]
                
                grad[i] = (corr_plus - corr_minus) / (2*epsilon)
            
            # Gradient step
            h_new = h_current - learning_rate * grad
            h_new = project_to_feasible(h_new)
            
            # Line search to maintain constraint
            step_size = learning_rate
            while step_size > min_step_size:
                h_trial = h_current - step_size * grad
                h_trial = project_to_feasible(h_trial)
                corr_trial = compute_auto_correlation(h_trial)
                min_trial = corr_trial[k_min_idx]
                if min_trial >= current_lambda:
                    h_current = h_trial
                    break
                step_size *= line_search_decay
            
            if step_size <= min_step_size:
                h_current = h_trial
            
            # Early stopping if we're too far from convergence
            if abs(min_correlation - current_lambda) < 1e-6:
                if evaluate_erdos_solution(h_current, dx) < best_c5:
                    best_h = h_current.copy()
                    best_c5 = evaluate_erdos_solution(h_current, dx)
                break
        
        # Update binary search bounds based on feasibility
        if was_feasible:
            lambda_low = current_lambda
        else:
            lambda_high = current_lambda
    
    # Final check and refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
