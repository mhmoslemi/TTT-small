import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy with adaptive mutation
    population_size = 50
    mutation_strength = 0.1
    elite_size = 5
    max_generations = 10000
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, mutation_strength, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    generation = 0
    stagnation_count = 0
    max_stagnation = 20
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > max_stagnation:
            # Restart with new random population
            population = []
            for _ in range(population_size):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population.append(h)
            stagnation_count = 0
        
        # Create new population
        new_population = []
        # Keep elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring via mutation
        for _ in range(population_size - elite_size):
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Mutate
            child = population[winner_idx] + np.random.normal(0, mutation_strength, n_points)
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Adapt mutation strength
        if generation % 10 == 0 and generation > 0:
            if stagnation_count > 10:
                mutation_strength *= 1.1
            else:
                mutation_strength *= 0.99
        
        generation += 1
    
    # Final refinement with local optimization
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
