import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        # Create a smoother initial guess
        h_start = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
        # Add a few strategic peaks
        for i in range(5):
            center = 0.4 + i * 0.3
            idx = int(center / dx)
            if 0 <= idx < n_points:
                h_start[idx] = 1.0
    
    # Normalize
    h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    max_iter = 2000
    step_size = 0.01
    patience = 100
    
    # Track patience
    last_improvement = 0
    iter_count = 0
    
    def compute_gradient(h, k):
        # Compute gradient of overlap w.r.t h for shift k
        # Using finite differences for better stability
        grad = np.zeros_like(h)
        # For shift k, derivative of overlap w.r.t h[i] is:
        # (1 - h[i+k]) - (1-h[i]) * delta_{k==0}
        # But we're computing integral, so it's more complex
        # Simplified here: only direct effect of h[i] on h[i+k]
        shift = k - (n_points - 1) // 2  # Convert correlation index to shift
        if shift >= 0:
            grad[shift:] = -(1 - h[:-shift]) if shift < n_points else 0
        else:
            grad[:shift] = -(1 - h[-shift:]) if -shift < n_points else 0
        if k == (n_points - 1) // 2:  # k=0 case
            grad += -h
        grad = -grad  # Negative gradient for minimization
        return grad
    
    def backtracking_line_search(h, grad, k, alpha=0.01, beta=0.5):
        # Simple backtracking line search
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            return new_h, alpha
        return h, alpha
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        iter_count += 1
        
        # Compute overlap for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Find k with maximum overlap
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        max_overlap = corr[max_idx] * dx
        if max_overlap == 0:
            continue
            
        # Compute gradient
        grad = compute_gradient(best_h, max_idx)
        # Normalize gradient
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm
        
        # Try line search
        new_h, alpha = backtracking_line_search(best_h, grad, max_idx)
        
        # Update
        best_h = new_h.copy()
        new_c5 = evaluate_erdos_solution(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            last_improvement = iter_count
        
        if iter_count - last_improvement > patience:
            # Reset step size and try a different strategy
            step_size *= 0.9
            # Add a small random perturbation
            perturbation = np.random.normal(0, 0.001, n_points)

                best_c5 = new_c5
                last_improvement = iter_count

                    return evaluate_erdos_solution(h, dx)

                    return np.sum(h) - n_points/2

                if res.success:
                    refined_h = res.x
                    refined_c5 = evaluate_erdos_solution(refined_h, dx)
                    if refined_c5 < best_c5:
                        best_h = refined_h
                        best_c5 = refined_c5
            except:
                pass
    
    # Final verification
    if not np.all((best_h >= 0) & (best_h <= 1)):
        best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
