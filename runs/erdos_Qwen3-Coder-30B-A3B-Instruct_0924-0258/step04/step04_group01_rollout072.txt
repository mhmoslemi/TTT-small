import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import MOSEK

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess with three clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.3, 1.0, 1.7]
        cluster_width = 0.4
        for c in cluster_centers:
            start = max(0, int((c - cluster_width/2)/dx))
            end = min(n_points, int((c + cluster_width/2)/dx))
            h_start[start:end] = 1.0 / (cluster_width/dx)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    tolerance = 1e-6
    max_iterations = 50
    
    # SCP method for feasibility checking
    def solve_feasibility(h_init, C_val):
        # Create CVXPY variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Linearize the overlap constraint
            # ∑ h[i] * h[i+k] >= 1 - C_val * dx
            # using h[i+k] ≈ h_init[i+k] (constant for linearization)
            overlap_expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    overlap_expr += h[i] * h_init[i+k] + h_init[i] * h[i+k] - h_init[i] * h_init[i+k]
            constraints.append(overlap_expr >= 1 - C_val * dx)
        
        # Objective (minimize a dummy function)
        objective = cp.Minimize(0)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        
        return None
    
    # Binary search
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        mid = (low + high) / 2.0
        h_solution = solve_feasibility(best_h, mid)
        
        if h_solution is not None:
            # Check if this is actually a feasible solution
            h_test = h_solution.copy()
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points/2) / np.sum(h_test)
            c5_test = evaluate_erdos_solution(h_test, dx)
            if c5_test < best_c5:
                best_h = h_test.copy()
                best_c5 = c5_test
            
            high = mid
        else:
            low = mid
        
        iteration += 1
    
    # Final improvement with local optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try several local optimizations from current best
    local_iters = 5
    for _ in range(local_iters):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Multiple restarts with slight perturbation
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    c5_final = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), c5_final, n_points)
