import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach
    resolutions = [35, 40, 50]
    
    # First, optimize at low resolution
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        current_n = res
        current_dx = 2.0 / current_n
        
        # Initialize h for this resolution
        if res == 35:
            h_current = h_start.copy()
        else:
            # Interpolate from previous resolution
            h_current = np.interp(np.linspace(0, 2, current_n), 
                                np.linspace(0, 2, n_points), h_start)
        
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (current_n/2) / np.sum(h_current)
        
        # Local optimization with random restarts
        max_restarts = 20
        restart_count = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_current + np.random.normal(0, 0.02, current_n)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (current_n/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            def objective(h):
                return evaluate_erdos_solution(h, current_dx)
            
            def constraint_sum(h):
                return np.sum(h) - current_n / 2.0
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(current_n)]
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 150, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0  # Reset counter on improvement
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
        
        if current_n > n_points:
            h_start = h_current
            n_points = current_n
    
    # Final optimization on best solution found
    if time() - start_time < budget_s * 0.9:
        try:
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
