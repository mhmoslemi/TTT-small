def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

            h_start[start_idx:end_idx] = 0.5

        h_start = h_start * (n_points/2) / np.sum(h_start)

    best_h = h_start.copy()

            corr = np.correlate(h, 1 - h, mode='full')

            if np.max(corr) * dx > c_val:
                return False

            # This helps escape local minima
            idx = np.argmax(corr)

                h_shifted = np.roll(h, -shift)
