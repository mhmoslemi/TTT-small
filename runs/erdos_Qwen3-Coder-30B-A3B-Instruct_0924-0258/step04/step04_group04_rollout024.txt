import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Problem, Minimize, sum as cp_sum, norm

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    n_points = 35 if 'initial_h_values' in globals() else 20
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Start with a structured construction
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create two peaks
        peak1_start = int(0.3 / dx)
        peak1_end = int(0.7 / dx)
        peak2_start = int(1.3 / dx)
        peak2_end = int(1.7 / dx)
        h[peak1_start:peak1_end] = 1.0
        h[peak2_start:peak2_end] = 1.0
        # Normalize
        h = h * (n_points/2) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Define penalty method optimization
    def solve_with_penalties(h_in, max_iter=100):
        h = h_in.copy()
        penalties = np.ones(len(h) * 2 - 1)  # full convolution
        
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Calculate overlaps
            corr = np.correlate(h, 1 - h, mode='full')
            
            # Update penalties (adaptive)
            for i in range(len(penalties)):
                if corr[i] > penalties[i]:
                    penalties[i] *= 1.1
                else:
                    penalties[i] *= 0.99
            
            # Define CVXPY problem for this iteration
            h_var = cp.Variable(n_points)
            objective = cp.Minimize(cp_sum(cp.maximum(corr + penalties, 0)) * dx)
            constraints = [
                cp_sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            prob = Problem(objective, constraints)
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status == 'optimal':
                    new_h = h_var.value
                    if np.all(new_h >= 0) and np.all(new_h <= 1):
                        h = new_h
                        c5_new = evaluate_erdos_solution(h, dx)
                        if c5_new < best_c5:
                            best_h = h.copy()
                            best_c5 = c5_new
            except:
                # fallback to gradient-based approach
                pass
                
        return h, best_h, best_c5
    
    # Upsample to 40 points
    if n_points < 40:
        h_upsampled = np.interp(np.linspace(0, 2, 40), np.linspace(0, 2, n_points), h)
        h_upsampled = h_upsampled * 20 / np.sum(h_upsampled)
        h, best_h, best_c5 = solve_with_penalties(h_upsampled, 100)
        n_points = 40
    
    # Refine with gradient-based optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    # Try SLSQP
    try:
        res = minimize(obj, h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 200, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final overall optimization with L-BFGS-B
    try:
        res = minimize(obj, best_h, method='L-BFGS-B', 
                      bounds=bounds, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    # Final validation
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points/2) > 1e-6:
        best_h = best_h * (n_points/2) / final_sum
    
    return (list(best_h), best_c5, n_points)
