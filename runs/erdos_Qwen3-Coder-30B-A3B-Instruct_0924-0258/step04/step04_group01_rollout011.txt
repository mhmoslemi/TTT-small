import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a structured initial guess with 3 clusters
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # BOBYQA optimization parameters
    max_iter = 500
    initial_trust_radius = 0.5
    min_trust_radius = 1e-5
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Set up bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # BOBYQA-based optimization loop - 100 iterations max
    for i in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Use BOBYQA
        try:
            res = minimize(
                objective, 
                h_start, 
                method='bobyqa',
                bounds=bounds,
                constraints=cons_sum,
                options={
                    'maxiter': 10,
                    'rhoend': min_trust_radius,
                    'catol': 1e-6
                }
            )
            if res.success:
                h_candidate = res.x
                # Ensure feasibility
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                h_start = h_candidate.copy()
            else:
                # Fallback to random perturbation
                h_perturbed = h_start + np.random.normal(0, 0.01, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                c5_perturbed = objective(h_perturbed)
                if c5_perturbed < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = c5_perturbed
                h_start = h_perturbed.copy()
        except:
            # Fallback
            h_perturbed = h_start + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
            h_start = h_perturbed.copy()
    
    # Post-processing cleanup to ensure constraints are met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
