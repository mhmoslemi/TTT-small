```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    n_points = 35
    dx = 2.0 / n_points
    
    # Get initial solution if available, otherwise create random one
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Add some diversity
        population = [initial_h.copy()]
        for _ in range(9):
            pert = initial_h + np.random.normal(0, 0.02, n_points)
            pert = np.clip(pert, 0, 1)
            pert = pert * (n_points/2) / np.sum(pert)
            population.append(pert)
    else:
        # Create random initial solutions
        population = []
        for _ in range(10):
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            population.append(h)
    
    # Add some random valid solutions
    for _ in range(5):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)

    # Tabu memory to prevent revisiting similar solutions
    tabu_list = []
    tabu_limit = 50
    best_h = None
    best_c5 = float('inf')
    
    # Genetic algorithm parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_counter = 0
    max_stagnation = 50
    
    # Generate diverse initial population if needed
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Keep only desired number of solutions
    population = population[:population_size]
    
    # Main evolutionary loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        if time() - start_time > budget_s * 0.9:
            break
            
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            h_clipped = np.clip(h, 0, 1)
            h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
            c5 = evaluate_erdos_solution(h_normalized, dx)
            fitness.append(c5)
        
        # Update best solution
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Adaptive parameter adjustment
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)

        # Selection and reproduction
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:10]]
        
        # Local refinement every 10 generations
        if gen % 10 == 0:
            for i in range(min(5, len(selected))):
                h_refined = selected[i].copy()
                try:
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                    
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    bounds = [(0, 1) for _ in range(n_points)]
                    
                    res = minimize(objective, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_c5 = c5_refined
                            best_h = h_refined.copy()
                except:
                    pass
        
        # Create new population
        new_population = []
        
        # Keep best individuals (elitism) 
        for i in range(5):
            new_population.append(selected[i].copy())
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            if random.random() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                # Single point crossover
                crossover_point = random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Mutation
                if random.random() < mutation_rate:
                    # Flip a few random positions
                    for _ in range(3):
                        pos = random.randint(0, n_points-1)
                        child[pos] = random.random()
                
                # Normalize and clip
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
                # Avoid tabu solutions
                tabu_similar = False
                for tabu_h in tabu_list[-tabu_limit:]:
                    if np.allclose(child, tabu_h, atol=1e-3):
                        tabu_similar = True
                        break
                
                if not tabu_similar:
                    new_population.append(child)
                    if len(tabu_list) >= tabu_limit:
                        tabu_list.pop(0)
                    tabu_list.append(child.copy())
            else:
                # Just mutation
                parent = random.choice(selected)
                child = parent.copy()
                if random.random() < mutation_rate:
                    for _ in range(2):
                        pos = random.randint(0, n_points-1)
                        child[pos] = random.random()
                
                # Normalize and clip
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
                # Avoid tabu solutions
                tabu_similar = False
                for tabu_h in tabu_list[-tabu_limit:]:
                    if np.allclose(child, tabu_h, atol=1e-3):
                        tabu_similar = True
                        break
                
                if not tabu_similar:
                    new_population.append(child)
                    if len(tabu_list) >= tabu_limit:
                        tabu_list.pop(0)
                    tabu_list.append(child.copy())
    
        population = new_population[:population_size]
        
        # Early stopping if we've hit our target
        if best_c5 < 0.3807:
            break
    
    # Final refinement of best solution
    if best_h is not None:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                best_h = res.x
                best_c5 = objective(best_h)
        except:
            pass

    final_h = best_h.copy() if best_h is not None else population[0]
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
```
