import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization
    n_coarse = 20
    h_coarse = np.linspace(0, 2, n_coarse + 1)
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), h_coarse[:-1], h_start)
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    dx_coarse = 2.0 / n_coarse
    
    # Optimization loop with penalty method
    # For coarse grid
    max_iter = 100
    penalties = np.ones(2 * n_coarse - 1)
    learning_rate = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Solve subproblem
        h_var = cp.Variable(n_coarse)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_coarse/2]
        
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        penalty_terms = [penalties[i] * corr[i] for i in range(len(corr))]
        objective = cp.Minimize(cp.sum(penalty_terms))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            h_new = h_var.value
        except:
            h_new = h_coarse.copy()
        
        # Update penalties
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        penalties = np.maximum(penalties * (corr_new / (penalties + 1e-8)), 1e-6)
        
        # Update coarse solution
        h_coarse = h_new.copy()
        h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
        
        # Evaluate
        cost_new = evaluate_erdos_solution(h_coarse, dx_coarse)
        if cost_new < best_c5:
            best_h = h_coarse.copy()
            best_c5 = cost_new
            
        # Upsample for fine grid
        h_fine = np.linspace(0, 2, n_points + 1)
        f_interp = interp1d(np.linspace(0, 2, n_coarse), h_coarse, kind='linear')
        h_fine = f_interp(np.linspace(0, 2, n_points))
        h_fine = h_fine * (n_points/2) / np.sum(h_fine)
        dx_fine = 2.0 / n_points
        
        # Fine optimization
        max_iter_fine = 200
        penalties_fine = np.ones(2 * n_points - 1)
        learning_rate_fine = 0.2
        
        for fine_iter in range(max_iter_fine):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_fine_var = cp.Variable(n_points)
            constraints_fine = [h_fine_var >= 0, h_fine_var <= 1, cp.sum(h_fine_var) == n_points/2]
            
            corr_fine = np.correlate(h_fine, 1 - h_fine, mode='full')
            penalty_terms_fine = [penalties_fine[i] * corr_fine[i] for i in range(len(corr_fine))]
            objective_fine = cp.Minimize(cp.sum(penalty_terms_fine))
            prob_fine = cp.Problem(objective_fine, constraints_fine)
            
            try:
                prob_fine.solve(solver=cp.SCS, verbose=False)
                h_fine_new = h_fine_var.value
            except:
                h_fine_new = h_fine.copy()
                
            # Update penalties
            corr_fine_new = np.correlate(h_fine_new, 1 - h_fine_new, mode='full')
            penalties_fine = np.maximum(penalties_fine * (corr_fine_new / (penalties_fine + 1e-8)), 1e-6)
            
            h_fine = h_fine_new.copy()
            h_fine = h_fine * (n_points/2) / np.sum(h_fine)
            
            cost_fine = evaluate_erdos_solution(h_fine, dx_fine)
            if cost_fine < best_c5:
                best_h = h_fine.copy()
                best_c5 = cost_fine
    
    # Final refinement 
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
