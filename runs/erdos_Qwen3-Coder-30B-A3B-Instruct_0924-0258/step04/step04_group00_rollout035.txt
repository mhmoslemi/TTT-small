import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Strategy with Augmented Lagrangian
    pop_size = 50
    mut_strength = 0.1
    max_gen = 1000
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def augmented_lagrangian(h, lambda_sum, lambda_bounds, penalty_sum, penalty_bounds):
        h_feas = project_to_feasible(h)
        obj = evaluate_erdos_solution(h_feas, dx)
        # Penalty terms
        penalty_sum_term = penalty_sum * (np.sum(h_feas) - n_points/2)**2
        penalty_bounds_term = penalty_bounds * np.sum(np.maximum(0, h_feas-1)**2 + np.maximum(0, -h_feas)**2)
        return obj + penalty_sum_term + penalty_bounds_term
    
    # Initial population
    population = [project_to_feasible(h_start + np.random.normal(0, mut_strength, n_points)) 
                  for _ in range(pop_size)]
    lambda_sum = 0
    lambda_bounds = np.zeros(n_points)
    penalty_sum = 1.0
    penalty_bounds = 1.0
    
    gen = 0
    while time() - start_time < budget_s * 0.95 and gen < max_gen:
        # Evaluate each individual
        fitness = [augmented_lagrangian(indiv, lambda_sum, lambda_bounds, penalty_sum, penalty_bounds) 
                   for indiv in population]
        
        # Tournament selection
        selected = []
        tournament_size = 5
        for _ in range(pop_size):
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        
        # Reproduction with mutation
        next_gen = []
        for indiv in selected:
            mutated = indiv + np.random.normal(0, mut_strength, n_points)
            next_gen.append(project_to_feasible(mutated))
        
        population = next_gen
        
        # Update best
        current_fitness = [evaluate_erdos_solution(indiv, dx) for indiv in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = current_fitness[best_idx]
        
        # Adapt mutation strength based on diversity
        distances = [np.linalg.norm(population[i] - population[j]) 
                     for i in range(pop_size) for j in range(i+1, pop_size)]
        if distances:
            diversity = np.mean(distances)
            mut_strength = max(0.001, min(0.5, mut_strength * (1.0 + 0.01 * (diversity - 0.05))))
        
        gen += 1
    
    # Final validation with local refinement
    try:
        h_final = project_to_feasible(best_h)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
