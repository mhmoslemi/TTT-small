import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Define grid sizes and optimization steps
    grid_sizes = [20, 40]
    max_iter_coarse = 100
    max_iter_fine = 100
    temp_start = 0.1
    temp_decay = 0.98
    perturb_mag_coarse = 0.03
    perturb_mag_fine = 0.02
    
    # Initialize with parent's starting point or uniform distribution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
        # Make sure initial is compatible with coarse grid
        if n_points not in grid_sizes:
            # Downsample to closest coarse size
            target_points = grid_sizes[0]
            h_resampled = np.interp(
                np.linspace(0, 2, target_points),
                np.linspace(0, 2, n_points),
                h_start
            )
            h_start = h_resampled
            n_points = target_points
    else:
        # Start with basic uniform distribution
        n_points = grid_sizes[0]
        h_start = np.ones(n_points) * 0.5
        # Normalize to sum to n_points / 2
        h_start = h_start * (n_points / 2) / np.sum(h_start)

    # Ensure proper normalization
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Multi-grid optimization
    for grid_idx, grid_size in enumerate(grid_sizes):
        if time() - start_time > budget_s * 0.95:
            break
            
        # If upgrading resolution, interpolate
        if grid_idx > 0:
            prev_n = grid_sizes[grid_idx - 1]
            # Ensure we're at the correct resolution
            if grid_size > prev_n:
                # Create reference grid for interpolation
                old_x = np.linspace(0, 2, prev_n)
                new_x = np.linspace(0, 2, grid_size)
                h_new = np.interp(new_x, old_x, best_h)
                # Normalize to ensure sum constraint
                scale = (grid_size / 2) / np.sum(h_new)
                h_new = h_new * scale
                best_h = h_new
                n_points = grid_size
                dx = 2.0 / n_points
                
        # Simulated Annealing for coarse grid
        if grid_idx == 0:
            temp = temp_start
            for iteration in range(max_iter_coarse):
                if time() - start_time > budget_s * 0.95:
                    break
                h_new = best_h + np.random.normal(0, perturb_mag_coarse, n_points)
                h_new = np.clip(h_new, 0, 1)
                # Normalize to correct sum
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                c5_old = best_c5
                if c5_new < c5_old or np.random.rand() < math.exp(-(c5_new - c5_old) / temp):
                    best_h = h_new
                    best_c5 = c5_new
                temp *= temp_decay
        
        # Fine grid optimization using SLSQP
        elif grid_idx == 1:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            for iteration in range(max_iter_fine):
                if time() - start_time > budget_s * 0.95:
                    break
                try:  
                    res = minimize(obj, best_h, method='SLSQP', 
                                  bounds=bounds, constraints=cons, 
                                  options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        best_h = res.x
                        best_c5 = evaluate_erdos_solution(best_h, dx)
                except:
                    pass
                    
            # Final gradient refinement
            # Simple gradient ascent with small steps
            num_refinements = 100
            step_size = 1e-3
            for iteration in range(num_refinements):
                if time() - start_time > budget_s * 0.95:
                    break
                # Simple gradient approximation by finite differences
                # This is just a rough local search
                h_trial = best_h + np.random.normal(0, step_size * 0.1, n_points)
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                c5_trial = evaluate_erdos_solution(h_trial, dx)
                if c5_trial < best_c5:
                    best_h = h_trial
                    best_c5 = c5_trial

    # Ensure final solution is feasible and correct
    if best_c5 > 1.0 or np.isnan(best_c5):
        # fallback in case of numerical issues
        h_check = np.clip(h_start, 0, 1)
        h_check = h_check * (n_points / 2) / np.sum(h_check)
        best_c5 = evaluate_erdos_solution(h_check, dx)
        best_h = h_check
    
    return (list(best_h), best_c5, n_points)
