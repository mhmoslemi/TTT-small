import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy parameters
    population_size = 50
    num_generations = 500
    mutation_strength = 0.1
    elite_size = 5
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = h_start + np.random.normal(0, mutation_strength, n_points)
        h = project_to_feasible(h)
        population.append(h)
    
    # Store current best
    current_best = best_h.copy()
    current_best_c5 = best_c5
    
    # Evolutionary loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_values = [fitness(h) for h in population]
        sorted_indices = np.argsort(fitness_values)
        
        # Update best solution
        if fitness_values[sorted_indices[0]] < current_best_c5:
            current_best = population[sorted_indices[0]].copy()
            current_best_c5 = fitness_values[sorted_indices[0]]
            
        # Selection and reproduction
        elite = [population[i] for i in sorted_indices[:elite_size]]
        parents = [population[i] for i in sorted_indices[:population_size // 2]]
        
        # Create new population through mutation
        new_population = elite.copy()
        while len(new_population) < population_size:
            parent = parents[np.random.randint(0, len(parents))]
            child = parent + np.random.normal(0, mutation_strength, n_points)
            child = project_to_feasible(child)
            new_population.append(child)
            
        population = new_population
        
        # Adapt mutation strength based on diversity
        if generation % 20 == 0:
            diffs = np.array([np.std([pop[i] for pop in population]) for i in range(n_points)])
            mutation_strength = 0.1 * np.mean(diffs)
    
    # Final check of the best solution
    try:
        final_h = project_to_feasible(current_best)
        final_c5 = fitness(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
