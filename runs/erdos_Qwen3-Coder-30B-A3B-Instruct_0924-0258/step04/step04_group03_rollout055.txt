import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO Parameters
    swarm_size = 50
    max_iterations = 1000
    inertia_weight = 0.9
    cognitive_param = 2.0
    social_param = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = h * (n_points/2) / np.sum(h)  # Enforce sum constraint
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h, dx)
        personal_best_positions.append(h.copy())
        personal_best_scores.append(score)
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_param * r1 * (personal_best_positions[i] - particles[i])
            social = social_param * r2 * (global_best_position - particles[i])
            velocities[i] = (inertia_weight * velocities[i] + cognitive + social)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Maintain sum constraint
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate current position
            current_score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best_position = particles[i].copy()
        
        # Update inertia weight linearly
        inertia_weight = 0.9 - (0.5 * iteration / max_iterations)
        
        # Occasionally refine with convex optimization
        if iteration % 50 == 0 and time() - start_time < budget_s * 0.9:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add overlap constraints
                for k in range(n_points):
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    constraints.append(corr_sum <= global_best_score * 1.1)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < global_best_score:
                            global_best_position = h_refined
                            global_best_score = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final SLSQP polish with best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, global_best_position, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < global_best_score:
                    global_best_position = h_final
                    global_best_score = c5_final
        except Exception:
            pass
    
    return (list(global_best_position), global_best_score, n_points)
