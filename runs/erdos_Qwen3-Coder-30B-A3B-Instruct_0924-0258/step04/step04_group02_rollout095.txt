from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters for simulated annealing
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    delta_max = 0.2
    
    T = T_initial
    h_current = h_start.copy()
    c5_current = best_c5
    
    # Simulated annealing loop
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Generate candidate by perturbing two points
        i, j = np.random.randint(0, n_points, 2)
        if i == j:
            j = (j + 1) % n_points
        
        # Compute max allowed delta
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Adjust h' while maintaining sum constraint
        h_new = h_current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Clamp to [0,1]
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Adjust the other points to maintain sum
        diff = np.sum(h_new) - (n_points / 2)

            # Distribute diff across all points except i and j
            for k in range(n_points):

        c5_new = evaluate_erdos_solution(h_new, dx)

            c5_current = c5_new
        
        # Update best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
        
        # Cool down
        T *= alpha
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
