import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with momentum
    def objective_and_gradient(h):
        # Normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
            
        c5 = evaluate_erdos_solution(h, dx)
        
        # Compute gradient numerically
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Optimizer with adaptive learning rate
    h_current = best_h.copy()
    lr = 1e-2
    momentum = 0.9
    velocity = np.zeros_like(h_current)
    
    iter_count = 0
    max_iter = int(1e6)
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        c5_current, grad = objective_and_gradient(h_current)
        
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
            
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        h_current = h_current + velocity
        
        # Project to valid range and normalize
        h_current = np.clip(h_current, 0, 1)
        h_sum = np.sum(h_current)
        if h_sum > 0:
            h_current = h_current * (n_points / 2) / h_sum
            
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count % 1000 == 0:
            lr *= 0.95
    
    # Final refinement with constrained optimization
    def objective(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 1000, 'ftol': 1e-10})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    h_sum = np.sum(best_h)
    if h_sum > 0:
        best_h = best_h * (n_points / 2) / h_sum
    
    return (list(best_h), best_c5, n_points)
