import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization using scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Initial optimization using L-BFGS-B
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb initial solution for restart
        h_restart = h_start + np.random.normal(0, 0.05, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        # Normalize to correct sum
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Additional fine-tuning with local search
    remaining_time = budget_s - (time() - start_time)

        # Local search around best solution
        for _ in range(30):
            if time() - start_time > budget_s * 0.95:

            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert
    
    return (list(best_h), best_c5, n_points)
