import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create clusters at strategic positions
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use BOBYQA for the optimization
    # Define constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint for the sum
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # BOBYQA optimization
    max_iter = 500
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Just use the current best as starting point for BOBYQA
        try:
            # Wrap objective to handle constraint as penalty
            def obj(h):
                h_clipped = np.clip(h, 0, 1)
                h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
                return evaluate_erdos_solution(h_normalized, dx)
            
            # BOBYQA with trust region, bound constraints handled manually
            res = minimize(obj, best_h, method='BOBYQA', 
                           options={'maxiter': 1, 'catol': 1e-6, 'rhoend': 1e-5},
                           bounds=bounds, constraints=cons_sum)
            
            if res.success:
                h_refined = res.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            # Revert to best solution if optimization fails
            pass
        
        # Occasionally re-initialize with a perturbed version to avoid local optima
        if iteration % 10 == 0 and iteration > 0:
            h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
