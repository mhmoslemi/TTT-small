import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with better resolution and existing solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Adjust to target resolution
    if n_points != 40:
        # Interpolate or resize to 40 points
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 40)
        h_start = np.interp(x_new, x_orig, h_start)
        n_points = 40
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive grid search for feasibility check
    search_range = np.linspace(0.37, best_c5, 20)
    best_c5_grid = best_c5
    
    # Grid search with feasibility check
    for c_mid in search_range:
        if time() - start_time > budget_s * 0.5:
            break
            
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Exact correlation computation and constraint
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                part1 = h[:n_points-k]
                part2 = 1 - h[k:]
                corr_sum = cp.sum(cp.multiply(part1, part2))
            else:
                part1 = h[k-n_points+1:]
                part2 = 1 - h[:n_points-(k-n_points+1)]
                corr_sum = cp.sum(cp.multiply(part1, part2))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5_grid:
                    best_c5_grid = c5_feasible
                    best_h = h_feasible.copy()
        except:
            continue
    
    # Gradient-based refinement with multiple restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use different solvers and increase restart attempts
    solvers = ['SLSQP', 'COBYLA']
    for restart in range(20):
        if time() - start_time > budget_s * 0.9:
            break
            
        if restart == 0:
            h_init = best_h.copy()
        else:
            # Random perturbation with adaptive sigma
            sigma = 0.02 + 0.01 * np.random.rand()
            h_init = best_h + np.random.normal(0, sigma, n_points)
            h_init = np.clip(h_init, 0, 1)
            # Normalize
            h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Try multiple solvers
        for solver in solvers:
            if time() - start_time > budget_s * 0.95:
                break
            try:
                res = minimize(objective, h_init, method=solver, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        if best_c5 < 0.3807:
                            break
            except:
                continue
    
    # Final polish with higher precision if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final validation with equal sum constraint
    best_h = np.clip(best_h, 0, 1)
    scale_factor = (n_points / 2) / np.sum(best_h)
    best_h = best_h * scale_factor
    
    return (list(best_h), best_c5, n_points)
