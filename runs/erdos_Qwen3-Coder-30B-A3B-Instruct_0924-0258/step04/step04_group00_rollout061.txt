import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Evolutionary strategy with adaptive mutation
    population_size = 50
    mutation_strength = 0.1
    elite_size = 10
    
    population = []
    fitness = []
    
    # Generate initial population
    for _ in range(population_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitness.append(objective(h))
    
    # Sort by fitness
    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]
    fitness = [fitness[i] for i in sorted_indices]
    
    # Evolutionary loop
    max_generations = 500
    generation = 0
    
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Selection: keep top individuals
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        
        # Mutation and recombination
        while len(new_population) < population_size:
            # Select parents
            parent1 = elite[np.random.randint(0, elite_size)]
            parent2 = elite[np.random.randint(0, elite_size)]
            
            # Crossover (uniform)
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            noise = np.random.normal(0, mutation_strength, n_points)
            child = child + noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        # Evaluate new population
        new_fitness = [objective(h) for h in new_population]
        
        # Sort and keep best
        sorted_indices = np.argsort(new_fitness)
        population = [new_population[i] for i in sorted_indices]
        fitness = [new_fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Adaptive mutation
        if generation % 10 == 0:
            # Adjust mutation strength based on diversity
            diversity = np.std([np.std(h) for h in population])
            mutation_strength = max(0.01, 0.1 * (1.0 - diversity))
        
        generation += 1
    
    # Final validation and refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
