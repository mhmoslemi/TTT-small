import numpy as np
from scipy.signal import correlate
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    elite_size = 10
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
            h = np.clip(h, 0, 1)
            population.append(h)
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def tournament_selection(pop, k=2):
        indices = np.random.choice(len(pop), size=k, replace=False)
        candidates = [pop[i] for i in indices]
        fitnesses = [evaluate_erdos_solution(h, dx) for h in candidates]
        winner_idx = np.argmin(fitnesses)
        return candidates[winner_idx]
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return normalize(child1), normalize(child2)
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h[i] += np.random.normal(0, 0.01)
        return normalize(h)
    
    # Main evolutionary loop
    generation = 0
    while time() - start_time < budget_s * 0.95:
        # Evaluate fitness of current population
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Track best solution
        best_in_gen_idx = np.argmin(fitnesses)
        if fitnesses[best_in_gen_idx] < best_c5:
            best_h = population[best_in_gen_idx].copy()
            best_c5 = fitnesses[best_in_gen_idx]
        
        # Create new population
        new_population = []
        
        # Elite preservation
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        generation += 1
    
    # Final refinement with local search
    try:
        # Try a few random restarts with local search
        for _ in range(10):
            h_rand = best_h + np.random.normal(0, 0.01, n_points)
            h_rand = normalize(h_rand)
            c5_rand = evaluate_erdos_solution(h_rand, dx)
            if c5_rand < best_c5:
                best_h = h_rand
                best_c5 = c5_rand
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
