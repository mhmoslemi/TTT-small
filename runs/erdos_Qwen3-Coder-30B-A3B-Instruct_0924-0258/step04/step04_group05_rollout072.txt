import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    population_size = 40
    n_points = 35
    dx = 2.0 / n_points
    population = []
    tabu_memory = set()
    
    # Initialize with perturbed version of initial_h_values or uniform
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
    else:
        base_h = np.ones(n_points) * 0.5

    # Generate diverse population
    for i in range(10):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        tabu_memory.add(tuple(h.round(8)))
        
    for i in range(5):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        tabu_memory.add(tuple(h.round(8)))
        
    # Add some random valid solutions
    for i in range(population_size - 15):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        tabu_memory.add(tuple(h.round(8)))
    
    # Parameters
    mutation_rate = 0.15
    crossover_probability = 0.7
    tabu_limit = 50
    stagnation_count = 0
    best_c5 = float('inf')
    best_individual = None
    
    # Genetic algorithm with tabu search
    max_generations = 500
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitnesses.append(c5)
            
        # Track best solution so far
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_individual = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive parameters
        if stagnation_count > 50:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_probability = max(0.3, crossover_probability - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_probability = min(0.9, crossover_probability + 0.05)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_size = population_size // 4
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if random.random() < crossover_probability and len(elite) >= 2:
                parent1 = random.choice(elite)
                parent2 = random.choice(elite)
                
                # Uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Select random parent
                parent = random.choice(elite)
                child = parent.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                mutation_idx = random.randint(0, n_points - 1)
                child[mutation_idx] = random.random()
                
            # Normalize and ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            # Avoid tabu solutions
            child_tuple = tuple(child.round(8))
            if child_tuple in tabu_memory and len(new_population) < population_size - 10:
                continue
                
            new_population.append(child)
            tabu_memory.add(child_tuple)
            
            # Manage tabu memory size
            if len(tabu_memory) > tabu_limit:
                tabu_memory.pop()
                
        population = new_population
        
        # Local refinement every 10 generations
        if gen % 10 == 0 and gen > 0:
            refined_population = []
            for h in population[:5]:  # Top 5
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)

                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0

                cons = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h, method='SLSQP', 
                                 bounds=bounds, constraints=cons, 
                                 options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                    else:
                        h_refined = h
                except:
                    h_refined = h
                    
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                refined_population.append(h_refined) 
                
            # Replace first few with refined versions
            for i in range(min(len(refined_population), len(population) // 4)):
                population[i] = refined_population[i]
    
    # Final refinement of the best solution found
    if best_individual is not None:
        try:
            def objective(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
                
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2.0

            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_individual, method='SLSQP', 
                         bounds=bounds, constraints=cons, 
                         options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                best_individual = res.x
                
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = best_individual * (n_points / 2) / np.sum(best_individual)
            
            best_c5 = evaluate_erdos_solution(best_individual, dx)
            
        except Exception as e:
            pass
    
    final_individual = best_individual.copy()
    final_c5 = best_c5
    
    return (list(final_individual), final_c5, n_points)
