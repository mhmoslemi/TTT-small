import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Coarse-to-fine optimization approach
    n_points_list = [20, 40]
    best_h = None
    best_c5 = float('inf')
    current_n = 20
    
    # Initialize with parent construction or uniform
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start[:current_n]
        # Resize to current_n using linear interpolation if necessary
        if len(h_start) < current_n:
            x_old = np.linspace(0, 2, len(h_start))
            x_new = np.linspace(0, 2, current_n)
            h_start = np.interp(x_new, x_old, h_start)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (current_n/2) / np.sum(h_start)
    else:
        h_start = np.ones(current_n) * 0.5
        
    dx = 2.0 / current_n
    
    # Simulated Annealing for coarse grid
    max_iter = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    h_current = h_start.copy()
    current_c5 = evaluate_erdos_solution(h_current, dx)
    if current_c5 < best_c5:
        best_h = h_current.copy()
        best_c5 = current_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        h_pert = h_current + np.random.normal(0, perturb_mag, current_n)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (current_n/2) / np.sum(h_pert)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < math.exp(-(new_c5 - current_c5) / temp):
            h_current = h_pert
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = current_c5
            
        temp *= cool_rate
    
    # Upsample to 40 points
    if current_n < 40:
        h_upsampled = np.zeros(40)
        x_old = np.linspace(0, 2, current_n)
        x_new = np.linspace(0, 2, 40)
        for i in range(40):
            h_upsampled[i] = np.interp(x_new[i], x_old, best_h)
        best_h = h_upsampled
        current_n = 40
        dx = 2.0 / current_n
    
    # Fine optimization with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 100:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - current_n/2
            
        bounds = [(0, 1) for _ in range(current_n)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # Final refinement with projected gradient descent
    if remaining_time > 50:
        h_refined = best_h.copy()
        lr = 0.01
        for i in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros(current_n)
            # Estimate gradient via finite differences
            eps = 1e-4
            current_obj = evaluate_erdos_solution(h_refined, dx)
            for j in range(current_n):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (current_n/2) / np.sum(h_plus)
                obj_plus = evaluate_erdos_solution(h_plus, dx)
                grad[j] = (obj_plus - current_obj) / eps
            
            # Update with gradient descent
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (current_n/2) / np.sum(h_new)
            
            # Check for improvement
            new_obj = evaluate_erdos_solution(h_new, dx)
            if new_obj < current_obj:
                h_refined = h_new
                if new_obj < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = new_obj
            else:
                lr *= 0.995  # Reduce learning rate
    
    return (list(best_h), best_c5, current_n)
