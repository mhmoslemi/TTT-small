import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Use CVXPY for convex optimization
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
    
    # Initial dual variables
    lambda_vars = [cp.Variable(pos=True) for _ in range(2*n_points-1)]
    
    def solve_convex_relaxation(h_current, lambda_vals):
        # Compute C_k for k = -n+1 to n-1
        # We'll computes this explicitly in loop below
        
        # For each k, we compute h_k = shift(h) and evaluate h^T (1-h_k)
        objective_terms = []
        for k in range(-n_points+1, n_points):
            # Construct circular shift
            k_shift = k
            # create shift indices
            if k_shift >= 0:
                h_shifted = np.concatenate([h_current[k_shift:], h_current[:k_shift]])
            else:
                k_shift = -k_shift
                h_shifted = np.concatenate([h_current[k_shift:], h_current[:-k_shift]])
            
            # Compute C_k(h)
            C_k = np.dot(h_current, 1 - h_shifted)
            
            if k < 0:
                k_idx = k + 2*n_points - 2  # Match the full correlation index
            else:
                k_idx = k
            objective_terms.append(lambda_vals[k_idx] * C_k)
        
        objective = cp.sum(objective_terms) - cp.sum(cp.entr(lambda_vars))
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_var.value
        except:
            pass
        return h_current
    
    # Main optimization loop
    max_iter = 200
    dual_learning_rate = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update dual variables based on current h
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        for k_idx in range(len(lambda_vars)):
            new_val = max(lambda_vars[k_idx].value * (corr[k_idx] / lambda_vars[k_idx].value), 1e-6)
            lambda_vars[k_idx].value = new_val
            
        # Optimize h with current duals
        h_new = solve_convex_relaxation(best_h, [var.value for var in lambda_vars])
        
        if h_new is not None:
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            curr_c5 = evaluate_erdos_solution(h_new, dx)
            
            if curr_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = curr_c5

    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
