import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with pairwise perturbations
    def neighbor(h, step_size):
        h_new = h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-step_size, step_size)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Adjust to maintain sum constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        return h_new
    
    current_h = h_start.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    
    # Track best
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.9995
    step_base = 0.05
    max_steps = 20000
    step_count = 0
    
    while temperature > min_temp and (time() - start_time) < budget_s * 0.9:
        # Adaptive step size
        step_size = step_base * temperature
        
        # Generate neighbor
        new_h = neighbor(current_h, step_size)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept with SA probability
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
        
        # Periodically refine with convex optimization
        if step_count % 500 == 0 and (time() - start_time) < budget_s * 0.9:
            try:
                # Define convex variables and constraints
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Convex relaxation for best objective so far
                # We'll add a constraint to bound the overlap
                max_corr = cp.Variable()
                # Use main objective as soft constraint
                max_corr_val = evaluate_erdos_solution(best_h, dx) * 1.1 # Safety margin
                
                # Add some constraints on relevant correlations
                for k in range(0, min(15, 2 * n_points - 1), 2):
                    if k < n_points:
                        # Sum of products for correlation k 
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= max_corr_val * dx)
                
                # Minimize max overlap
                constraints.append(max_corr >= cp.max(cp.multiply(h[:n_points-1], 1-h[1:])))
                prob = cp.Problem(cp.Minimize(max_corr), constraints)
                
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
            except:
                pass
    
    # Final SLSQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
