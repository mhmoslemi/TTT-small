import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    tol = 1e-5
    lambda_low = 0.0
    lambda_high = 1.0
    lambda_mid = 0.5
    
    # Gradient descent parameters
    max_iter = 1000
    lr = 0.1
    tolerance = 1e-6
    
    def compute_correlation(h, mode='full'):
        corr = np.correlate(h, 1 - h, mode=mode)
        return corr * dx
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Helper to get the k that minimizes correlation
    def get_min_correlation_index(corr_full):
        # The middle index is zero shift
        mid = len(corr_full) // 2
        min_idx = np.argmin(corr_full)
        # Convert to actual shift
        shift = min_idx - mid
        return shift, corr_full[min_idx]
    
    # Main binary search loop
    while time() - start_time < budget_s * 0.95 and lambda_high - lambda_low > tol:
        lambda_mid = (lambda_low + lambda_high) / 2.0
        # Try to find h satisfying the constraint
        current_h = best_h.copy()
        current_h = project_h(current_h)
        
        # Use gradient descent to try to satisfy the constraint
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.9:
                break
                
            corr_full = compute_correlation(current_h)
            shift_k, min_corr = get_min_correlation_index(corr_full)
            
            if min_corr >= lambda_mid:
                # Feasible, update best solution
                c5 = evaluate_erdos_solution(current_h, dx)
                if c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = c5
                break
            
            # Compute gradient of the minimum
            # Gradient of integral of h(x)(1-h(x+k)) with respect to h(x)
            # = 1 - h(x+k) - h(x) (for x in range of valid overlap)
            grad = np.zeros(n_points)
            for i in range(n_points):
                # At position i, the contribution to overlap at k is h[i] * (1-h[i+k])
                # So, d/dh[i] = 1 - h[i+k] - h[i] for overlap at shift k
                k = shift_k
                if 0 <= i + k < n_points:
                    grad[i] = 1 - current_h[i+k] - current_h[i]
                else:
                    # This is for boundary conditions - ignoring for simplicity
                    grad[i] = 1 - current_h[(i+k) % n_points] - current_h[i]
            
            # Take gradient step
            current_h -= lr * grad
            current_h = project_h(current_h)
        
        # Check if solution is feasible
        corr_full = compute_correlation(current_h)
        shift_k, min_corr = get_min_correlation_index(corr_full)
        
        if min_corr >= lambda_mid:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid

    # Final optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass

    return (list(best_h), best_c5, n_points)
