import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[5:10] = 1.0
        h_start[25:30] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse grid optimization (n_points = 20)
    n_coarse = 20
    h_coarse = h_start[::n_points//n_coarse]
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    dx_coarse = 2.0 / n_coarse
    
    # Initialize penalty terms
    penalties = np.ones(2*n_coarse - 1)
    max_iter = 100
    penalty_lr = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        
        # Update penalties
        for k in range(len(penalties)):
            penalties[k] = max(penalties[k] * (corr[k] / penalties[k]), 1e-6)
        
        # Solve convex optimization problem
        h_var = cp.Variable(n_coarse)
        obj = cp.Minimize(cp.sum(cp.pos(corr + penalties)))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_coarse/2]
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_coarse = h_new
        except:
            pass
            
        # Re-evaluate and update best
        c5_new = evaluate_erdos_solution(h_coarse, dx_coarse)
        if c5_new < best_c5:
            best_h = h_coarse.copy()
            best_c5 = c5_new
    
    # Upsample to fine grid
    n_fine = 40
    upsampled_h = np.interp(
        np.linspace(0, 2, n_fine),
        np.linspace(0, 2, n_coarse),
        h_coarse
    )
    upsampled_h = upsampled_h * (n_fine/2) / np.sum(upsampled_h)
    
    # Fine grid optimization
    dx_fine = 2.0 / n_fine
    h_fine = upsampled_h.copy()
    penalties = np.ones(2*n_fine - 1)
    penalty_lr = 0.2
    
    for iteration in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        
        # Update penalties - use L2 for refinement
        for k in range(len(penalties)):
            penalties[k] = max(penalties[k] * (corr[k] / penalties[k]), 1e-6)
        
        # Solve convex optimization
        h_var = cp.Variable(n_fine)
        obj = cp.Minimize(cp.sum(cp.pos(corr + penalties)))
        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_fine/2]
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_fine = h_new
        except:
            pass
            
        # Re-evaluate and update best
        c5_new = evaluate_erdos_solution(h_fine, dx_fine)
        if c5_new < best_c5:
            best_h = h_fine.copy()
            best_c5 = c5_new

    # Final refinement using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        def obj(h):
            return evaluate_erdos_solution(h, dx_fine)
            
        def con_sum(h):
            return np.sum(h) - n_fine/2
            
        bounds = [(0, 1) for _ in range(n_fine)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_fine, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx_fine)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_fine)
