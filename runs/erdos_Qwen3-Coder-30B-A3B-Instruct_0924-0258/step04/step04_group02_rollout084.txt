from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

            h = compute_h_from_z(z + b)
            return (np.sum(h) - target_sum) ** 2
        # Binary search approach
        low_b, high_b = -10.0, 10.0
        for _ in range(20):
            mid_b = (low_b + high_b) / 2.0
            if obj(mid_b) < 1e-10:
                return mid_b
            elif np.sum(compute_h_from_z(z + low_b)) > target_sum:
                high_b = mid_b
            else:
                low_b = mid_b
        return (low_b + high_b) / 2.0

    b = find_b(z, n_points / 2)
    h = compute_h_from_z(z + b)
    
    # Ensure it satisfies constraint
    h = h * (n_points / 2) / np.sum(h)
    z = np.log(h / (1 - h + 1e-12))

    max_iterations = 5000

        # Double-check by re-evaluating final solution

    return (list(best_h), best_c5, n_points)
