import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with 3 clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.4, 1.0, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / cluster_size
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm Parameters
    population_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_perturb = h_start + np.random.normal(0, 0.02, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
        population.append(h_perturb)
    
    # Main evolution loop
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(c5_val)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: top 50%
        elite_size = population_size // 2
        elite = population[:elite_size]
        
        # Create new population through crossover and mutation
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(elite) >= 2:
                # Tournament selection
                idx1, idx2 = np.random.choice(len(elite), 2, replace=False)
                parent1, parent2 = elite[idx1], elite[idx2]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                new_population.extend([child1, child2])
            else:
                # Mutation: random perturbation
                parent_idx = np.random.randint(0, len(elite))
                child = population[parent_idx].copy()
                if np.random.rand() < mutation_rate:
                    # Gaussian perturbation
                    perturbation = np.random.normal(0, 0.03, n_points)
                    child += perturbation
                    child = np.clip(child, 0, 1)
                    child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Refinement with SQP for top candidates
        for i in range(min(5, len(population))):
            h_refined = population[i].copy()
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_refined, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                    population[i] = h_refined
            except:
                pass
    
    # Final cleanup to make sure it's feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
