import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        # Create a structured initial guess with a more sophisticated pattern
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place a few spikes in strategic locations
        spikes = [0.4, 0.6, 1.4, 1.6]
        for i, pos in enumerate(spikes):
            start = max(0, int((pos - 0.15)/dx))
            end = min(n_points, int((pos + 0.15)/dx))
            h_start[start:end] = 0.5 * (end - start) / n_points
        # Normalize sum
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = None
    
    # Binary search to find tighter bound on C5
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iterations = 60
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
        mid = (low + high) / 2.0
        C = mid / dx
        
        # CVXPY approach for feasibility check
        h = Variable(n_points)
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2.0
        ]
        
        # Overlap constraints for all possible lags
        # For each lag k, the constraint becomes:
        # sum over a of h[a] * (1 - h[a + k]) <= C
        # where a + k is wrapped around. We use explicit indexing
        # to make all valid index combinations.
        # This can be formulated in constraints:
        for k in range(n_points):
            # Get indices for the shifted version of h
            # For lag k, we're computing sum(h[a] * (1 - h[a+k])) where indices wrap
            # This is a bit tricky; but since h is assumed to be circular
            # we simulate this using np.roll for indexing
            # Let's form it carefully
        
            # A better idea is to build the matrix for each lag
            # But we're really constraining the overlap over all fixed lags
            # Cluster overlap constraints so we can handle efficiently
            # Let us compute correlation constraint directly over fixed lags
            
            # For any fixed k, we want sum(h[i] * (1 - h[(i+k)%n])) <= C
            # Instead of loop over shifting, use valid ranges
            lag_constraints = []
            for i in range(n_points):
                j = (i + k) % n_points
                lag_constraints.append(h[i] * (1 - h[j]))
            constraints.append(sum(lag_constraints) <= C)
        
        # Create and solve the optimization problem
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=CVXOPT, verbose=False)
            if prob.status == "optimal":
                # Then the value mid is feasible
                high = mid
                # Extract the solution h
                h_sol = np.array([h.value[i] for i in range(n_points)])
                if best_c5 is None or mid < best_c5:
                    best_c5 = mid
                    best_h = h_sol.copy()
            else:
                # Infeasible, raise lower bound
                low = mid
        except:
            # If CVXPY fails, narrow search
            low = mid
            
        # Barrier: check first that we can achieve better than 0.38085
        if high < 0.38085:
            break
    
    # If we didn't solve to this level, evaluate best h obtained
    if best_c5 is None:
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Final validation to make sure h is feasible
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
