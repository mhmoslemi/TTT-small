import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search bounds
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    
    # Track best solution during binary search
    best_feasible_h = None
    best_feasible_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        # LP feasibility check
        h_vars = Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2.0
        ]
        
        # Overlap constraints using linearization
        # For each lag k, we have: sum over a of h[a] * (1 - h[a+k]) <= mid/dx
        # This is non-linear; use relaxation or reformulation approach directly with optimization framework
        # The constraint is that for all valid shifts k:
        # sum_{a} h[a] * (1 - h[a+k]) <= mid/dx
        
        # Linearization is hard, so just solve using CVXPY
        # Since h is a variable, we can't directly compute correlation
        # Let's reformulate exactly as a constraint on the whole problem
        # Use the fact that all constraints of form sum h[i]* (1 - h[i+k]) <= C/dx
        # will result into a set of linear inequalities when h is binary or linear in the constraints
        
        # Since we must reformulate or use a direct computational check, this current approach method to iterate 
        # and refine with LP may not be realistic to include in one time-step.
        # Direct empirical optimization is needed for performance across different stages.
        
        # Instead of reinventing, we use the existing and proven approach with fine tunning:
        
        # Filter for efficiency when overlaps are small
        if mid >= best_c5:
            high = mid
            continue
            
        try:
            # Direct search with a heuristic restart approach in CLS-based relaxations would be more robust
            # Here let's proceed with LSQN or direct perturbation starting at current best,
            # in another role to improve near frontier solutions.
            
            # Use geometrical iterative refinement with random diversification:
            best_local_score = float('inf')
            best_local_h = h_start.copy()
            
            for i in range(5):
                # Start from modified version of best
                h_cand = best_h.copy()
                # Perturb with discrete adds
                if i > 0:
                    # small randomize added on sign/norm
                    indices = np.random.choice(n_points, size=n_points // 10, replace=False)
                    for k in indices:
                        h_cand[k] = max(0, min(1, h_cand[k] + np.random.normal(0, 0.02)))
                    h_cand = h_cand * (n_points/2) / np.sum(h_cand)
                
                # Optimize using scipy if still valid
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                constraints = [{'type': 'eq', 'fun': constraint_sum}]
                try:
                    # Use L-BFGS-B for max efficiency with bounds
                    res = minimize(obj, h_cand, method='L-BFGS-B', 
                                   bounds=bounds, constraints=constraints, 
                                   options={'maxiter': 100, 'ftol': 1e-6}))
                    if res.success:
                        h_local = res.x
                        c5_local = obj(h_local)
                        if c5_local < best_local_score:
                            best_local_score = c5_local
                            best_local_h = h_local.copy()
                except:
                    pass
                    
            if best_local_score < best_c5:
                best_h = best_local_h
                best_c5 = best_local_score
                
        except Exception:
            pass
        
        # Update bounds
        if best_c5 <= mid:
            high = mid
            if best_c5 < best_feasible_c5:
                best_feasible_c5 = best_c5
                best_feasible_h = best_h.copy()
        else:
            low = mid
            
        if abs(high - low) < precision:
            break

    # Final verification (maybe there is a better approach)
    try:
        final_h = best_h.copy()
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_feasible_c5:
            best_feasible_h = final_h
            best_feasible_c5 = final_c5
        return (list(best_feasible_h), best_feasible_c5, n_points)
    except:
        return (list(best_h), best_c5, n_points)
