import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def is_feasible(h, n_points, dx):
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-10)

def repair_solution(h, n_points, dx):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    h = h * (1.0 / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based approach: combined genetic + local search
    n_points = 35
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 40
    population = []
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        # Add perturbations of the initial solution
        for _ in range(10):
            perturbed = h_start + np.random.normal(0, 0.02, n_points)
            population.append(repair_solution(perturbed, n_points, dx))
        
        # Add some random valid solutions
        for _ in range(5):
            h = np.random.rand(n_points)
            h = repair_solution(h, n_points, dx)
            population.append(h)
    else:
        # Start from clean initializations
        for _ in range(15):
            h = np.random.rand(n_points)
            h = repair_solution(h, n_points, dx)
            population.append(h)
    
    # Add diversity with uniform solution
    uniform_h = np.ones(n_points) * 0.5
    uniform_h = repair_solution(uniform_h, n_points, dx)
    population.append(uniform_h)
    
    # Default values
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_list = []
    tabu_limit = 50
    stagnation_count = 0
    best_c5 = float('inf')
    best_h = None
    
    def get_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def mutate(h):
        # Create mutation
        h_mut = h.copy()
        mask = np.random.rand(n_points) < mutation_rate
        h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
        return repair_solution(h_mut, n_points, dx)
    
    def crossover(h1, h2):
        # Uniform crossover
        mask = np.random.rand(n_points) < 0.5
        h_child = np.where(mask, h1, h2)
        return repair_solution(h_child, n_points, dx)
    
    # Main GA loop
    max_gens = 5000
    gen = 0
    
    while gen < max_gens and (time() - start_time) < budget_s * 0.98:
        # Evaluate population
        fitness_values = [get_fitness(h) for h in population]
        
        # Track best
        for i, fitness in enumerate(fitness_values):
            if fitness < best_c5 and is_feasible(population[i], n_points, dx):
                best_c5 = fitness
                best_h = population[i].copy()
                # Early stopping if target is met
                if best_c5 < 0.3807:
                    break
        
        # Selection: tournament
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            candidates = random.sample(range(len(population)), tournament_size)
            best_candidate = min(candidates, key=lambda i: fitness_values[i])
            selected.append(population[best_candidate])
        
        # Reproduction step
        new_population = []
        for i in range(0, population_size, 2):
            if random.random() < crossover_prob and i+1 < len(selected):
                child1 = crossover(selected[i], selected[i+1])
                child2 = crossover(selected[i+1], selected[i])
                new_population.extend([child1, child2])
            else:
                new_population.extend([selected[i], selected[i+1]])
        
        # Mutation
        for i in range(len(new_population)):
            if random.random() < mutation_rate:
                new_population[i] = mutate(new_population[i])
        
        # Local refinement of top solutions
        if gen % 10 == 0:
            top_indices = np.argsort(fitness_values)[:5]
            for idx in top_indices:
                h_orig = population[idx].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_orig, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5 and is_feasible(h_refined, n_points, dx):
                            best_c5 = c5_refined
                            best_h = h_refined.copy()
                except:
                    pass
        
        # Update population
        population = new_population[:population_size]
        
        # Adaptive parameters
        if gen % 10 == 0 and gen > 0:
            if stagnation_count > 50:
                mutation_rate = min(0.5, mutation_rate + 0.1)
                crossover_prob = max(0.3, crossover_prob - 0.1)
                tabu_limit = max(10, tabu_limit // 2)
                stagnation_count = 0
            else:
                mutation_rate = max(0.05, mutation_rate - 0.05)
                crossover_prob = min(0.9, crossover_prob + 0.05)
        
        gen += 1
        
        # Check time
        if (time() - start_time) > budget_s * 0.95:
            break
    
    # Final refinement of best solution
    if best_h is not None:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5 and is_feasible(final_h, n_points, dx):
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # Ensure final solution is valid
    if best_h is None:
        # Fallback: create a valid solution
        best_h = np.ones(n_points) * 0.5
        best_h = repair_solution(best_h, n_points, dx)
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
