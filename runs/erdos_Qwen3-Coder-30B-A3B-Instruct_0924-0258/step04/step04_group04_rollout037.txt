import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iter = 60
    
    # Optimizer parameters
    max_feval = 1000
    learning_rate = 0.1
    backtracking_factor = 0.5
    min_step = 1e-6
    
    def compute_auto_correlation(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr * dx
    
    def is_feasible(h, lambda_val, dx):
        corr = compute_auto_correlation(h, dx)
        min_corr = np.min(corr)
        return min_corr >= lambda_val
    
    def project_h(h, dx):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-10:
            h = h * (n_points/2) / h_sum
        return h
    
    def compute_gradient(h, k_min):
        # Compute gradient of the auto-correlation with respect to h at shift k_min
        # For simplicity and efficiency, use finite differences
        eps = 1e-6
        n = len(h)
        grad = np.zeros(n)
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(n):
            h_plus[i] += eps
            h_minus[i] -= eps
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (corr_plus[k_min + n-1] - corr_minus[k_min + n-1]) / (2*eps)
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        return grad
    
    # Start of binary search
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        h_candidate = best_h.copy()
        
        # Attempt optimization for this lambda
        # Initialize with best known h
        h_current = h_candidate.copy()
        
        # Gradient descent to find feasible h
        for i in range(max_feval):
            if time() - start_time > budget_s * 0.95:
                break
                
            if is_feasible(h_current, lambda_mid, dx):
                if lambda_mid > lambda_low:
                    lambda_low = lambda_mid
                    best_h = h_current.copy()
                    best_c5 = evaluate_erdos_solution(best_h, dx)
                break
            
            # Compute current auto-correlation
            corr = compute_auto_correlation(h_current, dx)
            k_min = np.argmin(corr)
            min_corr = corr[k_min]
            
            # Compute gradient
            gradient = compute_gradient(h_current, k_min)
            
            # Gradient step
            step_size = learning_rate
            h_new = h_current - step_size * gradient
            
            # Backtracking line search
            while True:
                h_test = project_h(h_new, dx)
                corr_test = compute_auto_correlation(h_test, dx)
                test_min = corr_test[k_min]
                if test_min >= min_corr or step_size < min_step:
                    h_current = h_test
                    break
                step_size *= backtracking_factor
                h_new = h_current - step_size * gradient
                
            # Early termination if no improvement
            if abs(min_corr - np.min(corr)) < 1e-10:
                break
            
        else:
            # Feasible solution not found for lambda_mid
            lambda_high = lambda_mid
    
    # Final optimization of best h
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
