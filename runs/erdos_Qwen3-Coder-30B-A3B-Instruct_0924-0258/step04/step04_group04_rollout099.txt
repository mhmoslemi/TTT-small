import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.zeros(n_points)
        h_start[5:15] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse grid optimization (20 points)
    for iter_count in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlations
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        
        # CVXPY approach to solve penalty problem
        h_var = cp.Variable(n_points)
        penalty = cp.Variable(len(corr))
        
        # Objective: minimize maximum of (correlation + penalty)
        objective = cp.Minimize(cp.max(corr + penalty))
        
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Use a smoother approach for updating penalty terms
        # Just run a single step with current approximates
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                new_h = h_var.value
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                new_c5 = evaluate_erdos_solution(new_h, dx)
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
        except:
            pass
        
        # Update penalties (simple version to avoid complex constraint handling)
        # We upgrade the current solution with minor increase in complexity if improved
        # But if not, let's just use simple SQP for possible progress
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 20, 'ftol': 1e-6})
            
            if res.success:
                res_h = res.x
                res_h = np.clip(res_h, 0, 1)
                res_h = res_h * (n_points/2) / np.sum(res_h)
                res_c5 = evaluate_erdos_solution(res_h, dx)
                if res_c5 < best_c5:
                    best_h = res_h.copy()
                    best_c5 = res_c5
        except:
            pass
    
    # Upsampling to 40 points for better resolution
    if n_points < 40 and time() - start_time < budget_s * 0.95:
        n_points = 40
        dx = 2.0 / n_points
        f = interp1d(np.linspace(0, 2, len(best_h)), best_h, kind='linear')
        x_new = np.linspace(0, 2, n_points)
        h_upsampled = f(x_new)
        h_upsampled = np.clip(h_upsampled, 0, 1)
        h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
        best_h = h_upsampled
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
        # Fine optimization with more iterations on 40 points
        for iter_count in range(150):
            if time() - start_time > budget_s * 0.95:
                break
                
            try:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                    
                def con_sum(h):
                    return np.sum(h) - n_points/2
                    
                bounds = [(0, 1) for _ in range(n_points)]
                cons = {'type': 'eq', 'fun': con_sum}
                
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 30, 'ftol': 1e-6})
                
                if res.success:
                    res_h = res.x
                    res_h = np.clip(res_h, 0, 1)
                    res_h = res_h * (n_points/2) / np.sum(res_h)
                    res_c5 = evaluate_erdos_solution(res_h, dx)
                    if res_c5 < best_c5:
                        best_h = res_h.copy()
                        best_c5 = res_c5
            except:
                pass
    
    # Final refinement stage with higher precision and gradient-based method
    if time() - start_time < budget_s * 0.9:
        try:
            # Clear the possible noise from optimization and use granular method
            n_points_refine = 35
            dx_ref = 2.0 / n_points_refine
            # Generate a refined sequence with step optimization
            h_refine = np.zeros(n_points_refine)
            
            # Use the best previous result as a seed for interpolation
            f = interp1d(np.linspace(0, 2, len(best_h)), best_h, kind='linear')
            x_new = np.linspace(0, 2, n_points_refine)
            h_refine = f(x_new)
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points_refine/2) / np.sum(h_refine)
            
            # Final optimization
            def obj(h):
                return evaluate_erdos_solution(h, dx_ref)
                
            def con_sum(h):
                return np.sum(h) - n_points_refine/2
                
            bounds = [(0, 1) for _ in range(n_points_refine)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            # Try a few different solvers to get precision
            try:
                res = minimize(obj, h_refine, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    final_h = res.x
                    final_h = np.clip(final_h, 0, 1)
                    final_h = final_h * (n_points_refine/2) / np.sum(final_h)
                    final_c5 = evaluate_erdos_solution(final_h, dx_ref)
                    if final_c5 < best_c5:
                        best_h = final_h
                        best_c5 = final_c5
            except:
                pass
        except:
            pass
    
    # Final return
    return (list(best_h), best_c5, len(best_h))
