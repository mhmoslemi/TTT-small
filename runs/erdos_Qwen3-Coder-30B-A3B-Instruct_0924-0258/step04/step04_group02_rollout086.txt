import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize Fourier coefficients
    n_coeffs = n_points // 2 + 1
    a = np.random.normal(0, 0.1, n_coeffs)
    b = np.random.normal(0, 0.1, n_coeffs-1)
    
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def compute_h(a_coeffs, b_coeffs):
        x = np.linspace(0, 2, n_points, endpoint=False)
        # Build Fourier series
        h_fourier = a_coeffs[0] * np.ones(n_points)
        for k in range(1, n_coeffs):
            h_fourier += a_coeffs[k] * np.cos(k * np.pi * x) + b_coeffs[k-1] * np.sin(k * np.pi * x)
        h = sigmoid(h_fourier)
        return h
    
    def compute_overlap_spectrum(h):
        # Compute FFT of h and (1-h)
        H = fft(h)
        H_conj = np.conj(H)
        OneMinusH = fft(1 - h)
        # Convolution in frequency domain
        overlap_spectrum = H_conj * OneMinusH
        # Inverse FFT to get overlap convolution
        overlap_conv = ifft(overlap_spectrum).real
        return overlap_conv
    
    # Optimization loop
    learning_rate = 0.01
    reg_weight = 0.001
    
    max_iterations = 5000
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute h from coefficients
        h = compute_h(a, b)
        # Normalize sum to 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlaps
        overlap_conv = compute_overlap_spectrum(h)
        # We are interested in shifts from 0 to n_points-1 spanning from k=0 to k=n_points-1
        # The full convolution has size 2*n_points-1, we want the overlapping part
        # The actual overlap we care about is over shift k from 0 to n_points-1
        # The middle element of overlap_conv (index n_points-1) corresponds to zero shift
        # from index 0 to n_points-1 (corresponds to ks from 0 to n_points-1)
        # This is equivalent to full convolution of h with (1-h) 
        # But shifted so that indices [0, n_points - 1] correspond to shifts [0, n_points - 1]
        overlap_values = overlap_conv[:n_points] * dx  # Get the first n_points values which is the full overlap
        max_overlap_k = np.argmax(overlap_values)
        max_overlap = overlap_values[max_overlap_k]
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient with respect to Fourier coefficients
        # Use finite differences for simplicity in this implementation
        eps = 1e-5
        grad_a = np.zeros_like(a)
        grad_b = np.zeros_like(b)
        
        # Compute base gradient for all coefficients
        h_base = compute_h(a, b)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        overlap_base = compute_overlap_spectrum(h_base)
        base_value = overlap_base[max_overlap_k] * dx
        
        # Compute gradient for a
        for i in range(len(a)):
            a_delta = a.copy()
            a_delta[i] += eps
            h_plus = compute_h(a_delta, b)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlap_plus = compute_overlap_spectrum(h_plus)
            plus_value = overlap_plus[max_overlap_k] * dx
            grad_a[i] = (plus_value - base_value) / eps
        
        # Compute gradient for b
        for i in range(len(b)):
            b_delta = b.copy()
            b_delta[i] += eps
            h_plus = compute_h(a, b_delta)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlap_plus = compute_overlap_spectrum(h_plus)
            plus_value = overlap_plus[max_overlap_k] * dx
            grad_b[i] = (plus_value - base_value) / eps
        
        # Update coefficients
        a -= learning_rate * (grad_a + reg_weight * a)
        b -= learning_rate * (grad_b + reg_weight * b)
        
        # Adjust learning rate
        learning_rate *= 0.99999
    
    # Final refinement with local minimization
    def objective(coeffs):
        aa = coeffs[:n_coeffs]
        bb = coeffs[n_coeffs:]
        h = compute_h(aa, bb)
        h = h * (n_points / 2) / np.sum(h)  # Normalize
        return evaluate_erdos_solution(h, dx)
    
    coeffs = np.concatenate([a, b])
    result = minimize(objective, coeffs, method='L-BFGS-B', bounds=[(None,None) for _ in coeffs], options={'maxiter':500})
    if result.success:
        best_a = result.x[:n_coeffs]
        best_b = result.x[n_coeffs:]
        h_final = compute_h(best_a, best_b)
        h_final = h_final * (n_points / 2) / np.sum(h_final)  # Normalize
        final_c5 = evaluate_erdos_solution(h_final, dx)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
