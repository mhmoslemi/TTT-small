import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with pairwise perturbations
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.999
    max_steps = 10000
    step_base = 0.05
    
    # Track best solution found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Generate pairwise neighbor
        h_new = current_h.copy()
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        # Step size based on temperature
        step = step_base * temperature
        
        # Ensure different indices
        while idx2 == idx1:
            idx2 = np.random.randint(0, n_points)
        
        # Perturb two points in opposite directions
        delta = np.random.uniform(-step, step)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        
        # Ensure bounds
        h_new[idx1] = np.clip(h_new[idx1], 0, 1)
        h_new[idx2] = np.clip(h_new[idx2], 0, 1)
        
        # Adjust for sum constraint
        sum_diff = np.sum(h_new) - n_points / 2.0
        if abs(sum_diff) > 1e-10:
            # Distribute the adjustment evenly
            adjustment = sum_diff / n_points
            h_new -= adjustment
            h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = h_new
            current_c5 = new_c5
            
            # Check if it's better than best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
        
        # Occasionally refine with convex optimization
        if step_count % 100 == 0 and (time() - start_time) < budget_s * 0.9:
            try:
                # More robust convex refinement
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Constraint to ensure the max overlap doesn't exceed current bound
                # Manually compute and add constraints from correlations (simplified)
                # We'll just refine with few strong constraints from highest overlaps
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                max_k = np.argmax(corr)
                max_value = corr[max_k] * dx
                
                # Add a constraint to directly bound the maximum overlap
                # This helps guide the search earlier on
                if max_value < best_c5 * 1.1:  # Only add if tight enough
                    # Add a few simple overlap constraints
                    for i in range(min(5, n_points)):
                        if i < n_points:
                            # Constrain one lag
                            corr_sum = cp.sum(cp.multiply(h[:n_points-i], 1 - h[i:]))
                            constraints.append(corr_sum <= max_value * 2)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        h_refined = np.clip(h_refined, 0, 1)
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Slightly perturb the current solution
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            # Ensure it stays within bounds
            h_restart = np.clip(h_restart, 0, 1)
            # Adjust sum to be correct
            h_restart *= (n_points / 2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
