import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for i in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            h += np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to sum=1
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        obj = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_vals.append(obj)
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to valid range [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Maintain sum constraint
            particles[i] *= (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new solution
            new_obj = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if new_obj < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = new_obj
                
                # Update global best
                if new_obj < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = new_obj
                    best_h = global_best.copy()
                    best_c5 = global_best_val
    
    # Convex refinement step
    if time() - start_time < budget_s * 0.9:
        try:
            # Reformulate using CVXPY to tighten bound
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add overlap constraints using upper bound from PSO
            max_k = 2 * n_points - 1
            h_full = np.concatenate([h, np.zeros(n_points - 1)])
            h_reversed = np.concatenate([np.zeros(n_points - 1), h])
            for k in range(0, max_k, 2):
                if k == 0:
                    corr = cp.sum(cp.multiply(h, 1 - h))
                elif k < n_points:
                    corr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr <= global_best_val / dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final polish with SLSQP
    if time() - start_time < budget_s * 0.9:
        remaining_budget = budget_s - (time() - start_time)
        if remaining_budget > 20:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, best_h, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
