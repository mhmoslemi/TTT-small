import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    max_iter = 500
    step_size = 0.01
    patience = 100
    
    def compute_gradient(h, k):
        # Gradient of overlap w.r.t. h for shift k
        n = len(h)
        grad = np.zeros(n)
        if k == 0:
            # For k=0, it's just -h[i] at each point
            grad = -h
        else:
            # Otherwise we compute based on correlation derivative
            # This is an approximation assuming shift k is fixed
            # We construct correlation vector manually
            corr = np.correlate(h, 1-h, mode='full')
            max_k = len(corr) - 1
            # This simplifies to: 1 - h[i+k] for position i+k for positive shifts
            # and h[i-k] for negative shifts. We only compute gradient for fixed shift k
            for i in range(n):
                if i+k >= 0 and i+k < n:
                    grad[i] = -(1 - h[i+k])
                else:
                    grad[i] = 0
                if i-k >= 0 and i-k < n:
                    grad[i] += h[i-k]
        return grad
    
    def backtracking_line_search(h, grad, k, alpha=0.01, beta=0.5):
        # Perform backtracking line search to find good step size for gradient descent
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        while new_c5 >= evaluate_erdos_solution(h, dx):
            alpha *= beta
            if alpha < 1e-10:
                break
            new_h = h - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
        return new_h
    
    patience_count = 0
    for iteration in range(max_iter):
        now = time()
        if now - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps to find max
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_overlap = np.max(corr)
        max_k = np.argmax(corr) - (len(best_h) - 1)
        
        # Compute gradient
        gradient = compute_gradient(best_h, max_k)
        
        # Apply step with line search
        new_h = backtracking_line_search(best_h, gradient, max_k)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            break
    
    # Final eval
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
