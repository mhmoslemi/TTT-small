import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolution Strategy parameters
    pop_size = 50
    elite_size = 5
    mutation_strength = 0.1
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Augmented Lagrangian parameters
    lambda_sum = 0.0
    lambda_bounds = np.zeros(n_points)
    rho = 1.0
    
    # Evolutionary optimization loop
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Select elite
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            
            parent1 = elite[parent1_idx]
            parent2 = elite[parent2_idx]
            
            # Blend crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Gaussian mutation
            noise = np.random.normal(0, mutation_strength, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            
            # Normalize to satisfy constraint
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Adaptive mutation strength
        if generation % 10 == 0:
            fitness_std = np.std(fitness_scores[:elite_size])
            if fitness_std < 1e-4:
                mutation_strength *= 0.95
            elif fitness_std > 1e-2:
                mutation_strength *= 1.05
                
        # Increase rho slightly for augmented Lagrangian
        rho = min(rho * 1.01, 100.0)
        
        generation += 1
    
    # Final refinement with local optimization
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
        
        # SLSQP local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_refined, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-7})
        
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        else:
            h_final = h_refined
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
