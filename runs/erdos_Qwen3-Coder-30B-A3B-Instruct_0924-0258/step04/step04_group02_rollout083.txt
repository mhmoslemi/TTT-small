import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Reparameterize h = sigmoid(z + b) and adjust b to satisfy constraint
    def reparameterize(z):
        h = 1 / (1 + np.exp(-z))
        return h
    
    def compute_b(h):
        # Adjust b to make sum(h) = n_points / 2
        # This is a bit tricky due to the sigmoidal nature
        # We'll only improve upon the constraint satisfaction
        return 0  # Just recompute using normalization below
    
    def normalize_h(h):
        sum_h = np.sum(h)
        if sum_h > 0:
            return h * (n_points / 2) / sum_h
        return h
    
    def compute_gradients(z):
        h = reparameterize(z)
        h = normalize_h(h)
        
        # Compute overlap for all shifts and find maximum
        h1 = np.concatenate([h, np.zeros(n_points - 1)])
        h2 = np.concatenate([np.zeros(n_points - 1), 1 - h])
        corr = np.convolve(h1, h2, mode='valid')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Now compute the gradient with respect to h at shift k_max
        # We need gradient of C_k_max = sum(h[i] * (1 - h[i+k_max]))
        # dC/dh[i] = (1 - h[i + k_max]) + (1 - h[i - k_max]) 
        # But we evaluate only the relevant shift k_max
        
        # For full correlation, we compute it manually
        # Now we compute: dC_kmax / dh
        dC_dh = np.zeros(n_points)
        for i in range(n_points):
            # Soft derivative (not exact, but approximates one)
            # It has to be: if k_max > 0, then neighbor to the right
            # if k_max < n_points-1 then neighbor to the left
            # So align both sides appropriately
            
            # Actually use numerical differentiation to get gradient w.r.t. current h
            # To avoid analytical complexity for now, we approximate with small perturbations
            if k_max < n_points and k_max > 0:
                # Let's say we're computing overlap with h at shift k_max
                # dC/dh[i] = (1 - h[i + k_max]) if i+k_max exists, else 0
                # Also there's contribution from h[i-k_max], if that exists 
                # But it's actually:
                # C = sum_i h[i] * (1 - h[i + k_max])
                # dC/dh[i] = 1 - h[i + k_max] (when i+k_max < n)
                if i + k_max < n_points:
                    dC_dh[i] = 1 - h[i + k_max]
                else:
                    dC_dh[i] = 0
            
            # Important: 
            # The shift k_max can be 0 to 2*(n_points - 1) 
            # But relevant summary is: C_k = sum_{i=0}^{n-1} h[i] * (1 - h[i+k])
            # Which we're interpreting as convolution: h * (1 - h) but with proper indexing
            # The overlap for each shift k = C_k = sum_{i=0}^{n-1} h[i] * (1 - h[i+k])  
            # So dC_k/dh[i] = (1 - h[i + k]) if i+k < n, else 0
            # So once k_max is identified:
            # Gradient dC_kmax/dh = [1 - h[i + k_max]] if appropriate else 0
            
        return dC_dh
    
    # Begin optimization
    # Initialize z appropriately
    z_best = np.log(h_start / (1 - h_start + 1e-8))  # logit transform
    z_best = np.clip(z_best, -20, 20)  # prevent overflow
    h_best = reparameterize(z_best)
    h_best = normalize_h(h_best)
    best_c5 = evaluate_erdos_solution(h_best, dx)
    
    # Optimization with gradient updates in logit space
    max_iter = 5000
    learning_rate = 0.05
    patience = 100  # early stopping
    patience_count = 0
    last_best = best_c5
    
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        h = reparameterize(z_best)
        h = normalize_h(h)
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            if patience_count > patience:
                break
                
        # Compute gradient in logit space approximately, manually for small error
        shifts = np.arange(n_points)
        mins = []
        for s in shifts:
            if s > 0:
                # Convolution appropriate for shift in the cross-correlation
                corr_i = np.correlate(h, 1 - np.roll(h, s), mode='valid')
            else:
                corr_i = np.correlate(h, 1 - h, mode='valid')
            mins.append(np.max(corr_i) * dx)
        
        # Find k_max such that overlap is maximum
        k_max = np.argmin(mins) 
        # For k_max we compute the gradient dC/dh= 1 - h[i+k_max] 
        dC_dh = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                dC_dh[i] = 1 - h[i + k_max]
        
        # Now transform gradient to logit space
        # dh/dz = sigmoid(z) (1 - sigmoid(z))
        # We approximate by evaluating at current z:
        sigma_z = reparameterize(z_best)
        grad_logit = dC_dh * sigma_z * (1 - sigma_z)
        
        # Update in logit space
        z_new = z_best - learning_rate * grad_logit
        z_best = z_new
        
        # Clip to prevent numerical issues
        z_best = np.clip(z_best, -20, 20)  # prevent overflow
        
        # Adjust learning rate if stagnated
        if abs(last_best - best_c5) < 1e-8 and learning_rate > 1e-5:
            learning_rate *= 0.8
        last_best = best_c5
        
    return (list(h_best), best_c5, n_points)
