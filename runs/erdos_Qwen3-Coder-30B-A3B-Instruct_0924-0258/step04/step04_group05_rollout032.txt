import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Linearly resample h_in to n_out points"""
    n_in = len(h_in)
    x_in = np.arange(n_in)
    x_out = np.linspace(0, n_in - 1, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, target_sum):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init, dx)
    
    # Coarse grid heuristics
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = linear_resample(h_init, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse / 2.0)
    
    # Fine grid
    n_fine = 100
    dx_fine = 2.0 / n_fine
    h_fine = linear_resample(h_init, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine / 2.0)
    
    # Adaptive scheduling
    max_coarse_steps = 150
    max_fine_steps = 200
    perturb_scale = 0.03
    perturb_decay = 0.99
    
    # Coarse search
    coarse_best_c5 = np.inf
    coarse_best_h = h_coarse.copy()
    
    for step in range(max_coarse_steps):
        if time() - start_time > budget_s * 0.7:
            break
            
        # Perturb
        h_pert = h_coarse + np.random.normal(0, perturb_scale, n_coarse) 
        h_pert = project_to_constraints(h_pert, n_coarse / 2.0)
        
        # Evaluate
        c5_test = evaluate_erdos_solution(h_pert, dx_coarse)
        
        # Update
        if c5_test < coarse_best_c5:
            coarse_best_c5 = c5_test
            coarse_best_h = h_pert.copy()
            
        # Adapt perturbation
        if step % 10 == 0:
            perturb_scale *= perturb_decay
    
    # Transfer to fine grid
    h_fine = linear_resample(coarse_best_h, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine / 2.0)
    
    # Fine-grid refinement
    fine_best_h = h_fine.copy()
    fine_best_c5 = evaluate_erdos_solution(h_fine, dx_fine)
    
    for step in range(max_fine_steps):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb with smaller scale
        h_pert = h_fine + np.random.normal(0, perturb_scale * 0.3, n_fine)
        h_pert = project_to_constraints(h_pert, n_fine / 2.0)
        
        # Evaluate
        c5_test = evaluate_erdos_solution(h_pert, dx_fine)
        
        # Update
        if c5_test < fine_best_c5:
            fine_best_c5 = c5_test
            fine_best_h = h_pert.copy()
            
        # Adapt perturbation
        if step % 10 == 0:
            perturb_scale *= perturb_decay
    
    # Further local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx_fine)
        
    def constraint_sum(h):
        return np.sum(h) - n_fine / 2.0
        
    bounds = [(0, 1) for _ in range(n_fine)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        res = minimize(objective, fine_best_h, method='SLSQP',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < fine_best_c5:
                fine_best_h = h_final
                fine_best_c5 = c5_final
    except:
        pass
    
    # Compensate with final evaluation
    final_c5 = evaluate_erdos_solution(fine_best_h, dx_fine)
    final_h = project_to_constraints(fine_best_h, n_points / 2.0)
    
    # Return to original resolution
    if len(final_h) != n_points:
        final_h = linear_resample(final_h, n_points)
        final_h = project_to_constraints(final_h, n_points / 2.0)
    
    return (list(final_h), final_c5, n_points)
