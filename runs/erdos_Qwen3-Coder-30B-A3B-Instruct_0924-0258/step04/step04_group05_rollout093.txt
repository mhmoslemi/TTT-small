import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 40
    population = []
    
    # Generate diverse initial population
    for i in range(10):
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
        
    for i in range(5):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
    
    # Initialize tabu search
    tabu_list = []
    tabu_limit = 50
    stagnation_counter = 0
    max_stagnation = 50
    
    # Adaptive parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Main genetic search loop
    gen = 0
    while time() - start_time < budget_s * 0.9:
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            f = evaluate_erdos_solution(h, dx)
            fitness.append(f)
        
        # Find best individual
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Check if we're close to target
        if best_c5 < 0.3807:
            break
            
        # Adaptive parameter adjustment
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        elif stagnation_counter > 0:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Select parents using tournament selection (size=3)
        new_population = []
        
        # Keep best individuals
        sorted_indices = np.argsort(fitness)
        for i in range(5):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament select two parents
            parent1_idx = random.choice(sorted_indices[:10])
            parent2_idx = random.choice(sorted_indices[:10])
            
            if random.random() < crossover_prob:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, population[parent1_idx], population[parent2_idx])
            else:
                # Clone one parent
                child = population[parent1_idx].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
                for i in mut_indices:
                    child[i] += np.random.normal(0, 0.02)
            
            # Repair solution
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Avoid tabu solutions
            child_tuple = tuple(child)
            if child_tuple not in tabu_list:
                new_population.append(child)
                tabu_list.append(child_tuple)
                if len(tabu_list) > tabu_limit:
                    tabu_list.pop(0)
        
        population = new_population[:population_size]
        
        # Local refinement every 10 generations
        if gen % 10 == 0:
            for i in range(min(5, len(population))):
                h_refined = population[i].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x.copy()
                        h_refined = np.clip(h_refined, 0, 1)
                        h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                        fitness[i] = evaluate_erdos_solution(h_refined, dx)
                        population[i] = h_refined
                except:
                    pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(lambda h: evaluate_erdos_solution(h, dx), best_h, method='SLSQP', 
                           bounds=[(0, 1) for _ in range(n_points)], 
                           constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
