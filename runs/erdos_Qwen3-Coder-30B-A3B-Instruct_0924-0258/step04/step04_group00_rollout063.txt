import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolution Strategy with Augmented Lagrangian
    population_size = 50
    mutation_strength = 0.1
    elite_size = 10
    stagnation_threshold = 50
    stagnation_count = 0
    best_fitness_history = []
    
    # Initialize population
    population = []
    fitnesses = []
    
    for _ in range(population_size):
        h = h_start + np.random.normal(0, mutation_strength, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
        fitnesses.append(evaluate_erdos_solution(h, dx))
    
    # Update best
    min_fitness_idx = np.argmin(fitnesses)
    if fitnesses[min_fitness_idx] < best_c5:
        best_h = population[min_fitness_idx].copy()
        best_c5 = fitnesses[min_fitness_idx]
    
    iteration = 0
    max_iter = int(budget_s * 0.8 / 0.05) if budget_s < 100 else 10000
    
    while iteration < max_iter and time() - start_time < budget_s * 0.95:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check stagnation
        if stagnation_count > stagnation_threshold:
            # Respawn some individuals with higher mutation
            for i in range(population_size//4, population_size):
                h = population[0] + np.random.normal(0, mutation_strength*2, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
                population[i] = h
                fitnesses[i] = evaluate_erdos_solution(h, dx)
        
        # Selection and reproduction
        elite = population[:elite_size]
        offspring = []
        
        # Keep elite
        offspring.extend(elite)
        
        # Generate offspring via mutation
        while len(offspring) < population_size:
            parent = np.random.choice(elite)
            child = parent + np.random.normal(0, mutation_strength, n_points)
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            offspring.append(child)
        
        # Evaluate offspring
        population = offspring
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Adapt mutation strength
        if iteration % 10 == 0:
            diversity = np.std([np.sum((p - population[0])**2) for p in population])
            if diversity > 0.1:
                mutation_strength *= 0.95
            elif diversity < 0.05:
                mutation_strength *= 1.05
            mutation_strength = np.clip(mutation_strength, 0.01, 0.5)
        
        iteration += 1
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
