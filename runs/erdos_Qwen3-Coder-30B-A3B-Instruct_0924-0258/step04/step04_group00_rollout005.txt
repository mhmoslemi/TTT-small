import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameters
    population_size = 100
    num_generations = 1000
    mutation_rate = 0.1
    tournament_size = 2
    elite_size = 5
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
        fitnesses.append(best_c5)
    
    # Fill with random individuals
    for _ in range(population_size - len(population)):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        population.append(h_rand)
        fitnesses.append(evaluate_erdos_solution(h_rand, dx))
    
    # Genetic algorithm

    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(ind, dx) for ind in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
        
        # Create new population
        new_population = []
        
        # Elite preservation
        elite_idx = np.argsort(fitnesses)[:elite_size]
        for i in elite_idx:
            new_population.append(population[i].copy())
        
        # Tournament selection, crossover and mutation
        while len(new_population) < population_size:
            # Selection
            parent1_idx = tournament_selection(population, fitnesses, tournament_size)
            parent2_idx = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            
            # Mutation
            mutate(child1, mutation_rate)
            mutate(child2, mutation_rate)
            
            # Ensure feasibility
            child1 = enforce_constraints(child1, n_points, dx)
            child2 = enforce_constraints(child2, n_points, dx)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
    
    # Final evaluation of best
    final_fitness = evaluate_erdos_solution(best_h, dx)
    if final_fitness < best_c5:
        best_c5 = final_fitness
    
    return (list(best_h), best_c5, n_points)

def tournament_selection(population, fitnesses, k):
    idxs = np.random.choice(len(population), size=k, replace=False)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return best_idx

def crossover(parent1, parent2):
    # Single-point crossover
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(individual, rate):
    for i in range(len(individual)):
        if np.random.rand() < rate:
            individual[i] += np.random.normal(0, 0.01)
    return individual

def enforce_constraints(h, n_points, dx):
    h = np.clip(h, 0, 1)
    h = h * (n_points/2) / np.sum(h)
    return h
