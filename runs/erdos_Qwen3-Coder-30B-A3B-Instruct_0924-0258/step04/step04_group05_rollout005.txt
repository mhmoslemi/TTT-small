import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Linearly resample h_in to n_out points"""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, target_sum):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialization
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init, dx)
    
    # Coarse grid setup
    n_coarse = 50
    h_coarse = linear_resample(h_init, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse / 2)
    dx_coarse = 2.0 / n_coarse
    
    # Fine grid setup
    n_fine = 100
    h_fine = linear_resample(h_init, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine / 2)
    dx_fine = 2.0 / n_fine
    
    # Coarse-grid global search
    n_coarse_steps = 100
    perturbation_scale = 0.03
    for step in range(n_coarse_steps):
        if time() - start_time > budget_s * 0.6:
            break
            
        # Perturb
        h_pert = h_coarse + np.random.normal(0, perturbation_scale, n_coarse)
        # Repair constraints
        h_pert = project_to_constraints(h_pert, n_coarse / 2)
        # Evaluate
        c5_coarse = evaluate_erdos_solution(h_pert, dx_coarse)
        if c5_coarse < best_c5:
            best_h = linear_resample(h_pert, n_points)
            best_c5 = c5_coarse
        # Update perturbation scale (adaptive decay)
        perturbation_scale *= 0.99
        
    # Fine-grid refinement
    n_fine_steps = 200
    perturbation_scale = 0.02
    for step in range(n_fine_steps):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Interpolate from best coarse solution
        h_fine = linear_resample(best_h, n_fine)
        # Perturb
        h_pert = h_fine + np.random.normal(0, perturbation_scale, n_fine)
        # Repair constraints
        h_pert = project_to_constraints(h_pert, n_fine / 2)
        # Evaluate
        c5_fine = evaluate_erdos_solution(h_pert, dx_fine)
        if c5_fine < best_c5:
            best_h = linear_resample(h_pert, n_points)
            best_c5 = c5_fine
        # Update perturbation scale
        perturbation_scale *= 0.995
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 300, 'ftol': 1e-9, 'eps': 1e-6})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
