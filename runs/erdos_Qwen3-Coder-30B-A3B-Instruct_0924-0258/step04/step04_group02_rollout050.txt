import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple random initializations to escape local minima
    num_restarts = 5
    for restart in range(num_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random restart with different strategy
        h_init = h_start.copy()
        if restart > 0:
            # Perturb with more diversity
            h_init += np.random.normal(0, 0.1, n_points)
            h_init = np.clip(h_init, 0, 1)
            # Normalize to correct sum
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            
        # Use scipy.optimize with different methods
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Try different optimization methods
            methods = ['SLSQP', 'L-BFGS-B']
            for method in methods:
                if time() - start_time > budget_s * 0.95:
                    break
                    
                h_test = h_init.copy()
                try:
                    res = minimize(objective, h_test, method=method, 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8})
                    if res.success:
                        h_candidate = res.x
                        c5_candidate = objective(h_candidate)
                        if c5_candidate < best_c5:
                            best_h = h_candidate
                            best_c5 = c5_candidate
                except:
                    continue
        except:
            continue

    # Final refinement with global method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Use gradient-based algorithm with multiple start points
        h_final = best_h.copy()
        h_refined = h_final.copy()
        best_refined_c5 = best_c5
        
        for _ in range(10):  # Multiple refinement attempts
            if time() - start_time > budget_s * 0.95:
                break
                
            h_test = h_final + np.random.normal(0, 0.03, n_points)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            try:
                res = minimize(evaluate_erdos_solution, h_test, args=(dx,), 
                               method='L-BFGS-B', bounds=[(0,1) for _ in range(n_points)],
                               options={'maxiter': 100, 'ftol': 1e-9})
                if res.success:
                    h_candidate = res.x
                    c5_candidate = evaluate_erdos_solution(h_candidate, dx)
                    if c5_candidate < best_refined_c5:
                        best_refined_c5 = c5_candidate
                        h_refined = h_candidate
            except:
                continue
                
        if best_refined_c5 < best_c5:
            best_h = h_refined
            best_c5 = best_refined_c5
    
    return (list(best_h), best_c5, n_points)
