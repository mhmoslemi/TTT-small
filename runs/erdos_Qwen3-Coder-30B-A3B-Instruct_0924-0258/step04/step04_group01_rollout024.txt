from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

            # OR sum(h[i] * h[i+k]) >= 1 - c/dx

            corr_approx = np.correlate(h, h, mode='full')

            max_corr = np.max(corr_approx)
