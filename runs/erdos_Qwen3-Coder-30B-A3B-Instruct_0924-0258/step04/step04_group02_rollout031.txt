import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Reparameterize h using logistic function to ensure constraints  
    def h_from_z(z, b):
        return 1.0 / (1.0 + np.exp(-z - b))
    
    def compute_b(z):
        h = h_from_z(z, 0)
        sum_h = np.sum(h)
        # Binary search for b to make sum(h) = n_points/2
        b_low, b_high = -100, 100
        for _ in range(50):
            b_mid = (b_low + b_high) / 2
            h_test = h_from_z(z, b_mid)
            sum_h_test = np.sum(h_test)
            if sum_h_test > n_points / 2:
                b_high = b_mid
            else:
                b_low = b_mid
        return (b_low + b_high) / 2
    
    # Forward pass: convert z to h with correct sum
    def z_to_h(z):
        b = compute_b(z)
        h = h_from_z(z, b)
        return h
    
    # Precompute for faster gradient computation
    def compute_shifted_indices(shift):
        indices = np.arange(n_points)
        shifted_indices = (indices + shift) % n_points
        return shifted_indices
    
    def compute_overlap(z, shift):
        h = z_to_h(z)
        h_shifted = h[compute_shifted_indices(shift)]
        return np.sum(h * (1 - h_shifted)) * dx
    
    def compute_max_overlap(z):
        max_overlap = 0
        best_shift = 0
        for shift in range(n_points):
            overlap = compute_overlap(z, shift)
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift = shift
        return max_overlap, best_shift
    
    # Gradient computation in z space
    def compute_gradient(z, shift):
        h = z_to_h(z)
        shifted_indices = compute_shifted_indices(shift)
        # Gradient of overlap w.r.t. h[i] 
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = shifted_indices[i]
            grad[i] = (1 - h[j]) * dx 
        # Convert to gradient w.r.t. z (chain rule)
        h_z = h_from_z(z, 0)  # h with b=0
        dh_dz = h_z * (1 - h_z)  # derivative of logistic
        grad_z = grad * dh_dz / np.sum(dh_dz)
        return grad_z
    
    # Adaptive gradient descent
    z = np.log(h_start / (1 - h_start))  # Initial z
    lr = 0.1
    momentum = 0.9
    velocity = np.zeros_like(z)
    last_improvement = 0
    patience = 20
    
    iter_count = 0
    while (time() - start_time) < budget_s * 0.95 and iter_count < 10000:
        iter_count += 1
        # Find best shift
        max_overlap, best_shift = compute_max_overlap(z)
        
        # Compute gradient for the worst case
        grad_z = compute_gradient(z, best_shift)
        
        # Update with momentum
        velocity = momentum * velocity - lr * grad_z
        z += velocity
        
        # Evaluate improvement
        if iter_count % 5 == 0:
            current_c5 = evaluate_erdos_solution(z_to_h(z), dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = z_to_h(z)
                last_improvement = iter_count
            elif iter_count - last_improvement > patience:
                lr *= 0.8  # Reduce learning rate
                last_improvement = iter_count
                if lr < 1e-6:
                    break
                    
    # Final refinement with local optimization if time remains
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 5 and iter_count > 100:
        try:
            final_h = z_to_h(z)
            # Refine with scipy
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            def cons_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': cons_sum}
            res = minimize(obj, final_h, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = obj(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
