import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    def scp_feasibility_check(h0, C):
        h = cp.Variable(n_points)
        constraints = []
        constraints.append(cp.sum(h) == n_points/2)
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        
        # Linearize the overlap constraint for each shift
        for k in range(n_points):
            if k == 0:
                continue
            # Approximate sum h[i]*h[i+k] using linearization around h0
            expr = 0
            for i in range(n_points):
                if i + k < n_points:
                    expr += h0[i]*h[i+k] + h0[i+k]*h[i] - h0[i]*h0[i+k]
                else:
                    break
            constraints.append(expr >= 1 - C/dx)
        objective = Minimize(0)
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value
        except:
            pass
        return None

    def binary_search_scp(low, high, eps=1e-6):
        best_h = h_start
        best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
        
        while time() - start_time < budget_s * 0.95 and high - low > eps:
            mid = (low + high) / 2
            approx_h = scp_feasibility_check(best_h, mid)
            if approx_h is not None:
                # Refine our solution using local optimization
                refined_h = approx_h.copy()
                scaled_h = refined_h * (n_points/2) / np.sum(refined_h)
                scaled_h = np.clip(scaled_h, 0, 1)
                
                def obj(h):
                    return evaluate_erdos_solution(h, 2.0/n_points)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, scaled_h, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-6})
                    if res.success:
                        final_h = res.x
                        final_c5 = obj(final_h)
                        if final_c5 < best_c5:
                            best_h = final_h.copy()
                            best_c5 = final_c5
                except:
                    pass
                
                # Check feasibility with fixed C
                h_test = scaled_h.copy()
                c5_test = evaluate_erdos_solution(h_test, 2.0/n_points)
                if c5_test <= mid:
                    high = mid
                    best_c5 = min(best_c5, c5_test)
                else:
                    low = mid
            else:
                low = mid
        
        return best_h, best_c5
    
    # Binary search the minimum C₅ value
    try:
        best_h, best_c5 = binary_search_scp(0.3, 0.38085, 1e-5)
    except:
        best_h = h_start
        best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Final refinement
    try:
        def obj(h):
            return evaluate_erdos_solution(h, 2.0/n_points)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final projection to constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
