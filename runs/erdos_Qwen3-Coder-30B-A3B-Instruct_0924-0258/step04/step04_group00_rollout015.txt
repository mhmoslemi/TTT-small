import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary relaxation with local search refinement
    def binary_refine(h):
        # Convert to binary (0 or 1) with proper scaling
        h_scaled = h * (n_points / 2.0) / np.sum(h)
        h_bin = (h_scaled > 0.5).astype(float)
        # Scale back to correct sum
        h_bin = h_bin * (n_points / 2.0) / np.sum(h_bin)
        return h_bin
    
    def local_search(h, max_iter=100):
        h_current = h.copy()
        c_current = evaluate_erdos_solution(h_current, dx)
        for _ in range(max_iter):
            # Slightly perturb
            h_perturbed = h_current + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Renormalize
            h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
            c_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c_perturbed < c_current:
                h_current = h_perturbed
                c_current = c_perturbed
        return h_current, c_current
    
    # Generate constraints dynamically
    def gen_constraints(h_vec, max_shift):
        constraints = []
        for k in range(1, max_shift):
            corr = np.sum(h_vec[:-k] * (1 - h_vec[k:])) if k < len(h_vec) else 0
            constraints.append(corr)
        return constraints
    
    # Efficient Loop
    iter_count = 0
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.9:
        iter_count += 1
        
        # Binary drift search + local refinement
        h_binary = binary_refine(best_h.copy())
        h_refined, c5_refined = local_search(h_binary, max_iter=20)
        
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
            restart_count = 0
        else:
            restart_count += 1
        
        # Periodic global refinement
        if iter_count % 10 == 0:
            # Try global perturbation
            h_perturb = best_h + np.random.normal(0, 0.01, n_points)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2.0) / np.sum(h_perturb)
            c5_perturb = evaluate_erdos_solution(h_perturb, dx)
            if c5_perturb < best_c5:
                best_h = h_perturb
                best_c5 = c5_perturb
                restart_count = 0
        
        # Early termination check
        if restart_count > max_restarts:
            break
    
    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2.0) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
