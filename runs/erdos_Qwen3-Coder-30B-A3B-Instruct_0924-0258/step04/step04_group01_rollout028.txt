import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a structured initial guess with 3 clusters
        cluster_positions = [0.4, 1.0, 1.6]
        cluster_width = 0.3
        for pos in cluster_positions:
            start_idx = max(0, int((pos - cluster_width/2)/dx))
            end_idx = min(n_points, int((pos + cluster_width/2)/dx))
            h_start[start_idx:end_idx] = 1.0
        # Normalize to satisfy constraint
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Initialize population with diverse samples
    population_size = 20
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            # Create diversified initial guesses
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Evolutionary approach with SQP refinement
    max_iter = 300
    mutation_scale = 0.02
    crossover_rate = 0.5
    n_keep = population_size // 2
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all population members
        fitness = [evaluate_erdos_solution(h, 2.0/n_points) for h in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update the best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection: keep top half
        selected = population[:n_keep]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        # Crossover: combine pairs
        for i in range(n_keep, population_size):
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                parent1 = selected[np.random.randint(0, len(selected))]
                parent2 = selected[np.random.randint(0, len(selected))]
                # Uniform crossover
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Clone or mutate
                if len(selected) > 0:
                    child = selected[np.random.randint(0, len(selected))].copy()
                else:
                    child = h_start.copy()
            
            # Mutation
            if np.random.rand() < 0.7:
                noise = np.random.normal(0, mutation_scale, n_points)
                child = child + noise
                child = np.clip(child, 0, 1)
            
            # Ensure sum constraint
            child = child * (n_points/2) / np.sum(child)
            new_population.append(child)
        
        population = new_population
        
        # Refine top candidates with SQP
        for i in range(min(5, len(population))):
            h_current = population[i]
            
            def obj(h):
                return evaluate_erdos_solution(h, 2.0/n_points)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    candidate = res.x
                    if evaluate_erdos_solution(candidate, 2.0/n_points) < fitness[i]:
                        population[i] = candidate
            except:
                pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, 2.0/n_points)
    
    return (list(best_h), final_c5, n_points)
