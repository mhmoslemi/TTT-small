import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Resample h_in to n_out points using linear interpolation."""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, target_sum):
    """Project h to [0,1] and normalize to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

        n_points = len(h_init)
    else:
        n_points = 35
        h_init = np.ones(n_points) * 0.5
    dx = 2.0 / n_points

    # Initialize coarse and fine resolution grids
    n_coarse = 50
    n_fine = 100
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    h_coarse = linear_resample(h_init, n_coarse)
    h_fine = linear_resample(h_init, n_fine)
    
    # Project to feasible region
    h_coarse = project_to_constraints(h_coarse, n_coarse / 2)
    h_fine = project_to_constraints(h_fine, n_fine / 2)
    
    # Track best solutions
    best_h_fine = h_fine.copy()
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    best_h_coarse = h_coarse.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    
    # Set bounds and constraints for optimization
    bounds = [(0, 1) for _ in range(n_fine)]
    
    def constraint_sum_fine(h):
        return np.sum(h) - n_fine / 2.0
    
    cons_sum_fine = {'type': 'eq', 'fun': constraint_sum_fine}
    
    def constraint_sum_coarse(h):
        return np.sum(h) - n_coarse / 2.0
    
    cons_sum_coarse = {'type': 'eq', 'fun': constraint_sum_coarse}
    
    # Evaluate objective function
    def objective_fine(h):
        return evaluate_erdos_solution(h, dx_fine)
    
    # Coarse grid search with perturbations
    pert_scale_coarse = 0.05
    for i in range(100):
        if time() - start_time > budget_s * 0.5:
            break
            
        # Perturb the coarse solution
        h_pert = h_coarse + np.random.normal(0, pert_scale_coarse, n_coarse)
        h_pert = project_to_constraints(h_pert, n_coarse / 2)
        
        # Evaluate
        c5_val = evaluate_erdos_solution(h_pert, dx_coarse)
        if c5_val < best_c5_coarse:
            best_c5_coarse = c5_val
            best_h_coarse = h_pert.copy()
            
        pert_scale_coarse *= 0.99
        
    # Refine coarse solution to fine grid and optimize
    h_fine = linear_resample(best_h_coarse, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine / 2)
    
    # Local optimization using SLSQP
    for restart in range(10):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Start from coarse solution interpolated to fine grid
        if restart == 0:
            h_start = h_fine.copy()
        else:
            h_start = h_fine + np.random.normal(0, 0.02, n_fine)
            h_start = project_to_constraints(h_start, n_fine / 2)
            
        try:
            res = minimize(objective_fine, h_start, method='SLSQP',
                           bounds=bounds, constraints=cons_sum_fine,
                           options={'maxiter': 100, 'ftol': 1e-8})

                h_refined = res.x
                c5_refined = objective_fine(h_refined)
                if c5_refined < best_c5_fine:
                    best_h_fine = h_refined
                    best_c5_fine = c5_refined
        except:
            continue
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            # Use a more aggressive refinement
            res = minimize(objective_fine, best_h_fine, method='SLSQP',
                           bounds=bounds, constraints=cons_sum_fine,
                           options={'maxiter': 200, 'ftol': 1e-10})

                h_final = res.x
                c5_final = objective_fine(h_final)
                if c5_final < best_c5_fine:
                    best_h_fine = h_final
                    best_c5_fine = c5_final
        except:
            pass
    
    # Return the finest resolution result
    return (list(best_h_fine), best_c5_fine, n_fine)
