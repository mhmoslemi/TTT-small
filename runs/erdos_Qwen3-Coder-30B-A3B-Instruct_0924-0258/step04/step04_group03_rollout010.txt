from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with evolutionary steps and gradient refinement
    population_size = 50
    mutation_step = 0.05
    selection_rate = 0.2
    max_generations = 1000
    min_mutation_step = 0.001
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Start with initial solution and perturbations
        population.append(h_start.copy())
        for _ in range(population_size - 1):
            h_pert = h_start + np.random.normal(0, 0.05, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert) 
            population.append(h_pert)
    else:
        # Uniform start
        for _ in range(population_size):
            h_init = np.ones(n_points) * 0.5
            h_init += np.random.normal(0, 0.05, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init)
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate entire population
        fitness = []
        for h in population:
            c_val = evaluate_erdos_solution(h, dx)
            fitness.append(c_val)
        
        # Sort by fitness
        idx_sort = np.argsort(fitness)
        population_sorted = [population[i] for i in idx_sort]
        fitness_sorted = [fitness[i] for i in idx_sort]
        
        # Select top performers
        num_select = max(1, int(population_size * selection_rate))
        selected = population_sorted[:num_select]
        selected_fitness = fitness_sorted[:num_select]
        
        # Keep best
        if fitness_sorted[0] < best_c5:
            best_h = population_sorted[0].copy()
            best_c5 = fitness_sorted[0]
        
        # Create new population
        new_population = selected.copy()
        
        # Mutation and crossover
        while len(new_population) < population_size:
            # Mutation
            h_mut = selected[np.random.randint(0, len(selected))].copy()
            noise = np.random.normal(0, mutation_step, n_points)
            h_mut += noise
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            
            # Occasionally crossover
            if np.random.rand() < 0.3:
                h2 = selected[np.random.randint(0, len(selected))].copy()
                alpha = np.random.rand()

                h_mut = np.clip(h_mut, 0, 1)
                h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            
            new_population.append(h_mut)

        if mutation_step > min_mutation_step:
            mutation_step *= 0.995

    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        top_candidates = []
        top_fitness = []
        
        for h in population:
            c_val = evaluate_erdos_solution(h, dx)

                    # Keep top 5 only
                    idx_sort = np.argsort(top_fitness)
                    top_candidates = [top_candidates[i] for i in idx_sort[:5]]
                    top_fitness = [top_fitness[i] for i in idx_sort[:5]]
        
        # Refine each with SLSQP
        for h_ref in top_candidates:
            h_start = h_ref.copy()
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_start, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
