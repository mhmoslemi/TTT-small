import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from cvxpy import MOSEK

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place three clusters
        cluster_centers = [0.3, 1.0, 1.7]
        cluster_sizes = [3, 4, 3]
        for c, size in zip(cluster_centers, cluster_sizes):
            start = max(0, int((c - size/2)/dx))
            end = min(n_points, int((c + size/2)/dx))
            h_start[start:end] = 1.0 / (2*size)
        # Normalize sum to n_points / 2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search approach with SCP feasibility checking
    low = 0.3
    high = 0.38085
    precision = 1e-6
    max_iterations = 50
    
    last_feasible_h = h_start.copy()
    last_feasible_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Feasibility check via SCP
        h_current = last_feasible_h.copy()
        success = False
        max_scp_iter = 5
        
        for scp_iter in range(max_scp_iter):
            # Formulate convex constraints
            h_vars = cp.Variable(n_points)
            
            # Linearized constraints (for each k)
            constraints = []
            
            # Add normalization constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Add box constraints
            constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
            constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
            
            # Linearize the overlap constraints
            for k in range(n_points):
                # The overlap term for shift k:
                # sum_i h[i]*h[i+k] >= 1 - mid/dx
                # Unknown let's just treat it as a quadratic constraint but solve in rounds using linearization
                # Approximate sum(h[i]*h[i+k]) = h_current[i]*h_current[i+k] + h[i]*h_current[i+k] + h_current[i]*h[i] - h_current[i]*h_current[i+k]
                # But compiler treats it as non-quadratic, so we write in a way it understands
                
                # Linearizing around current h_current
                overlap_sum = 0.0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap_sum += h_vars[i] * h_current[j] + h_current[i] * h_vars[j] - h_current[i] * h_current[j]
                # Actual overlap should be >= 1 - mid/dx
                # This is technically a convex constraint of the form sum terms >= const
                # Thus make sure that
                constraints.append(overlap_sum >= 1 - mid / dx)
                
            # Objective (minimize a dummy 0 to solve the feasibility)
            objective = cp.Minimize(0)
            
            # Solve the convex program
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=MOSEK, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.UNBOUNDED]:
                    h_current = h_vars.value
                    h_current = np.clip(h_current, 0, 1)
                    # Normalize
                    h_current = h_current * (n_points/2) / np.sum(h_current)
                    success = True
                    break
                else:
                    success = False
                    break
            except Exception as e:
                success = False
                break
        
        if success:
            # Evaluate feasibility
            c5_val = evaluate_erdos_solution(h_current, dx)
            if c5_val < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_val
            last_feasible_h = h_current.copy()
            last_feasible_c5 = c5_val
            high = mid  # Try smaller C
        else:
            low = mid  # Try larger C
    
    # Final refinement with local optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Final optimization
    try:
        res = minimize(obj, best_h, method='SLSQP',
                      bounds=bounds, constraints=cons_sum,
                      options={'maxiter': 1000, 'ftol': 1e-8})
        if res.success:
            refined_h = res.x
            refined_c5 = obj(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
