import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple initialization strategies
    initial_strategies = []
    initial_strategies.append(h_start.copy())
    
    # Random initialization with normalization
    h_random = np.random.uniform(0, 1, n_points)
    h_random = h_random * (n_points / 2) / np.sum(h_random)
    initial_strategies.append(h_random)
    
    # Slightly perturbed version
    h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
    initial_strategies.append(h_perturbed)
    
    # Optimizer with adaptive strategy
    max_iter = 500
    iter_count = 0
    
    for h_init in initial_strategies:
        if time() - start_time > budget_s * 0.95:
            break
            
        h_current = h_init.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Simple gradient descent approach with projection
        step_size = 0.01
        max_inner_iter = 200
        
        for _ in range(max_inner_iter):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute gradient numerically (simplified)
            epsilon = 1e-4
            grad = np.zeros(n_points)
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += epsilon
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                h_minus = h_current.copy()
                h_minus[i] -= epsilon
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                grad[i] = (evaluate_erdos_solution(h_plus, dx) - 
                          evaluate_erdos_solution(h_minus, dx)) / (2 * epsilon)
            
            # Update step
            h_new = h_current - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            # Accept if better
            if c5_new < c5_current:
                h_current = h_new
                c5_current = c5_new
                step_size *= 1.1  # Increase step size
            else:
                step_size *= 0.9  # Decrease step size
            
            # Update best
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h_current.copy()
            
            iter_count += 1
            
            if iter_count >= max_iter:
                break
    
    # Final local refinement with SLSQP
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Try to refine further with different starting points
        for _ in range(5):
            if time() - start_time > budget_s * 0.95:
                break
            h_refine = best_h + np.random.normal(0, 0.01, n_points)
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points/2) / np.sum(h_refine)
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(objective, h_refine, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-8})
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final
                        best_c5 = c5_final
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
