import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy with mutation and crossover
    population_size = 50
    mutation_step = 0.05
    selection_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    population = []
    population.append(h_start)
    
    # Generate diverse initial population
    for i in range(population_size - 1):
        h_mut = h_start + np.random.normal(0, mutation_step, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        population.append(h_mut)
    
    # Main evolutionary loop
    for generation in range(max_generations):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c = evaluate_erdos_solution(h, dx)
            fitness.append(c)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        best_fitness = fitness[sorted_indices[0]]
        if best_fitness < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = best_fitness
            
        # Select top performers
        num_selected = max(1, int(population_size * selection_rate))
        selected = [population[i] for i in sorted_indices[:num_selected]]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            p1, p2 = np.random.choice(selected, 2, replace=False)
            
            # Crossover: arithmetic mean
            h_child = (p1 + p2) / 2.0
            
            # Mutation
            noise = np.random.normal(0, mutation_step, n_points)
            h_child += noise
            h_child = np.clip(h_child, 0, 1)
            h_child = h_child * (n_points / 2) / np.sum(h_child)
            
            new_population.append(h_child)
        
        population = new_population
        
        # Adaptively reduce mutation step
        if generation % 100 == 0 and mutation_step > 0.001:
            mutation_step *= 0.9
    
    # Final gradient refinement of top solutions
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Test top 10 solutions
        fitness = []
        for h in population:
            c = evaluate_erdos_solution(h, dx)
            fitness.append(c)
            
        sorted_indices = np.argsort(fitness)
        for i in range(min(10, len(sorted_indices))):
            h_candidate = population[sorted_indices[i]].copy()
            
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Run refinement
            try:
                res = minimize(objective, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
                
    # Final check
    c5_final = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), c5_final, n_points)
