import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Population-based evolutionary algorithm with gradient refinement
    n_points = 35
    population_size = 50
    mutation_rate = 0.05
    selection_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_seed = np.array(initial_h_values)
        population = [h_seed.copy()]
        # Add random perturbations around seed
        for _ in range(population_size - 1):
            h_pert = h_seed + np.random.normal(0, mutation_rate/2, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            population.append(h_pert)
    else:
        # Default initialization
        population = [np.ones(n_points) * 0.5 for _ in range(population_size)]
        for i in range(population_size):
            population[i] = population[i] * (n_points/2) / np.sum(population[i])
    
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Track best solution
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Evolutionary loop
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all populations
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:int(population_size * selection_rate)]
        
        # Keep top performers
        elite = [population[i].copy() for i in top_indices]
        best_c5_now = min(fitness)
        if best_c5_now < best_c5:
            best_c5 = best_c5_now
            best_h = population[sorted_indices[0]].copy()
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover (averaging)
            child = (parent1 + parent2) / 2.0
            
            # Mutation
            mutation_strength = max(0.001, mutation_rate * (1 - generation/max_generations))
            noise = np.random.normal(0, mutation_strength, n_points)
            child = child + noise
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Gradient-based refinement of top performers
    top_performers = []
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
        top_performers.append((c5, h.copy()))
    
    top_performers.sort(key=lambda x: x[0])
    
    # Refine top 5
    for i in range(min(5, len(top_performers))):
        c5, h_refine = top_performers[i]
        
        # Use SLSQP to refine
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except Exception:
            pass
    
    # Final validation and return
    if best_h is None:
        # Fallback to last best found
        best_h = population[-1]
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
