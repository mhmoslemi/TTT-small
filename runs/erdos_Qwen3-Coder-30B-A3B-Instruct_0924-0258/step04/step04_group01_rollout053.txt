import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available

        for i in range(n_points):
            x = i * dx
            if 0.4 <= x <= 0.8 or 1.2 <= x <= 1.6 or 1.8 <= x <= 2.2:
                h_start[i] = 0.5
            else:
                h_start[i] = 0.0
        # Normalize to ensure sum is n/2
        h_start = h_start * (n_points/2) / np.sum(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population parameters
    population_size = 25
    generations = 200
    mutation_rate = 0.3
    crossover_rate = 0.5
    elite_size = 3
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.5:
            # Random initialization
            h = np.random.rand(n_points)
            h = h * (n_points/2) / np.sum(h)
        else:
            # Copy and perturb from best
            h = best_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Exploit or explore switching
    exploitation = False

    for gen in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]

        # Update global best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new population
        new_population = []
        
        # Elite
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select two parents
            parent1_idx = np.random.randint(0, population_size//2)
            parent2_idx = np.random.randint(0, population_size//2)
            
            if np.random.rand() < crossover_rate:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, population[parent1_idx], population[parent2_idx])
            else:
                # Pick one parent
                child = population[np.random.randint(0, population_size//2)].copy()

            if np.random.rand() < mutation_rate:
                # GA-style mutation - perturb with normal noise
                noise = np.random.normal(0, 0.03, n_points)
                child += noise

                # Renormalize
                child = child * (n_points/2) / np.sum(child)
            
            # Further refinement with SQP for high-quality candidates
            if np.random.rand() < 0.2:  # 20% chance to refine
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                try:
                    res = minimize(obj, child, method='SLSQP',

                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        child = res.x
                except:
                    pass
            
            new_population.append(child)
        
        population = new_population[:population_size]

    def obj(h):
        return evaluate_erdos_solution(h, dx)

        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}

                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass

    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
