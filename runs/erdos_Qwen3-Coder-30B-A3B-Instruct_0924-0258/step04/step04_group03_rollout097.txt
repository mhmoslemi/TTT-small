import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_init = 0.7
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.5
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    # Generate swarm around the initial solution
    for i in range(swarm_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_values.append(evaluate_erdos_solution(h, dx))
        
        if personal_best_values[-1] < best_c5:
            best_h = h.copy()
            best_c5 = personal_best_values[-1]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        w = w_init * (1.0 - iteration / max_iter)  # Linearly decreasing inertia
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (personal_best[0] - swarm[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Clip and renormalize
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if fitness < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = fitness
                
                if fitness < best_c5:
                    best_h = swarm[i].copy()
                    best_c5 = fitness
        
        # Force one Particle to be the best for better diversity
        swarm[0] = personal_best[0].copy()
    
    # Convex relaxation refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add linear relaxation of the overlap
            corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
            max_overlap = np.max(corr)
            
            # Add multiple constraints for different overlaps
            for k in range(0, min(20, 2 * n_points - 1), 2):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= max_overlap * 1.1)  # Soft constraint
            
            prob = cp.Problem(cp.Minimize(cp.max(cp.multiply(h_opt, 1 - h_opt))), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h_opt.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                continue
    
    return (list(best_h), best_c5, n_points)
