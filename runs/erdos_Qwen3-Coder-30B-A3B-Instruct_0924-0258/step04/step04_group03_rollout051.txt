import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing with pairwise perturbations and adaptive step size
    def pairwise_neighbor(h, step=0.05):
        h_new = h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-step, step)
        h_new[idx1] += delta
        h_new[idx2] -= delta
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        # Maintain sum constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        return h_new
    
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.999
    step_base = 0.05
    
    # Track best solution found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Adaptive step size based on temperature
        step = step_base * temperature
        
        # Generate neighbor with pairwise perturbation
        new_h = pairwise_neighbor(current_h, step)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            # Check if it's better than best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
        
        # Occasionally refine with convex optimization every 200 steps
        if step_count % 200 == 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                # Add stronger constraints using current best value
                max_overlap_constraint = 0
                corr_full = np.correlate(best_h, 1 - best_h, mode='full')
                max_overlap_constraint = np.max(corr_full) * dx
                
                # Add safety margin to encourage improvement
                safe_c5 = max_overlap_constraint * 0.99
                
                for k in range(2 * n_points - 1):
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    # Add constraint to prevent worse than current
                    constraints.append(corr_sum <= safe_c5 * dx)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-5)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
