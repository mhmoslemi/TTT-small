import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Parameterize h as a smooth function with learnable parameters
    # h(x) = sigmoid(f(x; theta) + beta)
    # But we'll use a simpler approach with linear activation + scaling
    
    # Initialize parameters
    theta = np.random.randn(n_points) * 0.1
    beta = 0.0
    lam = 1.0
    
    # Gradient descent with adaptive learning rate
    lr = 1e-3
    momentum = 0.9
    velocity_theta = np.zeros_like(theta)
    velocity_beta = 0.0
    velocity_lam = 0.0
    
    # Simple backtracking line search
    def line_search(h_func, grad_h, lr):
        alpha = 1.0
        max_iter = 100
        for _ in range(max_iter):
            new_h = h_func(alpha * grad_h)
            if np.sum(new_h) > 1e-8:
                return alpha
            alpha *= 0.5
        return alpha
    
    # Optimization loop
    iter_count = 0
    max_iter = 50000
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Construct h
        h_unscaled = np.maximum(0, theta + beta)
        h_sum = np.sum(h_unscaled)
        if h_sum > 0:
            h = np.minimum(1, h_unscaled * (n_points / 2) / h_sum)
        else:
            h = np.ones_like(h_unscaled) * 0.5
        
        # Evaluate current objective
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad_theta = np.zeros_like(theta)
        grad_beta = 0.0
        grad_lam = 0.0
        
        # Compute partial derivatives
        h_base = h.copy()
        for i in range(n_points):
            theta_eps = theta.copy()
            theta_eps[i] += eps
            h_plus = np.maximum(0, theta_eps + beta)
            h_sum_plus = np.sum(h_plus)
            if h_sum_plus > 0:
                h_plus = np.minimum(1, h_plus * (n_points / 2) / h_sum_plus)
            else:
                h_plus = np.ones_like(h_plus) * 0.5
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            
            grad_theta[i] = (c5_plus - c5) / eps
        
        # Beta gradient
        beta_eps = beta + eps
        h_plus = np.maximum(0, theta + beta_eps)
        h_sum_plus = np.sum(h_plus)
        if h_sum_plus > 0:
            h_plus = np.minimum(1, h_plus * (n_points / 2) / h_sum_plus)
        else:
            h_plus = np.ones_like(h_plus) * 0.5
        c5_plus = evaluate_erdos_solution(h_plus, dx)
        grad_beta = (c5_plus - c5) / eps
        
        # Update with momentum
        velocity_theta = momentum * velocity_theta + lr * grad_theta
        velocity_beta = momentum * velocity_beta + lr * grad_beta
        
        theta -= velocity_theta
        beta -= velocity_beta
        
        # Add noise occasionally to escape local minima
        if iter_count % 100 == 0:
            theta += np.random.normal(0, 0.01, n_points)
        
        iter_count += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:

            c5_new = evaluate_erdos_solution(new_h, dx)
            if c5_new < best_c5:
                best_h = new_h.copy()
                best_c5 = c5_new
            last_h = new_h
    
    return (list(best_h), best_c5, n_points)
