import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent's initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        h_start[5:15] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse grid optimization
    max_iter = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    n_points_coarse = 20
    
    # Initialize population
    pop_size = 10
    population = [h_start.copy()]
    for _ in range(pop_size-1):
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate and update best
        for i, h in enumerate(population):
            curr_c5 = evaluate_erdos_solution(h, dx)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
                
        # Refinement with SLSQP
        refined_population = []
        for h in population:
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
            
            def con_sum(h_vec):
                return np.sum(h_vec) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                else:
                    refined_h = h
            except:
                refined_h = h
            refined_population.append(refined_h)
        
        # Generate new population
        new_population = []
        for h in refined_population:
            h_pert = h + np.random.normal(0, perturb_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            # Simulated annealing acceptance
            old_c5 = evaluate_erdos_solution(h, dx)
            new_c5 = evaluate_erdos_solution(h_pert, dx)
            if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
                new_population.append(h_pert)
            else:
                new_population.append(h)
        
        population = new_population
        temp *= cool_rate
        
        # Progress print
        if iteration % 20 == 0:
            print("Iter:", iteration, "C5:", best_c5)
    
    # Upsample to 40 points
    n_upsample = 40
    h_upsample = np.interp(np.linspace(0, 2, n_upsample), 
                          np.linspace(0, 2, n_points), 
                          best_h)
    h_upsample = np.clip(h_upsample, 0, 1)
    h_upsample = h_upsample * (n_upsample/2) / np.sum(h_upsample)
    n_points = n_upsample
    dx = 2.0 / n_points
    
    # Refine the upsampled solution
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, h_upsample, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            refined_h = res.x
            refined_c5 = evaluate_erdos_solution(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    # Final refinement with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Gradient descent-like refinement
        h_refined = best_h.copy()
        step_size = 0.001
        for i in range(500):
            if time() - start_time > budget_s * 0.95:
                break
            # Simple gradient descent step
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            grad = np.zeros_like(h_refined)
            for j in range(len(h_refined)):
                temp_h = h_refined.copy()
                temp_h[j] += 0.001
                temp_corr = np.correlate(temp_h, 1 - temp_h, mode='full')
                grad[j] = (np.max(temp_corr) - np.max(corr)) / 0.001
            
            h_refined = h_refined - step_size * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points/2) / np.sum(h_refined)
            
            new_c5 = evaluate_erdos_solution(h_refined, dx)
            if new_c5 < best_c5:
                best_h = h_refined.copy()
                best_c5 = new_c5
        
    return (list(best_h), best_c5, n_points)
