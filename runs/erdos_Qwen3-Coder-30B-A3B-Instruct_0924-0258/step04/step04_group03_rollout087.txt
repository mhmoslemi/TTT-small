from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based approach with evolutionary strategies
    population_size = 50
    num_generations = 1000
    elite_ratio = 0.2
    mutation_rate = 0.05
    
    # Create initial population
    population = []
    if 'initial_h_values' in globals():
        # Start with parent solution and add variations
        population.append(h_start)
        for _ in range(population_size - 1):
            perturbed = h_start + np.random.normal(0, mutation_rate * 0.5, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points/2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        # Random initialization
        for _ in range(population_size):
            h_init = np.random.rand(n_points)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            population.append(h_init)
    
    # Evolutionary loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Selection: keep elite
        elite_indices = np.argsort(fitness)[:int(elite_ratio * population_size)]
        elite_population = [population[i] for i in elite_indices]
        
        # Create new population via mutation and crossover
        new_population = elite_population.copy()
        
        # Add mutated individuals
        while len(new_population) < population_size:
            # Select parent
            parent = np.random.choice(elite_population)
            
            # Mutation
            mutated = parent + np.random.normal(0, mutation_rate, n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * (n_points/2) / np.sum(mutated)
            
            new_population.append(mutated)
            
        # Ensure we maintain exact population size
        population = new_population[:population_size]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
    
    # Gradient refinement on top candidates
    top_candidates = []
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    # Get top candidates (best ones)
    top_indices = np.argsort(fitness)[:min(5, len(population))]
    top_candidates = [population[i] for i in top_indices]

    for h_candidate in top_candidates:
        if time() - start_time > budget_s * 0.95:
            break
            
        def objective(h):
            return evaluate_erdos_solution(h, dx)

            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
