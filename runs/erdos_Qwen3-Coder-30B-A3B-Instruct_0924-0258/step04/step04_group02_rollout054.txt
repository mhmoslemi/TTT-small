import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define reparameterization: h = sigmoid(z)
    def sigmoid(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    # Optimization using quasi-Newton method on reparameterized variables
    def objective_and_gradient(z):
        # Reparameterize h
        h = sigmoid(z)
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Compute objective
        c5 = evaluate_erdos_solution(h, dx)
        # Compute gradient using finite difference for simplicity
        eps = 1e-6
        grad = np.zeros_like(z)
        for i in range(len(z)):
            z_plus = z.copy()
            z_plus[i] += eps
            h_plus = sigmoid(z_plus)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5) / eps
        return c5, grad
    
    # Use scipy.optimize.minimize with L-BFGS-B
    max_iter = 1000
    iteration = 0
    
    z_start = np.log(h_start / (1 - h_start))  # Logit transformation
    
    # Multiple restarts with different initializations
    restarts = 5
    for restart in range(restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize z with different methods
        if restart == 0:
            z_current = z_start.copy()
        else:
            z_current = z_start + np.random.normal(0, 0.1, n_points)
            
        # Clip to avoid numerical issues
        z_current = np.clip(z_current, -20, 20)
        
        # Optimize using L-BFGS-B
        try:
            res = minimize(
                lambda z: objective_and_gradient(z)[0],
                z_current,
                method='L-BFGS-B',
                jac=lambda z: objective_and_gradient(z)[1],
                options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-6}
            )
            if res.success:
                z_opt = res.x
                h_opt = sigmoid(z_opt)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
    
    # Additional refinement using random local search
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5_pert = evaluate_erdos_solution(h_pert, dx)
        if c5_pert < best_c5:
            best_h = h_pert
            best_c5 = c5_pert
    
    return (list(best_h), best_c5, n_points)
