import numpy as np
from scipy.optimize import minimize
from time import time
import math
from cvxpy import *

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 30
        # Create a balanced initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place mass in two regions to break symmetry
        h_start[5:10] = 0.8
        h_start[20:25] = 0.8
        # Normalize to sum to n_points/2
        h_start *= (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Use modified Nelder-Mead for early optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points/2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Coarse optimization first
    for iter_coarse in range(20):
        if time() - start_time > budget_s * 0.5:
            break
            
        # Perturb slightly
        h_pert = h_start + np.random.normal(0, 0.03, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        try:
            res = minimize(objective, h_pert, method='Nelder-Mead', 
                          options={'maxiter': 50, 'xatol': 1e-6, 'fatol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_new = evaluate_erdos_solution(h_refined, dx)
                if c5_new < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_new
        except:
            pass
    
    # Refine with L-BFGS
    remaining_budget = budget_s - (time() - start_time)
    if remaining_budget > 50:
        try:
            res = minimize(objective, best_h, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # Final refinement using SLSQP
    if time() - start_time < budget_s * 0.95:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
