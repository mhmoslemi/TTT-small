import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Config
    population_size = 40
    max_generations = 1000
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    stagnation_limit = 50
    
    # Initialize population with diverse solutions
    population = []
    tabu_list = set()
    
    # Create initial population
    for _ in range(population_size):
        if random.random() < 0.5:
            # Start with perturbed initial solution
            h_pert = h_start + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
        else:
            # Random valid solution
            h_pert = np.random.random(n_points)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Keep track of best solution and stagnation counter
    current_best = best_h.copy()
    current_best_c5 = best_c5
    stagnation_count = 0
    last_improvement = 0
    
    # Main GA loop
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitnesses.append(c5)
        
        # Track best individual in population
        pop_best_idx = np.argmin(fitnesses)
        if fitnesses[pop_best_idx] < current_best_c5:
            current_best = population[pop_best_idx].copy()
            current_best_c5 = fitnesses[pop_best_idx]
            stagnation_count = 0
            last_improvement = gen
        else:
            stagnation_count += 1
        
        # Update global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()
            
        # Early stopping
        if best_c5 < 0.3807:
            break
            
        # Check for stagnation and adapt parameters
        if stagnation_count > stagnation_limit:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            if gen - last_improvement > 20:
                mutation_rate = max(0.05, mutation_rate - 0.05)
                crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Create next generation via tournament selection, crossover, and mutation
        next_gen = []
        fitness_ranked = sorted(enumerate(fitnesses), key=lambda x: x[1])
        selected_indices = [i for i, _ in fitness_ranked[:population_size//2]]
        
        while len(next_gen) < population_size:
            # Selection by tournament
            if random.random() < crossover_prob and len(selected_indices) >= 2:
                idx1, idx2 = random.sample(selected_indices, 2)
                parent1, parent2 = population[idx1], population[idx2]
                
                # Crossover with mask
                mask = np.random.random(n_points) > 0.5
                child1 = np.where(mask, parent1, parent2)
                child2 = np.where(mask, parent2, parent1)
                
                # Mutation
                if random.random() < mutation_rate:
                    mutate_idx = random.randint(0, n_points-1)
                    child1[mutate_idx] = random.random()
                if random.random() < mutation_rate:
                    mutate_idx = random.randint(0, n_points-1)
                    child2[mutate_idx] = random.random()
                
                # Ensure constraints
                child1 = child1 * (n_points/2) / np.sum(child1)
                child2 = child2 * (n_points/2) / np.sum(child2)
                
                next_gen.extend([child1, child2])
            else:
                # Clone top individual
                idx = random.choice(selected_indices)
                child = population[idx].copy()
                if random.random() < mutation_rate:
                    mutate_idx = random.randint(0, n_points-1)
                    child[mutate_idx] = random.random()
                child = child * (n_points/2) / np.sum(child)
                next_gen.append(child)
        
        # Trim and update population
        population = next_gen[:population_size]
    
    # Final gradient refinement on best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-9})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
