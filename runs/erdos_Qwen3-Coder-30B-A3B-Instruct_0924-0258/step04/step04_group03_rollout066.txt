import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with projected updates
    def project(h):
        # Ensure bounds and sum constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_gradient(h, k):
        # Compute gradient of overlap w.r.t. h for shift k
        n = len(h)
        grad = np.zeros(n)
        if k < n:
            # h[i] contributes to overlap with (1-h[i+k]) for i in [0,n-k-1]
            for i in range(n - k):
                grad[i] += 1 - h[i + k]
            # h[i+k] contributes to overlap with (1-h[i]) for i in [0,k-1]  
            for i in range(k):
                grad[i + k] += -1 + h[i]
        else:
            # k >= n, shifted version of h
            k_shift = k - n + 1
            for i in range(n - k_shift):
                grad[i + k_shift] += -1 + h[i]
            for i in range(k_shift):
                grad[i] += 1 - h[i + n - k_shift]
        return grad
    
    # Main optimization loop
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    decay = 0.999
    max_iter = 5000
    
    step = 0
    while step < max_iter and (time() - start_time) < budget_s * 0.95:
        # Compute all overlaps to find critical shift
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_best = np.argmax(corr)
        max_overlap = corr[k_best] * dx
        
        # Compute gradient
        grad = compute_gradient(current_h, k_best)
        
        # Update with gradient step
        new_h = current_h - alpha * grad
        new_h = project(new_h)
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
            alpha = min(alpha * decay, 1e-2)
        else:
            alpha *= 0.9
            
        step += 1
        
        # Occasionally refine with convex optimization
        if step % 50 == 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add top overlap constraints
                corr = np.correlate(current_h, 1 - current_h, mode='full')
                indices = np.argsort(corr)[-10:][::-1]  # Top 10 shifts
                
                for k in indices:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
