import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Main annealing loop
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute max delta to maintain constraints
        delta_max = min(1.0 - current_h[i], current_h[j])
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Perturb the points
        h_new = current_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds are maintained
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Adjust to maintain sum constraint
        diff = np.sum(h_new) - (n_points / 2)
        if abs(diff) > 1e-10:
            # Adjust by redistributing difference
            h_new += diff / n_points
            
        # Re-clip to ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject move
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cooling schedule
        T *= alpha
        iteration += 1
    
    return (list(best_h), best_c5, n_points)
