import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization
    max_iter_coarse = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    population = []
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        for i, h in enumerate(population):
            curr_c5 = evaluate_erdos_solution(h, dx)
            if curr_c5 < best_c5:
                best_h = h.copy()
                best_c5 = curr_c5
                
        # Generate new population with perturbations
        new_population = []
        for h in population:
            h_pert = h + np.random.normal(0, perturb_mag, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            
            old_c5 = evaluate_erdos_solution(h, dx)
            new_c5 = evaluate_erdos_solution(h_pert, dx)
            if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
                new_population.append(h_pert)
            else:
                new_population.append(h)
        
        population = new_population
        temp *= cool_rate
    
    # Upsample to 40 points
    n_points = 40
    dx = 2.0 / n_points
    h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    h_upsampled = h_upsampled * (n_points/2) / np.sum(h_upsampled)
    
    # Fine grid optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, h_upsampled, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 100, 'ftol': 1e-6})
        if res.success:
            h_fine = res.x
            fine_c5 = evaluate_erdos_solution(h_fine, dx)
            if fine_c5 < best_c5:
                best_h = h_fine
                best_c5 = fine_c5
    except:
        pass
    
    # Final refinement with gradient-based method
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        h_refined = best_h.copy()
        for _ in range(100):
            if time() - start_time > budget_s * 0.95:
                break
            grad = np.zeros(n_points)
            # Approximate gradient via finite differences
            eps = 1e-4
            base_c5 = evaluate_erdos_solution(h_refined, dx)
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            
            # Gradient descent step
            step_size = 0.1 / (1.0 + np.linalg.norm(grad))
            h_new = h_refined - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
            h_refined = h_new.copy()
    
    return (list(best_h), best_c5, n_points)
