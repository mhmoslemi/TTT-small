import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected gradient descent with adaptive step size
    max_iter = 10000
    alpha = 0.01
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 50
    
    h_current = best_h.copy()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        h_shifted = np.concatenate([h_current, np.zeros(n_points)])
        
        # For each possible shift k, compute contribution to gradient
        for k in range(-n_points+1, n_points):
            if k >= 0:
                # h[i] * (1 - h[i+k]) for i in range(n_points-k)
                grad[:n_points-k] += 1 - h_shifted[k:n_points]
                # h[i+k] * (1 - h[i]) for i in range(n_points-k)
                grad[k:n_points] -= h_current[:n_points-k]
            else:
                # k < 0, shift forward
                # h[i] * (1 - h[i+k]) for i in range(-k, n_points)
                grad[-k:] += 1 - h_shifted[:n_points+k]
                # h[i+k] * (1 - h[i]) for i in range(-k, n_points)
                grad[:n_points+k] -= h_current[-k:]
        
        # Update with gradient and step size
        h_new = h_current - alpha * grad
        
        # Project to [0,1]
        h_proj = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        h_proj = h_proj * (n_points/2) / np.sum(h_proj)
        
        # Compute new value
        c5_new = evaluate_erdos_solution(h_proj, dx)
        
        # Accept if improvement or if we're in the last third of time
        if c5_new < best_c5:
            h_current = h_proj
            best_c5 = c5_new
            last_improvement = iteration
            # Increase step size
            alpha *= 1.2
        else:
            # Decrease step size and potentially revert
            alpha *= 0.5
            # Optionally revert to best solution
            if iteration - last_improvement > 10:
                h_current = best_h.copy()
        
        # Check for stagnation
        if iteration - last_improvement > max_stagnation:
            break
    
    # Final validation and refinement
    final_h = h_current.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    if final_c5 < best_c5:
        best_h = final_h
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
