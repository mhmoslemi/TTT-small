import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def pairwise_perturb(h, step_size):
        h_new = h.copy()
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-step_size, step_size)
        h_new[i1] += delta
        h_new[i2] -= delta
        # Ensure bounds
        h_new[i1] = np.clip(h_new[i1], 0, 1)
        h_new[i2] = np.clip(h_new[i2], 0, 1)
        # Adjust to maintain sum
        h_new *= (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Simulated Annealing
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.999
    step_size = 0.05
    max_steps = 10000
    
    # Track best solution found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Generate neighbor
        new_h = pairwise_perturb(current_h, step_size)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            # Check if it's better than best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down and adjust step size
        temperature *= alpha
        step_size = max(0.01, step_size * alpha)
        step_count += 1
        
        # Periodic convex refinement
        if step_count % 500 == 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add constraint for better upper bound
                corr_max = cp.Variable()
                for i in range(n_points):
                    for j in range(i, n_points):
                        if i == j:
                            continue
                        k = j - i
                        if k < 0:
                            k = n_points + k
                        if k < n_points:
                            corr_term = cp.multiply(h[i], 1 - h[j])
                            constraints.append(corr_term <= corr_max * dx)
                
                prob = cp.Problem(cp.Minimize(corr_max), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
