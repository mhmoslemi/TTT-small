import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()

        n_points = len(h_start)

    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Objective function for optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint to maintain sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Constraints for optimization
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Gradient-based optimization with SLSQP
    max_iterations = 200
    iteration = 0
    
    # Generate a diverse population using different strategies
    population_size = 10
    population = []
    
    # Start with best current solution
    population.append(h_start.copy())
    
    # Add random perturbations
    for i in range(population_size - 1):
        # Create a more diverse initialization
        h_pert = h_start + np.random.normal(0, 0.03, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        # Normalize to sum to n_points/2
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Use the best from population with focussed local search
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Run optimization
    while (time() - start_time) < budget_s * 0.95 and iteration < max_iterations:
        # Test all solutions in population
        obj_values = [evaluate_erdos_solution(h, dx) for h in population]

        min_idx = np.argmin(obj_values)

            best_c5 = obj_values[min_idx]

        for i in range(3):
            h_current = population[top_indices[i]].copy()
            
            # Local optimization around each top solution
            try:
                res = minimize(objective, h_current, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 10, 'ftol': 1e-6})
                if res.success:
                    new_h = res.x
                    new_c5 = objective(new_h)

                    population.append(h_current)

        mutation_strength = 0.05 * (1 - iteration / max_iterations)

            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)

            try:

                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
