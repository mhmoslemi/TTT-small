import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Ensure h_start satisfies constraints
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = 100000
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Perturbation delta
        delta_max = 0.1 * T
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Compute how much we can actually change
        h_copy = current_h.copy()
        
        # Adjust h[i] and h[j] while maintaining sum
        if h_copy[i] + delta > 1:
            delta = 1 - h_copy[i]
        elif h_copy[i] + delta < 0:
            delta = -h_copy[i]
            
        h_copy[i] += delta
        h_copy[j] -= delta
        
        # Clamp to ensure constraints
        h_copy = np.clip(h_copy, 0, 1)
        h_copy = h_copy * (n_points / 2) / np.sum(h_copy)
        
        # Evaluate new solution
        candidate_c5 = evaluate_erdos_solution(h_copy, dx)
        
        # Accept with probability based on simulated annealing
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = h_copy.copy()
            current_c5 = candidate_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Cool temperature
        T *= alpha
        
    return (list(best_h), best_c5, n_points)
