import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # PSO parameters
    swarm_size = 50
    max_iterations = 1000
    omega_start = 0.9
    omega_end = 0.4
    c1 = c2 = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    # Initialize particles
    for i in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            h += np.random.normal(0, 0.01, n_points)
        else:
            h = np.random.uniform(0, 1, n_points)
        
        # Normalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_scores.append(evaluate_erdos_solution(h, dx))
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        omega = omega_start - (omega_start - omega_end) * iteration / max_iterations
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (omega * velocities[i] + 
                           c1 * r1 * (personal_best[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply constraints: clip to [0,1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Normalize to maintain sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            current_score = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_scores[i] = current_score
                
                # Update global best if needed
                if current_score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = current_score
    
    # Convex optimization refinement
    if time() - start_time < budget_s * 0.9:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Tighten constraints
            for k in range(1, n_points):
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                constraints.append(corr_sum <= global_best_score * dx * n_points)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < global_best_score:
                    global_best = h_refined
                    global_best_score = c5_refined
        except Exception:
            pass
    
    # SLSQP polish
    if time() - start_time < budget_s - 20:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, global_best, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                global_best = res.x
                global_best_score = objective(global_best)
        except Exception:
            pass
    
    # Final validation
    final_h = np.clip(global_best, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_score = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_score, n_points)
