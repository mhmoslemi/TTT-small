from scipy.optimize import linprog
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iterations = 60
    current_c5 = high
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # LP-based binary search
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        # Formulate LP feasibility problem
        # min 0 s.t. sum(h[a] * (1 - h[a+k])) <= mid / dx for all k
        #       sum(h) = n_points / 2
        #       0 <= h[i] <= 1
        
        # Constraint matrix for all overlap constraints

        h_vars = cp.Variable(n_points)
        constraints = [cp.constraints.Zero(h_vars >= 0),
                       cp.constraints.Zero(h_vars <= 1),
                       cp.sum(h_vars) == n_points / 2.0]
        
        # Add all overlap constraints using convolution-like masking
        for k in range(2 * n_points - 1):
            # Compute valid a indices for this shift k
            a_indices = []
            b_indices = []

                b = a + k

                    a_indices.append(a)

                expr = sum(h_vars[a] * (1 - h_vars[b]) for a,b in zip(a_indices, b_indices))
                constraints.append(expr <= mid / dx)
        
        # Formulate and solve the problem
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)

                    current_c5 = c5_feasible
                    best_h = h_feasible.copy()
                    if abs(high - low) < precision:
                        # Stop early if we're close to precision
                        break
                high = mid
            else:
                low = mid
        except:
            # If optimization fails, fall back to the latest solution
            low = mid

    return (list(best_h), current_c5, n_points)
