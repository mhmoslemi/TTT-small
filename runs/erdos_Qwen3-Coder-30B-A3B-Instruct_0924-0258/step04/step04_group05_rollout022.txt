import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    for _ in range(10):
        h = initial_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h.copy())
        
    # Add random solutions
    for _ in range(5):
        h = np.random.random(n_points)
        h = h * (n_points/2) / np.sum(h)

    # Initialize best solution
    best_h = initial_h.copy()
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(best_h, dx)

    for gen in range(500):
        if time() - start_time > budget_s * 0.95:

        fitness = []

        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:

            best_c5 = fitness[best_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        if res.success:
            best_h = res.x
            best_c5 = objective(best_h)
    except:
        pass
        
    return (list(best_h), best_c5, n_points)
