import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use gradient-based optimization approach
    # We'll work with z parameters where h = sigmoid(z + b)
    z = np.log(h_start / (1 - h_start)) if np.all((h_start > 0) & (h_start < 1)) else np.zeros(n_points)
    
    def get_h_from_z(z):
        return 1 / (1 + np.exp(-z))
    
    def adjust_b(z):
        h = get_h_from_z(z)
        b = np.log((n_points/2) / np.sum(h))
        return b
    
    def compute_Ck(h, k):
        # Compute correlation at shift k
        corr = np.correlate(h, 1 - h, mode='full')
        return corr[len(h) - 1 + k] * dx
    
    def compute_overlaps(z):
        h = get_h_from_z(z)
        b = adjust_b(z)
        shifted_h = get_h_from_z(z + b)
        corr = np.correlate(shifted_h, 1 - shifted_h, mode='full')
        return corr * dx
    
    def compute_gradient(z):
        h = get_h_from_z(z)
        b = adjust_b(z)
        shifted_h = get_h_from_z(z + b)
        
        # Find k that maximizes overlap
        corr = np.correlate(shifted_h, 1 - shifted_h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient of overlap for k_max
        # For shift k_max, we need to differentiate корреляция(h, 1-h) with respect to h
        n = len(shifted_h)
        grad = np.zeros_like(shifted_h)
        
        # By definition of correlation with shifted array
        # For shift k_max, overlap = sum_{i} h[i] (1 - h[i+k_max])
        # So d/dh[j] = (1 - h[j+k_max]) if j >= 0 and j+k_max < n, 0 elsewhere
        # But this is naive - the gradient wrt h at some index partly depends on h at 
        # indices in both directions due to overlap. Let's compute it via differences.
        # Actually, we compute dC_k/dh exactly:
        # C_k = sum_{i} h[i] * (1 - h[i+k]) 
        # So dC_k / dh[j] = 1 - h[j+k] if j+k < n, and 1 - h[j-k] if j-k >= 0 and j-k < n
        
        # Simplifying: we compute gradient from correlation directly
        # We compute gradient using chain rule and properties of sigmoid:
        # derivative of sigmoid(z) is sigmoid(z) * (1 - sigmoid(z))
        # Then total derivative of overlap w.r.t. z is adjoint * d/dz[overlap]
        # For direct computation we get the gradient at each position from the derivative
        # of the sum w.r.t. h, and transform it to z-space.
        forward = np.zeros(n)
        backward = np.zeros(n)
        for i in range(n):
            # Contribution of h[i]*(1-h[i+k_max]) to gradient at h[i]  
            j = i + k_max
            if j < n:
                forward[i] = 1 - shifted_h[j]
            j = i - k_max
            if j >= 0:
                backward[i] = 1 - shifted_h[j]
        
        # Since derivative of sigmoid(z) = sigmoid(z)*(1-sigmoid(z)):
        # 
        # dC/dz = sum_i { dC/dh_i * dh_i/dz }
        # 
        # Let's define the derivative of sigmoid:
        s = shifted_h
        ds_dz = s * (1 - s)
        
        # As each gradient term dC_k/dh_i:
        # It's a useful fact that the derivative of correlation is based on convolution
        # of the two functions' derivatives. 
        # But we're stuck with using simpler poewrful strategies
        
        # Simplified but effective approach:
        # Compute gradient of overlap w.r.t. actual h[i] values
        # This requires indexing through overlaps  
        # So we compute dC_k/dz by computing backward differences weights
        # We compute for k_max
        
        # Regular gradient for each position
        # dC_k/dh[i] = 1 - h[i + k_max]     if i+k_max < n
        #              1 - h[i - k_max]     if i-k_max >= 0
        # We compute this per i and then multiply by ds/dz[i] = h[i] * (1 - h[i])
        
        grad = np.zeros(n)
        for i in range(n):
            if i + k_max < n:
                grad[i] += (1 - shifted_h[i + k_max]) * ds_dz[i]
            if i - k_max >= 0:
                grad[i] += (1 - shifted_h[i - k_max]) * ds_dz[i]
        
        # Transform to z-space
        # This is the crucial part. The link is:
        #    dC/dz = dC/dh * dh/dz 
        # Here dh/dz = h*(1-h)
        return grad
    
    # Main optimization loop
    lr = 0.01
    iters = 0
    max_iters = 5000
    patience = 50
    
    last_improvement = 0
    
    # Random restarts
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Initialize z with random values
        z = np.random.normal(0, 0.1, n_points)
        best_z = z.copy()
        best_loss = np.inf
        
        for iteration in range(max_iters):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Adjust b
            b = adjust_b(z)
            
            # Get h from z
            h = get_h_from_z(z + b)
            
            # Evaluate C5
            loss = evaluate_erdos_solution(h, dx)
            
            if loss < best_loss:
                best_loss = loss
                best_z = z.copy()
                if loss < best_c5:
                    best_h = h.copy()
                    best_c5 = loss            
            else:
                if iteration - last_improvement > patience:
                    # Early stopping if no improvement in a while
                    break
            
            # Compute gradient (approximate)
            try:
                grad = compute_gradient(z)
                z_new = z - lr * grad
                
                # Line search to ensure descent
                alpha = 1.0
                step_size = 0.1
                
                while True:
                    z_candidate = z - alpha * lr * grad
                    b_candidate = adjust_b(z_candidate)
                    h_candidate = get_h_from_z(z_candidate + b_candidate)
                    candidate_loss = evaluate_erdos_solution(h_candidate, dx)
                    if candidate_loss < loss or alpha < step_size:
                        z = z_candidate
                        break
                    else:
                        alpha *= 0.5
                        
            except Exception:
                # If gradient computation fails, fallback to random step
                z += np.random.normal(0, 0.001, n_points)
            
            # Just for speed check
            if iteration % 500 == 0:
                lr = max(0.001, lr * 0.999)
                
    return (list(best_h), best_c5, n_points)
