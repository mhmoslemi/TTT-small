import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create clustered initial guess
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define objective function for optimization
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    # Define constraint for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try direct optimization using L-BFGS-B (good for smooth problems)
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', 
                       bounds=bounds, options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = obj(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    except:
        pass
    
    # Try more aggressive optimization with bounds and constraints
    try:
        # Use Trust Region Constrained method
        res = minimize(obj, h_start, method='trust-constr',
                       bounds=bounds, constraints=cons_sum,
                       options={'maxiter': 200, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = obj(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined.copy()
                best_c5 = c5_refined
    except:
        pass
    
    # Perform local random search as fallback
    local_search_iters = 100
    for _ in range(local_search_iters):
        if time() - start_time > budget_s * 0.95:
            break
        # Small random perturbations
        h_perturbed = h_start + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        c5_val = evaluate_erdos_solution(h_perturbed, dx)
        if c5_val < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5_val
    
    # Final cleanup and verification
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
