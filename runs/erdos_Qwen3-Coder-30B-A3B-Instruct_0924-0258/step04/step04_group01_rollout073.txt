import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from math import inf

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create an initial guess with 3 clusters
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 3
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5 bound
    def check_feasibility(C, h_init):
        # Convex relaxation using sequential convex programming
        h = h_init.copy()
        # Iteratively solve a sequence of convex programs
        # Linearization: h[i]*h[i+k] ≈ h[i] * h_prev[i+k] + h_prev[i] * h[i+k] - h_prev[i] * h_prev[i+k]
        for _ in range(5):
            # Create convex problem
            h_vars = cp.Variable(n_points)
            # Normalize constraint
            constraints = [cp.sum(h_vars) == n_points / 2]
            # Overlap constraints
            for k in range(n_points):
                # Approximate sum_i h[i]*h[i+k] 
                # Using current point estimate to linearize
                sum_expr = 0
                for i in range(n_points):
                    if i + k < n_points:
                        sum_expr += h[i] * h_vars[i + k] + h_vars[i] * h[i + k] - h[i] * h[i + k]
                # Enforce 1 - C <= sum_expr * dx
                constraints.append(sum_expr * dx >= 1 - C)
                
            # Bound constraints
            constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
            constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
            
            # Objective: not used, just solve feasibility
            prob = cp.Problem(cp.Minimize(0), constraints)
            
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h = h_vars.value
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                else:
                    # If infeasible, return False
                    return False
            except:
                return False
        return True
    
    # Binary search parameters
    low, high = 0.3, 0.38085
    precision = 1e-6
    
    # Reference the best solution from the initial set
    h_ref = best_h.copy()
    while time() - start_time < budget_s * 0.95 and high - low > precision:
        mid = (low + high) / 2
        if check_feasibility(mid, h_ref):
            high = mid
            # Update reference solution
            h_ref = h_ref.copy()
        else:
            low = mid

    # Final polishing with local optimization
    # Evaluate the final feasible solution 
    final_c5 = evaluate_erdos_solution(h_ref, dx)
    last_best = final_c5
    refined_h = h_ref.copy()
    
    # Try to do better with short local optimization many times
    for _ in range(5):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        pert_h = refined_h + np.random.normal(0, 0.005, n_points)
        pert_h = np.clip(pert_h, 0, 1)
        pert_h = pert_h * (n_points/2) / np.sum(pert_h)
        
        # Optimize it with SLSQP
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj, pert_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < last_best:
                    last_best = c5_refined
                    refined_h = h_refined
        except:
            pass
    
    # Ensure final is feasible
    final_h = np.clip(refined_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
