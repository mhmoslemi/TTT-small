import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a more structured initial guess
        h_start[:n_points//3] = 0.8
        h_start[n_points//3:2*n_points//3] = 0.2
        h_start[2*n_points//3:] = 0.8
        # Normalize sum to target value
        h_start = h_start * (n_points/2) / np.sum(h_start)
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization with fewer points
    n_coarse = min(20, n_points)
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), best_h)
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    dx_coarse = 2.0 / n_coarse
    
    # penalty update function
    def update_penalties(penalties, overlaps, factor=0.1):
        new_penalties = np.maximum(penalties * (overlaps / (penalties + 1e-10)), 1e-6)
        return new_penalties + factor * (overlaps - penalties) 
    
    # Optimization with penalty method  
    max_iter_coarse = 100
    penalties = np.ones(n_coarse * 2 - 1)  # overlaps for all shifts
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr_coarse = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        
        # Set up convex problem
        h_vars = cp.Variable(n_coarse)
        obj = cp.Minimize(cp.sum(cp.maximum(corr_coarse + penalties, 0)) * dx_coarse) 
        
        constraints = [
            cp.sum(h_vars) == n_coarse/2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                h_new = np.maximum(0, np.minimum(1, h_new))
                h_new = h_new * (n_coarse/2) / np.sum(h_new)
                
                # Update penalties based on constraint violations
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                penalties = update_penalties(penalties, corr_new)
                
                # Evaluate and update best solution
                c5_new = evaluate_erdos_solution(h_new, dx_coarse)
                if c5_new < best_c5:
                    # Interpolate back to fine grid
                    h_interpolated = np.interp(np.linspace(0, 2, n_points), 
                                            np.linspace(0, 2, n_coarse), h_new)
                    h_interpolated = h_interpolated * (n_points/2) / np.sum(h_interpolated)
                    new_c5 = evaluate_erdos_solution(h_interpolated, dx)
                    if new_c5 < best_c5:
                        best_h = h_interpolated
                        best_c5 = new_c5
        except Exception as e:
            pass
        
        h_coarse = h_new.copy()
    
    # Upsample and refine with optimization
    n_fine = min(40, n_points * 2)
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), h_coarse)
    h_fine = h_fine * (n_fine/2) / np.sum(h_fine)
    dx_fine = 2.0 / n_fine
    penalties_fine = np.ones(n_fine * 2 - 1)
    
    for iteration in range(200):  # More iterations for fine tuning
        if time() - start_time > budget_s * 0.95:
            break
            
        corr_fine = np.correlate(h_fine, 1 - h_fine, mode='full')
        
        h_vars = cp.Variable(n_fine)  
        obj = cp.Minimize(cp.sum(cp.maximum(corr_fine + penalties_fine, 0)) * dx_fine)
        constraints = [
            cp.sum(h_vars) == n_fine/2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_vars.value
                h_new = np.maximum(0, np.minimum(1, h_new))
                h_new = h_new * (n_fine/2) / np.sum(h_new)
                
                corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                penalties_fine = penalties_fine * 0.9 + 0.1 * (corr_new - penalties_fine)  # adaptive
                
                if iteration % 50 == 0:  # Alternate penalty types
                    penalties_fine = update_penalties(penalties_fine, corr_new, 0.2)
                
                # Interpolate to original grid and evaluate 
                h_interp = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_fine), h_new)
                h_interp = h_interp * (n_points/2) / np.sum(h_interp)
                c5_new = evaluate_erdos_solution(h_interp, dx)
                if c5_new < best_c5:
                    best_h = h_interp 
                    best_c5 = c5_new
        except Exception as e:
            pass
        
        h_fine = h_new.copy()
    
    # Final SQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            # start with best known solution
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
