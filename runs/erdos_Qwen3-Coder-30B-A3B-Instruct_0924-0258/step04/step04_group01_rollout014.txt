import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create a structured initial guess with 3 clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # BOBYQA optimization parameters
    max_iter = 500
    initial_radius = 0.5
    min_radius = 1e-5
    
    # Function wrapper for BOBYQA that handles constraints
    def objective(h):
        # Clip h to feasible range
        h_clipped = np.clip(h, 0, 1)
        # Normalize sum constraint
        h_normalized = h_clipped * (n_points/2) / np.sum(h_clipped)
        return evaluate_erdos_solution(h_normalized, dx)
    
    # Base optimization using BOBYQA
    try:
        # Use bounds: [0,1] for each h[i], sum constraint handled internally
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(objective, h_start, method='bobyqa',
                       bounds=bounds, options={
                           'maxiter': max_iter,
                           'catol': 1e-6,
                           'rhobeg': initial_radius,
                           'rhoend': min_radius
                       })
        
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception as e:
        pass

    # Final cleanup to ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Verify final result
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
