import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_old, n_new):
    """Resample h_old to n_new points using linear interpolation"""
    old_points = np.linspace(0, 2, len(h_old))
    new_points = np.linspace(0, 2, n_new)
    return np.interp(new_points, old_points, h_old)

def project_to_constraints(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fine grid for main optimization
    n_fine = 100
    dx_fine = 2.0 / n_fine
    h_fine = linear_resample(h_start, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    # Optimization loop
    remaining_time = budget_s
    step = 0
    pert_scale = 0.03
    max_iter = 1000
    
    while step < max_iter and (time() - start_time) < budget_s * 0.95:
        step += 1
        remaining_time = budget_s - (time() - start_time)
        
        if remaining_time < 10:
            break
            
        # Random perturbation with adaptive scale
        h_pert = h_fine + np.random.normal(0, pert_scale, n_fine) * 0.5
        h_pert = project_to_constraints(h_pert, n_fine)
        
        # Local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx_fine)
            
        def constraint_sum(h):
            return np.sum(h) - n_fine / 2.0
            
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_fine)]
        
        try:
            res = minimize(objective, h_pert, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = linear_resample(h_refined, n_points)
                    best_c5 = c5_refined
                    # Early stopping
                    if best_c5 < 0.3807:
                        break
        except:
            pass
        
        # Update perturbation scale
        if step % 20 == 0:
            pert_scale *= 0.99
            
        # Periodic constraint projection
        if step % 10 == 0:
            h_fine = project_to_constraints(h_fine, n_fine)
    
    # Final evaluation and refinement
    h_final = project_to_constraints(best_h, n_points)
    if n_points < len(h_final):
        h_final = h_final[:n_points]
    elif n_points > len(h_final):
        h_final = linear_resample(h_final, n_points)
    
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
