import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create an initial guess
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Concentrate mass in two regions for better overlap reduction
        h_start[5:15] = 0.8
        h_start[20:30] = 0.4
        # Normalize to sum to n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    max_iter = 1000
    step_size = 0.01
    
    for iter_count in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient for shift k_max
        grad = np.zeros(n_points)
        if k_max < n_points:
            # Add contributions from h[i] * (1 - h[i+k_max])
            for i in range(n_points - k_max):
                grad[i] += (1 - best_h[i + k_max])
            # Add contributions from h[i+k_max] * (1 - h[i])
            for i in range(n_points - k_max):
                grad[i + k_max] -= best_h[i] / dx
        else:
            # Cross-correlation case - this shift is in the middle
            shift = k_max - (n_points - 1)
            for i in range(n_points + shift):
                grad[i] += (1 - best_h[i - shift])
            for i in range(n_points + shift):
                grad[i - shift] -= best_h[i] / dx
        
        # Update h in gradient descent direction
        h_new = best_h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            
        # Adaptive step size
        if new_c5 >= max_overlap:
            step_size *= 0.5
        else:
            step_size = min(step_size * 1.1, 0.1)
    
    # Final optimization of best solution with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        from scipy.optimize import minimize
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
