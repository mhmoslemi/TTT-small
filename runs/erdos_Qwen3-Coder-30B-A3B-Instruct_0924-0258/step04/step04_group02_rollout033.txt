import numpy as np
from scipy.optimize import minimize
from time import time
import math

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-z))

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Convert to z-space representation
    def h_from_z(z, b):
        return sigmoid(z + b)
    
    def compute_b(z, target_sum):
        # Binary search to find b such that sum(h(z+b)) = target_sum
        low, high = -10, 10
        while high - low > 1e-8:
            mid = (low + high) / 2
            h_val = sigmoid(z + mid)
            if np.sum(h_val) > target_sum:
                high = mid
            else:
                low = mid
        return (low + high) / 2
    
    # Initialize z from h_start, find b
    z = np.log(h_start / (1 - h_start))
    b = compute_b(z, n_points / 2)
    h = h_from_z(z, b)
    
    # Target objective
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Gradient computation for overlap at shift k
    def compute_overlap_gradient(h, k):
        # Compute gradient of overlap with shift k
        # d/dh_i C_k = 1 - 2*h[i] - sum_{j: j+k=i} h[j]
        # But sum_{j: j+k=i} h[j] for j >= 0, j+k < n_points
        # This is the number of times h[i] contributes to C_k
        grad = np.zeros_like(h)
        # For h[i], this contributes to C_k when i+k < n_points (and i >= 0)
        # For these cases, we have h[i+k]
        # This increments 1 for h[i] and -1 for h[i+k]
        for i in range(len(h)):
            if i + k < len(h):
                grad[i] += 1
                if i >= k:
                    grad[i-k] -= 1
        grad -= 2 * h
        return grad
    
    # Compute new z update using gradient descent with line search
    def line_search_step(z, b, lr):
        h = h_from_z(z, b)
        # Find worst shift k
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        # Compute gradient of C_k_max
        grad = compute_overlap_gradient(h, k_max)
        # Project gradient to valid direction
        grad_h = grad * h * (1 - h)
        # Update in z space
        z_new = z - lr * grad_h
        new_b = compute_b(z_new, n_points / 2)
        h_new = h_from_z(z_new, new_b)
        return z_new, new_b, h_new
    
    def optimize_step():
        # Gradient-based optimization
        max_iter = 200
        lr = 0.1
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.9:
                break
            h = h_from_z(z, b)
            corr = np.correlate(h, 1-h, mode='full')
            k_max = np.argmax(corr)
            grad = compute_overlap_gradient(h, k_max)
            grad_h = grad * h * (1 - h)
            # Line search
            for _ in range(20):
                z_new = z - lr * grad_h
                new_b = compute_b(z_new, n_points / 2)
                h_new = h_from_z(z_new, new_b)
                new_c5 = evaluate_erdos_solution(h_new, dx)
                if new_c5 < best_c5:
                    z = z_new
                    b = new_b
                    h = h_new
                    if new_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = new_c5
                    break
                lr *= 0.5
            else:
                lr *= 0.9
    
    # Optimizing main loop
    for iteration in range(100):
        if time() - start_time > budget_s * 0.9:
            break
        optimize_step()
        # Try local refinement
        try:
            h_refined = h_from_z(z, b)
            x0 = z.copy()
            
            def obj_func(z_vec):
                b_val = compute_b(z_vec, n_points / 2)
                h_val = h_from_z(z_vec, b_val)
                return evaluate_erdos_solution(h_val, dx)
            
            def jac_func(z_vec):
                b_val = compute_b(z_vec, n_points / 2)
                h_val = h_from_z(z_vec, b_val)
                corr = np.correlate(h_val, 1-h_val, mode='full')
                k_max = np.argmax(corr)
                grad = compute_overlap_gradient(h_val, k_max)
                # Apply chain rule dh/dz
                dh_dz = h_val * (1 - h_val)
                return -grad * dh_dz  # Negative gradient for minimization
            
            # Use BFGS-based method for refinement
            res = minimize(obj_func, x0, method='L-BFGS-B', jac=jac_func, bounds=[(-10,10)]*n_points, options={'maxiter': 50})
            if res.success:
                new_b = compute_b(res.x, n_points / 2)
                new_h = h_from_z(res.x, new_b)
                new_c5 = evaluate_erdos_solution(new_h, dx)
                if new_c5 < best_c5:
                    z = res.x
                    b = new_b
                    h = new_h
                    best_h = h.copy()
                    best_c5 = new_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
