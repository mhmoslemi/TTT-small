import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_orig, n_new):
    """Linearly resample h_orig to n_new points"""
    n_orig = len(h_orig)
    x_orig = np.linspace(0, 2, n_orig)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_orig, h_orig)

def project_to_constraints(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h_proj = np.clip(h, 0, 1)
    h_sum = np.sum(h_proj)
    if h_sum > 0:
        h_proj = h_proj * (n_points / 2) / h_sum
    return h_proj

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(35) * 0.5
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    # Initialize coarse and fine grids
    n_coarse = 50
    n_fine = 100
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    h_coarse = linear_resample(h_init, n_coarse)
    h_fine = linear_resample(h_init, n_fine)
    
    h_coarse = project_to_constraints(h_coarse, n_coarse)
    h_fine = project_to_constraints(h_fine, n_fine)
    
    # Initial evaluations
    c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    best_c5 = min(c5_coarse, c5_fine)
    best_h = h_fine.copy()
    
    # Coarse grid search
    for i in range(100):
        if time() - start_time > budget_s * 0.5:
            break
        # Perturb h_coarse
        perturb_scale = 0.03 * (0.99 ** i)
        h_pert = h_coarse + np.random.normal(0, perturb_scale, n_coarse)
        h_pert = project_to_constraints(h_pert, n_coarse)
        c5_test = evaluate_erdos_solution(h_pert, dx_coarse)
        if c5_test < best_c5:
            best_c5 = c5_test
            h_coarse = h_pert
            h_fine = linear_resample(h_pert, n_fine)
            h_fine = project_to_constraints(h_fine, n_fine)
            c5_fine = c5_test
    
    # Fine grid local refinement
    for i in range(200):
        if time() - start_time > budget_s * 0.9:
            break
        # Perturb h_fine
        perturb_scale = 0.02 * (0.99 ** i)
        h_pert = h_fine + np.random.normal(0, perturb_scale, n_fine)
        h_pert = project_to_constraints(h_pert, n_fine)
        c5_test = evaluate_erdos_solution(h_pert, dx_fine)
        if c5_test < best_c5:
            best_c5 = c5_test
            h_fine = h_pert
            # Interpolate back to coarse for future coarse search
            h_coarse = linear_resample(h_pert, n_coarse)
            h_coarse = project_to_constraints(h_coarse, n_coarse)
            c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx_fine)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_fine)]
        h_init_refine = project_to_constraints(h_fine, n_fine)
        
        try:
            res = minimize(objective, h_init_refine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = project_to_constraints(res.x, n_fine)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Return the best solution at original resolution
    final_h = linear_resample(best_h, n_points)
    final_h = project_to_constraints(final_h, n_points)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
