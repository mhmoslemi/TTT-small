import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Resample h_in to n_out points using linear interpolation."""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum(h) = target_sum with bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set up initial solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid search with warm start
    n_coarse = 50
    h_coarse = linear_resample(h_start, n_coarse)
    h_coarse = project_to_constraints(h_coarse, n_coarse / 2.0)
    dx_coarse = 2.0 / n_coarse
    best_coarse = h_coarse.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    
    # Fine grid for refinement
    n_fine = 100
    h_fine = linear_resample(h_start, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine / 2.0)
    dx_fine = 2.0 / n_fine
    best_fine = h_fine.copy()
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse grid exploration
    for step in range(100):
        if time() - start_time > budget_s * 0.4:
            break
        # Perturb
        perturbation = np.random.normal(0, 0.03, n_coarse)
        h_tmp = h_coarse + perturbation
        h_tmp = project_to_constraints(h_tmp, n_coarse / 2.0)
        
        c5_tmp = evaluate_erdos_solution(h_tmp, dx_coarse)
        if c5_tmp < best_c5_coarse:
            best_c5_coarse = c5_tmp
            best_coarse = h_tmp.copy()
    
    # Transfer to fine grid
    h_fine = linear_resample(best_coarse, n_fine)
    h_fine = project_to_constraints(h_fine, n_fine / 2.0)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Fine grid refinement with local optimization
    def objective_fine(h):
        return evaluate_erdos_solution(h, dx_fine)
    
    def constraint_fine(h):
        return np.sum(h) - n_fine / 2.0
    
    cons_fine = {'type': 'eq', 'fun': constraint_fine}
    bounds_fine = [(0, 1) for _ in range(n_fine)]
    
    for restart in range(30):
        if time() - start_time > budget_s * 0.9:
            break
            
        if restart == 0:
            h_init = h_fine.copy()
        else:
            # Perturb the solution
            perturbation = np.random.normal(0, 0.01, n_fine)
            h_init = h_fine + perturbation
            h_init = project_to_constraints(h_init, n_fine / 2.0)
            
        try:
            res = minimize(objective_fine, h_init, method='SLSQP',
                           bounds=bounds_fine, constraints=cons_fine,
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective_fine(h_refined)
                if c5_refined < best_c5_fine:
                    best_c5_fine = c5_refined
                    best_fine = h_refined
        except:
            continue
    
    # Final refinement with high precision
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective_fine, best_fine, method='SLSQP',
                           bounds=bounds_fine, constraints=cons_fine,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective_fine(h_final)
                if c5_final < best_c5_fine:
                    best_c5_fine = c5_final
                    best_fine = h_final
        except:
            pass
    
    # Return best solution
    final_h = best_fine
    final_c5 = best_c5_fine
    final_n = n_fine
    
    return (list(final_h), final_c5, final_n)
