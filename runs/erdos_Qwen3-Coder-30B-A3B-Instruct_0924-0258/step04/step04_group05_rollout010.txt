import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local refinement first
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Multi-perturbation search with adaptive perturbations
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Parameters
    perturb_sigma = 0.02
    perturb_decay = 0.95
    max_iter = 500
    retry_limit = 3
    current_retry = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb
        h_perturbed = h_current + np.random.normal(0, perturb_sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Local refinement on perturbed point
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                
                if c5_refined < c5_current:
                    h_current = h_refined
                    c5_current = c5_refined
                    current_retry = 0
                    perturb_sigma *= perturb_decay
                else:
                    current_retry += 1
                    if current_retry > retry_limit:
                        perturb_sigma *= 1.1
                        current_retry = 0
            else:
                current_retry += 1
                if current_retry > retry_limit:
                    perturb_sigma *= 1.1
                    current_retry = 0
        except:
            current_retry += 1
            if current_retry > retry_limit:
                perturb_sigma *= 1.1
                current_retry = 0
        
        # Update global best
        if c5_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c5_current
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
