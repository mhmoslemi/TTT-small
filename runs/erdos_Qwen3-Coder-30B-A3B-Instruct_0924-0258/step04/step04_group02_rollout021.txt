import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        # Perturb slightly
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    # Current and best solutions
    current_h = h.copy()
    best_h = h.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    T = T_initial
    max_iterations = int(1e5)
    
    # Iteration counter and time tracking
    iteration = 0
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Generate new candidate
        h_new = current_h.copy()
        
        # Select two random indices to perturb
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute perturbation
        delta_max = 0.1 * T
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Ensure bounds
        if h_new[i] + delta > 1:
            delta = 1 - h_new[i]
        elif h_new[i] + delta < 0:
            delta = -h_new[i]
        
        # Apply perturbation
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds and maintain sum
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / T):
            current_h = h_new
            current_c5 = c5_new
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        # Cool down
        T *= alpha
        iteration += 1
    
    return (list(best_h), best_c5, n_points)
