import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initialize population
    population_size = 40
    population = []
    
    # Generate diverse initial population
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random valid solutions
    for _ in range(5):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
    
    # Initialize tabu list and parameters
    tabu_list = []
    tabu_limit = 50
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_counter = 0
    best_fitness = best_c5
    
    # Genetic algorithm main loop
    max_generations = 500
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        fitness_scores = []
        for h in population:
            c5_score = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5_score)
        
        # Select best
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness_scores[best_idx]
            stagnation_counter = 0
            best_fitness = best_c5
            
        # Stop if target reached
        if best_c5 < 0.3807:
            break
            
        # Check for stagnation
        if stagnation_counter > 50:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
            
        # Create next generation
        new_population = []
        sorted_indices = np.argsort(fitness_scores)
        
        # Keep top 10
        for i in range(10):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate rest through crossover/mutation
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1 = population[random.choice(sorted_indices[:20])]
            parent2 = population[random.choice(sorted_indices[:20])]
            
            # Crossover with probability
            if random.random() < crossover_prob:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mut_idx = random.randint(0, n_points-1)
                child[mut_idx] = random.random()
                
            # Ensure constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Avoid tabu solutions
            if tuple(child.round(6)) not in tabu_list:
                new_population.append(child)
                if len(tabu_list) < tabu_limit:
                    tabu_list.append(tuple(child.round(6)))
                else:
                    tabu_list.pop(0)
                    tabu_list.append(tuple(child.round(6)))
        
        population = new_population[:population_size]
        stagnation_counter += 1
        
        # Periodically refine with gradient descent
        if generation % 10 == 0 and generation > 0:
            for i in range(min(5, len(population))):
                h_refine = population[i].copy()
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_refine, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-6})
                    if res.success:
                        h_new = res.x
                        h_new = h_new * (n_points/2) / np.sum(h_new)
                        c5_new = objective(h_new)
                        if c5_new < evaluate_erdos_solution(population[i], dx):
                            population[i] = h_new
                except:
                    pass
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
