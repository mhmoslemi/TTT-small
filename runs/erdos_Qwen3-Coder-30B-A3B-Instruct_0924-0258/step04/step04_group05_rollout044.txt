import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Read initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm with tabu search
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    stagnation_limit = 50
    max_gens = 1000
    
    # Initialize population
    population = []
    fitness = []
    
    # Initialize with varied solutions
    population.append(h_start.copy())
    fitness.append(best_c5)
    
    # Add diverse solutions
    for i in range(1, population_size):
        if i <= 10:
            # Perturbation of initial solution
            h_var = h_start + np.random.normal(0, 0.02, n_points)
            h_var = np.clip(h_var, 0, 1)
            h_var = h_var * (n_points/2) / np.sum(h_var)
        else:
            # Random valid solution
            h_var = np.random.random(n_points)
            h_var = h_var * (n_points/2) / np.sum(h_var)
        
        population.append(h_var)
        fitness.append(evaluate_erdos_solution(h_var, dx))
    
    tabu_list = []
    best_global = best_c5
    best_global_h = best_h.copy()
    
    gen = 0
    stagnation_count = 0
    
    while gen < max_gens and (time() - start_time) < budget_s * 0.98:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        if sorted_fit[0] < best_global:
            best_global = sorted_fit[0]
            best_global_h = sorted_pop[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Apply adaptive parameters based on stagnation
        if stagnation_count > stagnation_limit:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
            
        # Refine top solutions with local search
        if gen % 10 == 0 and gen > 0:
            for i in range(min(5, len(sorted_pop))):
                h_refined = sorted_pop[i].copy()
                # Local refinement using SLSQP
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_refined, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        new_fit = objective(h_refined)
                        if new_fit < sorted_fit[i]:
                            sorted_pop[i] = h_refined
                            sorted_fit[i] = new_fit
                            if new_fit < best_global:
                                best_global = new_fit
                                best_global_h = h_refined.copy()
                except:
                    pass
        
        # Create new generation
        new_pop = []
        new_fitness = []
        
        # Elitism: keep best half
        keep_count = population_size // 2
        for i in range(keep_count):
            new_pop.append(sorted_pop[i].copy())
            new_fitness.append(sorted_fit[i])
        
        # Generate offspring
        while len(new_pop) < population_size:
            if random.random() < crossover_prob:
                # Tournament selection
                parent1_id = random.randint(0, keep_count-1)
                parent2_id = random.randint(0, keep_count-1)
                while parent2_id == parent1_id:
                    parent2_id = random.randint(0, keep_count-1)
                    
                parent1 = sorted_pop[parent1_id]
                parent2 = sorted_pop[parent2_id]
                
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
                
                # Add mutation
                if random.random() < mutation_rate:
                    mutations = np.random.randint(0, n_points, 
                                                max(1, int(n_points * mutation_rate)))
                    for i in mutations:
                        child[i] = random.uniform(0, 1)
                
                # Repair solution to satisfy constraint
                child = child * (n_points/2) / np.sum(child)
                child = np.clip(child, 0, 1)
                
                # Check if already in tabu list
                tabu_key = tuple(np.round(child, decimals=6))
                if tabu_key not in tabu_list:
                    new_pop.append(child)
                    new_fitness.append(evaluate_erdos_solution(child, dx))
                    if len(tabu_list) >= tabu_limit:
                        tabu_list.pop(0)
                    tabu_list.append(tabu_key)
            else:
                # Random solution
                h_rand = np.random.random(n_points)
                h_rand = h_rand * (n_points/2) / np.sum(h_rand)
                child = np.clip(h_rand, 0, 1)
                
                tabu_key = tuple(np.round(child, decimals=6))
                if tabu_key not in tabu_list:
                    new_pop.append(child)
                    new_fitness.append(evaluate_erdos_solution(child, dx))
                    if len(tabu_list) >= tabu_limit:
                        tabu_list.pop(0)
                    tabu_list.append(tabu_key)
        
        population = new_pop
        fitness = new_fitness
        
        gen += 1
        
        # Early stopping if target met
        if best_global < 0.3807:
            break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        # Refine the best found solution
        h_final = best_global_h.copy()
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            res = minimize(objective, h_final, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_global:
                    best_global = c5_final
                    best_global_h = h_final.copy()
        except:
            pass
    
    return (list(best_global_h), best_global, n_points)
