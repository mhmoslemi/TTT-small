import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    # Ensure sum constraint
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    # Ensure sum is exactly correct (avoid floating point issues)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    T = T_initial
    
    # Track best solution
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Compute maximum delta that respects bounds
        delta_max = min(1.0 - current_h[i], current_h[j])
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Create candidate
        h_candidate = current_h.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Ensure sum constraint (maintain sum to n_points/2)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute overlap for candidate
        candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = h_candidate
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        T *= alpha
    
    # Final refinement with local search (limited iterations)
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Simple hill climbing
        for _ in range(1000):
            if time() - start_time > budget_s * 0.95:
                break
                
            i, j = np.random.choice(n_points, 2, replace=False)
            delta_max = min(1.0 - best_h[i], best_h[j])
            delta = np.random.uniform(-delta_max, delta_max)
            
            h_candidate = best_h.copy()
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            candidate_c5 = evaluate_erdos_solution(h_candidate, dx)
            
            if candidate_c5 < best_c5:
                best_h = h_candidate
                best_c5 = candidate_c5
    
    return (list(best_h), best_c5, n_points)
