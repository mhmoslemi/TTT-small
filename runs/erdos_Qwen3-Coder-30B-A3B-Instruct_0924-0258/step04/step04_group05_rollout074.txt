import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    
    # Initialize population with variations of the starting solution
    population = [h_start.copy()]
    for _ in range(9):
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
    
    # Add random solutions
    for _ in range(5):
        h_random = np.random.random(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
    
    # Enforce population size
    population = population[:population_size]
    
    # Tabu memory for solutions
    tabu_list = set()
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # Helper to make solution hashable
    def hash_solution(h):
        return tuple(np.round(h, decimals=6))
    
    def repair_solution(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Objective function for local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Iterate
    stagnation_count = 0
    max_stagnation = 50
    
    for generation in range(1000):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            # Ensure feasibility
            h = repair_solution(h)
            fit = objective(h)
            fitness.append(fit)
            
            # Update best solution
            if fit < best_c5:
                best_c5 = fit
                best_h = h.copy()
                
                # Early termination
                if best_c5 < 0.3807:
                    break
                    
        # Apply tabu search: avoid revisiting recent solutions
        for i, h in enumerate(population):
            h_hash = hash_solution(h)
            if h_hash in tabu_list:
                # Replace with random valid solution
                new_h = np.random.random(n_points)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                population[i] = new_h
                
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:5]]
        
        # Refine top solutions with gradient descent
        for h in elite:
            if time() - start_time > budget_s * 0.95:
                break
            try:
                # Make a copy for refinement
                h_refined = h.copy()
                h_refined = repair_solution(h_refined)
                
                # Local optimization
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(objective, h_refined, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    h_refined = repair_solution(h_refined)
                    fit = objective(h_refined)
                    if fit < best_c5:
                        best_c5 = fit
                        best_h = h_refined.copy()
                        
                    # Add to population
                    if len(population) < population_size:
                        population.append(h_refined)
                    else:
                        # Replace worst element
                        worst_idx = np.argmax([objective(p) for p in population])
                        population[worst_idx] = h_refined
            except:
                continue
                
        # Genetic operations
        if len(population) < population_size:
            continue
            
        new_population = elite[:]
        while len(new_population) < population_size:
            if time() - start_time > budget_s * 0.95:
                break
                
            # Selection based on fitness (tournament selection)
            tournament_size = 3
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            parent1 = population[parent1_idx]
            
            # Sample parent2
            tournament_indices2 = random.sample(range(len(population)), tournament_size)
            tournament_fitness2 = [fitness[i] for i in tournament_indices2]
            parent2_idx = tournament_indices2[np.argmin(tournament_fitness2)]
            parent2 = population[parent2_idx]

            # Crossover
            if random.random() < crossover_prob:
                # Uniform crossover mask
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                # Add random noise
                noise = np.random.normal(0, 0.05, n_points)
                child = child + noise
                child = repair_solution(child)
                
            # Ensure diversity
            child_hash = hash_solution(child)
            if child_hash in tabu_list:
                # Mutate further or make a new random solution
                child = repair_solution(np.random.random(n_points))
                child = child * (n_points/2) / np.sum(child)
                
            new_population.append(child)
            
            # Add to tabu list
            if len(tabu_list) >= tabu_limit:
                # Remove oldest
                tabu_list.pop()
            tabu_list.add(child_hash)
            
        population = new_population
        
        # Adaptive parameters
        if best_c5 < 0.3807 or stagnation_count > max_stagnation:
            if mutation_rate < 0.5:
                mutation_rate += 0.1
            if crossover_prob > 0.3:
                crossover_prob -= 0.1
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 < 0.3807:
            break
            
    if best_h is None:
        best_h = h_start
        best_c5 = evaluate_erdos_solution(best_h, dx)
        
    return (list(best_h), best_c5, n_points)
