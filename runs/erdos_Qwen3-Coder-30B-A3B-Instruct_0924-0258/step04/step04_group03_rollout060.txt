import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy
    population_size = 50
    num_generations = 1000
    elite_ratio = 0.2
    mutation_step_min = 0.001
    mutation_step_max = 0.05
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        # Perturb initial solution
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Main evolutionary loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort and select elites
        sorted_indices = np.argsort(fitness)
        elite_count = int(elite_ratio * population_size)
        elites = [population[i] for i in sorted_indices[:elite_count]]
        best_in_gen = fitness[sorted_indices[0]]
        if best_in_gen < best_c5:
            best_c5 = best_in_gen
            best_h = population[sorted_indices[0]].copy()
        
        # Generate new population
        new_population = elites.copy()
        
        # Mutation and crossover
        while len(new_population) < population_size:
            # Mutation
            if np.random.rand() < 0.8:
                parent = elites[np.random.randint(0, len(elites))]
                mutation_step = np.random.uniform(mutation_step_min, mutation_step_max)
                child = parent + np.random.normal(0, mutation_step, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            else:
                # Crossover
                parent1 = elites[np.random.randint(0, len(elites))]
                parent2 = elites[np.random.randint(0, len(elites))]
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Gradient refinement of top candidates
    num_refinements = min(5, len(population))
    fitness = []
    for h in population:
        c5 = evaluate_erdos_solution(h, dx)
        fitness.append(c5)
    
    sorted_indices = np.argsort(fitness)
    for i in range(num_refinements):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_refined = population[sorted_indices[i]].copy()
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_refined, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
