import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a 3-cluster structure with some randomness
        cluster_centers = [0.4, 1.0, 1.6]
        cluster_width = 0.3
        for center in cluster_centers:
            start = max(0, int((center - cluster_width/2)/dx))
            end = min(n_points, int((center + cluster_width/2)/dx))
            h_start[start:end] = 0.8
        # Add random perturbations
        h_start += np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        # Normalize sum
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    pop_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main evolution loop
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness (inverse of C5)
        fitness = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness.append(1.0 / (c5_val + 1e-10))  # Avoid division by zero
        
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitness)[::-1]
        selected_indices = sorted_indices[:pop_size//2]
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population via crossover and mutation
        new_population = selected_population.copy()
        
        # Elitism: keep best solution
        best_idx = np.argmin([evaluate_erdos_solution(h, dx) for h in population])
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(selected_indices)
            parent2_idx = np.random.choice(selected_indices)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.02)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.02)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points/2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        # Refine population with SQP
        for i in range(len(new_population)):
            h_current = new_population[i]
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_new = res.x
                    new_population[i] = h_new
            except:
                pass
        
        population = new_population
        
        # Update global best
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            if c5_val < best_c5:
                best_h = h.copy()
                best_c5 = c5_val
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
