import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Resample h_in to n_out points using linear interpolation."""
    x_in = np.linspace(0, 2, len(h_in))
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Precompute and store for performance
    h_fine = linear_resample(h_start, 100) if n_points < 100 else h_start.copy()
    h_fine = project_to_feasible(h_fine, 100)
    dx_fine = 2.0 / 100
    
    # Coarse grid search to get a good starting point
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = linear_resample(h_start, n_coarse)
    h_coarse = project_to_feasible(h_coarse, n_coarse)
    
    # Tweak parameters for more effective searching
    max_coarse_steps = 100  # cooldown effective search steps
    max_fine_steps = 200
    perturb_scale = 0.03
    perturb_decay = 0.99
    adaptive_scale = 0.03
    
    # Coarse grid global search
    for step in range(max_coarse_steps):
        if time() - start_time > budget_s * 0.5:
            break
            
        # Perturbation
        h_perturbed = h_coarse + np.random.normal(0, adaptive_scale, n_coarse)
        h_perturbed = project_to_feasible(h_perturbed, n_coarse)
        
        # Evaluate
        c5_current = evaluate_erdos_solution(h_perturbed, dx_coarse)
        if c5_current < best_c5:
            best_h = linear_resample(h_perturbed, n_points)
            best_c5 = c5_current
            
        # Reduce perturbation over time
        adaptive_scale *= perturb_decay
    
    # Fine grid refinement
    # Take refined best solution on coarse grid and bring to fine grid
    h_refined = linear_resample(best_h, 100)
    h_refined = project_to_feasible(h_refined, 100)
    dx_refined = 2.0 / 100
    
    # Local optimization at fine level
    def objective(h):
        return evaluate_erdos_solution(h, dx_refined)
    
    def constraint_sum(h):
        return np.sum(h) - 50.0  # corresponds to n_points = 100
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(100)]
    
    for _ in range(max_fine_steps):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Perturb and project
        h_perturbed = h_refined + np.random.normal(0, 0.01, 100)
        h_perturbed = project_to_feasible(h_perturbed, 100)
        
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_refined = project_to_feasible(res.x, 100)
                new_c5 = objective(h_refined)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = linear_resample(h_refined, n_points)
        except:
            continue
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        def objective_final(h):
            return evaluate_erdos_solution(h, dx)
        
        cons_final = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
        bounds_final = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective_final, best_h, method='SLSQP', 
                           bounds=bounds_final, constraints=cons_final,
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                final_h = res.x
                final_c5 = objective_final(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
