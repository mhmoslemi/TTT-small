import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create structured initial guess with 50 points
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Trust region optimization approach
    # Using L-BFGS-B with bounds and constraints
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization parameters
    max_iter = 1000
    last_improvement = 0
    patience = 50
    
    # Main optimization loop
    iteration = 0
    while iteration < max_iter and (time() - start_time) < budget_s * 0.95:
        # Apply random perturbation to current best and try to optimize
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-5})
            
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    last_improvement = iteration
        except:
            pass
        
        # Additional local refinement with different method
        try:
            # Try SLSQP on current best
            res2 = minimize(obj, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 30, 'ftol': 1e-6})
            
            if res2.success:
                h_refined2 = res2.x
                c5_refined2 = obj(h_refined2)
                if c5_refined2 < best_c5:
                    best_h = h_refined2.copy()
                    best_c5 = c5_refined2
                    last_improvement = iteration
        except:
            pass
        
        # Occasionally apply a global perturbation
        if iteration % 30 == 0:
            # Create new random candidate
            h_new = np.random.rand(n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            c5_new = obj(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                last_improvement = iteration
        
        # Check for stagnation
        if iteration - last_improvement > patience:
            break
            
        iteration += 1
    
    # Final cleanup to ensure constraint satisfaction
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
