import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary relaxation + local search approach
    def binary_relaxation(h):
        # Threshold and binarize
        h_binary = (h > 0.5).astype(float)
        # Scale to meet constraint
        h_binary = h_binary * (n_points/2) / np.sum(h_binary)
        return np.clip(h_binary, 0, 1)
    
    def local_search(h, max_iter=100):
        h = h.copy()
        for _ in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            # Create neighbor by random mutation
            h_new = h.copy()
            # Mutate about 10% of entries
            indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in indices:
                h_new[i] = 1 - h_new[i]  # Flip between 0 and 1
            # Renormalize to sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        return h
    
    # Main optimization loop
    max_iterations = 1000
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Binary relaxation of current best
        h_binary = binary_relaxation(best_h)
        
        # Perform local search refinement
        h_refined = local_search(h_binary)
        
        # Check if we made progress
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
        
        # Ensure sum constraint after randomization
        best_h = best_h * (n_points/2) / np.sum(best_h)
        best_h = np.clip(best_h, 0, 1)
    
    # Final refinement with nonlinear optimization
    if time() - start_time < budget_s * 0.95:
        try:
            # Use a fractional solution as starting point
            h_frac = best_h.copy()
            h_frac = h_frac * (n_points/2) / np.sum(h_frac)
            
            # Use SLSQP for fine-tuning
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, h_frac, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
