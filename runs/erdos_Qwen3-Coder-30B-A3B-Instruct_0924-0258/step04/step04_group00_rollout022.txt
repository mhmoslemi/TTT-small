import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points/2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # PGD with adaptive step size and Lagrangian
    alpha = 0.01
    max_iter = 1000
    patience = 50
    patience_count = 0
    last_improvement = 0
    
    # Gradient-based optimization with projected updates
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap and find max shift
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_shift_idx = np.argmax(corr)
        max_overlap = corr[max_shift_idx] * dx
        
        # For gradient approximation, compute finite differences
        # We directly compute gradients for the main term that we'll minimize
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Shift adjustment contributions
            for k in range(-n_points+1, n_points):
                if 0 <= i+k < n_points:
                    if k < 0:
                        grad[i] += 1 - h_current[i+k]
                    elif k > 0:
                        grad[i] += h_current[i+k]
        
        # Projected gradient step
        h_new = h_current - alpha * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum to n_points/2
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject update
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_count = 0
            last_improvement = iteration
            # Increase step size
            alpha *= 1.2
        else:
            patience_count += 1
            # Reduce step size
            alpha *= 0.5
            
            # Restore previous point if no improvement for a while
            if patience_count > patience:
                h_new = best_h.copy()
                c5_new = best_c5
        
        h_current = h_new
        
        # Early termination criteria
        if patience_count > patience * 2:
            break
    
    # Final refinement with simple smoothing
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    if c5_final < best_c5:
        best_h = h_final
        best_c5 = c5_final
    
    return (list(best_h), best_c5, n_points)
