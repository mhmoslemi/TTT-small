import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iterations = 60
    
    # Gradient descent parameters
    max_iter_per_check = 1000
    learning_rate = 0.1
    
    # Warm start
    current_h = best_h.copy()
    
    # Check feasibility using gradient method
    def check_feasibility(h, lambda_target):
        h = h.copy()
        for _ in range(max_iter_per_check):
            if time() - start_time > budget_s * 0.95:
                return False, h
                
            # Compute all auto-correlations
            corr_full = np.correlate(h, 1 - h, mode='full')
            # Compute correlation for shifts in [-1, 1]
            corr_shifts = corr_full[len(corr_full)//2 - n_points//2 : len(corr_full)//2 + n_points//2]
            k_min = np.argmin(corr_shifts)
            min_corr = corr_shifts[k_min]
            
            if min_corr >= lambda_target * dx:
                return True, h
                
            # Compute gradient with respect to h for the minimum correlation
            # We'll compute gradient using finite differences for simplicity
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(n_points):
                # Perturb h[i] and compute change in correlation
                h_pert = h.copy()
                h_pert[i] += eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                
                corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
                corr_shifts_pert = corr_pert[len(corr_pert)//2 - n_points//2 : len(corr_pert)//2 + n_points//2]
                min_corr_pert = corr_shifts_pert[k_min]
                
                grad[i] = (min_corr_pert - min_corr) / eps
            
            # Update h in direction of gradient ascent
            h += learning_rate * grad
            # Project to feasible region
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
            
        return False, h
    
    # Binary search loop
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        
        # Check if feasible with lambda_mid
        feasible, h_new = check_feasibility(current_h, lambda_mid)
        
        if feasible:
            lambda_low = lambda_mid
            current_h = h_new
            
            # Update best solution if better
            c5_new = evaluate_erdos_solution(current_h, dx)
            if c5_new < best_c5:
                best_h = current_h.copy()
                best_c5 = c5_new
        else:
            lambda_high = lambda_mid
            
        # Early stopping
        if abs(lambda_high - lambda_low) < tolerance:
            break
    
    # Final optimization with direct search
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
