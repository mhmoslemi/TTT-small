import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create multi-cluster structure
        cluster_positions = [0.5, 1.0, 1.5]
        cluster_width = 0.3
        for pos in cluster_positions:
            start_idx = max(0, int((pos - cluster_width) / dx))
            end_idx = min(n_points, int((pos + cluster_width) / dx))
            h_start[start_idx:end_idx] = 1.0
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing loop
    max_iter = 1500
    temp = 0.1
    temp_decay = 0.995
    perturbation_scale = 0.05
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Create new candidate
        h_candidate = best_h + np.random.normal(0, perturbation_scale, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum constraint
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # SQP refinement with trust region
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj, h_candidate, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
        
        # Simulated annealing acceptance
        c5_current = evaluate_erdos_solution(best_h, dx)
        if c5_current > 0:
            accept_prob = math.exp(-abs(best_c5 - c5_current) / temp)
            if np.random.rand() < accept_prob:
                best_h = h_refined.copy()
                best_c5 = c5_refined
        
        temp *= temp_decay
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
