import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with adaptive learning rate
    def compute_gradient(h, k_shift):
        # Compute gradient for shift k
        grad = np.zeros_like(h)
        if k_shift < len(h):
            # Convolution-like gradient 
            grad[:len(h)-k_shift] += (1 - h[k_shift:])
            grad[k_shift:] += -h[:len(h)-k_shift]
        else:
            # Reverse case
            k_shift_rev = k_shift - len(h) + 1
            grad[k_shift_rev:] += (1 - h[:len(h)-k_shift_rev])
            grad[:len(h)-k_shift_rev] += -h[k_shift_rev:]
        return grad
    
    def project_to_feasible(h):
        # Ensure h is in [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            h = h * (n_points/2) / current_sum
        return h
    
    # Use adaptive gradient descent
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    max_iter = 5000
    iter_count = 0
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Adaptive learning rate scheduling
    step_decay = 0.99
    min_alpha = 1e-5
    decay_every = 100
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.95:
        # Find the most binding shift
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        k_best = np.argmax(corr)
        max_val = corr[k_best]
        
        # Compute gradient
        grad = compute_gradient(current_h, k_best)
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
            
        # Update with adaptive learning rate
        h_new = current_h - alpha * grad
        h_new = project_to_feasible(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement
        if new_c5 < current_c5:
            current_h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Adaptive learning rate
        if iter_count % decay_every == 0 and alpha > min_alpha:
            alpha *= step_decay
            
        iter_count += 1
        
        # Occasionally refine with convex optimization
        if iter_count % 50 == 0 and (time() - start_time) < budget_s * 0.9:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                # Add few top overlap constraints
                corr_full = np.correlate(current_h, 1 - current_h, mode='full')
                sorted_indices = np.argsort(-corr_full)
                top_k = min(10, len(sorted_indices))
                for k in sorted_indices[:top_k]:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        k_rev = k - n_points + 1
                        corr_sum = cp.sum(cp.multiply(h[k_rev:], 1 - h[:n_points-k_rev]))
                    constraints.append(corr_sum <= max(0.1, np.max(corr_full) * dx) + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    # Final validation
    best_h = project_to_feasible(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
