import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with Augmented Lagrangian approach
    max_iter = 1000
    alpha = 0.01
    lambda_mult = 0.0
    mu = 0.1
    stagnation_count = 0
    max_stagnation = 50
    
    h_current = best_h.copy()
    c5_current = best_c5
    
    # Gradient computation for overlap integral
    def compute_gradient(h):
        # Compute all shifts and find max overlap index
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        
        # Compute gradient components
        grad = np.zeros_like(h)
        # For the given maximal overlap, compute gradient
        # This is a tricky part - we discretize it and do finite difference
        n = len(h)
        # For any shift k, the overlap is sum h[i] * (1 - h[i+k])
        
        # We approximate gradient via finite differences for each h[i]
        # and use only the maximizer k in a simplified way
        
        # Simplified gradient: for each i, consider influences of neighbors
        # Better approach: compute exact gradient from KKT conditions
        
        # exact gradient calculation
        # for each i, let's compute the contribution to the max overlap
        # Because we're looking at max_k sum h[i](1-h[i+k]) for some k
        # We'll compute it in the actual maximum shift direction
        
        # Start with gradient for the specific max shift
        # shift_k corresponds to max overlap
        shift_k = max_idx - (n - 1)  # Convert full correlation index to actual shift
        # We need to compute derivative of sum h[i] * (1 - h[i+shift_k])
        # Avoid tedious symbols, use a simplified method
        # for now, just use numerical gradient via small perturbations
        eps = 1e-6
        grad_approx = np.zeros_like(h)
        for i in range(n):
            h_pert = h.copy()
            h_pert[i] += eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n/2) / np.sum(h_pert)
            c5_plus = evaluate_erdos_solution(h_pert, dx)
            
            h_pert = h.copy()
            h_pert[i] -= eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n/2) / np.sum(h_pert)
            c5_minus = evaluate_erdos_solution(h_pert, dx)
            grad_approx[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return grad_approx
    
    # Main optimization loop
    for it in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        if stagnation_count >= max_stagnation:
            break
            
        # Get current gradient
        try:
            grad = compute_gradient(h_current)
        except:
            # fallback to a simple gradient descent with possible projection
            grad = np.random.randn(n_points)
            
        # Update via projected gradient descent
        h_new = h_current - alpha * grad
        
        # Project into [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Respect sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate the new point
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < c5_current:

    refined_c5 = evaluate_erdos_solution(smoothed_h, dx)
    
    if refined_c5 < best_c5:
        best_h = smoothed_h.copy()
        best_c5 = refined_c5
    
    # Final local search with simulated annealing
    h_ref = best_h.copy()
    c5_ref = best_c5
    temperature = 1.0
    
    while temperature > 1e-6:
        if time() - start_time > budget_s * 0.99:
            break
            
        # Perturb
        h_p = h_ref + np.random.normal(0, 0.01, n_points)
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (n_points / 2) / np.sum(h_p)
        
        # Evaluate
        c5_p = evaluate_erdos_solution(h_p, dx)
        
        # Accept or reject
        if c5_p < c5_ref or np.random.rand() < np.exp((c5_ref - c5_p) / temperature):
            h_ref = h_p
            c5_ref = c5_p
            
        temperature *= 0.999
            
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
    
    return (list(best_h), best_c5, n_points)
