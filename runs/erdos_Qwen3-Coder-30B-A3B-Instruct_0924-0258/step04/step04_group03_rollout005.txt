import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_scores = []
    
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
        # Normalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        score = evaluate_erdos_solution(h, dx)
        personal_best.append(h.copy())
        personal_best_scores.append(score)
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary constraints and sum constraint
            particles[i] = np.clip(particles[i], 0, 1)
            # Normalize to sum = n_points/2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
    
    # Convex refinement of the best solution
    try:
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # This constraint attempts to bound the max overlap
        # While not directly enforcing the max, it's a good approximation to 
        # what we're trying to minimize.
        # Here we just add enough constraints to make the problem actually solvable
        # and add a soft inequality.
        e = cp.Variable()
        constraints.append(e >= 0)
        for k in range(0, 2 * n_points - 1):
            corr_sum = 0.0
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= e)
        
        prob = cp.Problem(cp.Minimize(e), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-8)
            if prob.status == cp.OPTIMAL:
                refined_h = h.value
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < global_best_score:
                    global_best = refined_h
                    global_best_score = refined_c5
        except Exception:
            pass
    except Exception:
        pass
    
    # Final SLSQP polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            res = minimize(objective, global_best, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < global_best_score:
                    global_best = final_h
                    global_best_score = final_c5
        except:
            pass
    
    return (list(global_best), global_best_score, n_points)
