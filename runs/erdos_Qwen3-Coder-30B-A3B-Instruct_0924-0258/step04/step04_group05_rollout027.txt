import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Ensure we have exactly the right number of points
    if len(h_start) != n_points:
        h_start = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h_start)), h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Coarse grid search
    grid_size = 60
    search_range = np.linspace(0.365, min(best_c5, 0.38085), grid_size)
    best_c5_grid = best_c5
    
    for c_mid in search_range:
        if time() - start_time > budget_s * 0.45:
            break
            
        # Convex feasibility check with updated constraints
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all possible shifts
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5_grid:
                    best_c5_grid = c5_feasible
                    best_h = h_feasible.copy()
        except:
            continue
    
    # Local refinement with multiple strategies
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try different optimization strategies
    optimization_strategies = [
        {'method': 'SLSQP', 'options': {'maxiter': 150, 'ftol': 1e-8}},
        {'method': 'COBYLA', 'options': {'maxiter': 200, 'rhobeg': 0.05}},
        {'method': 'L-BFGS-B', 'options': {'maxiter': 150, 'ftol': 1e-8}}
    ]
    
    # Multiple restarts
    for restart in range(20):
        if time() - start_time > budget_s * 0.90:
            break
            
        # Alternate between exploitation and exploration
        if restart % 3 == 0:
            # Random perturbation
            h_init = best_h + np.random.normal(0, 0.03, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
        else:
            # Perturb best solution with targeted noise
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h_init = best_h * (1 + noise)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points/2) / np.sum(h_init)
            
        # Try multiple optimizers
        for strategy in optimization_strategies:
            if time() - start_time > budget_s * 0.95:
                break
                
            try:
                res = minimize(objective, h_init, method=strategy['method'], 
                               bounds=bounds, constraints=cons_sum, 
                               options=strategy['options'])
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        # Early stopping if we're close to target
                        if best_c5 < 0.3807:
                            break
            except:
                continue
    
    # Final refinement with high precision
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30 and best_c5 > 0.37:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-10})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    # Final safety check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
