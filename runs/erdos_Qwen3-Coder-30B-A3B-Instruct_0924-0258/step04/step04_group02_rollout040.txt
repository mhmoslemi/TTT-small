import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define objective and constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Gradient-free optimization with random restarts
    max_restarts = 10
    max_iter_per_restart = 500
    
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random initialization
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': max_iter_per_restart, 
                                   'ftol': 1e-8, 'gtol': 1e-6})
            
            if res.success:
                h_opt = res.x
                c5_opt = objective(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
        except:
            pass
            
        # Noise injection to escape local minima
        if restart % 5 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_noisy = best_h + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points/2) / np.sum(h_noisy)
            c5_noisy = objective(h_noisy)
            if c5_noisy < best_c5:
                best_h = h_noisy
                best_c5 = c5_noisy
    
    # Final refinement with custom optimization if time allows
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 60:
        # Try a hybrid approach - simulated annealing with local refinement
        h_current = best_h.copy()
        c5_current = best_c5
        temperature = 1.0
        cooling_rate = 0.95
        max_iterations = 1000
        
        for i in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Propose a new solution
            h_proposed = h_current + np.random.normal(0, 0.02, n_points)
            h_proposed = np.clip(h_proposed, 0, 1)
            h_proposed = h_proposed * (n_points / 2) / np.sum(h_proposed)
            
            c5_proposed = evaluate_erdos_solution(h_proposed, dx)
            
            # Accept or reject based on Metropolis criterion
            if c5_proposed < c5_current or np.random.rand() < np.exp((c5_current - c5_proposed) / temperature):
                h_current = h_proposed
                c5_current = c5_proposed
            
            if c5_current < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_current
                
            temperature *= cooling_rate
            
    return (list(best_h), best_c5, n_points)
