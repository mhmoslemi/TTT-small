import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Adjust to proper resolution
    if n_points < 40:
        # Interpolate to higher resolution
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 40)
        h_start = np.interp(x_new, x_orig, h_start)
        n_points = 40
    elif n_points > 40:
        # Downsample
        h_start = h_start[::n_points//40]
        n_points = 40
    
    # Ensure correct integral
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2.0) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive grid search with step size control
    grid_size = 50
    search_range = np.linspace(0.36, best_c5, grid_size)
    grid_search_step = 0.0005
    improvement_count = 0
    
    for i, c_mid in enumerate(search_range):
        if time() - start_time > budget_s * 0.5:
            break
            
        # Convex feasibility check using CVXPY
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Add overlap constraints for all possible shifts
            corr_size = 2 * n_points - 1
            for k in range(corr_size):
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= c_mid * dx)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-7, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_c5 = c5_feasible
                    best_h = h_feasible.copy()
                    improvement_count += 1
        except Exception:
            continue
    
    # Local refinement with multiple restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Track best locally refined solution
    local_best_h = best_h.copy()
    local_best_c5 = best_c5
    
    for restart in range(20):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the solution
        if restart == 0:
            h_init = best_h.copy()
        else:
            h_init = best_h.copy()
            # Add noise with adaptive variance
            sigma = 0.02 if restart < 10 else 0.01
            noise = np.random.normal(0, sigma, n_points)
            h_init = h_init + noise
            h_init = np.clip(h_init, 0, 1)
            # Renormalize
            h_init = h_init * (n_points / 2.0) / np.sum(h_init)
        
        # Try multiple optimizers
        try:
            # SLSQP
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < local_best_c5:
                    local_best_c5 = c5_refined
                    local_best_h = h_refined
                    if local_best_c5 < 0.3807:
                        break
        except Exception:
            continue
            
        if restart == 0:
            try:
                # COBYLA as backup
                res2 = minimize(objective, h_init, method='COBYLA', 
                                bounds=bounds, constraints=cons_sum,
                                options={'maxiter': 300, 'rhobeg': 0.01})
                if res2.success:
                    h_refined2 = res2.x
                    c5_refined2 = objective(h_refined2)
                    if c5_refined2 < local_best_c5:
                        local_best_c5 = c5_refined2
                        local_best_h = h_refined2
            except Exception:
                continue
    
    # Final optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        try:
            res = minimize(objective, local_best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 500, 'ftol': 1e-10})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < local_best_c5:
                    local_best_h = final_h
                    local_best_c5 = final_c5
        except Exception:
            pass
    
    return (list(local_best_h), local_best_c5, n_points)
