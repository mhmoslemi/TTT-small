import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Using logit parameterization
    def h_from_y(y, b):
        # Ensure h is in [0,1] and integral constraint
        exp_y = np.exp(y + b)
        return exp_y / (1.0 + exp_y)
    
    def compute_b(y):
        # b such that integral of h from y is 1
        return math.log((n_points/2) / (1.0 + np.sum(np.exp(y))))
    
    def compute_c5_from_y(y):
        b = compute_b(y)
        h = h_from_y(y, b)
        return evaluate_erdos_solution(h, dx)
    
    # Gradient computation using finite differences
    def compute_gradient(y):
        eps = 1e-6
        grad = np.zeros_like(y)
        base_c5 = compute_c5_from_y(y)
        for i in range(len(y)):
            y_plus = y.copy()
            y_plus[i] += eps
            c5_plus = compute_c5_from_y(y_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Frank-Wolfe style approach with adaptive step size
    max_iter = 1000
    y = np.log(h_start / (1.0 - h_start))  # logit transform of h_start
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        grad = compute_gradient(y)
        # Frank-Wolfe style update
        step_size = 0.1
        y_new = y - step_size * grad
        y = y_new
        
        # Evaluate current solution
        current_c5 = compute_c5_from_y(y)
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Convert back to h for evaluation
            b = compute_b(y)
            best_h = h_from_y(y, b)
            
    # Post-optimize with local refinement
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points/2.0
    
    # Further refine with optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Random restarts with local optimization
        best_refined = best_h.copy()
        best_c5_refined = best_c5
        
        for _ in range(5):
            # Start from current best
            h_restart = best_h + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 1e-6, 1 - 1e-6)
            # Normalize
            h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
            
            try:
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(1e-6, 1 - 1e-6) for _ in range(n_points)]
                
                res = minimize(obj_func, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj_func(h_refined)
                    if c5_refined < best_c5_refined:
                        best_c5_refined = c5_refined
                        best_refined = h_refined
            except:
                pass
        
        if best_c5_refined < best_c5:
            best_h = best_refined
            best_c5 = best_c5_refined
    
    return (list(best_h), best_c5, n_points)
