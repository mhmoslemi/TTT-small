import numpy as np
from cvxpy import *
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create high density at center
        center_start = int(0.4 / dx)
        center_end = int(1.6 / dx)
        h_start[center_start:center_end] = 1.0
        # Normalize to sum = n_points/2
        h_start = h_start * (n_points/2) / np.sum(h_start)

    # Binary search on C
    low, high = 0.0, 0.38085
    precision = 1e-6
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    max_iter = 60
    iter_count = 0
    
    # For each iteration, we will solve an LP feasibility problem
    while (high - low > precision) and (iter_count < max_iter) and (time() - start_time < budget_s * 0.95):
        iter_count += 1
        mid = (low + high) / 2.0
        
        # Formulate LP for this mid
        h_vars = Variable(n_points)
        
        # Bounds
        constraints = [h_vars >= 0, h_vars <= 1]
        
        # Sum constraint
        constraints.append(sum(h_vars) == n_points / 2.0)
        
        # Overlap constraints for all lags
        dx = 2.0 / n_points
        for lag in range(n_points):
            # Lag k = lag in terms of point indices
            # This computes a discrete approximation of integral
            # ∫ h(x)(1 - h(x+k)) dx
            overlap_expr = 0.0
            if lag < n_points:
                for i in range(n_points - lag):
                    overlap_expr += h_vars[i] * (1 - h_vars[i + lag])
                # Incorporate the fact that this is a convolution-like structure
                # To make it a proper overlap integral (we are treating it as sum over all lags)
                # We just enforce one constraint which must hold for all lags:
                constraints.append(overlap_expr * dx <= mid)
            else:
                # Handle wraparound properly in the circular case for correlation
                # For large lag, this computation is based on correlation
                # But with typical handling of circular convolution or full convolution
                # Simplification: Enforce the key constraint using indices and a proxy
                for i in range(lag - n_points + 1, n_points):
                    if i < 0: continue
                    overlap_expr += h_vars[i] * (1 - h_vars[i + lag - n_points])
                constraints.append(overlap_expr * dx <= mid)

        try:
            prob = Problem(Minimize(0), constraints)
            prob.solve(solver=CVXOPT, verbose=False)
            
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_sol = h_vars.value
                # check if we really satisfy
                c5_val = evaluate_erdos_solution(h_sol, dx)
                if c5_val <= mid:
                    # Accept
                    best_h = h_sol
                    best_c5 = c5_val
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except:
            low = mid

    # Post processing to make sure best_h is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    return (list(best_h), best_c5, n_points)
