import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with better resolution and/or initial solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40  # Increase resolution
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Refine and optimize with heuristic improvements
    # Grid search first to find promising area
    grid_size = 50
    search_range = np.linspace(0.37, best_c5, grid_size)
    best_c5_grid = best_c5
    
    for c_mid in search_range:
        if time() - start_time > budget_s * 0.45:
            break
            
        # Convex feasibility check using CVXPY
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2.0,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints via explicit formulation
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h_vars[:n_points-k], 1 - h_vars[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h_vars[k-n_points+1:], 1 - h_vars[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                # Simple projection to ensure constraints
                h_feasible = np.clip(h_feasible, 0, 1)
                h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5_grid:
                    best_c5_grid = c5_feasible
                    best_h = h_feasible.copy()
        except:
            continue
    
    # Local refinement using multiple strategies
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multiple restarts with different initialization schemes
    for restart in range(20):
        if time() - start_time > budget_s * 0.9:
            break
            
        # Use prior best solution with noise or perturbations
        if restart == 0:
            h_initial = best_h.copy()
        else:
            # Randomized start with appropriate Gaussian noise
            noise = np.random.normal(0, 0.025, n_points)
            h_initial = best_h + noise
            h_initial = np.clip(h_initial, 0, 1)
            # Ensure sum constraint is preserved
            h_initial = h_initial * (n_points/2) / np.sum(h_initial)
            
        try:
            # Try both SLSQP and COBYLA
            methods = ['SLSQP', 'COBYLA']
            for method in methods:
                if time() - start_time > budget_s * 0.95:
                    break
                    
                res = minimize(objective, h_initial, method=method, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        if best_c5 < 0.3807:
                            break
        except:
            continue
    
    # Final fine-grained optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Further refine with high precision
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
