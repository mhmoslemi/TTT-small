import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

        h_start = np.zeros(n_points)

        h_start[5:15] = 0.5  # First cluster
        h_start[20:30] = 0.5  # Second cluster
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search for optimal C5
    low, high = 0.3, 0.38085
    precision = 1e-6
    max_iter = 50

    # Initialize a history of solutions for warm starting
    best_feasible_h = h_start.copy()
    best_feasible_c5 = best_c5
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # Try to solve feasibility problem for this C5 value using SCP
        h_current = best_feasible_h.copy() if iteration > 0 else h_start.copy()
        
        # SCP iterations (use 5 iterations as a compromise)
        for scp_iter in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            # For CVXPY formulation
            h_vars = cp.Variable(n_points)
            
            # Objective: just feasibility
            constraints = []
            
            # Sum constraint
            constraints.append(cp.sum(h_vars) == n_points / 2)
            
            # Bound constraints
            constraints.append(h_vars >= 0)
            constraints.append(h_vars <= 1)
            
            # Overlap constraints
            # We're looking for a feasible h such that
            # for all k, \int h(x) * (1 - h(x+k)) dx <= mid
            # This becomes a set of linearized constraints
            for k in range(n_points):
                # In the full correlation, we consider shifts k = 0 to n_points-1
                # Flip h to match correlation convention:
                corr_sum = 0.0
                for i in range(n_points):
                    j = (i + k) % n_points
                    corr_sum += h_vars[i] * (1 - h_vars[j])
                # Add scaled constraint so that simplification matches problem
                constraints.append(corr_sum * dx <= mid)
            
            # Try to solve the convex problem
            try:
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.MOSEK, verbose=False)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_new = h_vars.value
                    # Ensure proper constraints
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    h_current = h_new.copy()
                else:
                    # If infeasible, proceed with current h
                    pass
            except:
                # If problem is not solvable, just continue with current h
                pass
        
        # Check feasibility of the final h from SCP
        c5_check = evaluate_erdos_solution(h_current, dx)
        
        # Note that, in our context, we're actually minimizing C5
        # So, if the current h satisfies the constraint, it's potentially better
        if c5_check <= mid:
            high = mid
            if c5_check < best_feasible_c5:
                best_feasible_h = h_current.copy()
                best_feasible_c5 = c5_check
        else:
            low = mid

    # Return the best feasible solution found
    final_h = best_feasible_h.copy()
    final_c5 = best_feasible_c5
    
    return (list(final_h), final_c5, n_points)
