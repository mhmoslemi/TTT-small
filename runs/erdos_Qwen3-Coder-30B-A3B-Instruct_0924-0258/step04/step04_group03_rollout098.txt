import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization approach
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognitive = 2.0
    social = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Add the initial solution as the first particle
    h_init = h_start.copy()
    h_init *= (n_points / 2) / np.sum(h_init)
    swarm.append(h_init)
    velocities.append(np.zeros(n_points))
    pbest.append(h_init.copy())
    pbest_values.append(evaluate_erdos_solution(h_init, dx))
    
    # Generate remaining particles
    for i in range(1, swarm_size):
        h = h_start + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(evaluate_erdos_solution(h, dx))
    
    # Find global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (pbest[i] - swarm[i]) + 
                           social * r2 * (gbest - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Apply bounds and normalization
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] *= (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            current_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = swarm[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < gbest_value:
                    gbest = swarm[i].copy()
                    gbest_value = current_value
        
        # Occasionally refine with convex optimization
        if iter_num % 50 == 0 and iter_num > 0:
            try:
                # Create a simple convex relaxation of the problem
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add a soft constraint based on current best
                # Just a partial convex relaxation for feasibility
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try several random restarts with refinement
    for restart in range(10):
        if time() - start_time > budget_s * 0.95:
            break
        
        h_restart = best_h + np.random.normal(0, 0.02, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
