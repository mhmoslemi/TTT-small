import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic algorithm parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    max_stagnation = 50
    
    # Initialize population
    population = []
    fitness = []
    
    # Add perturbed versions of initial solution
    for _ in range(10):
        h_pert = h_start + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        population.append(h_pert)
    
    # Add random valid solutions
    for _ in range(5):
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Fill up to population_size with random valid solutions
    while len(population) < population_size:
        h_rand = np.random.random(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        population.append(h_rand)
    
    # Tabu list to prevent revisiting solutions
    tabu_list = set()
    
    # Track stagnation
    stagnation_count = 0
    best_fitness = best_c5
    
    # Main genetic algorithm loop
    gen = 0
    while time() - start_time < budget_s * 0.95 and stagnation_count < max_stagnation:
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            fit = evaluate_erdos_solution(h, dx)
            fitness.append(fit)
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            stagnation_count = 0
            best_fitness = best_c5
        else:
            stagnation_count += 1
            
        # Adaptive parameters
        if stagnation_count > 30:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Selection (tournament selection)
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = random.sample(list(enumerate(population)), tournament_size)
            winner = min(tournament, key=lambda x: fitness[x[0]])
            selected.append(winner[1].copy())
        
        # Create new population
        new_population = []
        
        # Elitism: keep best solution
        new_population.append(best_h.copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Randomly select parents
            if random.random() < crossover_prob:
                parent1 = random.choice(selected)
                parent2 = random.choice(selected)
                # Single-point crossover
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                parent = random.choice(selected)
                child1 = parent.copy()
                child2 = parent.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += random.gauss(0, 0.03)
                    child1[i] = np.clip(child1[i], 0, 1)
                if random.random() < mutation_rate:
                    child2[i] += random.gauss(0, 0.03)
                    child2[i] = np.clip(child2[i], 0, 1)
            
            # Repair and normalize
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            # Avoid Tabu
            child_tuple1 = tuple(child1)
            child_tuple2 = tuple(child2)
            if child_tuple1 not in tabu_list and len(new_population) < population_size:
                new_population.append(child1)
            if child_tuple2 not in tabu_list and len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
        
        # Update Tabu list
        for h in population:
            tabu_list.add(tuple(h))
            if len(tabu_list) > tabu_limit:
                # Remove oldest
                tabu_list.pop()
    
    # Final gradient refinement of best solution
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (list(best_h), final_c5, n_points)
