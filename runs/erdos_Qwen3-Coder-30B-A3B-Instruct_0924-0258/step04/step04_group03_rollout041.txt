import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based approach with constraint projection
    max_iter = 5000
    alpha = 0.1
    alpha_decay = 0.99
    
    # Initialize gradient descent
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Track best solution found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    def project(h):
        # Ensure h sums to n_points/2 and stays in [0,1]
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2.0) / sum_h
        return h
    
    # Compute gradients for overlap with shift k
    def compute_gradient(h, k):
        # Compute gradient of overlap with shift k
        grad = np.zeros_like(h)
        if k < len(h):
            grad[:len(h)-k] += 1 - h[k:]
        else:
            grad[k-len(h)+1:] += 1 - h[:len(h)-(k-len(h)+1)]
        return grad
    
    step_count = 0
    for i in range(max_iter):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Find the maximum overlap and its index
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        k = max_idx - (len(current_h) - 1)  # Convert to shift index
        
        # Compute gradient (a little more careful)
        grad = compute_gradient(current_h, k)
        
        # Gradient update
        current_h = current_h - alpha * grad
        
        # Project back to feasible region
        current_h = project(current_h)
        
        # Evaluate
        current_c5 = evaluate_erdos_solution(current_h, dx)
        
        # Check if we improved
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Adapt learning rate
        if i % 100 == 0:
            alpha *= alpha_decay
            
        # Occasionally refine using convex program
        if i > 0 and i % 50 == 0:
            try:
                # Identify top k shifts that might be problematic
                corr = np.correlate(best_h, 1 - best_h, mode='full')
                top_shifts = np.argsort(corr)[-10:][::-1]
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add constraints for top shifts
                for k in top_shifts:
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
                
        step_count += 1

    # Final polish
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try a few restarts with slight perturbations
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
