import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search on the overlap
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iterations = 60
    
    # Gradient descent parameters
    max_iter_per_check = 1000
    learning_rate = 0.1
    backtracking_factor = 0.5
    min_improvement = 1e-6
    
    current_h = best_h.copy()
    
    # Function to check feasibility and improve
    def check_feasibility_and_improve(lambda_target, h_init):
        h = h_init.copy()
        h = h * (n_points/2) / np.sum(h)  # Ensure sum is correct
        
        for iteration in range(max_iter_per_check):
            if time() - start_time > budget_s * 0.95:
                break
                
            # Compute all auto-correlations
            corr = np.correlate(h, 1 - h, mode='full')
            # corr_indices = np.arange(-len(h)+1, len(h))
            # corr_values = corr
            
            # For maximum overlap constraint
            # auto_correlations have length 2*n_points - 1
            # We're only interested in the "internal" overlaps (shifts from -n+1 to n-1)
            # so positions n-1 to 2*n-2 in the full correlation
            full_corr = corr
            corr_vals = full_corr[n_points-1:]  # Just left half
            
            min_idx = np.argmin(corr_vals)
            min_shift = min_idx - (n_points-1)
            min_corr = corr_vals[min_idx]
            
            if min_corr >= lambda_target:
                return h, True
                
            # Compute gradient of min_corr w.r.t. h
            # When we do h(x) * (1 - h(x+k)) for some shift k,
            # the contribution to the correlation at shift k comes from products:
            # For shift k: product h[i] * (1-h[i+k]) for all valid i
            # Gradient of correlation k w.r.t. h[i] is (1-h[i+k]) * (1 if k+i >= 0 else 0)
            # Or, A_ij = 1 if i+k >= 0 (i.e., the pair contributes to corr_k)
            # and A_ij = 0 otherwise
            
            # This is a complex but doable calculation. Let's instead take a 
            # more heuristic approach: just improve the critical point
            # by a gradient step in direction that reduces the negative gap
            
            # Simple heuristic: move h towards increasing the minimum correlation
            # by slightly increasing/decreasing components near the critical positions
            grad = np.zeros_like(h)
            # Need to compute gradient of the minimum term, which is correlation at k=min_idx
            
            # Compute pairwise derivatives for the correlation contribution 
            # for shift k_min to a specific component h[i]
            # This is exact for the current linearization
            for i in range(len(h)):
                # If i+k >= 0 (i.e. valid index for correlation term)
                # Then contribution to correlation k_min is h[i] * (1 - h[i + k_min]) 
                j = i + min_shift
                if 0 <= j < len(h):
                    grad[i] += 1 - h[j]
            
            # We want to increase the correlation, so take descent step on -(min_corr)  
            # So actually we want to update in direction of grad * (1 - h) to increase it
            # This is getting tricky due to coupling. Simpler: just take a gradient step
            # on the minimum value directly.
            
            # Modify h in the direction that increases the minimum correlation
            # Let's take a projected gradient ascent step (so we're increasing min_corr)
            h_new = h + learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            # Renormalize to sum to n_points/2
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            # But some backtracking to ensure we did improve
            h_test = h_new.copy()
            h_test = h_test * (n_points/2) / np.sum(h_test)
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            corr_vals_test = corr_test[n_points-1:]
            min_corr_test = corr_vals_test[np.argmin(corr_vals_test)]
            
            # Backtracking line search
            step_size = learning_rate
            while step_size > 1e-8:
                h_trial = h + step_size * grad
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points/2) / np.sum(h_trial)
                corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
                corr_vals_trial = corr_trial[n_points-1:]
                min_corr_trial = corr_vals_trial[np.argmin(corr_vals_trial)]
                
                if min_corr_trial >= lambda_target:
                    h = h_trial
                    break
                else:
                    step_size *= backtracking_factor
            
            if min_corr_test >= lambda_target:
                h = h_test.copy()
            
            h = h_test.copy()
            
        # Final check
        corr_final = np.correlate(h, 1 - h, mode='full')
        corr_vals_final = corr_final[n_points-1:]
        min_corr_final = corr_vals_final[np.argmin(corr_vals_final)]
        still_feasible = min_corr_final >= lambda_target
        
        return h, still_feasible
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        h_new, feasible = check_feasibility_and_improve(lambda_mid, current_h)
        
        # Update best solution
        curr_c5 = evaluate_erdos_solution(h_new, dx)
        if curr_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = curr_c5
            
        if feasible:
            lambda_low = lambda_mid
            current_h = h_new
        else:
            lambda_high = lambda_mid
        current_h = h_new
    
    return (list(best_h), best_c5, n_points)
