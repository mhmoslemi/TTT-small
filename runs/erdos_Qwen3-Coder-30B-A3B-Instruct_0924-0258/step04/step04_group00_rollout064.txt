import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Strategy with Augmented Lagrangian
    population_size = 50
    mutation_strength = 0.1
    tournament_size = 5
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def augmented_lagrangian(h, multipliers, penalty_param):
        obj = evaluate_erdos_solution(h, dx)
        # Constraint violations
        sum_violation = np.abs(np.sum(h) - n_points / 2.0)
        bound_violation = np.sum(np.maximum(0, -h)) + np.sum(np.maximum(0, h - 1))
        return obj + penalty_param * (sum_violation + bound_violation)
    
    # Initialize population
    population = [project_to_feasible(h_start + np.random.normal(0, 0.05, n_points)) 
                  for _ in range(population_size)]
    multipliers = np.array([0.0] * (n_points + 2))  # Sum and bound multipliers
    penalty_param = 1.0
    
    # Main evolutionary loop
    iterations = 0
    max_iterations = int(budget_s * 0.9 / 0.01)  # Rough estimate
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [augmented_lagrangian(h, multipliers, penalty_param) for h in population]
        
        # Selection: Tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Reproduction: Crossover and Mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            child1 += np.random.normal(0, mutation_strength, n_points)
            child2 += np.random.normal(0, mutation_strength, n_points)
            
            new_population.append(project_to_feasible(child1))
            if len(new_population) < population_size:
                new_population.append(project_to_feasible(child2))
        
        population = new_population
        
        # Update best
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Update penalty and mutation strength (adaptive)
        if iteration % 10 == 0 and penalty_param > 1e-6:
            penalty_param *= 0.95
        if iteration % 20 == 0:
            # Check diversity
            stds = [np.std([h[i] for h in population]) for i in range(n_points)]
            avg_std = np.mean(stds)
            if avg_std < 0.01:
                mutation_strength *= 1.1  # Increase mutation if too homogeneous
            else:
                mutation_strength *= 0.99  # Decrease mutation if diverse
            mutation_strength = np.clip(mutation_strength, 0.001, 0.5)
    
    # Final refinement using local optimization
    best_h = project_to_feasible(best_h)
    try:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        res = minimize(objective, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons_sum, 
                      options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except Exception:
        pass
    
    # Final projection ensures constraints
    best_h = project_to_feasible(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
