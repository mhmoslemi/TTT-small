import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Initialize population
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    population = [normalize_h(h_start)]
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h)
        population.append(h)
    
    # Evaluate initial population
    fitness_scores = [evaluate_erdos_solution(h, dx) for h in population]
    best_in_gen = min(fitness_scores)
    
    gen = 0
    while time() - start_time < budget_s * 0.95 and gen < num_generations:
        # Tournament selection
        def tournament_selection(k=2):
            tournament_indices = np.random.choice(population_size, k)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            return population[winner_idx]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            # Crossover
            if random.random() < crossover_rate:
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.01)
                if random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.01)
            
            child1 = normalize_h(child1)
            child2 = normalize_h(child2)
            
            new_population.extend([child1, child2])
        
        # Replace population
        population = new_population[:population_size]
        
        # Evaluate new population
        fitness_scores = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness_scores[current_best_idx]
        
        gen += 1
    
    # Final validation
    best_h = normalize_h(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
