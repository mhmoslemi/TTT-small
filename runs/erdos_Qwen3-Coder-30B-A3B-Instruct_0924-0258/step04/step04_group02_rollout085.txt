import numpy as np
from scipy.fftpack import fft, ifft
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Initialize parameters as Fourier coefficients
    # We'll use a truncated Fourier series with sine/cosine terms
    n_coeffs = 10
    a = np.random.normal(0, 0.1, n_coeffs)
    b = np.random.normal(0, 0.1, n_coeffs)
    
    # Fourier parameterization
    x = np.linspace(0, 2, n_points, endpoint=False)
    dx = 2.0 / n_points
    
    def sigmoid(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def compute_h(a, b):
        h = a[0] * np.ones(n_points)  # constant term
        for k in range(1, n_coeffs):
            h += a[k] * np.cos(k * np.pi * x) + b[k] * np.sin(k * np.pi * x)
        h = sigmoid(h)
        # Ensure constraint: sum(h) * dx = 1
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Precompute frequencies for FFT
    freqs = np.fft.fftfreq(n_points, dx)
    
    def compute_overlap_spectrum(h):
        # Use FFT to compute overlap efficiently
        H = fft(h)
        H_conj = np.conj(H)
        # Overlap is h * (1 - h) convolution
        overlap_spectrum = H * H_conj  # This is approximately what we need
        # More accurately, we want: 1-h = 1 - h, so (1-h) has FFT = 1 - H
        # So overlap spectrum of h * (1-h) is H * (1 - H_conj) = H - H * H_conj
        # But since we want convolution, it's computed via FFT properly.
        # Let's explicitly compute overlap via FFT
        h1 = np.concatenate([h, np.zeros(n_points)])
        h2 = np.concatenate([1 - h, np.zeros(n_points)])
        F1 = fft(h1)
        F2 = fft(h2)
        conv = ifft(F1 * np.conj(F2))
        out = np.real(conv[:n_points])
        return out * dx
    
    best_h = compute_h(a, b)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient-based optimization adapted for our problem
    def objective(params):
        """Wrapper for optimization"""
        a_new = params[:n_coeffs]
        b_new = params[n_coeffs:]
        h = compute_h(a_new, b_new)
        return evaluate_erdos_solution(h, dx)
    
    def gradient(params):
        """Compute gradient of objective"""
        a_new = params[:n_coeffs]
        b_new = params[n_coeffs:]
        h = compute_h(a_new, b_new)
        
        # Approximate with finite differences for now
        eps = 1e-6
        grad = np.zeros_like(params)
        
        for i in range(len(params)):
            params_plus = params.copy()
            params_plus[i] += eps
            params_minus = params.copy()
            params_minus[i] -= eps
            
            a_plus = params_plus[:n_coeffs]
            b_plus = params_plus[n_coeffs:]
            a_minus = params_minus[:n_coeffs]
            b_minus = params_minus[n_coeffs:]
            
            h_plus = compute_h(a_plus, b_plus)
            h_minus = compute_h(a_minus, b_minus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return grad
    
    # Adam optimizer state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    learning_rate = 0.01
    
    # Adam step to be used in optimization
    t = 0
    
    max_iterations = 1000
    patience = 0
    best_obj = float('inf')
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        t += 1
        # Compute current objective and gradient
        current_params = np.concatenate([a, b])
        obj = objective(current_params)
        
        if obj < best_obj:
            best_obj = obj
            best_h = compute_h(a, b)
            patience = 0
        else:
            patience += 1
            
        if patience > 50:
            break
            
        # Use gradient descent with Adam update
        grad = gradient(current_params)
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad[:n_coeffs]
        v = beta2 * v + (1 - beta2) * (grad[:n_coeffs] ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        m = beta1 * m + (1 - beta1) * grad[n_coeffs:]
        v = beta2 * v + (1 - beta2) * (grad[n_coeffs:] ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        b -= learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
    
    return (list(best_h), best_c5, n_points)
