import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a three-cluster construction
        cluster_size = 5
        cluster_centers = [0.6, 1.0, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            for i in range(start, end):
                h_start[i] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with restarts
    max_restarts = 50
    max_iter_per_restart = 50
    
    for restart in range(max_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Random perturbation
        h_current = h_start + np.random.normal(0, 0.01, n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        # Optimize using L-BFGS-B
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(obj, h_current, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': max_iter_per_restart, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
        
        # Alternative: grid-based refinement
        if time() - start_time > budget_s * 0.95:
            break
        grid_size = 10
        for g in range(grid_size):
            h_grid = h_start + np.random.uniform(-0.05, 0.05, n_points)
            h_grid = np.clip(h_grid, 0, 1)
            h_grid = h_grid * (n_points/2) / np.sum(h_grid)
            
            try:
                res = minimize(obj, h_grid, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 30, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except:
                pass
    
    # Final refinement with local search
    if time() - start_time < budget_s * 0.95:
        h_current = best_h.copy()

            if time() - start_time > budget_s * 0.95:
                break

            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points/2) / np.sum(h_perturb)
            
            try:
                res = minimize(obj, h_perturb, method='L-BFGS-B', 
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 20, 'ftol': 1e-6})

                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined.copy()
                        best_c5 = c5_refined
            except:
                pass
            
            h_current = best_h.copy()
    
    # Ensure final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
