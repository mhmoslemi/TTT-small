import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define logistic function and its derivative
    def sigma(z):
        return 1.0 / (1.0 + np.exp(-z))
    
    def sigma_deriv(z):
        s = sigma(z)
        return s * (1.0 - s)
    
    # Optimization using gradient ascent on the worst-case overlap
    # We reparameterize h as h = sigma(z + b) and optimize over z, b
    z = np.log(h_start / (1 - h_start))  # inverse logit
    b = 0.0
    
    # Compute current h from z and b
    def compute_h(z, b):
        return sigma(z + b)
    
    # Adjust b to satisfy constraint sum(h) = n_points / 2
    def adjust_b(z):
        h = compute_h(z, 0)
        sum_h = np.sum(h)
        # We want sum_h * exp(b) = n_points / 2, so b = log(n_points / 2 / sum_h)
        return np.log((n_points / 2) / sum_h)
    
    b = adjust_b(z)
    h = compute_h(z, b)
    
    # Precompute shift indices and correlation operations
    full_corr_len = 2 * n_points - 1
    shifts = np.arange(-n_points + 1, n_points)
    
    # Helper to compute max overlap for given h
    def compute_max_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Main optimization loop
    max_iter = 5000
    learning_rate = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.99:
            break
            
        # Compute overlap for all shifts and find max  
        h = compute_h(z, b)
        corr_full = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr_full)
        max_overlap = corr_full[k_max] * dx
        
        # Early stopping if we've found a good solution
        if max_overlap < 0.3807:
            break
            
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
        
        # Compute gradient with respect to h for the worst shift
        grad_h = np.zeros_like(h)
        # For the k_max shift, gradient is -h[i] for all i
        # (Considering the derivative d/dh[i] of h[i](1-h[i+k_max]))
        # We have two cases:
        # over all valid i: d/dh[i] h[i](1-h[i+k_max]) = (1-h[i+k_max]) - h[i]
        # but we only consider the shift k_max, so use only corresponding terms
        shift_idx = k_max - n_points + 1
        if shift_idx >= 0:
            # Everything from left
            grad_h[:n_points-shift_idx] += -(1 - h[shift_idx:])
            grad_h[shift_idx:] += -(1 - h[:n_points-shift_idx]) 
            # sum over overlapping ones twice: -1 for each term, except self
        else:
            # Everything from right
            shift_idx_neg = -shift_idx
            grad_h[shift_idx_neg:] += -(1 - h[:n_points-shift_idx_neg]) 
            grad_h[:n_points-shift_idx_neg] += -(1 - h[shift_idx_neg:])
            
        # Normalize by nx
        grad_h = grad_h / (n_points - abs(shift_idx))
        
        # Compute gradient with respect to z
        sig_z_b = sigma(z + b)
        sig_deriv = sigma_deriv(z + b)
        grad_z = sig_deriv * ( grad_h - np.sum(sig_deriv * grad_h) / np.sum(sig_deriv))
        
        # Take step
        z_new = z - learning_rate * grad_z
        
        # Update b to maintain the sum constraint
        b_new = adjust_b(z_new)
        
        # Apply the change
        z = z_new
        b = b_new
        
        # Small line search for learning rate
        # More accurate line search approach:
        test_lr = 0.01
        while test_lr > 1e-8:
            z_test = z - test_lr * grad_z
            b_test = adjust_b(z_test)
            h_test = compute_h(z_test, b_test)
            c5_test = compute_max_overlap(h_test)
            if c5_test <= max_overlap:
                z = z_test
                b = b_test
                break
            test_lr /= 2
        
        # Ensure a minimum step even if not improving
        if abs(c5_test - max_overlap) < 1e-8 and test_lr == 0.01:
            z = z - 0.001 * grad_z
            b = adjust_b(z)
    
    # Final validation
    h_final = compute_h(z, b)
    final_c5 = compute_max_overlap(h_final)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final
    
    return (list(best_h), best_c5, n_points)
