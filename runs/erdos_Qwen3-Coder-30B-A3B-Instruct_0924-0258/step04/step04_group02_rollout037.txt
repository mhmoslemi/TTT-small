import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the provided construction if it exists
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Fourier parameterization approach:
    # Use Fourier series with sigmoid activation to model h(x)
    # Let h(x) = sigmoid( a0 + sum_{k=1}^n (ak*cos(k*pi*x) + bk*sin(k*pi*x)) )
    
    # Initialize Fourier coefficients
    # We'll use 15 harmonics for a reasonable balance between complexity and
    # computation speed. But we'll start with a refined version on the
    # initial construction.
    n_harmonics = 10  # Will be the number of cosine + sine terms
    # We'll use the provided init as a starting point
    
    # x positions
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Fourier basis functions
    def fourier_basis(x, n_harmonics):
        basis = np.ones((len(x), 1 + 2 * n_harmonics))
        basis[:, 0] = 1.0
        for k in range(1, n_harmonics + 1):
            basis[:, 2*k-1] = np.cos(k * np.pi * x)
            basis[:, 2*k] = np.sin(k * np.pi * x)
        return basis
    
    # To make the sigmoid behave like 0 at -infinity and 1 at +infinity we use
    # 1 / (1 + exp(-slope * z))
    # We'll use a moderate slope to ensure this behaves well
    SLOPE = 10.0
    
    # Initialize coefficients
    # Modify these based on initial_h_values for better starting point
    def h_from_coeffs(coeffs, x):
        # coeffs are [a0, a1, b1, a2, b2, ...]
        basis = fourier_basis(x, n_harmonics)
        # Expand and get h_vals
        linear_comb = np.dot(basis, coeffs.reshape(-1, 1)).flatten()
        h_vals = 1.0 / (1.0 + np.exp(-SLOPE * linear_comb))
        # Ensure h_vals are in [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals
    
    # If we have initial_h_values, try to fit starting coeffs
    initial_coeffs = np.zeros(1 + 2 * n_harmonics)
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # this is a simplified fitting approach:
        # we'll take a few iterations of gradient descent on an
        # approximation with identity coefficients, then do a refined fit
        # Manually fit a line to get direction, then momentum optimization
        bas = fourier_basis(x, n_harmonics)
        y = np.log((initial_h + 1e-10) / (1 - initial_h + 1e-10)) / SLOPE
        try:
            # Use SVD to get initial coefficients
            coeffs = np.linalg.lstsq(bas, y, rcond=None)[0]
            initial_coeffs = coeffs
        except:
            # fallback
            pass
    
    # Optimize coefficients directly
    def objective(coeffs):
        h_vals = h_from_coeffs(coeffs, x)
        # Ensure sum equals 1.0 / dx = n_points / 2
        h_vals *= (n_points / 2) / (np.sum(h_vals) + 1e-12)
        return evaluate_erdos_solution(h_vals, dx)
    
    # Let's try both approaches: gradient-free (Nelder-Mead) and very coarse
    # gradient (simple finite differences) + gradient optimization
    import scipy.optimize as opt

    current_coeffs = initial_coeffs.copy()
    
    # Two Phase Optimization

    try:

            h_vals = h_from_coeffs(coeffs, x)

            coeffs_trial = current_coeffs + np.random.normal(0, 0.2, len(current_coeffs))

            if score < best_score:
                best_score = score
                best_coeffs = coeffs_trial.copy()
        
        # Perform a final optimization
        res = minimize(objective_for_scipy, best_coeffs, method='Nelder-Mead', 
                       options={'maxiter': 200, 'disp': False})
        if res.success:
            current_coeffs = res.x
        # else continue with current
    except Exception as e:
        pass
    
    # Refinement
    h_final = h_from_coeffs(current_coeffs, x)
    h_final *= (n_points / 2) / (np.sum(h_final) + 1e-12)
    current_c5 = evaluate_erdos_solution(h_final, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_final.copy()
    
    # Try to refine with local search if we're close to time limit
    if (time() - start_time) < budget_s * 0.85:
        try:
            # Refinement around current point
            def refine_objective(h_vals):
                h_vals = np.clip(h_vals, 0, 1)
                # Ensure sum constraint
                h_vals *= (n_points / 2) / (np.sum(h_vals) + 1e-12)
                return evaluate_erdos_solution(h_vals, dx)
            
            # SLSQP local refinement
            res = minimize(refine_objective, best_h, method='SLSQP', 
                           bounds=[(0, 1) for _ in range(n_points)],
                           options={'maxiter': 100})
            if res.success:
                refined_h = res.x
                refined_c5 = evaluate_erdos_solution(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
