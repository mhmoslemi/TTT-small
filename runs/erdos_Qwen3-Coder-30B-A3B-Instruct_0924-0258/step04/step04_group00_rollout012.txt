import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolution Strategy with Augmented Lagrangian
    pop_size = 50
    mutation_strength = 0.1
    tournament_size = 5
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, mutation_strength, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    for gen in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner])
        
        # Create new population via mutation
        new_population = []
        for h in selected:
            # Mutate
            h_mut = h + np.random.normal(0, mutation_strength, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            # Normalize
            h_mut = h_mut * (n_points/2) / np.sum(h_mut)
            new_population.append(h_mut)
        
        population = new_population
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
        
        # Adapt mutation strength
        if gen % 10 == 0:
            diversity = np.std([np.sum((h - best_h)**2) for h in population])
            if diversity < 0.01:
                mutation_strength *= 1.1
            elif diversity > 0.1:
                mutation_strength *= 0.9
                
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
