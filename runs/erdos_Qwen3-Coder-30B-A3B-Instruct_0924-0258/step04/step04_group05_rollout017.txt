```python
import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    n_points = 35
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.ones(n_points) * 0.5
    
    # Population initialization
    population_size = 40
    population = []
    
    # Generate diverse initial population
    for _ in range(10):
        perturbed = initial_h + np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points/2) / np.sum(perturbed)
        population.append(perturbed)
        
    for _ in range(5):
        h_random = np.random.random(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
    
    # Initialize tabu list
    tabu_list = set()
    best_h = None
    best_c5 = float('inf')
    
    # Parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    stagnation_counter = 0
    max_stagnation = 50
    
    # Main genetic search loop
    gen = 0
    while time() - start_time < budget_s * 0.95:
        gen += 1
        if gen % 10 == 0:
            # Local refinement every 10 generations
            current_best = min(population, key=lambda x: evaluate_erdos_solution(x, dx))
            current_best_c5 = evaluate_erdos_solution(current_best, dx)
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = current_best.copy()
                
                if best_c5 < 0.3807:
                    break
        
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitnesses.append(c5_val)
        
        # Selection via tournament
        selected = []
        tournament_size = 3
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitnesses)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Crossover and mutation
        new_population = []
        
        # Elitism: keep best solution
        elite_idx = np.argmin(fitnesses)
        new_population.append(population[elite_idx].copy())
        
        while len(new_population) < population_size:
            if random.random() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                # Clone a parent randomly
                child = selected[random.randint(0, len(selected)-1)].copy()
            
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, 
                                                   size=max(1, int(n_points * mutation_rate * 0.5)), 
                                                   replace=False)
                for idx in mutation_indices:
                    child[idx] = random.random()
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Check tabu
            child_tuple = tuple(child)
            if child_tuple not in tabu_list:
                new_population.append(child)
                tabu_list.add(child_tuple)
                
                if len(tabu_list) > tabu_limit:
                    # Remove oldest entry
                    tabu_list.pop()
        
        # Update population
        population = new_population[:population_size]
        
        # Adaptive parameters
        if gen % 5 == 0:
            if stagnation_counter > max_stagnation:
                mutation_rate = min(0.5, mutation_rate + 0.1)
                crossover_prob = max(0.3, crossover_prob - 0.1)
                tabu_limit = max(10, tabu_limit // 2)
                stagnation_counter = 0
            else:
                mutation_rate = max(0.05, mutation_rate - 0.05)
                crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Check if we're making progress
        if gen % 10 == 0:
            current_best = min(population, key=lambda x: evaluate_erdos_solution(x, dx))
            current_best_c5 = evaluate_erdos_solution(current_best, dx)
            if current_best_c5 < best_c5:
                stagnation_counter = 0
            else:
                stagnation_counter += 1
    
    # Final local refinement
    if best_h is None:
        best_h = min(population, key=lambda x: evaluate_erdos_solution(x, dx))
        
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Final gradient descent on best solution
    try:
        res = minimize(objective, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = objective(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
```
