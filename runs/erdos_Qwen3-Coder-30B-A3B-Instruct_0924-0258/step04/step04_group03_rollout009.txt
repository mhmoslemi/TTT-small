import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    n_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 2.0, 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
        else:
            h = np.random.rand(n_points) * 0.5
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.random.rand(n_points) * 0.1)
        personal_best.append(h.copy())
        personal_best_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    # PSO loop
    for iteration in range(n_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        w = w_start - (w_start - w_end) * iteration / n_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Restore sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if current_val < personal_best_vals[i]:
                personal_best[i] = particles[i].copy()
                personal_best_vals[i] = current_val
                
                # Update global best
                if current_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_val
                    if global_best_val < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_val
    
    # Convex refinement
    if time() - start_time < budget_s * 0.9:
        try:
            # Try to find the tightest upper bound on the overlap
            h_ref = cp.Variable(n_points)
            constraints = [
                cp.sum(h_ref) == n_points / 2.0,
                h_ref >= 0,
                h_ref <= 1
            ]
            
            # Add constraints for the overlap terms
            max_overlap = best_c5 * 1.1  # Initial estimate
            for k in range(n_points):
                corr_term = cp.sum(cp.multiply(h_ref[:n_points-k], 1 - h_ref[k:]))
                constraints.append(corr_term <= max_overlap * dx * 1.01)
            
            # Solve to find minimal overlap
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-7)
                if prob.status == cp.OPTIMAL:
                    h_refined = h_ref.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final SLSQP polish
    if time() - start_time < budget_s * 0.95:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
