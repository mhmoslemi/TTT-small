```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Mixed approach: gradient-based with integer refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Local search with hybrid restarts
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Try randomized local search
        h_rand = best_h + np.random.normal(0, 0.02, n_points)
        h_rand = project_to_feasible(h_rand)
        
        try:
            # Try multiple local search methods
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # SLSQP
            res_slsqp = minimize(objective, h_rand, method='SLSQP', 
                                bounds=bounds, constraints=cons_sum, 
                                options={'maxiter': 50, 'ftol': 1e-7})
            
            # L-BFGS-B
            res_lbfgs = minimize(objective, h_rand, method='L-BFGS-B',
                                bounds=bounds, options={'maxiter': 50})
            
            # Pick the best local result
            candidates = [res_slsqp, res_lbfgs]
            for res in candidates:
                if res.success:
                    h_local = res.x
                    c5_local = objective(h_local)
                    if c5_local < best_c5:
                        best_h = h_local
                        best_c5 = c5_local
                        restart_count = 0
                        
        except Exception:
            pass
        
        # Occasionally do discrete coarse search
        if restart_count % 7 == 0 and time() - start_time < budget_s * 0.6:
            try:
                # Try binary search for integer-like vectors
                h_binary = (best_h > 0.5).astype(float)
                # Scale to ensure sum=1
                h_binary = h_binary * (n_points/2) / np.sum(h_binary)
                c5_binary = evaluate_erdos_solution(h_binary, dx)
                if c5_binary < best_c5:
                    best_h = h_binary
                    best_c5 = c5_binary
            except Exception:
                pass
        
        restart_count += 1
        if restart_count > max_restarts:
            break
    
    # Final thorough local optimization
    try:
        h_final = project_to_feasible(best_h)
        res = minimize(objective, h_final, method='L-BFGS-B',
                      bounds=[(0, 1) for _ in range(n_points)],
                      options={'maxiter': 100})
        if res.success:
            h_final = project_to_feasible(res.x)
            c5_final = objective(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
```
