import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create a structured initial guess 
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a two-peak structure
        peak1_center = 0.6
        peak2_center = 1.6
        peak_width = 0.3
        for i in range(n_points):
            x = i * dx
            h_start[i] = max(0, 1 - abs(x - peak1_center) / peak_width)
            if abs(x - peak2_center) < peak_width:
                h_start[i] = max(h_start[i], 1 - abs(x - peak2_center) / peak_width)
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Stochastic gradient descent with momentum
    max_iter = 3000
    step_size = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # Find the shift with maximum overlap
        max_idx = np.argmax(corr)
        # The actual shift k in [0, 2)
        k = max_idx - (n_points - 1)
        k_adjusted = k * dx
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        # For shift k, gradient at position i is (1 - h[i+k]) - h[i] * (k != 0)
        if k == 0:
            # Handle k = 0 case separately
            for i in range(n_points):
                grad[i] = -best_h[i]  # We only consider contribution of h[i] to correlation for k=0
        else:
            # For strictly shifted case
            for i in range(n_points):
                if i + k >= 0 and i + k < n_points:
                    grad[i] = -best_h[i + k] - best_h[i]
                else:
                    grad[i] = -best_h[i]  # Boundary case

        # Update with momentum
        velocity = momentum * velocity - step_size * grad
        new_h = best_h + velocity
        
        # Project to feasible region and normalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            
        # Adjust step size based on improvement
        if iteration % 100 == 0 and iteration > 0:
            if new_c5 >= best_c5:  # No improvement, reduce step size
                step_size *= 0.95
            else:
                step_size *= 1.02  # Small increase if improving
    
    # Final refinement using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        def con_sum(h):
            return np.sum(h) - n_points/2
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
