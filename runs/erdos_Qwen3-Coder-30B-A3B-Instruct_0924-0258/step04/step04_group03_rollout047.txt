import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization approach
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognative = 2.0
    social = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    best_positions = []
    best_values = []
    
    # Start with the initial solution
    h_init = h_start.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    swarm.append(h_init)
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    best_positions.append(h_init.copy())
    best_values.append(evaluate_erdos_solution(h_init, dx))
    
    # Add more swarm particles by perturbing the initial solution
    for _ in range(swarm_size - 1):
        h_pert = h_init + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        swarm.append(h_pert)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        best_positions.append(h_pert.copy())
        best_values.append(evaluate_erdos_solution(h_pert, dx))
    
    global_best_idx = np.argmin(best_values)
    global_best = best_positions[global_best_idx]
    global_best_value = best_values[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognative * r1 * (best_positions[i] - swarm[i]) + 
                           social * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Ensure bounds
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Maintain sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            current_value = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if current_value < best_values[i]:
                best_values[i] = current_value
                best_positions[i] = swarm[i].copy()
                
                # Update global best
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
        
        # Occasionally refine with convex optimization
        if iteration % 20 == 0 and iteration > 0:
            try:
                # Refine using convex problem
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add a few key constraints to regularize
                for k in range(0, min(10, 2 * n_points - 1), 3):
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    # Add a soft constraint with a small penalty
                    constraints.append(corr_sum <= global_best_value * dx * 1.1)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < global_best_value:
                            global_best = h_refined
                            global_best_value = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polish with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Use global best as starting point for final polish
            res = minimize(objective, global_best, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < global_best_value:
                    global_best = final_h
                    global_best_value = final_c5
        except Exception:
            pass
    
    return (list(global_best), global_best_value, n_points)
