import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with convex refinement
    def compute_gradient(h, k):
        # Compute gradient of overlap integral for shift k
        grad = np.zeros_like(h)
        # Convolution gradient using finite differences
        for i in range(len(h)):
            # Difference approximation of gradient
            h_plus = h.copy()
            h_plus[i] += 1e-6
            overlap_plus = np.correlate(h_plus, 1-h_plus, mode='full')[k]
            
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            overlap_minus = np.correlate(h_minus, 1-h_minus, mode='full')[k]
            
            grad[i] = (overlap_plus - overlap_minus) * 2e-6 * dx
            
        return grad
    
    def project_h(h):
        # Ensure h[i] in [0, 1] and sum = n/2
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h = h * scale_factor
        return h
    
    # Gradient descent optimization
    h_current = h_start.copy()
    c_current = best_c5
    alpha = 0.1
    max_iter = 5000
    iter_count = 0
    
    # Cache for overlap computations
    prev_k = -1
    prev_overlap = np.zeros(2 * n_points - 1)
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient for the maximum overlap
        grad = compute_gradient(h_current, k_max)
        
        # Update with gradient descent
        h_new = h_current - alpha * grad
        h_new = project_h(h_new)
        
        # Check if this is an improvement
        new_overlap = np.correlate(h_new, 1 - h_new, mode='full')[k_max] * dx
        if new_overlap < max_overlap:
            h_current = h_new
            c_current = new_overlap
        else:
            # If no improvement, decrease step size or perturb
            if alpha > 1e-6:
                alpha *= 0.99
            else:
                # Add noise to escape local minimum
                h_current = h_current + np.random.normal(0, 1e-4, n_points)
                h_current = project_h(h_current)
                c_current = evaluate_erdos_solution(h_current, dx)
        
        # Update best
        if c_current < best_c5:
            best_h = h_current.copy()
            best_c5 = c_current
            
        iter_count += 1
        
        # Occasionally perform convex refinement
        if i % 50 == 0 and i > 0:
            try:
                h_var = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_var) == n_points / 2.0,
                    h_var >= 0,
                    h_var <= 1
                ]
                
                for k in range(0, min(15, 2 * n_points - 1), 3):
                    corr_sum = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                    constraints.append(corr_sum <= best_c5 * dx * 1.2)
                    
                prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_refined = h_var.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        h_current = h_refined
            except:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-7})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
