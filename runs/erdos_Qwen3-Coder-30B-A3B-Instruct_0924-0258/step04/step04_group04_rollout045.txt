import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[10:20] = 1.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization
    n_coarse = 20
    coarse_h = np.linspace(0, 2, n_coarse)
    if n_points >= n_coarse:
        coarse_indices = np.linspace(0, n_points-1, n_coarse, dtype=int)
        h_coarse = h_start[coarse_indices]
    else:
        f = interp1d(np.linspace(0, 2, n_points), h_start, kind='linear', fill_value='extrapolate')
        h_coarse = f(coarse_h)
    
    # Normalize coarse h
    h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
    
    # Penalty method optimization for coarse grid
    max_iter_coarse = 50
    penalty = np.ones(2*n_coarse-1)  # Full correlation size
    learning_rate = 0.1
    
    for iter_coarse in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Define optimization problem
        h_var = cp.Variable(n_coarse)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_coarse/2]
        
        # Compute overlap terms
        corr_terms = []
        for k in range(2*n_coarse-1):
            shift = np.zeros(2*n_coarse-1)
            if k < n_coarse:
                shift[k:n_coarse] = h_coarse[:n_coarse-k]
            else:
                shift[k-n_coarse+1:] = h_coarse[n_coarse-(2*n_coarse-1-k):]
            term = cp.sum(cp.multiply(h_var, 1 - shift[:n_coarse]))
            corr_terms.append(term)
        
        # Create optimization problem with penalties
        objective = cp.Minimize(cp.max(corr_terms) + cp.sum(cp.multiply(penalty, cp.pos(cp.max(corr_terms) - penalty))))
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_var.value
                h_coarse = np.clip(h_opt, 0, 1)
                h_coarse = h_coarse * (n_coarse/2) / np.sum(h_coarse)
                
                # Update penalties based on current correlation
                corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
                for k in range(len(penalty)):
                    penalty[k] = np.max([penalty[k] * (corr[k] / penalty[k] if penalty[k] > 1e-8 else 1), 1e-6])
                
                # Test if this coarse solution better than best
                h_full = h_coarse
                if n_points > n_coarse:
                    f = interp1d(coarse_h, h_coarse, kind='linear', fill_value='extrapolate')
                    h_full = f(np.linspace(0, 2, n_points))
                    h_full = np.clip(h_full, 0, 1)
                    h_full = h_full * (n_points/2) / np.sum(h_full)
                
                c5_test = evaluate_erdos_solution(h_full, dx)
                if c5_test < best_c5:
                    best_h = h_full.copy()
                    best_c5 = c5_test
        except:
            break
            
    # Upsample to fine grid
    if n_points > n_coarse:
        fine_h = np.linspace(0, 2, n_points)
        f_interp = interp1d(coarse_h, h_coarse, kind='linear', fill_value='extrapolate')
        h_fine = f_interp(fine_h)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points/2) / np.sum(h_fine)
        best_h = h_fine.copy()
        best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Fine grid solvers with penalty updating
    max_iter_fine = 100
    penalty_fine = np.ones(2*n_points-1)
    learning_rate_fine = 0.2
    
    for iter_fine in range(max_iter_fine):
        if time() - start_time > budget_s * 0.95:
            break
        
        try:
            # CVX optimization
            h_var = cp.Variable(n_points)
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
            
            # Compute overlap coefficients
            corr_terms_fine = []
            for k in range(2*n_points-1):
                shift_fine = np.zeros(2*n_points-1)
                if k < n_points:
                    shift_fine[k:n_points] = best_h[:n_points-k]
                else:
                    shift_fine[k-n_points+1:] = best_h[n_points-(2*n_points-1-k):]
                term = cp.sum(cp.multiply(h_var, 1 - shift_fine[:n_points]))
                corr_terms_fine.append(term)
            
            objective_fine = cp.Minimize(cp.max(corr_terms_fine) + 
                                       cp.sum(cp.multiply(penalty_fine, cp.pos(cp.max(corr_terms_fine) - penalty_fine))))
            
            prob_fine = cp.Problem(objective_fine, constraints)
            prob_fine.solve(solver=cp.SCS, verbose=False)
            
            if prob_fine.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points/2) / np.sum(h_opt)
                
                # Update penalties
                corr_fine = np.correlate(h_opt, 1 - h_opt, mode='full')
                for k in range(len(penalty_fine)):
                    if penalty_fine[k] > 0:
                        penalty_fine[k] = np.max([penalty_fine[k] * (corr_fine[k] / penalty_fine[k]), 1e-6])
                    else:
                        penalty_fine[k] = 1e-6
                
                # Test solution
                c5_fine = evaluate_erdos_solution(h_opt, dx)
                if c5_fine < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_fine
                    
        except Exception as e:
            continue
            
    # Final SQP refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
