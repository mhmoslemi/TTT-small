import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Ensure h_start is feasible
    h_start = np.clip(h_start, 0, 1)
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by adjusting two random points
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Determine allowed delta range
        delta_max = min(0.1, T)
        
        # Generate delta for i and adjust j accordingly
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Ensure we don't exceed bounds after adjustment
        new_h = current_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Clamp to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain sum constraint
        sum_diff = np.sum(new_h) - (n_points / 2)
        new_h[0] -= sum_diff
        new_h = np.clip(new_h, 0, 1)
        
        # Compute new objective
        candidate_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on simulated annealing criteria
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cooling schedule
        T *= alpha
    
    return (list(best_h), best_c5, n_points)
