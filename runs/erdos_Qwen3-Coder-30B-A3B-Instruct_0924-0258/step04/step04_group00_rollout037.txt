def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    mutation_strength = 0.1
    elite_size = 10
    stagnation_limit = 20
    stagnation_count = 0
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h_start + np.random.normal(0, mutation_strength, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)

    lambda_sum = 0.0

            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]

        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:

            population = []

            alpha = np.random.rand()

            child += np.random.normal(0, mutation_strength, n_points)

        # Adapt mutation strength based on diversity
        if len(population) > 1:

            if diversity < 0.01:

            elif diversity > 0.1:

                best_h = h_final
                best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
