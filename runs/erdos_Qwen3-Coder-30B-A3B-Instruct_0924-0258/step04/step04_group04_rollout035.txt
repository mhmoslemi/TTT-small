import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * (1.0 / n_points)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    max_iter = 100
    learning_rate = 0.1
    max_inner_iter = 1000
    
    def compute_auto_correlation(h, dx):
        # Compute correlation using numpy (full mode)
        corr = np.correlate(h, 1 - h, mode='full')
        # Only keep the valid shifts (from -n+1 to n-1)
        mid = len(corr) // 2
        corr_valid = corr[mid+1:mid+n_points]
        return corr_valid * dx
    
    def project_h(h, target_sum):
        # Project h to [0,1] and adjust sum to target_sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    def gradient_step(h, k_min, target_lambda, dx):
        # Compute the gradient of h[k_min] wrt auto-correlation
        # For a shift k_min, correlation term involves h[i] and h[i+k_min]
        # We compute this for the gradient
        grad = np.zeros_like(h)
        n = len(h)
        # For each position i, if k_min is valid, then h[i] and h[i+k_min] contribute
        for i in range(n):
            j = i + k_min
            if 0 <= j < n:
                grad[i] += 1  # grad of correlation wrt h[i] for h[i]*h[j] term
            j = i - k_min
            if 0 <= j < n:
                grad[i] += 1  # grad of correlation wrt h[i] for h[j]*h[i] term
        grad = grad * dx
        # Project and take step
        h_new = h + learning_rate * grad
        h_new = project_h(h_new, n_points / 2)
        return h_new
    
    # Iterate binary search on lambda
    iter_count = 0
    while (lambda_high - lambda_low > tolerance) and (iter_count < max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        lambda_mid = (lambda_low + lambda_high) / 2
        # Try to find a feasible h for this lambda
        h_temp = best_h.copy()
        
        # Do gradient-based feasibility check
        inner_iter = 0
        while inner_iter < max_inner_iter:
            if time() - start_time > budget_s * 0.95:
                break
            corr_vals = compute_auto_correlation(h_temp, dx)
            k_min = np.argmin(corr_vals)
            min_val = corr_vals[k_min]
            
            if min_val >= lambda_mid:
                # Feasible, update best solution
                c5_temp = evaluate_erdos_solution(h_temp, dx)
                if c5_temp < best_c5:
                    best_h = h_temp.copy()
                    best_c5 = c5_temp
                lambda_low = lambda_mid
                break
            else:
                # Not feasible, adjust h to increase min correlation
                h_temp = gradient_step(h_temp, k_min, lambda_mid, dx)
                inner_iter += 1
        
        if inner_iter >= max_inner_iter:
            lambda_high = lambda_mid
        iter_count += 1
    
    # Final refined optimization of best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
