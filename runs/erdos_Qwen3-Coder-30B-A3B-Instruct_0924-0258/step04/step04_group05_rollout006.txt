import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def linear_resample(h_in, n_out):
    """Resample a function from input grid to output grid."""
    n_in = len(h_in)
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)

def project_and_normalize(h, target_sum):
    """Project h to [0,1] and normalize to target sum."""
    h_proj = np.clip(h, 0, 1)
    return h_proj * (target_sum / np.sum(h_proj))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Setup
    n_coarse = 50
    n_fine = 100
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine
    
    # Initialize solution
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_coarse = linear_resample(h_start, n_coarse)
    else:
        h_coarse = np.ones(n_coarse) * 0.5
        
    h_coarse = project_and_normalize(h_coarse, n_coarse / 2)
    h_fine = linear_resample(h_coarse, n_fine)
    h_fine = project_and_normalize(h_fine, n_fine / 2)
    
    best_coarse = h_coarse.copy()
    best_fine = h_fine.copy()
    best_c5_coarse = evaluate_erdos_solution(h_coarse, dx_coarse)
    best_c5_fine = evaluate_erdos_solution(h_fine, dx_fine)
    
    # Coarse grid global search
    pert_scale = 0.03
    max_coarse_steps = 100
    for step in range(max_coarse_steps):
        if time() - start_time > budget_s * 0.6:
            break
        h_pert = h_coarse + np.random.normal(0, pert_scale, n_coarse) 
        h_pert = project_and_normalize(h_pert, n_coarse / 2)
        c5 = evaluate_erdos_solution(h_pert, dx_coarse)
        if c5 < best_c5_coarse:
            best_c5_coarse = c5
            best_coarse = h_pert.copy()
        # Decay perturbation
        pert_scale *= 0.99
    
    # Transfer to fine grid and refine
    h_fine = linear_resample(best_coarse, n_fine)
    h_fine = project_and_normalize(h_fine, n_fine / 2)
    raref = 0.01
    max_fine_steps = 200
    for step in range(max_fine_steps):
        if time() - start_time > budget_s * 0.95:
            break
        h_pert = h_fine + np.random.normal(0, raref, n_fine)
        h_pert = project_and_normalize(h_pert, n_fine / 2)
        c5 = evaluate_erdos_solution(h_pert, dx_fine)
        if c5 < best_c5_fine:
            best_c5_fine = c5
            best_fine = h_pert.copy()
        raref *= 0.99
    
    # Final local refinement using SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx_fine)
        
        def constraint_sum(h):
            return np.sum(h) - n_fine / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_fine)]
        
        try:
            res = minimize(objective, best_fine, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5_fine:
                    best_fine = h_final
                    best_c5_fine = c5_final
        except:
            pass
    
    # Return best result
    final_h = best_fine
    final_c5 = best_c5_fine
    final_n = n_fine
    
    return (list(final_h), final_c5, final_n)
