import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Set up dual variables for saddle-point formulation
    n_shifts = 2 * n_points - 1
    lambda_vals = np.ones(n_shifts)
    
    def solve_convex_relaxation(h_current, lambda_vals, dx):
        h = cp.Variable(n_points)
        objective = 0
        # Constraint: sum(h) = n_points/2
        constraints = [cp.sum(h) == n_points/2, h >= 0, h <= 1]
        # Add dual term to the objective
        for k in range(n_shifts):
            # Compute the correlation term
            offset = k - (n_points - 1)
            # Shift and compute overlap
            if offset >= 0:
                shifted_ones = np.concatenate([np.ones(offset), np.zeros(n_points - offset)])
            else:
                shifted_ones = np.concatenate([np.zeros(-offset), np.ones(n_points + offset)])
            # Convolution-like term using CVXPY
            # Not standard, so we'll reconstruct it directly
            # Equivalent to: h.T @ (shifted_ones - h)
            overlap = cp.sum(cp.multiply(h, shifted_ones - h)) * dx
            if k < n_shifts:
                objective += lambda_vals[k] * overlap
        
        # Solve convex optimization
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_current
    
    # Main optimization loop
    max_iter = 1000
    dual_update_prob = 0.4 # probability to update dual
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_h = best_h.copy()
            best_c5 = current_c5
        
        # Sample overlap computation for dual update
        corr_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
        
        # Update dual variables
        if np.random.rand() < dual_update_prob:
            # Update lambda based on current overlaps
            for k in range(n_shifts):
                # Small perturbation to avoid numerical issues
                if corr_vals[k] > 0 and lambda_vals[k] > 1e-8:
                    # Update as per projected gradient
                    lambda_vals[k] = max(lambda_vals[k] * (corr_vals[k] / (lambda_vals[k] + 1e-10)), 1e-6)
                else:
                    lambda_vals[k] = max(lambda_vals[k] * 0.9999, 1e-6)
        
        # Relaxation via convex optimization
        new_h = solve_convex_relaxation(best_h, lambda_vals, dx)
        if new_h is not None:
            # Project to feasible set
            new_h = np.clip(new_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(new_h)
            new_h = new_h * scale_factor
            best_h = new_h
        
        # Optimization within a subset of directions
        if iteration % 10 == 0 and iteration > 0:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    best_h = res.x
                    best_c5 = evaluate_erdos_solution(best_h, dx)
            except:
                pass

    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
