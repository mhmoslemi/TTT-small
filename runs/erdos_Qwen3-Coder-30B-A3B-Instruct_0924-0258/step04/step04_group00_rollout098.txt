import numpy as np
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # PGD with adaptive step size
    max_iter = 1000
    alpha = 0.01
    tolerance = 1e-6
    stagnation_count = 0
    max_stagnation = 50
    
    def compute_gradient(h):
        # Compute gradient of C_5 with respect to h
        # Based on the derivative: 
        # d/dh[i] of max_k sum_j h[j] * (1 - h[j+k]) 
        # = sum_k (1 - h[i+k]) - sum_k h[i+k] if i+k < n_points
        # Simplified: compute all overlaps and their gradients
        grad = np.zeros_like(h)
        # For regions where the derivative computed more efficiently:
        # we compute numerical gradients in the full correlation
        # Simplified approach computes gradient more directly:
        # we consider the overlapping sum for all shifts k
        # This is too complex for exact gradient compute, but we can do 
        # a semi-gradient approach or direct FD
        
        # Better approach: compute finite differences
        eps = 1e-4
        grad_approx = np.zeros_like(h)
        for i in range(n_points):
            # compute small perturbation for gradients
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # enforce constraint
            h_plus = h_plus * (n_points/2) / np.sum(h_plus)
            h_minus = h_minus * (n_points/2) / np.sum(h_minus)
            
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            c5_minus = evaluate_erdos_solution(h_minus, dx)
            grad_approx[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return grad_approx
    
    # Main loop
    iter_count = 0
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1
        
        # Compute current gradient
        grad = compute_gradient(h_current)
        
        # Update h
        h_new = h_current - alpha * grad
        # Project h_new to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Adjust to meet sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Update if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
            alpha *= 1.2  # Increase step size for faster convergence
        else:
            stagnation_count += 1
            alpha *= 0.5  # Decrease step size if no improvement
            if stagnation_count > max_stagnation:
                break
            # Revert to previous h
            h_new = h_current.copy()
            alpha = max(1e-6, alpha)
        
        h_current = h_new.copy()
    
    # Final refinement: try some local search
    if time() - start_time < budget_s * 0.99:
        # Refine with local search starting from the current solution
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        # Add some random restarts with local search
        for _ in range(10):  # few restarts
            h_restart = h_current + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                # Try to improve using gradient-based method
                grad = compute_gradient(h_restart)
                h_improved = h_restart - alpha * grad
                h_improved = np.clip(h_improved, 0, 1)
                h_improved = h_improved * (n_points/2) / np.sum(h_improved)
                
                c5_improved = objective(h_improved)
                if c5_improved < best_c5:
                    best_h = h_improved.copy()
                    best_c5 = c5_improved
            except Exception:
                pass
            
            # Small random perturbation to test
            if time() - start_time > budget_s * 0.99:
                break
    
    # Final validation
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
