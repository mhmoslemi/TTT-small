import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use logistic reparameterization to ensure [0,1] bounds
    # h = sigma(z + b) where sigma is sigmoid
    # Need to ensure sum(h) = n_points/2
    
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def h_from_z(z, b):
        return sigmoid(z + b)
    
    def compute_b(z):
        # Find b such that sum(h_from_z(z,b)) = n_points/2
        h = h_from_z(z, 0.0)  # b=0 for initial guess
        target_sum = n_points / 2.0
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return 0.0
        # Binary search for b to match sum
        b_lo, b_hi = -100.0, 100.0
        for _ in range(50):
            b_mid = (b_lo + b_hi) / 2.0
            h_test = h_from_z(z, b_mid)
            test_sum = np.sum(h_test)
            if test_sum < target_sum:
                b_lo = b_mid
            else:
                b_hi = b_mid
        return (b_lo + b_hi) / 2.0
    
    # Compute initial z values from h_start and find b
    z_start = np.log(h_start / (1.0 - h_start))  # inverse sigmoid
    b_start = compute_b(z_start)
    h_start_reparam = h_from_z(z_start, b_start)
    if not np.allclose(np.sum(h_start_reparam), n_points / 2.0):
        b_start = compute_b(z_start)
    
    # Gradient descent optimization
    z = z_start.copy()
    b = b_start
    
    # Gradient of C_k wrt h[i] for shift k
    def compute_gradient_shift(h, k):
        # C_k = sum h[i] * (1 - h[i + k]) for all i in valid range
        # dC_k/dh[i] = 1 - h[i + k] - h[i] if k is not zero shift
        # But simpler:
        # If we fix k:
        # dC_k/dh[i] = (1 - h[i + k]) for i such that i+k < n_points
        # dC_k/dh[i] = 1 - h[i + k] (copied from shifted)
        # dC_k/dh[i] = -h[i] + 1 for i such that i-k >= 0, but past shift
        # Actually, we compute overlap difference to find the maximum
        # For worst k shift w.r.t C_k we use full convolution approach
        # The derivative for a fixed shift k is then:
        # dC_k/dh[i] = (1 - h[i + k]) if i+k < n_points
        #             = 0 if i+k >= n_points 
        # But better: first compute the correlations, then the derivative
        # Let's just compute C_k = sum h[i] * (1 - h[i+k]), 
        # then d(C_k)/dh[i] = 1 - h[i+k] if i + k < n (or sum over valid i)
        
        # We compute C_k = sum_{i} h[i] * (1 - h[i+k]) 
        # This is a convolution: h * (1 - h_k.shifted) 
        # So for k shift: dC_k / dh[i] = 1 - h[i+k] if i+k < n, else 0
        
        # Actually, this is simpler - it's the convolution of h with (1-h) 
        # So dC_k / dh[i] = conv(h, 1-h)[i + k] when we interpret shifts properly
        # But think this more simply:
        # C_k = sum_{i = 0 to n-1} h[i] * (1 - h[i+k])
        # dC_k / dh[j] = 1 - h[j + k]
        
        grad = np.zeros_like(h)
        # differentiate: C_k = sum h[i] * (1 - h[i+k])
        for i in range(len(h)):
            if i + k < len(h):
                grad[i] = 1.0 - h[i + k]
            else:
                grad[i] = 0.0
        return grad

    # Optimizing a general function with gradients is tricky
    # The better approach: loop to find worst k first, then instantly move toward it
    # Inequality: max_k int h(x)(1-h(x+k)) dx = max_k corr(h, 1-h)[k] * dx
    
    # Gradient descent on z parameter
    alpha = 0.5
    for iter in range(10000):
        if time() - start_time > budget_s * 0.95:
            break
        if iter % 100 == 0:
            current_h = h_from_z(z, b)
            c5 = evaluate_erdos_solution(current_h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
                # print(f"Iteration {iter}: C5 = {c5}")

        # Compute all shifts and find k that maximizes C_k
        h = h_from_z(z, b)
        corr_matrix = np.correlate(h, 1-h, mode='full')
        corr = corr_matrix * dx
        
        # Find maximum correlation index
        k_max = np.argmax(corr)
        # k_max = k_max - (len(h) - 1)  # Convert from full conv index to actual k
        # Actually, k is offset by h_length - 1 in full mode
        actual_k = k_max - (len(h) - 1)
        # So if full index is 0, actual k = -(n-1)
        # If full index is n-1, actual k = 0
        # If full index is 2*(n-1), actual k = n-1
        # k = full_index - (n-1) = k_max - (n-1)
        # So useful k = k_max - (n-1) if we want actual k shift value
        
        # But k_max allows 0 to 2*(n-1)
        # shift k = 0 to n-1 -> index k_max = n-1 to 2*(n-1)  
        # shift k = 0 to -(n-1) -> index k_max = 0 to n-1 ? No it's symmetric

        # Full convolution numpy: mode='full', result length = 2*n - 1
        # correlate index 0 to 2*n-2
        # k = index - (n-1)
        # k = k_max - (n_points - 1) = shift k = index - (n-1)
        
        actual_k = k_max - (n_points - 1)
        # From full convolution index to shift: 
        # real k = index - (n_points - 1)
        # if k_max == 0: actual_k = -(n-1)
        # if k_max == n-1: actual_k = 0  
        # if k_max == 2*(n-1): actual_k = n-1
        
        # Now compute the gradient of this
        grad_h = compute_gradient_shift(h, actual_k)
        
        # Convert gradient from h to z (chain rule)
        # Let h = sigmoid(z + b)
        # dh/dz = h * (1 - h) 
        # So: dC/dz = dC/dh * dh/dz
        # dh/dz = h * (1 - h) only if we also shift b
        # Actually: 
        # If h = sigma(z + b), then dh/dz = h * (1-h) 
        # and dh/db = h * (1-h) (different w.r.t b for the same z)
        # How quickly does h change with z and b?
        # dh/dz = sigma'(z+b) = h*(1-h) 
        # So dC/dz = grad_h * h * (1 - h) = grad_h * h * (1 - h)
        
        dh_dz = h * (1.0 - h)
        grad_z = grad_h * dh_dz
        
        # Update z with gradient step
        z_new = z - alpha * grad_z
        # Recompute b such that sum(h_new) = n_points / 2
        b_new = compute_b(z_new)
        
        # Update z, b
        z = z_new
        b = b_new

    # Final adjustment to sum  
    h_final = h_from_z(z, b)
    b_final = compute_b(z)
    h_final = h_from_z(z, b_final)
    
    # Fine-tune using scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial refinement to fine tune
    attempt_count = 0
    while attempt_count < 5 and (time() - start_time) < budget_s * 0.9:
        h_restart = h_final + np.random.normal(0, 0.02, n_points)  
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        res = minimize(objective, h_restart, method='SLSQP', 
                       bounds=bounds, constraints=cons, 
                       options={'maxiter': 50})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
        attempt_count += 1

    return (list(best_h), best_c5, n_points)
