import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 40, 50]
    for res in resolutions:
        if time() - start_time > budget_s * 0.95:
            break
            
        if res > n_points:
            # Interpolate to higher resolution
            h_new = np.interp(np.linspace(0, 2, res), np.linspace(0, 2, n_points), h_start)
        else:
            h_new = h_start[:res].copy()
        
        # Adjust for sum constraint
        h_new = h_new * (res/2) / np.sum(h_new)
        n_points = res
        dx = 2.0 / n_points
        
        # Local optimization with random restarts
        max_restarts = 20
        restart_count = 0
        
        while time() - start_time < budget_s * 0.95 and restart_count < max_restarts:
            # Random perturbation
            h_restart = h_new + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(evaluate_erdos_solution, h_restart, 
                               args=(dx,), method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 150, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        restart_count = 0
                else:
                    restart_count += 1
            except Exception:
                restart_count += 1
        
        h_start = best_h.copy()
    
    # Final convex optimization refinement if time allows
    if time() - start_time < budget_s * 0.9:
        try:
            h_opt = cp.Variable(n_points)
            constraints = [
                cp.sum(h_opt) == n_points / 2.0,
                h_opt >= 0,
                h_opt <= 1
            ]
            
            # Add overlap constraints - linear approximation
            for k in range(1, n_points):  # Consider a subset of shifts
                if k < n_points // 2:
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h_opt[k:], 1 - h_opt[:n_points-k]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_opt.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
