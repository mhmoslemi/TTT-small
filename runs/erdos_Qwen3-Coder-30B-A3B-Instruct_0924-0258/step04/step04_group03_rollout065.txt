import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with adaptive learning rate
    def compute_gradient(h, k):
        # Compute gradient of overlap integral w.r.t. h for a specific shift k
        grad = np.zeros_like(h)
        if k < len(h):
            grad[:len(h)-k] += (1 - h[k:])
        else:
            grad[k-len(h)+1:] += (1 - h[:len(h)-(k-len(h)+1)])
        return grad
    
    # Optimization parameters
    max_iter = 5000
    alpha = 0.1
    decay_rate = 0.99
    patience = 200
    patience_counter = 0
    
    current_h = h_start.copy()
    current_c5 = best_c5
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Update step
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlaps for all shifts
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        
        # Compute gradient
        grad = compute_gradient(current_h, max_k)
        
        # Update with gradient step
        new_h = current_h - alpha * grad
        
        # Project to feasible region [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Project to sum constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject the new solution
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
        
        # Adaptive learning rate
        if patience_counter > patience:
            alpha *= decay_rate
            patience_counter = 0
            # Add noise to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            current_h = current_h + noise
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
    
    # Convex relaxation refinement every 200 steps
    if time() - start_time < budget_s * 0.9:
        # Try convex relaxation
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            # Add constraints for top 15 shifts based on current solution
            corr_current = np.correlate(best_h, 1 - best_h, mode='full')
            top_k_shifts = np.argsort(corr_current)[-15:][::-1]
            
            for k in top_k_shifts:
                if k < n_points:
                    corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                else:
                    corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                constraints.append(corr_sum <= 0.6 * dx * n_points)  # Soft constraint
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            # Try several random restarts
            for restart in range(5):
                if time() - start_time > budget_s * 0.95:
                    break
                    
                # Start from a slightly perturbed best solution
                h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
                
                res = minimize(objective, h_perturbed, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 200, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
        except:
            pass

    return (list(best_h), best_c5, n_points)
