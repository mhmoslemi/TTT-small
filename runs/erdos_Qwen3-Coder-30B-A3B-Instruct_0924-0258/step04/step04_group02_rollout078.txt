import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple initialization strategies
    init_strategies = []
    
    # Strategy 1: use initial_h_values directly
    init_strategies.append(h_start.copy())
    
    # Strategy 2: random with normalization
    for _ in range(3):
        h_rand = np.random.rand(n_points) * 0.8 + 0.1
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        init_strategies.append(h_rand)
    
    # Strategy 3: some periodic construction
    h_periodic = np.ones(n_points)
    half_len = n_points // 2
    h_periodic[:half_len] = 0.8
    h_periodic[half_len:] = 0.2
    h_periodic = h_periodic * (n_points / 2) / np.sum(h_periodic)
    init_strategies.append(h_periodic)
    
    # Simple optimization with multiple restarts
    for i, h_init in enumerate(init_strategies):
        if time() - start_time > budget_s * 0.95:
            break
        
        current_h = h_init.copy()
        
        # Local optimization
        def objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h_vec):
            h = np.clip(h_vec, 0, 1)
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, current_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success:
                h_final = np.clip(res.x, 0, 1)
                h_final = h_final * (n_points/2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final.copy()
                    best_c5 = c5_final
        except:
            pass
    
    # Additional local search improvement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20:
        # Use a derivative-free method
        h = best_h.copy()
        current_c5 = best_c5
        max_iter = min(500, int(remaining_time * 10))
        
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            # Random perturbation with adaptive step size
            step_size = max(0.001, 1.0 / (iter_num + 10))
            perturbation = np.random.normal(0, step_size, n_points)
            
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            
            # Occasionally jitter global solution
            if iter_num % 10 == 0:
                h = h + np.random.normal(0, 0.005, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points/2) / np.sum(h)
    
    return (list(best_h), best_c5, n_points)
