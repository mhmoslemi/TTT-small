import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Frank-Wolfe style optimization with reparameterization
    def logistic(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def reparameterize_z_to_h(z, b):
        return logistic(z + b)
    
    def compute_b_from_z(z):
        integral = np.sum(logistic(z)) * dx
        if integral > 0:
            return np.log(1.0 / integral) - np.mean(z)
        else:
            return 0.0
    
    # Objective function for optimization
    def objective_and_gradient(z):
        b = compute_b_from_z(z)
        h = reparameterize_z_to_h(z, b)
        c5 = evaluate_erdos_solution(h, dx)
        
        # Compute gradient
        # This is a simplified gradient computation for demonstration
        # In practice, one would derive analytical gradients
        grad = np.zeros_like(z)
        eps = 1e-6
        for i in range(len(z)):
            z_plus = z.copy()
            z_plus[i] += eps
            b_plus = compute_b_from_z(z_plus)
            h_plus = reparameterize_z_to_h(z_plus, b_plus)
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5) / eps
        
        return c5, grad
    
    # Optimization with adaptive learning rate and restarts
    max_iterations = 500
    learning_rate = 0.01
    
    # Try multiple random restarts
    for restart in range(5):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Initialize z
        if restart == 0:
            z = np.log(h_start / (1 - h_start)) if np.all(h_start > 0) and np.all(h_start < 1) else np.random.normal(0, 0.1, n_points)
        else:
            z = np.random.normal(0, 0.05, n_points)
        
        for it in range(max_iterations):
            if time() - start_time > budget_s * 0.95:
                break
                
            c5, grad = objective_and_gradient(z)
            
            if c5 < best_c5:
                # Update best solution
                b = compute_b_from_z(z)
                h_best = reparameterize_z_to_h(z, b)
                best_h = h_best.copy()
                best_c5 = c5
            
            # Step update with adaptive learning
            z -= learning_rate * grad
            if np.linalg.norm(grad) < 1e-8:
                break
    
    # Final refinement with numerical optimization
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Try with best solution as start
    if time() - start_time < budget_s * 0.9:
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
