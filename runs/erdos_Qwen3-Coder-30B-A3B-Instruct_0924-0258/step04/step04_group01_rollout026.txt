import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place 3 clusters with different heights
        cluster_positions = [0.3, 1.0, 1.7]
        cluster_sizes = [3, 4, 3]
        heights = [0.5, 0.8, 0.4]
        
        for i, (pos, size, height) in enumerate(zip(cluster_positions, cluster_sizes, heights)):
            start_idx = max(0, int((pos - size/2 * dx) / dx))
            end_idx = min(n_points, int((pos + size/2 * dx) / dx))
            h_start[start_idx:end_idx] = height / size
        
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 20
    generations = 300
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h_candidate = h_start + np.random.normal(0, 0.02, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Store best individual
    for h in population:
        c5_val = evaluate_erdos_solution(h, dx)
        if c5_val < best_c5:
            best_h = h.copy()
            best_c5 = c5_val

    # Main GA loop
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5_val = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5_val)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Keep top half
        elite_size = population_size // 2
        next_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(next_population) < population_size:
            # Selection (tournament)
            tournament_size = 3
            selected_indices = []
            for _ in range(2):
                tournament = np.random.choice(elite_size, tournament_size, replace=False)
                winner = tournament[np.argmin([sorted_fitness[i] for i in tournament])]
                selected_indices.append(winner)
            
            parent1, parent2 = sorted_population[selected_indices[0]], sorted_population[selected_indices[1]]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Gaussian mutation
                    noise = np.random.normal(0, 0.02, n_points)
                    child += noise
                    # Clip values
                    child = np.clip(child, 0, 1)
                    # Normalize
                    child = child * (n_points/2) / np.sum(child)
            
            if len(next_population) < population_size:
                next_population.append(child1)
            if len(next_population) < population_size:
                next_population.append(child2)
        
        population = next_population[:]
        
        # Refinement using SQP
        for i in range(len(population)):
            h_candidate = population[i]
            
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    refined_h = res.x
                    c5_refined = obj(refined_h)
                    if c5_refined < best_c5:
                        best_h = refined_h
                        best_c5 = c5_refined
                    population[i] = refined_h
            except:
                pass
        
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]

    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
