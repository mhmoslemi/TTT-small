import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 2
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    population = [best_h.copy()]
    for _ in range(population_size - 1):
        population.append(create_individual())
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def tournament_selection(pop, k=tournament_size):
        tournament_indices = np.random.choice(len(pop), k)
        tournament_fitness = [fitness(pop[i]) for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        return pop[winner_idx]
    
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        # Repair feasibility
        child1 = np.clip(child1, 0, 1)
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = np.clip(child2, 0, 1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        indices = np.random.choice(n_points, int(mutation_rate * n_points), replace=False)
        for i in indices:
            mutated[i] += np.random.normal(0, 0.01)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    for gen in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Find best in current generation
        best_in_gen = min(fitness_scores)
        if best_in_gen < best_c5:
            best_idx = np.argmin(fitness_scores)
            best_h = population[best_idx].copy()
            best_c5 = best_in_gen
        
        # Create new population
        new_population = [population[np.argmin(fitness_scores)]]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final refinement with local optimization
    for _ in range(10):
        if time() - start_time > budget_s * 0.95:
            break
        # Random restart with good individual
        restart_idx = np.argmin([fitness(ind) for ind in population])
        h_restart = population[restart_idx].copy()
        
        # Local optimization with SLSQP
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        try:
            res = minimize(fitness, h_restart, method='SLSQP', 
                          bounds=bounds, constraints=cons_sum, 
                          options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                h_opt = res.x
                c5_opt = fitness(h_opt)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
