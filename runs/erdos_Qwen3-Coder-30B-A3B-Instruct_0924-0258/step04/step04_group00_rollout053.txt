import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution optimization
    resolutions = [35, 40, 50]
    for res in resolutions:
        if time() - start_time > budget_s * 0.9:
            break
            
        n_points = res
        dx = 2.0 / n_points
        
        # Initialize with previous solution interpolated if possible
        if len(best_h) < n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(best_h)), best_h)
        else:
            h_init = best_h[:n_points]
            h_init = h_init * (n_points/2) / np.sum(h_init)
        
        # Local optimization with multiple restarts
        for restart in range(20):
            if time() - start_time > budget_s * 0.9:
                break
                
            # Perturb solution
            h_restart = h_init + np.random.normal(0, 0.02, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            # SLSQP local optimization
            try:
                cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
                bounds = [(0, 1) for _ in range(n_points)]
                res = minimize(evaluate_erdos_solution, h_restart, 
                               args=(dx,), method='SLSQP', bounds=bounds, 
                               constraints=cons_sum, options={'maxiter': 50})
                if res.success:
                    h_refined = res.x
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
            
            # Occasionally add convex relaxation
            if restart % 4 == 0:
                try:
                    h_opt = cp.Variable(n_points)
                    constraints = [
                        cp.sum(h_opt) == n_points / 2.0,
                        h_opt >= 0,
                        h_opt <= 1
                    ]
                    
                    # Sample some overlap constraints
                    shifts = np.arange(1, min(n_points//2+1, 10))
                    for k in shifts:
                        if k < n_points//2:
                            corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                        else:
                            corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                        constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                    
                    prob = cp.Problem(cp.Minimize(0), constraints)
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_feasible = h_opt.value
                        h_feasible = np.clip(h_feasible, 0, 1)
                        h_feasible = h_feasible * (n_points/2) / np.sum(h_feasible)
                        c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                        if c5_feasible < best_c5:
                            best_h = h_feasible
                            best_c5 = c5_feasible
                except:
                    pass
    
    # Final refinement with higher precision
    if time() - start_time < budget_s * 0.95:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        try:
            cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(evaluate_erdos_solution, h_final, args=(dx,), 
                           method='SLSQP', bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = evaluate_erdos_solution(h_final, dx)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
