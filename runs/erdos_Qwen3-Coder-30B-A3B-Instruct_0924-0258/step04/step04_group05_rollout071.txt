import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Tabu search parameters
    tabu_list = []
    tabu_limit = 50
    stagnation_counter = 0
    max_stagnation = 50
    
    # Construct initial diverse population
    population = []
    population_fitness = []
    
    # Initial solutions based on perturbed initial_h_values or uniform distribution
    for i in range(10):
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
        
    # Add some random solutions
    for i in range(5):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
    
    # Ensure population size
    while len(population) < 40:
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points/2) / np.sum(h_random)
        population.append(h_random)
        
    population = population[:40]
    
    # Genetic algorithm parameters
    pop_size = len(population)
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    gen = 0
    max_generations = 5000
    
    # Main evolution loop
    while time() - start_time < budget_s * 0.85 and gen < max_generations:
        gen += 1
        # Evaluate population
        population_fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            population_fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(population_fitness)
        if population_fitness[min_fitness_idx] < best_c5:
            best_c5 = population_fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnation_counter = 0
            
            if best_c5 < 0.3807:
                break
        else:
            stagnation_counter += 1
            
        # Adaptive parameters
        if stagnation_counter > max_stagnation:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_counter = 0
        else:
            if mutation_rate > 0.05:
                mutation_rate = max(0.05, mutation_rate - 0.05)
            if crossover_prob < 0.9:
                crossover_prob = min(0.9, crossover_prob + 0.05)
        
        # Selection (tournament)
        def tournament_select():
            tournament_size = 3
            tournament = random.sample(list(enumerate(population)), tournament_size)
            winner = min(tournament, key=lambda x: population_fitness[x[0]])
            return winner[1]
            
        # Crossover and mutation
        new_population = []
        
        # Keep elite
        elite_idx = np.argsort(population_fitness)[:3]
        for idx in elite_idx:
            new_population.append(population[idx].copy())
        
        while len(new_population) < pop_size:
            if random.random() < crossover_prob:
                # Crossover
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Single-point crossover
                crossover_point = random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Mutation
                if random.random() < mutation_rate:
                    mut_point = random.randint(0, n_points-1)
                    child[mut_point] += np.random.normal(0, 0.05)
                    child[mut_point] = np.clip(child[mut_point], 0, 1)
                    
                # Normalize sum
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
            else:
                # Just mutation
                parent = tournament_select()
                child = parent.copy()
                if random.random() < mutation_rate:
                    mut_point = random.randint(0, n_points-1)
                    child[mut_point] += np.random.normal(0, 0.05)
                    child[mut_point] = np.clip(child[mut_point], 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:pop_size]
        
        # Tabu search - prevent revisiting the same solution
        if len(tabu_list) >= tabu_limit:
            tabu_list.pop(0)
            
    # Local refinement of best solution using minimize 
    remaining_time = budget_s - (time() - start_time)
    
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
