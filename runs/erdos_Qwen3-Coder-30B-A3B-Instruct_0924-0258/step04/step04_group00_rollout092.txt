import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use binary relaxation for reduced complexity
    def binary_relax(h):
        return np.round(h).clip(0, 1)
    
    def project_to_feasible(h):
        h = h.copy()
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    def binary_project(h):
        h_binary = binary_relax(h)
        return project_to_feasible(h_binary)
    
    # Main optimization loop
    max_iters = 500
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iters:
        # 1. Binary relaxation and constraint generation
        try:
            # Create binary (relaxed) version
            h_binary = binary_project(best_h)
            
            # Local improvement using constrained optimization
            def obj_local(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # SLSQP local optimization
            res = minimize(obj_local, h_binary, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                h_local = res.x
                c5_local = obj_local(h_local)
                if c5_local < best_c5:
                    best_h = h_local
                    best_c5 = c5_local
        except:
            pass
        
        # 2. Intelligent local search
        try:
            h_search = best_h.copy()
            # Random binary mutation
            mutation_indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for i in mutation_indices:
                # Toggle between 0, 0.5, 1 with equal probability
                r = np.random.rand()
                if r < 0.33:
                    h_search[i] = 0
                elif r < 0.66:
                    h_search[i] = 0.5
                else:
                    h_search[i] = 1
            
            h_search = project_to_feasible(h_search)
            c5_search = obj_local(h_search)
            if c5_search < best_c5:
                best_h = h_search
                best_c5 = c5_search
        except:
            pass
        
        # 3. Convex relaxation check (only from time to time)
        if iter_count % 5 == 0 and time() - start_time < budget_s * 0.8:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add few overlap constraints
                for k in range(1, min(10, n_points)):
                    corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except:
                pass
        
        iter_count += 1
    
    # Final projective refinement
    try:
        h_final = project_to_feasible(best_h)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
