import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # CVXPy parameters
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var)))
    
    # Constraints
    sum_constraint = cp.sum(h_var) == n_points / 2
    bounds = [h_var >= 0, h_var <= 1]
    
    # Add some performance tuning
    solver = 'ECOS'  # Should work well with CVXPY
    
    # Optimization parameters
    dual_updates = 0.8  # Probability of updating dual variables
    max_iter = 100
    
    # Initialize dual variables
    lambda_vars = np.ones(2 * n_points - 1)  # For all shifts
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(best_h, dx)
        if current_c5 < best_c5:
            best_h = best_h.copy()
            best_c5 = current_c5
            
        # Compute overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Update dual variables (adaptive)
        if np.random.random() < dual_updates:
            lambda_vars = np.maximum(lambda_vars * (overlaps / (lambda_vars + 1e-8)), 1e-6)
        
        # Solve convex problem 
        total_objective = sum(lambda_vars[i] * (np.dot(h_var, np.roll(1 - best_h, i)) * dx) 
                              for i in range(len(lambda_vars)))
        prob = cp.Problem(cp.Minimize(total_objective), bounds + [sum_constraint])
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points/2) / np.sum(new_h)
                current_c5_new = evaluate_erdos_solution(new_h, dx)
                if current_c5_new < best_c5:
                    best_h = new_h.copy()
                    best_c5 = current_c5_new
        except:
            pass
            
        # Random perturbation with decay
        perturb_magnitude = 0.01 / (1 + iteration * 0.01)
        perturbed_h = best_h + np.random.normal(0, perturb_magnitude, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points/2) / np.sum(perturbed_h)
        current_c5_pert = evaluate_erdos_solution(perturbed_h, dx)
        if current_c5_pert < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = current_c5_pert
    
    # Final refinement if time permits
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
