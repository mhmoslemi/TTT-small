import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Two stage approach: 1) Binary MIP refinement, 2) Continuous local search
    
    # Stage 1: Binary solution generation
    # Using integer programming to find extremal binary patterns
    
    # Initialize binary problem
    h_binary = (h_start > 0.5).astype(float)
    h_binary = h_binary * (n_points/2) / np.sum(h_binary)  # Rescale
    
    max_iter = 0
    binary_feasible = False
    try:
        # Scale the binary problem to match continuous case
        h_binary_init = h_binary
        h_current = h_binary_init.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        if c5_current < best_c5:
            best_h = h_current
            best_c5 = c5_current

        # Constraint generation approach: iteratively tighten overlap constraints
        c_max = 0.4  # Start with a generous upper bound
        c_tol = 1e-5  # Tighten gradually

        for i in range(50):
            if time() - start_time > budget_s * 0.6:
                break
                
            # Try to find a satisfying solution for current c_max
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Add overlap constraints for all shifts
                for k in range(1, n_points):
                    if k < n_points // 2:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= c_max * dx + 1e-8)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    sol = h_opt.value
                    c5_sol = evaluate_erdos_solution(sol, dx)
                    if c5_sol < best_c5:
                        best_h = sol
                        best_c5 = c5_sol
                    c_max = c5_sol  # Tighter upper bound
                else:
                    # Infeasible; relax some constraints
                    c_max += 0.001
            except Exception:
                break
                
    except Exception:
        pass

    # Stage 2: Continuous local search on best solution
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95:
        if restart_count > max_restarts:
            break
            
        # Random perturbation
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # SLSQP local optimization
        try:
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(evaluate_erdos_solution, h_restart, 
                           args=(dx,), method='SLSQP', bounds=bounds, 
                           constraints=cons_sum, options={'maxiter': 100, 'ftol': 1e-7})
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0  # Reset counter on improvement
            else:
                restart_count += 1
        except Exception:
            restart_count += 1

    # Final consistency correction
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass

    return (list(best_h), best_c5, n_points)
