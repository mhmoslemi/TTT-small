import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from math import inf

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a three-cluster initial guess to avoid local optima
        cluster_centers = [0.3, 1.0, 1.7]
        cluster_size = 5
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on the bound C5
    low = 0.3
    high = 0.38085
    eps = 1e-6
    
    # SCP feasibility check with linearization
    def is_feasible(h_init, C_val):
        h = h_init.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        
        # Number of iterations for SCP
        max_iter = 5
        for _ in range(max_iter):
            # Build convex constraints via linearization
            # For each k, sum of h[i] * h[i+k] >= 1 - C_val * dx
            constraints = []
            for k in range(n_points):
                # Linear approximation around current h
                lhs = 0
                for i in range(n_points):
                    j = i + k
                    if j < n_points:
                        # Linearize h[i]*h[j] ≈ h[i] * h[j]_current + h[j] * h[i]_current - h[i]_current * h[j]_current
                        lhs += h[i] * h[j] + h[j] * h[i] - h[i] * h[j]
                    else:
                        # Wrap-around case (for periodic boundary condition)
                        j = j - n_points
                        lhs += h[i] * h[j] + h[j] * h[i] - h[i] * h[j]
                # At least (1 - C_val * dx) for this shift
                constraints.append(lhs >= 1 - C_val * dx)
            
            # Just to make a valid problem, solve with cvxpy
            h_vars = cp.Variable(n_points)
            eq_con = cp.Equality(h_vars, h)
            obj = cp.Minimize(0)  # Dummy objective
            prob = cp.Problem(obj, [eq_con] + constraints)
            try:
                prob.solve(solver=cp.MOSEK, verbose=False)
                h_new = h_vars.value
                if h_new is not None:
                    h = h_new
                    h = np.clip(h, 0, 1)
                    h = h * (n_points/2) / np.sum(h)
                else:
                    h = h_init
                    break
            except:
                # Fallback to Euclidean projection
                break
        
        # Check final solution
        corr = np.correlate(h, 1-h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap <= C_val
    
    # Binary search
    while time() - start_time < budget_s * 0.95 and high - low > eps:
        mid = (low + high) / 2.0
        if is_feasible(best_h, mid):
            high = mid
            best_c5 = mid
        else:
            low = mid
    
    # Verbose optimization to improve the solution further
    def obj(h):
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}

    # Use L-BFGS-B for improvement
    res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons_sum, options={'maxiter': 100, 'ftol': 1e-6})

    if res.success:
        final_h = res.x
        final_c5 = obj(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    # Final check to re-support constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
