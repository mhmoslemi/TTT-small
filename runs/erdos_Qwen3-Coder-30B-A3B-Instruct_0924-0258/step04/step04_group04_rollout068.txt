import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Initialize dual variables
    dual_vars = np.ones(2 * n_points - 1)  # For all shifts
    dual_vars *= 1.0
    
    # Main optimization loop
    max_iter = 500
    dual_update_prob = 0.8
    dual_step_size = 0.1
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current overlaps
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * dx
        
        # Update dual variables based on current overlaps
        if np.random.rand() < dual_update_prob:
            # Update only a subset of dual variables (batching)
            update_indices = np.random.choice(len(dual_vars), 
                                            size=max(1, int(0.1 * len(dual_vars))), 
                                            replace=False)
            for idx in update_indices:
                if overlaps[idx] > dual_vars[idx]:
                    dual_vars[idx] *= (1 + dual_step_size)
                else:
                    dual_vars[idx] *= (1 - dual_step_size)
                dual_vars[idx] = max(dual_vars[idx], 1e-6)
        
        # Form and solve convex problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(dual_vars, 
                                                  [cp.quad_form(h_var, np.eye(n_points)) - 
                                                   cp.dot(h_var, 1 - best_h)])))
        constraints = [cp.sum(h_var) == n_points / 2,
                       h_var >= 0,
                       h_var <= 1]
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points/2) / np.sum(h_new)
                    h_new = np.maximum(0, np.minimum(1, h_new))
                    
                    curr_c5 = evaluate_erdos_solution(h_new, dx)
                    if curr_c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = curr_c5
        except:
            pass
        
        # Perturb best solution occasionally to escape local optima
        if iteration % 20 == 0 and iteration > 0:
            small_perturb = np.random.normal(0, 0.005, n_points)
            best_h = np.clip(best_h + small_perturb, 0, 1)
            best_h = best_h * (n_points/2) / np.sum(best_h)
    
    # Final refinement with local optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
