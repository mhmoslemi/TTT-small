import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Use initial construction if available

        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_size = 5
        cluster_centers = [0.6, 1.6]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Steepest descent approach
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    patience = 100
    
    def compute_gradient(h, k_shift):
        # Compute gradient of overlap w.r.t h for given shift
        n = len(h)
        grad = np.zeros(n)
        # The derivative of sum(h(x)(1-h(x+k))) wrt h[i]
        # is (1-h[i+k]) - h[i] (when k != 0)
        # For k=0, it's just -1
        if k_shift == 0:
            grad = -np.ones(n)
        else:
            # Convolution with 1-h
            ones_minus_h = 1 - h
            # Compute cross-correlation between h and 1-h with shift k_shift
            # (this is a manual compute for efficiency without padding)
            for i in range(n):
                # get index in 1-h array
                j = i + k_shift
                if 0 <= j < n:
                    grad[i] = -ones_minus_h[j]
                else:
                    grad[i] = 0
        return grad

    def backtracking_line_search(h, grad, k_shift, step_size, max_iter=10):
        c1 = 0.5  # Armijo condition constant
        alpha = step_size
        old_overlap = evaluate_erdos_solution(h, dx)
        for _ in range(max_iter):
            # Compute new h
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Normalize to sum constraint
            h_new = h_new * (n_points/2) / np.sum(h_new)
            new_overlap = evaluate_erdos_solution(h_new, dx)
            # Armijo condition
            if new_overlap <= old_overlap + c1 * alpha * np.dot(grad, h-h_new):
                return alpha, h_new
            alpha *= 0.5
        # return best step size with at least some improvement
        return alpha, h_new

    patience_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break

        # Compute all overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')

        k_shift = max_idx - (len(best_h) - 1)
        max_overlap = corr[max_idx] * dx
        
        if max_overlap < best_c5:

            best_h = best_h.copy()

        # Compute gradient
        grad = compute_gradient(best_h, k_shift)

        step, new_h = backtracking_line_search(best_h, grad, k_shift, step_size)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Update
        best_h = new_h
        new_c5 = evaluate_erdos_solution(best_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            patience_count = 0
        else:
            patience_count += 1

                break  # Early termination if no improvement

    remaining_time = budget_s - (time() - start_time)

            return evaluate_erdos_solution(h, dx)

            return np.sum(h) - n_points/2

                          bounds=bounds, constraints=cons,

                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
