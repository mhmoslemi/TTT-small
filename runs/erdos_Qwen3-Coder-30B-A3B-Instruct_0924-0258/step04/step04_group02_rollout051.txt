import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Try multiple initializations to avoid local optima
    initializations = []
    
    # Start with the provided initial solution
    initializations.append(h_start.copy())
    
    # Add random perturbations 
    for _ in range(3):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        initializations.append(h_pert)
    
    # Add some carefully constructed alternatives
    # Piecewise constant approximations
    for _ in range(2):
        h_alt = np.ones(n_points)
        h_alt[:n_points//3] = 0.2
        h_alt[n_points//3:2*n_points//3] = 0.8
        h_alt = h_alt * (n_points / 2) / np.sum(h_alt)
        initializations.append(h_alt)
    
    # Try to optimize with different solvers
    solvers = ['SLSQP', 'L-BFGS-B']  # Reduced to avoid excessive running time
    
    for h_init in initializations:
        if time() - start_time > budget_s * 0.95:
            break
            
        current_h = h_init.copy()
        
        # Soft constraints with penalty
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def penalty(h):
            # Make sure integral is close to 1
            integral = np.sum(h) * dx
            return 100 * (integral - 1)**2
        
        def combined_objective(h):
            return objective(h) + penalty(h)
        
        # Apply bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}
        
        # Use multiple solver instances with different restarts
        for solver in solvers:
            if time() - start_time > budget_s * 0.95:
                break
                
            for _ in range(2):  # Restart trials
                try:
                    h_rand = current_h + np.random.normal(0, 0.01, n_points)
                    h_rand = np.clip(h_rand, 0, 1)
                    h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                    
                    res = minimize(combined_objective, h_rand, 
                                   method=solver, 
                                   bounds=bounds, 
                                   constraints=cons_sum,
                                   options={'maxiter': 500, 'ftol': 1e-8})
                    
                    if res.success:
                        h_final = res.x
                        c5_final = objective(h_final)
                        if c5_final < best_c5:
                            best_h = h_final.copy()
                            best_c5 = c5_final
                except:
                    pass
            
            # Do one final fine-tune with direct optimization
            try:
                h_fine = best_h + np.random.normal(0, 0.001, n_points)
                h_fine = np.clip(h_fine, 0, 1)
                h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
                
                res = minimize(objective, h_fine,
                               method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 300})
                
                if res.success:
                    h_final = res.x
                    c5_final = objective(h_final)
                    if c5_final < best_c5:
                        best_h = h_final.copy()
                        best_c5 = c5_final
                        
            except:
                pass
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
