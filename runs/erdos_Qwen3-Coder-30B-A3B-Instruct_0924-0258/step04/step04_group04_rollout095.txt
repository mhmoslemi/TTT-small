```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
        h_start[5:15] = 1.0
        # Normalize sum to exactly n_points / 2 
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Coarse grid optimization
    max_iter_coarse = 100
    penalties = np.ones(n_points * 2 - 1)  # full correlation length
    lr = 0.1
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.8:
            break
            
        # Compute overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        
        # Update penalties
        for i in range(len(penalties)):
            penalties[i] = max(penalties[i] * (corr[i] / penalties[i] if penalties[i] != 0 else 1), 1e-6)
        
        # Convex optimization of penalties
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.maximum(corr + penalties, 0)) * dx)
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = project_h(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
        except:
            pass

    # Upsample to 40 points
    if n_points < 40:
        # Interpolate
        x_old = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 40)
        f = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_upsampled = f(x_new)
        h_upsampled = project_h(h_upsampled)
        n_points = 40
        best_h = h_upsampled
        
        # Refine with same approach
        max_iter_fine = 100
        penalties = np.ones(n_points * 2 - 1)
        lr = 0.2
        
        for iteration in range(max_iter_fine):
            if time() - start_time > budget_s * 0.95:
                break
                
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            
            # Update penalties with aggressive learning
            for i in range(len(penalties)):
                # A bit of smoothing
                if i > 0:
                    penalties[i] = max(penalties[i] * (corr[i] / penalties[i] if penalties[i] != 0 else 1), 1e-6)
                else:
                    penalties[i] = max(penalties[i] * 0.9, 1e-6)
            
            # Convex optimization
            h_var = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(cp.maximum(corr + penalties, 0)) * dx)
            constraints = [
                cp.sum(h_var) == n_points/2,
                h_var >= 0,
                h_var <= 1
            ]
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_new = h_var.value
                    h_new = project_h(h_new)
                    c5_new = evaluate_erdos_solution(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
            except:
                pass

    # Final refinement with SQP
    if time() - start_time < budget_s * 0.95:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
```
