import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create sinusoidal-like initial guess
        x = np.linspace(0, 2, n_points)
        h_start = 0.5 + 0.3 * np.sin(3*np.pi*x)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # CVXPy optimization parameters
    max_iter = 50
    dual_updates = 0.2  # fraction of dual variables to update at each step
    dual_step = 0.1
    warm_start = True
    
    # Initial dual variables
    n_shifts = 2 * n_points - 1
    lambda_k = np.ones(n_shifts)
    
    def solve_convex_subproblem(h_current, lambda_k_vec):
        h = cp.Variable(n_points)
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points/2]
        
        # Convex approximation
        obj_expr = 0
        for k in range(n_shifts):
            # Construct shift matrix manually for efficiency
            shift = k - n_points + 1
            if shift >= 0:
                shifted_h = cp.hstack([h[shift:], cp.repeat(h[0], shift)])
            else:
                shifted_h = cp.hstack([cp.repeat(h[-1], -shift), h[:shift]])
            obj_expr += lambda_k_vec[k] * cp.sum(cp.multiply(h, 1 - shifted_h))
        
        objective = cp.Minimize(obj_expr)
        prob = cp.Problem(objective, constraints)
        try:
            if warm_start and 'h_prev' in globals():
                prob.solve(solver=cp.SCS, warm_start=True, verbose=False)
            else:
                prob.solve(solver=cp.SCS, verbose=False)
        except:
            return h_current.copy()
        
        return np.array(h.value).flatten()
    
    # Main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Compute overlaps
        overlaps = np.correlate(best_h, 1 - best_h, 'full') * dx
        
        # Update dual variables
        idxs = np.argsort(overlaps)[::-1][:max(1, int(dual_updates * n_shifts))]
        for idx in idxs:
            lambda_k[idx] *= np.exp(dual_step * (overlaps[idx] - lambda_k[idx]))
            lambda_k[idx] = max(lambda_k[idx], 1e-6)
        
        # Solve convex subproblem
        h_new = solve_convex_subproblem(best_h, lambda_k)
        
        # Project and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate and update best
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        # Add some randomness to prevent local trapping
        if iteration % 5 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h_noisy = h_new + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (n_points/2) / np.sum(h_noisy)
            
            c5_noisy = evaluate_erdos_solution(h_noisy, dx)
            if c5_noisy < best_c5:
                best_h = h_noisy.copy()
                best_c5 = c5_noisy
    
    return (list(best_h), best_c5, n_points)
