import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with Augmented Lagrangian
    max_iter = 1000
    alpha = 0.01
    lambda_ = 0.0
    mu = 1.0
    stagnation_count = 0
    stagnation_limit = 50
    
    # Initialize step size
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute current C5 and max overlap shift
        h_current = best_h.copy()
        c5_current = evaluate_erdos_solution(h_current, dx)
        
        # Compute gradient of objective w.r.t. h
        # Gradient for overlapping terms
        gradient = np.zeros(n_points)
        # Compute the full convolution and corresponding gradients
        # For this we compute element-wise derivatives for all possible shifts
        for k in range(1, n_points):
            if k <= n_points - 1:
                # s = sum(h[i] * (1 - h[i+k]))
                # d(s)/dh[i] = (1 - h[i+k]) if i < n_points - k else 0
                # d(s)/dh[i+k] = -h[i] if i < n_points - k else 0
                for i in range(n_points - k):
                    gradient[i] += (1 - h_current[i + k])
                    if i + k < n_points:
                        gradient[i + k] -= h_current[i]
            else:
                # Reverse case
                for i in range(n_points - (k - n_points + 1)):
                    if i + k - n_points + 1 < n_points:
                        gradient[i] += (1 - h_current[i + k - n_points + 1])
                    if i + k - n_points + 1 < n_points:
                        gradient[i + k - n_points + 1] -= h_current[i]
        
        # Augmented Lagrangian gradient w.r.t. sum constraint term
        # ∇_h (λ * (sum(h) - n_points/2) + μ * (sum(h) - n_points/2)^2) = λ + 2*μ*(sum(h) - n_points/2)
        lagrangian_grad = lambda_ + 2 * mu * (np.sum(h_current) - n_points / 2)
        gradient += lagrangian_grad * np.ones(n_points)
        
        # Apply gradient descent
        h_new = h_current - step_size * gradient
        
        # Project onto [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < c5_current:
            # Accept update
            best_h = h_new
            best_c5 = c5_new
            step_size *= 1.2
            stagnation_count = 0
        else:
            # Reject update
            step_size *= 0.5
            stagnation_count += 1
            if stagnation_count > stagnation_limit:
                break
                
        # Update Lagrange multipliers
        lambda_ += mu * (np.sum(h_new) - n_points / 2)
    
    # Final refinement
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), final_c5, n_points)
