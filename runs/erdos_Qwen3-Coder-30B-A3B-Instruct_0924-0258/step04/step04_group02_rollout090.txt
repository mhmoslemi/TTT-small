import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Optimization using L-BFGS-B with custom constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Ensure that h values stay within [0,1] and sum to n_points/2
    bounds = [(0, 1) for _ in range(n_points)]
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try multiple random restarts
    num_restarts = 5
    for restart in range(num_restarts):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb initial solution
        h_restart = h_start.copy()
        # Mix in some random noise
        noise = np.random.normal(0, 0.05, n_points)
        h_restart = h_restart + noise
        h_restart = np.clip(h_restart, 0, 1)
        # Rescale to make sum correct
        h_restart = h_restart * (n_points / 2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='L-BFGS-B', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_candidate = res.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        except:
            pass
    
    # Try a few more local refinements with different methods
    methods = ['SLSQP', 'TNC']
    for method in methods:
        if time() - start_time > budget_s * 0.95:
            break
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method=method, 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final refinement using BFGS with higher accuracy
    if time() - start_time < budget_s * 0.95:
        h_final = best_h + np.random.normal(0, 0.005, n_points)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        
        try:
            res = minimize(objective, h_final, method='BFGS', 
                           options={'maxiter': 150, 'gtol': 1e-10})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
