import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization with simulated annealing
    max_iter_coarse = 100
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    
    h_current = best_h.copy()
    
    for iteration in range(max_iter_coarse):
        if time() - start_time > budget_s * 0.95:
            break
            
        h_pert = h_current + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        old_c5 = evaluate_erdos_solution(h_current, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            h_current = h_pert
            if new_c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = new_c5
                
        temp *= cool_rate
    
    # Upsample to 40 points
    if n_points < 40:
        n_points = 40
        h_upsampled = np.interp(np.linspace(0, 2, n_points), 
                               np.linspace(0, 2, len(h_current)), 
                               h_current)
        h_current = h_upsampled.copy()
        h_current = h_current * (n_points/2) / np.sum(h_current)
        
        dx = 2.0 / n_points
        c5 = evaluate_erdos_solution(h_current, dx)
        if c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = c5
    
    # Fine grid optimization with SLSQP
    max_iter_fine = 100
    for iteration in range(max_iter_fine):
        if time() - start_time > budget_s * 0.95:
            break
            
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_current, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 10, 'ftol': 1e-6})
            if res.success:
                h_current = res.x
                c5 = evaluate_erdos_solution(h_current, dx)
                if c5 < best_c5:
                    best_h = h_current.copy()
                    best_c5 = c5
        except:
            pass
    
    # Final refinement with gradient ascent
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Simple gradient descent with clipping
        for _ in range(50):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_old = best_h.copy()
            c5_old = best_c5
            
            # Compute gradient approximation
            eps = 1e-4
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_pert = h_old.copy()
                h_pert[i] += eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                c5_plus = evaluate_erdos_solution(h_pert, dx)
                
                h_pert = h_old.copy()
                h_pert[i] -= eps
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points/2) / np.sum(h_pert)
                c5_minus = evaluate_erdos_solution(h_pert, dx)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update step
            step_size = 1e-3
            h_new = h_old - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_old:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    return (list(best_h), best_c5, n_points)
