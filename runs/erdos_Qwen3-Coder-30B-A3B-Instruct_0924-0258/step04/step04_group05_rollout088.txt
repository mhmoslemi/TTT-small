import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the provided initial construction
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Initial local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(objective, h_start, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-7})
        if res.success:
            h_refined = res.x
            c5_refined = objective(h_refined)
            if c5_refined < best_c5:
                best_h = h_refined
                best_c5 = c5_refined
    except:
        pass
    
    # Multi-perturbation search with adaptive strategy
    max_iterations = 500
    perturbation_scale = 0.02
    perturbation_decay = 0.95
    retry_count = 0
    max_retries = 3
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the current best solution
        h_perturbed = best_h + np.random.normal(0, perturbation_scale, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Normalize to maintain sum constraint
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Local refinement of perturbed solution
        try:
            res = minimize(objective, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 100, 'ftol': 1e-7})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                
                if c5_refined < best_c5:
                    # Success: update best solution
                    best_h = h_refined
                    best_c5 = c5_refined
                    retry_count = 0  # Reset retries on success
                    # Early stopping if target reached
                    if best_c5 < 0.3807:
                        break
                else:
                    # No improvement: increase retries and perturbation
                    retry_count += 1
                    if retry_count >= max_retries:
                        perturbation_scale *= 1.1  # Increase perturbation
                        retry_count = 0  # Reset retry count
                        perturbation_scale = min(perturbation_scale, 0.2)  # Cap perturbation
            else:
                # SLSQP failed, increase perturbation
                retry_count += 1
                if retry_count >= max_retries:
                    perturbation_scale *= 1.1
                    retry_count = 0
                    perturbation_scale = min(perturbation_scale, 0.2)
                    
        except Exception as e:
            retry_count += 1
            if retry_count >= max_retries:
                perturbation_scale *= 1.1
                retry_count = 0
                perturbation_scale = min(perturbation_scale, 0.2)
        
        # Decay perturbation scale
        perturbation_scale *= perturbation_decay
        perturbation_scale = max(perturbation_scale, 0.001)  # Minimum scale
    
    # Final refinement with tight tolerance
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
