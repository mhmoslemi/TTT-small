import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_current = h_start.copy()
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = evaluate_erdos_solution(h_current, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    T = T_initial
    max_iterations = int(1e5)
    
    # Perturbation range
    delta_max = 0.1
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two points
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Ensure we don't violate bounds
        h_new = h_current.copy()
        if h_new[i] + delta > 1:
            delta = 1 - h_new[i]
        elif h_new[i] + delta < 0:
            delta = -h_new[i]
            
        h_new[i] += delta
        h_new[j] -= delta
        
        # Keep sum invariant (adjust one more point)
        if i != j:
            # Adjust the third point to maintain sum
            k = np.random.choice([idx for idx in range(n_points) if idx != i and idx != j])
            h_new[k] += delta
        
        # Ensure all values remain in [0, 1]
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate candidate
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h_current = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Cool down
        T *= alpha
        
        # Occasionally adjust delta
        if iteration % 1000 == 0:
            delta_max *= 0.99
            
    return (list(best_h), best_c5, n_points)
