import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:

        # Simple initial guess - uniform distribution

    dx = 2.0 / n_points
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5

    def compute_auto_correlation(h):
        # Compute all correlations with 1-h
        corr = np.correlate(h, 1 - h, mode='full')
        # Only keep the relevant parts (shifts from -1 to 1)
        # Assuming we want correlation for shifts from 0 to 2
        # Since h is defined on [0,2] and correlate with 1-h shifted
        # The full correlation will have length 2*n_points - 1
        mid = len(corr) // 2
        return corr, mid
    
    def gradient_step_h(h, shift_idx, lambda_target, bias_term=0):
        # For the system: int h(x) * (1-h(x+shift_idx)) dx >= lambda_target
        # We need to increase the current correlation at shift_idx
        # This means increasing h where h(x)*h(x+shift_idx) contributes to it
        # The gradient w.r.t h at point k is:
        # For corr shift_idx: it involves h[k] * (1-h[k+shift_idx]) 
        # The gradient wrt h[k] is (1 - h[k+shift_idx])
        # The gradient wrt h[k+shift_idx] is -h[k]

        # corr[shift_idx] = sum_{x} h[x] * (1-h[x-shift_idx]) for all x <- need to be corrected

        # If shift=0, corr index is n-1.  
        # If shift=k,_corr index is n-1+k  
        # What's the actual shift indicated by the index j?
        # corr index = n-1 + shift => shift = index - (n-1)  
        # So for shift_idx = i - (n-1) unit shift
        # So in the actual correlation computation:
        # Input indices: i, j, where corr[i] = sum_x h[x] * h[x - (i-(n-1))]

        # correlation arrays: corr = np.correlate(h, 1-h, mode='full')

        # corr is indexed from 0 to len(h)-1+len(h)-1 = 2*len(h)-2  
        # Each element corr[k] corresponds to convolution:
        # sum_{x=0}^{min(len(h), len(h)-k)} h[x] * (1-h[x-k])

        shift = shift_idx - (len(h) - 1)
        # Correct indices to reflect convolution now...
        grad = np.zeros_like(h)
        # Contributing pairs
        for x in range(len(h)):
            y = x - shift
            if 0 <= y < len(h):
                grad[x] += (1 - h[y])  # d/dh[x] (h[x] * (1 - h[x-shift]))
                if x != y:
                    grad[y] += -h[x]   # d/dh[y] (h[x] * (1 - h[x-shift]))
        return grad

    def get_gradient_shift_zero(h, prev_grad=None, shift=0):
        # Gradient of Cshift = sum_{x} h[x]*(1 - h[x+shift]) = sum h - sum h(x) h(x+shift) 
        # d/dh_i Cshift = 1 - h[i+shift] - h[i-shift]  
        # But this creates problems with boundary
        # Better:  delta_C = sum_x h[x] (1 - h[x+shift])  
        # Gradient w.r.t h_i is:
        # d_i = 1 if x = i and x+shift = i or x = i - shift, so then 1 
        # We have two cases: 
        # 1) i corresponds to x such that i = x  -> :, contribution from h[x] * (1 - h[x-shift])  
        # So  d/dh[i] partial term in sum = (1 - h[i-shift])  
        # 2) i corresponds to x-shift = i, so x = i+shift -> contribution: h[i+shift] *(1-h[i])
        # So again  d/dh[i] partial term = - h[i+shift]
        # But this has boundary problem  
        
        # More carefully, we write out convolution:
        # corr[i] = sum_{x = 0}^{L-1} h[x] * (1 - h[x-i]) 
        # The gradient wrt h[k] = d_hk corr_index, how many terms have h[k]?
        # When index x = k, add (1 - h[k-i]) to component
        # When x-i = k, so i=x-k, add -h[k] to component
        # But x must be in [0, L-1], would put k in [0, L-1], similarly for k-i  
        # so x = k  
        # gradient wrt h[k] you have term: (1 - h[k-i]) such that  
        # x = k, and k-i >= 0, i in [0, L-1], so k in [i, L-1], k-i bounded in [0, L-1]  
        # y = x-i, x = k, so y = k-i, y >= 0, y < L  
        # x = k, y = k-i  
        # let's simplify  
        # If i-th coefficent to be derived:
        # Elements in sum for corr idx where i is considered are:
        # For each x, h[x] * (1 - h[x - (i - (L-1))]))
        # But that's likely incorrect...
        # Let's reason from index shift_idx = shift:
        # Take corr that maps to shift at k = 0,1,2,..., 2L-2  
        # corr[k] = sum_{x} h[x] * 1 - h[x - shift_index]  
        # i is the position in h we want derivative of corr with  
        # In [0, L-1], but 1-h has period L so no issue  
        # Each term in summation is h[x] * 1 - h[x-shift]  
        # So with respect to h[i]:

    # More practically, use finite difference to approximate  
    def get_gradient_shift(h, shift_index, eps=1e-6):
        # approx grad of corr[index] = sum_h h[x]*h[x-shift]  = h[x] * (1-h[x-shift])
        # the non-convolution part  
        # If we did want approx
        # Consider as objective fn: C = sum_h h[x] (1 - h[x-shift])
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Current value
            h_orig = h[i]
            h[i] = h_orig + eps
            corr_new = np.correlate(h, 1-h, mode='full')
            h[i] = h_orig - eps
            corr_neg = np.correlate(h, 1-h, mode='full') 
            h[i] = h_orig  
            # Approximate derivative
            diff = (corr_new[shift_index] - corr_neg[shift_index]) / (2*eps)
            grad[i] = diff
        return grad
        
    def check_feasibility(h, lambda_target):
        corr, mid = compute_auto_correlation(h)
        # Check that all corr[i] >= lambda_target
        if np.any(corr < lambda_target):
            return False, np.argmin(corr)
        return True, None  

    def project_to_feasible(h):
        # Normalize sum to n_points/2
        h = h * (n_points/2) / np.sum(h)
        # clip to [0,1]
        h = np.clip(h, 0, 1)
        return h

    while time() - start_time < budget_s * 0.95:

        found_feasible = False

                    best_h = current_h_trial.copy()
                    best_c5 = trial_c5

            worst_shift_idx = np.argmin(corr)
            
            # Compute gradient that decreases (np.sign of) the at-worst shift

            # For speed, use approx gradient
            if worst_shift_idx < len(corr) - 10 and worst_shift_idx > 10:
                grad = get_gradient_shift(current_h_trial, worst_shift_idx)

                # Depending on edges, better to optimize specifically  
                grad = np.zeros_like(current_h_trial) 
                # flavor the gradient to push that direction

                grad_x = np.zeros_like(current_h_trial)

                for i in range(len(current_h_trial)):
                    # Get gradients of terms that contribute to corr wrt x is i

                        grad_x[i] += -current_h_trial[i-shift_val]  # d/dh[i] of contrast term (the h[i-shift_val])
                # Use strategy to retain shift val
                grad = grad_x
                
            # gradient step to increase this value
            step_size = learning_rate
            while step_size > 1e-10:
                temp_h = current_h_trial - step_size * grad
                temp_h = project_to_feasible(temp_h)
                temp_corr, _ = compute_auto_correlation(temp_h)
                if np.min(temp_corr) >= lambda_mid:
                    current_h_trial = temp_h
                    break
                step_size *= backtracking_factor
                
            # If no improvement, let's still do backtracking
            if step_size <= 1e-10:
                # Simply try to improve by small increments or psychologically(naive!)
                current_h_trial = current_h_trial + np.random.normal(0, 1e-4, len(current_h_trial))
                current_h_trial = project_to_feasible(current_h_trial)
                
        if found_feasible:
            lambda_low = lambda_mid
        else:
            lambda_high = lambda_mid
            
    # Final optimization step
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 50:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}

                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)

                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
