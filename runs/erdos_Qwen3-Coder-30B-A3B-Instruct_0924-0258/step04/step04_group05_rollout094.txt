import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize from parent solution or create a new one
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Initialize population with diverse solutions
    population_size = 40
    population = []
    fitness = []
    
    # Add the initial solution and its variations
    for i in range(10):
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        population.append(h_perturbed)
        fitness.append(evaluate_erdos_solution(h_perturbed, dx))
    
    # Add more random valid solutions
    for i in range(30):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
        fitness.append(evaluate_erdos_solution(h_rand, dx))
    
    # Tabu list for visited solutions
    tabu_list = []
    tabu_limit = 50
    
    # Adaptive parameters
    mutation_rate = 0.15
    crossover_prob = 0.7
    stagnation_count = 0
    best_fitness = min(fitness)
    best_solution = population[fitness.index(best_fitness)].copy()
    
    # Genetic algorithm
    max_generations = 300
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best if we found a better one
        if fitness[0] < best_fitness:
            best_fitness = fitness[0]
            best_solution = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adapt parameters if stagnating
        if stagnation_count > 50:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
            
        # Create new population
        new_population = [population[0].copy()]  # Keep best
        new_fitness = [fitness[0]]
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            if time() - start_time > budget_s * 0.99:
                break
                
            # Tournament selection
            parent1_idx = random.randint(0, population_size // 3)
            parent2_idx = random.randint(0, population_size // 3)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover with probability
            if random.random() < crossover_prob:
                # Two-point crossover
                point1 = random.randint(1, n_points - 2)
                point2 = random.randint(point1 + 1, n_points - 1)
                child = np.copy(parent1)
                child[point1:point2] = parent2[point1:point2]
            else:
                child = parent1.copy()
            
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += random.gauss(0, 0.03)
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points/2) / np.sum(child)
            
            # Check tabu list
            child_tuple = tuple(child)
            if child_tuple in tabu_list:
                continue
                
            # Add to new population
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, dx))
            
            # Add to tabu list
            tabu_list.append(child_tuple)
            if len(tabu_list) > tabu_limit:
                tabu_list.pop(0)
                
        population = new_population
        fitness = new_fitness
        
        # Local refinement every 10 generations
        if generation % 10 == 0:
            # Refine top 5 individuals using SLSQP
            for i in range(min(5, len(population))):
                h_init = population[i].copy()
                
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(objective, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        new_f = evaluate_erdos_solution(h_refined, dx)
                        if new_f < fitness[i]:
                            # Update if better
                            population[i] = h_refined
                            fitness[i] = new_f
                            if new_f < best_fitness:
                                best_fitness = new_f
                                best_solution = h_refined.copy()
                except:
                    pass
    
    # Final refinement step
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Global refinement
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            for i in range(5):
                if time() - start_time > budget_s * 0.99:
                    break
                # Perturb best solution
                h_init = best_solution + np.random.normal(0, 0.01, n_points)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points/2) / np.sum(h_init)
                
                res = minimize(objective, h_init, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-9})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_fitness:
                        best_fitness = c5_refined
                        best_solution = h_refined.copy()
        except:
            pass
    
    # Ensure final solution is valid
    final_h = np.clip(best_solution, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
