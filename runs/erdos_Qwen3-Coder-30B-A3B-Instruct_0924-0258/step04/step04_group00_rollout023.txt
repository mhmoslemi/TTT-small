import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with augmented Lagrangian
    max_iter = 2000
    alpha = 0.01
    beta = 0.001
    lagrange_mult = 0.0
    stagnation_count = 0
    max_stagnation = 50
    
    h_current = best_h.copy()
    h_prev = h_current.copy()
    c5_current = best_c5
    
    for iter_num in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient direction
        grad = np.zeros(n_points)
        # Correlation with shifted version of (1-h)
        h_shifted = 1 - h_current
        corr = np.correlate(h_current, h_shifted, mode='full')
        
        # Find the maximum overlap index
        max_idx = np.argmax(corr)
        # For overlap at max_idx, compute gradient contributions
        # For shift k = max_idx - (n_points-1), i.e., shift from 0 to n_points-1
        if max_idx >= n_points - 1:
            # Positive shift k = max_idx - (n_points-1) (forward shift)
            k = max_idx - (n_points - 1)
            for i in range(n_points - k):
                grad[i] += (1 - h_current[i + k]) 
                grad[i + k] -= h_current[i]
        else:
            # Negative shift k = -(n_points-1 - max_idx) (backwards shift)
            k = -(n_points - 1 - max_idx)
            for i in range(n_points + k):
                grad[i] += (1 - h_current[i - k])
                grad[i - k] -= h_current[i]
                
        # Add Lagrangian terms for sum constraint
        grad += lagrange_mult - beta * (np.sum(h_current) - n_points / 2.0)
        
        # Update
        h_new = h_current - alpha * grad
        
        # Project to feasible region
        h_proj = np.clip(h_new, 0, 1)
        h_proj = h_proj * (n_points/2) / np.sum(h_proj)
        
        c5_new = evaluate_erdos_solution(h_proj, dx)
        
        # Adaptive step size
        if c5_new < c5_current:
            alpha *= 1.2
            c5_current = c5_new
            h_current = h_proj.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_proj.copy()
            stagnation_count = 0
        else:
            alpha *= 0.5
            h_current = h_prev.copy()
            stagnation_count += 1
            
        if stagnation_count > max_stagnation:
            break
            
        h_prev = h_current.copy()
        
    # Final smoothing/refinement
    try:
        h_smooth = best_h.copy()
        window_size = min(5, n_points // 4)
        if window_size > 0:
            h_smooth = np.convolve(h_smooth, np.ones(window_size)/window_size, mode='same')
            h_smooth = np.clip(h_smooth, 0, 1)
            h_smooth = h_smooth * (n_points/2) / np.sum(h_smooth)
        c5_smooth = evaluate_erdos_solution(h_smooth, dx)
        if c5_smooth < best_c5:
            best_h = h_smooth
            best_c5 = c5_smooth
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
