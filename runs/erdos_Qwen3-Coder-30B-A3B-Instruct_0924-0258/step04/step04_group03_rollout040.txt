import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Gradient-based optimization with constraint projection
    def project_h(h):
        # Ensure h is in [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_gradient(h, k_shift):
        # Compute gradient of overlap with respect to h for shift k
        # Convolution: sum_i h[i] * (1 - h[i+k])
        # Gradient: -1 for elements contributing to overlap at shift k
        grad = np.zeros_like(h)
        if k_shift < len(h):
            grad[k_shift:] += -h[:len(h)-k_shift]
            grad[:len(h)-k_shift] += -(1 - h[k_shift:])
        else:
            grad[:k_shift-len(h)+1] += -(1 - h[k_shift-len(h)+1:])
            grad[k_shift-len(h)+1:] += -h[:len(h)-(k_shift-len(h)+1)]
        return grad
    
    # Main gradient descent loop
    current_h = h_start.copy()
    current_c5 = best_c5
    alpha = 0.1
    max_iter = 5000
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and (time() - start_time) < budget_s * 0.9:
        # Compute all overlaps and find the largest one
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        k_shift = max_idx - (len(current_h) - 1)
        
        # Update tracking if better
        if max_overlap < best_c5:
            best_h = current_h.copy()
            best_c5 = max_overlap
        
        # Compute gradient for this shift
        grad = compute_gradient(current_h, k_shift)
        
        # Update with gradient descent
        new_h = current_h - alpha * grad
        
        # Project to feasible region
        new_h = project_h(new_h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 20:
                alpha *= 0.9
                no_improvement_count = 0
        
        iter_count += 1
        
        # Occasionally refine with convex optimization
        if iter_count % 50 == 0 and (time() - start_time) < budget_s * 0.9:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                # Add constraints for top 5 or 10 shifts (lower index shifts are more critical)
                corr_values = np.correlate(current_h, 1 - current_h, mode='full')
                top_k = np.argsort(corr_values)[-10:][::-1]
                for k_idx in top_k:
                    k_shift = k_idx - (len(current_h) - 1)
                    if k_shift < 0:
                        # Cross-correlation: h[-k_shift:] * (1 - h[:len(h)+k_shift])
                        temp_corr = cp.sum(cp.multiply(h[-k_shift:], 1 - h[:len(h)+k_shift]))
                        constraints.append(temp_corr <= max(corr_values[k_idx] * dx, 0.38 * dx))
                    else:
                        # h[:len(h)-k_shift] * (1 - h[k_shift:])
                        temp_corr = cp.sum(cp.multiply(h[:len(h)-k_shift], 1 - h[k_shift:]))
                        constraints.append(temp_corr <= max(corr_values[k_idx] * dx, 0.38 * dx))
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final polish with SLSQP if we have time
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try restarts
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    # Ensure final solution meets constraints exactly
    final_h = project_h(best_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (list(final_h), final_c5, n_points)
