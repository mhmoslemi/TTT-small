import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Projected Gradient Descent with Augmented Lagrangian
    max_iter = 10000
    alpha = 0.01
    lambda_lag = 0.0
    mu = 1.0
    stagnation_count = 0
    max_stagnation = 50
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap values for all shifts
        overlaps = []
        for k in range(-n_points+1, n_points):
            if k >= 0:
                corr = np.sum(h_start[:n_points-k] * (1 - h_start[k:]))
            else:
                corr = np.sum(h_start[-k:] * (1 - h_start[:n_points+k]))
            overlaps.append(corr)
        
        c5_current = max(overlaps) * dx
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for k in range(-n_points+1, n_points):
            if k >= 0:
                grad[:n_points-k] += (1 - h_start[k:]) - h_start[k:][::-1]
            else:
                grad[-k:] += (1 - h_start[:n_points+k]) - h_start[:n_points+k][::-1]
        
        # Augmented Lagrangian update
        lagrangian_grad = grad + lambda_lag + 2 * mu * (np.sum(h_start) - n_points/2)
        
        # Gradient descent step
        h_new = h_start - alpha * lagrangian_grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new C5
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
            alpha *= 1.1  # Increase step size when improving
        else:
            stagnation_count += 1
            alpha *= 0.5  # Decrease step size
            if stagnation_count > max_stagnation:
                # Try random restart
                h_new = np.random.rand(n_points) * 0.5 + 0.25
                h_new = h_new * (n_points/2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    stagnation_count = 0
                    alpha = 0.01  # Reset step size
        
        h_start = h_new
        
        # Update Lagrange multiplier and penalty
        lambda_lag += mu * (np.sum(h_start) - n_points/2)
        
        if stagnation_count > 10:
            mu *= 1.05  # Increase penalty parameter
        
    # Final smoothing adjustment
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    final_c5 = evaluate_erdos_solution(h_final, dx)
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
