import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # ES-style approach with mutation and selection
    pop_size = 50
    elite_size = 10
    mutation_strength = 0.05
    max_generations = 10000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_trial = h_start + np.random.normal(0, mutation_strength, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points/2) / np.sum(h_trial)
        population.append(h_trial)
    
    # Main evolutionary loop
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Selection and reproduction
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(elite_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Mutation
            h_mutant = population[winner_idx] + np.random.normal(0, mutation_strength, n_points)
            h_mutant = np.clip(h_mutant, 0, 1)
            h_mutant = h_mutant * (n_points/2) / np.sum(h_mutant)
            
            new_population.append(h_mutant)
        
        population = new_population
        generation += 1
        
        # Adjust mutation strength based on diversity
        if generation % 10 == 0:
            if generation > 100:
                diversity = np.std([np.sum((h - best_h)**2) for h in population])
                if diversity < 0.01 or generation > 500:
                    mutation_strength *= 0.95
        
        # Occasional constraint optimization
        if generation % 100 == 0:
            try:
                # Try to optimize best solution with scipy
                x0 = best_h.copy()
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                res = minimize(obj, x0, method='SLSQP', 
                              bounds=bounds, constraints=cons_sum, 
                              options={'maxiter': 50, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    c5_refined = obj(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
    
    # Final refinement of best solution
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
