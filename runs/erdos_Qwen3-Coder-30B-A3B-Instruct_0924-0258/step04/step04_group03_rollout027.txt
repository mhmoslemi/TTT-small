import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Improve using a hybrid approach
    def neighbor(h, step_size=0.05):
        h_new = h.copy()
        idx1, idx2 = np.random.choice(n_points, 2, replace=False)
        
        # Apply symmetric perturbation to maintain sum constraint
        perturbation = np.random.uniform(-step_size, step_size)
        h_new[idx1] += perturbation
        h_new[idx2] -= perturbation
        
        # Clamp to feasible bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_new)
        adjustment = (n_points / 2) / current_sum
        h_new *= adjustment
        return h_new
    
    # Simulated Annealing with adaptive step size
    current_h = h_start.copy()
    current_c5 = evaluate_erdos_solution(current_h, dx)
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.9995
    step_base = 0.05
    
    # Track best solution found
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Adaptive step size based on temperature
        current_step = step_base * temperature
        new_h = neighbor(current_h, current_step)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
    
    # Convex optimization refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40 and best_c5 > 0.37:
        try:
            h = cp.Variable(n_points)
            constraints = [
                cp.sum(h) == n_points / 2.0,
                h >= 0,
                h <= 1
            ]
            
            # Enhance convex relaxation with tight overlap constraint
            corr = cp.Variable(2 * n_points - 1)
            # Add constraints to enforce overlap bound
            for i in range(2 * n_points - 1):
                if i < n_points:
                    # Index i corresponds to correlation at offset
                    offset = i
                    corr_sum = cp.sum(cp.multiply(h[:n_points-offset], 1 - h[offset:]))
                else:
                    offset = 2 * n_points - 2 - i
                    corr_sum = cp.sum(cp.multiply(h[offset:], 1 - h[:n_points-offset]))
                # This constraint is for indirect maximization
                constraints.append(corr_sum <= best_c5 * dx)
            
            # Objective is to minimize the bound
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.ECOS, eps=1e-6)
                if prob.status == cp.OPTIMAL:
                    h_refined = h.value
                    c5_refined = evaluate_erdos_solution(h_refined, dx)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except Exception:
                pass
        except Exception:
            pass
    
    # Final gradient-based refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        h_restart = best_h + np.random.normal(0, 0.01, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        try:
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
