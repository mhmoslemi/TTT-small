import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a structured initial guess
        h_start[:n_points//3] = 1.0
        h_start[n_points//3:2*n_points//3] = 0.0
        h_start[2*n_points//3:] = 1.0
        # Normalize to ensure sum is correct
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Parameters
    max_iter = 1000
    step_size = 0.01
    patience = 100
    patience_counter = 0
    
    h = h_start.copy()
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (len(h) - 1)
        
        # Compute gradient for shift k_max
        grad = np.zeros_like(h)
        if k_max == 0:
            # Gradient is 1 - 2*h for k=0 case
            grad = 1 - 2 * h
        else:
            # Gradient: 1 - h[i+k] for i such that i+k < len(h) and i >= 0,
            # also -h[i] if k != 0
            for i in range(len(h)):
                if 0 <= i + k_max < len(h):
                    grad[i] += 1 - h[i + k_max]
                if k_max != 0:
                    grad[i] -= h[i]
        
        # Update h
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Compute new overlap
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept if improvement
        if new_c5 < best_c5:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            step_size *= 0.9
            patience_counter = 0
    
    # Final optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30 and best_c5 > 0.3807:
        # Try localized optimization if still above target
        try:
            from scipy.optimize import minimize
            def obj(h_vec):
                return evaluate_erdos_solution(h_vec, dx)
                
            def con_sum(h_vec):
                return np.sum(h_vec) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-7})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
