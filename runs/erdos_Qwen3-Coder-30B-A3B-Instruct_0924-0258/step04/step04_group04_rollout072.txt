```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Set up initial localized structure
        h_start[5:15] = 0.8
        h_start[20:30] = 0.2
        # Normalize to sum to n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization first on n=20
    coarse_n = 20
    coarse_dx = 2.0 / coarse_n
    h_coarse = h_start[::(n_points // coarse_n)] if n_points > coarse_n else h_start.copy()
    h_coarse = h_coarse * (coarse_n/2) / np.sum(h_coarse)
    
    # Optimization variables
    h_var = cp.Variable(coarse_n)
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == coarse_n/2]
    
    # Penalty method optimization
    max_iter = 100
    penalty = 1.0
    lr = 0.05
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full')
        max_corr = np.max(corr)
        
        # Objective with penalty: minimize max overlap
        objective = cp.Minimize(cp.sum(corr) + penalty * cp.sum(cp.pos(corr - max_corr)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Update penalty based on violation
                violation = np.max(corr) - max_corr
                penalty = max(1e-6, penalty * (1.0 + lr * violation))
                
                # Update h_coarse
                h_coarse = h_new
                h_coarse = h_coarse * (coarse_n/2) / np.sum(h_coarse)
            
                # Evaluate current solution
                c5_val = evaluate_erdos_solution(h_coarse, coarse_dx)
                if c5_val < best_c5:
                    best_h = h_coarse.copy()
                    best_c5 = c5_val
        except:
            pass
    
    # Upsample to fine grid (n=35)
    if n_points > coarse_n:
        # Interpolate
        x_original = np.linspace(0, 2, coarse_n)
        x_fine = np.linspace(0, 2, n_points)
        f_interp = interp1d(x_original, h_coarse, kind='linear', fill_value='extrapolate')
        h_fine = f_interp(x_fine)
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points/2) / np.sum(h_fine)
        
        # Refine with SQP if time permits
        remaining_time = budget_s - (time() - start_time)
        if remaining_time > 30:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, h_fine, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    final_h = res.x
                    final_c5 = evaluate_erdos_solution(final_h, dx)
                    if final_c5 < best_c5:
                        best_h = final_h
                        best_c5 = final_c5
            except:
                pass
    
    # Final optimization check
    if best_c5 > 0.3807:
        remaining_time = budget_s - (time() - start_time)
        if remaining_time > 30:
            # Try with full step function optimization now
            def obj(h):
                return evaluate_erdos_solution(h, dx)
                
            def con_sum(h):
                return np.sum(h) - n_points/2
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': con_sum}
            
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons, 
                              options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    final_h = res.x
                    final_c5 = evaluate_erdos_solution(final_h, dx)
                    if final_c5 < best_c5:
                        best_h = final_h
                        best_c5 = final_c5
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
```
