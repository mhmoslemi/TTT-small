import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with Augmented Lagrangian approach
    max_iter = 1000
    alpha = 0.01  # initial step size
    lambda_lag = 0.0  # lagrange multiplier
    mu = 1.0  # penalty parameter
    patience = 0
    max_patience = 50
    
    def compute_gradient(h):
        # compute gradient of overlap w.r.t. h
        grad = np.zeros_like(h)
        # sum over all possible shifts
        for k in range(1, n_points):
            if k <= n_points // 2:
                # shift forward
                for i in range(n_points - k):
                    grad[i] += (1 - h[i + k])
                    grad[i + k] -= h[i]
            else:
                # shift backward
                for i in range(n_points - (n_points - k)):
                    grad[i] += (1 - h[i + k - n_points])
                    grad[i + k - n_points] -= h[i]
        # add Lagrangian gradient
        grad += lambda_lag + 2 * mu * (np.sum(h) - n_points / 2.0)
        return grad
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        return h
    
    def compute_lagrangian(h, lambda_lag, mu):
        overlap = evaluate_erdos_solution(h, dx)
        penalty = lambda_lag * (np.sum(h) - n_points / 2.0) + mu * (np.sum(h) - n_points / 2.0)**2
        return overlap + penalty
    
    h_current = best_h.copy()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient using the full convolution
        grad = compute_gradient(h_current)
        
        # Update step
        h_new = h_current - alpha * grad
        
        # Project to feasible region
        h_projected = project_h(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_projected, dx)
        
        # Adaptive step size
        if c5_new < best_c5:
            best_h = h_projected.copy()
            best_c5 = c5_new
            last_improvement = iteration
            alpha *= 1.2
        else:
            alpha *= 0.5
            h_projected = h_current.copy()
            c5_new = evaluate_erdos_solution(h_projected, dx)
        
        h_current = h_projected
        
        # Check for stagnation
        if iteration - last_improvement > max_patience:
            break
            
        # Update Lagrange multiplier
        sum_violation = np.sum(h_current) - n_points / 2.0
        lambda_lag += mu * sum_violation
        
        # Slight smoothing to avoid numerical issues
        if iteration % 20 == 0:
            h_smooth = h_current.copy()
            for i in range(1, n_points - 1):
                h_smooth[i] = 0.8 * h_current[i] + 0.1 * (h_current[i-1] + h_current[i+1])
            h_current = project_h(h_smooth)
    
    # Final validation
    try:
        h_final = project_h(best_h)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
