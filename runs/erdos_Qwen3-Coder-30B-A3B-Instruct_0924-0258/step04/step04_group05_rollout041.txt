import numpy as np
from scipy.optimize import minimize
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initial population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Population parameters
    pop_size = 40
    tabu_limit = 50
    max_stagnation = 50
    mutation_rate = 0.15
    crossover_prob = 0.7
    
    # Initialize population with variations of initial solution
    population = []
    for i in range(10):
        h = initial_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Add random valid solutions
    for i in range(30):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Initialize tabu list
    tabu_list = []
    
    # Evaluate initial population
    fitness = []
    for h in population:
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Search loop
    generation = 0
    stagnation_count = 0
    last_best = best_c5
    
    while time() - start_time < budget_s * 0.95:
        if stagnation_count > max_stagnation:
            # Increase mutation rate if stuck
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        elif generation % 10 == 0:
            # Perform local refinement
            top_indices = np.argsort(fitness)[:5]
            for idx in top_indices:
                h_refined = population[idx].copy()
                def obj(x):
                    return evaluate_erdos_solution(x, 2.0 / n_points)
                def constraint_sum(x):
                    return np.sum(x) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_new = res.x
                        c5_new = obj(h_new)
                        if c5_new < fitness[idx]:
                            population[idx] = h_new
                            fitness[idx] = c5_new
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_new.copy()
                except:
                    pass
        
        # Create new generation
        new_population = []
        new_fitness = []
        
        # Elitism: Keep best solution
        new_population.append(best_h.copy())
        new_fitness.append(best_c5)
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection (tournament)
            tournament_size = 3
            selected_indices = []
            for _ in range(2):
                tournament = random.sample(range(pop_size), min(tournament_size, pop_size))
                winner = min(tournament, key=lambda i: fitness[i])
                selected_indices.append(winner)
            
            parent1, parent2 = population[selected_indices[0]], population[selected_indices[1]]
            
            if random.random() < crossover_prob:
                # Uniform crossover with mask
                mask = np.random.random(n_points) > 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutation
            if random.random() < mutation_rate:
                mutation_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                for i in mutation_indices:
                    child[i] = random.random()
                
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            # Avoid tabu solutions
            child_hash = tuple(np.round(child, decimals=6))
            if child_hash in tabu_list:
                tabu_list.remove(child_hash)  # Remove from tabu
                child = np.random.random(n_points)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            new_fitness.append(evaluate_erdos_solution(child, 2.0 / n_points))
            
            # Add to tabu
            tabu_list.append(child_hash)
            if len(tabu_list) > tabu_limit:
                tabu_list.pop(0)
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Check for improvement
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            stagnation_count = 0
            last_best = best_c5
        else:
            stagnation_count += 1
            
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
            
        generation += 1
    
    # Final refinement of best solution
    try:
        def obj(x):
            return evaluate_erdos_solution(x, 2.0 / n_points)
        def constraint_sum(x):
            return np.sum(x) - n_points / 2.0
        
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            best_h = res.x
            best_c5 = obj(best_h)
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
