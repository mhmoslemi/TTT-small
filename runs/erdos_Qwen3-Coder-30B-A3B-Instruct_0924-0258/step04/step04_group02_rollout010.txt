import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Account for initial evaluation cost
    if time() - start_time > budget_s * 0.1:
        return (list(best_h), best_c5, n_points)
    
    # Parameterize with Fourier coefficients
    # h(x) = sigmoid(a0 + sum_{k=1}^n [ak*cos(kπx) + bk*sin(kπx)])
    # We'll optimize the coefficients [a0, a1, b1, a2, b2, ...]
    
    # Initialize coefficients
    if 'initial_h_values' in globals():
        # Try to initialize from parent
        h0 = h_start
        # Simple fit to get initial coefficients (accurate enough)
        coefs = np.zeros(n_points)  # Placeholder
        if n_points > 1:
            # Use a simple linear approach (avoiding complex optimization for init)
            coefs[0] = np.log(np.mean(h0) / (1 - np.mean(h0)))
            # Try to capture simple harmonic components
            if n_points > 2:
                # Not critical, just start near zero
                coefs[1:] = np.random.normal(0, 0.1, n_points-1) 
        else:
            # To prevent all-zero coefficients in n_points==1
            coefs = np.random.normal(0, 0.1, n_points)
    else:
        # Random initialization
        coefs = np.random.normal(0, 0.1, n_points)
    
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    def h_from_coefs(coefs):
        x = np.linspace(0, 2, n_points)
        # The first coefficient is constant term
        if len(coefs) < 1:
            return np.ones(n_points)*0.5
        result = coefs[0] * np.ones(n_points)
        # Use frequencies k=1 to len(coefs)-1
        for k in range(1, min(len(coefs), n_points)):
            if k % 2 == 1:
                # odd indices (b_k) -> sin(kπx)
                result += coefs[k] * np.sin(k * np.pi * x)
            else:
                # even indices (a_k) -> cos(kπx)
                result += coefs[k] * np.cos(k * np.pi * x)
        h = sigmoid(result)
        return h
    
    def normalize_h(h):
        """Normalize so that sum(h)*dx = 1"""
        h_sum = np.sum(h) * dx
        return h * (1.0 / h_sum)
    
    def compute_overlap(coefs):
        h = h_from_coefs(coefs)
        h_norm = normalize_h(h)
        return evaluate_erdos_solution(h_norm, dx)
    
    def compute_gradient(coefs, k_max):
        """Numerical gradient at the k_max overlap"""
        h = h_from_coefs(coefs)
        h_norm = normalize_h(h)
        h_norm = h_norm * dx
        
        grad = np.zeros_like(coefs)
        eps = 1e-5
        
        # Compute Jacobian via finite differences for the final overlap integral 
        for i in range(len(coefs)):
            coefs_h = coefs.copy()
            coefs_l = coefs.copy()
            coefs_h[i] += eps
            coefs_l[i] -= eps
            
            h_h = h_from_coefs(coefs_h)
            h_l = h_from_coefs(coefs_l)
            
            h_h = normalize_h(h_h)
            h_l = normalize_h(h_l)
            
            # Compute C_k using convolution logic (approximate)
            corr_h = np.correlate(h_h, 1 - h_h, mode='full')
            corr_l = np.correlate(h_l, 1 - h_l, mode='full')
            
            # We will approximate partial derivative at k_max = k_max
            # Using finite difference of correlation at k_max
            grad[i] = (corr_h[k_max] - corr_l[k_max]) / (2 * eps)
        
        return grad
    
    # Optimizer parameters
    max_iter = 500
    learning_rate = 1e-2
    adaptive_lr = True
    
    for iter in range(max_iter):
        if time() - start_time > budget_s * 0.9:
            break
            
        c5 = compute_overlap(coefs)
        if c5 < best_c5:
            h_best = h_from_coefs(coefs)
            h_best = normalize_h(h_best)
            best_h = h_best
            best_c5 = c5
            
        # Estimate gradient with respect to overlap
        h = h_from_coefs(coefs)
        h_norm = normalize_h(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        k_max = np.argmax(corr)
        
        grad = compute_gradient(coefs, k_max)
        
        # Update
        if adaptive_lr and iter > 10:
            # Simple adaptive - reduce LR if not improving
            if iter % 20 == 0:
                if compute_overlap(coefs) >= c5:
                    learning_rate *= 0.9
                
        coefs -= learning_rate * grad
        coefs = np.clip(coefs, -10, 10)  # Prevent gradient explosion
        
    # Final cleanup
    h_final = h_from_coefs(coefs)
    h_final = normalize_h(h_final)
    c5_final = evaluate_erdos_solution(h_final, dx)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final
    
    # Also do a final check to make sure normalization is correct
    normalization = np.sum(best_h) * dx
    if abs(normalization - 1.0) > 1e-4:
        best_h = best_h * (1.0 / normalization)
    
    return (list(best_h), best_c5, n_points)
