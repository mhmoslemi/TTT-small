import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial local search with different methods
    methods_to_try = ['SLSQP', 'L-BFGS-B', 'TNC']
    
    for method in methods_to_try:
        if time() - start_time > budget_s * 0.95:
            break
            
        try:
            # Small random perturbations
            for _ in range(3):
                h_pert = h_start + np.random.normal(0, 0.02, n_points)
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                
                res = minimize(objective, h_pert, method=method, 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-8})
                
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        
        except:
            continue
    
    # Additional optimization using simulated annealing inspired approach
    if time() - start_time < budget_s * 0.9:
        # Try to improve using local gradient-based approach
        def objective_with_reg(h):
            return evaluate_erdos_solution(h, dx) + 1e-6 * np.sum((h - 0.5)**2)
            
        # Simulated annealing-like approach - restarts with small random perturbations
        max_restarts = 5
        for restart in range(max_restarts):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            res = minimize(objective_with_reg, h_restart, method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 100, 'ftol': 1e-8})
            
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
    
    return (list(best_h), best_c5, n_points)
