import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Reparameterize h to work in logit space for easier optimization
    def h_from_z(z, b):
        return 1.0 / (1.0 + np.exp(-z - b))
    
    def z_from_h(h, b):
        return np.log(h / (1.0 - h)) - b
    
    # Binary search to find b that makes sum(h) = n_points / 2
    def find_b(z):
        def sum_h(b):
            h = h_from_z(z, b)
            return np.sum(h) - n_points / 2.0
            
        b_low, b_high = -100.0, 100.0
        for _ in range(50):
            b_mid = (b_low + b_high) / 2.0
            if sum_h(b_mid) < 0:
                b_low = b_mid
            else:
                b_high = b_mid
        return b_mid
    
    # Initialize with a good starting point
    z_start = z_from_h(h_start, 0.0)
    z_current = z_start.copy()
    b_current = find_b(z_current)
    
    # Optimization parameters
    max_iterations = 3000
    learning_rate = 0.1
    tolerance = 1e-6
    
    # SGD with momentum for optimization
    momentum = 0.9
    velocity = np.zeros_like(z_current)
    
    for iteration in range(max_iterations):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Compute current h and b
        h_current = h_from_z(z_current, b_current)
        
        # Evaluate overlap for all shifts
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        max_index = np.argmax(corr)
        k_max = max_index - (len(h_current) - 1)
        
        # Compute gradient of overlap for k_max
        # For shift k_max, overlap = sum h[i] * (1 - h[i + k_max])
        # Gradient is d/dh[i] of that overlap
        grad = np.zeros_like(h_current)
        
        # Set up shift for correlation
        for i in range(len(h_current)):
            j = i + k_max
            if 0 <= j < len(h_current):
                grad[i] += 1 - h_current[j]
            j = i - k_max
            if 0 <= j < len(h_current):
                grad[i] += h_current[j]
        
        # Convert gradient to z-space: dC/dz = dC/dh * dh/dz
        # We have h = sigma(z + b) so dh/dz = h * (1 - h)
        dh_dz = h_current * (1.0 - h_current)
        
        # Avoid division by zero
        dh_dz = np.clip(dh_dz, 1e-10, 1.0)
        
        grad_z = grad / dh_dz
        
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad_z
        z_current += velocity
        
        # Update b to maintain constraint
        b_current = find_b(z_current)
        
        # Check for convergence
        if np.linalg.norm(velocity) < tolerance:
            break
            
        # Check for improvement
        h_new = h_from_z(z_current, b_current)
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        # If too many iterations, reduce learning rate
        if iteration % 500 == 0 and iteration > 0:
            learning_rate *= 0.9
            
    # Local refinement
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Refine using L-BFGS-B
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        best_h = np.clip(best_h, 1e-5, 1.0 - 1e-5)
        
        # Create approximated version for better initial guess
        h_refine = best_h.copy()
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        try:
            res = minimize(objective, h_refine, method='L-BFGS-B', 
                          bounds=[(1e-5, 1.0-1e-5) for _ in range(n_points)],
                          options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
            
    return (list(best_h), best_c5, n_points)
