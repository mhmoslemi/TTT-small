import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        
    # Normalize so sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated annealing parameters
    T = 1.0
    alpha = 0.995
    max_iterations = 100000
    delta_max = 0.1
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Make a perturbation to two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Create new h
        h_new = h.copy()
        
        # Adjust the two points
        if h_new[i] + delta > 1:
            delta = 1 - h_new[i]
        elif h_new[i] + delta < 0:
            delta = -h_new[i]
            
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure the sum constraint
        sum_diff = np.sum(h_new) - (n_points / 2)
        if abs(sum_diff) > 1e-10:
            # Adjust one more point to maintain sum
            k = np.random.choice(n_points)
            h_new[k] -= sum_diff
            
        # Clamp values
        h_new = np.clip(h_new, 0, 1)
        
        # Recompute sum and adjust to exactly match target
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / current_sum
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept with simulated annealing probability
        if new_c5 < best_c5 or np.random.random() < np.exp(-(new_c5 - best_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_h = h_new
                best_c5 = new_c5
                
        # Cool down
        T *= alpha
        if T < 1e-10:
            T = 1e-10
    
    return (list(best_h), best_c5, n_points)
