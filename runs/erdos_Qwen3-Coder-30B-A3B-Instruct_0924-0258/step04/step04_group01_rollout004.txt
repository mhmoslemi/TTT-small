import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # 3-cluster construction
        cluster_size = 5
        cluster_centers = [0.5, 1.0, 1.5]
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (2*cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Population-based GA with SQP refinement
    population_size = 25
    population = []
    
    for _ in range(population_size):
        h_candidate = h_start + np.random.normal(0, 0.015, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        population.append(h_candidate)
    
    # Add some diverse random initializations
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points/2) / np.sum(h_rand)
        population.append(h_rand)
    
    # GA parameters
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    for iteration in range(300):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0/n_points)
            fitness.append(c5)
        
        # Selection: top half
        sorted_indices = np.argsort(fitness)
        top_half = sorted_indices[:population_size//2]
        selected = [population[i] for i in top_half]
        
        # Create new population through crossover and mutation
        new_population = selected.copy()
        
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected) >= 2:
                # Crossover
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                crossover_point = np.random.randint(1, n_points-1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                # Mutation or take parent
                parent = np.random.choice(selected)
                child = parent.copy()
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.02, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    child = child * (n_points/2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # SQP refinement for each candidate
        for i in range(len(population)):
            h_current = population[i]
            
            def obj(h):
                return evaluate_erdos_solution(h, 2.0/n_points)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_current, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    if evaluate_erdos_solution(h_refined, 2.0/n_points) < evaluate_erdos_solution(population[i], 2.0/n_points):
                        population[i] = h_refined
            except:
                pass
            
            # Update best
            current_c5 = evaluate_erdos_solution(population[i], 2.0/n_points)
            if current_c5 < best_c5:
                best_h = population[i].copy()
                best_c5 = current_c5
    
    # Final normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
