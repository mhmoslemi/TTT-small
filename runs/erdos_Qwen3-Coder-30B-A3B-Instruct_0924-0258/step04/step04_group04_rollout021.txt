import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.interpolate import interp1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        # Ensure sum is correct
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Optimization with alternating approach
    max_iter = 500
    penalty_rate = 0.1
    
    # Coarse grid first
    coarse_points = 20
    coarse_h = np.interp(np.linspace(0, 2, coarse_points), 
                         np.linspace(0, 2, n_points), h_start)
    coarse_h = coarse_h * (coarse_points/2) / np.sum(coarse_h)
    
    # Penalty terms initialization
    penalty_terms = np.ones(2 * coarse_points - 1)
    
    def optimize_coarse():
        nonlocal coarse_h, penalty_terms
        # Define variable
        h_var = cp.Variable(coarse_points)
        # Define constraints
        constraints = [
            cp.sum(h_var) == coarse_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        # Compute correlations
        corr = np.correlate(coarse_h, 1 - coarse_h, mode='full')
        # Add penalties for each shift
        obj = cp.Minimize(
            cp.sum(
                cp.multiply(
                    cp.maximum(corr + penalty_terms, 0),
                    dx
                )
            )
        )
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                # Update penalties
                updates = np.correlate(new_h, 1 - new_h, mode='full') - corr
                penalty_terms = np.maximum(penalty_terms * (1 + penalty_rate * updates), 1e-6)
                return new_h
        except Exception:
            pass
        return coarse_h
    
    # Coarse optimization phase
    for iter_num in range(100):
        if time() - start_time > budget_s * 0.9:
            break
        coarse_h = optimize_coarse()
        coarse_h = np.maximum(np.minimum(coarse_h, 1.0), 0.0)
        # Interpolate back
        fine_h = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, coarse_points), coarse_h)
        fine_h = fine_h * (n_points/2) / np.sum(fine_h)
        curr_c5 = evaluate_erdos_solution(fine_h, dx)
        if curr_c5 < best_c5:
            best_h = fine_h.copy()
            best_c5 = curr_c5
    
    # Upsample to medium resolution
    medium_points = 35
    medium_h = np.interp(np.linspace(0, 2, medium_points), 
                         np.linspace(0, 2, n_points), best_h)
    medium_h = medium_h * (medium_points/2) / np.sum(medium_h)
    penalty_terms = np.ones(2 * medium_points - 1)
    
    def optimize_medium():
        nonlocal medium_h, penalty_terms
        h_var = cp.Variable(medium_points)
        constraints = [
            cp.sum(h_var) == medium_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        corr = np.correlate(medium_h, 1 - medium_h, mode='full')
        obj = cp.Minimize(
            cp.sum(
                cp.multiply(
                    cp.maximum(corr + penalty_terms, 0),
                    dx
                )
            )
        )
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                # Update penalties
                updates = np.correlate(new_h, 1 - new_h, mode='full') - corr
                penalty_terms = np.maximum(penalty_terms * (1 + penalty_rate * updates), 1e-6)
                return new_h
        except Exception:
            pass
        return medium_h
    
    # Refinement on medium grid
    for iter_num in range(100):
        if time() - start_time > budget_s * 0.9:
            break
        medium_h = optimize_medium()
        medium_h = np.maximum(np.minimum(medium_h, 1.0), 0.0)
        curr_c5 = evaluate_erdos_solution(medium_h, dx)
        if curr_c5 < best_c5:
            best_h = medium_h.copy()
            best_c5 = curr_c5
    
    # Final fine optimization on original grid
    fine_h = medium_h
    
    # Final NL optimization with SQP
    def obj_func(h):
        return evaluate_erdos_solution(h, dx)
    
    def con_sum(h):
        return np.sum(h) - n_points/2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    final_h = fine_h.copy()
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    # Refinement with scipy
    for i in range(100):
        if time() - start_time > budget_s * 0.95:
            break
        try:
            res = minimize(obj_func, final_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h.copy()
                    best_c5 = final_c5
                else:
                    break  # Early stop if no improvement
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
