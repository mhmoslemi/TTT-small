import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create a structured initial guess 
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two clusters to start
        cluster1 = np.linspace(0.2, 0.6, 15)
        cluster2 = np.linspace(1.2, 1.6, 15)
        for c in cluster1:
            idx = int(c / dx)
            if 0 <= idx < n_points:
                h_start[idx] = 1.0
        for c in cluster2:
            idx = int(c / dx)
            if 0 <= idx < n_points:
                h_start[idx] = 1.0
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    max_iter = 1000
    step_size = 0.01
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps and find max
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Compute gradient for the k_max shift
        grad = np.zeros(n_points)
        # For shift k_max, the derivative of the overlap w.r.t h[i] 
        # is (1 - h[i+k_max]) if k_max > 0, and -h[i] if k_max == 0
        if k_max < n_points:
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] = 1 - best_h[i + k_max]
        else:
            # wrap-around case (for mode='full')
            # k_max >= n_points means we need to handle wrap-around
            wrap_k = k_max - n_points + 1
            for i in range(n_points):
                if i + wrap_k < n_points:
                    grad[i] = 1 - best_h[i + wrap_k]
                else:
                    grad[i] = 1 - best_h[i + wrap_k - n_points]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
        
        # Backtracking line search
        alpha = step_size
        while True:
            # Update step
            h_new = best_h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            if new_c5 < max_overlap - 1e-10 or alpha < 1e-8:
                break
            alpha *= 0.5
        
        # Accept new solution
        best_h = h_new
        best_c5 = new_c5

    return (list(best_h), best_c5, n_points)
