import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Mixed heuristic approach
    max_restarts = 50
    restart_count = 0
    
    while time() - start_time < budget_s * 0.95:
        # Local search with different strategies
        strategies = []
        
        # Strategy 1: Random perturbation with normalization
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        strategies.append(h_pert)
        
        # Strategy 2: Sigmoid smoothing + local optimization
        h_smooth = 1.0 / (1.0 + np.exp(-10*(best_h - 0.5)))
        h_smooth = h_smooth * (n_points/2) / np.sum(h_smooth)
        strategies.append(h_smooth)
        
        # Strategy 3: Binary relaxation with scaling
        h_binary = (best_h > 0.5).astype(float)
        h_binary = h_binary * (n_points/2) / np.sum(h_binary)
        strategies.append(h_binary)
        
        # Strategy 4: Piecewise constant approximation
        h_piecewise = best_h.copy()
        for i in range(0, n_points, 4):
            end = min(i + 4, n_points)
            avg_val = np.mean(best_h[i:end])
            h_piecewise[i:end] = avg_val
        h_piecewise = h_piecewise * (n_points/2) / np.sum(h_piecewise)
        strategies.append(h_piecewise)
        
        # Evaluate all strategies
        for h_test in strategies:
            try:
                c5_test = evaluate_erdos_solution(h_test, dx)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_test
                    restart_count = 0
            except:
                continue
        
        # SLSQP local optimization with current best
        try:
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Local optimization around current best
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 50, 'ftol': 1e-6})
            
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
        
        # Occasionally use CVXPY convex relaxation
        if time() - start_time < budget_s * 0.7 and restart_count % 7 == 0:
            try:
                h_opt = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_opt) == n_points / 2.0,
                    h_opt >= 0,
                    h_opt <= 1
                ]
                
                # Use a subset of constraint bindings for efficiency
                max_k = n_points // 2
                for k in range(1, min(20, max_k+1)):
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h_opt[:n_points-k], 1 - h_opt[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_opt[k-n_points+1:], 1 - h_opt[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=50)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_opt.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        # Final refinement
        try:
            h_final = np.clip(best_h, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
            
        if restart_count > max_restarts:
            break
    
    return (list(best_h), best_c5, n_points)
