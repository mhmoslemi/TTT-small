import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a 3-cluster construction
        centers = [0.4, 1.0, 1.6]
        widths = [0.2, 0.2, 0.2]
        heights = [0.8, 0.6, 0.9]
        for i, (c, w, h) in enumerate(zip(centers, widths, heights)):
            start = max(0, int((c - w/2)/dx))
            end = min(n_points, int((c + w/2)/dx))
            h_start[start:end] = h
        # Normalize sum
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0 / n_points)
    
    # Genetic algorithm parameters
    population_size = 20
    generations = 150
    mutation_rate = 0.3
    crossover_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if np.random.rand() < 0.7:
            # Perturb initial solution
            h = h_start + np.random.normal(0, 0.02, n_points)
        else:
            # Fresh random
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    # Main GA loop
    for generation in range(generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, 2.0 / n_points)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        best_fitness = fitness[sorted_indices[0]]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[sorted_indices[0]].copy()
        
        # Select top 50%
        selected_indices = sorted_indices[:population_size//2]
        selected_population = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = selected_population.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(selected_population) >= 2:
                # Crossover
                parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                # Clone one parent
                parent = np.random.choice(selected_population)
                child1 = parent.copy()
                child2 = parent.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child1[mutate_idx] = np.clip(child1[mutate_idx] + np.random.normal(0, 0.03), 0, 1)
            if np.random.rand() < mutation_rate:
                mutate_idx = np.random.randint(0, n_points)
                child2[mutate_idx] = np.clip(child2[mutate_idx] + np.random.normal(0, 0.03), 0, 1)
            
            # Normalize
            child1 = child1 * (n_points/2) / np.sum(child1)
            child2 = child2 * (n_points/2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        
        # SQP refinement step
        for i in range(len(population)):
            h_candidate = population[i]
            
            def obj(h):
                return evaluate_erdos_solution(h, 2.0 / n_points)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(obj, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-6})
                if res.success:
                    h_refined = res.x
                    h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                    population[i] = h_refined
            except:
                pass
                
    # Final check
    best_c5 = evaluate_erdos_solution(best_h, 2.0 / n_points)
    return (list(best_h), best_c5, n_points)
