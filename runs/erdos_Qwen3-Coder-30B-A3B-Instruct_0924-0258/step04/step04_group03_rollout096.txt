```python
import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO approach
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_bests = []
    personal_best_vals = []
    
    # Central particle + perturbations
    central = h_start.copy()
    central = central * (n_points / 2) / np.sum(central)
    swarm.append(central)
    velocities.append(np.zeros(n_points))
    personal_bests.append(central.copy())
    personal_best_vals.append(evaluate_erdos_solution(central, dx))
    
    for i in range(1, swarm_size):
        h = central + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        personal_bests.append(h.copy())
        personal_best_vals.append(evaluate_erdos_solution(h, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_bests[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    if global_best_val < best_c5:
        best_h = global_best.copy()
        best_c5 = global_best_val
    
    # PSO iteration
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - swarm[i]) + 
                           c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            # Clamp to [0,1]
            swarm[i] = np.clip(swarm[i], 0, 1)
            # Normalize to sum to n_points/2
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate
            c_val = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if c_val < personal_best_vals[i]:
                personal_best_vals[i] = c_val
                personal_bests[i] = swarm[i].copy()
                
                # Update global best
                if c_val < global_best_val:
                    global_best_val = c_val
                    global_best = swarm[i].copy()
                    
                    if global_best_val < best_c5:
                        best_h = global_best.copy()
                        best_c5 = global_best_val
                        
        # Occasionally refine with convex optimization
        if iteration % 50 == 0 and time() - start_time < budget_s * 0.9:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Linear relaxation of max constraint
                # For actual convex problem, we can create an auxiliary variable
                # and add a constraint that it's >= all correlations
                # But here we'll refine the global best solution
                
                h_refined = global_best.copy()
                try:
                    # Try SLSQP with global best
                    def objective(h):
                        return evaluate_erdos_solution(h, dx)
                        
                    def constraint_sum(h):
                        return np.sum(h) - n_points / 2.0
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    cons_sum = {'type': 'eq', 'fun': constraint_sum}
                    
                    res = minimize(objective, h_refined, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-8})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except:
                    pass
            except:
                pass
    
    # Final polishing with SLSQP using best so far
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        h_restart = best_h.copy()
        try:
            res = minimize(objective, h_restart, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
```
