import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    # Normalize to desired sum
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2) / np.sum(h_start)
    
    # Ensure bounds
    h_start = np.clip(h_start, 0, 1)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Simulated Annealing parameters
    T_initial = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    T = T_initial
    current_h = h_start.copy()
    current_c5 = best_c5
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two random points
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.uniform(-0.1, 0.1)
        
        # Make sure we don't violate bounds
        h_new = current_h.copy()
        if h_new[i] + delta > 1:
            delta = 1 - h_new[i]
        elif h_new[i] + delta < 0:
            delta = -h_new[i]
        
        h_new[i] += delta
        h_new[j] -= delta
        
        # Maintain sum constraint by adjusting one more point
        if j != 0:
            h_new[0] += delta
        else:
            h_new[1] -= delta
            
        # Ensure bounds and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        candidate_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept with probability based on temperature
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            current_h = h_new
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cool down
        T *= alpha
        
    return (list(best_h), best_c5, n_points)
