import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create a structured initial guess
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Three clusters with varying heights to avoid uniform solutions
        cluster_starts = [0.3, 1.0, 1.7]
        cluster_sizes = [3, 5, 4]
        for i, (start, size) in enumerate(zip(cluster_starts, cluster_sizes)):
            lo = max(0, int((start - size/2*dx)/dx))
            hi = min(n_points, int((start + size/2*dx)/dx))
            h_start[lo:hi] = 1.0 / (size * 2)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Hyperparameters
    population_size = 20
    max_iter = 300
    mutation_scale = 0.02
    crossover_rate = 0.5
    elite_count = 5
    
    # Create initial population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h = h_start + np.random.normal(0, mutation_scale, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        population.append(h)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness = [evaluate_erdos_solution(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Select elite
        elites = [population[i] for i in sorted_indices[:elite_count]]
        best_c5_candidate = fitness[sorted_indices[0]]
        if best_c5_candidate < best_c5:
            best_h = population[sorted_indices[0]].copy()
            best_c5 = best_c5_candidate
            
        # Create offspring
        offspring = elites.copy()
        
        # Crossover: blend pairs of solutions
        for _ in range(population_size - elite_count):
            if time() - start_time > budget_s * 0.95:
                break
            parent1 = population[np.random.choice(sorted_indices[:population_size//2])]
            parent2 = population[np.random.choice(sorted_indices[:population_size//2])]
            
            if np.random.rand() < crossover_rate:
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                child = parent1.copy()
                
            # Mutation
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, mutation_scale, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                
            # Refinement with SQP if not just mutated
            if np.random.rand() < 0.7:
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(obj, child, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 50, 'ftol': 1e-6})
                    if res.success:
                        child = res.x
                except:
                    pass
            
            offspring.append(child)
        
        population = offspring.copy()
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), best_c5, n_points)
