import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp
from scipy.ndimage import gaussian_filter1d

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    remaining_time = budget_s
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.zeros(n_points)
        h_start[5:15] = 1.0
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    def solve_convex(h_in, penalties):
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.maximum(
            [np.correlate(h_in, 1 - h_in, 'full')[k] + penalties[k] for k in range(len(penalties))])))
        constraints = [
            cp.sum(h) == n_points/2,
            h >= 0,
            h <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_in
    
    # Coarse grid optimization
    for iteration in range(100):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(best_h, 1 - best_h, 'full')
        penalties = np.maximum(corr, 1e-6)
        
        # Modify the penalty terms
        penalty_scaling = 0.1
        penalties = penalties * penalty_scaling
        
        # Solve convex subproblem
        new_h = solve_convex(best_h, penalties)
        if new_h is not None:
            new_h = np.maximum(np.minimum(new_h, 1), 0)
            new_h = new_h * (n_points/2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        
        # Update penalties
        penalties = np.maximum(0.9 * penalties, 1e-6)
        
    # Upsample to fine grid
    fine_n_points = 40
    fine_h = np.interp(np.linspace(0, 2, fine_n_points), 
                       np.linspace(0, 2, n_points), best_h)
    fine_h = fine_h * (fine_n_points/2) / np.sum(fine_h)
    dx = 2.0 / fine_n_points
    
    # Fine grid optimization
    for iteration in range(200):
        if time() - start_time > budget_s * 0.95:
            break
            
        corr = np.correlate(fine_h, 1 - fine_h, 'full')
        penalties = np.maximum(corr, 1e-6)
        
        penalty_scaling = 0.2
        penalties = penalties * penalty_scaling
        
        new_h = solve_convex(fine_h, penalties)
        if new_h is not None:
            new_h = np.maximum(np.minimum(new_h, 1), 0)
            new_h = new_h * (fine_n_points/2) / np.sum(new_h)
            new_c5 = evaluate_erdos_solution(new_h, dx)
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
        
        # Apply gradient smoothing
        fine_h = gaussian_filter1d(fine_h, sigma=0.1)
        fine_h = fine_h * (fine_n_points/2) / np.sum(fine_h)
        
    # Final refinement with SQP
    remaining_time = budget_s - (time() - start_time)
    n_points = fine_n_points
    dx = 2.0 / n_points
    
    def obj(h):
        return evaluate_erdos_solution(h, dx)
        
    def con_sum(h):
        return np.sum(h) - n_points/2
        
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': con_sum}
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                      bounds=bounds, constraints=cons, 
                      options={'maxiter': 50, 'ftol': 1e-6})
        if res.success:
            final_h = res.x
            final_c5 = evaluate_erdos_solution(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
