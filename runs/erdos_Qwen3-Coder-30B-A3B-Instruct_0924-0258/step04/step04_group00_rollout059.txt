import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()

    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    elite_size = 10
    tournament_size = 2
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start.copy())
    else:
        population.append(np.ones(n_points) * 0.5)
    
    # Generate random individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points/2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        fitness = evaluate_erdos_solution(h, dx)
        fitnesses.append(fitness)
    
    # GA main loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitnesses[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitnesses[0]
        
        # Select parents using tournament selection
        new_population = []
        # Elite individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament_indices = np.random.choice(len(sorted_population), 
                                                    tournament_size, replace=False)
                tournament_fitnesses = [sorted_fitnesses[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
                parents.append(sorted_population[winner_idx].copy())
            
            # Crossover
            parent1, parent2 = parents
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            for child in [child1, child2]:
                # Apply mutation to child
                for i in range(n_points):
                    if random.random() < mutation_rate:
                        child[i] += np.random.normal(0, 0.01)
                # Repair constraints
                child = np.clip(child, 0, 1)
                child = child * (n_points/2) / np.sum(child)
                new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Re-evaluate fitness
        fitnesses = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitnesses.append(fitness)
    
    # Final refinement with local optimization
    if time() - start_time < budget_s * 0.98:
        # Try local optimization on the best solution found
        try:
            from scipy.optimize import minimize
            h_refined = best_h.copy()
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            cons_sum = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2.0}

            res = minimize(objective, h_refined, method='SLSQP', 
                        bounds=bounds, constraints=cons_sum,

                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
