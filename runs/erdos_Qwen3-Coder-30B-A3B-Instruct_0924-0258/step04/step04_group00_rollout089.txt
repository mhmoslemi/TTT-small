import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary strategy with adaptive mutation
    population_size = 50
    mutation_strength = 0.1
    elite_size = 5
    max_generations = int(budget_s * 1000 / (n_points * population_size)) if n_points > 0 else 1000
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def mutate(h, strength):
        mutated = h + np.random.normal(0, strength, len(h))
        return project_to_feasible(mutated)
    
    def evaluate_individual(h):
        return evaluate_erdos_solution(h, dx)
    
    # Create initial population
    population = [project_to_feasible(h_start + np.random.normal(0, 0.01, n_points)) 
                  for _ in range(population_size)]
    
    for generation in range(max_generations):
        if time() - start_time > budget_s * 0.95:
            break
        
        # Evaluate all individuals
        fitness = [evaluate_individual(ind) for ind in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Create new generation
        new_population = population[:elite_size]  # Elitism
        
        # Mutation and selection
        for _ in range(population_size - elite_size):
            # Tournament selection
            tournament_size = 5
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            
            # Mutate the winner
            child = mutate(population[winner_index], mutation_strength)
            new_population.append(child)
        
        population = new_population
        
        # Adaptive mutation strength
        if generation % 10 == 0:
            diversity = np.std([np.std(ind) for ind in population[:10]])
            if diversity < 0.01:
                mutation_strength *= 1.1
            else:
                mutation_strength *= 0.98
    
    # Final validation and refinement
    try:
        h_final = project_to_feasible(best_h)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    return (list(best_h), best_c5, n_points)
