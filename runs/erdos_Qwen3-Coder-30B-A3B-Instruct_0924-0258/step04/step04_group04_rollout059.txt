import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create an initial guess with triangulated structure
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Make a triangular-shaped function that's normalized to sum to n_points/2
        mid = n_points // 2
        for i in range(n_points):
            dist = abs(i - mid)
            h_start[i] = max(0, 1 - dist / (n_points/3))
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Steepest descent for maximum overlap shift
    max_iter = 1000
    step_size = 0.01
    iter_since_improvement = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_shift_idx = np.argmax(corr)
        # The actual shift is from the center
        shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient for shift k (additive with proper indexing)
        # gradient wrt h[i] is (1 - h[i+k]) if k >= 0, (1 - h[i-k]) if k < 0
        gradient = np.zeros(n_points)
        if shift >= 0:
            for i in range(n_points):
                if i + shift < n_points:
                    gradient[i] = 1 - best_h[i + shift]
        else:
            for i in range(n_points):
                if i - shift >= 0:
                    gradient[i] = 1 - best_h[i - shift]
                    
        # Handle special case where shift = 0:
        if shift == 0:
            for i in range(n_points):
                gradient[i] = 1 - 2 * best_h[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-10:
            gradient = gradient / grad_norm

        # Update h in gradient direction
        new_h = best_h - step_size * gradient
        
        # Clip to feasible region [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Re-normalize
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Reduce step size if no improvement for a while
        if iter_since_improvement >= 10:
            step_size *= 0.95
            
        # Early exit if no significant improvement
        if iter_since_improvement >= max_no_improvement:
            break
            
    # Final refinement with optimization
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        from scipy.optimize import minimize
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
