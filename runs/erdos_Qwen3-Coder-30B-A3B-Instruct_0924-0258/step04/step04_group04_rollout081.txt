import numpy as np
from scipy.signal import correlate
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient descent parameters
    max_iter = 1000
    step_size = 0.01
    tolerance = 1e-6
    
    # Keep track of last few iterations for early stopping
    recent_c5 = []
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute the full correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        # The correlation result length is 2*n_points - 1
        # The maximum overlap is at the middle (n_points-1 index)
        shift_indices = np.arange(2 * n_points - 1)
        k_max_index = np.argmax(corr)
        k_max = shift_indices[k_max_index] - (n_points - 1)
        max_overlap = corr[k_max_index] * dx
        
        # Compute gradient of overlap w.r.t. h
        # For shift k, ∂/∂h_i overlap = (1 - h_{i+k})  if k ≠ 0
        #                               (1 - h_i - h_i) = 1 - 2*h_i if k = 0
        grad = np.zeros(n_points)
        
        # Handle difference case
        if k_max != 0:
            # Correlation with (1 - h) starts at index 0 but gets 
            # shifted by k_max to match pattern: h[i] * (1 - h[i+k_max])
            # So we're looking at h[i] * (1 - h[i+k_max])
            # Gradient contribution: -(1 - h[i+k_max]) when i+k_max in [0,n_points-1]
            
            for i in range(n_points):
                if 0 <= i + k_max < n_points:
                    grad[i] -= (1 - best_h[i + k_max])
        else:
            # This is when k_max = 0
            for i in range(n_points):
                grad[i] = -(1 - 2 * best_h[i])  # Derivative of h[i] * (1 - h[i])
        
        # Apply step in negative gradient direction
        h_new = best_h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Normalize to the correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        
        # Check for convergence
        recent_c5.append(best_c5)
        if len(recent_c5) > 5:
            recent_c5.pop(0)
        if len(recent_c5) == 5 and np.all(np.abs(np.array(recent_c5) - recent_c5[0]) < tolerance):
            break
    
    # Final gradient step if time allows
    if time() - start_time < budget_s - 10:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        shift_indices = np.arange(2 * n_points - 1)
        k_max_index = np.argmax(corr)
        k_max = shift_indices[k_max_index] - (n_points - 1)
        max_overlap = corr[k_max_index] * dx
        
        grad = np.zeros(n_points)
        if k_max != 0:
            for i in range(n_points):
                if 0 <= i + k_max < n_points:
                    grad[i] -= (1 - best_h[i + k_max])
        else:
            for i in range(n_points):
                grad[i] = -(1 - 2 * best_h[i])
        
        h_final = best_h - step_size * grad
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        final_c5 = evaluate_erdos_solution(h_final, dx)
        if final_c5 < best_c5:
            best_h = h_final
            best_c5 = final_c5
    
    return (list(best_h), best_c5, n_points)
