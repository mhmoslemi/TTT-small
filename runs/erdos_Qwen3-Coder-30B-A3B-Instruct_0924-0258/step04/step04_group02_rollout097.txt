import numpy as np
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 35
        h = np.ones(n_points) * 0.5
        # Add some random perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Simulated annealing parameters
    T = 1.0
    alpha = 0.995
    max_iterations = int(1e5)
    
    # Track iterations
    for i in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate candidate by perturbing two random points
        i1, i2 = np.random.choice(n_points, 2, replace=False)
        
        # Ensure valid range with small changes
        delta_max = min(0.1, T)
        delta = np.random.uniform(-delta_max, delta_max)
        
        # Adjust h values
        h_new = h.copy()
        h_new[i1] += delta
        h_new[i2] -= delta
        
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        diff = np.sum(h_new) - n_points / 2
        h_new -= diff / n_points
        
        # Ensure boundary constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new objective
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        # Accept with simulated annealing probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        T *= alpha
    
    return (list(best_h), best_c5, n_points)
