import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a more refined initial construction
        h_start[5:15] = 0.5
        h_start[20:30] = 0.5
        # Normalize to ensure sum is correct
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Gradient-based iterative optimization
    max_iter = 1000
    step_size = 0.01
    patience = 100
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_shift_idx = np.argmax(corr)
        max_overlap = corr[max_shift_idx] * dx
        
        # Identify which shift k causes the max overlap
        # The correlation has length 2*n_points - 1
        # Shift k=0 corresponds to index n_points-1
        # Shift k=2 corresponds to index n_points+1
        # In general shift k corresponds to index n_points-1+k 
        # For shifts k in [-n_points+1, n_points-1]
        # This gives us k = index - (n_points-1)
        k_max = max_shift_idx - (n_points - 1)
        
        # Compute gradient of overlap with respect to h at shift k_max
        # Overlap = sum h(x)(1-h(x+k)) dx
        # Gradient wrt h[i] = (1 - h[i+k]) - h[i] (assuming k > 0)
        # For the case that k_max is negative, we shift elements forward
        # We can approximate gradient as:
        #   if k_max > 0: gradient[i] = 1 - h[i+k_max]
        #   if k_max < 0: gradient[i] = 1 - h[i-k_max]
        # Note: if k_max = 0, we manually check the gradient
        grad = np.zeros(n_points)
        
        if k_max > 0:
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] = 1 - best_h[i + k_max]
                else:
                    grad[i] = 0
        elif k_max < 0:
            for i in range(n_points):
                if i - k_max >= 0:
                    grad[i] = 1 - best_h[i - k_max]
                else:
                    grad[i] = 0
        else:
            # Gradient for k=0 case: ∂/∂h_i [h_i (1 - h_i)] = 1 - 2*h_i 
            grad = 1 - 2 * best_h
        
        # Update h in direction of negative gradient
        new_h = best_h - step_size * grad
        
        # Project to feasible region
        new_h = np.clip(new_h, 0, 1)
        # Ensure sum constraint
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Check if new solution is better
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            # Reduce step size if we make progress
            step_size *= 0.99
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Increase step size if we haven't improved recently
            step_size = min(step_size * 1.01, 0.1)
            
        # Terminate if we haven't improved in a while
        if no_improve_count > patience:
            break
    
    # Final refinement with SQP to ensure quality
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
