import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # More structured initial with two peaks
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two peaks with different heights
        peak1_pos = 0.6
        peak2_pos = 1.6
        width = 0.3
        for i in range(n_points):
            x = i * dx
            if abs(x - peak1_pos) < width:
                h_start[i] = 0.8 * np.exp(-0.5*((x - peak1_pos)/width)**2)
            elif abs(x - peak2_pos) < width:
                h_start[i] = 1.2 * np.exp(-0.5*((x - peak2_pos)/width)**2)
        # Normalize to sum = n_points/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Parameters
    max_iter = 100
    dual_update_prob = 0.8
    dual_update_batch = max(1, n_points // 10)
    dual_slack = 1e-6
    dual_bound = 100.0

    # Dual variables
    lambda_k = np.ones(2 * n_points - 1)  # Full correlation length
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        current_maximum = np.max(corr)
        
        # Update lambda with simulated annealing-like schedule
        if np.random.rand() < dual_update_prob:
            # Select random subset to update
            indices = np.random.choice(len(lambda_k), dual_update_batch, replace=False)
            for k in indices:
                # Project lambda to feasible range
                lambda_k[k] = np.clip(lambda_k[k] * (1.0 + 0.01 * (corr[k] - lambda_k[k])), 
                                    dual_slack, dual_bound)
                if np.isnan(lambda_k[k]) or np.isinf(lambda_k[k]):
                    lambda_k[k] = 1.0
        
        # Create and solve convex program using CVXPY
        h_var = cp.Variable(n_points)
        
        # Objective: minimize weighted sum of overlaps
        obj = cp.Minimize(cp.sum(cp.multiply(lambda_k, corr)))
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points/2,
            h_var >= 0,
            h_var <= 1
        ]
        
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=50)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    # Re-normalize
                    new_h = new_h * (n_points/2) / np.sum(new_h)
                    
                    # Compute new overlap
                    new_c5 = evaluate_erdos_solution(new_h, dx)
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
        except:
            pass  # Continue with previous best if optimization fails
            
    # Final refinement using scipy
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
