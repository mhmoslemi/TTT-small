import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 35
        initial_h = np.ones(n_points) * 0.5
    
    # Population parameters
    population_size = 40
    mutation_rate = 0.15
    crossover_prob = 0.7
    tabu_limit = 50
    stagnation_limit = 50
    
    # Generate initial population
    population = []
    fitness = []
    tabu_list = set()
    
    # Seed with initial solution and variations
    for i in range(10):
        h = initial_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h.copy())
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Add random solutions
    for i in range(5):
        h = np.random.rand(n_points)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h.copy())
        fit = evaluate_erdos_solution(h, 2.0 / n_points)
        fitness.append(fit)
    
    # Current best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Store tabu solutions
    tabu_list.add(tuple(best_h.round(8)))
    
    gen = 0
    stagnation_count = 0
    target_c5 = 0.3807
    
    while time() - start_time < budget_s * 0.95 and best_c5 > target_c5:
        gen += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitness[i] for i in sorted_indices]
        
        # Keep top 50% for next generation
        elite_count = population_size // 2
        selected_pop = sorted_pop[:elite_count]
        selected_fit = sorted_fit[:elite_count]
        
        # Generate offspring
        new_population = selected_pop.copy()
        new_fitness = selected_fit.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            if np.random.random() < crossover_prob and len(selected_pop) >= 2:
                # Tournament selection
                idx1, idx2 = np.random.choice(len(selected_pop), 2, replace=False)
                parent1 = selected_pop[idx1]
                parent2 = selected_pop[idx2]
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.random() < mutation_rate:
                    mut_idx = np.random.randint(n_points)
                    child1[mut_idx] = np.clip(child1[mut_idx] + np.random.normal(0, 0.05), 0, 1)
                
                if np.random.random() < mutation_rate:
                    mut_idx = np.random.randint(n_points)
                    child2[mut_idx] = np.clip(child2[mut_idx] + np.random.normal(0, 0.05), 0, 1)
                
                # Repair and normalize
                child1 = child1 * (n_points / 2.0) / np.sum(child1)
                child2 = child2 * (n_points / 2.0) / np.sum(child2)
                
                # Check tabu
                key1 = tuple(child1.round(8))
                key2 = tuple(child2.round(8))
                
                if key1 not in tabu_list:
                    new_population.append(child1)
                    new_fitness.append(evaluate_erdos_solution(child1, 2.0 / n_points))
                    tabu_list.add(key1)
                    if len(tabu_list) > tabu_limit:
                        tabu_list.pop()
                
                if len(new_population) < population_size and key2 not in tabu_list:
                    new_population.append(child2)
                    new_fitness.append(evaluate_erdos_solution(child2, 2.0 / n_points))
                    tabu_list.add(key2)
                    if len(tabu_list) > tabu_limit:
                        tabu_list.pop()
            
            # If not enough offspring, add random individuals
            if len(new_population) < population_size:
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0) / np.sum(h)
                new_population.append(h)
                new_fitness.append(evaluate_erdos_solution(h, 2.0 / n_points))
        
        # Replace population with new generation
        population = new_population[:population_size]
        fitness = new_fitness[:population_size]
        
        # Find best in this generation
        current_best_idx = np.argmin(fitness)
        current_best_h = population[current_best_idx]
        current_best_c5 = fitness[current_best_idx]
        
        # Update global best
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive parameter adjustment
        if stagnation_count > stagnation_limit:
            mutation_rate = min(0.5, mutation_rate + 0.1)
            crossover_prob = max(0.3, crossover_prob - 0.1)
            tabu_limit = max(10, tabu_limit // 2)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate - 0.05)
            crossover_prob = min(0.9, crossover_prob + 0.05)
            
        # Refinement with local optimization every 10 generations
        if gen % 10 == 0:
            # Select top solutions for refinement
            top_indices = np.argsort(fitness)[:5]
            for idx in top_indices:
                h_init = population[idx]
                dx = 2.0 / n_points
                
                def obj(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                bounds = [(0, 1) for _ in range(n_points)]
                
                try:
                    res = minimize(obj, h_init, method='SLSQP', 
                                   bounds=bounds, constraints=cons_sum, 
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        refined_h = res.x
                        refined_c5 = obj(refined_h)
                        if refined_c5 < fitness[idx]:
                            population[idx] = refined_h
                            fitness[idx] = refined_c5
                except:
                    pass
                
                # Update best if refined solution is better
                if fitness[idx] < best_c5:
                    best_c5 = fitness[idx]
                    best_h = population[idx].copy()
                    
        # Early exit if target achieved
        if best_c5 <= target_c5:
            break
            
    # Final refinement
    dx = 2.0 / n_points
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        res = minimize(obj, best_h, method='SLSQP', 
                       bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 200, 'ftol': 1e-9})
        if res.success:
            final_h = res.x
            final_c5 = obj(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
