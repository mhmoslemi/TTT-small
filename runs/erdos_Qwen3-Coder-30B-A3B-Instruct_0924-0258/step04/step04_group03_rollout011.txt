import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Evolutionary Strategy with adaptive mutation
    population_size = 50
    generations = 1000
    elite_ratio = 0.2
    mutation_rate = 0.05
    
    population = []
    for i in range(population_size):
        if i == 0:
            h = h_start.copy()
        else:
            h = h_start + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution found
    best_h = best_h.copy()
    best_c5 = best_c5
    
    # Number of generations
    for gen in range(generations):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Evaluate population
        fitness_scores = []
        for h in population:
            fitness = evaluate_erdos_solution(h, dx)
            fitness_scores.append(fitness)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
            
        # Select elite
        elite_count = max(1, int(elite_ratio * population_size))
        elite = sorted_population[:elite_count]
        
        # Generate new population
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            # Tournament selection for parents
            parent1 = elite[np.random.randint(0, len(elite))]
            parent2 = elite[np.random.randint(0, len(elite))]
            
            # Crossover (arithmetic)
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            mutation_step = max(0.001, mutation_rate * (1.0 - gen / generations))
            noise = np.random.normal(0, mutation_step, n_points)
            child = child + noise
            
            # Ensure bounds
            child = np.clip(child, 0, 1)
            # Normalize to fix sum constraint
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Occasionally refine using gradient-based optimization
        if gen % 20 == 0 and (time() - start_time) < budget_s * 0.9:
            top_candidates = new_population[:5]
            for h_refine in top_candidates:
                if (time() - start_time) > budget_s * 0.95:
                    break
                def objective(h):
                    return evaluate_erdos_solution(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h_refine, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 50, 'ftol': 1e-8})
                    if res.success:
                        h_final = res.x
                        c5_final = objective(h_final)
                        if c5_final < best_c5:
                            best_h = h_final
                            best_c5 = c5_final
                except Exception:
                    pass
    
    # Final refinement of the best solution
    if (time() - start_time) < budget_s * 0.1:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = objective(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
