import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, 2.0/n_points)
    
    # Use BOBYQA for local optimization
    # Define constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint dictionary
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # BOBYQA parameters
    max_iter = 500
    trust_radius = 0.5
    min_trust_radius = 1e-5
    patience = 20
    
    # Track best
    patience_counter = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Generate a new candidate via perturbation
        h_candidate = h_start + np.random.normal(0, 0.01, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        # Normalize sum
        h_candidate = h_candidate * (n_points/2) / np.sum(h_candidate)
        
        # Use BOBYQA (or fallback to SLSQP) to refine
        try:
            # For simplicity, we'll use SLSQP -- practically more efficient
            result = minimize(
                lambda h: evaluate_erdos_solution(h, 2.0/n_points),
                h_candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_refined = result.x
                c5_refined = evaluate_erdos_solution(h_refined, 2.0/n_points)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    patience_counter = 0
                    last_improvement = iteration
                else:
                    patience_counter += 1
            else:
                continue
                
        except Exception as e:
            # If optimization fails, skip this iteration
            continue
            
        # Early stopping on no improvement
        if patience_counter > patience:
            break
    
    # Final cleanup
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points/2) / np.sum(final_h)
    
    return (list(final_h), best_c5, n_points)
