import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Local search with random restarts and gradient-based refinement
    max_restarts = 100
    restart_count = 0
    max_iter = 200
    iter_count = 0
    
    while time() - start_time < budget_s * 0.95 and iter_count < max_iter:
        # Random perturbation
        h_restart = best_h + np.random.normal(0, 0.03, n_points)
        h_restart = np.clip(h_restart, 0, 1)
        h_restart = h_restart * (n_points/2) / np.sum(h_restart)
        
        # SLSQP local optimization
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(objective, h_restart, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
            else:
                restart_count += 1
        except Exception:
            restart_count += 1
            
        # Mix of gradient-free and local refinement
        if iter_count % 10 == 0 and time() - start_time < budget_s * 0.8:
            # Binary relaxation approach with scaling
            h_binary = (best_h > 0.5).astype(float)
            h_binary = h_binary * (n_points/2) / np.sum(h_binary)
            # Small perturbations around binary solution
            h_pert = h_binary + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points/2) / np.sum(h_pert)
            c5_pert = evaluate_erdos_solution(h_pert, dx)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert
            
        iter_count += 1
        if restart_count > max_restarts:
            break
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
