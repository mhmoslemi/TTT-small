import numpy as np
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a more structured initial guess
        h_start[5:15] = 0.5
        h_start[20:30] = 0.5
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Stochastic gradient descent with maximum overlap tracking
    max_iter = 1000
    step_size = 0.01
    patience = 100
    
    last_improve = 0
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap for all shifts
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for max_k shift
        grad = np.zeros(n_points)
        if max_k == 0:
            # Gradient for shift 0: (1 - h[i+0]) - h[i] 
            for i in range(n_points):
                grad[i] = 1 - best_h[i] - best_h[i]
        else:
            # For non-zero shifts
            for i in range(n_points):
                if i + max_k >= 0 and i + max_k < n_points:
                    grad[i] = 1 - best_h[i + max_k]
                else:
                    grad[i] = 1
            
        # Apply gradient step
        new_h = best_h - step_size * grad
        
        # Project to feasible region
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points/2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept if better or with some probability if worse (simulated annealing)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            last_improve = iteration
        elif np.random.rand() < 0.01:  # Some chance of accepting worse solutions
            best_h = new_h.copy()
            best_c5 = new_c5
            last_improve = iteration
            
        # Early stopping if no improvement in a while
        if iteration - last_improve > patience:
            break
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Try a simpler optimization approach locally
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            from scipy.optimize import minimize
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
