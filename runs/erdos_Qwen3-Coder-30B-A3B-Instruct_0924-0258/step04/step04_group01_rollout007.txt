import numpy as np
from cvxpy import *
import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a step-like construction
        step1 = int(0.4 / dx)
        step2 = int(1.6 / dx)
        h_start[:step1] = 1.0
        h_start[step1:step2] = 0.0
        h_start[step2:] = 1.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary search on C5
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iterations = 60
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        mid = (low + high) / 2.0
        
        # LP formulation
        h_vars = Variable(n_points)
        
        # Constraint: sum(h) = n_points/2
        sum_constraint = [sum(h_vars) == n_points / 2]
        
        # Bounds: 0 <= h[i] <= 1
        bound_constraints = [h_vars >= 0, h_vars <= 1]
        
        # Overlap constraints for all lags
        constraints = sum_constraint + bound_constraints
        
        # For each lag k, create the overlap constraint
        for k in range(2 * n_points - 1):
            # Indices for h[a] and h[a+k]
            if k < n_points:
                # h[a] * (1 - h[a + k])
                overlap = 0
                for a in range(n_points - k):
                    overlap += h_vars[a] * (1 - h_vars[a + k])
            else:
                # Indices need to be adjusted for overlap constraint
                overlap = 0
                for a in range(k - n_points + 1, n_points):
                    overlap += h_vars[a] * (1 - h_vars[a + k - n_points + 1])
            
            constraints.append(overlap <= mid / dx)
        
        # Solve LP feasibility problem
        prob = Problem(Minimize(0), constraints)
        
        try:
            prob.solve(solver=SCIP, verbose=False, maxTime=budget_s * 0.95 - (time.time() - start_time))
            if prob.status in ["optimal", "optimal_inaccurate"]:
                # Extract solution
                h_sol = h_vars.value
                if h_sol is not None:
                    # Check if this solution is better
                    c5_sol = evaluate_erdos_solution(h_sol, dx)
                    if c5_sol < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = c5_sol
                # Update binary search bounds
                high = mid
            else:
                # Infeasible
                low = mid
        except:
            low = mid
        
        # Break if we've reached precision
        if high - low < precision:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
