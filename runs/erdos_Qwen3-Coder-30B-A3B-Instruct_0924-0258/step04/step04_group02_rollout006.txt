import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert to logistic space for easier constraint handling
    def h_from_z(z, b):
        return 1.0 / (1.0 + np.exp(-z - b))
    
    def z_from_h(h, b):
        return np.log(h / (1 - h)) - b
    
    def adjust_b(z, target_sum):
        # Binary search for correct b to satisfy sum(h) = target_sum
        low, high = -10, 10
        for _ in range(50):
            b = (low + high) / 2
            h = h_from_z(z, b)
            current_sum = np.sum(h)
            if current_sum < target_sum:
                high = b
            else:
                low = b
        b = (low + high) / 2
        return b
    
    # Initialize in z space
    z = z_from_h(h_start, 0)
    b = adjust_b(z, n_points / 2)
    
    # Line search parameters
    alpha = 0.01
    beta = 0.5
    max_linesearch = 10
    
    # Optimization loop
    iteration = 0
    max_iterations = 2000
    
    while iteration < max_iterations and (time() - start_time) < budget_s * 0.95:
        # Compute all overlaps for each shift
        h = h_from_z(z, b)
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr * dx
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Check if we've improved
        if max_overlap < best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
        
        # Find gradient w.r.t. h for the worst shift
        # Compute partial derivatives of correlation w.r.t. h
        # This is simplified - we compute gradient for just k_max
        # For shift k_max, contribution to gradient is: d/dh_i (h_i * (1 - h_{i+k_max}))
        # = 1 - h_{i+k_max} if i is in domain of h, 0 otherwise
        
        # Compute the gradient at k_max
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad_h[i] = 1 - h[i + k_max]
        
        # Convert gradient to z-space
        # dh/dz = h * (1 - h)
        h_val = h_from_z(z, b)
        dh_dz = h_val * (1 - h_val)
        # Gradient in z space: dz/dh * dh/dz
        grad_z = grad_h / (dh_dz + 1e-10)  # Avoid division by zero
        
        # Line search
        step_size = alpha
        for _ in range(max_linesearch):
            z_new = z - step_size * grad_z
            b_new = adjust_b(z_new, n_points / 2)
            h_new = h_from_z(z_new, b_new)
            new_c5 = evaluate_erdos_solution(h_new, dx)
            
            if new_c5 <= max_overlap + 1e-10 or step_size < 1e-6:
                break
            step_size *= beta
        
        # Update
        z = z_new
        b = b_new
        
        iteration += 1
    
    # Final check
    h_final = h_from_z(z, b)
    final_c5 = evaluate_erdos_solution(h_final, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_final.copy()
    
    # Refine final using scipy
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        # Convert to float array
        h_init = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, h_init, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200})
            if res.success:
                h_refined = res.x
                c5_refined = objective(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
