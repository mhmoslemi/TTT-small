import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Combined approach: hybrid with adaptive perturbations and convex refinement
    def adaptive_neighbor(h, temperature):
        # Perturb two random indices, adjust to maintain sum
        h_new = h.copy()
        i, j = np.random.choice(n_points, size=2, replace=False)
        step = 0.05 * temperature
        delta = np.random.uniform(-step, step)
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0, 1)
        h_new[j] = np.clip(h_new[j], 0, 1)
        
        # Adjust to maintain sum constraint (exact)
        sum_diff = np.sum(h_new) - n_points / 2.0
        if abs(sum_diff) > 1e-10:
            h_new[0] -= sum_diff
            h_new[0] = np.clip(h_new[0], 0, 1)
        
        return h_new

    # Simulated Annealing parameters
    current_h = h_start.copy()
    current_c5 = best_c5
    temperature = 1.0
    min_temp = 1e-6
    alpha = 0.9995
    max_steps = 10000
    
    # Track best solution
    best_h = current_h.copy()
    best_c5 = current_c5
    
    step_count = 0
    while temperature > min_temp and (time() - start_time) < budget_s * 0.95:
        # Generate neighbor
        new_h = adaptive_neighbor(current_h, temperature)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(new_h, dx)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
            # Check if it's better than best
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        
        # Cool down
        temperature *= alpha
        step_count += 1
        
        # Periodically enrich with convex optimization for tightening
        if step_count % 200 == 0:
            # Convex optimization to refine
            try:
                h = cp.Variable(n_points)
                h_sum = cp.sum(h)
                h_bounds = [h >= 0, h <= 1]
                h_sum_constraint = cp.Equality(h_sum, n_points / 2.0)
                
                # Empirical constraint - compute a soft constraint on max overlap 
                # based on current best
                overlap_bound = best_c5
                
                # Create convex constraints for a range of indices with overlapping terms
                constraints = h_bounds + [h_sum_constraint]
                for i in range(n_points):
                    for k in range(1, min(n_points - i, n_points - 1) + 1):
                        # consider approximating: sum_{j=0}^{n_points-k} h[i+j]* (1-h[i+j+k])
                        # can compute it with reshaping, or submatrices
                        if i + k < n_points:
                            corr = cp.sum(cp.multiply(h[i:i+k], 1 - h[i+k:i+2*k]))
                        else:
                            corr = cp.sum(cp.multiply(h[i:], 1 - h[:n_points-(i+k)]))
                        constraints.append(corr <= overlap_bound * dx * 2)  # Prior constraint on refinement
                        
                prob = cp.Problem(cp.Minimize(0), constraints)
                
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass

    # Final gradient-based refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Multiple restarts
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x 
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass

    return (list(best_h), best_c5, n_points)
