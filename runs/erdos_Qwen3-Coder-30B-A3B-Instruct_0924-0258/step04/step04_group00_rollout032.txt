import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h_start)

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # GA parameters
    population_size = 100
    num_offspring = 20
    mutation_rate = 0.1
    tournament_size = 2
    max_generations = 10000
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def normalize(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start.copy()]
    else:
        population = []
    
    while len(population) < population_size:
        individual = create_individual()
        population.append(individual)
    
    def evaluate_fitness(h):
        return evaluate_erdos_solution(h, dx)
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(2):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx])
        return selected
    
    def crossover(parent1, parent2):
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return normalize(child1), normalize(child2)
    
    def mutate(individual):
        mutated = individual.copy()
        num_mutation = int(mutation_rate * n_points)
        indices = np.random.choice(n_points, num_mutation, replace=False)
        for i in indices:
            mutated[i] += np.random.normal(0, 0.01)
        return normalize(mutated)
    
    # Main GA loop
    generation = 0
    while time() - start_time < budget_s * 0.8 and generation < max_generations:
        fitnesses = [evaluate_fitness(ind) for ind in population]
        best_in_generation = min(fitnesses)
        if best_in_generation < best_c5:
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_c5 = best_in_generation
        
        # Selection
        selected_parents = []
        for _ in range(num_offspring):
            parents = tournament_selection(population, fitnesses)
            selected_parents.extend(parents)
        
        # Crossover
        offspring = []
        for i in range(0, len(selected_parents), 2):
            if i + 1 < len(selected_parents):
                child1, child2 = crossover(selected_parents[i], selected_parents[i+1])
                offspring.extend([child1, child2])
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.random() < 0.3:  # 30% chance of mutation
                offspring[i] = mutate(offspring[i])
        
        # Replacement: keep best individuals from previous population
        mixed_population = population + offspring
        mixed_fitnesses = [evaluate_fitness(ind) for ind in mixed_population]
        
        # Sort by fitness and keep best
        sorted_indices = np.argsort(mixed_fitnesses)
        population = [mixed_population[i] for i in sorted_indices[:population_size]]
        
        generation += 1
    
    # Final refinement with local search
    if time() - start_time < budget_s * 0.95:
        try:
            h_final = best_h.copy()
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points/2) / np.sum(h_final)
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
