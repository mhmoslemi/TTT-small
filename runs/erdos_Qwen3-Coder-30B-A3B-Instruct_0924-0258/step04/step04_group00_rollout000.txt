import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Multi-resolution approach with refinement
    resolutions = [35, 40, 50]
    current_n = n_points
    
    # Function to project h to valid domain
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (current_n / 2) / np.sum(h)
        return h
    
    # Local search using Nelder-Mead with regular restarts
    def local_search(h0, maxiter=100):
        start_h = project_h(h0)
        try:
            # Use a simple local search with restarts
            # For high-dim problems, it's effective to restart often
            for _ in range(10):
                res = minimize(evaluate_erdos_solution, 
                               start_h, 
                               args=(dx,), 
                               method='Nelder-Mead', 
                               options={'maxiter': maxiter, 'xatol': 1e-6, 'fatol': 1e-6})
                if res.success:
                    h_test = project_h(res.x)
                    c5_test = evaluate_erdos_solution(h_test, dx)
                    if c5_test < best_c5:
                        return h_test, c5_test
                # Random restart
                start_h = np.random.uniform(0, 1, current_n)
                start_h = project_h(start_h)
            return start_h, evaluate_erdos_solution(start_h, dx)
        except Exception:
            return start_h, evaluate_erdos_solution(start_h, dx)
    
    # Use convex relaxation when possible for better constraints
    def cvx_relaxation(h0, c5_bound):
        try:
            h_var = cp.Variable(current_n)
            constraints = [
                cp.sum(h_var) == current_n / 2.0,
                h_var >= 0,
                h_var <= 1
            ]
            # Add a few key constraints from h0 to constrain search space
            # Correlations for a few shifts
            for k in range(1, min(20, current_n)):  # Limited number for efficiency
                if k < current_n:
                    corr = cp.sum(cp.multiply(h_var[:current_n-k], 1 - h_var[k:]))
                else:
                    corr = cp.sum(cp.multiply(h_var[k-current_n:], 1 - h_var[:current_n-(k-current_n)]))
                constraints.append(corr <= c5_bound * dx + 1e-4)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_feasible = h_var.value
                return project_h(h_feasible)
        except Exception:
            pass
        return h0

    # Main loop of multi-resolution optimization
    all_h = [best_h.copy()]
    for n in resolutions:
        if n <= current_n:
            continue
        current_n = n
        dx = 2.0 / current_n
        # Extend h_start to higher resolution
        h_extended = np.interp(np.linspace(0, 2, current_n),
                               np.linspace(0, 2, len(best_h)),
                               best_h)
        # Add a bit of noise
        h_extended += np.random.normal(0, 0.01, current_n)
        h_extended = project_h(h_extended)
        best_h = h_extended
        best_c5 = evaluate_erdos_solution(h_extended, dx)
        
        # Local optimization
        for iter_ in range(20):
            if time() - start_time > budget_s * 0.95:
                break
            h_trial, c5_trial = local_search(best_h, 50)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                
            # Occasionally apply convex relaxation
            if iter_ % 5 == 0 and time() - start_time < budget_s * 0.8:
                h_cvx = cvx_relaxation(best_h, best_c5)
                if h_cvx is not None:
                    c5_cvx = evaluate_erdos_solution(h_cvx, dx)
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
        
        if time() - start_time > budget_s * 0.95:
            break
    
    # Final clean up and additional refinement if time
    while time() - start_time < budget_s * 0.98:
        h_trial, c5_trial = local_search(best_h, 100)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
        else:
            break
    
    # Final projection
    best_h = project_h(best_h)
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (list(best_h), final_c5, current_n)
