import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PGD with augmented Lagrangian approach
    max_iter = 2000
    alpha = 0.01
    beta = 0.99  # for diminishing step size
    tolerance = 1e-6
    stagnation_counter = 0
    max_stagnation = 50
    
    h_current = best_h.copy()
    lagrange_mult = 0.0
    penalty_param = 1.0
    
    # Precompute all possible correlations
    def compute_correlations(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr
    
    def compute_gradient(h):
        # This is a simplified gradient computation
        # In a real implementation, we'd more carefully compute gradients for
        # the max operation over all k.
        # For now, we use a heuristic approach based on the current max overlap
        correlations = compute_correlations(h)
        max_idx = np.argmax(correlations)
        max_k = max_idx - (len(h) - 1)
        grad = np.zeros_like(h)
        # Compute gradient with respect to one specific maximizer
        # This is an approximation - real gradient is more complex 
        # (involves subgradients at the max)
        if max_k >= 0:
            for i in range(len(h) - max_k):
                grad[i] += (1 - h[i + max_k])
                grad[i + max_k] -= h[i]
        else:
            max_k = -max_k
            for i in range(len(h) - max_k):
                grad[i] += (1 - h[i + max_k])
                grad[i + max_k] -= h[i]
        return grad * dx
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient and value
        c5_current = evaluate_erdos_solution(h_current, dx)
        grad = compute_gradient(h_current)
        
        # Augmented Lagrangian step
        # Constraint: sum(h) == n_points / 2  
        constraint_violation = np.sum(h_current) - n_points / 2.0
        lagrange_mult += penalty_param * constraint_violation
        
        # Compute actual gradient including Lagrange term
        grad += lagrange_mult
        
        # Update h
        h_new = h_current - alpha * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Adaptive step size
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < c5_current:
            alpha *= 1.2
            h_current = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                stagnation_counter = 0
        else:
            alpha *= 0.5
            stagnation_counter += 1
            
        # Lower the penalty param gradually if we're converging
        if abs(c5_current - c5_new) < tolerance:
            penalty_param *= 0.999
        
        # Early termination if stagnation
        if stagnation_counter >= max_stagnation:
            break
            
        # Adjustment for step size
        alpha = max(alpha, 1e-6)
    
    # Final refinement if necessary
    try:
        h_refined = best_h.copy()
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2.0) / np.sum(h_refined)
        c5_final = evaluate_erdos_solution(h_refined, dx)
        if c5_final < best_c5:
            best_h = h_refined
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
