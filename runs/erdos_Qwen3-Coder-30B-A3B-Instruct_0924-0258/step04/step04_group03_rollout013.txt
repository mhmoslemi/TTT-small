import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based optimization with evolutionary approach
    population_size = 50
    mutation_step = 0.05
    selection_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h_start)
        for _ in range(population_size - 1):
            h_pert = h_start + np.random.normal(0, 0.02, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert *= (n_points / 2) / np.sum(h_pert)
            population.append(h_pert)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h *= (n_points / 2) / np.sum(h)
            population.append(h)
    
    for generation in range(max_generations):
        if (time() - start_time) > budget_s * 0.95:
            break
            
        # Evaluate all population members
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 20%
        top_count = max(1, int(population_size * selection_rate))
        selected = population[:top_count]
        best_in_generation = fitness[0]
        if best_in_generation < best_c5:
            best_c5 = best_in_generation
            best_h = population[0].copy()
            
        # Generate new population
        new_population = selected[:]
        
        # Mutation and crossover
        while len(new_population) < population_size:
            # Selection for crossover
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover - arithmetic
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutation
            mutation_magnitude = np.random.normal(0, mutation_step)
            child += mutation_magnitude
            child = np.clip(child, 0, 1)
            child *= (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
        
        # Reduce mutation step
        if mutation_step > 0.001:
            mutation_step *= 0.995
    
    # Gradient refinement of the best solutions
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        top_candidates = []
        # Collect top 5 candidates
        for h in population[:5]:
            top_candidates.append(h.copy())
            
        for h_candidate in top_candidates:
            if time() - start_time > budget_s * 0.95:
                break
                
            def objective(h):
                return evaluate_erdos_solution(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            # Run refinement with SLSQP
            try:
                res = minimize(objective, h_candidate, method='SLSQP',
                               bounds=bounds, constraints=cons_sum,
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
