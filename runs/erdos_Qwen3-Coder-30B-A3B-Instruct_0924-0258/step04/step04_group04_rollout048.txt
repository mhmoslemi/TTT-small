import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()

    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        h_start[:n_points//2] = 1.0
        h_start[n_points//2:] = 0.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Parameters
    max_iter = 300
    penalty_lr = 0.1
    smoothing_sigma = 0.1
    l1_penalty = True
    
    # Optimization loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update penalties based on current solution
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_corr = np.max(corr)
        
        # Sample a few key shifts for penalty updates
        shifts = range(0, len(corr), max(1, len(corr) // 20))
        penalty_updates = []
        for k in shifts:
            penalty_updates.append(corr[k] / (max_corr + 1e-8))
        
        if len(penalty_updates) > 0:
            avg_update = np.mean(penalty_updates)
            penalty_lr = max(0.01, penalty_lr * 0.995)
            
        # Smoothing step
        h_smooth = np.copy(best_h)
        if smoothing_sigma > 0:
            # Apply Gaussian smoothing
            kernel = np.exp(-np.arange(-3, 4)**2 / (2 * smoothing_sigma**2))
            kernel /= np.sum(kernel)
            h_smooth = np.convolve(h_smooth, kernel, mode='same')
        
        # Clip to [0,1]
        h_smooth = np.clip(h_smooth, 0, 1)
        # Normalize sum
        h_smooth = h_smooth * (n_points/2) / np.sum(h_smooth)
        
        # Local optimization
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_smooth, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 20, 'ftol': 1e-6})
            if res.success:
                h_opt = res.x
                c5_opt = evaluate_erdos_solution(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt
            else:
                raise Exception("Optimization failed")
        except:
            pass
            
        # Adaptive refinement using SQP for final few iterations
        if max_iter - iteration < 50:
            try:
                res = minimize(obj, best_h, method='SLSQP', 
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_opt = res.x
                    c5_opt = evaluate_erdos_solution(h_opt, dx)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
            except:
                pass
                
        # Print progress every 50 iterations
        if iteration % 50 == 0:
            if best_c5 < 0.381:
                break

    if time() - start_time < budget_s - 30:
        try:
            res = minimize(evaluate_erdos_solution, best_h,

                          bounds=[(0, 1) for _ in range(n_points)],
                          constraints={'type': 'eq', 'fun': con_sum},
                          options={'maxiter': 100, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
