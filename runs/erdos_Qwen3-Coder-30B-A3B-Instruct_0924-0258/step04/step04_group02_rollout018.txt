import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use gradient-based approach with a smooth approximation
    def smooth_h(x, params):
        # params: [theta1, theta2, ..., theta_n, beta, lambda]
        theta = params[:-2]
        beta = params[-2]
        lmbda = params[-1]
        
        # Linear combination of basis functions
        # Using a simple linear model: sum of shifted weights
        h_vals = np.zeros_like(x)
        for i in range(len(theta)):
            h_vals += theta[i] * np.exp(-((x - i * dx) ** 2) / (2 * (dx/2) ** 2))
        h_vals += beta
        h_vals = np.maximum(h_vals, 0)  #ReLU
        h_vals = np.minimum(h_vals, 1)  #clip to [0,1]
        
        # Rescale by lambda
        scaled_h = lmbda * h_vals
        scaled_h = np.minimum(scaled_h, 1)
        return scaled_h
    
    def compute_objective(params):
        # Compute h values
        x = np.linspace(0, 2, n_points)
        h = smooth_h(x, params)
        
        # Ensure the constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            lmbda = (n_points / 2) / h_sum
        else:
            lmbda = 1.0
        h = np.minimum(lmbda * h, 1.0)
        
        # Compute objective
        return evaluate_erdos_solution(h, dx)
    
    def project_params(params):
        # Ensure lambda is positive, and effective h boundary values
        theta = params[:-2]
        beta = params[-2]
        lmbda = params[-1]
        
        # Clip lambda to avoid numerical issues
        lmbda = max(0.01, min(100, lmbda))
        
        # Clip beta (sensible bound for initial function)
        beta = max(-10, min(10, beta))
        
        # Avoid extreme theta
        theta = np.clip(theta, -10, 10)
        
        return np.append(theta, [beta, lmbda])
    
    # Initial parameters (theta, beta, lambda)
    # Start with some educated guesses
    x = np.linspace(0, 2, n_points)
    if 'initial_h_values' in globals():

        # Initialize with small values, then project as needed
        params = np.random.normal(0, 0.1, n_points + 2)
        params[-2] = 0
        params[-1] = 1.0
    else:
        # Initialize from scratch
        params = np.random.normal(0, 0.1, n_points + 2)
        params[-2] = 0
        params[-1] = 1.0
    
    # TMVQN optimizer approach 
    # Use a simple coordinate descent/restart based approach
    # More advanced optimizers might be too complex for this constraint
    iteration = 0

        idx = np.random.randint(0, len(params))

                    best_c5 = objective_new
                    last_improvement = iteration

        if iteration - last_improvement > patience:

            params = np.random.normal(0, 0.1, n_points + 2)

        final_h = (n_points / 2) / final_h_sum * final_h
    final_h = np.minimum(final_h, 1.0)
    
    # Double check final evaluation
    final_c5 = evaluate_erdos_solution(final_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    return (list(best_h), best_c5, n_points)
