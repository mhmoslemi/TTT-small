from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

    decay = 0.95

        h = 1 / (1 + np.exp(-current_z))

        integral = np.sum(h) * dx
