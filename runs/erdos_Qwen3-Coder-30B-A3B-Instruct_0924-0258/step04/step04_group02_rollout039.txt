import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Fourier-based parametrization for smooth functions
    n_freq = n_points // 2
    
    # Initialize with small random coefficients
    a = np.random.normal(0, 0.1, n_freq)
    b = np.random.normal(0, 0.1, n_freq)
    a0 = 0.0
    
    # Activation function: sigmoid
    def sigmoid(x):
        return 1.0 / (1.0 + np.exp(-x))
    
    # Compute h(x) from Fourier coefficients
    def compute_h():
        x = np.linspace(0, 2, n_points, endpoint=False)
        # Build the sum: a0 + sum(a_k * cos(k*pi*x) + b_k * sin(k*pi*x))
        arg = a0 + np.sum([a[k] * np.cos((k+1) * np.pi * x) + b[k] * np.sin((k+1) * np.pi * x) 
                          for k in range(n_freq)], axis=0)
        h = sigmoid(arg)
        return h
    
    # Normalize h so that sum(h) * dx = 1
    def normalize_h(h):
        h_sum = np.sum(h) * dx
        return h / h_sum * (n_points / 2)
    
    # Compute overlap using FFT (convolution theorem)
    def compute_overlaps(h):
        # Ensure h is in [0,1]
        h_clipped = np.clip(h, 0, 1)
        # Compute 1 - h with same bounds
        one_minus_h = 1 - h_clipped
        
        # Use FFT convolution
        H_fft = fft(h_clipped)
        one_minus_H_fft = fft(one_minus_h)
        
        # Convolution is spectrum multiplication
        conv_spectrum = H_fft * np.conj(one_minus_H_fft)
        overlap = ifft(conv_spectrum).real
        
        # The overlaps are for all k in [0, 2*n_points-2]
        # Extract only meaningful shifts [0, 2*n_points-2]
        # As we're doing it on [0,2], we want the first n_points elements
        # and then wrap around properly.
        # Extract the first n_points - 1 non-trivial (for k=1 to 2*n_points-2)
        # But faster, use symmetric property
        C_k = overlap[:n_points]
        # C_k[k] corresponds to shift k, C_k[0] is for shift 0
        # Return the max, avoiding self-overlap at shift 0
        return np.max(C_k[1:])  # omit k=0

    # Main optimization loop
    best_c5 = float('inf')
    best_a, best_b, best_a0 = a.copy(), b.copy(), a0
    
    # Optimizer parameters
    learning_rate = 0.1
    momentum = 0.9
    decay = 0.99
    reg_lambda = 1e-4
    
    # Gradient and momentum accumulators
    m_a = np.zeros_like(a)
    m_b = np.zeros_like(b)
    m_a0 = 0.0
    
    for iteration in range(500):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute h
        h = compute_h()
        # Normalize
        h = normalize_h(h)
        
        # Compute overlaps
        c5 = compute_overlaps(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_a = a.copy()
            best_b = b.copy()
            best_a0 = a0
            
        # Compute gradient approximated via finite difference with small perturbations
        eps = 1e-5
        grad_a = np.zeros_like(a)
        grad_b = np.zeros_like(b)
        grad_a0 = 0.0
        
        for k in range(n_freq):
            # Perturb coefficient a[k]
            a_plus = a.copy()
            a_plus[k] += eps
            h_plus = sigmoid(a0 + np.sum([a_plus[j] * np.cos((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) + 
                                         b[j] * np.sin((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) for j in range(n_freq)], axis=0))
            h_plus = normalize_h(h_plus)
            c5_plus = compute_overlaps(h_plus)
            
            a_minus = a.copy()
            a_minus[k] -= eps
            h_minus = sigmoid(a0 + np.sum([a_minus[j] * np.cos((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) + 
                                          b[j] * np.sin((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) for j in range(n_freq)], axis=0))
            h_minus = normalize_h(h_minus)
            c5_minus = compute_overlaps(h_minus)
            
            grad_a[k] = (c5_plus - c5_minus) / (2 * eps)
        
        for k in range(n_freq):
            # Perturb coefficient b[k]
            b_plus = b.copy()
            b_plus[k] += eps
            h_plus = sigmoid(a0 + np.sum([a[j] * np.cos((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) + 
                                         b_plus[j] * np.sin((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) for j in range(n_freq)], axis=0))
            h_plus = normalize_h(h_plus)
            c5_plus = compute_overlaps(h_plus)
            
            b_minus = b.copy()
            b_minus[k] -= eps
            h_minus = sigmoid(a0 + np.sum([a[j] * np.cos((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) + 
                                          b_minus[j] * np.sin((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) for j in range(n_freq)], axis=0))
            h_minus = normalize_h(h_minus)
            c5_minus = compute_overlaps(h_minus)
            
            grad_b[k] = (c5_plus - c5_minus) / (2 * eps)
        
        # Perturb a0
        a0_plus = a0 + eps
        h_plus = sigmoid(a0_plus + np.sum([a[j] * np.cos((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) + 
                                          b[j] * np.sin((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) for j in range(n_freq)], axis=0))
        h_plus = normalize_h(h_plus)
        c5_plus = compute_overlaps(h_plus)
        
        a0_minus = a0 - eps
        h_minus = sigmoid(a0_minus + np.sum([a[j] * np.cos((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) + 
                                            b[j] * np.sin((j+1) * np.pi * np.linspace(0, 2, n_points, endpoint=False)) for j in range(n_freq)], axis=0))
        h_minus = normalize_h(h_minus)
        c5_minus = compute_overlaps(h_minus)
        
        grad_a0 = (c5_plus - c5_minus) / (2 * eps)
        
        # Update gradients with momentum
        m_a = momentum * m_a + (1 - momentum) * grad_a
        m_b = momentum * m_b + (1 - momentum) * grad_b
        m_a0 = momentum * m_a0 + (1 - momentum) * grad_a0
        
        # Apply gradients with learning rate
        a -= learning_rate * m_a + reg_lambda * a
        b -= learning_rate * m_b + reg_lambda * b
        a0 -= learning_rate * m_a0
        
        # Apply decay to learning rate
        learning_rate *= decay
            
    # Refine best solution using optimization with better gradient
    def objective(coeffs):
        a_try = coeffs[:n_freq]
        b_try = coeffs[n_freq:2*n_freq]
        a0_try = coeffs[2*n_freq]
        
        # Compute h
        x = np.linspace(0, 2, n_points, endpoint=False)
        arg = a0_try + np.sum([a_try[k] * np.cos((k+1) * np.pi * x) + 
                               b_try[k] * np.sin((k+1) * np.pi * x) for k in range(n_freq)], axis=0)
        h = sigmoid(arg)
        h = normalize_h(h)
        
        # Compute C5
        return compute_overlaps(h)
    
    # Initial guess
    coeffs_opt = np.concatenate([best_a, best_b, [best_a0]])
    
    # Use scipy minimize for better performance
    try:
        res = minimize(objective, coeffs_opt, method='L-BFGS-B', options={'maxiter': 50},
                       bounds=[(-10, 10)] * (2*n_freq+1))
        if res.success:
            best_a = res.x[:n_freq]
            best_b = res.x[n_freq:2*n_freq]
            best_a0 = res.x[2*n_freq]
    except:
        pass
    
    # Compute final h
    x = np.linspace(0, 2, n_points, endpoint=False)
    arg = best_a0 + np.sum([best_a[k] * np.cos((k+1) * np.pi * x) + 
                           best_b[k] * np.sin((k+1) * np.pi * x) for k in range(n_freq)], axis=0)
    final_h = sigmoid(arg)
    final_h = normalize_h(final_h)
    
    final_c5 = compute_overlaps(final_h)
    
    # Ensure bounds
    final_h = np.clip(final_h, 0, 1)
    final_h_scaled = normalize_h(final_h)
    final_c5 = compute_overlaps(final_h_scaled)
    
    return (final_h_scaled.tolist(), final_c5, n_points)
