import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization approach
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    exploration = 2.0
    exploitation = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    personal_bests = []
    personal_best_scores = []
    
    # Main particle
    h_main = h_start.copy()
    h_main = h_main * (n_points / 2) / np.sum(h_main)
    particles.append(h_main.copy())
    velocities.append(np.zeros(n_points))
    personal_bests.append(h_main.copy())
    personal_best_scores.append(evaluate_erdos_solution(h_main, dx))
    
    # Other particles
    for _ in range(swarm_size - 1):
        h = h_start + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_bests.append(h.copy())
        personal_best_scores.append(evaluate_erdos_solution(h, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            exploration * r1 * (personal_bests[i] - particles[i]) + 
                            exploitation * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Clamp to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Maintain sum constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            score = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_bests[i] = particles[i].copy()
                personal_best_scores[i] = score
            
            # Update global best
            if score < global_best_score:
                global_best = particles[i].copy()
                global_best_score = score
    
    # Refinement with convex optimization
    try:
        # Try to polish the global best with a constrained convex optimization
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add a rough bound on the correlation
        corr = cp.Variable(2*n_points - 1)
        for k in range(2*n_points - 1):
            lo = max(0, k - n_points + 1)
            hi = min(k, n_points - 1) + 1
            corr[k] = cp.sum(cp.multiply(h[lo:hi], 1 - h[k-hi+1:k-lo]))
        
        # Soft constraint on the maximum correlation
        max_corr = cp.max(corr)
        constraints.append(max_corr <= global_best_score + 0.01)
        
        prob = cp.Problem(cp.Minimize(max_corr), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_refined = h.value
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < global_best_score:
                    global_best = h_refined
                    global_best_score = c5_refined
        except Exception:
            pass
    except Exception:
        pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Improve with more restarts
        for restart in range(10):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = np.clip(global_best + np.random.normal(0, 0.02, n_points), 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 50, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < global_best_score:
                        global_best = h_refined
                        global_best_score = c5_refined
            except:
                pass
    
    # Final validation
    if not np.allclose(np.sum(global_best), n_points / 2, atol=1e-6):
        global_best = global_best * (n_points / 2) / np.sum(global_best)
    if not np.all((global_best >= 0) & (global_best <= 1)):
        global_best = np.clip(global_best, 0, 1)
    
    return (list(global_best), global_best_score, n_points)
