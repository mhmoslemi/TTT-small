import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Binary relaxation approach combined with continuous refinement
    def binary_relaxation(h):
        return np.clip(np.round(h), 0, 1)
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Constraint generation with MIP-like tightening
    max_iter = 50
    iter_count = 0
    tight_bound = 0.4
    
    while time() - start_time < budget_s * 0.8 and iter_count < max_iter:
        iter_count += 1
        
        # Improved binary initialization
        h_binary = binary_relaxation(best_h)
        h_binary = project_to_feasible(h_binary)
        
        # Local improvement with optimization
        h_local = h_binary.copy()
        for _ in range(5):  # Local search iterations
            # Perturb 10% of values
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            for i in indices:
                h_local[i] = np.clip(h_local[i] + np.random.normal(0, 0.05), 0, 1)
            h_local = project_to_feasible(h_local)
            
            c5_current = evaluate_erdos_solution(h_local, dx)
            if c5_current < best_c5:
                best_h = h_local.copy()
                best_c5 = c5_current
        
        # Constraint tightening for future iterations
        if best_c5 < tight_bound:
            tight_bound = best_c5 - 0.001
        if tight_bound < 0.30:  # Safeguard threshold
            break
    
    # Convex optimization to smooth and tighten
    max_iter_cvx = 50
    iter_cvx = 0
    
    while time() - start_time < budget_s * 0.95 and iter_cvx < max_iter_cvx:
        iter_cvx += 1
        
        try:
            h_cvx = cp.Variable(n_points)
            constraints = [
                cp.sum(h_cvx) == n_points / 2.0,
                h_cvx >= 0,
                h_cvx <= 1,
            ]
            
            # Generate a subset of constraints for efficiency
            max_shift = min(10, n_points // 2)
            for k in range(1, max_shift):
                corr_sum = cp.sum(cp.multiply(h_cvx[:n_points-k], 1 - h_cvx[k:]))
                constraints.append(corr_sum <= best_c5 * dx + 1e-6)
            
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=100)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_cvx.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_h = h_feasible
                    best_c5 = c5_feasible
        except Exception:
            pass
    
    # Final refinement
    try:
        h_refined = project_to_feasible(best_h)
        c5_refined = evaluate_erdos_solution(h_refined, dx)
        if c5_refined < best_c5:
            best_h = h_refined
            best_c5 = c5_refined
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
