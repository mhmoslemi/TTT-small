import numpy as np
from scipy.optimize import linprog
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 100
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Create a piecewise constant function with 6 segments
        segments = 6
        segment_width = 2.0 / segments
        for i in range(segments):
            start_idx = int(i * segment_width / dx)
            end_idx = int((i + 1) * segment_width / dx)
            if i % 2 == 0:
                h_start[start_idx:end_idx] = 0.5
            else:
                h_start[start_idx:end_idx] = 0.0
        # Normalize to ensure sum is n/2
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    h_opt = h_start.copy()
    c5_opt = evaluate_erdos_solution(h_start, dx)
    
    # Binary search parameters
    low = 0.3
    high = 0.38085
    precision = 1e-5
    max_iter = 60
    iter_count = 0
    
    # Binary search loop
    while (high - low > precision) and (time() - start_time < budget_s * 0.95) and iter_count < max_iter:
        mid = (low + high) / 2.0
        
        # Formulate and solve LP for feasibility check
        try:
            # Create variables
            h = cp.Variable(n_points)
            
            # Objective is to verify feasibility - we minimize 0
            objective = cp.Minimize(0)
            
            # Add constraints: sum = n_points/2
            sum_constraint = cp.Constraint(cp.sum(h), cp.Equality, n_points / 2.0)
            
            # Add constraints: 0 <= h[i] <= 1
            bound_constraints = [cp.Constraint(h[i], cp.LessThan, 1.0) for i in range(n_points)]
            bound_constraints += [cp.Constraint(h[i], cp.GreaterThan, 0.0) for i in range(n_points)]
            
            # Add overlap constraints: sum h[i]*(1-h[i+k]) <= mid / dx for all valid lags k
            constraints = [sum_constraint] + bound_constraints
            for k in range(2 * n_points - 1):
                # Compute valid range of i
                if k < n_points:
                    valid_i = range(n_points - k)
                    expr = cp.sum(cp.multiply(h[valid_i], 1 - h[valid_i + k]))
                else:
                    valid_i = range(2 * n_points - 1 - k)
                    expr = cp.sum(cp.multiply(h[valid_i], 1 - h[k - n_points + 1 + valid_i]))
                constraints.append(cp.Constraint(expr, cp.LessThan, mid / dx))
            
            # Create LP problem
            prob = cp.Problem(objective, constraints)
            
            # Solve problem
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
                if prob.status == cp.OPTIMAL or prob.status == cp.UNBOUNDED:
                    # The problem is feasible, refine upper bound
                    high = mid
                    h_opt = h.value
                    c5_opt = mid
                else:
                    # The problem is infeasible, refine lower bound
                    low = mid
            except:
                # Default case - infeasible or error
                low = mid
                
        except Exception as e:
            # Fallback in case of error
            low = mid
        
        iter_count += 1
        
    # Final cleanup of the best solution
    h_final = np.clip(h_opt, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final) if np.sum(h_final) > 0 else h_final
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (list(h_final), c5_final, n_points)
