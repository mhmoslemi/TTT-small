import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Particle Swarm Optimization approach
    swarm_size = 50
    max_iter = 1000
    inertia = 0.7
    cognition = 2.0
    social = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_bests = []
    personal_best_values = []
    
    # Start with initial solution as one particle
    swarm.append(h_start.copy())
    velocities.append(np.zeros(n_points))
    personal_bests.append(h_start.copy())
    personal_best_values.append(best_c5)
    
    # Initialize remaining particles with small perturbations
    for i in range(1, swarm_size):
        h_pert = h_start + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        swarm.append(h_pert)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_bests.append(h_pert.copy())
        personal_best_values.append(evaluate_erdos_solution(h_pert, dx))
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_bests[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognition * r1 * (personal_bests[i] - swarm[i]) + 
                            social * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Clamp and normalize
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Evaluate fitness
            fit = evaluate_erdos_solution(swarm[i], dx)
            
            # Update personal best
            if fit < personal_best_values[i]:
                personal_bests[i] = swarm[i].copy()
                personal_best_values[i] = fit
                
                # Update global best
                if fit < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = fit
        
        # Periodically refine with convex optimization
        if iteration % 20 == 0:
            try:
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add some overlap constraints for feasibility
                for k in range(0, min(10, 2 * n_points - 1), 2):
                    if k < n_points:
                        corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(corr_sum <= 0.5 * dx * n_points)
                
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < global_best_value:
                            global_best = h_refined
                            global_best_value = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
    
    # Final polishing with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        for restart in range(5):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = global_best + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < global_best_value:
                        global_best = h_refined
                        global_best_value = c5_refined
            except:
                pass
    
    return (list(global_best), global_best_value, n_points)
