import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 40
        h_start = np.ones(n_points) * 0.5
    
    # Rescale to fit the 2-unit interval with sum = 1
    dx = 2.0 / n_points
    h_start = h_start * (n_points / 2.0) / np.sum(h_start)
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Adaptive grid search
    max_grid_iter = 50
    grid_search_time = 0.6 * budget_s
    initial_c5 = best_c5
    
    # Start from a narrowed down search space
    search_range = np.linspace(0.37, min(0.38085, initial_c5), max_grid_iter)
    
    print(f"Initial C5: {best_c5:.6f}")
    
    for i, c_mid in enumerate(search_range):
        if time() - start_time > grid_search_time:
            break
            
        # Convex feasibility check
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints
        corr_size = 2 * n_points - 1
        for k in range(corr_size):
            if k < n_points:
                corr_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
            constraints.append(corr_sum <= c_mid * dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                h_feasible = h.value
                c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                if c5_feasible < best_c5:
                    best_c5 = c5_feasible
                    best_h = h_feasible.copy()
                    print(f"Improved C5 at iteration {i}: {best_c5:.6f}")
        except:
            continue
    
    # Gradient-based refinement with multiple restarts
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Multiple restarts with perturbations
    if time() - start_time < 0.9 * budget_s:
        max_restarts = 30
        for restart in range(max_restarts):
            if time() - start_time > 0.9 * budget_s:
                break
                
            # Start with the best solution or random perturbation
            if restart == 0:
                h_init = best_h.copy()
            else:
                # Random perturbation with adaptive scaling
                scale = 0.03 * (1.0 - restart / max_restarts) + 0.005
                h_init = best_h + np.random.normal(0, scale, n_points)
                h_init = np.clip(h_init, 0, 1)
                # Normalize again
                h_init = h_init * (n_points / 2.0) / np.sum(h_init)
            
            try:
                res = minimize(objective, h_init, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 150, 'ftol': 1e-8})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
                        print(f"Refined C5 at restart {restart}: {best_c5:.6f}")
                        if best_c5 < 0.3807:
                            break
            except:
                continue
    
    # Final fine-tuning
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 20 and best_c5 > 0.3807:
        try:
            res = minimize(objective, best_h, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 200, 'ftol': 1e-9})
            if res.success:
                h_final = res.x
                c5_final = objective(h_final)
                if c5_final < best_c5:
                    best_h = h_final
                    best_c5 = c5_final
        except:
            pass
    
    print(f"Final C5: {best_c5:.6f}")
    return (list(best_h), best_c5, n_points)
