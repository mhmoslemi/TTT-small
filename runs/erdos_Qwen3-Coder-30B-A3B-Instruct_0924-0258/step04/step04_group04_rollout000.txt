import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 20
        h_start = np.ones(n_points) * 0.5  # Simple uniform start
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Coarse optimization with simulated annealing
    temp = 0.1
    cool_rate = 0.98
    perturb_mag = 0.03
    max_iter = 100
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
        # Perturb
        h_pert = h_start + np.random.normal(0, perturb_mag, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        old_c5 = evaluate_erdos_solution(h_start, dx)
        new_c5 = evaluate_erdos_solution(h_pert, dx)
        
        if new_c5 < old_c5 or np.random.rand() < math.exp(-(new_c5 - old_c5) / temp):
            h_start = h_pert
            if new_c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = new_c5
        temp *= cool_rate
    
    # Upsample to 40 points
    if time() - start_time < budget_s * 0.8:
        n_points_up = 40
        dx_up = 2.0 / n_points_up
        h_upsampled = np.interp(np.linspace(0, 2, n_points_up), 
                               np.linspace(0, 2, len(h_start)), 
                               h_start)
        h_upsampled = h_upsampled * (n_points_up/2) / np.sum(h_upsampled)
        h_start = h_upsampled
        n_points = n_points_up
    
    # Fine optimization with SLSQP
    if time() - start_time < budget_s * 0.9:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, h_start, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                h_start = res.x
        except:
            pass
    
    # Final refinement with gradient descent-like approach
    if time() - start_time < budget_s * 0.98:
        h_refined = h_start.copy()
        learning_rate = 0.01
        max_iters = 500
        for _ in range(max_iters):
            if time() - start_time > budget_s * 0.98:
                break
            c5_old = evaluate_erdos_solution(h_refined, dx)
            
            # Compute gradient approximation
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            for i in range(len(h_refined)):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = h_plus * (n_points/2) / np.sum(h_plus)
                c5_plus = evaluate_erdos_solution(h_plus, dx)
                grad[i] = (c5_plus - c5_old) / eps
            
            # Update step
            h_new = h_refined - learning_rate * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new, dx)
            if c5_new < c5_old:
                h_refined = h_new
                if c5_new < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_new
            else:
                break
    
    return (list(best_h), best_c5, n_points)
