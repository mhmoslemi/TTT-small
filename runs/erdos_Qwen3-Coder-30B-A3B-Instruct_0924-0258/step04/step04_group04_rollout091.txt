import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        # Create an initial guess with some structure
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Place two clusters
        for i in range(n_points):
            x = i * dx
            if 0.3 <= x <= 0.7 or 1.3 <= x <= 1.7:
                h_start[i] = 0.5
        # Normalize to satisfy constraint
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # CVXPY optimization helper
    def solve_convex_subproblem(h_curr, lambda_vals):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points/2,
            h >= 0,
            h <= 1
        ]
        
        # Objective: minimize sum of lambda * C_k(h) over all shifts
        obj = 0
        for k in range(-n_points+1, n_points):
            # Compute the contribution from shift k
            if k >= 0:
                # h shifted by k positions (circular)
                h_shifted = np.concatenate([h_curr[k:], h_curr[:k]])
            else:
                # h shifted by k positions (negative shift)
                h_shifted = np.concatenate([h_curr[k:], h_curr[:k]])
            # Contribution from k shift to objective
            obj += lambda_vals[k + n_points - 1] * cp.sum(cp.multiply(h, 1 - h_shifted))
        
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                return h_sol
        except:
            pass
        return h_curr

    # Alternate between optimizing h and updating lambda
    max_iter = 50
    lambda_vals = np.ones(2 * n_points - 1)
    
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Update lambda_vals based on current h
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        lambda_vals = np.maximum(lambda_vals * corr, 1e-6)
        
        # Solve the convex subproblem to improve h
        h_new = solve_convex_subproblem(best_h, lambda_vals)
        if h_new is not None:
            # Project h_new to feasible set
            h_new = np.clip(h_new, 0, 1)
            # Normalize again
            h_new = h_new * (n_points/2) / np.sum(h_new)
            corr_new = evaluate_erdos_solution(h_new, dx)
            if corr_new < best_c5:
                best_h = h_new.copy()
                best_c5 = corr_new
    
    # Final refinement using scipy minimize
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
