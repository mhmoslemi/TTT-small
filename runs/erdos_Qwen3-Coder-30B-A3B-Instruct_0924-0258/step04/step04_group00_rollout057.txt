import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    elite_size = 10
    tournament_size = 2
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def create_population():
        population = []
        if 'initial_h_values' in globals():
            population.append(h_start.copy())
            remaining = population_size - 1
        else:
            remaining = population_size
        
        for _ in range(remaining):
            population.append(create_individual())
        return population
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def tournament_selection(population, k=tournament_size):
        tournament = random.sample(population, k)
        return min(tournament, key=fitness)
    
    def crossover(parent1, parent2):
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        # Repair feasibility
        child1 = np.clip(child1, 0, 1)
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = np.clip(child2, 0, 1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.01)
        # Repair feasibility
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    population = create_population()
    
    generation = 0
    while time() - start_time < budget_s * 0.95 and generation < num_generations:
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best
        for i, ind in enumerate(population):
            if fitness_scores[i] < best_c5:
                best_h = ind.copy()
                best_c5 = fitness_scores[i]
        
        # Create new population
        new_population = []
        
        # Elitism
        sorted_indices = np.argsort(fitness_scores)
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Tournament selection, crossover and mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        generation += 1
    
    # Final refinement with local search
    if time() - start_time < budget_s * 0.99:
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        # Local improvement using gradient-free method
        from scipy.optimize import differential_evolution
        
        def obj(h):
            return evaluate_erdos_solution(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
        
        try:
            res = differential_evolution(obj, bounds, constraints=cons, 
                                        maxiter=50, popsize=15)
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass
    
    # Final check
    try:
        final_h = np.clip(best_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = evaluate_erdos_solution(final_h, dx)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
