import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # PSO parameters
    n_particles = 50
    n_iter = 1000
    w_start, w_end = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    inertia = w_start
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        if 'initial_h_values' in globals():
            h = h_start.copy()
            h += np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.uniform(0.2, 0.8, n_points)
        h = h * (n_points/2) / np.sum(h)  # normalize sum to n/2
        h = np.clip(h, 0, 1)  # bounds
        particles.append(h)
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_erdos_solution(h, dx))
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO loop
    for iter_idx in range(n_iter):
        if (time() - start_time) > budget_s * 0.95:
            break

        # Update inertia weight
        inertia = w_start - (w_start - w_end) * iter_idx / n_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and sum constraint
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points/2) / np.sum(particles[i])
            
            # Evaluate
            particle_val = evaluate_erdos_solution(particles[i], dx)
            
            # Update personal best
            if particle_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = particle_val
                
                # Update global best if needed
                if particle_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = particle_val
        
        if iter_idx % 100 == 0:
            # Occasionally refine with convex optimization
            try:
                # Find the maximum overlap value to bound the convex problem
                corr_vals = np.correlate(gbest, 1 - gbest, mode='full')
                max_over = np.max(corr_vals) * dx
                
                h = cp.Variable(n_points)
                constraints = [
                    cp.sum(h) == n_points / 2.0,
                    h >= 0,
                    h <= 1
                ]
                
                # Add overlap constraints for key lags
                for k in range(0, min(10, 2 * n_points - 1), 2):
                    if k < n_points:
                        conv_sum = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
                    else:
                        conv_sum = cp.sum(cp.multiply(h[k-n_points+1:], 1 - h[:n_points-(k-n_points+1)]))
                    constraints.append(conv_sum <= max_over * 1.1)  # relax a bit
                
                # Solve for the maximum
                prob = cp.Problem(cp.Minimize(0), constraints)
                try:
                    prob.solve(solver=cp.ECOS, eps=1e-6)
                    if prob.status == cp.OPTIMAL:
                        h_refined = h.value
                        c5_refined = evaluate_erdos_solution(h_refined, dx)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
            except Exception:
                pass
            
            # Early stopping if we're improving well
            if best_c5 < 0.38085:
                break
    
    # Final refinement with SLSQP
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        # Try several random restarts with refinement
        for restart in range(8):
            if time() - start_time > budget_s * 0.95:
                break
                
            h_restart = best_h + np.random.normal(0, 0.01, n_points)
            h_restart = np.clip(h_restart, 0, 1)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            
            try:
                res = minimize(objective, h_restart, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass

    # Enforce final constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
