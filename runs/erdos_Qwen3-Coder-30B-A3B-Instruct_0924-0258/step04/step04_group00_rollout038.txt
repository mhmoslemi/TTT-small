import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use Evolution Strategies with Augmented Lagrangian
    population_size = 50
    mutation_strength = 0.05
    elite_size = 5
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def augmented_lagrangian(h, Lambda_sum, Lambda_bounds, mu_sum, mu_bounds):
        h_feasible = project_to_feasible(h)
        obj = evaluate_erdos_solution(h_feasible, dx)
        
        # penalty for constraint violation
        violation_sum = np.abs(np.sum(h_feasible) - n_points / 2.0)
        violation_bounds = np.sum(np.maximum(0, h_feasible) + np.maximum(0, 1 - h_feasible))
        
        penalty = mu_sum * violation_sum + mu_bounds * violation_bounds
        return obj + penalty
    
    # ES optimization loop
    population = [project_to_feasible(h_start + np.random.normal(0, mutation_strength, n_points)) for _ in range(population_size)]
    Lambda_sum = 0.0
    Lambda_bounds = 0.0
    mu_sum = 1.0
    mu_bounds = 1.0
    
    max_generations = 5000
    generation = 0
    
    while time() - start_time < budget_s * 0.9 and generation < max_generations:
        # Evaluate population
        fitness = [augmented_lagrangian(ind, Lambda_sum, Lambda_bounds, mu_sum, mu_bounds) for ind in population]
        
        # Sort by fitness (lower is better)
        indices = np.argsort(fitness)
        top_indices = indices[:elite_size]
        
        # Update best solution
        if fitness[top_indices[0]] < best_c5:
            best_h = project_to_feasible(population[top_indices[0]])
            best_c5 = fitness[top_indices[0]]
        
        # Update Lagrange multipliers
        sum_violations = [np.abs(np.sum(ind) - n_points / 2.0) for ind in population]
        bound_violations = [np.sum(np.maximum(0, ind) + np.maximum(0, 1 - ind)) for ind in population]
        Lambda_sum = Lambda_sum + mu_sum * np.mean(sum_violations)
        Lambda_bounds = Lambda_bounds + mu_bounds * np.mean(bound_violations)
        
        # Update penalty parameters
        if np.mean(sum_violations) > 1e-3:
            mu_sum *= 1.1
        if np.mean(bound_violations) > 1e-3:
            mu_bounds *= 1.1
            
        # Selection and reproduction
        selected = [population[i] for i in top_indices]
        new_population = []
        
        # Elite
        new_population.extend(selected)
        
        # Mutation with recombination
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Blend crossover
            alpha = np.random.random()
            offspring = alpha * parent1 + (1 - alpha) * parent2
            
            # Gaussian mutation
            offspring += np.random.normal(0, mutation_strength, n_points)
            
            # Project
            offspring = project_to_feasible(offspring)
            new_population.append(offspring)
            
        population = new_population[:population_size]
        
        # Adjust mutation strength
        diversity = np.std([np.std(ind) for ind in population])
        if diversity < 0.01:
            mutation_strength *= 1.1
        elif diversity > 0.1:
            mutation_strength *= 0.9
            
        generation += 1
    
    # Final refinement with local optimization
    try:
        h_refined = project_to_feasible(best_h.copy())
        cons_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(evaluate_erdos_solution, h_refined, args=(dx,), 
                       method='SLSQP', bounds=bounds, constraints=cons_sum, 
                       options={'maxiter': 100, 'ftol': 1e-8})
        if res.success:
            h_final = res.x
            c5_final = evaluate_erdos_solution(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
