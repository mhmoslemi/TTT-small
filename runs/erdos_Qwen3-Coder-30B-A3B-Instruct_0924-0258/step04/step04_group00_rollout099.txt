import numpy as np
from time import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Helper function to compute gradient
    def compute_gradient(h, dx):
        n = len(h)
        grad = np.zeros(n)
        for k in range(-n+1, n):
            if k >= 0:
                prod = h[:n-k] * (1 - h[k:])
            else:
                prod = h[-k:] * (1 - h[:n+k])
            if k >= 0:
                grad[:n-k] += prod
                grad[k:] -= prod
            else:
                grad[-k:] += prod
                grad[:n+k] -= prod
        return grad * dx
    
    # Project h onto feasible region
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # PGD with adaptive step size
    max_iter = 10000
    alpha = 0.01
    tol = 1e-6
    last_improvement = 0
    patience = 50
    
    h_current = best_h.copy()
    
    for i in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient
        grad = compute_gradient(h_current, dx)
        
        # Update h
        h_new = h_current - alpha * grad
        h_proj = project_h(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_proj, dx)
        
        # Adaptive step size
        if c5_new < best_c5:
            h_current = h_proj
            best_c5 = c5_new
            alpha *= 1.2
            last_improvement = 0
        else:
            alpha *= 0.5
            last_improvement += 1
            if last_improvement > patience:
                break
            
    # Final refinement
    try:
        h_final = project_h(h_current)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
