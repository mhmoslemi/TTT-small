import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Use binary relaxation approach with local search refinement
    def binary_search_h(h_binary):
        # Convert binary h to continuous h
        h_cont = h_binary.astype(float)
        h_cont = h_cont * (n_points/2) / np.sum(h_cont)
        return h_cont
    
    def binary_objective(h_binary):
        h_cont = binary_search_h(h_binary)
        return evaluate_erdos_solution(h_cont, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Local optimization with discrete search
    max_restarts = 50
    restart_count = 0
    early_stop = 0
    
    while time() - start_time < budget_s * 0.95:
        # Try local binary search
        h_binary = (best_h > 0.5).astype(int)
        h_binary = h_binary * (n_points/2) / np.sum(h_binary)
        
        # Perturb and refine
        h_perturbed = best_h.copy()
        h_perturbed += np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        # Use optimization to refine
        try:
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(
                lambda h: evaluate_erdos_solution(h, dx), 
                h_perturbed, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=cons_sum,
                options={'maxiter': 100, 'ftol': 1e-7}
            )
            if res.success:
                h_refined = res.x
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
                    restart_count = 0
                    early_stop = 0
                else:
                    restart_count += 1
                    early_stop += 1
            else:
                restart_count += 1
                early_stop += 1
        except Exception:
            restart_count += 1
            early_stop += 1
        
        # Constraint satisfaction check
        if early_stop > 10:
            early_stop = 0
            # Try to improve using relaxation
            try:
                h_relaxed = cp.Variable(n_points)
                constraints = [
                    cp.sum(h_relaxed) == n_points / 2.0,
                    h_relaxed >= 0,
                    h_relaxed <= 1
                ]
                
                # Add a subset of constraints to manage complexity
                for k in range(1, min(10, n_points)):  
                    if k < n_points // 2:
                        corr_sum = cp.sum(cp.multiply(h_relaxed[:n_points-k], 1 - h_relaxed[k:]))
                    else:
                        corr_sum = cp.sum(cp.multiply(h_relaxed[k:], 1 - h_relaxed[:n_points-k]))
                    constraints.append(corr_sum <= best_c5 * dx + 1e-6)
                
                prob = cp.Problem(cp.Minimize(cp.sum(h_relaxed)), constraints)
                prob.solve(solver=cp.ECOS, eps=1e-7)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_relaxed.value
                    c5_feasible = evaluate_erdos_solution(h_feasible, dx)
                    if c5_feasible < best_c5:
                        best_h = h_feasible
                        best_c5 = c5_feasible
            except Exception:
                pass
        
        # Global restart
        if restart_count > max_restarts:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart * (n_points/2) / np.sum(h_restart)
            h_restart = np.clip(h_restart, 0, 1)
            restart_count = 0
            if time() - start_time > budget_s * 0.9:
                break
    
    # Final refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except Exception:
        pass
    
    # Force constraints to be exact
    h_final = best_h.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points/2) / np.sum(h_final)
    
    return (list(h_final), best_c5, n_points)
