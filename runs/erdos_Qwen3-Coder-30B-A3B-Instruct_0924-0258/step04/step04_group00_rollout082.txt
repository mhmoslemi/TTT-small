import numpy as np
from time import time
import random

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Genetic Algorithm parameters
    population_size = 100
    num_generations = 500
    mutation_rate = 0.1
    tournament_size = 2
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def create_feasible_individual():
        h = create_individual()
        # Ensure sum constraint more carefully
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h
    
    if 'initial_h_values' in globals():
        population = [h_start] + [create_feasible_individual() for _ in range(population_size - 1)]
    else:
        population = [create_feasible_individual() for _ in range(population_size)]
    
    def fitness(individual):
        return evaluate_erdos_solution(individual, dx)
    
    def tournament_selection(pop, k=tournament_size):
        selected = random.sample(pop, k)
        return min(selected, key=fitness)
    
    def crossover(parent1, parent2):
        crossover_point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        
        # Repair children
        child1 = np.clip(child1, 0, 1)
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = np.clip(child2, 0, 1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                mutated[i] += np.random.normal(0, 0.01)
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    for generation in range(num_generations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate fitness
        fitness_scores = [fitness(ind) for ind in population]
        
        # Track best individual
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fitness_scores[best_idx]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best
        elites = [population[i] for i in np.argsort(fitness_scores)[:2]]
        new_population.extend(elites)
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population)
            parent2 = tournament_selection(population)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final evaluation with projective refinement
    try:
        h_final = best_h.copy()
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points/2) / np.sum(h_final)
        c5_final = evaluate_erdos_solution(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (list(best_h), best_c5, n_points)
