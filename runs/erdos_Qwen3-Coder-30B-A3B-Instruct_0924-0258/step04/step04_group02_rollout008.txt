import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Convert to logit space for optimization
    def h_from_z(z, b):
        return 1.0 / (1.0 + np.exp(-z - b))
    
    def z_from_h(h, b):
        return np.log(h / (1 - h)) - b
    
    # Find appropriate b for given z
    def find_b(z):
        h = h_from_z(z, 0)
        b = 0.0
        target = n_points / 2.0
        current_sum = np.sum(h)
        if abs(current_sum - target) < 1e-10:
            return b
        
        # Binary search for b
        low_b, high_b = -10.0, 10.0
        for _ in range(100):
            b = (low_b + high_b) / 2.0
            h_test = h_from_z(z, b)
            current_sum = np.sum(h_test)
            if current_sum > target:
                high_b = b
            else:
                low_b = b
        return b
    
    # Convert to optimization space
    z_start = z_from_h(h_start, 0)
    b_start = find_b(z_start)
    z_start = z_start - b_start
    
    best_z = z_start.copy()
    best_b = b_start
    best_h_opt = h_from_z(best_z, best_b)
    
    # Optimization loop
    max_iterations = 2000
    learning_rate = 0.1
    
    for iteration in range(max_iterations):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Recompute h and constraint
        h_opt = h_from_z(best_z, best_b)
        # Ensure sum constraint is satisfied
        b_new = find_b(best_z)
        h_opt = h_from_z(best_z, b_new)
        
        # Evaluate the worst case overlap for all shifts
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        max_shift_idx = np.argmax(corr)
        max_overlap = corr[max_shift_idx] * dx
        
        # Compute gradient
        # Pseudo-gradient for overlap at the worst shift
        k_max = max_shift_idx - (n_points - 1)
        # Derivative of overlap with respect to h at k_max
        gradient_h = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max >= 0 and i + k_max < n_points:
                gradient_h[i] = 1 - 2*h_opt[i] - h_opt[i+k_max]
            elif i + k_max < 0:
                # Handle boundary case
                gradient_h[i] = 1 - 2*h_opt[i]
            else:
                gradient_h[i] = 1 - 2*h_opt[i]
        
        # Convert gradient to z space: dh/dz = h*(1-h)
        dh_dz = h_opt * (1 - h_opt)
        gradient_z = gradient_h * dh_dz
        
        # Update z
        z_new = best_z - learning_rate * gradient_z
        
        # Adjust b to maintain sum constraint
        b_new = find_b(z_new)
        if np.isnan(b_new):
            b_new = best_b
        z_new = z_new - b_new
        
        # Update best if improved
        h_new = h_from_z(z_new, b_new)
        c5_new = evaluate_erdos_solution(h_new, dx)
        
        if c5_new < best_c5:
            best_z = z_new.copy()
            best_b = b_new
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
    
    # Fine-tune with local optimization
    def objective(z):
        h = h_from_z(z, find_b(z))
        return evaluate_erdos_solution(h, dx)
        
    def constraint_sum(z):
        h = h_from_z(z, find_b(z))
        return np.sum(h) - n_points / 2.0
    
    bounds = [(-20, 20) for _ in range(n_points)]
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 10:
        try:
            z_ref = best_z + np.random.normal(0, 0.01, n_points)
            res = minimize(objective, z_ref, method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 50, 'ftol': 1e-8})
            if res.success:
                h_refined = h_from_z(res.x, find_b(res.x))
                c5_refined = evaluate_erdos_solution(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
