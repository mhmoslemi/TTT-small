import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search parameters
    lambda_low = 0.0
    lambda_high = 1.0
    tolerance = 1e-5
    
    # Feasibility check parameters
    max_iter = 1000
    learning_rate = 0.1
    
    def compute_auto_correlation(h):
        # Compute h(x)(1-h(x+k)) for all shifts k
        corr = np.correlate(h, 1 - h, mode='full')
        # Only keep the relevant part
        n_corr = len(corr)
        start_idx = (n_corr - n_points) // 2
        corr = corr[start_idx:start_idx + n_points]
        return corr
    
    def gradient_auto_correlation(h, k):
        # Compute gradient of auto correlation at shift k
        # h(x) * (1 - h(x + k)) with respect to h(x)
        # Only for degree 1 terms (linear in h)
        grad = np.zeros_like(h)
        if k >= 0:
            grad[k:] += (1 - h[k:])
            grad[:n_points-k] -= h[:n_points-k]
        else:
            k = -k
            grad[:k] -= h[k:]
            grad[k:] += (1 - h[:n_points-k])
        return grad
    
    def project_to_feasible(h):
        # Project h to [0,1] and normalize to sum = n_points/2
        h = np.clip(h, 0, 1)
        h = h * (n_points/2) / np.sum(h)
        return h
    
    def check_feasibility(h, lambda_val):
        # Check if h is feasible for lambda_val
        # i.e., all auto-correlations at least lambda_val
        corr = compute_auto_correlation(h)
        min_corr = np.min(corr)
        return min_corr >= lambda_val
    
    def project_h_to_constraint(h):
        # Project h to the constraint sum(h) * dx = 1
        h = h * (n_points/2) / np.sum(h)
        return h
    
    # Binary search on the minimum auto correlation
    while time() - start_time < budget_s * 0.95 and lambda_high - lambda_low > tolerance:
        lambda_mid = (lambda_low + lambda_high) / 2
        h_mid = best_h.copy()
        
        # Try to find a feasible solution
        converged = False
        for iter_num in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            
            corr = compute_auto_correlation(h_mid)
            min_k = np.argmin(corr)
            min_corr = corr[min_k]
            
            if min_corr >= lambda_mid:
                # Feasible, update best if better
                c5 = evaluate_erdos_solution(h_mid, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_mid.copy()
                # Success, move to higher lambda
                lambda_low = lambda_mid
                break
            
            # Not feasible, try to improve
            grad = gradient_auto_correlation(h_mid, min_k - n_points + 1)
            h_new = h_mid - learning_rate * grad
            
            # Line search for best step size
            step_size = 1.0
            for _ in range(20):
                h_test = h_mid - step_size * grad
                h_test = project_to_feasible(h_test)
                corr_test = compute_auto_correlation(h_test)
                if np.min(corr_test) >= lambda_mid:
                    h_mid = h_test
                    break
                step_size *= 0.5
            
            # Even if not perfect, reproject and advance
            h_mid = project_to_feasible(h_mid)
            
            # Check if we're stuck
            if iter_num > 100:
                # Try random perturbation if stuck
                if np.allclose(h_mid, best_h, atol=1e-5):
                    h_mid = np.random.uniform(0, 1, n_points)
                    h_mid = h_mid * (n_points/2) / np.sum(h_mid)
    
    # Final optimization step on best solution
    if time() - start_time < 60:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
