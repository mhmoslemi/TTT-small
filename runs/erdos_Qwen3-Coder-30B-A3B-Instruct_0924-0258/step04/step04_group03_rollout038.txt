import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary approach with gradient refinement
    population_size = 50
    mutation_rate = 0.05
    selection_rate = 0.2
    max_generations = 1000
    
    # Initialize population
    population = [h_start.copy()]
    for _ in range(population_size - 1):
        h_mut = h_start + np.random.normal(0, 0.05, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        population.append(h_mut)
    
    # Evolutionary loop
    generation = 0
    while generation < max_generations and (time() - start_time) < budget_s * 0.95:
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Keep top 20%
        num_keep = max(1, int(population_size * selection_rate))
        selected = population[:num_keep]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection with roulette wheel (biased towards better performers)
            p1 = selected[np.random.randint(0, len(selected))]
            p2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover: arithmetic mean
            alpha = np.random.rand()
            h_child = alpha * p1 + (1 - alpha) * p2
            
            # Mutation
            if np.random.rand() < 0.8:  # 80% chance of mutation
                mut_strength = max(0.001, mutation_rate * (1 - generation/max_generations))
                h_child += np.random.normal(0, mut_strength, n_points)
                h_child = np.clip(h_child, 0, 1)
            
            # Normalize
            h_child = h_child * (n_points / 2) / np.sum(h_child)
            new_population.append(h_child)
        
        population = new_population
        
        # Occasionally do one gradient refinement on top 5 individuals
        if generation % 5 == 0 and (time() - start_time) < budget_s * 0.9:
            top_individuals = population[:5]
            for h in top_individuals:
                # SLSQP optimization
                def objective(h_vec):
                    return evaluate_erdos_solution(h_vec, dx)
                
                def constraint_sum(h_vec):
                    return np.sum(h_vec) - n_points / 2.0
                
                bounds = [(0, 1) for _ in range(n_points)]
                cons_sum = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    res = minimize(objective, h, method='SLSQP',
                                   bounds=bounds, constraints=cons_sum,
                                   options={'maxiter': 100, 'ftol': 1e-7})
                    if res.success:
                        h_refined = res.x
                        c5_refined = objective(h_refined)
                        if c5_refined < best_c5:
                            best_h = h_refined
                            best_c5 = c5_refined
                except Exception:
                    pass
        
        generation += 1
    
    # Final gradient refinement on best solution
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def objective(h):
            return evaluate_erdos_solution(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons_sum = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(objective, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons_sum,
                           options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                best_h = res.x.copy()
                best_c5 = objective(best_h)
        except Exception:
            pass
    
    return (list(best_h), best_c5, n_points)
