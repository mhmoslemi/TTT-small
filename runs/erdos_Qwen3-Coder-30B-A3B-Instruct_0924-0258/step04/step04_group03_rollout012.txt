import numpy as np
from scipy.optimize import minimize
from time import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Population-based evolutionary approach
    def create_population(size, seed_h):
        """Create diverse population around seed"""
        population = [seed_h.copy()]
        for _ in range(size - 1):
            # Create perturbed version
            h = seed_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        return population
    
    def mutate(h, step_size):
        """Create mutated version"""
        h_new = h + np.random.normal(0, step_size, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    def crossover(h1, h2):
        """Arithmetic crossover"""
        alpha = np.random.random()
        h_new = alpha * h1 + (1 - alpha) * h2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Initialize population
    pop_size = 50
    population = create_population(pop_size, h_start)
    
    # Evolutionary parameters
    mutation_step = 0.05
    min_mutation_step = 0.001
    selection_rate = 0.2
    max_generations = 1000
    adapt_interval = 10
    
    # Track best
    best_h = population[0].copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    generation = 0
    while (time() - start_time) < budget_s * 0.95 and generation < max_generations:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_erdos_solution(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update global best
        if fitness_scores[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness_scores[0]
        
        # Selection
        num_keep = max(1, int(pop_size * selection_rate))
        selected = population[:num_keep]
        
        # Generate new population
        new_population = selected.copy()
        
        # Add mutated versions
        for _ in range(num_keep):
            parent = selected[np.random.randint(0, len(selected))]
            h_mut = mutate(parent, mutation_step)
            new_population.append(h_mut)
        
        # Add crossbred versions
        while len(new_population) < pop_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            h_cross = crossover(parent1, parent2)
            new_population.append(h_cross)
        
        # Ensure correct size
        population = new_population[:pop_size]
        
        # Adjust mutation rate
        if generation % adapt_interval == 0 and mutation_step > min_mutation_step:
            mutation_step *= 0.9
        
        generation += 1
    
    # Final gradient refinement of top candidates
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 40:
        top_candidates = population[:5]
        for h_candidate in top_candidates:
            # SLSQP refinement
            def objective(h):
                return evaluate_erdos_solution(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2.0
            
            bounds = [(0, 1) for _ in range(n_points)]
            cons_sum = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                res = minimize(objective, h_candidate, method='SLSQP', 
                               bounds=bounds, constraints=cons_sum, 
                               options={'maxiter': 100, 'ftol': 1e-7})
                if res.success:
                    h_refined = res.x
                    c5_refined = objective(h_refined)
                    if c5_refined < best_c5:
                        best_h = h_refined
                        best_c5 = c5_refined
            except:
                pass
    
    return (list(best_h), best_c5, n_points)
