import numpy as np
from scipy.optimize import minimize
from time import time

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 50
        # Create a structured initial guess with clusters
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        cluster_centers = [0.5, 1.0, 1.5]
        cluster_size = 8
        for c in cluster_centers:
            start = max(0, int((c - cluster_size/2)/dx))
            end = min(n_points, int((c + cluster_size/2)/dx))
            h_start[start:end] = 1.0 / (3 * cluster_size)
        # Normalize to ensure sum is n/2
        scale = (n_points/2) / np.sum(h_start)
        h_start *= scale
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Define constraint function for the sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use BOBYQA optimization
    # We define an objective function for minimize
    def obj(h):
        return evaluate_erdos_solution(h, dx)
    
    # Define constraint dictionary
    cons_sum = {'type': 'eq', 'fun': constraint_sum}
    
    # Try several restarts to escape local optima
    for restart in range(10):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Perturb the current best solution slightly
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Re-normalize the perturbed solution
        h_perturbed = h_perturbed * (n_points/2) / np.sum(h_perturbed)
        
        try:
            # Use SLSQP for refinement
            res = minimize(obj, h_perturbed, method='SLSQP', 
                           bounds=bounds, constraints=cons_sum, 
                           options={'maxiter': 300, 'ftol': 1e-6})
            if res.success:
                h_refined = res.x
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
            
        # Try another optimization approach (L-BFGS)
        try:
            res = minimize(obj, h_perturbed, method='L-BFGS-B', 
                           bounds=bounds, options={'maxiter': 200})
            if res.success:
                h_refined = res.x
                # Ensure it satisfies the sum constraint
                h_refined = h_refined * (n_points/2) / np.sum(h_refined)
                c5_refined = obj(h_refined)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except:
            pass
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points/2) / np.sum(best_h)
    
    return (list(best_h), best_c5, n_points)
