import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        n_points = 35
        h_start = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(h_start, dx)
    
    # Learnable parameters approach (gradient-based)
    def make_h_from_params(params):
        # Parameters: [n_points] weights + 1 bias
        weights = params[:-1]
        bias = params[-1]
        
        # Use a smooth approximation of the step function
        # via ReLU-like function
        h_raw = np.maximum(0, weights + bias)
        # Scale to correct integral
        h_scaled = h_raw * (n_points / 2) / np.sum(h_raw)
        # Clip to [0,1]
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped
    
    def objective(params):
        h = make_h_from_params(params)
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(params):
        h = make_h_from_params(params)
        return np.sum(h) - n_points / 2.0
    
    # Initialize parameters
    initial_weights = np.random.normal(0, 0.1, n_points)
    initial_bias = 0.0
    initial_params = np.append(initial_weights, initial_bias)
    
    # Single optimization run
    bounds = [(None, None)] * n_points + [(-2, 2)]  # Weights: free, bias: bounded
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Try different optimizers
    optimizers = ['SLSQP', 'L-BFGS-B', 'TNC']
    
    for opt in optimizers:
        if time() - start_time > budget_s * 0.9:
            break
        try:
            res = minimize(objective, initial_params, method=opt, 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-6})
            if res.success:
                h_opt = make_h_from_params(res.x)
                c5_opt = objective(res.x)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
        except Exception as e:
            continue
    
    # Local refinement using multiple restarts
    for restart in range(10):
        if time() - start_time > budget_s * 0.95:
            break
        # Generate random start
        weights_restart = np.random.normal(0, 0.5, n_points)
        bias_restart = np.random.normal(0, 0.5)
        params_restart = np.append(weights_restart, bias_restart)
        
        try:
            res = minimize(objective, params_restart, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-8})
            if res.success:
                h_try = make_h_from_params(res.x)
                c5_try = objective(res.x)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
        except:
            continue
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if abs(final_sum - n_points / 2.0) > 1e-6:
        # Renormalize
        best_h = best_h * (n_points / 2) / final_sum
    
    return (list(best_h), best_c5, n_points)
