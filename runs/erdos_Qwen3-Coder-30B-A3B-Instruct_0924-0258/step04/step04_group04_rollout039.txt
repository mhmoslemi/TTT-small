import numpy as np
from scipy.optimize import minimize
from time import time
import math

def evaluate_erdos_solution(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Create an initial guess
        n_points = 35
        h_start = np.zeros(n_points)
        dx = 2.0 / n_points
        # Spread with two peaks
        peak1_start = int(0.3 / dx)
        peak1_end = int(0.7 / dx)
        peak2_start = int(1.3 / dx)
        peak2_end = int(1.7 / dx)
        if peak1_end > peak1_start:
            h_start[peak1_start:peak1_end] = 1.0
        if peak2_end > peak2_start:
            h_start[peak2_start:peak2_end] = 1.0
        # Normalize
        h_start = h_start * (n_points/2) / np.sum(h_start)
    
    dx = 2.0 / n_points
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Binary search on λ to maximize the minimum auto-correlation
    tol = 1e-5
    lambda_low = 0.0
    lambda_high = 1.0
    lambda_mid = 0.5
    max_iter = 300
    
    # Helper function to compute constraints
    def compute_auto_correlations(h):
        # h: array of length n_points
        h_padded = np.concatenate([h, np.zeros(n_points)])
        ac = np.correlate(h_padded, h, mode='valid')[:n_points]
        return ac
    
    # Gradient descent to improve a solution for a specific lambda
    def improve_h(h_in, lambda_target, max_iter=1000, lr=0.1):
        h = h_in.copy()
        for iteration in range(max_iter):
            if time() - start_time > budget_s * 0.95:
                break
            # Compute auto-correlations
            ac = compute_auto_correlations(h)
            # Find k_min where ac is minimal
            k_min = np.argmin(ac)
            ac_min = ac[k_min]
            if ac_min >= lambda_target:
                break
                
            # Compute gradient of ac[k_min] with respect to h
            # d/dh ac[k_min] = sum_{j,k} delta[k_min + j - k] * h[j] * (1 - h[k])
            #              = h[k_min + j] * (1 - h[j]) for j in support(h)
            # But more precisely:
            # AC[k] = sum_i h[i] * (1 - h[i+k])
            # dAC[k]/dh[j] = - (1-h[j+k]) * delta[j, j'] + h[j+k] * delta[j+ k, j']
            # Since we compute for k_min, it's a bit more intricate.
            # So: let's recompute AC explicitly for small perturbations to estimate gradient
            
            # For efficiency, we use two-point finite difference estimation of gradient
            eps = 1e-5
            grad = np.zeros_like(h)
            h_plus = h.copy()
            h_minus = h.copy()
            for j in range(n_points):
                h_plus[j] = min(1.0, h[j] + eps)
                h_minus[j] = max(0.0, h[j] - eps)
                
                ac_plus = compute_auto_correlations(h_plus)
                ac_minus = compute_auto_correlations(h_minus)
                
                grad[j] = (ac_plus[k_min] - ac_minus[k_min]) / (2.0 * eps)
                
                # Reset
                h_plus[j] = h[j]
                h_minus[j] = h[j]
                
            # Update with gradient ascent (maximize ac_k_min)
            h = h + lr * grad
            # Project to feasible region
            h = np.clip(h, 0, 1)
            # Force sum constraint using normalization
            h = h * (n_points/2) / np.sum(h)
            
        return h
    
    # Binary search loop
    iter_count = 0
    for iteration in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        lambda_mid = (lambda_low + lambda_high) / 2.0
        
        # Try to improve solution for lambda_mid
        h_candidate = improve_h(best_h, lambda_mid)
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        if c5_candidate < lambda_mid and c5_candidate < best_c5:
            # This means improving didn't work or we're in the wrong direction
            lambda_high = lambda_mid
        elif c5_candidate < lambda_mid:
            # We weren't able to improve to satisfy lambda mid at the very worse case
            lambda_high = lambda_mid
        else:
            # We can satisfy lambda_mid constraint with this solution
            # So we want even higher lambda - update
            lambda_low = lambda_mid
            # This is a better solution (smaller overlap)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        iter_count += 1
    
    # Final refinement
    remaining_time = budget_s - (time() - start_time)
    if remaining_time > 30:
        def obj(h):
            return evaluate_erdos_solution(h, dx)
            
        def con_sum(h):
            return np.sum(h) - n_points/2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': con_sum}
        
        try:
            res = minimize(obj, best_h, method='SLSQP', 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 100, 'ftol': 1e-6})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    # One more deterministic optimization pass using L-BFGS
    if remaining_time > 30:
        try:
            def obj(h):
                return evaluate_erdos_solution(h, dx)
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, best_h, method='L-BFGS-B', 
                          bounds=bounds, options={'maxiter': 1000})
            if res.success:
                final_h = res.x
                final_c5 = evaluate_erdos_solution(final_h, dx)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass
    
    return (list(best_h), best_c5, n_points)
