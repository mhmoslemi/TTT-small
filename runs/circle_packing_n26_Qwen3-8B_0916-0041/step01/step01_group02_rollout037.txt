import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Step 1: Define initial configuration using a hexagonal grid
    # Number of rows: floor(sqrt(n))
    n = 26
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows

    # Generate initial centers
    centers = []
    for r in range(rows):
        for c in range(cols):
            x = c + 0.5 * (r % 2)
            y = r + 0.5
            if x < 1 and y < 1:
                centers.append([x, y])
    
    # Step 2: Convert to NumPy arrays
    centers = np.array(centers)
    radii = np.full(centers.shape[0], 0.05)

    # Step 3: Define objective function to maximize sum of radii
    def objective(x):
        # x is a flat array [x0, y0, r0, x1, y1, r1, ...]
        # Reconstruct centers and radii
        centers_flat = x[:2 * centers.shape[0]].reshape(-1, 2)
        radii_flat = x[2 * centers.shape[0]:]
        
        # Check if all radii are non-negative
        if np.any(radii_flat < 0):
            return -np.inf
        
        # Check if circles are within the unit square
        for i in range(len(radii_flat)):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        
        # Check if circles overlap
        for i in range(len(radii_flat)):
            for j in range(i + 1, len(radii_flat)):
                dx = centers_flat[i, 0] - centers_flat[j, 0]

                distance = np.sqrt(dx ** 2 + dy ** 2)

                    return -np.inf
        
        # Return the negative sum of radii for maximization
        return -np.sum(radii_flat)

    # Step 4: Set bounds for the optimization parameters
    # Each center has two coordinates in [0,1], and each radius is non-negative
    bounds = []
    for _ in range(centers.shape[0]):

    # Step 5: Run optimization
    result = optimize.minimize(
