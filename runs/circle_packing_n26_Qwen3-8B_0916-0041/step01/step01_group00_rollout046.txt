from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    h = np.sqrt(3) / 2  # hexagon height
    w = 1.0  # hexagon width
    
    # Generate initial centers in a hexagonal grid
    centers = []
    for row in range(5):
        for col in range(5 + (row % 2)):

            y = row * h

                x = col + 0.5

            x = x / w
            y = y / (h * 5)
            centers.append([x, y])
    centers = np.array(centers)
    
    # Initialize radii to a small value
    radii = np.full(26, 0.01)
    
    # Optimization: maximize sum of radii with constraints
    def objective(radii):

    def constraint_boundary(radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12

        return 0

    def constraint_overlap(radii):
