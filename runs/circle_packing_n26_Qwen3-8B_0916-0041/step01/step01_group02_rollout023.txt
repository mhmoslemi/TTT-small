import numpy as np
import cvxpy as cp

def run_packing():

    n = 26

    # Define variable for centers and radii
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize total radius sum
    obj = cp.Maximize(cp.sum(radii))

    # Constraints: Each circle is within the unit square
    constraints = []
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r <= 1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r <= 1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Constraints: No overlapping between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.OSQP, eps=1e-3)

    # Extract the results
    centers_val = centers.value
