from scipy.optimize import minimize

def run_packing():

    n_cols = int(np.ceil(N / n_rows))

    centers = np.zeros((N, 2))

        row = i // n_cols

    centers /= max(centers.max(), 1)

    def objective(radii):

    def constraint_bound(radii):
