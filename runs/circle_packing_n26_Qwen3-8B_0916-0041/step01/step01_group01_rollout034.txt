import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define the number of circles
    n = 26

    # Define initial grid for hexagonal packing
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows != 0 else 0)

    # Initial positions for a hexagonal grid
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row
        centers[i] = [x, y]

    # Initial radii values (equal for all)
    radii = np.full(n, 0.05)

    # Define constraints and bounds
    # Constraint: non-overlapping circles
    def non_overlap_constraints(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii_flat[i] + radii_flat[j]) >= 0)
        return np.array(constraints)

    # Boundary constraint: circles must be within the unit square
    def boundary_constraints(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        constraints = []
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            constraints.append(x_i - r_i >= 0)
            constraints.append(x_i + r_i <= 1)
            constraints.append(y_i - r_i >= 0)
            constraints.append(y_i + r_i <= 1)
        return np.array(constraints)

    # Define the objective: maximize sum of radii
    def objective(x):
        return -x[n*2:]  # Minimize negative of sum to maximize

    # Combine all constraints
    cons = []
    cons.append({'type': 'ineq', 'fun': non_overlap_constraints})
    cons.append({'type': 'ineq', 'fun': boundary_constraints})

    # Bounds: positions are in [0, 1], radii are non-negative
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)])  # for x and y of each circle
        bounds.append((0, 1))           # for radius of each circle

    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])

    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 500, 'disp': False}
    )

    # Extract final values
    centers_final = result.x[:n*2].reshape(n, 2)

    sum_radii = np.sum(radii_final)
    return (centers_final, radii_final, sum_radii)
