import scipy.optimize as optimize

def run_packing():

    # 26 circles:
    # - 1 in center
    # - 6 in first layer
    # - 12 in second layer
    # - 7 in third layer (adjusting)
    # Total: 1 + 6 + 12 + 7 = 26

    # We'll use a hexagonal grid with some variation for better fit within 1×1 square

    # Define required: centers (nx2), radii (n,)
    n_circles = 26

    centers = np.zeros((n_circles, 2))

    centers[0] = [0.5, 0.5]
    
    # First layer: 6 circles
    for i in range(1, 7):
        angle = 2 * np.pi * i / 6
        r = 0.2  # initial radius
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
    
    # Second layer: 12 circles
    for i in range(7, 19):
        angle = 2 * np.pi * i / 12
        r = 0.15
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
    
    # Third layer: 7 circles (to make total 26)
    for i in range(19, 26):
        angle = 2 * np.pi * i / 7
        r = 0.1
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
    
    # Initial guess for radii
    radii = np.full(n_circles, 0.1)

    # Define the bounds for optimization
    bounds = []
    for _ in range(n_circles):
        bounds.append((0, 1))  # Radius between 0 and 1
    
    # Define inequality constraints: center + radius <= 1 and center - radius >= 0
    # Total number of constraints = 2 * n_circles
    cons = []
    for i in range(n_circles):
        # x_i + r_i <= 1
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - (centers[i, 0] + r[i])})
        # x_i - r_i >= 0
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: (centers[i, 0] - r[i])})
    
    # Define pairwise distance constraints: for all i < j, distance >= r_i + r_j
    # Create a list of constraint functions
    for i in range(n_circles):

            def constraint(i=i, j=j):
                def c(r):
                    return np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (r[i] + r[j])

            cons.append({'type': 'ineq', 'fun': constraint})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
