    num_rows = 5
    num_cols = 5

            idx += 1

            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
        # Ensure circles are non-overlapping
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)

                    return False

    for _ in range(max_iterations):

        new_centers = points.copy()

        for i in range(len(new_centers)):

            max_radius = 0

        # Check if we can improve by shifting circles closer to boundaries
        for i in range(len(new_centers)):
            x, y = new_centers[i]
            r = new_radii[i]
            # Check if we can shift the circle closer to a boundary
            best_shift = 0

                dx, dy = dir
                new_x = x + dx * r
                new_y = y + dy * r
                # Check if this new position is valid
                valid = True

                        break
                if valid:
                    best_shift = max(best_shift, dx, dy)

            # Apply the best shift (only if it improves the packing)
            if best_shift != 0:
                new_centers[i] += best_shift * r

        # Update to new_centers and new_radii
        points = new_centers
        radii = new_radii

    # Final check and return
    centers = points
    radii = radii
