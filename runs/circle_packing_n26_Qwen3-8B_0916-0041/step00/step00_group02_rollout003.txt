def generate_hexagonal_packing(n, square_size=1.0):
    """
    Generate a hexagonal packing of n circles in a square of size square_size.
    """
    max_radius = 0.1
    spacing = 2 * max_radius
    rows = int(np.sqrt(n))
    cols = int(np.ceil(n / rows))
    
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            if x > square_size or y > square_size:
                continue
            centers.append((x, y))
    
    # Adjust to fit in square
    if not centers:
        return np.array([]), np.array([])
    
    centers = np.array(centers)
    scale = square_size / np.max(centers[:, 0])
    centers *= scale
    max_radius = np.min(np.min(centers, axis=0) * 0.5)
    
    return centers, np.full(len(centers), max_radius)

def run_packing():

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

            if dist < radii[i] + radii[j] - 1e-12:
