import numpy as np
import scipy.optimize as opt

def run_packing():

    n = 26

    # We'll generate a hexagonal grid with random offsets to avoid perfect symmetry

    rows = int(np.ceil(np.sqrt(n)))  # Number of rows
    cols = int(np.ceil(n / rows))  # Number of columns

    # Base grid positions
    grid = []

            x = j + 0.5 * (i % 2)  # Offset for staggered rows

            grid.append((x, y))

    jitter = 0.05
    centers = np.array([[x + np.random.uniform(-jitter, jitter), y + np.random.uniform(-jitter, jitter)] for x, y in grid])

    radii = np.full(n, 0.05)

    # Objective function: Maximize the sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint functions:
    # 1. All circles must be within [0,1]^2

    # 3. Radii must be non-negative

        return c - r

    def constraint_boundary(c, r):
        return c - r

    # Define constraints for circles
    constraints = []
    for i in range(n):

                return np.sqrt(dx**2 + dy**2) - (r[i] + r[j])
