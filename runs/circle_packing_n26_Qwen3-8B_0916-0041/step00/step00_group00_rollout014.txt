import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Densify the grid to allow a hybrid layout
    n = 26

    num_rows = 8
    row_sizes = [3, 5, 4, 4, 4, 3, 3, 3]

    centers = []
    for i in range(num_rows):
        row_size = row_sizes[i]

            x = j / (row_size - 1)
            y = i / (num_rows - 1)
            centers.append([x, y])

    centers = np.array(centers)

    def obj(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def constraint(radii, centers, i, j):
        # Return the squared distance between centers[i] and centers[j]
        return (centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2 - (radii[i] + radii[j])**2

    # Prepare constraints
    cons = []
    for i in range(n):

                'type': 'ineq',

    bounds = [(0, 1) for _ in range(n)]

    # Optimize
