```
@triton.jit
def _layer_norm_kernel(
    x_ptr, y_ptr, gamma_ptr, beta_ptr,
    M,
    C: tl.constexpr,
    eps,
    BLOCK_T: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid_t = tl.program_id(0)
    b = tl.program_id(1)
    t0 = pid_t * BLOCK_T
    t = t0 + tl.arange(0, BLOCK_T)
    tmask = t < M
    base = b * M * C
    x_base = x_ptr + base
    y_base = y_ptr + base
    s = tl.zeros((BLOCK_T,), dtype=tl.float32)
    sq = tl.zeros((BLOCK_T,), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C
        mask = tmask[:, None] & cmask[None, :]
        x = tl.load(x_base + t[:, None] * C + c[None, :], mask=mask, other=0.0).to(tl.float32)
        s += tl.sum(x, axis=1)
        sq += tl.sum(x * x, axis=1)
    mean = s / C
    var = sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)
    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C
        mask = tmask[:, None] & cmask[None, :]
        x = tl.load(x_base + t[:, None] * C + c[None, :], mask=mask, other=0.0).to(tl.float32)
        g = tl.load(gamma_ptr + c, mask=cmask, other=0.0)
        bet = tl.load(beta_ptr + c, mask=cmask, other=0.0)
        y = (x - mean[:, None]) * rstd[:, None] * g[None, :] + bet[None, :]
        tl.store(y_base + t[:, None] * C + c[None, :], y.to(tl.float16), mask=mask)
```

```
@triton.jit
def _prod_lr_kernel(
    x_ptr, L_ptr, R_ptr,
    wl_ptr, wr_ptr, wgl_ptr, wgr_ptr, mask_ptr,
    M, C: tl.constexpr, H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_T: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
    scale,
):
    pid_t = tl.program_id(0)
    pid_h = tl.program_id(1)
    b = tl.program_id(2)
    t0 = pid_t * BLOCK_T
    t = t0 + tl.arange(0, BLOCK_T)
    tmask = t < M
    h0 = pid_h * BLOCK_H
    h = h0 + tl.arange(0, BLOCK_H)
    hmask = h < H
    acc_l = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    acc_gl = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    acc_gr = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    x_base = x_ptr + b * M * C
    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C
        x_mask = tmask[:, None] & cmask[None, :]
        w_mask = hmask[:, None] & cmask[None, :]
        x = tl.load(x_base + t[:, None] * C + c[None, :], mask=x_mask, other=0.0)
        xT = tl.trans(x)
        wl = tl.load(wl_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        wgl = tl.load(wgl_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        wr = tl.load(wr_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        wgr = tl.load(wgr_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        acc_l = tl.dot(wl, xT, acc_l)
        acc_gl = tl.dot(wgl, xT, acc_gl)
        acc_r = tl.dot(wr, xT, acc_r)
        acc_gr = tl.dot(wgr, xT, acc_gr)
    if HAS_MASK:
        mt = tl.load(mask_ptr + b * M + t, mask=tmask, other=0.0).to(tl.float32)
    else:
        mt = tl.ones((BLOCK_T,), dtype=tl.float32)
    gl = 1.0 / (1.0 + tl.exp(-acc_gl))
    gr = 1.0 / (1.0 + tl.exp(-acc_gr))
    left = (acc_l * gl * mt[None, :]) * scale
    right = (acc_r * gr * mt[None, :]) * scale
    out_offs = b * H * M + h[:, None] * M + t[None, :]
    store_mask = hmask[:, None] & tmask[None, :]
    tl.store(L_ptr + out_offs, left.to(tl.float16), mask=store_mask)
    tl.store(R_ptr + out_offs, right.to(tl.float16), mask=store_mask)
```

```
@triton.jit
def _final_recompute_kernel(
    out_h_ptr, x_ptr, out_ptr,
    wg_ptr, wo_ptr, lnw_ptr, lnb_ptr,
    M, C: tl.constexpr, H: tl.constexpr,
    eps, BLOCK_T: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid_t = tl.program_id(0)
    b = tl.program_id(1)
    t = pid_t*BLOCK_T + tl.arange(0, BLOCK_T)
    tmask = t < M
    h = tl.arange(0, H)
    out_idx = h[:, None]*M + t[None, :]
    out_mask = tmask[None, :]
    out_base = out_h_ptr + b*H*M
    out16 = tl.load(out_base + out_idx, mask=out_mask, other=0.0)
    out = out16.to(tl.float32)
    s = tl.sum(out, axis=0)
    sq = tl.sum(out * out, axis=0)
    mean = s / H
    var = sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)
    og = tl.zeros((H, BLOCK_T), dtype=tl.float32)
    x_base = x_ptr + b * M * C
    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C
        x = tl.load(x_base + t[:, None]*C + c[None, :], mask=tmask[:,None] & cmask[None,:], other=0.0)
        xT = tl.trans(x)
        wg = tl.load(wg_ptr + h[:,None]*C + c[None,:], mask=cmask[None,:], other=0.0)
        og = tl.dot(wg, xT, og)
    gamma = tl.load(lnw_ptr + h, mask=(h < H), other=1.0)
    beta = tl.load(lnb_ptr + h, mask=(h < H), other=0.0)
    out32 = out16.to(tl.float32)
    y = (out32 - mean[None,:]) * rstd[None,:] * gamma[:,None] + beta[None,:]
    gate = 1.0 / (1.0 + tl.exp(-og))
    y = y * gate
    y16 = y.to(tl.float16)
    yT = tl.trans(y16)
    out_base2 = out_ptr + b * M * C
```

```python
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

LN_EPS = 1e-5
PROD_SCALE = 0.1  # scales L and R before the N^3 contraction; second LN is scale-invariant


@triton.jit
def _layer_norm_kernel(
    x_ptr, y_ptr, gamma_ptr, beta_ptr,
    M,
    C: tl.constexpr,
    eps,
    BLOCK_T: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """LayerNorm over the last dimension, writing float16 output."""
    pid_t = tl.program_id(0)
    b = tl.program_id(1)

    t0 = pid_t * BLOCK_T
    t = t0 + tl.arange(0, BLOCK_T)
    tmask = t < M

    base = b * M * C
    x_base = x_ptr + base
    y_base = y_ptr + base

    s = tl.zeros((BLOCK_T,), dtype=tl.float32)
    sq = tl.zeros((BLOCK_T,), dtype=tl.float32)

    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C
        mask = tmask[:, None] & cmask[None, :]
        x = tl.load(x_base + t[:, None] * C + c[None, :], mask=mask, other=0.0).to(tl.float32)
        s += tl.sum(x, axis=1)
        sq += tl.sum(x * x, axis=1)

    mean = s / C
    var = sq / C - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)

    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C
        mask = tmask[:, None] & cmask[None, :]
        x = tl.load(x_base + t[:, None] * C + c[None, :], mask=mask, other=0.0).to(tl.float32)
        g = tl.load(gamma_ptr + c, mask=cmask, other=0.0)
        bet = tl.load(beta_ptr + c, mask=cmask, other=0.0)
        y = (x - mean[:, None]) * rstd[:, None] * g[None, :] + bet[None, :]
        tl.store(y_base + t[:, None] * C + c[None, :], y.to(tl.float16), mask=mask)


@triton.jit
def _prod_lr_kernel(
    x_ptr, L_ptr, R_ptr,
    wl_ptr, wr_ptr, wgl_ptr, wgr_ptr, mask_ptr,
    M, C: tl.constexpr, H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_T: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
    scale,
):
    """
    Compute masked/gated left and right projections directly in hidden-major layout:
      L[b, h, i, k] = (LN(x) W_left^T)[b, i, k, h] * sigmoid(gate_left) * mask[b, i, k]
      R[b, h, j, k] = (LN(x) W_right^T)[b, j, k, h] * sigmoid(gate_right) * mask[b, j, k]
    A global scale is applied to both to keep the large N^3 contraction well-conditioned
    in fp16; the second LayerNorm is scale-invariant.
    """
    pid_t = tl.program_id(0)
    pid_h = tl.program_id(1)
    b = tl.program_id(2)

    t0 = pid_t * BLOCK_T
    t = t0 + tl.arange(0, BLOCK_T)
    tmask = t < M

    h0 = pid_h * BLOCK_H
    h = h0 + tl.arange(0, BLOCK_H)
    hmask = h < H

    acc_l = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    acc_gl = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    acc_r = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)
    acc_gr = tl.zeros((BLOCK_H, BLOCK_T), dtype=tl.float32)

    x_base = x_ptr + b * M * C

    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        cmask = c < C

        x_mask = tmask[:, None] & cmask[None, :]
        w_mask = hmask[:, None] & cmask[None, :]

        x = tl.load(x_base + t[:, None] * C + c[None, :], mask=x_mask, other=0.0)
        xT = tl.trans(x)

        wl = tl.load(wl_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        wgl = tl.load(wgl_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        wr = tl.load(wr_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)
        wgr = tl.load(wgr_ptr + h[:, None] * C + c[None, :], mask=w_mask, other=0.0)

        acc_l = tl.dot(wl, xT, acc_l)
        acc_gl = tl.dot(wgl, xT, acc_gl)
        acc_r = tl.dot(wr, xT, acc_r)
        acc_gr = tl.dot(wgr, xT, acc_gr)

    if HAS_MASK:
        mt = tl.load(mask_ptr + b * M + t, mask=tmask, other=0.0).to(tl.float32)
    else:
        mt = tl.ones((BLOCK_T,), dtype=tl.float32)

    gl = 1.0 / (1.0 + tl.exp(-acc_gl))
    gr = 1.0 / (1.0 + tl.exp(-acc_gr))

    left = (acc_l * gl * mt[None, :]) * scale
    right = (acc_r * gr * mt[None, :]) * scale

    out_offs = b * H * M + h[:, None] * M + t[None, :]
    store_mask = hmask[:, None] & tmask[None, :]

    tl.store(L_ptr + out_offs, left.to(tl.float16), mask=store_mask)
    tl.store(R_ptr + out_offs, right.to(tl.float16), mask=store_mask)


@triton.jit
def _final_recompute_kernel(
    out_h_ptr, x_ptr, out_ptr,
    wg_ptr, wo_ptr, lnw_ptr, lnb_ptr,
    M, C_IN: tl.constexpr, C_OUT: tl.constexpr, H: tl.constexpr,
    eps,
    BLOCK_T: tl.constexpr, BLOCK_C_IN: tl.constexpr, BLOCK_C_OUT: tl.constexpr,
):
    """
    Fused final stage for H <= 128 (power-of-two):
      - recompute out_gate = sigmoid(LN(x) W_out_gate^T)
      - LN over hidden dim from hidden-major contraction output
      - apply gate
      - final linear to output channels
    """
    pid_t = tl.program_id(0)
    b = tl.program_id(1)

    t = pid_t * BLOCK_T + tl.arange(0, BLOCK_T)
    tmask = t < M
    h = tl.arange(0, H)

    out_base = out_h_ptr + b * H * M
    out_idx = h[:, None] * M + t[None, :]
    out_mask = tmask[None, :]

    out16 = tl.load(out_base + out_idx, mask=out_mask, other=0.0)
    out = out16.to(tl.float32)

    s = tl.sum(out, axis=0)
    sq = tl.sum(out * out, axis=0)
    mean = s / H
    var = sq / H - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + eps)

    og = tl.zeros((H, BLOCK_T), dtype=tl.float32)
    x_base = x_ptr + b * M * C_IN

    for c0 in range(0, C_IN, BLOCK_C_IN):
        c = c0 + tl.arange(0, BLOCK_C_IN)
        cmask = c < C_IN
        x_mask = tmask[:, None] & cmask[None, :]

        x = tl.load(x_base + t[:, None] * C_IN + c[None, :], mask=x_mask, other=0.0)
        xT = tl.trans(x)
        wg = tl.load(wg_ptr + h[:, None] * C_IN + c[None, :], mask=cmask[None, :], other=0.0)
        og = tl.dot(wg, xT, og)

    gamma = tl.load(lnw_ptr + h, mask=(h < H), other=1.0)
    beta = tl.load(lnb_ptr + h, mask=(h < H), other=0.0)

    out32 = out16.to(tl.float32)
    y = (out32 - mean[None, :]) * rstd[None, :] * gamma[:, None] + beta[None, :]
    gate = 1.0 / (1.0 + tl.exp(-og))
    y = y * gate
    y16 = y.to(tl.float16)
    yT = tl.trans(y16)

    out_base2 = out_ptr + b * M * C_OUT

    for co in range(0, C_OUT, BLOCK_C_OUT):
        c_out = co + tl.arange(0, BLOCK_C_OUT)
        cmask_out = c_out < C_OUT

        w_out = tl.load(
            wo_ptr + h[:, None] * C_OUT + c_out[None, :],
            mask=cmask_out[None, :],
            other=0.0,
        )
        acc = tl.dot(yT, w_out).to(tl.float32)
        tl.store(
            out_base2 + t[:, None] * C_OUT + c_out[None, :],
            acc,
            mask=tmask[:, None] & cmask_out[None, :],
        )


def _block_c(c: int) -> int:
    c = int(c)
    if c <= 0:
        return 16
    return min(128, max(16, triton.next_power_of_2(c)))


def _block_h(h: int) -> int:
    h = int(h)
    if h <= 0:
        return 16
    if h <= 128:
        return max(16, triton.next_power_of_2(h))
    return 128


def custom_kernel(data):
    """
    Triton-optimized forward pass for AlphaFold3-style TriMul.

    Pipeline:
      1) Triton LN(input) -> fp16 x_norm
      2) Triton fused left/right projection + sigmoid gates + mask -> hidden-major fp16 L/R
      3) cuBLAS batched matmul over hidden dimension:
             out[b, h, i, j] = sum_k L[b, h, i, k] * R[b, h, j, k]
      4) Triton fused second LN + out_gate + final linear -> fp32 output
    """
    if len(data) == 4:
        x, mask, weights, config = data
    elif len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        raise ValueError("custom_kernel expects data of length 3 or 4")

    if config is None:
        config = {}

    x = x.contiguous()
    B = int(x.shape[0])
    N = int(x.shape[1])
    C_IN = int(x.shape[-1])
    M = N * N
    device = x.device

    H = int(weights["left_proj.weight"].shape[0])
    C_OUT = int(weights["to_out.weight"].shape[0])

    norm_w = weights["norm.weight"].to(torch.float32).contiguous()
    norm_b = weights["norm.bias"].to(torch.float32).contiguous()

    W_l = weights["left_proj.weight"].to(torch.float16).contiguous().view(H, C_IN)
    W_r = weights["right_proj.weight"].to(torch.float16).contiguous().view(H, C_IN)
    W_gl = weights["left_gate.weight"].to(torch.float16).contiguous().view(H, C_IN)
    W_gr = weights["right_gate.weight"].to(torch.float16).contiguous().view(H, C_IN)
    W_og = weights["out_gate.weight"].to(torch.float16).contiguous().view(H, C_IN)
    W_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous().view(H, C_OUT)

    ln2_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    ln2_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()

    use_mask = isinstance(mask, torch.Tensor) and mask.dim() >= 1 and not bool(config.get("nomask", False))
    if use_mask:
        mask = mask.reshape(B, M).to(torch.float32).contiguous()
        HAS_MASK = True
    else:
        mask = torch.empty(1, dtype=torch.float32, device=device)
        HAS_MASK = False

    x_flat = x.reshape(B, M, C_IN)
    xh = torch.empty(B, M, C_IN, dtype=torch.float16, device=device)

    BLOCK_C_IN = _block_c(C_IN)
    BLOCK_C_OUT = _block_c(C_OUT)
    BLOCK_H = _block_h(H)

    # 1) LayerNorm -> fp16
    BLOCK_T_LN = 128
    grid_ln = (triton.cdiv(M, BLOCK_T_LN), B)
    _layer_norm_kernel[grid_ln](
        x_flat, xh, norm_w, norm_b,
        M,
        C=C_IN,
        eps=LN_EPS,
        BLOCK_T=BLOCK_T_LN,
        BLOCK_C=BLOCK_C_IN,
        num_warps=8,
        num_stages=2,
    )

    # 2) Fused L/R projection + gates + mask, hidden-major fp16
    L_h = torch.empty(B, H, M, dtype=torch.float16, device=device)
    R_h = torch.empty(B, H, M, dtype=torch.float16, device=device)

    BLOCK_T_PROD = 64
    grid_prod = (triton.cdiv(M, BLOCK_T_PROD), triton.cdiv(H, BLOCK_H), B)
    prod_stages = 2 if C_IN <= 128 else 1
    _prod_lr_kernel[grid_prod](
        xh, L_h, R_h,
        W_l, W_r, W_gl, W_gr,
        mask,
        M,
        C=C_IN,
        H=H,
        HAS_MASK=HAS_MASK,
        BLOCK_T=BLOCK_T_PROD,
        BLOCK_H=BLOCK_H,
        BLOCK_C=BLOCK_C_IN,
        scale=PROD_SCALE,
        num_warps=8,
        num_stages=prod_stages,
    )

    # 3) Batched contraction over sequence dimension, per hidden channel
    Lb = L_h.reshape(B * H, N, N)
    Rb = R_h.reshape(B * H, N, N)
    out_mat = torch.bmm(Lb, Rb.transpose(1, 2))
    out_h = out_mat.reshape(B, H, M)

    del L_h, R_h, Lb, Rb

    # 4) Final LN + gate + projection
    # Since L and R were both scaled by PROD_SCALE, the contraction is scaled by PROD_SCALE^2.
    eps_final = LN_EPS * (PROD_SCALE * PROD_SCALE) ** 2

    use_triton_final = (
        H >= 16
        and H <= 128
        and H == triton.next_power_of_2(H)
    )

    if use_triton_final:
        out = torch.empty(B, N, N, C_OUT, dtype=torch.float32, device=device)
        BLOCK_T_FIN = 64
        grid_final = (triton.cdiv(M, BLOCK_T_FIN), B)
        _final_recompute_kernel[grid_final](
            out_h, xh, out,
            W_og, W_out_T, ln2_w, ln2_b,
            M,
            C_IN=C_IN,
            C_OUT=C_OUT,
            H=H,
            eps=eps_final,
            BLOCK_T=BLOCK_T_FIN,
            BLOCK_C_IN=BLOCK_C_IN,
            BLOCK_C_OUT=BLOCK_C_OUT,
            num_warps=8,
            num_stages=2,
        )
    else:
        # Correct generic path for uncommon hidden dims / non-power-of-two H.
        out_n = out_h.permute(0, 2, 1).contiguous().reshape(B, M, H).to(torch.float32)

        W_og_T = W_og.t().contiguous()
        gate = torch.sigmoid(torch.matmul(xh, W_og_T).to(torch.float32))

        out_n = F.layer_norm(out_n, (H,), ln2_w, ln2_b, eps=eps_final)
        y = (out_n * gate).to(torch.float16)
        out = torch.matmul(y, W_out_T).to(torch.float32).reshape(B, N, N, C_OUT)

    return out
```
