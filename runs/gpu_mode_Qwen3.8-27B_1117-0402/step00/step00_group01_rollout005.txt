```python
import torch
import triton
import triton.language as tl


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _trimul_input_kernel(
    x_ptr,
    mask_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    M,
    N2,
    H_N2,
    eps,
    C: tl.constexpr,
    H: tl.constexpr,
    C_PAD: tl.constexpr,
    BLOCK_P: tl.constexpr,
    HCH: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Fused:
      - LayerNorm over C
      - five linear projections to H
      - sigmoid gating
      - mask application to left/right features
    and stores L/R/OG in [B, H, N, N] layout for batched pair GEMM.
    """
    pid = tl.program_id(0)
    p = pid * BLOCK_P + tl.arange(0, BLOCK_P)
    p_mask = p < M
    p64 = p.to(tl.int64)

    offs_c = tl.arange(0, C_PAD)
    c_mask = offs_c < C

    x_ptrs = x_ptr + p64[:, None] * C + offs_c[None, :]
    x = tl.load(x_ptrs, mask=p_mask[:, None] & c_mask[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    xc = x - mean[:, None]
    xc = tl.where(c_mask[None, :], xc, 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    wn = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    wb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)

    xn = (x - mean[:, None]) * rstd[:, None] * wn[None, :] + wb[None, :]
    xn = tl.where(c_mask[None, :], xn, 0.0).to(tl.bfloat16)

    if HAS_MASK:
        m = tl.load(mask_ptr + p64, mask=p_mask, other=0.0).to(tl.float32)
    else:
        m = tl.zeros((BLOCK_P,), dtype=tl.float32) + 1.0

    b = p64 // N2
    rem = p64 - b * N2
    pos = b * H_N2 + rem

    for h0 in tl.static_range(0, H, HCH):
        h = h0 + tl.arange(0, HCH)
        h_mask = h < H
        w_mask = c_mask[:, None] & h_mask[None, :]

        wl = tl.load(wl_ptr + h[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0).to(tl.bfloat16)
        wr = tl.load(wr_ptr + h[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0).to(tl.bfloat16)
        wlg = tl.load(wlg_ptr + h[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0).to(tl.bfloat16)
        wrg = tl.load(wrg_ptr + h[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0).to(tl.bfloat16)
        wog = tl.load(wog_ptr + h[None, :] * C + offs_c[:, None], mask=w_mask, other=0.0).to(tl.bfloat16)

        l = tl.dot(xn, wl)
        r = tl.dot(xn, wr)
        lg = tl.dot(xn, wlg)
        rg = tl.dot(xn, wrg)
        og = tl.dot(xn, wog)

        lg = tl.sigmoid(lg)
        rg = tl.sigmoid(rg)
        og = tl.sigmoid(og)

        l = l * m[:, None] * lg
        r = r * m[:, None] * rg

        st_mask = h_mask[:, None] & p_mask[None, :]
        off = pos[None, :] + h[:, None] * N2

        tl.store(l_ptr + off, tl.trans(l.to(tl.bfloat16)), mask=st_mask)
        tl.store(r_ptr + off, tl.trans(r.to(tl.bfloat16)), mask=st_mask)
        tl.store(og_ptr + off, tl.trans(og), mask=st_mask)


@triton.jit
def _trimul_pairwise_kernel(
    l_ptr,
    r_ptr,
    out_ptr,
    N,
    N2,
    NUM_M,
    NUM_N,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BK: tl.constexpr,
    EVEN: tl.constexpr,
):
    """
    For each (b, h): out[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k].
    This is a batched GEMM with K = N.
    """
    pid = tl.program_id(0)

    pid_m = pid % NUM_M
    tmp = pid // NUM_M
    pid_n = tmp % NUM_N
    bh = tmp // NUM_N

    base = bh.to(tl.int64) * N2

    offs_i = pid_m * BM + tl.arange(0, BM)
    offs_j = pid_n * BN + tl.arange(0, BN)
    offs_k = tl.arange(0, BK)

    if EVEN:
        a_ptrs = l_ptr + base + offs_i[:, None] * N + offs_k[None, :]
        b_ptrs = r_ptr + base + offs_j[:, None] * N + offs_k[None, :]
        acc = tl.zeros((BM, BN), dtype=tl.float32)

        for _ in range(0, N, BK):
            a = tl.load(a_ptrs)
            b = tl.load(b_ptrs)
            acc = tl.dot(a, tl.trans(b), acc)
            a_ptrs += BK
            b_ptrs += BK

        out_ptrs = out_ptr + base + offs_i[:, None] * N + offs_j[None, :]
        tl.store(out_ptrs, acc)
    else:
        i_mask = offs_i < N
        j_mask = offs_j < N

        a_ptrs = l_ptr + base + offs_i[:, None] * N + offs_k[None, :]
        b_ptrs = r_ptr + base + offs_j[:, None] * N + offs_k[None, :]
        acc = tl.zeros((BM, BN), dtype=tl.float32)

        for k0 in range(0, N, BK):
            k_mask = (k0 + offs_k) < N
            a = tl.load(a_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
            acc = tl.dot(a, tl.trans(b), acc)
            a_ptrs += BK
            b_ptrs += BK

        out_ptrs = out_ptr + base + offs_i[:, None] * N + offs_j[None, :]
        tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _trimul_out_kernel(
    out_ptr,
    gate_ptr,
    y_ptr,
    w_ptr,
    ln_w_ptr,
    ln_b_ptr,
    N,
    H_N2,
    N2_C,
    eps,
    H: tl.constexpr,
    C: tl.constexpr,
    HCH: tl.constexpr,
    CCH: tl.constexpr,
    BJ: tl.constexpr,
):
    """
    Fused final step:
      - LayerNorm over H for each (b, i, j)
      - multiply by out_gate
      - linear projection H -> C
    Reads out/gate in [B, H, N, N] layout, writes final [B, N, N, C].
    """
    pid = tl.program_id(0)

    num_j = (N + BJ - 1) // BJ
    row = pid // num_j
    jblk = pid % num_j

    b = row // N
    i = row % N

    p = jblk * BJ + tl.arange(0, BJ)
    p_mask = p < N
    p_global = i * N + p

    b64 = b.to(tl.int64)
    base = b64 * H_N2

    h = tl.arange(0, HCH)

    if H <= HCH:
        h_mask = h < H
        offs = base + h[:, None] * N2 + p_global[None, :]

        ov = tl.load(out_ptr + offs, mask=h_mask[:, None] & p_mask[None, :], other=0.0).to(tl.float32)
        gate = tl.load(gate_ptr + offs, mask=h_mask[:, None] & p_mask[None, :], other=0.0).to(tl.float32)

        mean = tl.sum(ov, axis=0) / H
        cent = tl.where(h_mask[:, None], ov - mean[None, :], 0.0)
        var = tl.sum(cent * cent, axis=0) / H
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + eps)

        lw = tl.load(ln_w_ptr + h, mask=h_mask, other=0.0).to(tl.float32)
        lb = tl.load(ln_b_ptr + h, mask=h_mask, other=0.0).to(tl.float32)

        hn = (ov - mean[None, :]) * rstd[None, :] * lw[:, None] + lb[:, None]
        gated = tl.where(h_mask[:, None], hn * gate, 0.0).to(tl.bfloat16)

        for c0 in tl.static_range(0, C, CCH):
            cc = c0 + tl.arange(0, CCH)
            c_mask = cc < C

            w = tl.load(
                w_ptr + cc[:, None] * H + h[None, :],
                mask=c_mask[:, None] & h_mask[None, :],
                other=0.0,
            ).to(tl.bfloat16)

            yT = tl.dot(w, gated)
            y = tl.trans(yT)

            y_off = b64 * N2_C + p_global[None, :] * C + cc[None, :]
            tl.store(y_ptr + y_off, y, mask=p_mask[None, :] & c_mask[None, :])
    else:
        total_sum = tl.zeros((BJ,), dtype=tl.float32)
        total_sq = tl.zeros((BJ,), dtype=tl.float32)

        for h0 in tl.static_range(0, H, HCH):
            hh = h0 + h
            hm = hh < H
            offs = base + hh[:, None] * N2 + p_global[None, :]
            ov = tl.load(out_ptr + offs, mask=hm[:, None] & p_mask[None, :], other=0.0).to(tl.float32)
            total_sum += tl.sum(ov, axis=0)
            total_sq += tl.sum(ov * ov, axis=0)

        mean = total_sum / H
        var = total_sq / H - mean * mean
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + eps)

        for c0 in tl.static_range(0, C, CCH):
            cc = c0 + tl.arange(0, CCH)
            c_mask = cc < C
            yT = tl.zeros((CCH, BJ), dtype=tl.float32)

            for h0 in tl.static_range(0, H, HCH):
                hh = h0 + h
                hm = hh < H
                offs = base + hh[:, None] * N2 + p_global[None, :]

                ov = tl.load(out_ptr + offs, mask=hm[:, None] & p_mask[None, :], other=0.0).to(tl.float32)
                gate = tl.load(gate_ptr + offs, mask=hm[:, None] & p_mask[None, :], other=0.0).to(tl.float32)

                lw = tl.load(ln_w_ptr + hh, mask=hm, other=0.0).to(tl.float32)
                lb = tl.load(ln_b_ptr + hh, mask=hm, other=0.0).to(tl.float32)

                hn = (ov - mean[None, :]) * rstd[None, :] * lw[:, None] + lb[:, None]
                gated = tl.where(hm[:, None], hn * gate, 0.0).to(tl.bfloat16)

                w = tl.load(
                    w_ptr + cc[:, None] * H + hh[None, :],
                    mask=c_mask[:, None] & hm[None, :],
                    other=0.0,
                ).to(tl.bfloat16)

                yT = tl.dot(w, gated, yT)

            y = tl.trans(yT)
            y_off = b64 * N2_C + p_global[None, :] * C + cc[None, :]
            tl.store(y_ptr + y_off, y, mask=p_mask[None, :] & c_mask[None, :])


def custom_kernel(data):
    """
    Fused Triton forward for outgoing TriMul.

    Pipeline:
      1. Triton: input LayerNorm + projections + gates + mask, stored as [B,H,N,N].
      2. Triton: batched triangular pair update O(B*H*N^3).
      3. Triton: output LayerNorm + out gate + projection to final [B,N,N,C].
    """
    if len(data) == 4:
        x, mask, weights, config = data
    else:
        x, weights, config = data
        mask = None

    x = x.detach().contiguous()
    B, N, _, C_in = x.shape
    H = int(weights["left_proj.weight"].shape[0])
    C_out = int(weights["to_out.weight"].shape[0])
    eps = float(config.get("eps", 1e-5))

    N2 = N * N
    M = B * N2
    H_N2 = H * N2
    N2_C = N2 * C_out

    has_mask = mask is not None and not bool(config.get("nomask", False))
    if has_mask:
        mask_flat = mask.detach().to(torch.float32).contiguous().view(-1)
    else:
        mask_flat = x  # dummy pointer; not loaded when HAS_MASK=False

    def f32(name):
        return weights[name].detach().to(torch.float32).contiguous()

    def bf16(name):
        return weights[name].detach().to(torch.bfloat16).contiguous()

    norm_w = f32("norm.weight")
    norm_b = f32("norm.bias")

    wl = bf16("left_proj.weight")
    wr = bf16("right_proj.weight")
    wlg = bf16("left_gate.weight")
    wrg = bf16("right_gate.weight")
    wog = bf16("out_gate.weight")

    to_out_w = bf16("to_out.weight")
    to_out_norm_w = f32("to_out_norm.weight")
    to_out_norm_b = f32("to_out_norm.bias")

    dev = x.device

    L = torch.empty((B, H, N, N), device=dev, dtype=torch.bfloat16)
    R = torch.empty((B, H, N, N), device=dev, dtype=torch.bfloat16)
    OG = torch.empty((B, H, N, N), device=dev, dtype=torch.float32)

    C_PAD = _next_pow2(C_in)
    HCH = 128

    if C_PAD <= 128:
        BLOCK_P = 64
    elif C_PAD <= 512:
        BLOCK_P = 32
    else:
        BLOCK_P = 16

    grid = (triton.cdiv(M, BLOCK_P),)
    _trimul_input_kernel[grid](
        x,
        mask_flat,
        L,
        R,
        OG,
        norm_w,
        norm_b,
        wl,
        wr,
        wlg,
        wrg,
        wog,
        M,
        N2,
        H_N2,
        eps,
        C_in,
        H,
        C_PAD,
        BLOCK_P,
        HCH,
        has_mask,
        num_warps=8,
    )

    out_main = torch.empty((B, H, N, N), device=dev, dtype=torch.float32)

    BM = 128
    BN = 128
    BK = 64
    num_m = triton.cdiv(N, BM)
    num_n = triton.cdiv(N, BN)
    even = (N % BM == 0) and (N % BN == 0) and (N % BK == 0)

    grid = (B * H * num_m * num_n,)
    _trimul_pairwise_kernel[grid](
        L,
        R,
        out_main,
        N,
        N2,
        num_m,
        num_n,
        BM,
        BN,
        BK,
        even,
        num_warps=8,
        num_stages=3,
    )

    del L, R

    out = torch.empty((B, N, N, C_out), device=dev, dtype=torch.float32)

    BJ = 128
    CCH = 128
    num_j = triton.cdiv(N, BJ)
    grid = (B * N * num_j,)

    _trimul_out_kernel[grid](
        out_main,
        OG,
        out,
        to_out_w,
        to_out_norm_w,
        to_out_norm_b,
        N,
        H_N2,
        N2_C,
        eps,
        H,
        C_out,
        HCH,
        CCH,
        BJ,
        num_warps=8,
    )

    return out
```
