def trimul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H, B,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,

    num_pid_j = tl.cdiv(N, BLOCK_J)

    num_bh = B * H

    pid_ij = pid % num_ij
    pid_i = pid_ij // num_pid_j
    pid_j = pid_ij % num_pid_j

import triton
import triton.language as tl

@triton.jit
def trimul_core_kernel(

    N, stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_B: tl.constexpr,
):

    Computes out[b, i, j] = sum_k left[b, i, k] * right[b, j, k]
    where left, right, out all have shape [NUM_B, N, N]
    (NUM_B = B * H after permuting)
    """

    num_pid_i = tl.cdiv(N, BLOCK_I)
    num_pid_j = tl.cdiv(N, BLOCK_J)
    num_ij = num_pid_i * num_pid_j
    
    pid_b = pid // num_ij
    pid_ij = pid % num_ij
    pid_i = pid_ij // num_pid_j
    pid_j = pid_ij % num_pid_j
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # For permuted layout [B, H, N, N], treating B*H as one dim:

    b_ptrs = right_ptr + pid_b * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_pid_i = tl.cdiv(N, BLOCK_I)
    num_pid_j = tl.cdiv(N, BLOCK_J)
    num_ij = num_pid_i * num_pid_j
    
    pid_b = pid // num_ij
    pid_ij = pid % num_ij
    pid_i = pid_ij // num_pid_j
    pid_j = pid_ij % num_pid_j
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    a_ptrs = left_ptr + pid_b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
    b_ptrs = right_ptr + pid_b * stride_rb + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a = tl.load(a_ptrs, mask=mask_i[:, None] & k_mask[None, :], other=0.0)
        b_tile = tl.load(b_ptrs, mask=mask_j[:, None] & k_mask[None, :], other=0.0)
        acc = tl.dot(a, tl.trans(b_tile), acc)
        a_ptrs += BLOCK_K * stride_lk
        b_ptrs += BLOCK_K * stride_rk
    
    out_ptrs = out_ptr + pid_b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

    left_gate_weight = weights["left_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)  # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    right = right * mask_expanded

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    
    left = left * left_gate
    right = right * right_gate

    # left: [B, N, N, H] -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.bfloat16)
    
    # Batch dimension for the kernel

    # Flatten the batch*hidden dimension
    left_flat = left_perm.view(NUM_B, N, N)
    right_flat = right_perm.view(NUM_B, N, N)
    
    out_flat = torch.empty(NUM_B, N, N, device=device, dtype=torch.float32)
    
    # Launch kernel
    BLOCK_I = 64

        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
