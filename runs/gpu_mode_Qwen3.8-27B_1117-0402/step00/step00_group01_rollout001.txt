```
def fused_projections_kernel(
    x_ptr, mask_ptr,
    left_w_ptr, right_w_ptr, left_gw_ptr, right_gw_ptr, out_gw_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_size,  # B * N * N
    C: tl.constexpr, H: tl.constexpr,
    stride_x, stride_mask,
    stride_l, stride_r, stride_g,
    BLOCK_T: tl.constexpr,  # block over the sequence dimension
):
    pid = tl.program_id(0)
    offs_t = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    mask_t = offs_t < total_size
    
    offs_c = tl.arange(0, C)
    offs_h = tl.arange(0, H)
    
    # Load x: [BLOCK_T, C]
    x = tl.load(x_ptr + offs_t[:, None] * stride_x + offs_c[None, :], mask=mask_t[:, None], other=0.0)
    
    # Load mask: [BLOCK_T]
    m = tl.load(mask_ptr + offs_t * stride_mask, mask=mask_t, other=0.0)
    
    # Compute projections
    # left = x @ left_w.T: [BLOCK_T, H]
    left_w = tl.load(left_w_ptr + offs_h[:, None] * C + offs_c[None, :])  # [H, C]
    left = tl.dot(x, tl.trans(left_w))  # [BLOCK_T, H]
    
    # right = x @ right_w.T: [BLOCK_T, H]
    right_w = tl.load(right_w_ptr + offs_h[:, None] * C + offs_c[None, :])  # [H, C]
    right = tl.dot(x, tl.trans(right_w))  # [BLOCK_T, H]
    
    # Gates
    left_gw = tl.load(left_gw_ptr + offs_h[:, None] * C + offs_c[None, :])
    left_gate = tl.sigmoid(tl.dot(x, tl.trans(left_gw)))
    
    right_gw = tl.load(right_gw_ptr + offs_h[:, None] * C + offs_c[None, :])
    right_gate = tl.sigmoid(tl.dot(x, tl.trans(right_gw)))
    
    out_gw = tl.load(out_gw_ptr + offs_h[:, None] * C + offs_c[None, :])
    out_gate = tl.sigmoid(tl.dot(x, tl.trans(out_gw)))
    
    # Apply mask and gates
    left = left * m[:, None] * left_gate
    right = right * m[:, None] * right_gate
    
    # Store
    tl.store(left_ptr + offs_t[:, None] * stride_l + offs_h[None, :], left, mask=mask_t[:, None])
    tl.store(right_ptr + offs_t[:, None] * stride_r + offs_h[None, :], right, mask=mask_t[:, None])
    tl.store(out_gate_ptr + offs_t[:, None] * stride_g + offs_h[None, :], out_gate, mask=mask_t[:, None])
```

```python
    ln_w_ptr, ln_b_ptr, to_out_w_ptr,
    final_ptr,
    total_size,
    H: tl.constexpr, C: tl.constexpr,
    BLOCK_T: tl.constexpr, BLOCK_C_OUT: tl.constexpr,
    eps: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)
    offs_t = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    t_mask = offs_t < total_size
    offs_h = tl.arange(0, H)
    
    # Load out and out_gate
    out = tl.load(out_ptr + offs_t[:, None] * H + offs_h[None, :], mask=t_mask[:, None], other=0.0)
    og = tl.load(out_gate_ptr + offs_t[:, None] * H + offs_h[None, :], mask=t_mask[:, None], other=0.0)
    
    # LayerNorm
    mean = tl.sum(out, axis=1) / H
    x_centered = out - mean[:, None]
    var = tl.sum(x_centered * x_centered, axis=1) / H
    rstd = 1.0 / tl.sqrt(var + eps)
    
    ln_w = tl.load(ln_w_ptr + offs_h)
    ln_b = tl.load(ln_b_ptr + offs_h)
    
    normed = x_centered * rstd[:, None] * ln_w[None, :] + ln_b[None, :]
    gated = normed * og
    
    # Linear projection in chunks
    for c_off in range(0, C, BLOCK_C_OUT):
        offs_c = c_off + tl.arange(0, BLOCK_C_OUT)
        c_mask = offs_c < C
        w_chunk = tl.load(to_out_w_ptr + offs_c[None, :] * H + offs_h[:, None], mask=c_mask[None, :], other=0.0)
        result_chunk = tl.dot(gated, w_chunk)  # [BLOCK_T, BLOCK_C_OUT]
        tl.store(final_ptr + offs_t[:, None] * C + offs_c[None, :], result_chunk, 
                 mask=t_mask[:, None] & c_mask[None, :])
```
