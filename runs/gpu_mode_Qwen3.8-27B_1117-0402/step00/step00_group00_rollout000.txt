```python
@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,  # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,  # right strides
    stride_ob, stride_oi, stride_oj, stride_od,  # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # pid_batch is over B*H
    # pid_m is over N (rows of left, i.e., i dimension)
    # pid_n is over N (cols of right, i.e., j dimension)
    
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    
    pid_batch = pid // (num_pid_m * num_pid_n)
    pid_mn = pid % (num_pid_m * num_pid_n)
    pid_m = pid_mn // num_pid_n
    pid_n = pid_mn % num_pid_n
    
    b = pid_batch // H
    d = pid_batch % H
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # left[b, i, k, d] - we need to load blocks of [BLOCK_M, BLOCK_K]
    # right[b, j, k, d] - we need to load blocks of [BLOCK_N, BLOCK_K]
    # out[b, i, j, d] - we store blocks of [BLOCK_M, BLOCK_N]
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        
        # left[b, offs_m, offs_k, d]
        l_ptrs = left_ptr + b * stride_lb + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk + d * stride_ld
        # right[b, offs_n, offs_k, d]
        r_ptrs = right_ptr + b * stride_rb + offs_n[:, None] * stride_rj + offs_k[None, :] * stride_rk + d * stride_rd
        
        l_block = tl.load(l_ptrs)
        r_block = tl.load(r_ptrs)
        
        # We want: acc[m, n] += sum_k l[m, k] * r[n, k]
        # This is l_block @ r_block^T
        acc += tl.dot(l_block, tl.trans(r_block))
    
    # Store
    o_ptrs = out_ptr + b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj + d * stride_od
    tl.store(o_ptrs, acc.to(out_ptr.dtype.element_type))
```

```python
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. Input LayerNorm
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=1e-5)

    # 2. Linear projections
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # 3. Apply mask
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_expanded
        right = right * mask_expanded

    # 4. Apply sigmoid gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # 5. Core einsum via Triton: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    out = torch.empty(B, N, N, hidden_dim, device=input_tensor.device, dtype=torch.float32)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    s_lb = N * N * hidden_dim
    s_li = N * hidden_dim
    s_lk = hidden_dim
    s_ld = 1
    s_rb = N * N * hidden_dim
    s_rj = N * hidden_dim
    s_rk = hidden_dim
    s_rd = 1
    s_ob = N * N * hidden_dim
    s_oi = N * hidden_dim
    s_oj = hidden_dim
    s_od = 1

    total_batch = B * hidden_dim
    num_pid_m = triton.cdiv(N, BLOCK_M)
    num_pid_n = triton.cdiv(N, BLOCK_N)
    grid = (total_batch * num_pid_m * num_pid_n,)

    _trimul_einsum_kernel[grid](
        left, right, out,
        N, hidden_dim,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # 6. Output LayerNorm
    out = F.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps=1e-5)

    # 7. Apply out gate
    out = out * out_gate

    # 8. Final projection
    out = F.linear(out, to_out_weight)

    return out
```
