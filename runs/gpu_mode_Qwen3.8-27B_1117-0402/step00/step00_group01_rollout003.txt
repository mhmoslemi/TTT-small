def final_fused_kernel(core_ptr, og_ptr, wto_ptr, lnw_ptr, lnb_ptr, out_ptr,

    local_m = offs_m - b*N2

```python
"""
Fused Triton forward pass for the outgoing TriMul module.

This implementation fuses:
1. input LayerNorm,
2. left/right projection + gating + masking,
3. outgoing-gate projection,
4. the batched product sum over the sequence dimension via cuBLAS bmm,
5. output LayerNorm + outgoing_gate + final linear projection.

The heavy middle operation is computed as:
    out[b, i, j, :] = sum_k left[b, i, k, :] * right[b, j, k, :]
by converting left/right into [B, H, N, N] planes and using a batched
matrix multiplication:
    left_plane @ right_plane^T.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def _project_gate_kernel(
    x_ptr,
    mask_ptr,
    ln_w_ptr,
    ln_b_ptr,
    w_left_ptr,
    w_left_gate_ptr,
    w_right_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    left_plane_ptr,
    right_plane_ptr,
    out_gate_ptr,
    M,
    N2,
    C,
    H,
    eps,
    x_b_stride,
    x_m_stride,
    mask_b_stride,
    mask_m_stride,
    left_plane_b_stride,
    left_plane_h_stride,
    right_plane_b_stride,
    right_plane_h_stride,
    out_gate_b_stride,
    out_gate_m_stride,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    H_BLOCKS: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < M

    b = offs_m // N2
    local_m = offs_m - b * N2

    offs_c = tl.arange(0, BLOCK_C)
    c_mask = offs_c < C
    x_load_mask = m_mask[:, None] & c_mask[None, :]

    x = tl.load(
        x_ptr
        + b[:, None] * x_b_stride
        + local_m[:, None] * x_m_stride
        + offs_c[None, :],
        mask=x_load_mask,
        other=0.0,
    ).to(tl.float32)

    # Input LayerNorm over the last dimension C.
    mean = tl.sum(x, axis=1) / C
    xc = x - mean[:, None]
    xc = tl.where(c_mask[None, :], xc, 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    ln_w = tl.load(ln_w_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)
    ln_b = tl.load(ln_b_ptr + offs_c, mask=c_mask, other=0.0).to(tl.float32)

    # Normalized input used for all projections.
    a = (xc * rstd[:, None] * ln_w[None, :] + ln_b[None, :]).to(tl.bfloat16)

    if HAS_MASK:
        mask_vals = tl.load(
            mask_ptr + b * mask_b_stride + local_m * mask_m_stride,
            mask=m_mask,
            other=0.0,
        ).to(tl.float32)
    else:
        mask_vals = tl.zeros((BLOCK_M,), dtype=tl.float32) + 1.0

    for hb in range(H_BLOCKS):
        h0 = hb * BLOCK_H
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H
        w_mask = c_mask[:, None] & h_mask[None, :]
        store_plane_mask = h_mask[:, None] & m_mask[None, :]
        store_gate_mask = m_mask[:, None] & h_mask[None, :]

        w_left = tl.load(
            w_left_ptr + offs_c[:, None] * H + (h0 + offs_h[None, :]),
            mask=w_mask,
            other=0.0,
        ).to(tl.bfloat16)
        acc_left = tl.dot(a, w_left)

        w_left_gate = tl.load(
            w_left_gate_ptr + offs_c[:, None] * H + (h0 + offs_h[None, :]),
            mask=w_mask,
            other=0.0,
        ).to(tl.bfloat16)
        acc_left_gate = tl.dot(a, w_left_gate)

        left = acc_left * _sigmoid(acc_left_gate) * mask_vals[:, None]
        tl.store(
            left_plane_ptr
            + b[None, :] * left_plane_b_stride
            + local_m[None, :]
            + offs_h[:, None] * left_plane_h_stride,
            tl.trans(left.to(tl.bfloat16)),
            mask=store_plane_mask,
        )

        w_right = tl.load(
            w_right_ptr + offs_c[:, None] * H + (h0 + offs_h[None, :]),
            mask=w_mask,
            other=0.0,
        ).to(tl.bfloat16)
        acc_right = tl.dot(a, w_right)

        w_right_gate = tl.load(
            w_right_gate_ptr + offs_c[:, None] * H + (h0 + offs_h[None, :]),
            mask=w_mask,
            other=0.0,
        ).to(tl.bfloat16)
        acc_right_gate = tl.dot(a, w_right_gate)

        right = acc_right * _sigmoid(acc_right_gate) * mask_vals[:, None]
        tl.store(
            right_plane_ptr
            + b[None, :] * right_plane_b_stride
            + local_m[None, :]
            + offs_h[:, None] * right_plane_h_stride,
            tl.trans(right.to(tl.bfloat16)),
            mask=store_plane_mask,
        )

        w_out_gate = tl.load(
            w_out_gate_ptr + offs_c[:, None] * H + (h0 + offs_h[None, :]),
            mask=w_mask,
            other=0.0,
        ).to(tl.bfloat16)
        acc_out_gate = tl.dot(a, w_out_gate)
        out_gate = _sigmoid(acc_out_gate).to(tl.bfloat16)

        tl.store(
            out_gate_ptr
            + b[:, None] * out_gate_b_stride
            + local_m[:, None] * out_gate_m_stride
            + offs_h[None, :],
            out_gate,
            mask=store_gate_mask,
        )


@triton.jit
def _final_fused_kernel(
    core_ptr,
    out_gate_ptr,
    w_to_out_ptr,
    ln_w_ptr,
    ln_b_ptr,
    out_ptr,
    M,
    N2,
    C,
    eps,
    core_b_stride,
    core_h_stride,
    out_gate_b_stride,
    out_gate_m_stride,
    out_b_stride,
    out_m_stride,
    w_to_out_h_stride,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    H: tl.constexpr,
    C_BLOCKS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < M

    b = offs_m // N2
    local_m = offs_m - b * N2

    offs_h = tl.arange(0, H)

    core = tl.load(
        core_ptr
        + b[None, :] * core_b_stride
        + offs_h[:, None] * core_h_stride
        + local_m[None, :],
        mask=m_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(core, axis=0) / H
    xc = core - mean[None, :]
    var = tl.sum(xc * xc, axis=0) / H
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    ln_w = tl.load(ln_w_ptr + offs_h).to(tl.float32)
    ln_b = tl.load(ln_b_ptr + offs_h).to(tl.float32)

    out_gate = tl.load(
        out_gate_ptr
        + b[:, None] * out_gate_b_stride
        + local_m[:, None] * out_gate_m_stride
        + offs_h[None, :],
        mask=m_mask[:, None],
        other=0.0,
    ).to(tl.float32)

    z = (tl.trans(xc) * rstd[None, :] * ln_w[None, :] * out_gate + ln_b[None, :]).to(
        tl.bfloat16
    )

    for cb in range(C_BLOCKS):
        c0 = cb * BLOCK_C
        offs_c = c0 + tl.arange(0, BLOCK_C)
        c_mask = offs_c < C

        w = tl.load(
            w_to_out_ptr + offs_h[:, None] * w_to_out_h_stride + offs_c[None, :],
            mask=c_mask[None, :],
            other=0.0,
        ).to(tl.bfloat16)

        acc = tl.dot(z, w)

        out_ptrs = (
            out_ptr
            + b[:, None] * out_b_stride
            + local_m[:, None] * out_m_stride
            + offs_c[None, :]
        )
        tl.store(out_ptrs, acc.to(tl.float32), mask=m_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    x = x.contiguous()
    device = x.device

    B, N, _, C = x.shape
    N2 = N * N
    M = B * N2

    H = int(weights["left_proj.weight"].shape[0])
    C_out = int(weights["to_out.weight"].shape[0])

    eps = 1e-5
    bf16 = torch.bfloat16
    fp32 = torch.float32

    # Weight layout for the fused projection kernel: [C, H], row-major,
    # so that A[M, C] @ W[C, H] is a standard row-major GEMM.
    ln_w = weights["norm.weight"].to(fp32).contiguous()
    ln_b = weights["norm.bias"].to(fp32).contiguous()

    W_left = weights["left_proj.weight"].to(bf16).t().contiguous()
    W_left_gate = weights["left_gate.weight"].to(bf16).t().contiguous()
    W_right = weights["right_proj.weight"].to(bf16).t().contiguous()
    W_right_gate = weights["right_gate.weight"].to(bf16).t().contiguous()
    W_out_gate = weights["out_gate.weight"].to(bf16).t().contiguous()

    # Mask handling.
    has_mask = mask is not None
    if has_mask:
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask[..., 0]
        if mask.dim() == 2:
            mask = mask.unsqueeze(0)
        if mask.shape[0] == 1 and B > 1:
            mask = mask.expand(B, N, N).contiguous()
        elif mask.numel() == B * N2:
            mask = mask.view(B, N, N).contiguous()
        else:
            mask = mask.contiguous()
        mask_b_stride = mask.stride(0)
        mask_m_stride = mask.stride(2)
        mask_arg = mask
    else:
        mask_b_stride = 0
        mask_m_stride = 0
        mask_arg = x

    # Temporary tensors.
    left_plane = torch.empty((B, H, N, N), dtype=bf16, device=device)
    right_plane = torch.empty((B, H, N, N), dtype=bf16, device=device)
    out_gate = torch.empty((B, N, N, H), dtype=bf16, device=device)

    # Fused LayerNorm + projection + gating + masking.
    BLOCK_C = max(16, triton.next_power_of_2(C))
    BLOCK_M = 64 if C <= 128 else 32
    BLOCK_H = min(128, max(16, triton.next_power_of_2(H)))
    H_BLOCKS = triton.cdiv(H, BLOCK_H)

    grid = (triton.cdiv(M, BLOCK_M),)
    _project_gate_kernel[grid](
        x,
        mask_arg,
        ln_w,
        ln_b,
        W_left,
        W_left_gate,
        W_right,
        W_right_gate,
        W_out_gate,
        left_plane,
        right_plane,
        out_gate,
        M,
        N2,
        C,
        H,
        eps,
        x.stride(0),
        x.stride(2),
        mask_b_stride,
        mask_m_stride,
        H * N2,
        N2,
        H * N2,
        N2,
        N2 * H,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        BLOCK_H=BLOCK_H,
        H_BLOCKS=H_BLOCKS,
        HAS_MASK=has_mask,
        num_warps=4,
    )

    # Batched matrix product over sequence dimension N.
    lp = left_plane.view(B * H, N, N)
    rp = right_plane.view(B * H, N, N)
    core = torch.bmm(lp, rp.transpose(1, 2))  # [B*H, N, N]
    core4 = core.view(B, H, N, N)

    # Fast fused final LayerNorm + gate + linear for common power-of-two H.
    if H <= 256 and triton.next_power_of_2(H) == H:
        W_to_out = weights["to_out.weight"].to(bf16).t().contiguous()  # [H, C_out]
        final_ln_w = weights["to_out_norm.weight"].to(fp32).contiguous()
        final_ln_b = weights["to_out_norm.bias"].to(fp32).contiguous()

        out = torch.empty((B, N, N, C_out), dtype=fp32, device=device)

        F_BLOCK_M = 64 if H <= 128 else 32
        F_BLOCK_C = max(16, min(128, triton.next_power_of_2(C_out)))
        C_BLOCKS = triton.cdiv(C_out, F_BLOCK_C)

        final_grid = (triton.cdiv(M, F_BLOCK_M),)
        _final_fused_kernel[final_grid](
            core,
            out_gate,
            W_to_out,
            final_ln_w,
            final_ln_b,
            out,
            M,
            N2,
            C_out,
            eps,
            core4.stride(0),
            core4.stride(1),
            out_gate.stride(0),
            out_gate.stride(2),
            out.stride(0),
            out.stride(2),
            W_to_out.stride(0),
            BLOCK_M=F_BLOCK_M,
            BLOCK_C=F_BLOCK_C,
            H=H,
            C_BLOCKS=C_BLOCKS,
            num_warps=4,
        )
        return out

    # Generic fallback for unusual hidden dimensions. The main kernels are still
    # used for all projections and gating; this path only handles the final
    # normalization and linear map.
    core_last = core4.permute(0, 2, 3, 1).contiguous().to(fp32)
    out_gate_f = out_gate.to(fp32)

    final_ln_w = weights["to_out_norm.weight"].to(fp32).contiguous()
    final_ln_b = weights["to_out_norm.bias"].to(fp32).contiguous()

    normed = F.layer_norm(core_last, (H,), final_ln_w, final_ln_b, eps=eps)
    gated = normed * out_gate_f

    W_to_out = weights["to_out.weight"].to(fp32).contiguous()
    out = F.linear(gated, W_to_out)
    return out.view(B, N, N, C_out).contiguous()
```
