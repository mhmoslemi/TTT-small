import torch
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    # strides for left [B, N, N, H]
    lb, li, lk, ld,
    # strides for right [B, N, N, H]  
    rb, rj, rk, rd,
    # strides for out [B, N, N, H]
    ob, oi, oj, od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_I_BLOCKS: tl.constexpr, NUM_J_BLOCKS: tl.constexpr,
):
    pid = tl.program_id(0)
    
    # Decompose: i_block fastest, then j_block, then d, then b
    i_block = pid % NUM_I_BLOCKS
    pid1 = pid // NUM_I_BLOCKS
    j_block = pid1 % NUM_J_BLOCKS
    pid2 = pid1 // NUM_J_BLOCKS
    d = pid2 % H
    b = pid2 // H
    
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # left[b, i, k, d]
    left_ptrs = left_ptr + b*lb + offs_i[:, None]*li + (offs_k[None, :]) * lk + d*ld
    # right[b, j, k, d]
    right_ptrs = right_ptr + b*rb + offs_j[:, None]*rj + offs_k[None, :] * rk + d*rd
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        l_mask = (offs_i < N)[:, None] & k_mask[None, :]
        r_mask = (offs_j < N)[:, None] & k_mask[None, :]
        
        left_tile = tl.load(left_ptrs, mask=l_mask, other=0.0)
        right_tile = tl.load(right_ptrs, mask=r_mask, other=0.0)
        
        acc = tl.dot(left_tile, tl.trans(right_tile), acc)
        
        left_ptrs += BLOCK_K * lk
        right_ptrs += BLOCK_K * rk
    
    # Store out[b, i, j, d]
    out_ptrs = out_ptr + b*ob + offs_i[:, None]*oi + offs_j[None, :]*oj + d*od
    out_mask = (offs_i < N)[:, None] & (offs_j < N)[None, :]
    tl.store(out_ptrs, acc.to(out_ptr.dtype.element_type), mask=out_mask)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    B, N, N_, C = input_tensor.shape
    assert N == N_
    H = hidden_dim
    
    device = input_tensor.device
    
    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Step 1: LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (C,), norm_weight, norm_bias, eps=1e-5)
    
    # Step 2: Projections (in bf16 for speed)
    x_bf16 = x.to(torch.bfloat16)
    
    left = torch.nn.functional.linear(x_bf16, left_proj_weight.to(torch.bfloat16))
    right = torch.nn.functional.linear(x_bf16, right_proj_weight.to(torch.bfloat16))
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_bf16, left_gate_weight.to(torch.bfloat16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_bf16, right_gate_weight.to(torch.bfloat16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_bf16, out_gate_weight.to(torch.bfloat16)))
    
    # Step 3: Apply mask and gates
    mask_expanded = mask.unsqueeze(-1).to(torch.bfloat16)
    left = left * mask_expanded * left_gate

import triton
import triton.language as tl

"""
TriMul (Triangle Multiplicative Update) - AlphaFold3 core operation.
Implements the "outgoing" TriMul forward pass with:
1. Fused preprocessing in PyTorch (LayerNorm + 5 Linear projections + sigmoid gates + mask)
2. Triton kernel for the O(N^3 * H) einsum (batched matmul over d dimension)
3. Fused postprocessing in PyTorch (LayerNorm + gate + Linear projection)

@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    lid,  # stride_ld (1 usually)
    N_stride,  # N*H (stride for i or j dimension)
    H_stride,  # H (stride for k dimension)

    NUM_I_BLOCKS: tl.constexpr, NUM_J_BLOCKS: tl.constexpr,
):

    # Decompose pid: i_block (fastest), j_block, d, b (slowest)
    i_block = pid % NUM_I_BLOCKS
    pid1 = pid // NUM_I_BLOCKS
    j_block = pid1 % NUM_J_BLOCKS
    pid2 = pid1 // NUM_J_BLOCKS
    d = pid2 % H
    b = pid2 // H

    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # For the tile: i = offs_i, k = k_start + offs_k, d = d (fixed)

        r_mask = (offs_j < N)[:, None] & k_mask[None, :]
