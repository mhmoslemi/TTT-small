```python
import torch
import triton
import triton.language as tl


"""
TriMul outgoing forward.

Fused pipeline:
1) Triton fused LayerNorm + left/right projections/gates + out_gate.
   The projected tensors are written in a transposed [B, N_i, H, N_k] layout so
   that the hot TriMul core reads contiguous sequence blocks for each hidden dim.
2) Triton core:
   For every hidden feature h, compute out_h = L_h @ R_h^T over the sequence
   dimension K using bf16/fp16 tensor-core dots.
3) Triton fused final LayerNorm + out_gate + to_out linear, producing float32
   output [B, N, N, C].
"""


def _next_pow2(x: int) -> int:
    n = 1
    while n < x:
        n <<= 1
    return n


def _div_block(n: int, desired: int) -> int:
    """Largest power-of-two divisor of n not exceeding desired (>=1)."""
    if n <= 0:
        return 1
    b = min(desired, _next_pow2(n))
    while b > 1 and n % b != 0:
        b //= 2
    return b


def _block(n: int, desired: int, min_block: int = 16) -> int:
    """Power-of-two block near desired, divisible by n when possible."""
    if n <= 0:
        return min_block
    b = min(desired, _next_pow2(n))
    while b > min_block and n % b != 0:
        b //= 2
    if b < min_block:
        b = min_block
    return b


@triton.jit
def _sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))


@triton.jit
def stage1_fused_dn_proj_kernel(
    x_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    gate_ptr,
    norm_w_ptr,
    norm_b_ptr,
    w_left_proj_ptr,
    w_left_gate_ptr,
    w_right_proj_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    M, N, C, H,
    C_PAD: tl.constexpr,
    H_BLOCK: tl.constexpr,
    BM: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EVEN_M: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = (pid * BM + tl.arange(0, BM)).to(tl.int64)
    row_valid = offs_m < M

    col = tl.arange(0, C_PAD)
    col_valid = col < C

    # Load the row block and compute LayerNorm statistics over C.
    x_ptrs = x_ptr + offs_m[:, None] * C + col[None, :]
    x = tl.load(x_ptrs, mask=row_valid[:, None] & col_valid[None, :], other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    xc = tl.where(col_valid[None, :], x - mean[:, None], 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    norm_w = tl.load(norm_w_ptr + col, mask=col_valid, other=0.0).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + col, mask=col_valid, other=0.0).to(tl.float32)

    y = (x - mean[:, None]) * rstd[:, None] * norm_w[None, :] + norm_b[None, :]
    y = tl.where(col_valid[None, :], y, 0.0)
    y_f16 = y.to(tl.float16)

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_m, mask=row_valid, other=1.0).to(tl.float32)
    else:
        mask_val = tl.full((BM,), 1.0, tl.float32)
    mask_val = mask_val[:, None]

    row_div = (offs_m // N).to(tl.int64)
    row_mod = (offs_m % N).to(tl.int64)
    if EVEN_M:
        base_div = ((pid * BM) // N).to(tl.int64)

    # Iterate over hidden-feature blocks. y_f16 is reused for all five linears.
    for h0 in range(0, H, H_BLOCK):
        h_off = h0 + tl.arange(0, H_BLOCK)
        h_valid = h_off < H

        # Weight tile is loaded in natural [H_BLOCK, C_PAD] form (coalesced along C),
        # then transposed to [C_PAD, H_BLOCK] for tl.dot.
        h_col_mask = h_valid[:, None] & col_valid[None, :]

        wp = w_left_proj_ptr + h_off[:, None] * C + col[None, :]
        w_left_proj = tl.trans(tl.load(wp, mask=h_col_mask, other=0.0).to(tl.float16))
        acc_left_proj = tl.dot(y_f16, w_left_proj, out_dtype=tl.float32)

        wp = w_left_gate_ptr + h_off[:, None] * C + col[None, :]
        w_left_gate = tl.trans(tl.load(wp, mask=h_col_mask, other=0.0).to(tl.float16))
        acc_left_gate = tl.dot(y_f16, w_left_gate, out_dtype=tl.float32)

        left = acc_left_proj * _sigmoid(acc_left_gate) * mask_val
        left_val = left.to(tl.float16)

        wp = w_right_proj_ptr + h_off[:, None] * C + col[None, :]
        w_right_proj = tl.trans(tl.load(wp, mask=h_col_mask, other=0.0).to(tl.float16))
        acc_right_proj = tl.dot(y_f16, w_right_proj, out_dtype=tl.float32)

        wp = w_right_gate_ptr + h_off[:, None] * C + col[None, :]
        w_right_gate = tl.trans(tl.load(wp, mask=h_col_mask, other=0.0).to(tl.float16))
        acc_right_gate = tl.dot(y_f16, w_right_gate, out_dtype=tl.float32)

        right = acc_right_proj * _sigmoid(acc_right_gate) * mask_val
        right_val = right.to(tl.float16)

        wp = w_out_gate_ptr + h_off[:, None] * C + col[None, :]
        w_out_gate = tl.trans(tl.load(wp, mask=h_col_mask, other=0.0).to(tl.float16))
        acc_out_gate = tl.dot(y_f16, w_out_gate, out_dtype=tl.float32)
        gate_val = _sigmoid(acc_out_gate).to(tl.float16)

        # Store to transposed layout [B, N_i, H, N_k].
        # When a full M block lies inside one i-row, transpose the tile so that
        # the contiguous N_k dimension is the vectorized store dimension.
        if EVEN_M:
            lm_mask = h_valid[:, None] & row_valid[None, :]
            row_base = base_div * (H * N)

            tl.store(
                left_ptr + row_base + h_off[:, None] * N + row_mod[None, :],
                tl.trans(left_val),
                mask=lm_mask,
            )
            tl.store(
                right_ptr + row_base + h_off[:, None] * N + row_mod[None, :],
                tl.trans(right_val),
                mask=lm_mask,
            )
            tl.store(
                gate_ptr + row_base + h_off[:, None] * N + row_mod[None, :],
                tl.trans(gate_val),
                mask=lm_mask,
            )
        else:
            lm_mask = row_valid[:, None] & h_valid[None, :]
            row_base = row_div[:, None] * (H * N) + row_mod[:, None]

            tl.store(
                left_ptr + row_base + h_off[None, :] * N,
                left_val,
                mask=lm_mask,
            )
            tl.store(
                right_ptr + row_base + h_off[None, :] * N,
                right_val,
                mask=lm_mask,
            )
            tl.store(
                gate_ptr + row_base + h_off[None, :] * N,
                gate_val,
                mask=lm_mask,
            )


@triton.jit
def trimul_core_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BK: tl.constexpr,
    EVEN_N: tl.constexpr,
):
    pid = tl.program_id(0)

    num_i = tl.cdiv(N, BI)
    num_j = tl.cdiv(N, BJ)
    num_ij = num_i * num_j

    ij = pid % num_ij
    hb = pid // num_ij
    h = hb % H
    b = hb // H

    ib = ij // num_j
    jb = ij % num_j

    i0 = ib * BI
    j0 = jb * BJ

    offs_i = i0 + tl.arange(0, BI)
    offs_j = j0 + tl.arange(0, BJ)
    offs_k = tl.arange(0, BK)

    i_valid = offs_i < N
    j_valid = offs_j < N

    # left/right are in [B, N_i, H, N_k] layout.
    a_base = ((b * N + offs_i).to(tl.int64) * (H * N)) + (h * N)
    b_base = ((b * N + offs_j).to(tl.int64) * (H * N)) + (h * N)

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    for k0 in range(0, N, BK):
        a_ptrs = left_ptr + a_base[:, None] + k0 + offs_k[None, :]
        b_ptrs = right_ptr + b_base[:, None] + k0 + offs_k[None, :]

        if EVEN_N:
            a = tl.load(a_ptrs)
            b = tl.load(b_ptrs)
        else:
            k_valid = offs_k < (N - k0)
            a = tl.load(a_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=j_valid[:, None] & k_valid[None, :], other=0.0)

        # For fixed h: out[i,j] += sum_k L[i,k,h] * R[j,k,h]
        acc = tl.dot(a, tl.trans(b), acc, out_dtype=tl.float32)

    # Store out in [B, N_i, H, N_j] layout.
    out_base = a_base[:, None] + j0 + offs_j[None, :]
    if EVEN_N:
        tl.store(out_ptr + out_base, acc.to(tl.float16))
    else:
        tl.store(out_ptr + out_base, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])

    if mask is not None:
        mask = mask.contiguous()

    B, N, _, C = input_tensor.shape
    H = int(config.get("hidden_dim", C))
    M = B * N * N
    device = input_tensor.device

    # Weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    f16 = torch.float16

    # ---------------- Stage 1 ----------------
    C_PAD = _next_pow2(max(C, 1))
    if C_PAD <= 256:
        bm1 = _block(N, 64, 16)
        h_block1 = _block(H, 128, 16)
    elif C_PAD <= 512:
        bm1 = _block(N, 32, 16)
        h_block1 = _block(H, 64, 16)
    else:
        bm1 = _block(N, 16, 16)
        h_block1 = _block(H, 64, 16)

    left_t = torch.empty((B, N, H, N), dtype=f16, device=device)
    right_t = torch.empty((B, N, H, N), dtype=f16, device=device)
    gate_t = torch.empty((B, N, H, N), dtype=f16, device=device)

    has_mask = mask is not None
    mask_ptr = mask if has_mask else input_tensor

    grid1 = (triton.cdiv(M, bm1),)
    stage1_fused_dn_proj_kernel[grid1](
        input_tensor,
        mask_ptr,
        left_t,
        right_t,
        gate_t,
        norm_weight,
        norm_bias,
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
        out_gate_weight,
        M, N, C, H,
        C_PAD=C_PAD,
        H_BLOCK=h_block1,
        BM=bm1,
        HAS_MASK=has_mask,
        EVEN_M=(N % bm1 == 0),
        num_warps=4,
        num_stages=1,
    )

    # ---------------- Stage 2: TriMul core ----------------
    bi_core = _block(N, 128, 16)
    bj_core = _block(N, 128, 16)
    bk_core = _block(N, 128, 16)
    even_core = (N % bi_core == 0) and (N % bj_core == 0) and (N % bk_core == 0)

    out_t = torch.empty((B, N, H, N), dtype=f16, device=device)

    num_i = triton.cdiv(N, bi_core)
    num_j = triton.cdiv(N, bj_core)
    grid_core = (H * B * num_i * num_j,)

    trimul_core_kernel[grid_core](
        left_t,
        right_t,
        out_t,
        B, N, H,
        BI=bi_core,
        BJ=bj_core,
        BK=bk_core,
        EVEN_N=even_core,
        num_warps=8,
        num_stages=2,
    )

    del left_t, right_t

    # ---------------- Stage 3: final LN + gate + linear ----------------
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    bi_final = _div_block(N, 16 if N >= 1024 else 8 if N >= 512 else 4 if N >= 256 else 2)
    bj_final = _block(N, 128, 16)
    hc = _block(H, 128, 16)
    cb = _block(C, 128, 16)
    full_h = (H == hc) and (H <= 128)

    num_i_final = triton.cdiv(N, bi_final)
    num_j_final = triton.cdiv(N, bj_final)
    grid_final = (B * num_i_final * num_j_final,)

    final_fused_ln_gate_linear_kernel[grid_final](
        out_t,
        gate_t,
        out,
        to_out_weight,
        to_out_norm_weight,
        to_out_norm_bias,
        B, N, H, C,
        BI=bi_final,
        BJ=bj_final,
        HC=hc,
        CB=cb,
        FULL_H=full_h,
        num_warps=8,
        num_stages=2,
    )

    return out
```
