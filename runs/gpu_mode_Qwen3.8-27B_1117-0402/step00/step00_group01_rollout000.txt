```python
import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_kernel(
    left_ptr, right_ptr, lgate_ptr, rgate_ptr, mask_ptr,
    out_left_ptr, out_right_ptr,
    N,  # total number of positions (B*N*N)
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)
    
    mask_m = offs_m < N
    mask_h = offs_h < H
    
    mask_2d = mask_m[:, None] & mask_h[None, :]
    
    # Load
    left = tl.load(left_ptr + offs_m[:, None] * H + offs_h[None, :], mask=mask_2d, other=0.0)
    right = tl.load(right_ptr + offs_m[:, None] * H + offs_h[None, :], mask=mask_2d, other=0.0)
```

```python
import torch
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    left_ptr, right_ptr, lgate_ptr, rgate_ptr, ogate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr, out_ogate_ptr,
    total_elements,  # B * N * N
    H: tl.constexpr,
    BLOCK: tl.constexpr,
):
    """
    Computes:
        out_left = left * mask * sigmoid(lgate)
        out_right = right * mask * sigmoid(rgate)  
        out_ogate = sigmoid(ogate)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask_1d = offs < total_elements
    
    h_offs = tl.arange(0, H)
    
    # Load all inputs
    base = offs[:, None] * H + h_offs[None, :]
    mask_2d = mask_1d[:, None]  # [BLOCK, 1] broadcast to [BLOCK, H]
    
    left = tl.load(left_ptr + base, mask=mask_2d, other=0.0)
    right = tl.load(right_ptr + base, mask=mask_2d, other=0.0)
    lgate = tl.load(lgate_ptr + base, mask=mask_2d, other=0.0)
    rgate = tl.load(rgate_ptr + base, mask=mask_2d, other=0.0)
    ogate = tl.load(ogate_ptr + base, mask=mask_2d, other=0.0)
    
    m = tl.load(mask_ptr + offs, mask=mask_1d, other=0.0)
    m = m[:, None]  # [BLOCK, 1]
    
    # Apply operations
    out_left = left * m * tl.sigmoid(lgate)
    out_right = right * m * tl.sigmoid(rgate)
    out_ogate = tl.sigmoid(ogate)
    
    # Store
    tl.store(out_left_ptr + base, out_left, mask=mask_2d)
    tl.store(out_right_ptr + base, out_right, mask=mask_2d)
    tl.store(out_ogate_ptr + base, out_ogate, mask=mask_2d)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, N2, C = input_tensor.shape
    H = config["hidden_dim"]
    
    device = input_tensor.device
    
    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor, (C,), weights["norm.weight"], weights["norm.bias"]
    )
    
    # Flatten for batched operations
    x_flat = x.view(-1, C)  # [B*N*N, C]
    total = B * N * N
    
    # Fused projection: concatenate all 5 weight matrices
    # Each weight is [H, C], we want [5*H, C]
    W_fused = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate_weight"],
        weights["right_gate_weight"],
        weights["out_gate_weight"],
    ], dim=0)  # [5*H, C]
    
    # One big GEMM: [B*N*N, C] @ [C, 5*H] = [B*N*N, 5*H]
    proj = x_flat @ W_fused.t()  # [total, 5*H]
    
    # Split
    left = proj[:, :H]
    right = proj[:, H:2*H]
    lgate = proj[:, 2*H:3*H]
    rgate = proj[:, 3*H:4*H]
    ogate = proj[:, 4*H:5*H]
    
    # Apply mask and gates using Triton kernel
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)
    out_ogate = torch.empty_like(ogate)

    grid = (triton.cdiv(total, BLOCK),)

    if mask is not None and mask.numel() > 0:

    else:
        mask_flat = torch.ones(B, N, N, device=device)
        mask_flat = mask_flat.view(-1)
    
    fused_mask_gate_kernel[grid](
        left, right, lgate, rgate, ogate,

        out_left, out_right, out_ogate,
        total,
        H=H,

    )
    
    # Reshape for batched GEMM
```

```python
import torch
import triton
import triton.language as tl

@triton.jit
def fused_mask_gate_kernel(
    left_ptr, right_ptr, lgate_ptr, rgate_ptr, ogate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr, out_ogate_ptr,
    total,
    H: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    valid = offs < total
    
    h = tl.arange(0, H)
    idx = offs[:, None] * H + h[None, :]
    m2d = valid[:, None]
    
    left = tl.load(left_ptr + idx, mask=m2d, other=0.0)
    right = tl.load(right_ptr + idx, mask=m2d, other=0.0)
    lg = tl.load(lgate_ptr + idx, mask=m2d, other=0.0)
    rg = tl.load(rgate_ptr + idx, mask=m2d, other=0.0)
    og = tl.load(ogate_ptr + idx, mask=m2d, other=0.0)
    
    m = tl.load(mask_ptr + offs, mask=valid, other=0.0)
    m = m[:, None]
    
    out_l = left * m * tl.sigmoid(lg)
    out_r = right * m * tl.sigmoid(rg)
    out_og = tl.sigmoid(og)
    
    tl.store(out_left_ptr + idx, out_l, mask=m2d)
    tl.store(out_right_ptr + idx, out_r, mask=m2d)
    tl.store(out_ogate_ptr + idx, out_og, mask=m2d)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    device = input_tensor.device
    total = B * N * N
    
    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor, (C,), weights["norm.weight"], weights["norm.bias"]
    )
    x_flat = x.view(total, C)
    
    # 2. Fused projection
    W = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate_weight"],
        weights["right_gate_weight"],
        weights["out_gate_weight"],
    ], dim=0).t()  # [C, 5*H]
    
    proj = x_flat @ W  # [total, 5*H]
    
    # Split
    left = proj[:, :H]
    right = proj[:, H:2*H]
    lgate = proj[:, 2*H:3*H]
    rgate = proj[:, 3*H:4*H]
    ogate = proj[:, 4*H:5*H]
    
    # 3. Triton kernel for mask + gate
    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)
    out_ogate = torch.empty_like(ogate)
    
    BLOCK = 64
    grid = (triton.cdiv(total, BLOCK),)
    
    if mask is not None and mask.numel() > 0:
        mask_flat = mask.view(-1)
    else:
        mask_flat = torch.ones(total, device=device)
    
    fused_mask_gate_kernel[grid](
        left, right, lgate, rgate, ogate,
        mask_flat,
        out_left, out_right, out_ogate,
        total,
        H=H,
        BLOCK=BLOCK,
    )
    
    # 4. Batched GEMM
    left_4d = out_left.view(B, N, N, H)
    right_4d = out_right.view(B, N, N, H)
    
    left_bmm = left_4d.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_bmm = right_4d.permute(0, 3, 1, 2).reshape(B * H, N, N)
    
    # Use bf16 for the GEMM if possible
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))
    out = out_bmm.reshape(B, H, N, N).permute(0, 2, 3, 1)
    out = out.contiguous()
    
    # 5. Final LayerNorm + gate + projection
    out = torch.nn.functional.layer_norm(
        out, (H,), weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    )
    
    ogate_4d = out_ogate.view(B, N, N, H)
    out = out * ogate_4d
    
    out = out.view(total, H) @ weights["to_out_weight"].t()
    out = out.view(B, N, N, C)
    
    return out.float()
```
