```python
import torch
import triton
import triton.language as tl


"""
Fused Triton forward pass for AlphaFold3-style outgoing TriMul.

Pipeline:
  1) One Triton kernel computes LayerNorm over the last C dimension, then all
     five linear projections:
       - left_proj, right_proj
       - left_gate, right_gate, out_gate
     The projection outputs are masked and gated; left/right are stored in
     [B, hidden, N, N] layout for an efficient batched GEMM, while out_gate is
     stored token-major [B, N, N, hidden].
  2) The N^3 triangular multiplication is done as a batched GEMM:
       core[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
  3) A second Triton kernel fuses the final LayerNorm over hidden, the
     out_gate multiplication, and the final linear projection to dim.
"""


def _next_pow2(x: int) -> int:
    x = int(max(1, x))
    p = 1
    while p < x:
        p <<= 1
    return p


_WEIGHT_CACHE = {}


def _prepare_weights(weights: dict, C: int, H: int, device: torch.device):
    keys = (
        "norm.weight",
        "norm.bias",
        "left_proj.weight",
        "right_proj.weight",
        "left_gate.weight",
        "right_gate.weight",
        "out_gate.weight",
        "to_out_norm.weight",
        "to_out_norm.bias",
        "to_out.weight",
    )

    dev_key = (device.type, getattr(device, "index", None))
    key = tuple(weights[k].data_ptr() for k in keys) + (C, H) + dev_key
    if key in _WEIGHT_CACHE:
        return _WEIGHT_CACHE[key]

    def weighted(name: str) -> torch.Tensor:
        return weights[name].detach().to(device).float().contiguous()

    def proj_weight(name: str) -> torch.Tensor:
        # [H, C] -> [C, H], bf16 for Triton tensor cores.
        return weights[name].detach().to(device).t().contiguous().to(torch.bfloat16)

    norm_w = weighted("norm.weight")
    norm_b = weighted("norm.bias")

    w_l = proj_weight("left_proj.weight")
    w_r = proj_weight("right_proj.weight")
    w_lg = proj_weight("left_gate.weight")
    w_rg = proj_weight("right_gate.weight")
    w_og = proj_weight("out_gate.weight")

    to_out_norm_w = weighted("to_out_norm.weight")
    to_out_norm_b = weighted("to_out_norm.bias")

    # to_out.weight is [C, H]; transposes to [H, C] for y @ W.
    w_out = weights["to_out.weight"].detach().to(device).t().contiguous().to(torch.bfloat16)

    out = (
        norm_w,
        norm_b,
        w_l,
        w_r,
        w_lg,
        w_rg,
        w_og,
        w_out,
        to_out_norm_w,
        to_out_norm_b,
    )
    _WEIGHT_CACHE[key] = out
    return out


@triton.jit
def _trimul_projection_kernel(
    x_ptr,
    mask_ptr,
    gamma_ptr,
    beta_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    left_ptr,
    right_ptr,
    gate_ptr,
    T,
    N: tl.constexpr,
    C: tl.constexpr,
    H: tl.constexpr,
    HB: tl.constexpr,
    NN: tl.constexpr,
    BLOCK_T: tl.constexpr,
    BLOCK_C: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_t_raw = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    t_ok = offs_t_raw < T
    offs_t = tl.where(t_ok, offs_t_raw, 0)

    offs_h = tl.arange(0, HB)
    h_ok = offs_h < H
    oh = tl.where(h_ok, offs_h, 0)

    # Pass 1: raw LayerNorm statistics.
    if C <= BLOCK_C:
        offs_c = tl.arange(0, BLOCK_C)
        c_ok = offs_c < C
        cmask = t_ok[:, None] & (c_ok[None, :])
        xv = tl.load(
            x_ptr + offs_t[:, None] * C + offs_c[None, :],
            mask=cmask,
            other=0.0,
        ).to(tl.float32)

        sum_x = tl.sum(xv, axis=1)
        sum_x2 = tl.sum(xv * xv, axis=1)
    else:
        sum_x = tl.zeros((BLOCK_T,), dtype=tl.float32)
        sum_x2 = tl.zeros((BLOCK_T,), dtype=tl.float32)
        for c0 in range(0, C, BLOCK_C):
            offs_c = c0 + tl.arange(0, BLOCK_C)
            cmask = t_ok[:, None] & (offs_c[None, :] < C)
            xv = tl.load(
                x_ptr + offs_t[:, None] * C + offs_c[None, :],
                mask=cmask,
                other=0.0,
            ).to(tl.float32)
            sum_x += tl.sum(xv, axis=1)
            sum_x2 += tl.sum(xv * xv, axis=1)

    mean = sum_x / C
    var = sum_x2 / C - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + EPS)

    # Pass 2: normalize and compute all five projections.
    if C <= BLOCK_C:
        offs_c = tl.arange(0, BLOCK_C)
        c_ok = offs_c < C
        cmask = t_ok[:, None] & (c_ok[None, :])
        xv = tl.load(
            x_ptr + offs_t[:, None] * C + offs_c[None, :],
            mask=cmask,
            other=0.0,
        ).to(tl.float32)

        gamma_c = tl.load(gamma_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)
        beta_c = tl.load(beta_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)

        xn = (xv - mean[:, None]) * rstd[:, None] * gamma_c[None, :] + beta_c[None, :]
        xb = xn.to(tl.bfloat16)

        wmask = (c_ok[:, None]) & (h_ok[None, :])
        wl = tl.load(wl_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
        wr = tl.load(wr_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
        wlg = tl.load(wlg_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
        wrg = tl.load(wrg_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
        wog = tl.load(wog_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)

        acc_l = tl.dot(xb, wl)
        acc_r = tl.dot(xb, wr)
        acc_lg = tl.dot(xb, wlg)
        acc_rg = tl.dot(xb, wrg)
        acc_og = tl.dot(xb, wog)
    else:
        acc_l = tl.zeros((BLOCK_T, HB), dtype=tl.float32)
        acc_r = tl.zeros((BLOCK_T, HB), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_T, HB), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_T, HB), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_T, HB), dtype=tl.float32)

        for c0 in range(0, C, BLOCK_C):
            offs_c = c0 + tl.arange(0, BLOCK_C)
            cmask = t_ok[:, None] & (offs_c[None, :] < C)
            xv = tl.load(
                x_ptr + offs_t[:, None] * C + offs_c[None, :],
                mask=cmask,
                other=0.0,
            ).to(tl.float32)

            gamma_c = tl.load(gamma_ptr + offs_c, mask=offs_c < C, other=0.0).to(tl.float32)
            beta_c = tl.load(beta_ptr + offs_c, mask=offs_c < C, other=0.0).to(tl.float32)

            xn = (xv - mean[:, None]) * rstd[:, None] * gamma_c[None, :] + beta_c[None, :]
            xb = xn.to(tl.bfloat16)

            wmask = (offs_c[:, None] < C) & (h_ok[None, :])
            wl = tl.load(wl_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
            wr = tl.load(wr_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
            wlg = tl.load(wlg_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
            wrg = tl.load(wrg_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)
            wog = tl.load(wog_ptr + offs_c[:, None] * H + oh[None, :], mask=wmask, other=0.0)

            acc_l = tl.dot(xb, wl, acc_l)
            acc_r = tl.dot(xb, wr, acc_r)
            acc_lg = tl.dot(xb, wlg, acc_lg)
            acc_rg = tl.dot(xb, wrg, acc_rg)
            acc_og = tl.dot(xb, wog, acc_og)

    # Sigmoid gates.
    left_gate = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate = 1.0 / (1.0 + tl.exp(-acc_og))

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_t, mask=t_ok, other=0.0).to(tl.float32)
    else:
        mask_val = tl.full((BLOCK_T,), 1.0, dtype=tl.float32)

    m = mask_val[:, None]
    left_out = (acc_l * left_gate * m).to(tl.bfloat16)
    right_out = (acc_r * right_gate * m).to(tl.bfloat16)
    gate_out = out_gate.to(tl.bfloat16)

    store_mask = t_ok[:, None] & (h_ok[None, :])

    # Store left/right in [B, H, N, N] layout.
    outer = tl.where(t_ok, offs_t // NN, 0)
    inner = tl.where(t_ok, offs_t - outer * NN, 0)
    hbase = outer * (H * NN) + inner

    lh_ptrs = left_ptr + hbase[:, None] + oh[None, :] * NN
    rh_ptrs = right_ptr + hbase[:, None] + oh[None, :] * NN
    tl.store(lh_ptrs, left_out, mask=store_mask)
    tl.store(rh_ptrs, right_out, mask=store_mask)

    # Store out-gate token-major [B, N, N, H].
    g_ptrs = gate_ptr + offs_t[:, None] * H + oh[None, :]
    tl.store(g_ptrs, gate_out, mask=store_mask)


@triton.jit
def _trimul_final_kernel(
    core_ptr,
    gate_ptr,
    w_out_ptr,
    norm_w_ptr,
    norm_b_ptr,
    out_ptr,
    T,
    N: tl.constexpr,
    C: tl.constexpr,
    H: tl.constexpr,
    HB: tl.constexpr,
    NN: tl.constexpr,
    BLOCK_T: tl.constexpr,
    BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_t_raw = pid * BLOCK_T + tl.arange(0, BLOCK_T)
    t_ok = offs_t_raw < T
    offs_t = tl.where(t_ok, offs_t_raw, 0)

    offs_h = tl.arange(0, HB)
    h_ok = offs_h < H
    oh = tl.where(h_ok, offs_h, 0)

    outer = tl.where(t_ok, offs_t // NN, 0)
    inner = tl.where(t_ok, offs_t - outer * NN, 0)
    hbase = outer * (H * NN) + inner

    load_mask = t_ok[:, None] & (h_ok[None, :])

    # core is [B, H, N, N], gate is [B, N, N, H].
    core = tl.load(
        core_ptr + hbase[:, None] + oh[None, :] * NN,
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)

    gate = tl.load(
        gate_ptr + offs_t[:, None] * H + oh[None, :],
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(core, axis=1) / H
    var = tl.sum(core * core, axis=1) / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = tl.rsqrt(var + EPS)

    norm_w = tl.load(norm_w_ptr + oh, mask=h_ok, other=0.0).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + oh, mask=h_ok, other=0.0).to(tl.float32)

    y = (core - mean[:, None]) * rstd[:, None] * norm_w[None, :] + norm_b[None, :]
    y = y * gate
    y16 = y.to(tl.bfloat16)

    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        c_ok = offs_c < C

        wmask = h_ok[:, None] & c_ok[None, :]
        w = tl.load(w_out_ptr + oh[:, None] * C + offs_c[None, :], mask=wmask, other=0.0)

        out_block = tl.dot(y16, w)
        out_ptrs = out_ptr + offs_t[:, None] * C + offs_c[None, :]
        tl.store(
            out_ptrs,
            out_block.to(tl.float32),
            mask=t_ok[:, None] & c_ok[None, :],
        )

    # Block shapes.  H is padded to a power of two for tl.arange/dot tiling.
    HB = _next_pow2(max(16, H))
    BLOCK_C = max(16, min(128, _next_pow2(max(1, C))))

    if HB <= 128:
        PROJ_T = 32
    else:
        PROJ_T = 16
    PROJ_T = max(16, PROJ_T)
    PROJ_WARPS = 8 if HB <= 256 else 16

    left = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    right = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    out_gate = torch.empty((B, N, N, H), dtype=torch.bfloat16, device=device)

    mask_arg = mask if has_mask else x
    grid = (triton.cdiv(T, PROJ_T),)
    _trimul_projection_kernel[grid](
        x,
        mask_arg,
        norm_w,
        norm_b,
        w_l,
        w_r,
        w_lg,
        w_rg,
        w_og,
        left,
        right,
        out_gate,
        T,
        N=N,
        C=C,
        H=H,
        HB=HB,
        NN=NN,
        BLOCK_T=PROJ_T,
        BLOCK_C=BLOCK_C,
        HAS_MASK=has_mask,
        EPS=1e-5,
        num_warps=PROJ_WARPS,
        num_stages=2,
    )

    # Batched GEMM over hidden dimension:
    # core[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
    batch = B * H
    left_b2 = left.view(batch, N, N)
    right_b2 = right.view(batch, N, N)
    core = torch.bmm(left_b2, right_b2.transpose(1, 2)).view(B, H, N, N)

    del left, right, left_b2, right_b2

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    if HB <= 128:
        FINAL_T = 64
    elif HB <= 256:
        FINAL_T = 32
    else:
        FINAL_T = 16
    FINAL_T = max(16, FINAL_T)
    FINAL_WARPS = 8 if HB <= 256 else 16

    grid = (triton.cdiv(T, FINAL_T),)
    _trimul_final_kernel[grid](
        core,
        out_gate,
        w_out,
        to_out_norm_w,
        to_out_norm_b,
        out,
        T,
        N=N,
        C=C,
        H=H,
        HB=HB,
        NN=NN,
        BLOCK_T=FINAL_T,
        BLOCK_C=BLOCK_C,
        EPS=1e-5,
        num_warps=FINAL_WARPS,
        num_stages=2,
    )

    return out
```
