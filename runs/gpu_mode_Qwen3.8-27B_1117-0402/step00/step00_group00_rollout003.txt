    while n < x:

            ss += tl.sum(x_chunk * x_chunk, axis=1)

        mean = s * C_inv
        var = ss * C_inv - mean * mean
        var = tl.maximum(var, 0.0)
        rstd = 1.0 / tl.sqrt(var + 1e-5)

        for cb in range(C_BLOCKS):
            c = cb * CC + tl.arange(0, CC)
            c1d = c < C
            x_chunk = tl.load(
                x_ptr + x_base + pos[:, None] * C + c[None, :],
                mask=(c1d[None, :] & pos_mask[:, None]),
                other=0.0
            ).to(tl.float32)

            g = tl.load(gamma_ptr + c, mask=c1d, other=0.0).to(tl.float32)
            be = tl.load(beta_ptr + c, mask=c1d, other=0.0).to(tl.float32)

            x_norm = (x_chunk - mean[:, None]) * rstd[:, None] * g[None, :] + be[None, :]
            x_norm = tl.where((c1d[None, :] & pos_mask[:, None]), x_norm, 0.0)

            w_mask = d_mask[:, None] & (c1d[None, :])

            w = tl.load(wl_ptr + d[:, None] * C + c[None, :], mask=w_mask, other=0.0).to(tl.float32)
            acc_l = tl.dot(x_norm, tl.trans(w), acc_l, input_precision="tf32")

            w = tl.load(wr_ptr + d[:, None] * C + c[None, :], mask=w_mask, other=0.0).to(tl.float32)
            acc_r = tl.dot(x_norm, tl.trans(w), acc_r, input_precision="tf32")

            w = tl.load(wlg_ptr + d[:, None] * C + c[None, :], mask=w_mask, other=0.0).to(tl.float32)
            acc_lg = tl.dot(x_norm, tl.trans(w), acc_lg, input_precision="tf32")

            w = tl.load(wrg_ptr + d[:, None] * C + c[None, :], mask=w_mask, other=0.0).to(tl.float32)
            acc_rg = tl.dot(x_norm, tl.trans(w), acc_rg, input_precision="tf32")

            w = tl.load(wog_ptr + d[:, None] * C + c[None, :], mask=w_mask, other=0.0).to(tl.float32)
            acc_og = tl.dot(x_norm, tl.trans(w), acc_og, input_precision="tf32")

    if HAS_MASK:
        maskv = tl.load(mask_ptr + b * N2 + pos, mask=pos_mask, other=0.0).to(tl.float32)
    else:
        maskv = tl.zeros((BM,), dtype=tl.float32) + 1.0

    lg = 1.0 / (1.0 + tl.exp(-acc_lg))
    rg = 1.0 / (1.0 + tl.exp(-acc_rg))
    og = 1.0 / (1.0 + tl.exp(-acc_og))

    left = acc_l * lg * maskv[:, None]
    right = acc_r * rg * maskv[:, None]

    token_base = b * H * N2 + pos
    store_mask = (d[:, None] < H) & (pos_mask[None, :])
    ptrs = token_base[None, :] + d[:, None] * N2

    tl.store(l_ptr + ptrs, tl.trans(left.to(tl.bfloat16)), mask=store_mask)
    tl.store(r_ptr + ptrs, tl.trans(right.to(tl.bfloat16)), mask=store_mask)
    tl.store(og_ptr + ptrs, tl.trans(og.to(tl.bfloat16)), mask=store_mask)


@triton.jit
def _central_kernel(
    l_ptr, r_ptr, o_ptr,
    N, H, N2,
    BI: tl.constexpr,
    BJ: tl.constexpr,
    BK: tl.constexpr,
    EVEN: tl.constexpr,
):
    pid0 = tl.program_id(0)
    b = pid0 // H
    d = pid0 % H

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    base = (b * H + d) * N2

    offs_i = pid_i * BI + tl.arange(0, BI)
    offs_j = pid_j * BJ + tl.arange(0, BJ)
    offs_k = tl.arange(0, BK)

    a_ptrs = l_ptr + base + offs_i[:, None] * N + offs_k[None, :]
    r_ptrs = r_ptr + base + offs_j[:, None] * N + offs_k[None, :]

    acc = tl.zeros((BI, BJ), dtype=tl.float32)

    if EVEN:
        for _ in range(0, N, BK):
            a = tl.load(a_ptrs)
            r = tl.load(r_ptrs)
            acc = tl.dot(a, tl.trans(r), acc)
            a_ptrs += BK
            r_ptrs += BK

        c_ptrs = o_ptr + base + offs_i[:, None] * N + offs_j[None, :]
        tl.store(c_ptrs, acc)
    else:
        i_mask = offs_i < N
        j_mask = offs_j < N
        num_k = tl.cdiv(N, BK)

        for k0 in range(0, num_k):
            k_off = k0 * BK
            km = (k_off + offs_k) < N

            a = tl.load(a_ptrs, mask=i_mask[:, None] & km[None, :], other=0.0)
            r = tl.load(r_ptrs, mask=j_mask[:, None] & km[None, :], other=0.0)

            acc = tl.dot(a, tl.trans(r), acc)

            a_ptrs += BK
            r_ptrs += BK

        c_ptrs = o_ptr + base + offs_i[:, None] * N + offs_j[None, :]
        tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


@triton.jit
def _final_kernel(
    o_ptr, og_ptr, w_ptr, out_ptr,
    gamma_ptr, beta_ptr,
    N, H, C, N2, H_inv,
    BM: tl.constexpr,
    HP: tl.constexpr,
    CB: tl.constexpr,
    C_BLOCKS: tl.constexpr,
):
    pid = tl.program_id(0)
    b = tl.program_id(1)

    pos = pid * BM + tl.arange(0, BM)
    pos_mask = pos < N2

    token_base = b * H * N2 + pos

    d = tl.arange(0, HP)
    d_mask = d < H

    load_mask = d_mask[:, None] & pos_mask[None, :]

    o = tl.load(o_ptr + token_base[None, :] + d[:, None] * N2, mask=load_mask, other=0.0).to(tl.float32)
    og = tl.load(og_ptr + token_base[None, :] + d[:, None] * N2, mask=load_mask, other=0.0).to(tl.float32)

    mean = tl.sum(o, axis=0) * H_inv
    diff = o - mean[None, :]
    var = tl.sum(diff * diff, axis=0) * H_inv
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + 1e-5)

    gamma = tl.load(gamma_ptr + d, mask=d_mask, other=0.0).to(tl.float32)
    beta = tl.load(beta_ptr + d, mask=d_mask, other=0.0).to(tl.float32)

    z = (diff * rstd[None, :] * gamma[:, None] + beta[None, :]) * og
    z = tl.where(d_mask[:, None], z, 0.0)
    zt = tl.trans(z)

    out_base = b * N2 * C

    for cb in range(C_BLOCKS):
        c = cb * CB + tl.arange(0, CB)
        c_mask = c < C

        w_tile = tl.load(
            w_ptr + c[:, None] * H + d[None, :],
            mask=c_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        acc = tl.dot(zt, tl.trans(w_tile), input_precision="tf32")

        out_ptrs = out_ptr + out_base + pos[:, None] * C + c[None, :]
        tl.store(out_ptrs, acc, mask=pos_mask[:, None] & c_mask[None, :])


def custom_kernel(data):
    """
    Fused AlphaFold3-style outgoing TriMul.

    Pipeline:
      1. Triton kernel: LayerNorm + five linears + sigmoid gates + mask,
         writing L/R/out_gate in [B, H, N, N] layout.
      2. Triton kernel: per-channel batched GEMM:
         O[..., d, i, j] = sum_k L[..., d, i, k] * R[..., d, j, k].
      3. Triton kernel: LayerNorm over H, out-gate, final projection.
    """
    input_tensor, mask, weights, config = data

    B, N, _, C = input_tensor.shape
    H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))

    x = input_tensor.contiguous().float()

    has_mask = mask is not None
    if has_mask:
        mask = torch.as_tensor(mask).contiguous().float()
    else:
        mask = None

    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()
    wl = weights["left_proj.weight"].float().contiguous()
    wr = weights["right_proj.weight"].float().contiguous()
    wlg = weights["left_gate.weight"].float().contiguous()
    wrg = weights["right_gate.weight"].float().contiguous()
    wog = weights["out_gate.weight"].float().contiguous()
    to_norm_w = weights["to_out_norm.weight"].float().contiguous()
    to_norm_b = weights["to_out_norm.bias"].float().contiguous()
    to_out_w = weights["to_out.weight"].float().contiguous()

    N2 = N * N
    M = B * N2
    dev = x.device
    BF = torch.bfloat16

    # For the common fastest path, use fully Triton fusion for H <= 128.
    # A torch fallback is used only for larger hidden dimensions to keep
    # register usage sane while preserving the mandatory Triton central kernel.
    use_fast_triton = 16 <= H <= 128

    if use_fast_triton:
        L = torch.empty((B, H, N, N), dtype=BF, device=dev)
        R = torch.empty((B, H, N, N), dtype=BF, device=dev)
        OG = torch.empty((B, H, N, N), dtype=BF, device=dev)
        O = torch.empty((B, H, N, N), dtype=torch.float32, device=dev)
        out = torch.empty((B, N, N, C), dtype=torch.float32, device=dev)

        CC = 128
        C_BLOCKS = triton.cdiv(C, CC)
        HH = _next_pow2(H)

        if M <= 32768:
            BM_p = 16
        elif C > 384:
            BM_p = 16
        else:
            BM_p = 32

        grid_p = (triton.cdiv(N2, BM_p), B)
        _proj_kernel[grid_p](
            x, mask if has_mask else x,
            L, R, OG,
            wl, wr, wlg, wrg, wog,
            norm_w, norm_b,
            N, H, C, N2, 1.0 / C,
            C_BLOCKS=C_BLOCKS,
            BM=BM_p,
            HH=HH,
            CC=CC,
            HAS_MASK=has_mask,
            num_warps=8,
        )

        if N <= 128:
            BI, BJ, BK = 32, 64, 32
        elif N <= 256:
            BI, BJ, BK = 64, 64, 64
        else:
            BI, BJ, BK = 128, 64, 64

        EVEN = (N % BI == 0) and (N % BJ == 0) and (N % BK == 0)
        grid_c = (H * B, triton.cdiv(N, BI), triton.cdiv(N, BJ))
        _central_kernel[grid_c](
            L, R, O,
            N, H, N2,
            BI=BI, BJ=BJ, BK=BK,
            EVEN=EVEN,
            num_warps=8 if BI >= 128 else 4,
            num_stages=3,
        )

        HP = _next_pow2(H)
        CB = 128
        FC_BLOCKS = triton.cdiv(C, CB)

        if M <= 65536:
            BM_f = 32
        else:
            BM_f = 64

        grid_f = (triton.cdiv(N2, BM_f), B)
        _final_kernel[grid_f](
            O, OG, to_out_w, out,
            to_norm_w, to_norm_b,
            N, H, C, N2, 1.0 / H,
            BM=BM_f, HP=HP, CB=CB, C_BLOCKS=FC_BLOCKS,
            num_warps=8 if BM_f >= 64 else 4,
            num_stages=3,
        )

        return out

    else:
        # Robust path for larger hidden_dim. Still uses the Triton central kernel.
        x_norm = F.layer_norm(x, (C,), weight=norm_w, bias=norm_b)

        left = x_norm @ wl.t()
        right = x_norm @ wr.t()

        if has_mask:
            m4 = mask.unsqueeze(-1)
            left = left * m4
            right = right * m4

        left = left * torch.sigmoid(x_norm @ wlg.t())
        right = right * torch.sigmoid(x_norm @ wrg.t())
        out_gate = torch.sigmoid(x_norm @ wog.t())

        L = left.to(BF).permute(0, 3, 1, 2).contiguous()
        R = right.to(BF).permute(0, 3, 1, 2).contiguous()
        OG = out_gate.to(BF).permute(0, 3, 1, 2).contiguous()

        O = torch.empty((B, H, N, N), dtype=torch.float32, device=dev)

        if N <= 128:
            BI, BJ, BK = 32, 64, 32
        elif N <= 256:
            BI, BJ, BK = 64, 64, 64
        else:
            BI, BJ, BK = 128, 64, 64

        EVEN = (N % BI == 0) and (N % BJ == 0) and (N % BK == 0)
        grid_c = (H * B, triton.cdiv(N, BI), triton.cdiv(N, BJ))
        _central_kernel[grid_c](
            L, R, O,
            N, H, N2,
            BI=BI, BJ=BJ, BK=BK,
            EVEN=EVEN,
            num_warps=8 if BI >= 128 else 4,
            num_stages=3,
        )

        O_h = O.permute(0, 2, 3, 1).contiguous().float()
        OG_h = OG.permute(0, 2, 3, 1).float()

        O_h = F.layer_norm(O_h, (H,), weight=to_norm_w, bias=to_norm_b)
        O_h = O_h * OG_h
        return torch.matmul(O_h, to_out_w.t()).float()
