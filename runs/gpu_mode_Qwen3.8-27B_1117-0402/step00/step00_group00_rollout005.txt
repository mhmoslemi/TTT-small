def fused_ln_proj_gate_kernel(
    x_ptr, mask_ptr,
    gamma_ptr, beta_ptr,
    wl_ptr, wr_ptr, wlg_ptr, wrg_ptr, wog_ptr,
    left_ptr, right_ptr, og_ptr,
    M,
    D: tl.constexpr,
    H: tl.constexpr,

    stride_w_h,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
    NUM_H: tl.constexpr,
    NUM_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid_m = offs_m < M
    if HAS_MASK:
        mask_m = tl.load(mask_ptr + offs_m, mask=valid_m, other=0.0).to(tl.float32)
    row_sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
    row_sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)
    for dc in range(0, NUM_D):
        offs_d = dc * BLOCK_D + tl.arange(0, BLOCK_D)
        valid_d = offs_d < D
        x_chunk = tl.load(x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :],
                         mask=valid_m[:, None] & valid_d[None, :], other=0.0).to(tl.float32)
        row_sum += tl.sum(x_chunk, axis=1)
        row_sumsq += tl.sum(x_chunk * x_chunk, axis=1)
    mean = row_sum / D
    var = row_sumsq / D - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

        acc_rg = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)

import torch
import triton
import triton.language as tl


"""
TriMul forward, fused as:
  1) Triton kernel:
     - row-wise LayerNorm over the last dim of the input
     - 5 linear projections (left, right, left_gate, right_gate, out_gate)
     - sigmoid gating and optional mask application
     - outputs are stored in h-major layout [hidden, seq*seq]
  2) cuBLAS batched matmul for the N^3 multiplicative update:
     out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
  3) Triton kernel:
     - LayerNorm over hidden dim
     - multiply by out_gate
     - final linear projection to dim
"""


def _next_power_of_two(x: int) -> int:
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


@triton.jit
def fused_ln_proj_gate_kernel(
    x_ptr,
    mask_ptr,
    gamma_ptr,
    beta_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    left_ptr,
    right_ptr,
    og_ptr,
    M,
    D: tl.constexpr,
    H: tl.constexpr,
    stride_x_m,
    stride_out_m,
    stride_w_h,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
    NUM_H: tl.constexpr,
    NUM_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid_m = offs_m < M

    mask_m = tl.zeros((BLOCK_M,), dtype=tl.float32)
    if HAS_MASK:
        mask_m = tl.load(mask_ptr + offs_m, mask=valid_m, other=0.0).to(tl.float32)

    # Pass 1: LayerNorm statistics.
    row_sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
    row_sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for dc in range(0, NUM_D):
        offs_d = dc * BLOCK_D + tl.arange(0, BLOCK_D)
        valid_d = offs_d < D

        x_chunk = tl.load(
            x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :],
            mask=valid_m[:, None] & valid_d[None, :],
            other=0.0,
        ).to(tl.float32)

        row_sum += tl.sum(x_chunk, axis=1)
        row_sumsq += tl.sum(x_chunk * x_chunk, axis=1)

    mean = row_sum / D
    var = row_sumsq / D - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    # Pass 2: normalized projections + gating.
    for hb in range(0, NUM_H):
        offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        valid_h = offs_h < H

        acc_l = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_r = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_H, BLOCK_M), dtype=tl.float32)

        for dc in range(0, NUM_D):
            offs_d = dc * BLOCK_D + tl.arange(0, BLOCK_D)
            valid_d = offs_d < D
            x_mask = valid_m[:, None] & valid_d[None, :]

            x_chunk = tl.load(
                x_ptr + offs_m[:, None] * stride_x_m + offs_d[None, :],
                mask=x_mask,
                other=0.0,
            ).to(tl.float32)

            gamma = tl.load(gamma_ptr + offs_d, mask=valid_d, other=0.0).to(tl.float32)
            beta = tl.load(beta_ptr + offs_d, mask=valid_d, other=0.0).to(tl.float32)

            xn = (x_chunk - mean[:, None]) * rstd[:, None] * gamma[None, :] + beta[None, :]
            x_dot = tl.trans(xn.to(tl.bfloat16))  # [BLOCK_D, BLOCK_M]

            w_mask = valid_h[:, None] & valid_d[None, :]

            wl = tl.load(
                wl_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :],
                mask=w_mask,
                other=0.0,
            ).to(tl.bfloat16)
            acc_l += tl.dot(wl, x_dot)

            wlg = tl.load(
                wlg_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :],
                mask=w_mask,
                other=0.0,
            ).to(tl.bfloat16)
            acc_lg += tl.dot(wlg, x_dot)

            wr = tl.load(
                wr_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :],
                mask=w_mask,
                other=0.0,
            ).to(tl.bfloat16)
            acc_r += tl.dot(wr, x_dot)

            wrg = tl.load(
                wrg_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :],
                mask=w_mask,
                other=0.0,
            ).to(tl.bfloat16)
            acc_rg += tl.dot(wrg, x_dot)

            wog = tl.load(
                wog_ptr + offs_h[:, None] * stride_w_h + offs_d[None, :],
                mask=w_mask,
                other=0.0,
            ).to(tl.bfloat16)
            acc_og += tl.dot(wog, x_dot)

        left_val = acc_l * (1.0 / (1.0 + tl.exp(-acc_lg)))
        right_val = acc_r * (1.0 / (1.0 + tl.exp(-acc_rg)))
        og_val = 1.0 / (1.0 + tl.exp(-acc_og))

        if HAS_MASK:
            left_val = left_val * mask_m[None, :]
            right_val = right_val * mask_m[None, :]

        out_mask = valid_h[:, None] & valid_m[None, :]

        left_ptrs = left_ptr + offs_h[:, None] * stride_out_m + offs_m[None, :]
        tl.store(left_ptrs, left_val.to(tl.bfloat16), mask=out_mask)

        right_ptrs = right_ptr + offs_h[:, None] * stride_out_m + offs_m[None, :]
        tl.store(right_ptrs, right_val.to(tl.bfloat16), mask=out_mask)

        og_ptrs = og_ptr + offs_h[:, None] * stride_out_m + offs_m[None, :]
        tl.store(og_ptrs, og_val.to(tl.bfloat16), mask=out_mask)


@triton.jit
def final_norm_gate_proj_kernel(
    core_ptr,
    gate_ptr,
    wn_ptr,
    bn_ptr,
    wout_ptr,
    out_ptr,
    M,
    D: tl.constexpr,
    H: tl.constexpr,
    stride_core_m,
    stride_gate_m,
    stride_wd,
    stride_wh,
    stride_out_m,
    BLOCK_H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    NUM_H: tl.constexpr,
    NUM_D: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    valid_m = offs_m < M

    # LayerNorm statistics over hidden dim.
    row_sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
    row_sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for hb in range(0, NUM_H):
        offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
        valid_h = offs_h < H
        v_mask = valid_h[:, None] & valid_m[None, :]

        v = tl.load(
            core_ptr + offs_h[:, None] * stride_core_m + offs_m[None, :],
            mask=v_mask,
            other=0.0,
        ).to(tl.float32)

        row_sum += tl.sum(v, axis=0)
        row_sumsq += tl.sum(v * v, axis=0)

    mean = row_sum / H
    var = row_sumsq / H - mean * mean
    var = tl.maximum(var, 0.0)
    rstd = 1.0 / tl.sqrt(var + EPS)

    # Final layer norm + gate + linear projection.
    for db in range(0, NUM_D):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)
        valid_d = offs_d < D

        acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

        for hb in range(0, NUM_H):
            offs_h = hb * BLOCK_H + tl.arange(0, BLOCK_H)
            valid_h = offs_h < H
            h_mask = valid_h[:, None] & valid_m[None, :]

            v = tl.load(
                core_ptr + offs_h[:, None] * stride_core_m + offs_m[None, :],
                mask=h_mask,
                other=0.0,
            ).to(tl.float32)

            g = tl.load(
                gate_ptr + offs_h[:, None] * stride_gate_m + offs_m[None, :],
                mask=h_mask,
                other=0.0,
            ).to(tl.float32)

            wn_h = tl.load(wn_ptr + offs_h, mask=valid_h, other=0.0).to(tl.float32)
            bn_h = tl.load(bn_ptr + offs_h, mask=valid_h, other=0.0).to(tl.float32)

            val = (v - mean[None, :]) * rstd[None, :]
            val = val * wn_h[:, None] + bn_h[:, None]
            val = val * g
            val = tl.where(h_mask, val, 0.0)

            a = tl.trans(val.to(tl.bfloat16))  # [BLOCK_M, BLOCK_H]

            # Load [BLOCK_D, BLOCK_H], transpose to [BLOCK_H, BLOCK_D].
            wt = tl.load(
                wout_ptr + offs_d[:, None] * stride_wd + offs_h[None, :] * stride_wh,
                mask=valid_d[:, None] & valid_h[None, :],
                other=0.0,
            ).to(tl.bfloat16)
            w = tl.trans(wt)

            acc += tl.dot(a, w)

        out_ptrs = out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.float32),
            mask=valid_m[:, None] & valid_d[None, :],
        )

            mask_b = mask_dummy

        left = torch.empty((H, M), dtype=dtype, device=device)
        right = torch.empty((H, M), dtype=dtype, device=device)
        og = torch.empty((H, M), dtype=dtype, device=device)

        fused_ln_proj_gate_kernel[grid_f](
            x_b,
            mask_b,
            gamma,
            beta,
            wl,
            wr,
            wlg,
            wrg,
            wog,
            left,
            right,
            og,
            M,
            D,
            H,
            x_b.stride(0),
            M,
            D,
            BLOCK_M=BLOCK_M_F,
            BLOCK_H=BLOCK_H_F,
            BLOCK_D=BLOCK_D_F,
            NUM_H=NUM_H_F,
            NUM_D=NUM_D_F,
            HAS_MASK=has_mask,
            EPS=eps,
            num_warps=8,
        )

        # Core N^3 update: for each hidden h,
        # left_h [N, N] @ right_h^T [N, N] -> [N, N].
        left_view = left.view(H, N, N)
        right_view = right.view(H, N, N)
        core = torch.bmm(left_view, right_view.transpose(1, 2))

        final_norm_gate_proj_kernel[grid_fin](
            core,
            og,
            wn,
            bn,
            wout,
            out[b],
            M,
            D,
            H,
            M,
            M,
            H,
            1,
            D,
            BLOCK_H=BLOCK_H_FIN,
            BLOCK_M=BLOCK_M_FIN,
            BLOCK_D=BLOCK_D_FIN,
            NUM_H=NUM_H_FIN,
            NUM_D=NUM_D_FIN,
            EPS=eps,
            num_warps=8,
        )

    return out
