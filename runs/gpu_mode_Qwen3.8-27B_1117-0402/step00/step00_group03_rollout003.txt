def _layer_norm_kernel(x_ptr, out_ptr, w_ptr, b_ptr, R, C, eps, BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    rows = (pid * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    r_mask = rows < R
    cols = tl.arange(0, BLOCK_C)
    c_mask = cols < C
    x_ptrs = x_ptr + rows[:, None] * C + cols[None, :]
    x_mask = r_mask[:, None] & c_mask[None, :]
    x = tl.load(x_ptrs, mask=x_mask, other=0.0).to(tl.float32)
    inv_c = 1.0 / C
    mean = tl.sum(x, axis=1) * inv_c
    xc = x - mean[:, None]
    var = tl.sum(tl.where(c_mask[None, :], xc * xc, 0.0), axis=1) * inv_c
    inv_std = 1.0 / tl.sqrt(var + eps)
    w = tl.load(w_ptr + cols, mask=c_mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=c_mask, other=0.0).to(tl.float32)
    y = (xc * inv_std[:, None]) * w[None, :] + b[None, :]
    y = tl.where(c_mask[None, :], y, 0.0)
    tl.store(out_ptr + rows[:, None] * C + cols[None, :], y.to(tl.bfloat16), mask=x_mask)

```python
import torch
import triton
import triton.language as tl


def _npow2(n: int) -> int:
    n = max(1, int(n))
    return 1 << (n - 1).bit_length()


def _as_transposed_weight(w: torch.Tensor, dtype: torch.dtype, device: torch.device) -> torch.Tensor:
    # Model weights are [out_dim, in_dim].  Triton GEMMs here use [in_dim, out_dim].
    return w.to(dtype, device=device).t().contiguous()


@triton.jit
def _layer_norm_kernel(
    x_ptr, out_ptr, w_ptr, b_ptr,
    R, C, eps,
    BLOCK_R: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = (pid * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    r_mask = rows < R

    cols = tl.arange(0, BLOCK_C)
    c_mask = cols < C

    x_ptrs = x_ptr + rows[:, None] * C + cols[None, :]
    x_mask = r_mask[:, None] & c_mask[None, :]

    x = tl.load(x_ptrs, mask=x_mask, other=0.0).to(tl.float32)

    inv_c = 1.0 / C
    mean = tl.sum(x, axis=1) * inv_c
    xc = x - mean[:, None]
    var = tl.sum(tl.where(c_mask[None, :], xc * xc, 0.0), axis=1) * inv_c
    inv_std = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + cols, mask=c_mask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + cols, mask=c_mask, other=0.0).to(tl.float32)

    y = (xc * inv_std[:, None]) * w[None, :] + b[None, :]
    y = tl.where(c_mask[None, :], y, 0.0)

    tl.store(out_ptr + rows[:, None] * C + cols[None, :], y.to(tl.bfloat16), mask=x_mask)


@triton.jit
def _proj_two_kernel(
    x_ptr, w1_ptr, w2_ptr, mask_ptr, out_ptr,
    R, C, D, s_seq,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
    HAS_MASK: tl.constexpr, K_MASKED: tl.constexpr, D_MASKED: tl.constexpr,
):
    """
    Compute y = (x @ W1^T) * sigmoid(x @ W2^T) * mask,
    and store it transposed as [B, D, seq_r, seq_c].
    """
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)

    rows = (pid_r * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    r_mask = rows < R

    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
    d_mask = d < D

    k = tl.arange(0, BLOCK_K)

    x_ptrs = x_ptr + rows[:, None] * C + k[None, :]
    w1_ptrs = w1_ptr + k[:, None] * D + d[None, :]
    w2_ptrs = w2_ptr + k[:, None] * D + d[None, :]

    acc1 = tl.zeros((BLOCK_R, BLOCK_D), dtype=tl.float32)
    acc2 = tl.zeros((BLOCK_R, BLOCK_D), dtype=tl.float32)

    if K_MASKED:
        for k0 in range(0, tl.cdiv(C, BLOCK_K)):
            kc = k0 * BLOCK_K + k
            k_mask = kc < C

            a = tl.load(x_ptrs, mask=r_mask[:, None] & k_mask[None, :], other=0.0)
            wm = k_mask[:, None] & d_mask[None, :]

            w1 = tl.load(w1_ptrs, mask=wm, other=0.0)
            w2 = tl.load(w2_ptrs, mask=wm, other=0.0)

            acc1 += tl.dot(a, w1)
            acc2 += tl.dot(a, w2)

            x_ptrs += BLOCK_K
            w1_ptrs += BLOCK_K * D
            w2_ptrs += BLOCK_K * D
    else:
        for k0 in range(0, tl.cdiv(C, BLOCK_K)):
            a = tl.load(x_ptrs, mask=r_mask[:, None], other=0.0)

            if D_MASKED:
                w1 = tl.load(w1_ptrs, mask=d_mask[None, :], other=0.0)
                w2 = tl.load(w2_ptrs, mask=d_mask[None, :], other=0.0)
            else:
                w1 = tl.load(w1_ptrs)
                w2 = tl.load(w2_ptrs)

            acc1 += tl.dot(a, w1)
            acc2 += tl.dot(a, w2)

            x_ptrs += BLOCK_K
            w1_ptrs += BLOCK_K * D
            w2_ptrs += BLOCK_K * D

    gate = tl.sigmoid(acc2)

    if HAS_MASK:
        m = tl.load(mask_ptr + rows, mask=r_mask, other=1.0).to(tl.float32)
        y = acc1 * gate * m[:, None]
    else:
        y = acc1 * gate

    y = y.to(tl.bfloat16)

    # rows are flattened as [B, S, S].  Output layout is [B, D, S, S],
    # flattened as [(b * D + d) * S * S + (i * S + j)].
    b_idx = rows // s_seq
    ij = rows - b_idx * s_seq

    d64 = d.to(tl.int64)
    base = (b_idx[None, :] * D + d64[:, None]) * s_seq + ij[None, :]
    store_mask = d_mask[:, None] & r_mask[None, :]

    tl.store(out_ptr + base, tl.trans(y), mask=store_mask)


@triton.jit
def _proj_one_kernel(
    x_ptr, w_ptr, out_ptr,
    R, C, D,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr,
    K_MASKED: tl.constexpr, D_MASKED: tl.constexpr,
):
    """
    Compute sigmoid(x @ W^T) and store row-major as [B, seq_r, seq_c, D].
    """
    pid_r = tl.program_id(0)
    pid_d = tl.program_id(1)

    rows = (pid_r * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    r_mask = rows < R

    d = pid_d * BLOCK_D + tl.arange(0, BLOCK_D)
    d_mask = d < D

    k = tl.arange(0, BLOCK_K)

    x_ptrs = x_ptr + rows[:, None] * C + k[None, :]
    w_ptrs = w_ptr + k[:, None] * D + d[None, :]

    acc = tl.zeros((BLOCK_R, BLOCK_D), dtype=tl.float32)

    if K_MASKED:
        for k0 in range(0, tl.cdiv(C, BLOCK_K)):
            kc = k0 * BLOCK_K + k
            k_mask = kc < C

            a = tl.load(x_ptrs, mask=r_mask[:, None] & k_mask[None, :], other=0.0)
            wm = k_mask[:, None] & d_mask[None, :]
            w = tl.load(w_ptrs, mask=wm, other=0.0)

            acc += tl.dot(a, w)

            x_ptrs += BLOCK_K
            w_ptrs += BLOCK_K * D
    else:
        for k0 in range(0, tl.cdiv(C, BLOCK_K)):
            a = tl.load(x_ptrs, mask=r_mask[:, None], other=0.0)

            if D_MASKED:
                w = tl.load(w_ptrs, mask=d_mask[None, :], other=0.0)
            else:
                w = tl.load(w_ptrs)

            acc += tl.dot(a, w)

            x_ptrs += BLOCK_K
            w_ptrs += BLOCK_K * D

    g = tl.sigmoid(acc).to(tl.bfloat16)

    out_ptrs = out_ptr + rows[:, None] * D + d[None, :]
    store_mask = r_mask[:, None] & d_mask[None, :]
    tl.store(out_ptrs, g, mask=store_mask)


@triton.jit
def _tri_core_kernel(
    L_ptr, R_ptr, O_ptr,
    S, s_seq,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    K_MASKED: tl.constexpr, MN_MASKED: tl.constexpr,
    INT64: tl.constexpr,
):
    """
    For each batch index bd = b * D + d:
        O_bd[i, j] = sum_k L_bd[i, k] * R_bd[j, k]
    """
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    if INT64:
        base = pid_b.to(tl.int64) * s_seq
    else:
        base = pid_b * s_seq

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    m_mask = offs_m < S
    n_mask = offs_n < S

    a_ptrs = L_ptr + base + offs_m[:, None] * S + offs_k[None, :]
    b_ptrs = R_ptr + base + offs_n[:, None] * S + offs_k[None, :]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    if K_MASKED:
        for k0 in range(0, tl.cdiv(S, BLOCK_K)):
            kc = k0 * BLOCK_K + offs_k
            k_mask = kc < S

            a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=n_mask[:, None] & k_mask[None, :], other=0.0)

            acc += tl.dot(a, tl.trans(b))

            a_ptrs += BLOCK_K
            b_ptrs += BLOCK_K
    else:
        for _ in range(0, tl.cdiv(S, BLOCK_K)):
            if MN_MASKED:
                a = tl.load(a_ptrs, mask=m_mask[:, None], other=0.0)
                b = tl.load(b_ptrs, mask=n_mask[:, None], other=0.0)
            else:
                a = tl.load(a_ptrs)
                b = tl.load(b_ptrs)

            acc += tl.dot(a, tl.trans(b))

            a_ptrs += BLOCK_K
            b_ptrs += BLOCK_K

    o_ptrs = O_ptr + base + offs_m[:, None] * S + offs_n[None, :]
    o = acc.to(tl.bfloat16)

    if MN_MASKED:
        tl.store(o_ptrs, o, mask=m_mask[:, None] & n_mask[None, :])
    else:
        tl.store(o_ptrs, o)


@triton.jit
def _final_kernel(
    O_ptr, G_ptr, W_ptr, NW_ptr, NB_ptr, OUT_ptr,
    R, C, D, s_seq, eps,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """
    Final epilogue:
    on = LayerNorm(O, dim=D)
    og = on * G
    out = og @ W_to_out^T
    O is stored as [B, D, S, S], G as [B, S, S, D].
    """
    pid = tl.program_id(0)

    rows = (pid * BLOCK_R + tl.arange(0, BLOCK_R)).to(tl.int64)
    r_mask = rows < R

    d = tl.arange(0, BLOCK_D)
    d_mask = d < D

    b_idx = rows // s_seq
    ij = rows - b_idx * s_seq

    d64 = d.to(tl.int64)
    o_base = b_idx[None, :] * (D * s_seq) + d64[:, None] * s_seq + ij[None, :]

    raw = tl.load(O_ptr + o_base, mask=d_mask[:, None] & r_mask[None, :], other=0.0).to(tl.float32)
    raw = tl.trans(raw)  # [BLOCK_R, BLOCK_D]

    gate_ptrs = G_ptr + rows[:, None] * D + d[None, :]
    gate = tl.load(gate_ptrs, mask=r_mask[:, None] & d_mask[None, :], other=0.0).to(tl.float32)

    nw = tl.load(NW_ptr + d, mask=d_mask, other=0.0).to(tl.float32)
    nb = tl.load(NB_ptr + d, mask=d_mask, other=0.0).to(tl.float32)

    raw = tl.where(d_mask[None, :], raw, 0.0)

    inv_d = 1.0 / D
    mean = tl.sum(raw, axis=1) * inv_d
    xc = raw - mean[:, None]
    var = tl.sum(tl.where(d_mask[None, :], xc * xc, 0.0), axis=1) * inv_d
    inv_std = 1.0 / tl.sqrt(var + eps)

    y = (xc * inv_std[:, None]) * nw[None, :] + nb[None, :]
    y = y * gate
    y = tl.where(d_mask[None, :], y, 0.0).to(tl.bfloat16)

    c = tl.arange(0, BLOCK_C)
    for c0 in range(0, tl.cdiv(C, BLOCK_C)):
        cc = c0 * BLOCK_C + c
        c_mask = cc < C

        w_mask = d_mask[:, None] & c_mask[None, :]
        w = tl.load(W_ptr + d64[:, None] * C + cc[None, :], mask=w_mask, other=0.0)

        acc = tl.zeros((BLOCK_R, BLOCK_C), dtype=tl.float32)
        acc += tl.dot(y, w)

        out_ptrs = OUT_ptr + rows[:, None] * C + cc[None, :]
        o_mask = r_mask[:, None] & c_mask[None, :]
        tl.store(out_ptrs, acc, mask=o_mask)


def custom_kernel(data):
    """
    Fused Triton forward pass for Triangle Multiplicative Update.

    Pipeline:
      1. LayerNorm + bf16 cast.
      2. Fused projection/gate/mask kernels producing:
         - L_t [B, hidden, S, S]
         - R_t [B, hidden, S, S]
         - G   [B, S, S, hidden]
      3. Batched per-channel GEMM kernel:
         O_t[b, d, i, j] = sum_k L_t[b, d, i, k] * R_t[b, d, j, k]
      4. Fused LayerNorm(out) * out_gate * to_out.
    """
    input_tensor, mask, weights, config = data

    B, S, _, C = input_tensor.shape
    D = int(config["hidden_dim"])
    device = input_tensor.device

    R = B * S * S
    s_seq = S * S
    eps = 1e-5

    # Keep norm computation in float32, then use bf16 for tensor-core GEMMs.
    x = input_tensor.to(torch.float32, device=device).contiguous()

    norm_w = weights["norm.weight"].to(torch.float32, device=device).contiguous()
    norm_b = weights["norm.bias"].to(torch.float32, device=device).contiguous()

    xn = torch.empty((B, S, S, C), dtype=torch.bfloat16, device=device)

    norm_block_c = max(16, _npow2(C))
    if norm_block_c <= 128:
        norm_block_r = 64
    elif norm_block_c <= 512:
        norm_block_r = 32
    else:
        norm_block_r = 16
    norm_block_r = max(16, norm_block_r)
    norm_warps = 4 if norm_block_c <= 128 else 8

    _layer_norm_kernel[(triton.cdiv(R, norm_block_r),)](
        x, xn, norm_w, norm_b,
        R, C, eps,
        norm_block_r, norm_block_c,
        num_warps=norm_warps,
        num_stages=3,
    )

    # Pre-transpose small projection weights to [in, out] layout.
    bf16 = torch.bfloat16
    wl_t = _as_transposed_weight(weights["left_proj.weight"], bf16, device)
    wlg_t = _as_transposed_weight(weights["left_gate.weight"], bf16, device)
    wr_t = _as_transposed_weight(weights["right_proj.weight"], bf16, device)
    wrg_t = _as_transposed_weight(weights["right_gate.weight"], bf16, device)
    wog_t = _as_transposed_weight(weights["out_gate.weight"], bf16, device)
    wout_t = _as_transposed_weight(weights["to_out.weight"], bf16, device)

    out_norm_w = weights["to_out_norm.weight"].to(torch.float32, device=device).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(torch.float32, device=device).contiguous()

    if mask is None or (isinstance(mask, torch.Tensor) and mask.numel() == 0):
        has_mask = False
        mask_t = xn
    else:
        has_mask = True
        if not isinstance(mask, torch.Tensor):
            mask = torch.as_tensor(mask, device=device)
        if mask.dim() == 4 and mask.shape[-1] == 1:
            mask = mask.squeeze(-1)
        mask_t = mask.to(bf16, device=device).contiguous()
        if mask_t.numel() != R:
            mask_t = mask_t.reshape(B, S, S).contiguous()

    left_t = torch.empty((B, D, S, S), dtype=bf16, device=device)
    right_t = torch.empty((B, D, S, S), dtype=bf16, device=device)
    gate = torch.empty((B, S, S, D), dtype=bf16, device=device)
    out_t = torch.empty((B, D, S, S), dtype=bf16, device=device)
    out = torch.empty((B, S, S, C), dtype=torch.float32, device=device)

    # Projection / gate block sizes.
    proj_block_d = min(128, max(16, _npow2(D)))
    proj_block_r = 32 if (C >= 512 or D >= 512) else 64
    proj_block_r = max(16, proj_block_r)
    proj_block_k = 64 if C >= 64 else max(16, _npow2(C))

    proj_k_masked = (C % proj_block_k) != 0
    proj_d_masked = (D % proj_block_d) != 0
    proj_grid = (triton.cdiv(R, proj_block_r), triton.cdiv(D, proj_block_d))
    proj_warps = 4
    proj_stages = 3

    _proj_two_kernel[proj_grid](
        xn, wl_t, wlg_t, mask_t, left_t,
        R, C, D, s_seq,
        proj_block_r, proj_block_d, proj_block_k,
        has_mask, proj_k_masked, proj_d_masked,
        num_warps=proj_warps,
        num_stages=proj_stages,
    )

    _proj_two_kernel[proj_grid](
        xn, wr_t, wrg_t, mask_t, right_t,
        R, C, D, s_seq,
        proj_block_r, proj_block_d, proj_block_k,
        has_mask, proj_k_masked, proj_d_masked,
        num_warps=proj_warps,
        num_stages=proj_stages,
    )

    _proj_one_kernel[proj_grid](
        xn, wog_t, gate,
        R, C, D,
        proj_block_r, proj_block_d, proj_block_k,
        proj_k_masked, proj_d_masked,
        num_warps=proj_warps,
        num_stages=proj_stages,
    )

    # Core N^3 part: D independent batched GEMMs per (batch, channel).
    if S <= 128:
        core_bm = 64
        core_bn = 64
        core_bk = 32
        core_warps = 4
    else:
        core_bm = 128
        core_bn = 128
        core_bk = 64
        core_warps = 8

    core_bm = max(16, core_bm)
    core_bn = max(16, core_bn)
    core_bk = max(16, core_bk)

    core_k_masked = (S % core_bk) != 0
    core_mn_masked = (S % core_bm != 0) or (S % core_bn != 0)
    core_int64 = (B * D * s_seq) >= (1 << 31)
    num_batches = B * D

    core_grid = (
        num_batches,
        triton.cdiv(S, core_bm),
        triton.cdiv(S, core_bn),
    )

    _tri_core_kernel[core_grid](
        left_t, right_t, out_t,
        S, s_seq,
        core_bm, core_bn, core_bk,
        core_k_masked, core_mn_masked, core_int64,
        num_warps=core_warps,
        num_stages=3,
    )

    # Final epilogue: layer norm + gate + to_out.
    final_block_d = max(16, _npow2(D))
    final_block_c = 128 if C > 16 else max(16, _npow2(C))
    final_block_r = 32 if (D <= 128 and C <= 128) else 16
    final_block_r = max(16, final_block_r)
    final_warps = 4 if (final_block_r * final_block_d <= 4096) else 8

    final_grid = (triton.cdiv(R, final_block_r),)

    _final_kernel[final_grid](
        out_t, gate, wout_t, out_norm_w, out_norm_b, out,
        R, C, D, s_seq, eps,
        final_block_r, final_block_d, final_block_c,
        num_warps=final_warps,
        num_stages=2,
    )

    return out
```
