```python
import torch
import triton
import triton.language as tl

# Fused TriMul forward.
# Input LayerNorm -> five linear projections -> mask/sigmoid gating ->
# batched pairwise dot in sequence dimension -> hidden LayerNorm + out-gate
# -> final projection.  Heavy GEMMs use cuBLAS via torch.matmul, while
# LayerNorm, fused gating, and fused final-normalize/gate/project are Triton.


@triton.jit
def _stable_sigmoid(x):
    ax = tl.where(x < 0, -x, x)
    z = tl.exp(-ax)
    return tl.where(x >= 0, 1.0 / (1.0 + z), z / (1.0 + z))


@triton.jit
def _layernorm_kernel(
    x_ptr,
    y_ptr,
    w_ptr,
    b_ptr,
    M,
    C,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_c = tl.arange(0, BLOCK_C)

    m_ok = offs_m < M
    c_ok = offs_c < C
    load_mask = m_ok[:, None] & c_ok[None, :]

    x = tl.load(
        x_ptr + offs_m[:, None] * C + offs_c[None, :],
        mask=load_mask,
        other=0.0,
    ).to(tl.float32)
    x = tl.where(c_ok[None, :], x, 0.0)

    mean = tl.sum(x, axis=1) / C
    xhat = x - mean[:, None]
    xhat = tl.where(c_ok[None, :], xhat, 0.0)
    var = tl.sum(xhat * xhat, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)
    bb = tl.load(b_ptr + offs_c, mask=c_ok, other=0.0).to(tl.float32)

    y = (xhat * rstd[:, None]) * w[None, :] + bb[None, :]
    y = tl.where(load_mask, y, 0.0)

    tl.store(
        y_ptr + offs_m[:, None] * C + offs_c[None, :],
        y.to(tl.bfloat16),
        mask=load_mask,
    )


@triton.jit
def _gate_kernel(
    left_ptr,
    right_ptr,
    gates_ptr,
    mask_ptr,
    M,
    H: tl.constexpr,
    H_BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid_m = tl.program_id(0)
    b = tl.program_id(1)
    hb = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_hl = tl.arange(0, H_BLOCK)
    h0 = hb * H_BLOCK
    offs_h = h0 + offs_hl

    m_ok = offs_m < M
    h_ok = offs_h < H
    mask2 = m_ok[:, None] & h_ok[None, :]

    M64 = tl.cast(M, tl.int64)
    b64 = tl.cast(b, tl.int64)
    offs_m64 = tl.cast(offs_m, tl.int64)
    offs_h64 = tl.cast(offs_h, tl.int64)

    m_off = offs_m64[:, None]
    h_off = offs_h64[None, :] * M64

    left_base = left_ptr + b64 * (H * M64) + h_off + m_off
    right_base = right_ptr + b64 * (H * M64) + h_off + m_off
    gates_base = gates_ptr + b64 * ((3 * H) * M64) + h_off + m_off

    left = tl.load(left_base, mask=mask2, other=0.0).to(tl.float32)
    right = tl.load(right_base, mask=mask2, other=0.0).to(tl.float32)
    lg = tl.load(gates_base, mask=mask2, other=0.0).to(tl.float32)
    rg = tl.load(gates_base + H * M64, mask=mask2, other=0.0).to(tl.float32)

    if HAS_MASK:
        mf = tl.load(mask_ptr + b64 * M64 + offs_m64, mask=m_ok, other=0.0).to(tl.float32)[:, None]
    else:
        mf = 1.0

    left = left * mf * _stable_sigmoid(lg)
    right = right * mf * _stable_sigmoid(rg)

    tl.store(left_base, left.to(tl.bfloat16), mask=mask2)
    tl.store(right_base, right.to(tl.bfloat16), mask=mask2)

def _final_norm_gate_matmul_kernel(

    if H_POW2:

        for cb in tl.static_range(C_BLOCKS):

def _next_pow2_int(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (int(n) - 1).bit_length()


if hasattr(torch.backends.cuda.matmul, "allow_bf16_reduced_precision_reduction"):
    torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = False


def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B = int(config.get("bs", input_tensor.shape[0]))
    N = int(config.get("seqlen", input_tensor.shape[1]))
    C = int(config.get("dim", input_tensor.shape[-1]))
    H = int(config.get("hidden_dim", input_tensor.shape[-1]))

    device = input_tensor.device
    dt = torch.bfloat16
    eps = 1e-5
    N2 = N * N
    M_pair = N2
    M_total = B * N2

    if not input_tensor.is_contiguous():
        input_tensor = input_tensor.contiguous()

    # ------------------------- input LayerNorm -------------------------
    x_norm = torch.empty((M_total, C), dtype=dt, device=device)

    BLOCK_C = _next_pow2_int(C)
    if BLOCK_C <= 128:
        LN_BLOCK_M = 64
    elif BLOCK_C <= 512:
        LN_BLOCK_M = 32
    else:
        LN_BLOCK_M = 16

    _layernorm_kernel[(triton.cdiv(M_total, LN_BLOCK_M),)](
        input_tensor,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        M_total,
        C,
        eps,
        BLOCK_M=LN_BLOCK_M,
        BLOCK_C=BLOCK_C,
    )

    # ----------------------- projection weights ------------------------
    W_left = weights["left_proj.weight"].to(dt).contiguous()
    W_right = weights["right_proj.weight"].to(dt).contiguous()
    W_gates = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dt).contiguous()

    W_out = weights["to_out.weight"].to(dt).t().contiguous()
    nw = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    nb = weights["to_out_norm.bias"].to(torch.float32).contiguous()

    # -------------------------------- projections ----------------------
    # We compute the projections in transposed layout:
    #   left:   [B, H, N*N]
    #   right:  [B, H, N*N]
    #   gates:  [B, 3H, N*N]
    # This keeps each (b,h) matrix contiguous for the next bmm.
    left = torch.empty((B, H, N2), dtype=dt, device=device)
    right = torch.empty((B, H, N2), dtype=dt, device=device)
    gates = torch.empty((B, 3 * H, N2), dtype=dt, device=device)

    x_norm_2d = x_norm
    for bi in range(B):
        start = bi * N2
        Xb = x_norm_2d[start : start + N2]
        Xbt = Xb.t()
        torch.matmul(W_left, Xbt, out=left[bi])
        torch.matmul(W_right, Xbt, out=right[bi])
        torch.matmul(W_gates, Xbt, out=gates[bi])

    del x_norm, x_norm_2d, Xb, Xbt, W_left, W_right, W_gates

    # ------------------------- mask and gates --------------------------
    if mask is None or not torch.is_tensor(mask) or mask.numel() == 0:
        mask_t = torch.empty(0, dtype=dt, device=device)
        has_mask = False
    else:
        mask_t = mask.to(dt).reshape(B, N2).contiguous()
        has_mask = True

    GATE_BLOCK_M = 32
    GATE_H_BLOCK = 128
    _gate_kernel[(triton.cdiv(N2, GATE_BLOCK_M), B, triton.cdiv(H, GATE_H_BLOCK))](
        left,
        right,
        gates,
        mask_t,
        N2,
        H=H,
        H_BLOCK=GATE_H_BLOCK,
        HAS_MASK=has_mask,
        BLOCK_M=GATE_BLOCK_M,
    )

    del mask_t

    # --------------------------- core TriMul ---------------------------
    # For each h: out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]
    left_3d = left.view(B * H, N, N)
    right_3d = right.view(B * H, N, N)
    out_p = torch.bmm(left_3d, right_3d.transpose(1, 2))
    out_p = out_p.contiguous()

    del left, right, left_3d, right_3d

    # ------------------- final norm + gate + projection ----------------
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    # og is the third H-block of [B, 3H, N*N]
    og = gates[:, 2 * H : 3 * H, :]
    og_b_stride = 3 * H * N2
    og_h_stride = N2

    H_POW2 = (H & (H - 1)) == 0 and 16 <= H <= 256
    if H_POW2:
        if H <= 128:
            FINAL_BLOCK_M = 64
        else:
            FINAL_BLOCK_M = 32
        H_BLOCK = H
    else:
        FINAL_BLOCK_M = 32
        H_BLOCK = 128

    if C <= 16:
        C_BLOCK = 16
    elif C <= 32:
        C_BLOCK = 32
    elif C <= 64:
        C_BLOCK = 64
    else:
        C_BLOCK = 128

    C_BLOCKS = (C + C_BLOCK - 1) // C_BLOCK
    H_BLOCKS = (H + H_BLOCK - 1) // H_BLOCK

    _final_norm_gate_matmul_kernel[(triton.cdiv(N2, FINAL_BLOCK_M), B)](
        out_p,
        og,
        nw,
        nb,
        W_out,
        out,
        N2,
        C,
        eps,
        H=H,
        C_BLOCK=C_BLOCK,
        C_BLOCKS=C_BLOCKS,
        H_POW2=H_POW2,
        H_BLOCK=H_BLOCK,
        H_BLOCKS=H_BLOCKS,
        BLOCK_M=FINAL_BLOCK_M,
        OG_B_STRIDE=og_b_stride,
        OG_H_STRIDE=og_h_stride,
    )

    return out
```
