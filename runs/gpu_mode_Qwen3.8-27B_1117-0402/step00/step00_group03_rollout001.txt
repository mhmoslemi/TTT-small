    B, N, N, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Step 1-9: Use PyTorch for projections and gating
    # LayerNorm
    normed = F.layer_norm(input_tensor, (C,), weights['norm.weight'], weights['norm.bias'])
    
    # Projections
    left = torch.matmul(normed, weights['left_proj.weight'].T)  # [B, N, N, hidden_dim]
    right = torch.matmul(normed, weights['right_proj.weight'].T)  # [B, N, N, hidden_dim]

    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask
    
    # Gates
    left_gate = torch.sigmoid(torch.matmul(normed, weights['left_gate_weight'].T))
    right_gate = torch.sigmoid(torch.matmul(normed, weights['right_gate_weight'].T))
    out_gate = torch.sigmoid(torch.matmul(normed, weights['out_gate_weight'].T))
    
    left = left * left_gate
    right = right * right_gate
    
    # Step 10: Triton kernel for the einsum
    out = torch.empty(B, N, N, hidden_dim, device=input_tensor.device, dtype=input_tensor.dtype)
    # Launch kernel...
    
    # Step 11-13: LayerNorm, gate, projection
    out = F.layer_norm(out, (hidden_dim,), weights['to_out_norm.weight'], weights['to_out_norm.bias'])

    out = torch.matmul(out, weights['to_out_weight'].T)  # [B, N, N, C]
    
    return out

```python
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Grid: (B * hidden_dim * cdiv(N, BLOCK_M) * cdiv(N, BLOCK_N),)
    # Each program handles one (b, d, i_block, j_block)
    
    pid = tl.program_id(0)
    
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    
    # Decompose pid
    pid_ij = pid % (num_i_blocks * num_j_blocks)
    pid_bd = pid // (num_i_blocks * num_j_blocks)
    
    pid_i = pid_ij % num_i_blocks
    pid_j = pid_ij // num_i_blocks
    
    b = pid_bd // hidden_dim
    d = pid_bd % hidden_dim
    
    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Base pointers
    base = b * N * N * hidden_dim + d
    
    a_ptrs = left_ptr + base + offs_i[:, None] * (N * hidden_dim) + offs_k[None, :] * hidden_dim
    b_ptrs = right_ptr + base + offs_j[None, :] * (N * hidden_dim) + offs_k[:, None] * hidden_dim
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        a = tl.load(a_ptrs, mask=(offs_k[None, :] + k_start < N) & (offs_i[:, None] < N), other=0.0)
        b_mat = tl.load(b_ptrs, mask=(offs_k[:, None] + k_start < N) & (offs_j[None, :] < N), other=0.0)
        acc = tl.dot(a, b_mat, acc)
        a_ptrs += BLOCK_K * hidden_dim
        b_ptrs += BLOCK_K * hidden_dim
    
    # Store
    c_ptrs = out_ptr + base + offs_i[:, None] * (N * hidden_dim) + offs_j[None, :] * hidden_dim
    tl.store(c_ptrs, acc.to(out_ptr.dtype.element_ty), mask=(offs_i[:, None] < N) & (offs_j[None, :] < N))
```

- Extra transpose after kernel: one additional pass over the data

The extra transpose reads and writes B*N*N*hd elements. The strided writes in the kernel would waste bandwidth on every write. Since the kernel does N iterations of the inner loop (each writing a [BLOCK_M, BLOCK_N] tile), the total wasted bandwidth with strided writes would be much higher than one transpose pass.

So: kernel writes in transposed layout, then we transpose back.

But that's a view, and then LayerNorm would work on the last dimension.

- Write: B*N*N*hd*4 bytes

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_batched_matmul(
    left_ptr, right_ptr, out_ptr,
    N, num_bd,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    For each (b, d), compute:
        out_t[b, :, d, :] = left_t[b, :, d, :] @ right_t[b, :, d, :].T
    
    left_t has shape [B, N, hidden_dim, N] (contiguous)
    right_t has shape [B, N, hidden_dim, N] (contiguous)
    out_t has shape [B, N, hidden_dim, N] (contiguous)
    
    For fixed (b, d):
        left_t[b, :, d, :] is a [N, N] matrix with:
            row i, col k at offset (b*N + i) * hidden_dim * N + d * N + k
            row stride: hidden_dim * N
            col stride: 1
        right_t[b, :, d, :] is a [N, N] matrix with:
            row j, col k at offset (b*N + j) * hidden_dim * N + d * N + k
            row stride: hidden_dim * N
            col stride: 1
        out_t[b, :, d, :] is a [N, N] matrix with:
            row i, col j at offset (b*N + i) * hidden_dim * N + d * N + j
            row stride: hidden_dim * N
            col stride: 1
    """
    pid = tl.program_id(0)
    
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    
    # pid decodes to (bd, m_block, n_block)
    pid_mn = pid % (num_m_blocks * num_n_blocks)
    pid_bd = pid // (num_m_blocks * num_n_blocks)
    pid_m = pid_mn % num_m_blocks
    pid_n = pid_mn // num_m_blocks
    
    # Base offset for this (b, d) pair
    # left_t[b, i, d, k] at: ((b * N + i) * hidden_dim + d) * N + k
    # So the base for row i is: (b * N + i) * hidden_dim * N + d * N
    base = pid_bd * N  # this gives us (b*N + offset_in_hidden)... 
    # Actually, let me think about this differently.
    # pid_bd ranges from 0 to B*hidden_dim - 1
    # b = pid_bd // hidden_dim, d = pid_bd % hidden_dim
    # Base offset in the tensor for (b, d): b * N * hidden_dim * N + d * N
    # = (b * N * hidden_dim + d) * N
    # = pid_bd * N  (since pid_bd = b * hidden_dim + d, and b * N * hidden_dim + d = b*N*hd + d)
    # Wait: b * N * hd * N + d * N = (b * N * hd + d) * N
    # And pid_bd = b * hd + d
    # So (b * N * hd + d) * N ≠ pid_bd * N unless N = 1
    
    # Let me just compute it directly:
    b = pid_bd // hidden_dim
    d = pid_bd % hidden_dim
    base = (b * N * hidden_dim + d) * N
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers for A tile: left_t[b, i, d, k]
    # Address: base + i * (hidden_dim * N) + k
    a_ptrs = left_ptr + base + offs_m[:, None] * (hidden_dim * N) + offs_k[None, :]
    
    # Pointers for B tile: right_t[b, j, d, k], we need B.T[k, j]
    # B[k, j] = right_t[b, j, d, k] at address: base + j * (hidden_dim * N) + k
    # So B.T[k, j] has: row k, col j
    # Address of B.T[k, j] = base + j * (hidden_dim * N) + k
    # As a 2D pointer with [k, j] indexing:
    b_ptrs = right_ptr + base + offs_n[None, :] * (hidden_dim * N) + offs_k[:, None]
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        a_mask = (offs_k[None, :] + k_start < N) & (offs_m[:, None] < N)
        b_mask = (offs_k[:, None] + k_start < N) & (offs_n[None, :] < N)
        
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)
        
        acc = tl.dot(a, b, acc)
        
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K
    
    # Store: out_t[b, i, d, j] at base + i * (hidden_dim * N) + j
    c_ptrs = out_ptr + base + offs_m[:, None] * (hidden_dim * N) + offs_n[None, :]
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    
    num_m_blocks = tl.cdiv(N, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    num_blocks_per_bd = num_m_blocks * num_n_blocks
    
    pid_mn = pid % num_blocks_per_bd
    pid_bd = pid // num_blocks_per_bd
    pid_m = pid_mn % num_m_blocks
    pid_n = pid_mn // num_m_blocks
    
    b = pid_bd // HIDDEN_DIM
    d = pid_bd % HIDDEN_DIM
    base = (b * N * HIDDEN_DIM + d) * N
    
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    a_ptrs = left_ptr + base + offs_m[:, None] * (HIDDEN_DIM * N) + offs_k[None, :]
    b_ptrs = right_ptr + base + offs_n[None, :] * (HIDDEN_DIM * N) + offs_k[:, None]
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        a_mask = (offs_k[None, :] + k_start < N) & (offs_m[:, None] < N)
        b_mask = (offs_k[:, None] + k_start < N) & (offs_n[None, :] < N)
        
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.bfloat16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.bfloat16)
        
        acc = tl.dot(a, b, acc)
        
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K
    
    c_ptrs = out_ptr + base + offs_m[:, None] * (HIDDEN_DIM * N) + offs_n[None, :]
    c_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, N, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    
    # Step 1: LayerNorm
    normed = F.layer_norm(input_tensor, (C,), weights['norm.weight'], weights['norm.bias'])
    
    # Step 2-9: Projections, masking, gating (all in PyTorch)
    left = torch.matmul(normed, weights['left_proj.weight'].T)  # [B, N, N, hidden_dim]
    right = torch.matmul(normed, weights['right_proj.weight'].T)
    
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp
    
    left_gate = torch.sigmoid(torch.matmul(normed, weights['left_gate_weight'].T))
    right_gate = torch.sigmoid(torch.matmul(normed, weights['right_gate_weight'].T))
    out_gate = torch.sigmoid(torch.matmul(normed, weights['out_gate_weight'].T))
    
    left = left * left_gate
    right = right * right_gate
    
    # Transpose for efficient matmul: [B, N, N, hd] -> [B, N, hd, N]
    left_t = left.permute(0, 1, 3, 2).contiguous()  # [B, N, hd, N]
    right_t = right.permute(0, 1, 3, 2).contiguous()  # [B, N, hd, N]
    
    # Step 10: Triton kernel for batched matmul
    out_t = torch.empty(B, N, hidden_dim, N, device=device, dtype=torch.float32)
    
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    
    num_m_blocks = triton.cdiv(N, BLOCK_M)
    num_n_blocks = triton.cdiv(N, BLOCK_N)
    grid = (B * hidden_dim * num_m_blocks * num_n_blocks,)
    
    tri_mul_batched_matmul_kernel[grid](
        left_t, right_t, out_t,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )
    
    # Transpose back: [B, N, hd, N] -> [B, N, N, hd]
    out = out_t.permute(0, 1, 3, 2).contiguous()
    
    # Step 11-13: LayerNorm, gate, projection
    out = F.layer_norm(out, (hidden_dim,), weights['to_out_norm.weight'], weights['to_out_norm.bias'])
    out = out * out_gate
    out = torch.matmul(out, weights['to_out_weight'].T)  # [B, N, N, C]
    
    return out
