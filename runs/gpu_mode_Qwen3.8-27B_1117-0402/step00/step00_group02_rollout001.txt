```python
@triton.jit
def trimul_core_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,  # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,  # right strides
    stride_ob, stride_oi, stride_oj, stride_od,  # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_j = tl.cdiv(N, BLOCK_J)
    num_i = tl.cdiv(N, BLOCK_I)
    
    # Decode (b, d, i_block, j_block)
    pid_j = pid % num_j
    pid_rest = pid // num_j
    pid_i = pid_rest % num_i
    pid_bd = pid_rest // num_i
    d = pid_bd % D
    b = pid_bd // D
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        offs_k_cur = k_start + offs_k
        k_mask = offs_k_cur < N
        
        # Load left[b, offs_i, offs_k_cur, d]: shape [BLOCK_I, BLOCK_K]
        left_ptrs = left_ptr + b * stride_lb + offs_i[:, None] * stride_li + offs_k_cur[None, :] * stride_lk + d * stride_ld
        left_tile = tl.load(left_ptrs, mask=k_mask[None, :], other=0.0)
        
        # Load right[b, offs_j, offs_k_cur, d]: shape [BLOCK_J, BLOCK_K]
        right_ptrs = right_ptr + b * stride_rb + offs_j[:, None] * stride_rj + offs_k_cur[None, :] * stride_rk + d * stride_rd
        right_tile = tl.load(right_ptrs, mask=k_mask[None, :], other=0.0)
        
        # acc += left_tile @ right_tile^T
        # right_tile^T is [BLOCK_K, BLOCK_J]
        acc = tl.dot(left_tile, tl.trans(right_tile), acc)
    
    # Store
    out_ptrs = out_ptr + b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + d * stride_od
    i_mask = offs_i < N
    j_mask = offs_j < N
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,  # sequence length (same for all dims)
    stride_ab, stride_ad, stride_ai, stride_ak,  # A strides (b, d, i, k)
    stride_bb, stride_bd, stride_bj, stride_bk,  # B strides (b, d, j, k)
    stride_cb, stride_cd, stride_ci, stride_cj,  # C strides (b, d, i, j)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    num_j = tl.cdiv(N, BLOCK_J)
    num_i = tl.cdiv(N, BLOCK_I)
    
    pid_j = pid % num_j
    pid_rest = pid // num_j
    pid_i = pid_rest % num_i
    pid_bd = pid_rest // num_i
    
    # pid_bd encodes (b, d)
    # d = pid_bd % D, b = pid_bd // D (but we pass the combined stride)
    # Actually, let's just use pid_bd as a flat index into the [B*D] dimension
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Pointers
    a_base = a_ptr + pid_bd * stride_ad + offs_i[:, None] * stride_ai
    b_base = b_ptr + pid_bd * stride_bd + offs_j[:, None] * stride_bj
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a_ptrs = a_base + (k_start + offs_k)[None, :] * stride_ak
        b_ptrs = b_base + (k_start + offs_k)[None, :] * stride_bk
        
        a_tile = tl.load(a_ptrs, mask=k_mask[None, :], other=0.0)
        b_tile = tl.load(b_ptrs, mask=k_mask[None, :], other=0.0)
        
        acc = tl.dot(a_tile, tl.trans(b_tile), acc)
    
    # Store
    c_ptrs = c_ptr + pid_bd * stride_cd + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    i_mask = offs_i < N
    j_mask = offs_j < N
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])
```

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    B, N, _, C = input_tensor.shape
    device = input_tensor.device
    
    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # 1. LayerNorm
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)
    
    # Convert to bf16 for the matmul parts
    x_bf16 = x.to(torch.bfloat16)
    
    # 2. Projections (linear layers)
    left = F.linear(x_bf16, left_proj_weight.to(torch.bfloat16))
    right = F.linear(x_bf16, right_proj_weight.to(torch.bfloat16))
    
    # 3. Gates
    left_gate = torch.sigmoid(F.linear(x_bf16, left_gate_weight.to(torch.bfloat16)))
    right_gate = torch.sigmoid(F.linear(x_bf16, right_gate_weight.to(torch.bfloat16)))
    out_gate = torch.sigmoid(F.linear(x_bf16, out_gate_weight.to(torch.bfloat16)))
    
    # 4. Apply mask and gates
    if mask is not None and mask.numel() > 0:
        mask_expanded = mask.unsqueeze(-1).to(torch.bfloat16)
        left = left * mask_expanded
        right = right * mask_expanded
    
    left = left * left_gate
    right = right * right_gate
    
    # 5. Permute for matmul: [B, N, N, D] -> [B, D, N, N]
    D = hidden_dim
    # left is [B, N, N, D], we want [B, D, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, D, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, D, N, N]
    
    # Flatten batch dims: [B*D, N, N]
    total_batches = B * D
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)
    
    # 6. Triton matmul kernel
    out_flat = torch.empty(total_batches, N, N, device=device, dtype=torch.float32)
    
    BLOCK_I = 128
    BLOCK_J = 128
    BLOCK_K = 64
    
    # For N < BLOCK_I or N < BLOCK_J, adjust
    if N <= 128:
        BLOCK_I = min(128, triton.next_power_of_2(N))
        BLOCK_J = min(128, triton.next_power_of_2(N))
    
    grid = (total_batches * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)
    trimul_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, total_batches,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )
    
    # 7. Permute back: [B*D, N, N] -> [B, N, N, D]
    out_perm = out_flat.view(B, D, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, D]
    
    # 8. LayerNorm on hidden_dim
    out = F.layer_norm(out, (D,), to_out_norm_weight, to_out_norm_bias)
    
    # 9. Apply out_gate
    out = out * out_gate.to(torch.float32)
    
    # 10. Final projection
    out = F.linear(out, to_out_weight)
    
    return out.to(torch.float32)
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C[bd, i, j] = sum_k A[bd, i, k] * B[bd, j, k]
    where A, B, C are [total_batches, N, N] contiguous tensors.
    """
    pid = tl.program_id(0)
    num_j = tl.cdiv(N, BLOCK_J)
    num_i = tl.cdiv(N, BLOCK_I)
    
    pid_j = pid % num_j
    pid_rest = pid // num_j
    pid_i = pid_rest % num_i
    pid_bd = pid_rest // num_i
    
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)
    
    a_base = a_ptr + pid_bd * N * N
    b_base = b_ptr + pid_bd * N * N
    
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    
    for k_start in range(0, N, BLOCK_K):
        k_mask = (k_start + offs_k) < N
        a_ptrs = a_base + offs_i[:, None] * N + (k_start + offs_k)[None, :]
        b_ptrs = b_base + offs_j[:, None] * N + (k_start + offs_k)[None, :]
        
        a_tile = tl.load(a_ptrs, mask=k_mask[None, :], other=0.0)
        b_tile = tl.load(b_ptrs, mask=k_mask[None, :], other=0.0)
        
        acc = tl.dot(a_tile, tl.trans(b_tile), acc)
    
    c_ptrs = c_ptr + pid_bd * N * N + offs_i[:, None] * N + offs_j[None, :]
    i_mask = offs_i < N
    j_mask = offs_j < N
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    TriMul forward pass: fuses LayerNorm, projections, gating, 
    N^3 pairwise interaction (batched matmul via Triton), and output projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    D = hidden_dim
    
    # Access weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # 1. Input LayerNorm (over dim)
    x = F.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)
    
    # Convert to bf16 for efficient matmuls
    x_bf16 = x.to(torch.bfloat16)
    
    # 2. Projections
    left = F.linear(x_bf16, left_proj_weight.to(torch.bfloat16))
    right = F.linear(x_bf16, right_proj_weight.to(torch.bfloat16))
    
    # 3. Gates
    left_gate = torch.sigmoid(F.linear(x_bf16, left_gate_weight.to(torch.bfloat16)))
    right_gate = torch.sigmoid(F.linear(x_bf16, right_gate_weight.to(torch.bfloat16)))
    out_gate = torch.sigmoid(F.linear(x_bf16, out_gate_weight.to(torch.bfloat16)))
    
    # 4. Apply mask
    if mask is not None and mask.numel() > 0:
        mask_bf16 = mask.to(torch.bfloat16).unsqueeze(-1)
        left = left * mask_bf16
        right = right * mask_bf16
    
    # 5. Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # 6. Permute for matmul: [B, N, N, D] -> [B, D, N, N] -> [B*D, N, N]
    left_perm = left.permute(0, 3, 1, 2).reshape(B * D, N, N).contiguous()
    right_perm = right.permute(0, 3, 1, 2).reshape(B * D, N, N).contiguous()
    
    # 7. Triton batched matmul
    out_flat = torch.empty(B * D, N, N, device=device, dtype=torch.float32)
    
    # Choose block sizes
    if N >= 256:
        BLOCK_I = 128
        BLOCK_J = 128
    else:
        BLOCK_I = triton.next_power_of_2(N)
        BLOCK_J = triton.next_power_of_2(N)
    BLOCK_K = 64
    
    total_batches = B * D
    grid = (total_batches * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),)
    trimul_matmul_kernel[grid](
        left_perm, right_perm, out_flat,
        N,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )
    
    # 8. Permute back: [B*D, N, N] -> [B, N, N, D]
    out = out_flat.reshape(B, D, N, N).permute(0, 2, 3, 1).contiguous()
    
    # 9. Output LayerNorm (over hidden_dim)
    out = F.layer_norm(out, (D,), to_out_norm_weight, to_out_norm_bias)
    
    # 10. Apply out_gate
    out = out * out_gate.to(torch.float32)
    
    # 11. Final projection
    out = F.linear(out, to_out_weight)
    
    return out.to(torch.float32)
```
