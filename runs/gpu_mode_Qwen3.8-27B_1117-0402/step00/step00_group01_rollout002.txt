```
        x_c = tl.load(x_ptr + offs_m[:, None] * C + offs_c[None, :], mask=load_mask, other=0.0)
        sum_x += tl.sum(x_c, axis=1)
        sum_xx += tl.sum(x_c * x_c, axis=1)

    mean = sum_x / C
    var = sum_xx / C - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    b_idx = offs_m // N2
    local = offs_m - b_idx * N2
    m_base = b_idx * (H * N2) + local

    else:

        h_mask = offs_h < H
        acc_l = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_r = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

            offs_c = c0 + tl.arange(0, BLOCK_C)
            c_mask = offs_c < C

            x_c = tl.load(x_ptr + offs_m[:, None] * C + offs_c[None, :], mask=load_mask, other=0.0)
            nw = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0)
            nb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0)
            xn = (x_c - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            xn_b = xn.to(tl.bfloat16)

            w_mask = c_mask[:, None] & h_mask[None, :]
            wl = tl.load(wl_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0)
            wr = tl.load(wr_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0)
            wlg = tl.load(wlg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0)
            wrg = tl.load(wrg_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0)
            wog = tl.load(wog_ptr + offs_c[:, None] * H + offs_h[None, :], mask=w_mask, other=0.0)

            acc_l = tl.dot(xn_b, wl, acc_l)
            acc_r = tl.dot(xn_b, wr, acc_r)
            acc_lg = tl.dot(xn_b, wlg, acc_lg)
            acc_rg = tl.dot(xn_b, wrg, acc_rg)
            acc_og = tl.dot(xn_b, wog, acc_og)

        lg = 1.0 / (1.0 + tl.exp(-acc_lg))
        rg = 1.0 / (1.0 + tl.exp(-acc_rg))
        og = 1.0 / (1.0 + tl.exp(-acc_og))
        left = acc_l * lg * mask_vals[:, None]
        right = acc_r * rg * mask_vals[:, None]

        store_mask = h_mask[:, None] & m_mask[None, :]
        L_off = m_base[None, :] + offs_h[:, None] * N2
        R_off = m_base[None, :] + offs_h[:, None] * N2
        tl.store(L_ptr + L_off, tl.trans(left.to(tl.bfloat16)), mask=store_mask)
        tl.store(R_ptr + R_off, tl.trans(right.to(tl.bfloat16)), mask=store_mask)

        og_off = offs_m[:, None] * H + offs_h[None, :]
        tl.store(OG_ptr + og_off, og.to(tl.bfloat16), mask=m_mask[:, None] & h_mask[None, :])
```

```python
import torch
import triton
import triton.language as tl


"""
TriMul outgoing pass.

Fused Triton + cuBLAS implementation:
1. A Triton kernel fuses input LayerNorm, five linear projections
   (left, right, left_gate, right_gate, out_gate), sigmoid gating,
   and input masking. It writes:
     - L/R in [B, H, N, N] bf16 for efficient batched GEMM contraction
     - out_gate in [B*N*N, H] bf16 for the final fused kernel
2. The N^3 contraction is performed as H batched matrix products using
   torch.bmm:
     out[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
3. A fused Triton kernel applies the second LayerNorm (with learned
   weight/bias), multiplies by out_gate, and performs the final linear
   projection to float32 output.
"""


@triton.jit
def _trimul_proj_kernel(
    x_ptr,
    mask_ptr,
    L_ptr,
    R_ptr,
    OG_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wr_ptr,
    wlg_ptr,
    wrg_ptr,
    wog_ptr,
    M,
    N2,
    N2H,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < M

    # LayerNorm statistics over the feature dimension C.
    sum_x = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_xx = tl.zeros((BLOCK_M,), dtype=tl.float32)
    for c0 in range(0, C, BLOCK_C):
        offs_c = c0 + tl.arange(0, BLOCK_C)
        c_mask = offs_c < C
        load_mask = m_mask[:, None] & c_mask[None, :]
        x_c = tl.load(
            x_ptr + offs_m[:, None] * C + offs_c[None, :],
            mask=load_mask,
            other=0.0,
        )
        sum_x += tl.sum(x_c, axis=1)
        sum_xx += tl.sum(x_c * x_c, axis=1)

    mean = sum_x / C
    var = sum_xx / C - mean * mean
    rstd = 1.0 / tl.sqrt(tl.maximum(var, 0.0) + EPS)

    # Addresses for L/R stored as [B, H, N, N].
    b_idx = offs_m // N2
    local = offs_m - b_idx * N2
    m_base = b_idx * N2H + local

    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + offs_m, mask=m_mask, other=1.0).to(tl.float32)
    else:
        mask_vals = tl.where(m_mask, 1.0, 0.0)

    for h0 in range(0, H, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H

        acc_l = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_r = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_lg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_rg = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)
        acc_og = tl.zeros((BLOCK_M, BLOCK_H), dtype=tl.float32)

        for c0 in range(0, C, BLOCK_C):
            offs_c = c0 + tl.arange(0, BLOCK_C)
            c_mask = offs_c < C
            load_mask = m_mask[:, None] & c_mask[None, :]

            x_c = tl.load(
                x_ptr + offs_m[:, None] * C + offs_c[None, :],
                mask=load_mask,
                other=0.0,
            )

            nw = tl.load(norm_w_ptr + offs_c, mask=c_mask, other=0.0)
            nb = tl.load(norm_b_ptr + offs_c, mask=c_mask, other=0.0)
            xn = (x_c - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            xn_bf = xn.to(tl.bfloat16)

            w_mask = c_mask[:, None] & h_mask[None, :]
            wl = tl.load(
                wl_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=w_mask,
                other=0.0,
            )
            wr = tl.load(
                wr_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=w_mask,
                other=0.0,
            )
            wlg = tl.load(
                wlg_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=w_mask,
                other=0.0,
            )
            wrg = tl.load(
                wrg_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=w_mask,
                other=0.0,
            )
            wog = tl.load(
                wog_ptr + offs_c[:, None] * H + offs_h[None, :],
                mask=w_mask,
                other=0.0,
            )

            acc_l = tl.dot(xn_bf, wl, acc_l)
            acc_r = tl.dot(xn_bf, wr, acc_r)
            acc_lg = tl.dot(xn_bf, wlg, acc_lg)
            acc_rg = tl.dot(xn_bf, wrg, acc_rg)
            acc_og = tl.dot(xn_bf, wog, acc_og)

        # Store out_gate in [M, H] layout.
        og = 1.0 / (1.0 + tl.exp(-acc_og))
        og_off = offs_m[:, None] * H + offs_h[None, :]
        tl.store(
            OG_ptr + og_off,
            og.to(tl.bfloat16),
            mask=m_mask[:, None] & h_mask[None, :],
        )

        # Left/right, masked and gated.
        lg = 1.0 / (1.0 + tl.exp(-acc_lg))
        rg = 1.0 / (1.0 + tl.exp(-acc_rg))
        left = acc_l * lg * mask_vals[:, None]
        right = acc_r * rg * mask_vals[:, None]

        # L/R are [B, H, N, N]. Store as [BLOCK_H, BLOCK_M] tiles so that
        # each H row is contiguous over pair positions.
        store_mask = h_mask[:, None] & m_mask[None, :]
        loff = m_base[None, :] + offs_h[:, None] * N2
        roff = m_base[None, :] + offs_h[:, None] * N2
        tl.store(L_ptr + loff, tl.trans(left.to(tl.bfloat16)), mask=store_mask)
        tl.store(R_ptr + roff, tl.trans(right.to(tl.bfloat16)), mask=store_mask)


@triton.jit
def _trimul_final_kernel_small_h(
    Y_ptr,
    OG_ptr,
    W_ptr,
    TOW_W_ptr,
    TOW_B_ptr,
    OUT_ptr,
    M,
    N2,
    N2H,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_OUT: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < M

    offs_h = tl.arange(0, BLOCK_H)
    h_mask = offs_h < H

    b_idx = offs_m // N2
    local = offs_m - b_idx * N2
    row_base = b_idx * N2H + local

    # Y is [B, H, N, N]. Load as [BLOCK_H, BLOCK_M], then transpose.
    y_off = row_base[None, :] + offs_h[:, None] * N2
    y_h = tl.load(
        Y_ptr + y_off,
        mask=m_mask[None, :] & h_mask[:, None],
        other=0.0,
    ).to(tl.float32)
    y = tl.trans(y_h)

    sum_y = tl.sum(y, axis=1)
    sum_y2 = tl.sum(y * y, axis=1)
    mean_y = sum_y / H
    var_y = sum_y2 / H - mean_y * mean_y
    rstd_y = 1.0 / tl.sqrt(tl.maximum(var_y, 0.0) + EPS)

    tow_w = tl.load(TOW_W_ptr + offs_h, mask=h_mask, other=0.0)
    tow_b = tl.load(TOW_B_ptr + offs_h, mask=h_mask, other=0.0)

    g_off = offs_m[:, None] * H + offs_h[None, :]
    g = tl.load(
        OG_ptr + g_off,
        mask=m_mask[:, None] & h_mask[None, :],
        other=0.0,
    ).to(tl.float32)

    normed = (
        (y - mean_y[:, None]) * rstd_y[:, None] * tow_w[None, :] + tow_b[None, :]
    )
    scaled = normed * g
    y_bf = scaled.to(tl.bfloat16)

    for c0 in range(0, C, BLOCK_OUT):
        offs_c = c0 + tl.arange(0, BLOCK_OUT)
        c_mask = offs_c < C
        store_mask = m_mask[:, None] & c_mask[None, :]

        w_off = offs_h[:, None] * C + offs_c[None, :]
        w = tl.load(
            W_ptr + w_off,
            mask=h_mask[:, None] & c_mask[None, :],
            other=0.0,
        )

        acc = tl.dot(y_bf, w, tl.zeros((BLOCK_M, BLOCK_OUT), dtype=tl.float32))
        tl.store(
            OUT_ptr + offs_m[:, None] * C + offs_c[None, :],
            acc.to(tl.float32),
            mask=store_mask,
        )


@triton.jit
def _trimul_final_kernel_large_h(
    Y_ptr,
    OG_ptr,
    W_ptr,
    TOW_W_ptr,
    TOW_B_ptr,
    OUT_ptr,
    M,
    N2,
    N2H,
    C,
    H,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_OUT: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    m_mask = offs_m < M

    b_idx = offs_m // N2
    local = offs_m - b_idx * N2
    row_base = b_idx * N2H + local

    # First pass: LayerNorm statistics over H.
    sum_y = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum_y2 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    for h0 in range(0, H, BLOCK_K):
        offs_h = h0 + tl.arange(0, BLOCK_K)
        h_mask = offs_h < H
        y_off = row_base[None, :] + offs_h[:, None] * N2
        y_h = tl.load(
            Y_ptr + y_off,
            mask=m_mask[None, :] & h_mask[:, None],
            other=0.0,
        ).to(tl.float32)
        y = tl.trans(y_h)
        sum_y += tl.sum(y, axis=1)
        sum_y2 += tl.sum(y * y, axis=1)

    mean_y = sum_y / H
    var_y = sum_y2 / H - mean_y * mean_y
    rstd_y = 1.0 / tl.sqrt(tl.maximum(var_y, 0.0) + EPS)

    for c0 in range(0, C, BLOCK_OUT):
        offs_c = c0 + tl.arange(0, BLOCK_OUT)
        c_mask = offs_c < C
        store_mask = m_mask[:, None] & c_mask[None, :]

        acc = tl.zeros((BLOCK_M, BLOCK_OUT), dtype=tl.float32)

        # Second pass: apply norm + gate and accumulate final linear.
        for h0 in range(0, H, BLOCK_K):
            offs_h = h0 + tl.arange(0, BLOCK_K)
            h_mask = offs_h < H

            y_off = row_base[None, :] + offs_h[:, None] * N2
            y_h = tl.load(
                Y_ptr + y_off,
                mask=m_mask[None, :] & h_mask[:, None],
                other=0.0,
            ).to(tl.float32)
            y = tl.trans(y_h)

            tow_w = tl.load(TOW_W_ptr + offs_h, mask=h_mask, other=0.0)
            tow_b = tl.load(TOW_B_ptr + offs_h, mask=h_mask, other=0.0)
            normed = (
                (y - mean_y[:, None]) * rstd_y[:, None] * tow_w[None, :] + tow_b[None, :]
            )

            g_off = offs_m[:, None] * H + offs_h[None, :]
            g = tl.load(
                OG_ptr + g_off,
                mask=m_mask[:, None] & h_mask[None, :],
                other=0.0,
            ).to(tl.float32)

            scaled = normed * g
            y_bf = scaled.to(tl.bfloat16)

            w_off = offs_h[:, None] * C + offs_c[None, :]
            w = tl.load(
                W_ptr + w_off,
                mask=h_mask[:, None] & c_mask[None, :],
                other=0.0,
            )
            acc = tl.dot(y_bf, w, acc)

        tl.store(
            OUT_ptr + offs_m[:, None] * C + offs_c[None, :],
            acc.to(tl.float32),
            mask=store_mask,
        )


def custom_kernel(data):
    if len(data) == 4:
        x, mask, weights, config = data
    else:
        x, weights, config = data
        mask = None

    if not x.is_contiguous():
        x = x.contiguous()
    if x.dtype != torch.float32:
        x = x.to(torch.float32)

    B, N, N2a, C = x.shape
    N2 = N2a
    H = int(weights["left_proj.weight"].shape[0])
    M = B * N * N
    N2H = H * N2

    use_mask = mask is not None and not bool(config.get("nomask", False))
    if use_mask:
        if mask.dim() == 4:
            mask = mask.squeeze(-1)
        if mask.dtype != torch.float32:
            mask = mask.to(torch.float32)
        if not mask.is_contiguous():
            mask = mask.contiguous()

    bf16 = torch.bfloat16

    # Convert small weights once.
    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    def _linear_weight(name):
        return weights[name].float().t().contiguous().to(bf16)

    wl = _linear_weight("left_proj.weight")
    wr = _linear_weight("right_proj.weight")
    wlg = _linear_weight("left_gate.weight")
    wrg = _linear_weight("right_gate.weight")
    wog = _linear_weight("out_gate.weight")

    wout = weights["to_out.weight"].float().t().contiguous().to(bf16)
    tow_w = weights["to_out_norm.weight"].float().contiguous()
    tow_b = weights["to_out_norm.bias"].float().contiguous()

    # Intermediate activations.
    L = torch.empty((B, H, N, N), dtype=bf16, device=x.device)
    R = torch.empty((B, H, N, N), dtype=bf16, device=x.device)
    out_gate = torch.empty((M, H), dtype=bf16, device=x.device)

    # Fused LN + projections + gates.
    BLOCK_M = 32
    BLOCK_C = 128 if C > 128 else max(16, triton.next_power_of_2(C))
    BLOCK_H = 128 if H > 128 else max(16, triton.next_power_of_2(H))
    grid = (triton.cdiv(M, BLOCK_M),)

    _trimul_proj_kernel[grid](
        x,
        mask if use_mask else x,
        L,
        R,
        out_gate,
        norm_w,
        norm_b,
        wl,
        wr,
        wlg,
        wrg,
        wog,
        M,
        N2,
        N2H,
        C,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_C=BLOCK_C,
        BLOCK_H=BLOCK_H,
        HAS_MASK=use_mask,
        EPS=1e-5,
        num_warps=8,
        num_stages=2,
    )

    # Batched contraction over sequence:
    # Y[b,h,i,j] = sum_k L[b,h,i,k] * R[b,h,j,k]
    L_flat = L.view(B * H, N, N)
    R_flat = R.view(B * H, N, N)
    Y = torch.empty_like(L_flat)
    if hasattr(torch.backends.cuda.matmul, "allow_bf16_reduced_precision_reduction"):
        torch.backends.cuda.matmul.allow_bf16_reduced_precision_reduction = True
    torch.bmm(L_flat, R_flat.transpose(1, 2), out=Y)

    # Final fused LN + out_gate + linear projection.
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=x.device)
    BLOCK_OUT = 128 if C > 128 else max(16, triton.next_power_of_2(C))

    if H <= 128:
        BLOCK_HF = max(16, triton.next_power_of_2(H))
        BLOCK_MF = 64
        grid = (triton.cdiv(M, BLOCK_MF),)
        _trimul_final_kernel_small_h[grid](
            Y,
            out_gate,
            wout,
            tow_w,
            tow_b,
            out,
            M,
            N2,
            N2H,
            C,
            H,
            BLOCK_M=BLOCK_MF,
            BLOCK_H=BLOCK_HF,
            BLOCK_OUT=BLOCK_OUT,
            EPS=1e-5,
            num_warps=8,
            num_stages=2,
        )
    else:
        BLOCK_MF = 32
        BLOCK_K = 128
        grid = (triton.cdiv(M, BLOCK_MF),)
        _trimul_final_kernel_large_h[grid](
            Y,
            out_gate,
            wout,
            tow_w,
            tow_b,
            out,
            M,
            N2,
            N2H,
            C,
            H,
            BLOCK_M=BLOCK_MF,
            BLOCK_K=BLOCK_K,
            BLOCK_OUT=BLOCK_OUT,
            EPS=1e-5,
            num_warps=4,
            num_stages=2,
        )

    return out
```
