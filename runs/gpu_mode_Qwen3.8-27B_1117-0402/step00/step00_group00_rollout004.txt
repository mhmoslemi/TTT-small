def einsum_kernel(L, R, O, N, H,
                  stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                  stride_r_b, stride_r_j, stride_r_k, stride_r_h,
                  stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                  BM: tl.constexpr, BN: tl.constexpr, BK: tl.constexpr,
                  OUT_BF16: tl.constexpr):
  pid_i = tl.program_id(0); pid_j = tl.program_id(1); pid_h = tl.program_id(2)
  b = pid_h // H
  d = pid_h % H
  offs_i = pid_i*BM + tl.arange(0,BM)
  offs_j = pid_j*BN + tl.arange(0,BN)
  offs_k = tl.arange(0,BK)
  l_base = L + b*stride_l_b + d*stride_l_h + offs_i[:,None]*stride_l_i
  r_base = R + b*stride_r_b + d*stride_r_h + offs_j[None,:]*stride_r_j
  # A [BM,BK]: l_base + offs_k[None,:]*stride_l_k
  # B [BK,BN]: r_base + offs_k[:,None]*stride_r_k
  acc = tl.zeros((BM,BN), dtype=tl.float32)
  mask_i = offs_i < N; mask_j = offs_j < N
  for k0 in range(0, N, BK):
      l = tl.load(l_base + offs_k[None,:]*stride_l_k, mask=mask_i[:,None] & ((k0+offs_k)<N)[None,:], other=0.0)
      r = tl.load(r_base + offs_k[:,None]*stride_r_k, mask=((k0+offs_k)<N)[:,None] & mask_j[None,:], other=0)
      acc = tl.dot(l, r, acc)
      l_base += BK*stride_l_k; r_base += BK*stride_r_k

  if OUT_BF16: tl.store(out_ptrs, acc.to(tl.bfloat16), mask=mask)
  else: tl.store(out_ptrs, acc, mask=mask)

```python
"""
Fused/optimized forward TriMul.

Pipeline:
1) Triton LayerNorm to bf16 activations.
2) One cuBLAS bf16 GEMM for all 5 projections
   (left_proj, left_gate, right_proj, right_gate, out_gate).
3) Triton kernel splits/gates/masks the bundle and transposes left/right
   into h-major layout for the sequence contraction.
4) Triton batched-GEMM kernel computes:
       out[h, b, i, j] = sum_k left[h,b,i,k] * right[h,b,j,k]
5) Triton final kernel fuses LayerNorm, out_gate sigmoid, and the
   to_out projection to produce fp32 output.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _layer_norm_kernel(
    x_ptr,
    w_ptr,
    b_ptr,
    y_ptr,
    M,
    C,
    eps,
    BM: tl.constexpr,
    BC: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BM + tl.arange(0, BM)
    rmask = rows < M
    rows64 = rows.to(tl.int64)

    c = tl.arange(0, BC)
    cmask = c < C
    load_mask = rmask[:, None] & cmask[None, :]

    x = tl.load(x_ptr + rows64[:, None] * C + c[None, :], mask=load_mask, other=0.0).to(tl.float32)

    mean = tl.sum(x, axis=1) / C
    xc = x - mean[:, None]
    xc = tl.where(cmask[None, :], xc, 0.0)
    var = tl.sum(xc * xc, axis=1) / C
    rstd = 1.0 / tl.sqrt(var + eps)

    w = tl.load(w_ptr + c, mask=cmask, other=0.0).to(tl.float32)
    b = tl.load(b_ptr + c, mask=cmask, other=0.0).to(tl.float32)

    y = xc * rstd[:, None] * w[None, :] + b[None, :]
    tl.store(y_ptr + rows64[:, None] * C + c[None, :], y.to(tl.bfloat16), mask=load_mask)


@triton.jit
def _split_gate_kernel(
    bundle_ptr,
    mask_ptr,
    left_ptr,
    right_ptr,
    M,
    H,
    S,
    HAS_MASK: tl.constexpr,
    BM: tl.constexpr,
    HB: tl.constexpr,
):
    pid = tl.program_id(0)
    rows = pid * BM + tl.arange(0, BM)
    rmask = rows < M
    rows64 = rows.to(tl.int64)

    h = tl.arange(0, HB)
    hmask = h < H
    load_mask = rmask[:, None] & hmask[None, :]

    base = rows64[:, None] * S

    # Bundle column order:
    # [left_proj, left_gate, right_proj, right_gate, out_gate]
    lp = tl.load(bundle_ptr + base + h[None, :], mask=load_mask, other=0.0).to(tl.float32)
    lg = tl.load(bundle_ptr + base + H + h[None, :], mask=load_mask, other=0.0).to(tl.float32)
    rp = tl.load(bundle_ptr + base + 2 * H + h[None, :], mask=load_mask, other=0.0).to(tl.float32)
    rg = tl.load(bundle_ptr + base + 3 * H + h[None, :], mask=load_mask, other=0.0).to(tl.float32)

    if HAS_MASK:
        m = tl.load(mask_ptr + rows, mask=rmask, other=0.0).to(tl.float32)[:, None]
    else:
        m = 1.0

    left = lp * m * tl.sigmoid(lg)
    right = rp * m * tl.sigmoid(rg)

    # Write transposed / h-major: [H, M]
    dst = h.to(tl.int64)[None, :] * M + rows[:, None]
    tl.store(left_ptr + dst, left.to(tl.bfloat16), mask=load_mask)
    tl.store(right_ptr + dst, right.to(tl.bfloat16), mask=load_mask)


@triton.jit
def _einsum_kernel(
    l_ptr,
    r_ptr,
    o_ptr,
    B,
    N,
    M,
    N2,
    BM: tl.constexpr,
    BN: tl.constexpr,
    BK: tl.constexpr,
    EVEN: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Layout is [H, B, N, N], but passed flat.
    h = pid_bh // B
    b = pid_bh % B
    h64 = h.to(tl.int64)
    b64 = b.to(tl.int64)
    base = h64 * M + b64 * N2

    off_i = pid_i * BM + tl.arange(0, BM)
    off_j = pid_j * BN + tl.arange(0, BN)
    off_k = tl.arange(0, BK)

    l_ptrs = l_ptr + base + off_i[:, None] * N + off_k[None, :]
    r_ptrs = r_ptr + base + off_j[:, None] * N + off_k[None, :]

    acc = tl.zeros((BM, BN), dtype=tl.float32)

    if EVEN:
        for _ in range(0, N, BK):
            l = tl.load(l_ptrs)
            r = tl.load(r_ptrs)
            acc = tl.dot(l, tl.trans(r), acc)
            l_ptrs += BK
            r_ptrs += BK
    else:
        mask_i = off_i < N
        mask_j = off_j < N
        for k0 in range(0, N, BK):
            mask_k = (k0 + off_k) < N
            l = tl.load(l_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            r = tl.load(r_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
            acc = tl.dot(l, tl.trans(r), acc)
            l_ptrs += BK
            r_ptrs += BK

    o_ptrs = o_ptr + base + off_i[:, None] * N + off_j[None, :]
    if EVEN:
        tl.store(o_ptrs, acc)
    else:
        store_mask = (off_i < N)[:, None] & (off_j < N)[None, :]
        tl.store(o_ptrs, acc, mask=store_mask)


@triton.jit
def _final_kernel(
    out_ptr,
    og_ptr,
    og_off,
    og_stride,
    norm_w_ptr,
    norm_b_ptr,
    wout_ptr,
    res_ptr,
    M,
    C,
    H,
    eps,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_c = tl.program_id(1)

    rows = pid_m * BM + tl.arange(0, BM)
    rmask = rows < M
    rows64 = rows.to(tl.int64)

    c = pid_c * BC + tl.arange(0, BC)
    cmask = c < C

    h = tl.arange(0, BH)
    hmask = h < H

    # out is h-major: [H, M]
    out_mask = hmask[:, None] & rmask[None, :]
    o = tl.load(out_ptr + h.to(tl.int64)[:, None] * M + rows[None, :], mask=out_mask, other=0.0).to(tl.float32)

    # og is still the out_gate slice of the original row-major bundle:
    # row stride is S, column offset is og_off.
    og_mask = rmask[:, None] & hmask[None, :]
    og = tl.load(
        og_ptr + og_off + rows64[:, None] * og_stride + h[None, :],
        mask=og_mask,
        other=0.0,
    ).to(tl.float32)

    mean = tl.sum(o, axis=0) / H
    centered = o - mean[None, :]
    centered = tl.where(hmask[:, None], centered, 0.0)
    var = tl.sum(centered * centered, axis=0) / H
    rstd = 1.0 / tl.sqrt(var + eps)

    gamma = tl.load(norm_w_ptr + h, mask=hmask, other=0.0).to(tl.float32)
    beta = tl.load(norm_b_ptr + h, mask=hmask, other=0.0).to(tl.float32)

    # a has shape [BH, BM], transpose to [BM, BH] for the GEMM.
    a_hm = centered * (rstd[None, :] * gamma[:, None]) + beta[:, None]
    a_tm = tl.trans(a_hm)
    a_tm = a_tm * tl.sigmoid(og)
    a_tm = tl.where(hmask[None, :], a_tm, 0.0)
    ab = a_tm.to(tl.bfloat16)

    # to_out.weight is [C, H]; form B = [ BH, BC ].
    w = tl.load(wout_ptr + h[:, None] + c[None, :] * H, mask=hmask[:, None] & cmask[None, :], other=0.0)

    acc = tl.dot(ab, w)

    store_mask = rmask[:, None] & cmask[None, :]
    tl.store(res_ptr + rows64[:, None] * C + c[None, :], acc, mask=store_mask)


def custom_kernel(data):
    # Robust unpacking: expected [x, mask, weights, config],
    # but also support [x, weights, config].
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    elif len(data) == 3:
        if isinstance(data[1], dict):
            input_tensor, weights, config = data
            mask = None
        else:
            # Fallback for unlikely [x, mask, config]
            input_tensor, mask, config = data
            weights = {}
    else:
        raise ValueError("custom_kernel expects 3 or 4 elements in data")

    if config is None:
        config = {}

    device = input_tensor.device
    B, N, _, C_in = input_tensor.shape

    H = config.get("hidden_dim", None)
    if H is None:
        H = weights["left_proj.weight"].shape[0]

    C_out = weights["to_out.weight"].shape[0]
    eps = float(config.get("eps", 1e-5))

    M = B * N * N
    N2 = N * N
    S = 5 * H

    # Input LayerNorm -> bf16 activations.
    x_flat = input_tensor.reshape(-1, C_in).contiguous()
    norm_w = weights["norm.weight"].float().contiguous()
    norm_b = weights["norm.bias"].float().contiguous()

    BC_LN = triton.next_power_of_2(C_in)
    if C_in <= 128:
        BM_LN = 128
    elif C_in <= 256:
        BM_LN = 64
    else:
        BM_LN = 32

    x_norm = torch.empty((M, C_in), dtype=torch.bfloat16, device=device)
    _layer_norm_kernel[(triton.cdiv(M, BM_LN),)](
        x_flat, norm_w, norm_b, x_norm,
        M, C_in, eps,
        BM=BM_LN,
        BC=BC_LN,
        num_warps=4,
    )

    # Add projections/gates into a single wide GEMM.
    def _wbf(name):
        return weights[name].to(torch.bfloat16).contiguous()

    w_all = torch.cat(
        [
            _wbf("left_proj.weight"),
            _wbf("left_gate.weight"),
            _wbf("right_proj.weight"),
            _wbf("right_gate.weight"),
            _wbf("out_gate.weight"),
        ],
        dim=0,
    )
    w_all_t = w_all.t().contiguous()

    bundle = torch.mm(x_norm, w_all_t)  # [M, 5H], bf16
    del x_norm

    # Prepare mask.
    if mask is None or (isinstance(mask, torch.Tensor) and mask.numel() == 0):
        has_mask = False
        mask_flat = None
    else:
        has_mask = True
        mask_flat = mask.to(torch.float32, copy=False).reshape(-1).contiguous()

    # Allocate left/right in h-major layout [H, M] for efficient einsum.
    left = torch.empty((H, M), dtype=torch.bfloat16, device=device)
    right = torch.empty((H, M), dtype=torch.bfloat16, device=device)

    HB = triton.next_power_of_2(H)
    if H <= 128:
        BM_SPLIT = 128
    elif H <= 256:
        BM_SPLIT = 64
    else:
        BM_SPLIT = 32

    _split_gate_kernel[(triton.cdiv(M, BM_SPLIT),)](
        bundle,
        mask_flat if has_mask else left,  # dummy pointer when unused
        left,
        right,
        M, H, S,
        HAS_MASK=has_mask,
        BM=BM_SPLIT,
        HB=HB,
        num_warps=4,
    )
    if has_mask:
        del mask_flat

    # out from the pairwise sequence contraction, stored h-major: [H, M], fp32.
    out = torch.empty((H, M), dtype=torch.float32, device=device)

    if N <= 128:
        BM_E = 64
        BN_E = 64
        BK_E = 64
        num_warps_e = 4
    else:
        BM_E = 128
        BN_E = 128
        BK_E = 64
        num_warps_e = 8

    grid_e = (triton.cdiv(N, BM_E), triton.cdiv(N, BN_E), H * B)
    even_e = (N % BM_E == 0) and (N % BN_E == 0) and (N % BK_E == 0)

    _einsum_kernel[grid_e](
        left, right, out,
        B, N, M, N2,
        BM=BM_E,
        BN=BN_E,
        BK=BK_E,
        EVEN=even_e,
        num_warps=num_warps_e,
        num_stages=3,
    )

    # Final fused LayerNorm + out_gate + to_out projection.
    final = torch.empty((B, N, N, C_out), dtype=torch.float32, device=device)
    to_out_norm_w = weights["to_out_norm.weight"].float().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].float().contiguous()
    to_out_w_bf = weights["to_out.weight"].to(torch.bfloat16).contiguous()

    BH = triton.next_power_of_2(H)
    if H <= 128:
        BM_F = 64
        BC_F = 256 if C_out > 128 else 128
    else:
        BM_F = 32
        BC_F = 128

    grid_f = (triton.cdiv(M, BM_F), triton.cdiv(C_out, BC_F))

    _final_kernel[grid_f](
        out,
        bundle,
        4 * H,        # og_off: out_gate slice starts at column 4H
        S,            # og_stride: row stride of the original bundle
        to_out_norm_w,
        to_out_norm_b,
        to_out_w_bf,
        final,
        M, C_out, H, eps,
        BM=BM_F,
        BH=BH,
        BC=BC_F,
        num_warps=8 if BM_F >= 64 else 4,
        num_stages=2,
    )

    return final
```
