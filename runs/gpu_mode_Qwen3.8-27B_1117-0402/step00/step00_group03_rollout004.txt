```python
import torch
import triton
import triton.language as tl


def _np2(n: int) -> int:
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _ln_proj_gate_kernel(
    x_ptr, mask_ptr,
    norm_w_ptr, norm_b_ptr,
    wl_ptr, wr_ptr, wgl_ptr, wgr_ptr, wog_ptr,
    left_ptr, right_ptr, gate_ptr,
    N2, C, H,
    BLOCK_C: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BM: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    Fused LayerNorm + five projections/gates.

    Outputs:
      left[b, h, i, k]   = sigmoid(xn @ W_left_gate^T) * mask[b, i, k] * (xn @ W_left^T)
      right[b, h, i, k]  = sigmoid(xn @ W_right_gate^T) * mask[b, i, k] * (xn @ W_right^T)
      gate[b, h, i, k]   = sigmoid(xn @ W_out_gate^T)
    """
    pid_m = tl.program_id(0)
    b = tl.program_id(1)

    offs_m = pid_m * BM + tl.arange(0, BM)
    m_mask = offs_m < N2
    m_off = tl.where(m_mask, offs_m, 0)

    offs_c = tl.arange(0, BLOCK_C)
    c_mask = offs_c < C
    c_off = tl.where(c_mask, offs_c, 0)

    x_offs = b * N2 * C + m_off[:, None] * C + c_off[None, :]
    x = tl.load(x_ptr + x_offs, mask=(m_mask[:, None] & c_mask[None, :]), other=0.0).to(tl.float32)
    x = tl.where(c_mask[None, :], x, 0.0)

    mean = tl.sum(x, axis=1) / C
    diff = x - mean[:, None]
    diff = tl.where(c_mask[None, :], diff, 0.0)
    var = tl.sum(diff * diff, axis=1) / C
    inv = 1.0 / tl.sqrt(var + 1e-5)

    norm_w = tl.load(norm_w_ptr + c_off, mask=c_mask, other=0.0).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + c_off, mask=c_mask, other=0.0).to(tl.float32)

    xn_f32 = (x - mean[:, None]) * inv[:, None] * norm_w[None, :] + norm_b[None, :]
    xn = tl.where(c_mask[None, :], xn_f32, 0.0).to(tl.bfloat16)

    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + b * N2 + m_off, mask=m_mask, other=0.0).to(tl.float32)
        mask_vals = tl.where(m_mask, mask_vals, 0.0)
    else:
        mask_vals = tl.where(m_mask, 1.0, 0.0)

    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        h_mask = offs_h < H
        h_off = tl.where(h_mask, offs_h, 0)

        wl = tl.load(
            wl_ptr + h_off[None, :] * C + c_off[:, None],
            mask=(c_mask[:, None] & h_mask[None, :]),
            other=0.0,
        ).to(tl.bfloat16)
        proj_l = tl.dot(xn, wl)

        wgl = tl.load(
            wgl_ptr + h_off[None, :] * C + c_off[:, None],
            mask=(c_mask[:, None] & h_mask[None, :]),
            other=0.0,
        ).to(tl.bfloat16)
        gate_l = 1.0 / (1.0 + tl.exp(-tl.dot(xn, wgl)))

        left_val = proj_l * gate_l * mask_vals[:, None]
        l_ptrs = left_ptr + (b * H + h_off)[:, None] * N2 + m_off[None, :]
        tl.store(
            l_ptrs,
            tl.trans(left_val.to(tl.bfloat16)),
            mask=(h_mask[:, None] & m_mask[None, :]),
        )

        wr = tl.load(
            wr_ptr + h_off[None, :] * C + c_off[:, None],
            mask=(c_mask[:, None] & h_mask[None, :]),
            other=0.0,
        ).to(tl.bfloat16)
        proj_r = tl.dot(xn, wr)

        wgr = tl.load(
            wgr_ptr + h_off[None, :] * C + c_off[:, None],
            mask=(c_mask[:, None] & h_mask[None, :]),
            other=0.0,
        ).to(tl.bfloat16)
        gate_r = 1.0 / (1.0 + tl.exp(-tl.dot(xn, wgr)))

        right_val = proj_r * gate_r * mask_vals[:, None]
        r_ptrs = right_ptr + (b * H + h_off)[:, None] * N2 + m_off[None, :]
        tl.store(
            r_ptrs,
            tl.trans(right_val.to(tl.bfloat16)),
            mask=(h_mask[:, None] & m_mask[None, :]),
        )

        wog = tl.load(
            wog_ptr + h_off[None, :] * C + c_off[:, None],
            mask=(c_mask[:, None] & h_mask[None, :]),
            other=0.0,
        ).to(tl.bfloat16)
        gate_o = 1.0 / (1.0 + tl.exp(-tl.dot(xn, wog)))

        g_ptrs = gate_ptr + (b * H + h_off)[:, None] * N2 + m_off[None, :]
        tl.store(
            g_ptrs,
            tl.trans(gate_o.to(tl.bfloat16)),
            mask=(h_mask[:, None] & m_mask[None, :]),
        )


@triton.jit
def _core_kernel(
    left_ptr, right_ptr, out_ptr,
    N, N2, H,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EVEN: tl.constexpr,
):
    """
    Core TriMul:
      out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k]

    left/right/out are h-major: [B, H, N, N].
    """
    pid = tl.program_id(0)

    grid_i = tl.cdiv(N, BLOCK_I)
    grid_j = tl.cdiv(N, BLOCK_J)

    pid_j = pid % grid_j
    pid_i = (pid // grid_j) % grid_i
    pid_bh = pid // (grid_j * grid_i)

    b = pid_bh // H
    h = pid_bh % H

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    i_valid = offs_i < N
    j_valid = offs_j < N

    base = (b * H + h) * N2
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)
    num_k = tl.cdiv(N, BLOCK_K)

    if EVEN:
        a_ptrs = left_ptr + base + offs_i[:, None] * N + offs_k[None, :]
        b_ptrs = right_ptr + base + offs_j[:, None] * N + offs_k[None, :]

        for _ in range(0, num_k):
            a = tl.load(a_ptrs)
            bmat = tl.load(b_ptrs)
            acc += tl.dot(a, tl.trans(bmat))
            a_ptrs += BLOCK_K
            b_ptrs += BLOCK_K
    else:
        i_off = tl.where(i_valid, offs_i, 0)
        j_off = tl.where(j_valid, offs_j, 0)

        for k in range(0, num_k):
            kstart = k * BLOCK_K
            k_offs = kstart + offs_k
            k_valid = k_offs < N
            k_off = tl.where(k_valid, k_offs, 0)

            a = tl.load(
                left_ptr + base + i_off[:, None] * N + k_off[None, :],
                mask=(i_valid[:, None] & k_valid[None, :]),
                other=0.0,
            )
            bmat = tl.load(
                right_ptr + base + j_off[:, None] * N + k_off[None, :],
                mask=(j_valid[:, None] & k_valid[None, :]),
                other=0.0,
            )
            acc += tl.dot(a, tl.trans(bmat))

    if EVEN:
        out_ptrs = out_ptr + base + offs_i[:, None] * N + offs_j[None, :]
        tl.store(out_ptrs, acc.to(tl.bfloat16))
    else:
        i_off = tl.where(i_valid, offs_i, 0)
        j_off = tl.where(j_valid, offs_j, 0)
        out_ptrs = out_ptr + base + i_off[:, None] * N + j_off[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.bfloat16),
            mask=(i_valid[:, None] & j_valid[None, :]),
        )


@triton.jit
def _final_ln_gate_out_kernel(
    o_ptr, gate_ptr, w_ptr, out_ptr,
    ln_w_ptr, ln_b_ptr,
    N2, C, H,
    BM: tl.constexpr,
    BH: tl.constexpr,
    BC: tl.constexpr,
    FULL_H: tl.constexpr,
):
    """
    Final fused operation:
      z = LayerNorm(out) + gate
      y = z @ W_out^T

    out and gate are h-major [B, H, N, N]; output is [B, N, N, C].
    """
    pid_m = tl.program_id(0)
    pid_c = tl.program_id(1)
    b = tl.program_id(2)

    offs_m = pid_m * BM + tl.arange(0, BM)
    m_mask = offs_m < N2
    m_off = tl.where(m_mask, offs_m, 0)

    offs_c = pid_c * BC + tl.arange(0, BC)
    c_mask = offs_c < C
    c_off = tl.where(c_mask, offs_c, 0)

    base = b * H * N2

    if FULL_H:
        offs_h = tl.arange(0, BH)
        h_mask = offs_h < H
        h_off = tl.where(h_mask, offs_h, 0)

        og_mask = h_mask[:, None] & m_mask[None, :]
        o_ptrs = o_ptr + base + h_off[:, None] * N2 + m_off[None, :]
        g_ptrs = gate_ptr + base + h_off[:, None] * N2 + m_off[None, :]

        o = tl.load(o_ptrs, mask=og_mask, other=0.0).to(tl.float32)
        g = tl.load(g_ptrs, mask=og_mask, other=0.0).to(tl.float32)

        sum_o = tl.sum(o, axis=0)
        mean = sum_o / H

        diff = o - mean[None, :]
        diff = tl.where(h_mask[:, None], diff, 0.0)
        var = tl.sum(diff * diff, axis=0) / H
        inv = 1.0 / tl.sqrt(var + 1e-5)

        ln_w = tl.load(ln_w_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
        ln_b = tl.load(ln_b_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)

        p = (o - mean[None, :]) * inv[None, :] * ln_w[:, None] + ln_b[:, None]
        p = p * g
        p = tl.where(h_mask[:, None], p, 0.0)

        w = tl.load(
            w_ptr + h_off[:, None] * C + c_off[None, :],
            mask=(h_mask[:, None] & c_mask[None, :]),
            other=0.0,
        ).to(tl.bfloat16)

        acc = tl.dot(tl.trans(p.to(tl.bfloat16)), w)

        out_ptrs = out_ptr + (b * N2 + m_off)[:, None] * C + c_off[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.float32),
            mask=(m_mask[:, None] & c_mask[None, :]),
        )
    else:
        sum1 = tl.zeros((BM,), dtype=tl.float32)
        sum2 = tl.zeros((BM,), dtype=tl.float32)

        for h_start in range(0, H, BH):
            offs_h = h_start + tl.arange(0, BH)
            h_mask = offs_h < H
            h_off = tl.where(h_mask, offs_h, 0)

            mask_hm = h_mask[:, None] & m_mask[None, :]
            o_ptrs = o_ptr + base + h_off[:, None] * N2 + m_off[None, :]
            o = tl.load(o_ptrs, mask=mask_hm, other=0.0).to(tl.float32)

            sum1 += tl.sum(o, axis=0)
            sum2 += tl.sum(o * o, axis=0)

        mean = sum1 / H
        var = sum2 / H - mean * mean
        var = tl.where(var < 0.0, 0.0, var)
        inv = 1.0 / tl.sqrt(var + 1e-5)

        acc = tl.zeros((BM, BC), dtype=tl.float32)

        for h_start in range(0, H, BH):
            offs_h = h_start + tl.arange(0, BH)
            h_mask = offs_h < H
            h_off = tl.where(h_mask, offs_h, 0)

            mask_hm = h_mask[:, None] & m_mask[None, :]
            o_ptrs = o_ptr + base + h_off[:, None] * N2 + m_off[None, :]
            g_ptrs = gate_ptr + base + h_off[:, None] * N2 + m_off[None, :]

            o = tl.load(o_ptrs, mask=mask_hm, other=0.0).to(tl.float32)
            g = tl.load(g_ptrs, mask=mask_hm, other=0.0).to(tl.float32)

            ln_w = tl.load(ln_w_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)
            ln_b = tl.load(ln_b_ptr + h_off, mask=h_mask, other=0.0).to(tl.float32)

            p = (o - mean[None, :]) * inv[None, :] * ln_w[:, None] + ln_b[None, :]
            p = p * g
            p = tl.where(h_mask[:, None], p, 0.0)

            w = tl.load(
                w_ptr + h_off[:, None] * C + c_off[None, :],
                mask=(h_mask[:, None] & c_mask[None, :]),
                other=0.0,
            ).to(tl.bfloat16)

            acc += tl.dot(tl.trans(p.to(tl.bfloat16)), w)

        out_ptrs = out_ptr + (b * N2 + m_off)[:, None] * C + c_off[None, :]
        tl.store(
            out_ptrs,
            acc.to(tl.float32),
            mask=(m_mask[:, None] & c_mask[None, :]),
        )


def custom_kernel(data):
    """
    Optimized forward pass for outgoing TriMul.

    Pipeline:
      1) Fused LayerNorm + five projections/gates into h-major bf16 tensors.
      2) Batched h-centred Triton matmul for the O(N^3) TriMul core.
      3) Fused LayerNorm + out-gate + output projection to fp32.
    """
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    x = x.contiguous()
    B, N, _, C = x.shape
    N2 = N * N
    H = int(weights["left_proj.weight"].shape[0])
    device = x.device

    def _f32(name):
        return weights[name].to(device=device, dtype=torch.float32).contiguous()

    def _bf16(name):
        return weights[name].to(device=device, dtype=torch.bfloat16).contiguous()

    norm_w = _f32("norm.weight")
    norm_b = _f32("norm.bias")

    wl = _bf16("left_proj.weight")
    wr = _bf16("right_proj.weight")
    wgl = _bf16("left_gate.weight")
    wgr = _bf16("right_gate.weight")
    wog = _bf16("out_gate.weight")

    to_out_w = _bf16("to_out.weight")
    to_out_ln_w = _f32("to_out_norm.weight")
    to_out_ln_b = _f32("to_out_norm.bias")

    if mask is None:
        has_mask = False
        mask_arg = x
    else:
        mask = mask.contiguous()
        has_mask = True
        mask_arg = mask.to(torch.float32, non_blocking=True)

    # ---------------- Stage 1: LayerNorm + projections/gates ----------------
    BLOCK_C = max(16, _np2(C))
    if BLOCK_C <= 256:
        PROJ_BM = 128
    elif BLOCK_C <= 512:
        PROJ_BM = 64
    else:
        PROJ_BM = 32
    PROJ_BM = max(16, PROJ_BM)

    PROJ_BH = max(16, min(_np2(H), 128))

    grid_proj = (triton.cdiv(N2, PROJ_BM), B)
    left = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    right = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    gate = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

    _ln_proj_gate_kernel[grid_proj](
        x, mask_arg,
        norm_w, norm_b,
        wl, wr, wgl, wgr, wog,
        left, right, gate,
        N2, C, H,
        BLOCK_C=BLOCK_C,
        BLOCK_H=PROJ_BH,
        BM=PROJ_BM,
        HAS_MASK=has_mask,
        num_warps=8,
        num_stages=2,
    )

    # ---------------- Stage 2: Core N^3 batched contraction ----------------
    if N >= 128:
        core_bi = 128
        core_bj = 128
        core_bk = 64
    else:
        pn = max(16, _np2(N))
        core_bi = min(128, pn)
        core_bj = core_bi
        core_bk = min(64, core_bi)
        if core_bk < 16:
            core_bk = core_bi

    core_even = (N % core_bi == 0) and (N % core_bj == 0) and (N % core_bk == 0)
    grid_i = triton.cdiv(N, core_bi)
    grid_j = triton.cdiv(N, core_bj)
    grid_core = (B * H * grid_i * grid_j,)

    out_mid = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)

    _core_kernel[grid_core](
        left, right, out_mid,
        N, N2, H,
        BLOCK_I=core_bi,
        BLOCK_J=core_bj,
        BLOCK_K=core_bk,
        EVEN=core_even,
        num_warps=8,
        num_stages=4,
    )

    del left, right

    # ---------------- Stage 3: Final LayerNorm + gate + projection ----------------
    FINAL_BH = max(16, min(_np2(H), 128))
    if C <= 256:
        FINAL_BC = max(16, _np2(C))
    else:
        FINAL_BC = 256

    if FINAL_BC <= 128:
        FINAL_BM = 128
    else:
        FINAL_BM = 64

    FINAL_BM = max(16, FINAL_BM)
    full_h = H <= FINAL_BH

    num_m = triton.cdiv(N2, FINAL_BM)
    num_c = triton.cdiv(C, FINAL_BC)
    grid_final = (num_m, num_c, B)

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    _final_ln_gate_out_kernel[grid_final](
        out_mid, gate, to_out_w, out,
        to_out_ln_w, to_out_ln_b,
        N2, C, H,
        BM=FINAL_BM,
        BH=FINAL_BH,
        BC=FINAL_BC,
        FULL_H=full_h,
        num_warps=8,
        num_stages=2 if full_h else 3,
    )

    return out
```
