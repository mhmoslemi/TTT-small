```
@triton.jit
def _tri_mul_front_kernel(
    x_ptr, mask_ptr,
    norm_w_ptr, norm_b_ptr,
    wl_comb_ptr, wr_comb_ptr, wo_ptr,
    l_ptr, r_ptr, og_ptr,
    M, C, H, H2, N2, H_N2,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_2N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = offs_m < M
    row_b = (offs_m // N2).to(tl.int64)
    row_t = (offs_m - row_b * N2).to(tl.int64)

    offs_k = tl.arange(0, BLOCK_K)
    num_k = tl.cdiv(C, BLOCK_K)
    sum1 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum2 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    for k0 in range(0, num_k):
       k = k0 * BLOCK_K + offs_k
       km = k < C
       xmat = tl.load(x_ptr + offs_m[:, None] * C + k[None, :], mask=row_mask[:, None] & km[None, :], other=0.0).to(tl.float32)
       sum1 += tl.sum(xmat, axis=1)
       sum2 += tl.sum(xmat * xmat, axis=1)
    mean = sum1 / C
    var = sum2 / C - mean * mean
    var = tl.maximum(var, 0.0)

import torch
import triton
import triton.language as tl

"""
Fused Triton implementation of AlphaFold3 TriMul outgoing...
"""

@triton.jit
def _tri_mul_front_kernel(...):
   ...

@triton.jit
def _tri_mul_back_kernel(...):
   ...

def _pow2(n):
    return 1 << (n - 1).bit_length() if n > 1 else 1

def custom_kernel(data):
    x, mask, weights, config = data
    B, N, _, C = x.shape
    H = int(config.get('hidden_dim', config.get('c_z', 128)))
    M = B*N*N
    N2 = N*N
    H_N2 = H*N2
    eps = float(config.get('eps', 1e-5))
    dev = x.device
    if x.dtype != torch.float32:
       x = x.to(torch.float32)
    if not x.is_contiguous(): x = x.contiguous()
    has_mask = mask is not None
    if has_mask:
       if mask.dim() == 4: mask = mask.squeeze(-1)
       if mask.dtype != torch.float32: mask_f = mask.to(torch.float32)
       else: mask_f = mask
       if not mask_f.is_contiguous(): mask_f = mask_f.contiguous()
       mask_flat = mask_f.reshape(M)
    else:
       mask_flat = x

    # weights
    def wt(key): return weights[key].T.contiguous().to(torch.bfloat16)
    W_L = wt('left_proj.weight'); W_LG = wt('left_gate.weight')
    W_R = wt('right_proj.weight'); W_RG = wt('right_gate.weight')
    W_OG = wt('out_gate.weight')
    W_OUT = weights['to_out.weight'].T.contiguous().to(torch.bfloat16)
    norm_w = weights['norm.weight'].to(torch.float32).contiguous()
    norm_b = weights['norm.bias'].to(torch.float32).contiguous()
    out_norm_w = weights['to_out_norm.weight'].to(torch.float32).contiguous()
    out_norm_b = weights['to_out_norm.bias'].to(torch.float32).contiguous()

    L = torch.empty((B,H,N,N), dtype=torch.bfloat16, device=dev)
    R = torch.empty((B,H,N,N), dtype=torch.bfloat16, device=dev)
    OG = torch.empty((M,H), dtype=torch.bfloat16, device=dev)

    BLOCK_M = 64
    if H > 256: BLOCK_M=32
    BLOCK_K = 128 if C >= 128 else max(16, _pow2(C))
    BLOCK_N = 128 if H >= 128 else max(16, _pow2(H))
    grid = (triton.cdiv(M, BLOCK_M),)
    _tri_mul_front_kernel[grid](
       x, mask_flat, norm_w, norm_b,
       W_L, W_LG, W_R, W_RG, W_OG,
       L, R, OG,
       M, C, H, N2, H_N2, eps,
       BLOCK_M=BLOCK_M, BLOCK_K=BLOCK_K, BLOCK_N=BLOCK_N,
       HAS_MASK=has_mask,
       num_warps=4, num_stages=2,
    )

    BH = B*H
    out_h = torch.bmm(L.view(BH,N,N), R.view(BH,N,N).transpose(1,2))
    del L, R

    out = torch.empty((B,N,N,C), dtype=torch.float32, device=dev)
    BLOCK_MF = 64
    BLOCK_KF = 128 if H >=128 else max(16, _pow2(H))
    BLOCK_CF = 128 if C >=128 else max(16, _pow2(C))
    gridf = (triton.cdiv(M, BLOCK_MF),)
    _tri_mul_back_kernel[gridf](
       out_h, OG, out_norm_w, out_norm_b, W_OUT, out,
       M, H, C, N2, H_N2, eps,
       BLOCK_M=BLOCK_MF, BLOCK_K=BLOCK_KF, BLOCK_C=BLOCK_CF,
       num_warps=4, num_stages=2,
    )
    return out

```python
import torch
import triton
import triton.language as tl


"""
Triton implementation of the AlphaFold3 "outgoing" TriMul operator.

The implementation fuses:
  1. Input LayerNorm + left/right/out projection and gating.
     - Left and right are stored in a memory layout that is efficient for the
       core batched matmul: [B, hidden_dim, N, N].
     - Out_gate is stored as [B, N, N, hidden_dim] for efficient final use.
  2. The heavy core contraction:
         out[..., d] = sum_k left[..., k, d] * right[..., k, d]
     is performed by a bf16 batched GEMM over (B, hidden_dim) using cuBLAS.
  3. Output LayerNorm + out gate + final projection, fused into one kernel and
     written directly as float32 [B, N, N, dim].
"""


def _next_pow2(n: int) -> int:
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


@triton.jit
def _tri_mul_front_kernel(
    x_ptr,
    mask_ptr,
    norm_w_ptr,
    norm_b_ptr,
    wl_ptr,
    wlg_ptr,
    wr_ptr,
    wrg_ptr,
    wo_ptr,
    l_ptr,
    r_ptr,
    og_ptr,
    M,
    C,
    H,
    N2,
    H_N2,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = offs_m < M

    # For the transposed [B, H, N, N] output layout:
    # global token t = b * N2 + local token lt.
    row_b = (offs_m // N2).to(tl.int64)
    row_t = (offs_m - row_b * N2).to(tl.int64)

    # ---- LayerNorm statistics over the last input dimension C ----
    offs_k = tl.arange(0, BLOCK_K)
    num_k = tl.cdiv(C, BLOCK_K)

    sum1 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum2 = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k0 in range(0, num_k):
        k = k0 * BLOCK_K + offs_k
        km = k < C
        xmat = tl.load(
            x_ptr + offs_m[:, None] * C + k[None, :],
            mask=row_mask[:, None] & km[None, :],
            other=0.0,
        ).to(tl.float32)
        sum1 += tl.sum(xmat, axis=1)
        sum2 += tl.sum(xmat * xmat, axis=1)

    mean = sum1 / C
    var = sum2 / C - mean * mean
    var = tl.where(var > 0.0, var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    if HAS_MASK:
        mask_vals = tl.load(mask_ptr + offs_m, mask=row_mask, other=0.0).to(tl.float32)
    else:
        mask_vals = tl.zeros((BLOCK_M,), dtype=tl.float32) + 1.0

    offs_n = tl.arange(0, BLOCK_N)
    num_n = tl.cdiv(H, BLOCK_N)

    # ---- left projection + left gate + mask ----
    for n0 in range(0, num_n):
        n = n0 * BLOCK_N + offs_n
        hmask = n < H

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        accg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k0 in range(0, num_k):
            k = k0 * BLOCK_K + offs_k
            km = k < C

            xmat = tl.load(
                x_ptr + offs_m[:, None] * C + k[None, :],
                mask=row_mask[:, None] & km[None, :],
                other=0.0,
            ).to(tl.float32)

            nw = tl.load(norm_w_ptr + k, mask=km, other=0.0).to(tl.float32)
            nb = tl.load(norm_b_ptr + k, mask=km, other=0.0).to(tl.float32)

            xn = (xmat - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            xd = xn.to(tl.bfloat16)

            w = tl.load(
                wl_ptr + k[:, None] * H + n[None, :],
                mask=km[:, None] & hmask[None, :],
                other=0.0,
            )
            wg = tl.load(
                wlg_ptr + k[:, None] * H + n[None, :],
                mask=km[:, None] & hmask[None, :],
                other=0.0,
            )

            acc = tl.dot(xd, w, acc)
            accg = tl.dot(xd, wg, accg)

        gate = 1.0 / (1.0 + tl.exp(-accg))
        val = acc * gate * mask_vals[:, None]

        # Store transposed: [feature, token] to keep token dimension contiguous.
        val_t = tl.trans(val.to(tl.bfloat16))
        l_ptrs = (
            l_ptr
            + n[:, None].to(tl.int64) * N2
            + row_b[None, :] * H_N2
            + row_t[None, :]
        )
        tl.store(l_ptrs, val_t, mask=hmask[:, None] & row_mask[None, :])

    # ---- right projection + right gate + mask ----
    for n0 in range(0, num_n):
        n = n0 * BLOCK_N + offs_n
        hmask = n < H

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        accg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k0 in range(0, num_k):
            k = k0 * BLOCK_K + offs_k
            km = k < C

            xmat = tl.load(
                x_ptr + offs_m[:, None] * C + k[None, :],
                mask=row_mask[:, None] & km[None, :],
                other=0.0,
            ).to(tl.float32)

            nw = tl.load(norm_w_ptr + k, mask=km, other=0.0).to(tl.float32)
            nb = tl.load(norm_b_ptr + k, mask=km, other=0.0).to(tl.float32)

            xn = (xmat - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            xd = xn.to(tl.bfloat16)

            w = tl.load(
                wr_ptr + k[:, None] * H + n[None, :],
                mask=km[:, None] & hmask[None, :],
                other=0.0,
            )
            wg = tl.load(
                wrg_ptr + k[:, None] * H + n[None, :],
                mask=km[:, None] & hmask[None, :],
                other=0.0,
            )

            acc = tl.dot(xd, w, acc)
            accg = tl.dot(xd, wg, accg)

        gate = 1.0 / (1.0 + tl.exp(-accg))
        val = acc * gate * mask_vals[:, None]

        val_t = tl.trans(val.to(tl.bfloat16))
        r_ptrs = (
            r_ptr
            + n[:, None].to(tl.int64) * N2
            + row_b[None, :] * H_N2
            + row_t[None, :]
        )
        tl.store(r_ptrs, val_t, mask=hmask[:, None] & row_mask[None, :])

    # ---- out gate only ----
    for n0 in range(0, num_n):
        n = n0 * BLOCK_N + offs_n
        hmask = n < H

        accg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k0 in range(0, num_k):
            k = k0 * BLOCK_K + offs_k
            km = k < C

            xmat = tl.load(
                x_ptr + offs_m[:, None] * C + k[None, :],
                mask=row_mask[:, None] & km[None, :],
                other=0.0,
            ).to(tl.float32)

            nw = tl.load(norm_w_ptr + k, mask=km, other=0.0).to(tl.float32)
            nb = tl.load(norm_b_ptr + k, mask=km, other=0.0).to(tl.float32)

            xn = (xmat - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            xd = xn.to(tl.bfloat16)

            w = tl.load(
                wo_ptr + k[:, None] * H + n[None, :],
                mask=km[:, None] & hmask[None, :],
                other=0.0,
            )
            accg = tl.dot(xd, w, accg)

        val = 1.0 / (1.0 + tl.exp(-accg))
        og_ptrs = og_ptr + offs_m[:, None] * H + n[None, :]
        tl.store(og_ptrs, val.to(tl.bfloat16), mask=row_mask[:, None] & hmask[None, :])


@triton.jit
def _tri_mul_back_kernel(
    out_h_ptr,
    og_ptr,
    nw_ptr,
    nb_ptr,
    wout_ptr,
    out_ptr,
    M,
    H,
    C,
    N2,
    H_N2,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    row_mask = offs_m < M

    # Global token index in [B, N, N] is offs_m.
    # out_h is [B, H, N, N], so for fixed token and feature h:
    # index = b * (H * N2) + h * N2 + local_token.
    row_b = (offs_m // N2).to(tl.int64)
    row_t = (offs_m - row_b * N2).to(tl.int64)
    token_base = row_b[None, :] * H_N2 + row_t[None, :]

    offs_k = tl.arange(0, BLOCK_K)
    num_k = tl.cdiv(H, BLOCK_K)

    # ---- LayerNorm statistics over hidden dimension H ----
    sum1 = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sum2 = tl.zeros((BLOCK_M,), dtype=tl.float32)

    for k0 in range(0, num_k):
        h = k0 * BLOCK_K + offs_k
        hmask = h < H
        lm_t = hmask[:, None] & row_mask[None, :]

        # Load as [feature, token] so the token dimension is contiguous.
        y_t = tl.load(
            out_h_ptr + h[:, None].to(tl.int64) * N2 + token_base,
            mask=lm_t,
            other=0.0,
        ).to(tl.float32)

        # Sum over feature axis (axis=0 of y_t), giving per-token sums.
        sum1 += tl.sum(y_t, axis=0)
        sum2 += tl.sum(y_t * y_t, axis=0)

    mean = sum1 / H
    var = sum2 / H - mean * mean
    var = tl.where(var > 0.0, var, 0.0)
    rstd = 1.0 / tl.sqrt(var + eps)

    offs_c = tl.arange(0, BLOCK_C)
    num_c = tl.cdiv(C, BLOCK_C)

    # ---- LN + out_gate + final projection ----
    for c0 in range(0, num_c):
        c = c0 * BLOCK_C + offs_c
        cmask = c < C

        acc = tl.zeros((BLOCK_M, BLOCK_C), dtype=tl.float32)

        for k0 in range(0, num_k):
            h = k0 * BLOCK_K + offs_k
            hmask = h < H
            lm_t = hmask[:, None] & row_mask[None, :]

            # y_t: [feature, token], loaded coalesced along token.
            y_t = tl.load(
                out_h_ptr + h[:, None].to(tl.int64) * N2 + token_base,
                mask=lm_t,
                other=0.0,
            ).to(tl.float32)

            # y: [token, feature]
            y = tl.trans(y_t)

            # gate is naturally [token, feature] and coalesced.
            g = tl.load(
                og_ptr + offs_m[:, None] * H + h[None, :],
                mask=row_mask[:, None] & hmask[None, :],
                other=0.0,
            ).to(tl.float32)

            nw = tl.load(nw_ptr + h, mask=hmask, other=0.0).to(tl.float32)
            nb = tl.load(nb_ptr + h, mask=hmask, other=0.0).to(tl.float32)

            a = (y - mean[:, None]) * rstd[:, None] * nw[None, :] + nb[None, :]
            a = a * g
            ab = a.to(tl.bfloat16)

            w = tl.load(
                wout_ptr + h[:, None] * C + c[None, :],
                mask=hmask[:, None] & cmask[None, :],
                other=0.0,
            )
            acc = tl.dot(ab, w, acc)

        out_ptrs = out_ptr + offs_m[:, None] * C + c[None, :]
        tl.store(out_ptrs, acc, mask=row_mask[:, None] & cmask[None, :])


def custom_kernel(data):
    if len(data) == 3:
        x, weights, config = data
        mask = None
    else:
        x, mask, weights, config = data

    B, N, _, C = x.shape
    H = int(config.get("hidden_dim", weights["left_proj.weight"].shape[0]))
    M = B * N * N
    N2 = N * N
    H_N2 = H * N2
    eps = float(config.get("eps", 1e-5))

    device = x.device

    if x.dtype != torch.float32:
        x = x.to(torch.float32)
    if not x.is_contiguous():
        x = x.contiguous()

    has_mask = (mask is not None) and (not config.get("nomask", False))
    if has_mask:
        if mask.dim() == 4:
            mask = mask.squeeze(-1)
        if mask.dtype != torch.float32:
            mask_f = mask.to(torch.float32)
        else:
            mask_f = mask
        if not mask_f.is_contiguous():
            mask_f = mask_f.contiguous()
        mask_f = mask_f.reshape(M)
    else:
        mask_f = x  # dummy pointer; not read when HAS_MASK is False

    # Transposed dense weights for dot(A[X,K], W[K,Y])
    def _trans_bf16(key):
        return weights[key].T.contiguous().to(torch.bfloat16)

    W_L = _trans_bf16("left_proj.weight")
    W_LG = _trans_bf16("left_gate.weight")
    W_R = _trans_bf16("right_proj.weight")
    W_RG = _trans_bf16("right_gate.weight")
    W_OG = _trans_bf16("out_gate.weight")

    # Final linear: to_out.weight is [C, H], we need [H, C].
    W_OUT = weights["to_out.weight"].T.contiguous().to(torch.bfloat16)

    norm_w = weights["norm.weight"].to(torch.float32).contiguous()
    norm_b = weights["norm.bias"].to(torch.float32).contiguous()
    out_norm_w = weights["to_out_norm.weight"].to(torch.float32).contiguous()
    out_norm_b = weights["to_out_norm.bias"].to(torch.float32).contiguous()

    # Left/right in [B, H, N, N] to make the batched GEMM contiguous per head.
    L = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    R = torch.empty((B, H, N, N), dtype=torch.bfloat16, device=device)
    # Out gate is used along the final feature dimension, so keep [B, N, N, H].
    OG = torch.empty((M, H), dtype=torch.bfloat16, device=device)

    BLOCK_M = 64
    if H > 256:
        BLOCK_M = 32

    BLOCK_K = 128 if C >= 128 else max(16, _next_pow2(C))
    BLOCK_N = 128 if H >= 128 else max(16, _next_pow2(H))

    grid = (triton.cdiv(M, BLOCK_M),)
    _tri_mul_front_kernel[grid](
        x,
        mask_f,
        norm_w,
        norm_b,
        W_L,
        W_LG,
        W_R,
        W_RG,
        W_OG,
        L,
        R,
        OG,
        M,
        C,
        H,
        N2,
        H_N2,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        HAS_MASK=has_mask,
        num_warps=4,
        num_stages=2,
    )

    # Core contraction: for each (b, d):
    #   out[b,d] = L[b,d] @ R[b,d]^T
    BH = B * H
    out_h = torch.bmm(L.view(BH, N, N), R.view(BH, N, N).transpose(1, 2))
    del L, R

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    BLOCK_MF = 64
    BLOCK_KF = 128 if H >= 128 else max(16, _next_pow2(H))
    BLOCK_CF = 128 if C >= 128 else max(16, _next_pow2(C))

    gridf = (triton.cdiv(M, BLOCK_MF),)
    _tri_mul_back_kernel[gridf](
        out_h,
        OG,
        out_norm_w,
        out_norm_b,
        W_OUT,
        out,
        M,
        H,
        C,
        N2,
        H_N2,
        eps,
        BLOCK_M=BLOCK_MF,
        BLOCK_K=BLOCK_KF,
        BLOCK_C=BLOCK_CF,
        num_warps=4,
        num_stages=2,
    )

    return out
