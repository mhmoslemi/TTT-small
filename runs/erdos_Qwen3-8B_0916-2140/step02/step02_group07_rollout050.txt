import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, NonlinearConstraint
from numpy import linalg as la

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Create bounds for variables
    bounds = Bounds(0, 1)
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step with better settings
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints=[NonlinearConstraint(constraint_sum, 0, 0)],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 200}
        )
        return result.x if result.success else h
    
    # Initial guess
    h0 = initial_h.copy()
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Try different starting points
    for _ in range(5):
        # Random perturbation with decreasing magnitude
        perturbation = np.random.normal(0, 0.05 / np.sqrt(n_points), size=n_points)
        perturbed_h = np.clip(h0 + perturbation, 0, 1)
        
        # Optimize from the perturbed point
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            h0 = current_h
            print(f"New best C5: {best_C5}")
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
