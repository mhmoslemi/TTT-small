import numpy as np
import scipy.optimize as opt
import time
from functools import partial
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess: use known construction with 200 points
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0
    initial_h = initial_h_values

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # First optimization step
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Check if we can improve with a different initial guess
    def alternate_initial_guess(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                h[i] = 0.0
            elif x < 1.5:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h

    alternate_h = alternate_initial_guess(n_points)
    result_alt = minimize(objective, alternate_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    alt_best_c5 = compute_overlap(result_alt.x)
    best_h = result_alt.x
    best_c5 = alt_best_c5

    # Optional scaling up to 400 points if time allows
    if budget_s > 500:
        n_points = 400
        dx = 2.0 / n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        initial_h = np.clip(initial_h, 0, 1)

        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx

        def objective_new(x):
            return compute_overlap_new(x)

        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2

        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})

        result_new = minimize(objective_new, initial_h, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        best_h_new = result_new.x
        best_c5 = compute_overlap_new(best_h_new)

    return (best_h_new.tolist() if budget_s > 500 else best_h.tolist(), best_c5, n_points)
