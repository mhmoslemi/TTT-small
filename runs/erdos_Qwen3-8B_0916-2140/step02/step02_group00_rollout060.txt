    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # First optimization with SLSQP

    # Second optimization with Adam-like method for better exploration
    class AdamOptimizer:
        def __init__(self, x0, bounds, objective, step_size=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iter=100):
            self.x0 = x0
            self.bounds = bounds
            self.objective = objective
            self.step_size = step_size
            self.beta1 = beta1
            self.beta2 = beta2
            self.eps = eps
            self.max_iter = max_iter
            self.m = np.zeros_like(x0)
            self.v = np.zeros_like(x0)
            self.t = 0

        def optimize(self):
            x = self.x0.copy()
            m = np.zeros_like(x)
            v = np.zeros_like(x)
            t = 0
            best_x = x
            best_val = self.objective(x)
            for _ in range(self.max_iter):
                t += 1
                grad = np.gradient(self.objective, x, edge_order=2)
                m = self.beta1 * m + (1 - self.beta1) * grad
                v = self.beta2 * v + (1 - self.beta2) * grad**2
                m_hat = m / (1 - self.beta1**t)
                v_hat = v / (1 - self.beta2**t)
                step = self.step_size * m_hat / (np.sqrt(v_hat) + self.eps)
                x_new = x - step
                # Clip to bounds
                x_new = np.clip(x_new, self.bounds[0], self.bounds[1])
                x = x_new
                val = self.objective(x)
                if val < best_val:
                    best_x = x
                    best_val = val
            return best_x, best_val

        if best_c5_final < best_c5:
            best_h = best_h_final
            best_c5 = best_c5_final

    return (best_h.tolist(), best_c5, n_points)
