import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from sklearn.decomposition import PCA

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid
    n_points = 200
    dx = 2.0 / n_points
    
    # Load initial_h_values from the global variable
    initial_h_values = np.load("initial_h_values.npy")
    
    # Interpolate the initial_h_values to the new n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_h = np.interp(x_new, x_initial, initial_h_values)
    
    # Define objective function to minimize the maximum overlap
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define constraint for integral to be 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Define bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Optimization setup
    result = minimize(objective, initial_h, bounds=bounds, constraints=LinearConstraint(np.ones(n_points), 0, n_points / 2), method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
