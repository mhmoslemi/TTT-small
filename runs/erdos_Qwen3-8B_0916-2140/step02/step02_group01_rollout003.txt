```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess based on initial_h_values (n=100)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is pre-saved
    initial_h = initial_h_values[:n_points]
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem with more sophisticated constraints
    # We will use a constrained optimization method to find better solutions
    
    # Define the optimization problem with constraints
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the best solution found
    return best_h, best_c5, n_points
```
