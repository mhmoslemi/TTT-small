import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points for the function h
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We will use a smooth construction with periodicity and symmetry, and include a fine-tuning step
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=100
    initial_h = np.zeros(n_points)
    
    # We'll use a piecewise linear function that stays within bounds [0,1]
    # Define intervals [a_i, b_i] with values at the boundaries
    x = np.linspace(0, 2, n_points)
    
    # Define a function that rises from 0 to 1, stays flat from 1 to 1.5, then decreases to 0 at 2
    # This reduces the overlap by avoiding high density in overlapping regions
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
