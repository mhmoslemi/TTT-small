import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to perform a single perturbation and optimization step
    def perturb_and_optimize(h):
        perturbation = np.random.normal(0, 0.05, size=len(h))
        perturbed_h = np.clip(h + perturbation, 0, 1)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        return current_h, current_C5
    
    # Function to perform a single local search with multiple perturbations
    def local_search(h):
        best_h = h
        best_C5 = compute_C5(h)
        for _ in range(5):
            perturbed_h = np.clip(h + np.random.normal(0, 0.05, size=len(h)), 0, 1)
            current_h = optimize_step(perturbed_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_h = current_h
                best_C5 = current_C5
        return best_h, best_C5
    
    # Multi-stage optimization
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Stage 1: Local search with multiple perturbations
    for _ in range(5):
        current_h, current_C5 = local_search(best_h)
        if current_C5 < best_C5:
            best_h = current_h
            best_C5 = current_C5
    
    # Stage 2: Global search with random initialization and perturbations
    for _ in range(3):
        random_h = np.random.rand(n_points)
        random_h = np.clip(random_h, 0, 1)
        random_h = (random_h - np.min(random_h)) / (np.max(random_h) - np.min(random_h))
        random_h = np.clip(random_h, 0, 1)
        random_h = np.random.permutation(random_h)
        current_h, current_C5 = local_search(random_h)
        if current_C5 < best_C5:
            best_h = current_h
            best_C5 = current_C5
    
    # Stage 3: Final fine-tuning with a grid search around the best solution
    best_h = best_h
    best_C5 = compute_C5(best_h)
    
    # Create a grid of perturbations around the best solution
    grid_size = 10
    perturbations = np.random.normal(0, 0.05, size=(grid_size, n_points))
    perturbations = np.clip(perturbations, -0.1, 0.1)
    
    for i in range(grid_size):
        perturbed_h = np.clip(best_h + perturbations[i], 0, 1)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_h = current_h
            best_C5 = current_C5
    
    # Check time budget
    if time.time() - start_time > time_budget:
        print(f"Time budget exceeded. Returning best found solution.")
    
    return best_h, best_C5, len(best_h)
