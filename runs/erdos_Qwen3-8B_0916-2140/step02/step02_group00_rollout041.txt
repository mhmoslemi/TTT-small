import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points
    n_points = 100
    dx = 2.0 / n_points

    # Directly access the pre-imported initial_h_values for n=200
    initial_h_values = evaluate_erdos_solution()
    initial_h_values = np.array(initial_h_values, dtype=np.float64)
    
    # Interpolate initial_h_values to n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_h = np.interp(x_new, x_initial, initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization stage: local search
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        options={'ftol': 1e-10, 'maxiter': 200}
    )
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # If time allows, perform a second stage with more points
    if budget_s > 300:
        n_points = 200
        dx = 2.0 / n_points
        x_new = np.linspace(0, 2, n_points)
        initial_h = np.interp(x_new, np.linspace(0, 2, len(best_h)), best_h)
        initial_h = np.clip(initial_h, 0, 1)
        
        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx
        
        def objective_new(x):
            return compute_overlap_new(x)
        
        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2
        
        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})
        
        # Second optimization stage: more accurate search
        result_new = minimize(
            objective_new,
            initial_h,
            bounds=bounds_new,
            constraints=cons_new,
            method='L-BFGS-B',
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        
        best_h_new = result_new.x
        best_c5 = compute_overlap_new(best_h_new)
    
    return (best_h_new.tolist() if budget_s > 300 else best_h.tolist(), best_c5, n_points)
