import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 200
    dx = 2.0 / n_points
    
    # Define initial guess based on the provided initial_h_values (for n=100)
    # We'll use a more refined construction with a smooth, piecewise linear profile
    
    # Define the initial function with periodicity and symmetry to minimize overlap
    # This construction starts with a more uniform distribution of the function values
    
    # Define a smooth, symmetric, piecewise linear function with increased smoothness
    x = np.linspace(0, 2, n_points)
    initial_h = np.zeros(n_points)
    
    # Define a more refined function that starts at 0, increases slowly, peaks, then decreases
    # This function is designed to create a more even distribution to minimize overlap
    # It uses a piecewise linear approach with increased smoothness
    
    # Define intervals where the function is piecewise linear to reduce complexity
    # The function starts at 0, increases to 0.5 at 0.5, stays at 0.5 from 0.5 to 1.5, and decreases to 0 at 2
    # This ensures symmetry and a balanced distribution of values
    
    # Create the piecewise linear function
    for i in range(n_points):
        x_i = x[i]
        if x_i <= 0.5:
            initial_h[i] = x_i / 0.5
        elif x_i <= 1.5:
            initial_h[i] = 1.0
        else:
            initial_h[i] = (2 - x_i) / 0.5
    
    # Compute the current integral and scale the function to ensure the total is 1
    integral = np.sum(initial_h) * dx
    initial_h_scaled = initial_h / integral
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Set up the optimization problem
    result = minimize(objective, initial_h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500, 'disp': False})
    
    # Return the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
