    h_values = np.interp(indices, np.arange(len(initial_h_values)), initial_h_values)

    # Create the grid
    x = np.linspace(0, 2, n_points)

    # Compute the integral to ensure it's 1
    integral = np.sum(h_values) * dx
    h_values = h_values / integral

    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1

    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)

    # Use SLSQP for optimization
    result = minimize(objective, h_values, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun

    return best_h, best_c5, n_points
