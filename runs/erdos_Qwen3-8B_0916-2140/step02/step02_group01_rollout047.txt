```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial guess: use a symmetric, piecewise linear function with different segments
    # We use a design that gradually increases, peaks, and decreases to balance the overlap
    initial_h = np.zeros(n_points)
    
    # Define a function that increases from 0 to 1, stays flat from 1 to 1.5, then decreases to 0 at 2
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    integral = np.sum(h) * dx
    h_scaled = h / integral
    
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
```
