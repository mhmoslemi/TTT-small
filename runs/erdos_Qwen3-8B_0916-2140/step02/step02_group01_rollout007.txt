import numpy as np
import scipy.optimize as optimize
import time

from scipy.optimize import minimize, Bounds
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 200  # Increase resolution for better optimization
    dx = 2.0 / n_points
    
    # Use initial_h_values as a starting point
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is preloaded
    
    # Normalize the initial_h_values to have integral 1
    initial_h = initial_h_values[:n_points]
    integral = np.sum(initial_h) * dx
    initial_h_scaled = initial_h / integral
    
    # Define the objective function
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint for integral
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2  # Ensure sum(h) * dx = 1
    
    # Define bounds for h values (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Use a more efficient optimization method with enhanced settings
    result = minimize(
        objective,
        initial_h_scaled,
        bounds=bounds,
        constraints=({"type": "eq", "fun": constraint_sum}),
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 300, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
