import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a high-resolution grid
    n_points = 200
    dx = 2.0 / n_points
    
    # Load the initial construction from the global variable
    initial_h_values = np.array(evaluate_erdos_solution(), dtype=np.float64)
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization with SLSQP
    result = minimize(objective, initial_h_values, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try a different optimization method, e.g., COBYLA
    result_cobyla = minimize(objective, best_h, bounds=bounds, constraints=cons, method='COBYLA', options={'maxiter': 150, 'disp': False})
    best_h_cobyla = result_cobyla.x
    best_c5_cobyla = compute_overlap(best_h_cobyla)
    
    # Use the better of the two results
    if best_c5_cobyla < best_c5:
        best_h = best_h_cobyla
        best_c5 = best_c5_cobyla
    
    # Try a local search with a different starting point
    perturb_h = best_h + np.random.normal(0, 0.05, n_points)
    perturb_h = np.clip(perturb_h, 0, 1)
    
    result_perturb = minimize(objective, perturb_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h_perturb = result_perturb.x
    best_c5_perturb = compute_overlap(best_h_perturb)
    
    if best_c5_perturb < best_c5:
        best_h = best_h_perturb
        best_c5 = best_c5_perturb
    
    # Check if we have time to try a larger grid
    if budget_s > 300:
        n_points = 400

        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)

            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx
        
        def objective_new(x):
            return compute_overlap_new(x)
        
        result_new = minimize(objective_new, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h_new = result_new.x
        best_c5_new = compute_overlap_new(best_h_new)
        
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    return (best_h.tolist(), best_c5, n_points)
