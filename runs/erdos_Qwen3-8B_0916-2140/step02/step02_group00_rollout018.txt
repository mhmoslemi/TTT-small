import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Reuse the initial_h_values (n=200 samples)
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraint_sum = lambda x: np.sum(x) - n_points / 2

    # Optimization function
    def objective(x):
        return compute_overlap(x)

    # Initial guess
    initial_h = initial_h_values.copy()
    initial_h = np.clip(initial_h, 0, 1)

    # Perform local optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, method='L-BFGS-B', options={'ftol': 1e-12, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Perform a population-based optimization to escape local optima
    population_size = 20
    population = [np.clip(np.random.uniform(0, 1, n_points), 0, 1) for _ in range(population_size)]
    population = [p for p in population if np.sum(p) * dx >= 0.9 and np.sum(p) * dx <= 1.1]

    best_solution = best_h
    best_c5 = compute_overlap(best_solution)

    for _ in range(10):
        new_population = []
        for p in population:
            # Perturb the current solution
            perturbed = p + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            new_population.append(perturbed)
        population = new_population

        # Evaluate the population
        for p in population:
            c5 = compute_overlap(p)
            if c5 < best_c5:
                best_c5 = c5
                best_solution = p

    return (best_solution.tolist(), best_c5, n_points)
