def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, may scale up if needed
    dx = 2.0 / n_points

    # Initial guess based on known construction (n=100)
    # From the initial_h_values global variable which is not shown, we'll start with a simple piecewise construction
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Start with a local optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Optionally scale up to more points if needed (double the points for a more accurate search)
    if budget_s > 300:
        n_points = 200
        dx = 2.0 / n_points
        initial_h_new = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        initial_h_new = np.clip(initial_h_new, 0, 1)

        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx

        def objective_new(x):
            return compute_overlap_new(x)

        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2

        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})

        # Use differential evolution for global optimization with a heuristic
        def objective_with_robustness(x):
            x = np.clip(x, 0, 1)
            return compute_overlap_new(x)

        result_de = differential_evolution(objective_with_robustness, bounds_new, strategy='best1bin', maxiter=100, popsize=20, mutation=(0.5, 1), recombination=0.9, tol=1e-10, mutation_independent=True)
        best_h_new = result_de.x
        best_c5 = compute_overlap_new(best_h_new)

    return (best_h_new.tolist(), best_c5, n_points)
