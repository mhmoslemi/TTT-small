    else:
        # Default initial guess: linearly increasing function
        initial_h = np.linspace(0, 1, n_points)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Perform local optimization with multiple starting points to avoid local optima
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    # Try multiple starting points
    for i in range(5):
        # Perturb the initial guess
        perturb = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(initial_h + perturb, 0, 1)
        
        # Optimize
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        
        # Update best solution
        if result.success:
            current_c5 = result.fun
