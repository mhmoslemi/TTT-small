import numpy as np

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a smaller grid to speed up computations and avoid overfitting
    n_points = 200
    dx = 2.0 / n_points
    
    # Define initial guess based on a more refined construction
    # We use a combination of piecewise linear and piecewise constant segments
    # We start with a symmetric, piecewise linear function with increased smoothness
    
    initial_h = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    
    # Initial construction: piecewise linear from 0 to 1, flat from 1 to 1.5, linear from 1.5 to 2
    for i in range(n_points):
        x_i = x[i]
        if x_i <= 1:
            initial_h[i] = x_i
        elif x_i <= 1.5:
            initial_h[i] = 1
        else:
            initial_h[i] = 2 - x_i
    
    # Compute initial integral and scale to ensure total integral is 1
    integral = np.sum(initial_h) * dx
    initial_h = initial_h / integral
    
    # Define the objective function to minimize the max overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Use a more robust optimization method with additional constraints to improve results
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
