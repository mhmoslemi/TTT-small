def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We will use a refined version of the initial construction by interpolating or extending it
    # For this example, we use a smooth construction with periodicity and symmetry
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=100
    initial_h = np.zeros(n_points)
    
    # Define a piecewise linear function with increased smoothness
    # Define intervals where the function is piecewise linear to reduce complexity
    # To reduce experimentation time and avoid overfitting, we'll use a principled design
    
    # Define a design that gradually increases, peaks, then decreases
    # This helps in distributing the function values to reduce max overlap
    
    # Let's design a smooth, continuous function with higher-order behavior
    # We will use a smooth function that is a convolution of a triangular kernel
    # This ensures continuity and smoothness while keeping the total integral equal to 1
    
    # Define a smooth, continuous, symmetric function with two peaks
    # The function will be composed of two triangles that overlap smoothly
    
    # Define the function as follows:
    # - From x=0 to x=0.5, h(x) = x
    # - From x=0.5 to x=1.5, h(x) = 1 - 2*(x - 0.5)
    # - From x=1.5 to x=2, h(x) = 2 - x
    
    # Create the interval grid
    x = np.linspace(0, 2, n_points)
    
    # Compute the function
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 0.5:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1 - 2 * (xi - 0.5)
        else:
            h[i] = 2 - xi
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
