import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess with 100 points
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            initial_h_values[i] = 0.5 * (1 + np.cos(np.pi * x))
        else:
            initial_h_values[i] = 0.5 * (1 + np.cos(np.pi * (x - 1.0)))
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Define the optimization function
    def optimize_with_perturbations(x0, n_perturb=5, max_iter=100):
        best_c5 = float('inf')
        best_h = x0.copy()
        
        for _ in range(n_perturb):
            # Perturb the initial guess
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            x0_perturb = np.clip(x0 + perturb, 0, 1)
            
            # Perform optimization
            result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': max_iter})
            current_c5 = compute_overlap(result.x)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        return best_h, best_c5
    
    # Run the optimization with the initial guess
    start_time = time.time()
    x0 = initial_h_values.copy()
    best_h, best_c5 = optimize_with_perturbations(x0)
    
    # Check if we have time for more detailed optimization
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s * 0.7:
        # Try increasing the number of points for better resolution
        n_points = 200
        dx = 2.0 / n_points
        initial_h_values = np.array([0.0] * n_points)
        for i in range(n_points):
            x = i * dx
            if x < 1.0:
                initial_h_values[i] = 0.5 * (1 + np.cos(np.pi * x))
            else:
                initial_h_values[i] = 0.5 * (1 + np.cos(np.pi * (x - 1.0)))
        initial_h_values = np.clip(initial_h_values, 0, 1)
        
        # Define new objective and constraints
        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx
        
        def objective_new(x):
            return compute_overlap_new(x)
        
        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2
        
        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})
        
        # Run the optimization with the new points
        best_h_new, best_c5_new = optimize_with_perturbations(initial_h_values, n_perturb=3, max_iter=150)
        
        # Compare and choose the best solution
        if best_c5_new < best_c5:
            best_h, best_c5 = best_h_new, best_c5_new
    
    return (best_h.tolist(), best_c5, n_points)
