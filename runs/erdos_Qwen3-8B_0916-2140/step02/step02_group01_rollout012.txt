import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 400
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (n=100 samples)
    # We will use a more refined construction by interpolating or extending it
    # For this example, we use a smooth, symmetric, and piecewise linear function
    
    # Define the initial function
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    # Define a piecewise linear function that increases from 0 to 1, stays flat until 1.5, then decreases
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    # Compute the integral of the initial function
    initial_integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / initial_integral
    
    # The initial guess will be the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
