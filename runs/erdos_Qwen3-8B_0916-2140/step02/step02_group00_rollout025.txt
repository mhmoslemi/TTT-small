import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    # Starting with a larger number of points for better resolution
    n_points = 200
    dx = 2.0 / n_points

    # Load the initial construction (n=200 samples)
    initial_h_values = np.load("initial_h_values.npy")  # Assuming this file is available
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = np.clip(initial_h, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Use a more robust optimization strategy with multiple restarts and adaptive settings
    best_h = initial_h
    best_c5 = compute_overlap(initial_h)

    # Define a function to perform optimization with given initial guess
    def optimize_with_guess(initial_guess):
        result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 200})
        if result.success:
            return result.x
        return initial_guess

    # Perform multiple restarts with different initial guesses
    for _ in range(5):
        # Generate random perturbations around the initial guess
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        guess = np.clip(initial_h + perturbation, 0, 1)
        h_candidate = optimize_with_guess(guess)
        c5_candidate = compute_overlap(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    # If budget allows, scale up to even higher resolution
    if budget_s > 300:
        n_points = 400
        dx = 2.0 / n_points
        # Interpolate initial guess to higher resolution
        initial_high_res = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        initial_high_res = np.clip(initial_high_res, 0, 1)

        def compute_overlap_high_res(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx

        def objective_high_res(x):
            return compute_overlap_high_res(x)

        def constraint_sum_high_res(x):
            return np.sum(x) - n_points / 2

        bounds_high_res = [(0, 1) for _ in range(n_points)]
        cons_high_res = ({'type': 'eq', 'fun': constraint_sum_high_res})

        # Final optimization with higher resolution
        result_high_res = minimize(objective_high_res, initial_high_res, bounds=bounds_high_res, constraints=cons_high_res, method='L-BFGS-B', options={'ftol': 1e-9, 'maxiter': 200})
        best_h_high_res = result_high_res.x
        best_c5_high_res = compute_overlap_high_res(best_h_high_res)

        if best_c5_high_res < best_c5:
            best_h = best_h_high_res
            best_c5 = best_c5_high_res

    return (best_h.tolist(), best_c5, n_points)
