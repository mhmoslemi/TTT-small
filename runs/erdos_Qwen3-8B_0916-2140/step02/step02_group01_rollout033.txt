import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def initial_h_values(n_points):
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    integral = np.sum(h) * 2.0 / n_points
    h_scaled = h / integral
    return h_scaled

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a higher resolution to capture finer details
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with a symmetric, piecewise linear function
    h_initial = initial_h_values(n_points)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to maintain integral equal to 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Run the optimization
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
