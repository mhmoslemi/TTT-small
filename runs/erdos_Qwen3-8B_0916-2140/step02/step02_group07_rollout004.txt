import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to generate random perturbation
    def generate_perturbation(h, perturbation_factor=0.05):
        perturbation = np.random.normal(0, perturbation_factor, size=n_points)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        return perturbed_h
    
    # Function to perform a single perturbation and optimization step
    def perturb_and_optimize(h):
        perturbed_h = generate_perturbation(h)
        current_h = optimize_step(perturbed_h)
        return current_h
    
    # Function to perform a single local optimization step
    def local_optimize(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to perform a full optimization with multiple iterations
    def full_optimize(h, iterations=10, perturbation_factor=0.05):
        current_h = h
        best_h = current_h
        best_C5 = compute_C5(current_h)
        for _ in range(iterations):
            current_h = local_optimize(current_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
            # Add perturbation to avoid local minima
            current_h = generate_perturbation(current_h, perturbation_factor)
        return best_h, best_C5

    best_h = h0
    best_C5 = compute_C5(h0)

        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
        # Add perturbation and continue
        best_h = generate_perturbation(best_h)
    
    # Final optimization
    final_h = local_optimize(best_h)
    final_C5 = compute_C5(final_h)
    
    # Check time budget
    if time.time() - start_time > time_budget:
        print(f"Time budget exceeded. Returning best found solution.")
    
    return final_h, final_C5, len(final_h)
