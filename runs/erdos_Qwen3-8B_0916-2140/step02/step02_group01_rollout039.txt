import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points
    n_points = 200
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We will use a more refined and smoother construction to explore new possibilities
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=200
    
    # Define a piecewise linear function with more segments to capture finer details
    # The function will increase, plateau, and decrease to ensure integral is 1
    
    # Define the function as follows:
    # - From x=0 to x=0.5, h(x) = 2x
    # - From x=0.5 to x=1.0, h(x) = 1
    # - From x=1.0 to x=1.5, h(x) = 2 - 2(x - 1.0)
    # - From x=1.5 to x=2.0, h(x) = 0
    
    # Convert the grid to actual x positions
    x = np.linspace(0, 2, n_points)
    
    # Compute h(x)
    h = np.zeros(n_points)
    for i in range(n_points):
        x_i = x[i]
        if x_i <= 0.5:
            h[i] = 2 * x_i
        elif x_i <= 1.0:
            h[i] = 1.0
        elif x_i <= 1.5:
            h[i] = 2 - 2 * (x_i - 1.0)
        else:
            h[i] = 0.0
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
