import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.fft import fft, ifft
from scipy.optimize import LinearConstraint
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points and step size
    n_points = 200
    dx = 2.0 / n_points
    
    # Precompute the x grid
    x = np.linspace(0, 2, n_points)
    
    # Define a refined initial guess using a piecewise linear function that gradually increases, plateaus, then decreases
    # The function stays within [0,1] and ensures the integral is 1
    
    # Define the initial function as a piecewise linear function
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h_initial[i] = xi
        elif xi <= 1.5:
            h_initial[i] = 1
        else:
            h_initial[i] = 2 - xi
    
    # Compute the current integral and scale it to ensure the total is 1
    integral = np.sum(h_initial) * dx
    h_initial = h_initial / integral
    
    # Define the objective function to minimize the maximum overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Define bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem with an initial guess
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution and compute the bound
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
