import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5

    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)

    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Initial guess
    h0 = initial_h.copy()

    # Optimization settings
    time_budget = budget_s
    start_time = time.time()

    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h

    # Function to generate a random perturbation
    def perturb(h, sigma=0.05):
        perturbation = np.random.normal(0, sigma, size=len(h))
        perturbed_h = np.clip(h + perturbation, 0, 1)
        return perturbed_h

    # Function to generate random initial solution
    def random_initial_solution(n_points):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        scaling_factor = (n_points / 2) / np.sum(h)
        h = np.clip(h * scaling_factor, 0, 1)
        return h

    # Best solution found so far
    best_h = h0
    best_C5 = compute_C5(h0)

    # Perform multiple rounds of local optimization
    for _ in range(10):
        # Generate a perturbed solution
        perturbed_h = perturb(h0)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)

        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            h0 = current_h
            print(f"New best C5: {best_C5}")

        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break

    # Try optimizing from a random initial solution if not enough progress
    if best_C5 > 0.380876:
        print("Trying random initial solution for further improvement...")
        random_h = random_initial_solution(n_points)
        current_h = optimize_step(random_h)
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f"New best C5: {best_C5}")

    return best_h, best_C5, len(best_h)
