import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to compute the maximum overlap for a given h
    def compute_max_overlap(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    # Function to generate a random perturbation
    def perturb(h, magnitude=0.05):
        return np.clip(h + np.random.normal(0, magnitude, size=len(h)), 0, 1)
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to check if a solution is better than the current best
    def is_better(new_h, current_h, current_c5):
        new_c5 = compute_max_overlap(new_h)
        return new_c5 < current_c5
    
    time_budget = budget_s
    start_time = time.time()
    
    # Initial guess
    h0 = initial_h.copy()
    best_h = h0
    best_c5 = compute_max_overlap(h0)
    print(f"Initial best C5: {best_c5}")
    
    # Perform multiple rounds of local optimization
    for _ in range(20):
        # Perturb the current solution to avoid local optima
        perturbed_h = perturb(h0, magnitude=0.05)
        
        # Optimize from the perturbed point
        current_h = optimize_step(perturbed_h)
        current_c5 = compute_max_overlap(current_h)
        
        # Check if the new solution is better
        if is_better(current_h, best_h, best_c5):
            best_h = current_h
            best_c5 = current_c5
            print(f"New best C5: {best_c5}")
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_c5, len(best_h)
