import numpy as np
import scipy.optimize as opt
import time
from functools import partial
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points
    dx = 2.0 / n_points

    # Initial guess based on known construction (n=200)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this file contains the initial guess
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = np.clip(initial_h, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Start with a local optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Optionally try different starting points to avoid local optima
    if budget_s > 300:
        # Try a different initial guess based on known constructions
        # For example, use a piecewise constant function with different thresholds
        initial_h_diff = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            if x < 0.5:
                initial_h_diff[i] = 0.5
            elif x < 1.5:
                initial_h_diff[i] = 0.75
            else:
                initial_h_diff[i] = 0.25
        initial_h_diff = np.clip(initial_h_diff, 0, 1)

        result_diff = minimize(objective, initial_h_diff, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h_diff = result_diff.x
        best_c5_diff = compute_overlap(best_h_diff)

        # Compare and choose the best
        if best_c5_diff < best_c5:
            best_h = best_h_diff
            best_c5 = best_c5_diff

    return (best_h.tolist(), best_c5, n_points)
