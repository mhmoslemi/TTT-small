import numpy as np
import scipy.optimize as opt
import time
from functools import partial
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess from known construction (n=200)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is predefined
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Interpolate the initial guess to n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_h = np.interp(x_new, x_initial, initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Start with a local optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 300})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try a second round with different initial guess
    def random_initial_guess():
        return np.random.uniform(0, 1, n_points)
    
    result2 = minimize(objective, random_initial_guess(), bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 300})
    best_h2 = result2.x
    best_c52 = compute_overlap(best_h2)
    
    # Choose the better solution between the two
    if best_c5 < best_c52:
        best_h_final = best_h
        best_c5_final = best_c5
    else:
        best_h_final = best_h2
        best_c5_final = best_c52
    
    # Optionally try scaling up if budget allows
    if budget_s > 300:
        n_points = 400
        dx = 2.0 / n_points
        x_new = np.linspace(0, 2, n_points)
        initial_h = np.interp(x_new, np.linspace(0, 2, len(best_h_final)), best_h_final)
        initial_h = np.clip(initial_h, 0, 1)
        
        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx
        
        def objective_new(x):
            return compute_overlap_new(x)
        
        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2
        
        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})
        
        result_new = minimize(objective_new, initial_h, bounds=bounds_new, constraints=cons_new, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 300})
        best_h_new = result_new.x
        best_c5_new = compute_overlap_new(best_h_new)
        
        if best_c5_new < best_c5_final:
            best_h_final = best_h_new
            best_c5_final = best_c5_new
    
    return (best_h_final.tolist(), best_c5_final, n_points)
