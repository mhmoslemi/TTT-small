    dx = 2.0 / n_points

    # Define function to compute C_5
    def compute_c5(h):
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective function to minimize C_5
    def objective(h):
        return compute_c5(h)

        return np.sum(h) - n_points / 2

        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
