from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):

        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Local optimization with multiple restarts
    best_h = initial_h
    best_c5 = compute_overlap(initial_h)
    for _ in range(5):
        result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 150})
        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        initial_h = np.random.uniform(0, 1, n_points)
