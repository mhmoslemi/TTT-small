import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points for faster experimentation
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess: Use a refined version of the initial construction by interpolating or extending it
    # We'll use a piecewise linear function with increased smoothness
    
    # Define the initial function based on a sample construction
    # Create a symmetric, piecewise linear function that stays within [0,1]
    # Define the function as follows:
    # - From x=0 to x=1, h(x) = x
    # - From x=1 to x=1.5, h(x) = 1
    # - From x=1.5 to x=2, h(x) = 2 - x
    
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    # Compute the integral and scale to ensure it equals 1
    integral = np.sum(h) * dx
    h_scaled = h / integral
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Use a more robust solver with tighter tolerances
    result = minimize(objective, h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-12, 'maxiter': 1000, 'eps': 1e-8})
    
    # If the optimization fails or converges too slowly, try a local search from a perturbed point
    if not result.success or result.fun > 0.382:
        # Perturb the initial guess and retry
        perturbation = 0.01 * np.random.rand(n_points)
        perturbed_h = h_scaled + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-12, 'maxiter': 1000, 'eps': 1e-8})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
