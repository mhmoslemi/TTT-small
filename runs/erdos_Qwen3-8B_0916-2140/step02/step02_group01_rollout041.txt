import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    budget_remaining = budget_s
    
    # Use a refined construction with more points for higher accuracy
    n_points = 200
    dx = 2.0 / n_points
    total_mass = n_points / 2
    
    # Use the initial_h_values from the problem description (n=100 samples)
    # We'll perform interpolation to get the initial guess for n=200
    initial_h = np.zeros(n_points)
    initial_h_values = initial_h_values  # Provided by the problem
    
    # Interpolate initial_h_values to n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    x_new = np.linspace(0, 2, n_points)
    initial_h = np.interp(x_new, x_initial, initial_h_values)
    
    # Define objective function
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - total_mass
    
    # Define bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Use a more robust optimization algorithm with some tuning
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, 
                      method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500, 'disp': False})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, n_points
