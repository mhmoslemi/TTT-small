def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 120
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Interpolate initial_h_values to match the n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    initial_guess = np.interp(x, x_initial, initial_h_values)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the optimization problem
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1 --> sum(h) = n_points / 2
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Objective: minimize the maximum overlap
    def objective_func(h):
        return compute_c5(h)
    
    # Optimization with SLSQP and adaptive step size
    result = minimize(
        objective_func,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'maxiter': 500, 'eps': 1e-6, 'ftol': 1e-8, 'disp': False}
    )
    
    current_h = result.x
    current_c5 = compute_c5(current_h)
    
    # Try a few perturbations to avoid local optima
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(current_h + perturbation, 0, 1)
        perturbed_h = (perturbed_h - np.min(perturbed_h)) / (np.max(perturbed_h) - np.min(perturbed_h)) if np.max(perturbed_h) != np.min(perturbed_h) else perturbed_h
        perturbed_h = perturbed_h * np.sum(current_h) / np.sum(perturbed_h) if np.sum(perturbed_h) != 0 else perturbed_h
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < current_c5:
            current_h = perturbed_h
            current_c5 = perturbed_c5
    
    return (current_h, current_c5, n_points)
