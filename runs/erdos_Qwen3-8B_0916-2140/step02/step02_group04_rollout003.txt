import numpy as np

    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        
        # Use initial_h_values as a starting point
        if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n_points:
            h_values = np.array(initial_h_values, dtype=float)
        else:
            h_values = np.random.rand(n_points)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization with increased iterations and tighter tolerances
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 200, "ftol": 1e-10, "eps": 1e-10},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        current_h = result.x
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n = n_points
    
    return best_h, best_c5, best_n
