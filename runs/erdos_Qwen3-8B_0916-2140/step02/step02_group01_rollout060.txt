import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=100)
    # We will use a refined version of the initial construction by interpolating or extending it
    # For this example, we use a smooth construction with periodicity and symmetry
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=100
    
    # Define intervals
    x = np.linspace(0, 2, n_points)
    
    # Define a piecewise linear function that increases, plateaus, then decreases
    h = np.zeros(n_points)
    
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    integral = np.sum(h) * dx
    h_scaled = h / integral
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem with better constraints and settings
    result = minimize(objective, h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we can improve with a larger grid
    if n_points < 200:
        print("Increasing grid for better resolution...")
        new_n_points = 200
        new_dx = 2.0 / new_n_points
        new_x = np.linspace(0, 2, new_n_points)
        
        # Interpolate the previous solution to new grid
        def interpolate(h_old, x_old, x_new):
            h_new = np.zeros_like(x_new)
            for i in range(len(x_new)):
                xi = x_new[i]
                idx = np.searchsorted(x_old, xi)
                if idx == 0:
                    h_new[i] = h_old[0]
                elif idx == len(x_old):
                    h_new[i] = h_old[-1]
                else:
                    left = h_old[idx-1]
                    right = h_old[idx]
                    t = (xi - x_old[idx-1]) / (x_old[idx] - x_old[idx-1])
                    h_new[i] = left + t * (right - left)
            return h_new
        
        h_interpolated = interpolate(best_h, x, new_x)
        
        # Re-scale to ensure integral is 1
        integral = np.sum(h_interpolated) * new_dx
        h_scaled = h_interpolated / integral
        
        # Define the objective function
        def objective_new(h):
            h = h.reshape(new_n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * new_dx)
            return c5
        
        # Define the constraint to ensure the integral is 1
        def constraint_sum_new(h):
            return np.sum(h) - new_n_points / 2  # sum(h) * dx = 1

        bounds_new = Bounds(0, 1)
        
        # Define the optimization problem with better constraints and settings
        result_new = minimize(objective_new, h_scaled, bounds=bounds_new, constraints={"type": "eq", "fun": constraint_sum_new}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
        
        # Extract the best solution
        best_h_final = result_new.x

        return best_h_final, best_c5_final, new_n_points
