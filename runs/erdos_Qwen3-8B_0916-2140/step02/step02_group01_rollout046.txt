import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points for the function
    n_points = 200
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=200)
    # We'll use a smooth, symmetric function with periodicity to minimize overlap
    # The initial construction is a piecewise linear function with three segments
    
    # Define a simple, symmetric, and piecewise linear function with increased smoothness
    # The function is designed to increase, plateau, and decrease symmetrically
    # This helps in distributing the function values to reduce max overlap
    
    # Convert the grid to actual x positions
    x = np.linspace(0, 2, n_points)
    
    # Compute the piecewise linear function
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    # Compute the initial integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h_val = h.reshape(n_points)
        corr = np.correlate(h_val, 1 - h_val, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
