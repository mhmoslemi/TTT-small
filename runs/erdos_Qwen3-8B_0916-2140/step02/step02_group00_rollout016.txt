import numpy as np
import scipy.optimize as opt
import time
from functools import partial
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Start with a moderate number of points
    dx = 2.0 / n_points

    # Load the initial construction from initial_h_values
    initial_h_values = np.array(evaluate_erdos_solution().h_values)
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h_values, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Start with a local optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Optionally refine with a global optimization strategy
    if budget_s > 300:
        # Use a different starting point to avoid local optima
        initial_h2 = np.zeros(n_points)
        initial_h2[:n_points//3] = 1.0
        initial_h2[n_points//3:n_points//2] = 0.5
        initial_h2[n_points//2:] = 0.0
        initial_h2 = np.clip(initial_h2, 0, 1)

        result2 = minimize(objective, initial_h2, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h2 = result2.x
        best_c5 = compute_overlap(best_h2)

        # Further refinement with another starting point
        initial_h3 = np.zeros(n_points)
        initial_h3[:n_points//4] = 1.0
        initial_h3[n_points//4:n_points//2] = 0.75
        initial_h3[n_points//2:3*n_points//4] = 0.25
        initial_h3[3*n_points//4:] = 0.0
        initial_h3 = np.clip(initial_h3, 0, 1)

        result3 = minimize(objective, initial_h3, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 200})
        best_h3 = result3.x
        best_c5 = compute_overlap(best_h3)

    return (best_h3.tolist(), best_c5, n_points)
