import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import minimize_slsqp
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 200
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We will use a more refined piecewise linear / piecewise constant construction
    
    # Define the initial function based on a bi-linear construction
    # We'll use a piecewise linear function that stays within [0,1]
    # Define intervals [0,1] and [1,2] with different slopes to ensure the integral is 1
    
    # Define a function that increases from 0 to 1, stays flat from 1 to 1.5, then decreases to 0 at 2
    # This construction minimizes peak overlap by balancing the function over the interval
    
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    # Compute the function values
    for i in range(n_points):
        if x[i] <= 1:
            h[i] = x[i]
        elif x[i] <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - x[i]
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure the total integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the max overlap integral
    def objective(h):
        h = h.reshape(n_points)
        # Compute all possible shifted overlaps
        corr = np.correlate(h, 1 - h, mode="full")
        # We are interested in the maximum overlap at all shifts
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem with multiple stages to escape local optima
    
    # Stage 1: Rough optimization using SLSQP to get a good starting point
    result1 = minimize_slsqp(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        iter=50,
        ftol=1e-9,
        iprint=0
    )
    
    # Stage 2: Refine using a more aggressive optimizer with tighter tolerances
    result2 = minimize_slsqp(
        objective,
        result1.x,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        iter=100,
        ftol=1e-10,
        iprint=0
    )
    
    # Stage 3: Final refinement with a small number of iterations to polish the solution
    result3 = minimize_slsqp(
        objective,
        result2.x,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        iter=50,
        ftol=1e-12,
        iprint=0
    )
    
    # Extract the best solution
    best_h = result3.x
    best_c5 = result3.fun
    
    return best_h, best_c5, n_points
