import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.sparse.linalg import lsqr
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid for fine-tuning
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=100)
    # We will use a refined version of the initial construction by interpolating or extending it
    # For this example, we use a smooth construction with periodicity and symmetry
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=100
    initial_h = np.zeros(n_points)
    
    # Define the initial construction using a piecewise linear function
    # This is a symmetric and smooth construction that gradually increases, then decreases
    # The construction is designed to minimize overlaps by distributing the function values evenly
    
    # Define the initial piecewise function
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    # Define a piecewise linear function that increases from 0 to 1, stays flat from 1 to 1.5, then decreases to 0 at 2
    for i in range(n_points):
        if x[i] <= 1:
            h[i] = x[i]
        elif x[i] <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - x[i]
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the best solution
    return best_h, best_c5, n_points
