import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points to balance performance and resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Define initial guess for h: a piecewise constant function that alternates between 0 and 1
    # with a fixed pattern designed to minimize overlap with shifted versions
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points / 3:
            initial_h_values[i] = 1.0
        elif i < 2 * n_points / 3:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.0
    
    # Use interpolation to convert initial_h_values to the desired n_points
    x_initial = np.linspace(0, 2, len(initial_h_values))
    h_initial = initial_h_values
    h_interp = interp1d(x_initial, h_initial, kind='linear', fill_value='extrapolate')
    h_initial_n = h_interp(np.linspace(0, 2, n_points))
    initial_h = np.clip(h_initial_n, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Define an initial guess
    x0 = initial_h.copy()
    
    # Use the SLSQP optimizer with more aggressive settings
    opt_result = minimize(
        objective, 
        x0, 
        bounds=bounds, 
        constraints=cons, 
        method='SLSQP', 
        options={'ftol': 1e-8, 'maxiter': 150}
    )
    
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)
    
    # Perform a multi-start optimization to avoid local minima
    n_starts = 5
    best_c5 = float('inf')
    best_h = best_h
    
    for _ in range(n_starts):
        # Perturb the current best_h
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)
        
        # Perform a local optimization
        result = minimize(
            objective, 
            x0_perturb, 
            bounds=bounds, 
            constraints=cons, 
            method='SLSQP', 
            options={'ftol': 1e-8, 'maxiter': 150}
        )
        
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Optionally refine the solution with a different initial guess
    # Create a new initial guess based on a different pattern
    new_initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            new_initial_h[i] = 1.0
        elif i % 10 == 0:
            new_initial_h[i] = 0.5
        else:
            new_initial_h[i] = 0.0
    
    x0_new = new_initial_h.copy()
    x0_new = np.clip(x0_new, 0, 1)
    
    result = minimize(
        objective, 
        x0_new, 
        bounds=bounds, 
        constraints=cons, 
        method='SLSQP', 
        options={'ftol': 1e-8, 'maxiter': 150}
    )
    
    current_c5 = compute_overlap(result.x)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result.x
    
    # Final refinement with a slightly different initial pattern
    new_initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 7 == 0:
            new_initial_h[i] = 1.0
        elif i % 14 == 0:
            new_initial_h[i] = 0.5
        else:
            new_initial_h[i] = 0.0
    
    x0_new = new_initial_h.copy()
    x0_new = np.clip(x0_new, 0, 1)
    
    result = minimize(
        objective, 
        x0_new, 
        bounds=bounds, 
        constraints=cons, 
        method='SLSQP', 
        options={'ftol': 1e-8, 'maxiter': 150}
    )
    
    current_c5 = compute_overlap(result.x)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
