import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initial guess: use a piecewise-constant function with alternating segments
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 4 == 0 or i % 4 == 1:
            initial_h[i] = 0.5
        elif i % 4 == 2 or i % 4 == 3:
            initial_h[i] = 0.75
    initial_h = np.array(initial_h, dtype=float)

    # Ensure the integral constraint is satisfied
    total = np.sum(initial_h) * dx
    scaling = 1.0 / total
    initial_h *= scaling

    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def constraint(x):
        return np.sum(x) * dx - 1.0

    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization with increased iterations and tighter tolerances
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 200, "ftol": 1e-10, "eps": 1e-10},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )

    best_h = result.x
    best_c_c5 = objective(best_h)

    return best_h, best_c_c5, n_points
