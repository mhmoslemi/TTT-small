from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points for speed and quality balance
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We use a refined version with more points and a better construction
    
    # We will use a piecewise linear function that stays within [0,1]
    # Define the function with more balanced regions to spread the function over [0, 2]
    
    # Define a piecewise linear function that increases, plateaus, and then decreases
    # This helps in distributing the function values to reduce max overlap
    
    # Construct the initial guess
    x = np.linspace(0, 2, n_points)
    h_values = np.zeros(n_points)
    
    # Define a piecewise function with three segments:
    # 1. From 0 to 1: linear increase
    # 2. From 1 to 1.5: plateau
    # 3. From 1.5 to 2: linear decrease
    
    for i in range(n_points):
        x_i = x[i]
        if x_i <= 1:
            h_values[i] = x_i
        elif x_i <= 1.5:
            h_values[i] = 1
        else:
            h_values[i] = 2 - x_i
    
    # Compute the integral of the initial function
    initial_integral = np.sum(h_values) * dx
    
    # Scale the function to ensure the total integral is 1
    h_scaled = h_values / initial_integral
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
