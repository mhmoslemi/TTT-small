import numpy as np

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define a grid with a moderate number of points
    n_points = 500
    dx = 2.0 / n_points
    
    # Initialize with a refined version of the initial construction (interpolated for n=500)
    initial_h_values = np.zeros(n_points)
    
    # Start with a piecewise linear function that increases from 0 to 1, then decreases from 1 to 2
    x = np.linspace(0, 2, n_points)
    
    # Define a function that increases from 0 to 1, then decreases from 1 to 2
    h = np.zeros(n_points)
    for i in range(n_points):
        if x[i] <= 1:
            h[i] = x[i]
        else:
            h[i] = 2 - x[i]
    
    # Calculate the integral of the initial function
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    initial_h_values = h_scaled
    
    # Define the objective function
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint for the integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # We will perform a multi-start optimization to avoid local optima
    num_starts = 10
    best_result = np.inf
    best_h = np.zeros(n_points)
    
    for start_idx in range(num_starts):
        # Create a perturbation to the initial guess
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_guess = initial_h_values + perturbation
        perturbed_guess = np.clip(perturbed_guess, 0, 1)
        
        # Perform optimization
        result = minimize(objective, perturbed_guess, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
        
        # Update best result
        if result.fun < best_result:
            best_result = result.fun
            best_h = result.x
    
    # Return the best solution found
    return best_h, best_result, n_points
