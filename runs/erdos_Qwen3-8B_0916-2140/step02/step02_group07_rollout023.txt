import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to generate a perturbed version of h
    def perturb(h, magnitude=0.05):
        return np.clip(h + np.random.normal(0, magnitude, size=len(h)), 0, 1)
    
    # Function to perform a genetic-like search
    def genetic_search(population_size=20, generations=10, mutation_rate=0.1):
        population = [perturb(h0) for _ in range(population_size)]
        best_solution = min(population, key=objective)
        best_fitness = objective(best_solution)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            
            # Select parents (tournament selection)
            parents = []
            for _ in range(population_size // 2):
                idx1, idx2 = np.random.choice(range(len(population)), size=2, replace=False)
                if fitness[idx1] < fitness[idx2]:
                    parents.append(population[idx1])
                else:
                    parents.append(population[idx2])
            
            # Create offspring
            offspring = []
            for _ in range(population_size // 2):
                parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                child = np.copy(parent1)
                for i in range(len(child)):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.clip(parent2[i] + np.random.normal(0, 0.05), 0, 1)
                offspring.append(child)
            
            # Replace the population
            population = parents + offspring
            population = np.array([np.clip(p, 0, 1) for p in population])
            
            # Update best solution
            current_best = min(population, key=objective)
            current_fitness = objective(current_best)
            if current_fitness < best_fitness:
                best_fitness = current_fitness
                best_solution = current_best
        
        return best_solution, best_fitness
    
    # Main optimization loop
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Initial genetic search
    best_h, best_C5 = genetic_search()
    
    # Local optimization around best solution
    for _ in range(5):
        perturbed_h = perturb(best_h)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
    
    # Final check: if we have time, perform more local optimization
    if time.time() - start_time < time_budget - 10:
        for _ in range(3):
            perturbed_h = perturb(best_h)
            current_h = optimize_step(perturbed_h)
            current_C5 = compute_C5(current_h)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
    
    return best_h, best_C5, len(best_h)
