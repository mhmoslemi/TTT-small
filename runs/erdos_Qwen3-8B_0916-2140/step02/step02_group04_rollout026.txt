import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = initial_h_values

    # Function to convert h_values to a numpy array and scale to satisfy integral constraint
    def create_h_values(h):
        h = np.array(h, dtype=float)
        total = np.sum(h) * dx
        scaling = 1.0 / total
        h *= scaling
        return h

    # Function to compute C_5
    def compute_c5(h_values):
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Initial guess
    h_values = create_h_values(initial_h)

    # Optimization with reduced number of points for faster execution
    def objective(x):
        return compute_c5(x)

    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization with multiple restarts to avoid local minima
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    # Define a custom function for the optimization
    def optimize_with_restarts():
        nonlocal best_c5, best_h
        for restart in range(5):
            # Perturb the initial guess
            perturbed_h = create_h_values(initial_h)
            perturbed_h += np.random.uniform(-0.05, 0.05, size=n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)

            result = minimize(
                objective,
                perturbed_h,
                bounds=bounds,
                constraints={"type": "eq", "fun": lambda x: np.sum(x) * dx - 1.0},
                method="L-BFGS-B",
                options={"maxiter": 200, "ftol": 1e-10, "eps": 1e-10},
            )

            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                print(f"Restart {restart + 1}: Best C5 = {best_c5}")

            # Early stopping if budget is exhausted
            elapsed = time.time() - start_time
            if elapsed >= budget_s:
                print(f"Time budget exceeded, returning best solution so far.")
                return

    start_time = time.time()
    optimize_with_restarts()

    return best_h, best_c5, n_points
