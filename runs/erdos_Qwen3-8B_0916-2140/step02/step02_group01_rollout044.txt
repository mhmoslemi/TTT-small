import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 1000  # Increased to 1000 for better resolution
    dx = 2.0 / n_points

    # Define the initial guess based on the provided initial_h_values (reconstructed for n=1000)
    # We'll use a smooth construction with periodicity and symmetry
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=1000
    initial_h = np.zeros(n_points)
    
    # Define a piecewise linear function that increases, plateaus, then decreases
    # This helps in distributing the function values to reduce max overlap
    
    # Define intervals for the piecewise function
    interval = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Define the piecewise linear function
    h = np.zeros(n_points)
    for i in range(n_points):
        if x[i] <= 1.0:
            h[i] = x[i]
        elif x[i] <= 1.5:
            h[i] = 1.0
        else:
            h[i] = 2.0 - x[i]
    
    # Compute the integral of the piecewise function
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Use the scaled function as the initial guess
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
