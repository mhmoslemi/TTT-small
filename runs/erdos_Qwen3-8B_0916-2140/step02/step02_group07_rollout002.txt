import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Try different starting points to avoid local optima
    start_points = [initial_h]
    for _ in range(3):
        perturbation = np.random.normal(0, 0.1, size=n_points)
        perturbed_h = np.clip(initial_h + perturbation, 0, 1)
        start_points.append(perturbed_h)
    
    best_h = start_points[0]
    best_C5 = compute_C5(best_h)
    
    for start_h in start_points:
        current_h = optimize_step(start_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
    
    # Fine-tuning with different perturbations
    for _ in range(5):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
    
    # Check time budget
    if time.time() - start_time > time_budget:
        print(f"Time budget exceeded. Returning best found solution.")
    
    return best_h, best_C5, len(best_h)
