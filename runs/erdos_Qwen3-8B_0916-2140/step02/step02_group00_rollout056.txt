def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Starting with 200 points
    dx = 2.0 / n_points

    # Load the initial_h_values as a global variable
    initial_h_values = np.concatenate([
        np.linspace(0, 1, 50),
        np.linspace(1, 0, 50),
        np.linspace(0, 1, 50),
        np.linspace(1, 0, 50)
    ])

    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h.astype(np.float64)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Start with a local optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
