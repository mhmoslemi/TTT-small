import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 200
    dx = 2.0 / n_points
    
    # Define the initial guess based on the provided initial_h_values (reconstructed for n=100)
    # We'll use a refined version of the initial construction by interpolating or extending it
    # For this example, we'll use a smooth, symmetric, and piecewise linear function
    
    # Define a piecewise linear function that stays within [0,1]
    # With more points, we can achieve finer adjustments
    
    # Construct a smooth, symmetric, and piecewise linear function
    # Ensure integral is 1 and the function remains in [0,1]
    
    # Define initial guess with more points
    def construct_initial_h(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        
        # Define a piecewise linear function that increases, plateaus, then decreases
        for i in range(n_points):
            xi = x[i]
            if xi <= 1:
                h[i] = xi
            elif xi <= 1.5:
                h[i] = 1
            else:
                h[i] = 2 - xi
        
        # Compute the integral and scale to ensure it equals 1
        integral = np.sum(h) * dx
        h_scaled = h / integral
        
        return h_scaled
    
    initial_h_values = construct_initial_h(n_points)
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h_values, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
