import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a fine grid for better exploration
    n_points = 200
    dx = 2.0 / n_points
    
    # Generate a base construction
    base_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        base_h[i] = 0.5 * (1 - np.cos(2 * np.pi * x / 2))
    
    # Interpolate the base construction to a smaller grid for initial guess
    interpolation_points = np.linspace(0, 2, len(base_h))
    interp_func = interp1d(interpolation_points, base_h, kind='cubic', fill_value="extrapolate")
    initial_h = interp_func(np.linspace(0, 2, n_points))
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization parameters
    max_iter = 100
    tolerance = 1e-6
    n_starts = 5
    
    # Perform a multi-start optimization with different initial points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for _ in range(n_starts):
        # Generate a perturbed initial guess
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        x0 = np.clip(initial_h + perturb, 0, 1)
        
        # Use the L-BFGS-B method for faster convergence
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': tolerance, 'maxiter': max_iter})
        
        # Check if the result is valid
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Optionally perform a few more local optimizations to refine the result
    for _ in range(3):
        perturb = np.random.uniform(-0.01, 0.01, n_points)
        x0 = np.clip(best_h + perturb, 0, 1)
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': tolerance, 'maxiter': max_iter})
        if result.success:
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
