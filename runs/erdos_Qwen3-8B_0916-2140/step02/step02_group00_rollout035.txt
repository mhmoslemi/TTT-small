def run(seed=42, budget_s=600):

    initial_h_values = evaluate_erdos_solution()
    
    # Try starting with different constructions to avoid local optima
    constructions = [initial_h_values]
    for i in range(3):
        # Random perturbation of the initial construction
        perturbed = initial_h_values.copy()
        perturbed += np.random.normal(0, 0.05, len(perturbed))
        perturbed = np.clip(perturbed, 0, 1)
        constructions.append(perturbed)
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for i, construction in enumerate(constructions):
        n = len(construction)
        dx = 2.0 / n
        
        # Convert to step function
        h = np.array(construction, dtype=np.float64)
        h = np.clip(h, 0, 1)
        
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr[1:-1]) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
            best_n = n
    
    # Try higher resolution with best_n
    if best_n < 400 and budget_s > 300:
        n_points = best_n * 2
        dx = 2.0 / n_points
        
        # Interpolate the best construction to higher resolution
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_points)
        h_new = np.interp(x_new, x_old, best_h)
        h_new = np.clip(h_new, 0, 1)
        
        # Define the objective function
        def compute_overlap(h):
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx
        
        # Define the constraint function
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint_sum})
        
        # Use a more robust optimization method
        result = minimize(compute_overlap, h_new, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        
        if result.success:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
            best_n = n_points
    
    return (best_h.tolist(), best_c5, best_n)
