import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 500
    dx = 2.0 / n_points
    
    # Define initial guess based on a more refined construction
    # Start with a piecewise linear function that is symmetric, smooth, and has a peak at the center
    
    # Define the initial guess with a symmetric piecewise linear function
    # From x=0 to x=1: slope = 1
    # From x=1 to x=1.5: flat at 1
    # From x=1.5 to x=2: slope = -1
    # This ensures symmetry and a balance between different regions
    
    # Create a grid of x values
    x = np.linspace(0, 2, n_points)
    
    # Compute the initial function
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h_initial[i] = xi
        elif xi <= 1.5:
            h_initial[i] = 1
        else:
            h_initial[i] = 2 - xi
    
    # Compute the integral of the initial function
    integral_initial = np.sum(h_initial) * dx
    # Scale the function to ensure the total integral is 1
    h_initial_scaled = h_initial / integral_initial
    
    # Define the objective function to minimize the maximum overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Setup the optimization problem
    result = minimize(objective, h_initial_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
