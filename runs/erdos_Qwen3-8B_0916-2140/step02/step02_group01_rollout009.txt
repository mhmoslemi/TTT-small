import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 200
    dx = 2.0 / n_points
    
    # Define a more refined initial guess based on a piecewise linear function with increased smoothness
    # Initially, we use a simple symmetric function that increases to a peak and then decreases
    # We'll refine this by adding more intervals to capture more detailed variations
    
    # Create a grid of points
    x = np.linspace(0, 2, n_points)
    
    # Define an initial piecewise linear function with the following structure:
    # - From 0 to 1: linear increase from 0 to 1
    # - From 1 to 1.5: flat at 1
    # - From 1.5 to 2: linear decrease from 1 to 0
    
    # Compute the initial h(x)
    h = np.zeros(n_points)
    for i in range(n_points):
        if x[i] <= 1:
            h[i] = x[i]
        elif x[i] <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - x[i]
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure the total integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
