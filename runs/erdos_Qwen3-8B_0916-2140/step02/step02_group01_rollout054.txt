import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a moderate grid
    n_points = 200
    dx = 2.0 / n_points
    
    # Define initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We will use a refined version of the initial construction by interpolating or extending it
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # We'll use a more refined piecewise linear construction with increased smoothness
    
    # Define the initial function as a piecewise linear function with three key points:
    # - From x=0 to x=1, h(x) = x
    # - From x=1 to x=1.5, h(x) = 1
    # - From x=1.5 to x=2, h(x) = 2 - x
    # This construction is symmetric and balances the function values
    
    # Create the initial function
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    # Compute the integral to scale
    integral = np.sum(h) * dx
    h_scaled = h / integral
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Set up the optimization problem
    result = minimize(objective, h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
