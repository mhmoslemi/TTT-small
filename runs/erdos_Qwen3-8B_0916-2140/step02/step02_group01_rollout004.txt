import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with a larger grid to capture finer details
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess based on the provided initial_h_values (reconstructed for n=45)
    # We will use a smooth, symmetric, and piecewise linear function to minimize overlap
    # Starting with a refined version of the initial construction by interpolating or extending it
    
    # Construct a smooth, symmetric function with periodicity to minimize overlap
    # Start with a sample construction for n_points=200
    initial_h = np.zeros(n_points)
    
    # Define a piecewise linear function that increases from 0 to 1, stays flat from 1 to 1.5, then decreases to 0 at 2
    # This helps in distributing the function values to reduce max overlap
    
    # Define the function as follows:
    # - From x=0 to x=1, h(x) = x
    # - From x=1 to x=1.5, h(x) = 1
    # - From x=1.5 to x=2, h(x) = 2 - x
    
    # Convert the grid to actual x positions
    x = np.linspace(0, 2, n_points)
    
    # Compute h(x)
    h = np.zeros(n_points)
    for i in range(n_points):
        x_i = x[i]
        if x_i <= 1:
            h[i] = x_i
        elif x_i <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - x_i
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1 (this defines the domain)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
