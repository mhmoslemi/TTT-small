import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Using more points for better resolution
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    
    # Start with a symmetric, bimodal initial guess
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 1.0:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.5 * np.sin(np.pi * (x - 1.0))
    h_values = np.array(initial_h, dtype=float)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with multiple restarts to avoid local optima
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for _ in range(3):  # Try multiple restarts
        # Perturb the initial guess slightly
        h_values = np.clip(initial_h * (1 + np.random.normal(0, 0.05, n_points)), 0, 1)
        h_values = np.array(h_values, dtype=float)
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 200, "ftol": 1e-10, "eps": 1e-10},
            callback=lambda x: None
        )
        
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
