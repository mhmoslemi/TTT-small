import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess based on known construction (n=200)
    n_points = 200
    dx = 2.0 / n_points
    
    # Load initial_h_values from the global variable
    try:
        from evaluate_erdos_solution import initial_h_values
    except:
        initial_h_values = np.random.rand(n_points) * 2 - 1  # Fallback if initial_h_values is not available
    
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization with a local solver
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # If budget allows, perform a grid search on the best_h to explore better solutions
    if budget_s > 300:
        # Grid search parameters
        grid_points = 10
        grid_search = np.linspace(0, 1, grid_points)
        
        # Evaluate the best_h at grid points
        best_h_grid = np.array([best_h[i] * grid_search[j] for i in range(n_points) for j in range(grid_points)]).reshape(n_points, grid_points)
        best_h_grid = best_h_grid.mean(axis=1)
        
        # Refine the solution using a local optimization
        result_grid = minimize(objective, best_h_grid, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h_grid = result_grid.x
        best_c5_grid = compute_overlap(best_h_grid)
        
        # Use the grid-refined solution as the final result
        best_h = best_h_grid
        best_c5 = best_c5_grid
    
    return (best_h.tolist(), best_c5, n_points)
