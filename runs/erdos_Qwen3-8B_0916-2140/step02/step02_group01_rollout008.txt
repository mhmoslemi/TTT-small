    # We will use a smooth construction with periodicity and symmetry

    # We will use a combination of piecewise linear and smooth segments to create a more refined function
    
    # Define the initial function with a smooth transition
    def smooth_transition(x, x0, x1, a, b):
        # Smooth transition between x0 and x1 with values a and b
        t = (x - x0) / (x1 - x0)
        t = np.clip(t, 0, 1)
        return a + (b - a) * (1 - np.exp(-t * 10))
    
    # Define the initial function as a combination of several smooth segments
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    
    # Segment 1: 0 to 0.5, smooth increase from 0 to 0.5
    h[:int(n_points / 4)] = smooth_transition(x[:int(n_points / 4)], 0, 0.5, 0, 0.5)
    
    # Segment 2: 0.5 to 1.0, smooth increase from 0.5 to 1.0
    h[int(n_points / 4):int(n_points / 2)] = smooth_transition(x[int(n_points / 4):int(n_points / 2)], 0.5, 1.0, 0.5, 1.0)
    
    # Segment 3: 1.0 to 1.5, plateau at 1.0
    h[int(n_points / 2):int(3 * n_points / 4)] = 1.0
    
    # Segment 4: 1.5 to 2.0, smooth decrease from 1.0 to 0.0
    h[int(3 * n_points / 4):] = smooth_transition(x[int(3 * n_points / 4):], 1.5, 2.0, 1.0, 0.0)
    
    # Compute the current integral
    integral = np.sum(h) * dx
    # Scale the function to ensure its integral is 1
    h_scaled = h / integral
    
    # Our initial guess is the scaled function
    initial_h = h_scaled
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)

    # Define the constraint to ensure the integral is 1 (this defines the domain)

        return np.sum(h) - n_points / 2  # sum(h) * dx = 1

    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
