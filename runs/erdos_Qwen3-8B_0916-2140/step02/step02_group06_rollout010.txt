def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use a smaller number of points for faster computation
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Interpolate the initial guess to the new grid
    initial_x = np.linspace(0, 2, len(initial_h_values))
    initial_h = np.interp(x, initial_x, initial_h_values)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the objective function
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use SLSQP for optimization
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we've exceeded the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        return (best_h, best_c5, n_points)

    # Generate a few random perturbations to escape local optima

            best_c5 = perturbed_c5
