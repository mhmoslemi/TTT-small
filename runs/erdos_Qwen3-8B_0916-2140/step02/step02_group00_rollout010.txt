import numpy as np
import scipy.optimize as opt
import time
from functools import partial
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Starting with 200 points
    
    # Initial guess based on known construction (n=200)
    initial_h_values = np.load('initial_h_values.npy')
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * (2.0 / n_points)
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization with a local solver
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Second optimization with a different initial guess
    random_h = np.random.rand(n_points)
    random_h = (random_h - 0.5) * 2
    random_h = np.clip(random_h, 0, 1)
    random_h = (random_h * n_points / 2) / np.sum(random_h)
    
    result2 = minimize(objective, random_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 100})
    best_h2 = result2.x
    best_c52 = compute_overlap(best_h2)
    
    # Third optimization with a different initial guess
    random_h3 = np.random.rand(n_points)
    random_h3 = np.clip(random_h3, 0, 1)
    random_h3 = (random_h3 * n_points / 2) / np.sum(random_h3)
    
    result3 = minimize(objective, random_h3, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 100})
    best_h3 = result3.x
    best_c53 = compute_overlap(best_h3)
    
    # Fourth optimization with a different initial guess
    random_h4 = np.random.rand(n_points)
    random_h4 = np.clip(random_h4, 0, 1)
    random_h4 = (random_h4 * n_points / 2) / np.sum(random_h4)
    
    result4 = minimize(objective, random_h4, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 100})
    best_h4 = result4.x
    best_c54 = compute_overlap(best_h4)
    
    # Fifth optimization with a different initial guess
    random_h5 = np.random.rand(n_points)
    random_h5 = np.clip(random_h5, 0, 1)
    random_h5 = (random_h5 * n_points / 2) / np.sum(random_h5)
    
    result5 = minimize(objective, random_h5, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-10, 'maxiter': 100})
    best_h5 = result5.x
    best_c55 = compute_overlap(best_h5)
    
    # Find the best result among all optimizations
    best_result = min([(best_c5, best_h), (best_c52, best_h2), (best_c53, best_h3), (best_c54, best_h4), (best_c55, best_h5)])
    best_c5, best_h = best_result
    return (best_h.tolist(), best_c5, n_points)
