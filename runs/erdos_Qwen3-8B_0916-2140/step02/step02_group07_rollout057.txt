import numpy as np
import scipy.optimize as optimize

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration parameters
    n_points = 100  # Number of samples
    dx = 2.0 / n_points
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    h_initial = np.array(initial_h[:n_points])
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = h_initial
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step with advanced settings
    def optimize_step(h):
        result = minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones_like(x)}],
            method='SLSQP',
            options={
                'ftol': 1e-8,
                'maxiter': 100,
                'eps': 1e-8,
                'disp': False
            }
        )
        return result.x if result.success else h
    
    # Perform multiple rounds of local optimization
    best_h = h0
    best_C5 = compute_C5(h0)
    
    for _ in range(15):
        # Perturb the current solution to avoid local optima
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(h0 + perturbation, 0, 1)
        
        # Optimize from the perturbed point
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            h0 = current_h
            print(f"New best C5: {best_C5}")
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
