import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a high-resolution initial guess
    n_points = 400
    dx = 2.0 / n_points
    initial_h = np.load("initial_h_values.npy")  # Load the known construction

    # Ensure the initial guess is of the correct length
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Start with a local optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Apply a more aggressive optimization by introducing a penalty for large variations
    def penalty(x, penalty_weight=100):
        return np.sum(np.abs(np.gradient(x))) * penalty_weight

    def objective_with_penalty(x):
        return compute_overlap(x) + penalty(x)

    result_with_penalty = minimize(objective_with_penalty, best_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result_with_penalty.x
    best_c5 = compute_overlap(best_h)

    # If budget allows, try a coarser grid to find a better solution by sweeping through different constructions
    if budget_s > 400:
        n_points_sweep = 200
        dx_sweep = 2.0 / n_points_sweep
        best_c5_sweep = np.inf
        best_h_sweep = np.zeros(n_points_sweep)

        # Try a simple piecewise constant function
        h_sweep = np.zeros(n_points_sweep)
        h_sweep[0:100] = 0.5
        h_sweep[100:200] = 0.7
        h_sweep = np.clip(h_sweep, 0, 1)
        h_sweep = h_sweep / np.sum(h_sweep) * n_points_sweep / 2

        c5_sweep = compute_overlap(h_sweep)
        if c5_sweep < best_c5_sweep:
            best_c5_sweep = c5_sweep
            best_h_sweep = h_sweep

        # Try a more complex construction with alternating values
        h_sweep = np.zeros(n_points_sweep)
        h_sweep[::2] = 0.5
        h_sweep[1::2] = 0.7
        h_sweep = np.clip(h_sweep, 0, 1)
        h_sweep = h_sweep / np.sum(h_sweep) * n_points_sweep / 2

        c5_sweep = compute_overlap(h_sweep)
        if c5_sweep < best_c5_sweep:
            best_c5_sweep = c5_sweep
            best_h_sweep = h_sweep

        best_c5 = min(best_c5, best_c5_sweep)
        best_h = best_h_sweep

    return (best_h.tolist(), best_c5, n_points)
