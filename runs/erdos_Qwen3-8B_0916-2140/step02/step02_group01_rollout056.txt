import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess based on existing construction
    # We use a piecewise linear function with a central peak and symmetric design
    # This is a refined version of the initial construction to better minimize overlap
    x = np.linspace(0, 2, n_points)
    
    # Construct a piecewise linear function
    h = np.zeros(n_points)
    for i in range(n_points):
        xi = x[i]
        if xi <= 1:
            h[i] = xi
        elif xi <= 1.5:
            h[i] = 1
        else:
            h[i] = 2 - xi
    
    # Compute the current integral
    integral = np.sum(h) * dx
    h_scaled = h / integral
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
