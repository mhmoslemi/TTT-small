import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial construction from known solution (n=200)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.loadtxt("initial_h_values.txt", delimiter=",")
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)

    # Function to compute overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Objective function
    def objective(x):
        return compute_overlap(x)

    # Constraint: total integral is 1
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Local optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try doubling the points if time allows
    if budget_s > 300:
        n_points_new = 400
        dx_new = 2.0 / n_points_new
        initial_h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), best_h)
        initial_h_new = np.clip(initial_h_new, 0, 1)

        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx_new

        def objective_new(x):
            return compute_overlap_new(x)

        def constraint_sum_new(x):
            return np.sum(x) - n_points_new / 2

        bounds_new = [(0, 1) for _ in range(n_points_new)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})

        result_new = minimize(objective_new, initial_h_new, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        best_h_new = result_new.x
        best_c5 = compute_overlap_new(best_h_new)
        
        # Further scaling if time permits
        if budget_s > 400:
            n_points_3 = 800
            dx_3 = 2.0 / n_points_3
            initial_h_3 = np.interp(np.linspace(0, 2, n_points_3), np.linspace(0, 2, n_points_new), best_h_new)
            initial_h_3 = np.clip(initial_h_3, 0, 1)

            def compute_overlap_3(h):
                h = np.array(h, dtype=np.float64)
                corr = np.correlate(h, 1 - h, mode="full")
                return np.max(corr[1:-1]) * dx_3

            def objective_3(x):
                return compute_overlap_3(x)

            def constraint_sum_3(x):
                return np.sum(x) - n_points_3 / 2

            bounds_3 = [(0, 1) for _ in range(n_points_3)]
            cons_3 = ({'type': 'eq', 'fun': constraint_sum_3})

            result_3 = minimize(objective_3, initial_h_3, bounds=bounds_3, constraints=cons_3, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            best_h_3 = result_3.x
            best_c5 = compute_overlap_3(best_h_3)

    return (best_h_3.tolist() if budget_s > 400 else best_h_new.tolist() if budget_s > 300 else best_h.tolist(), best_c5, n_points_3 if budget_s > 400 else n_points_new if budget_s > 300 else n_points)
