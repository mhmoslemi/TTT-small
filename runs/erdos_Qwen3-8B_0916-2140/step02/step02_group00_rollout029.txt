import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial construction (n=200 points)
    initial_h_values = np.concatenate([
        np.linspace(0, 1, 100),
        np.linspace(1, 0, 100)
    ])
    initial_h_values = np.clip(initial_h_values, 0, 1)
    n_points = 200
    dx = 2.0 / n_points

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Initial optimization with SLSQP
    result = opt.minimize(objective, initial_h_values, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Consider a different initial guess for exploration
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0

    result2 = opt.minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h2 = result2.x
    best_c5_2 = compute_overlap(best_h2)

    # Use the better of the two solutions
    if best_c5 < best_c5_2:
        best_h = best_h
        best_c5 = best_c5
    else:
        best_h = best_h2
        best_c5 = best_c5_2

    # Optional: Try increasing n_points if budget allows
    if budget_s > 300:
        n_points = 400
        dx = 2.0 / n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        initial_h = np.clip(initial_h, 0, 1)

        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx

        def objective_new(x):
            return compute_overlap_new(x)

        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2

        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})

        result_new = opt.minimize(objective_new, initial_h, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h_new = result_new.x
        best_c5_new = compute_overlap_new(best_h_new)

        # Compare to keep the best
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new

    return (best_h.tolist(), best_c5, n_points)
