def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    initial_h = np.array(initial_h_values, dtype=np.float64)

    # Function to compute overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Objective function to minimize
    def objective(x):
        return compute_overlap(x)

    # Constraint: sum of h must be n_points/2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Local optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Try to improve with a global optimization strategy
    # Define a new objective to minimize the maximum overlap
    def objective_max(x):
        h = np.array(x, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Use differential evolution for global search
    result_de = opt.differential_evolution(
        fun=objective_max,
        bounds=bounds,
        constraints=cons,
        strategy='best1bin',
        popsize=30,
        maxiter=50,
        tol=1e-10,
        seed=seed
    )
    best_h_de = result_de.x
    best_c5_de = compute_overlap(best_h_de)
