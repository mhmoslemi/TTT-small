        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define a more refined optimization strategy
    def optimize_with_mixed_methods():
        # First use a Bayesian optimization to explore the space
        from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
        from sklearn.gaussian_process import GaussianProcessRegressor
        from sklearn.gaussian_process.acquisition_functions import expected_improvement
        
        # Initial points
        initial_points = np.random.uniform(0, 1, (n_points,)) * 2.0 / n_points
        initial_points = np.array(initial_points, dtype=np.float64)
        initial_points = initial_points * 2.0 / n_points
        initial_points = np.clip(initial_points, 0, 1)
        
        # Initialize GP
        kernel = C(1.0, (1e-3, 1e3)) * RBF(1.0, (1e-2, 1e2))
        gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=25)
        X = initial_points.reshape(-1, 1)
        y = np.array([objective(x) for x in X])
        
        # Bayesian optimization loop
        n_iter = 50
        for _ in range(n_iter):
            # Predict
            mu, sigma = gp.predict(X, return_std=True)
            # Expected Improvement
            ei = expected_improvement(mu, sigma, y.min())
            # Select next point
            next_idx = np.argmax(ei)
            next_x = X[next_idx]
            # Evaluate
            next_y = objective(next_x)
            # Update GP
            X = np.vstack((X, next_x.reshape(-1, 1)))
            y = np.append(y, next_y)
            gp.fit(X, y)
        
        # Use the best point from Bayesian optimization as initial guess
        best_idx = np.argmin(y)
        best_x = X[best_idx]
        best_y = y[best_idx]
        return best_x, best_y
    
    # Use Bayesian optimization to get a good starting point
    best_start, _ = optimize_with_mixed_methods()
    
    # Perform local optimization using SLSQP
    result = minimize(
        objective,
        best_start,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-10, "eps": 1e-10},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )
    
    best_h = result.x
    best_c_c5 = objective(best_h)
    
    return best_h, best_c_c5, n_points
