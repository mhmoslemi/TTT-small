import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try varying n_points to find best performance
    n_points = 200  # Choose a higher resolution for better accuracy
    dx = 2.0 / n_points
    
    # Initial guess: use a piecewise constant function with 4 segments
    # This is a heuristic to avoid local minima
    len_initial = 4
    initial_h_values = np.array([0.0, 0.5, 1.0, 0.5, 0.0])
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len_initial), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Define a function to perform a single optimization
    def optimize_with_perturbations(x0, max_iter=100, n_perturb=5):
        best_c5 = float('inf')
        best_h = x0.copy()
        
        for _ in range(n_perturb):
            # Perturb the initial guess
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            x0_perturb = np.clip(x0 + perturb, 0, 1)
            
            # Perform optimization
            result = minimize(
                objective,
                x0_perturb,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': max_iter}
            )
            
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        return best_h, best_c5
    
    # Initial optimization
    x0 = initial_h.copy()
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Perform multiple perturbations to avoid local minima
    best_h, best_c5 = optimize_with_perturbations(best_h, max_iter=100, n_perturb=5)
    
    # Optionally perform a multi-start optimization with different initial guesses
    # This is a heuristic to find the global minimum
    for _ in range(3):
        # Generate a new random initial guess
        x0 = np.random.uniform(0, 1, n_points)
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (n_points / 2) / np.sum(x0)
        
        # Perform optimization
        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
