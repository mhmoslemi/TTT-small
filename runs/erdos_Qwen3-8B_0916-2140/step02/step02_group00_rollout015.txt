import numpy as np
import scipy.optimize as opt
import time
from functools import partial
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial construction - starting from known good construction
    initial_h_values = np.load('initial_h.npy')  # Assume this is a precomputed array of 200 samples
    n_points = 200
    dx = 2.0 / n_points
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Start with a local optimization
    initial_h = initial_h_values
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try to refine by introducing some randomness and perturbation to escape local optima
    def perturb_h(h, perturbation_strength=0.05):
        perturbed = h.copy()
        for i in range(len(perturbed)):
            if np.random.rand() < 0.1:
                perturbed[i] += np.random.uniform(-perturbation_strength, perturbation_strength)
                perturbed[i] = np.clip(perturbed[i], 0, 1)
        return perturbed
    
    # Run multiple perturbed optimizations
    results = []
    for _ in range(5):
        perturbed = perturb_h(best_h)
        result = minimize(objective, perturbed, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        results.append((result.fun, result.x))
    
    # Select the best result from the perturbed optimizations
    results.sort()
    best_c5 = results[0][0]
    best_h = results[0][1]
    
    # If time allows, try even higher resolution
    if budget_s > 300:
        n_points = 400
        dx = 2.0 / n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        initial_h = np.clip(initial_h, 0, 1)
        
        def compute_overlap_new(h):
            h = np.array(h, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr[1:-1]) * dx
        
        def objective_new(x):
            return compute_overlap_new(x)
        
        def constraint_sum_new(x):
            return np.sum(x) - n_points / 2
        
        bounds_new = [(0, 1) for _ in range(n_points)]
        cons_new = ({'type': 'eq', 'fun': constraint_sum_new})
        
        # Run optimization on higher resolution
        result_new = minimize(objective_new, initial_h, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h_new = result_new.x
        best_c5_new = compute_overlap_new(best_h_new)
        
        best_c5 = best_c5_new
        best_h = best_h_new
    
    return (best_h.tolist(), best_c5, n_points)
