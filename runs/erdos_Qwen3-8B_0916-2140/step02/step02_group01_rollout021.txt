```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the number of points
    # Start with n_points = 200 to better capture different configurations
    n_points = 200
    dx = 2.0 / n_points
    
    # Define initial guess based on a more refined construction
    # Use a piecewise smooth function with variable slopes to reduce overlap
    
    # Define the initial function as a smooth, symmetric, and periodic function
    # We'll use a function that has multiple segments with different slopes
    # Constructing h as a piecewise linear function with variable slopes
    
    x = np.linspace(0, 2, n_points)
    
    # Define a multi-segment piecewise function
    # Start with a low slope from 0 to 0.5, then increase, then decrease and plateau
    # This aims to reduce overlapping with shifted versions
    
    # Define the function h(x) in segments
    h = np.zeros(n_points)
    
    # Define segments:
    # 1. From x=0 to x=0.5: linear increase
    # 2. From x=0.5 to x=1.0: linear increase with higher slope
    # 3. From x=1.0 to x=1.5: linear plateau
    # 4. From x=1.5 to x=2.0: linear decrease
    
    # Compute h(x) based on the segments
    for i in range(n_points):
        xi = x[i]
        if xi <= 0.5:
            h[i] = (xi / 0.5)
        elif xi <= 1.0:
            h[i] = 1.0
        elif xi <= 1.5:
            h[i] = 1.0
        else:
            h[i] = 2.0 - xi
    
    # Compute the integral of the current function
    integral = np.sum(h) * dx
    # Scale the function to ensure the total integral is 1
    h_scaled = h / integral
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint to ensure the integral is 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # sum(h) * dx = 1
    
    # Set bounds for h (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)
    
    # Define the optimization problem
    result = minimize(objective, h_scaled, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Check time and return the best solution
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return best_h, best_c5, n_points
    
    # Use a different initial guess based on a random perturbation
    initial_guess = np.copy(best_h)
    noise = np.random.normal(0, 0.05, n_points)
    initial_guess += noise
    initial_guess = np.clip(initial_guess, 0, 1)
    # Scale to ensure the integral is 1
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
    
    result = minimize(objective, initial_guess, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500, 'eps': 1e-8})
    
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
```
