import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Define a function to generate perturbations with different patterns
    def generate_perturbations(h, scale=0.05, num_perturbations=3):
        perturbations = []
        for _ in range(num_perturbations):
            perturbation = np.random.normal(0, scale, n_points)
            if np.random.rand() < 0.5:
                perturbation = np.abs(perturbation)
            perturbations.append(perturbation)
        return perturbations
    
    # Define a function to perform local search with different perturbation patterns
    def local_search(h, max_iter=10, scale=0.05):
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        for _ in range(max_iter):
            perturbations = generate_perturbations(h, scale=scale)
            for p in perturbations:
                perturbed_h = np.clip(h + p, 0, 1)
                result = opt.minimize(
                    objective,
                    perturbed_h,
                    bounds=bounds,
                    constraints=cons,
                    method='SLSQP',
                    options={'ftol': 1e-10, 'maxiter': 150}
                )
                if result.success:
                    current_h = result.x
                    current_c5 = compute_overlap(current_h)
                    if current_c5 < best_c5:
                        best_h = current_h
                        best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = local_search(initial_h)
    
    # Refine further with more perturbations and lower scale
    start_time = time.time()
    while time.time() - start_time < budget_s - 100:

        for p in perturbations:

                if current_c5 < best_c5:
