```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial grid points
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Define initial_h_values here if needed
    initial_h = np.array([0.5 for _ in range(n_points)])
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform optimization with adaptive perturbations and adaptive learning rate
    def optimize_with_adaptive_learning(initial_guess, num_perturbations=5, learning_rate=0.01):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        for _ in range(num_perturbations):
            perturbations = np.random.normal(0, learning_rate, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_adaptive_learning(initial_h)
    
    # If time allows, perform further local search with adaptive learning rate
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        learning_rate = 0.01 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, learning_rate, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
```
