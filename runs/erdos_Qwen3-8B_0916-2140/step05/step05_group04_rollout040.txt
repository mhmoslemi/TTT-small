    def compute_overlap(h):

        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

            return compute_overlap(individual)
        
        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [np.clip(ind, 0, 1) for ind in population]
