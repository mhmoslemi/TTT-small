import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define a base grid size
    n_base = 200
    n_points = n_base
    
    # Initial construction from initial_h_values
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * (2.0 / n_points)
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to optimize
    def optimize_with_adaptive_search(initial_guess, num_iterations=50, perturbation_scale=0.15, maxiter=150):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        
        for _ in range(num_iterations):
            # Generate perturbation
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            
            # Optimize
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': maxiter})
            
            # Check if improvement
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                perturbation_scale *= 0.95  # Decrease perturbation scale after improvement
            else:
                perturbation_scale *= 1.05  # Increase perturbation scale if no improvement
        
        return best_h, best_c5
    
    # Start optimization
    best_h, best_c5 = optimize_with_adaptive_search(initial_h)
    
    # Final local search with decreasing perturbation scale
    elapsed_time = 0
    start_time = time.time()
    perturbation_scale = 0.15
    while elapsed_time < budget_s - 100:
        # Generate perturbation
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        
        # Optimize
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        
        # Check if improvement
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
            perturbation_scale *= 0.95  # Decrease perturbation scale after improvement
        else:
            perturbation_scale *= 1.05  # Increase perturbation scale if no improvement
        
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
