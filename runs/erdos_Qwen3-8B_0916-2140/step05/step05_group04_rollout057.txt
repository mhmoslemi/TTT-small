import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with adaptive step size
    def optimize_with_adaptive_steps(initial_guess, num_steps=20, step_decay=0.95, step_min=0.01):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        current_step = 1.0
        for _ in range(num_steps):
            # Perturb the guess
            perturbation = np.random.normal(0, current_step, n_points)
            perturbed_h = np.clip(initial_guess + perturbation, 0, 1)
            # Local optimization
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            current_step *= step_decay
            if current_step < step_min:
                break
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_adaptive_steps(initial_h)

    elapsed_time = 0
