    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to optimize with a different strategy: local search with adaptive step size
    def optimize_with_adaptive_step_size(initial_guess, num_steps=100, step_size=0.1, tolerance=1e-6):

        best_c5 = compute_overlap(initial_guess)

            direction = np.random.randn(n_points)
            direction /= np.linalg.norm(direction)
            perturbation = step_size * direction
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tolerance, 'maxiter': 150})
