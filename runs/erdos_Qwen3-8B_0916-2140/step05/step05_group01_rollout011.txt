    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a single optimization step
    def optimize_step(initial_guess):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x, result.fun
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_overlap(best_h)
    
    # Initial perturbation
    perturbations = np.random.normal(0, 0.02, n_points)
    best_h = np.clip(best_h + perturbations, 0, 1)
    best_c5 = compute_overlap(best_h)
    
    # Main optimization loop with adaptive perturbations
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        # Adaptive perturbation scale
        perturbation_scale = 0.02 * np.exp(-best_c5 / 0.5)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        
        # Perform optimization
        result = optimize_step(perturbed_h)
        current_h, current_c5 = result[0], result[1]
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        
        elapsed_time = time.time() - start_time
