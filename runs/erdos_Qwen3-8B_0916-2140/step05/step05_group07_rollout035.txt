    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    def evaluate_solution(h):
        return compute_overlap(h)
    
    # Function to perform optimization with different strategies
    def optimize_solution(initial_guess):
        best_h = initial_guess
        best_c5 = evaluate_solution(best_h)

        learning_rate = 0.1
        for _ in range(100):
            grad = np.zeros(n_points)

                overlap = np.correlate(best_h, 1 - best_h, mode="full")[k]
                grad += np.gradient(overlap, dx)
