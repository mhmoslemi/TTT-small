import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a single optimization step
    def optimize_step(initial_guess, perturbation_scale=0.1):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x, result.fun
    
    # Function to perform a local search with multiple perturbations
    def local_search(initial_guess, num_steps=20, perturbation_scale=0.1):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_steps):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            current_h, current_c5 = optimize_step(perturbed_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = local_search(initial_h)

    elapsed_time = 0

        perturbations = np.random.normal(0, perturbation_scale, n_points)
