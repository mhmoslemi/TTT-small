    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform localized perturbations
    def localized_perturbations(h, scale=0.05):
        perturb = np.zeros_like(h)
        indices = np.random.choice(range(n_points), size=50, replace=False)
        for idx in indices:
            perturb[idx] = np.random.normal(0, scale)
        return np.clip(h + perturb, 0, 1)
    
    # Function to perform multi-scale optimization
    def multi_scale_optimization(initial_guess, num_scales=5, scale_factor=0.95):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        current_h = initial_guess
        scale = 0.1
        for _ in range(num_scales):
            perturbed_h = localized_perturbations(current_h, scale)
            result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h

    best_h, best_c5 = multi_scale_optimization(initial_h)
