import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial grid points

    # Initial construction from initial_h_values

    if len(initial_h) != n_points:

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local optimization
    def local_optimize(initial_guess, max_iter=150, tol=1e-10):
        result = opt.minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': tol, 'maxiter': max_iter}
        )
        return result.x
    
    # Function to perform adaptive perturbation
    def adaptive_perturb(h, perturbation_scale, num_perturbations):
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        for _ in range(num_perturbations):
            perturb = np.random.normal(0, perturbation_scale, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            result = local_optimize(perturbed)
            current_c5 = compute_overlap(result)
            if current_c5 < best_c5:
                best_h = result
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main optimization process
    best_h = initial_h.copy()
    best_c5 = compute_overlap(best_h)
    
    # Perform initial optimization with perturbations
    perturbation_scale = 0.1
    num_perturbations = 10
    best_h, best_c5 = adaptive_perturb(best_h, perturbation_scale, num_perturbations)
    
    # Additional refinement with decreasing perturbation scale
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale *= 0.95
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = local_optimize(perturbed_h)
        current_c5 = compute_overlap(result)
        if current_c5 < best_c5:
            best_h = result
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
