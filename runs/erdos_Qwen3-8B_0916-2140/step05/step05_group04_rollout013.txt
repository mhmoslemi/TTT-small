        return np.max(corr[1:-1]) * dx

    def objective(x):

            perturbations = np.random.normal(0, step_size, n_points)
            perturbed_h = np.clip(current_h + perturbations, 0, 1)

            current_c5 = compute_overlap(perturbed_h)
