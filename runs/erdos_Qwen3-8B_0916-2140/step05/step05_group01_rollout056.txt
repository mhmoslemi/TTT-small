```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define a function to construct h
    def construct_h(n_points, seed):
        np.random.seed(seed)
        h = np.zeros(n_points)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] = 1
            else:
                h[i] = 0
        return h
    
    # Define an initial construction
    initial_h_values = construct_h(200, seed)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Optimization function
    def optimize_h(initial_guess, n_points, max_iter=100, perturbation_scale=0.05):
        best_h = initial_guess.copy()
        best_c5 = compute_overlap(best_h)
        for _ in range(max_iter):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(best_h + perturbations, 0, 1)
            result = opt.minimize(
                lambda x: compute_overlap(x),
                perturbed_h,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 50}
            )
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Main algorithm
    n_points = 200
    dx = 2.0 / n_points
    best_h = initial_h_values
    best_c5 = compute_overlap(best_h)
    
    # Local search with adaptive perturbation scale
    elapsed_time = 0
    start_time = time.time()
    perturbation_scale = 0.05
    while elapsed_time < budget_s - 100:
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(
            lambda x: compute_overlap(x),
            perturbed_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 50}
        )
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        # Adaptive perturbation scale
        perturbation_scale *= np.exp(-best_c5 / 0.5)
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
```
