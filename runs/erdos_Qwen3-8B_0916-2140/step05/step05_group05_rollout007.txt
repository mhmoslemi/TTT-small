    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    def optimize_with_gradient_descent(initial_guess, num_steps=50, learning_rate=0.01):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)
        for _ in range(num_steps):
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                grad[i] = (compute_overlap(h_plus) - compute_overlap(h_minus)) / (2 * 1e-6)
            h = np.clip(h - learning_rate * grad, 0, 1)
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        return best_h, best_c5

                perturbed_h,
                bounds=bounds,

            )
