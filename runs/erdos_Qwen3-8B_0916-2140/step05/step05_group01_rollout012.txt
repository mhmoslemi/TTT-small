import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial grid points
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial construction from initial_h_values

    if len(initial_h) != n_points:

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a single optimization step
    def optimize_step(initial_guess):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x
    
    # Function to perform a perturbation-based search
    def perturbation_search(initial_guess, num_steps=5, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_steps):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = optimize_step(perturbed_h)
            current_c5 = compute_overlap(result)
            if current_c5 < best_c5:
                best_h = result
                best_c5 = current_c5
        return best_h, best_c5
    
    # Function to perform a grid-based refinement search
    def grid_refinement_search(h, num_refinements=5, num_points_per_refinement=20):
        for _ in range(num_refinements):
            refined_h = np.zeros(n_points)
            for i in range(n_points):
                # Find the nearest grid point in the refined grid
                refined_x = np.linspace(0, 2, n_points + num_points_per_refinement)
                idx = np.argmin(np.abs(refined_x - x[i]))
                refined_h[i] = h[idx]
            h = refined_h
        return h
    
    # Initial optimization
    best_h, best_c5 = perturbation_search(initial_h)
    
    # Local search with adaptive perturbation scaling
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        # Use a smaller perturbation scale as we get closer to the optimal
        perturbation_scale = 0.01 * np.exp(-best_c5 / 0.5)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_scale, n_points), 0, 1)
        result = optimize_step(perturbed_h)
        current_c5 = compute_overlap(result)
        if current_c5 < best_c5:
            best_h = result
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    # Final grid refinement
    best_h = grid_refinement_search(best_h)
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
