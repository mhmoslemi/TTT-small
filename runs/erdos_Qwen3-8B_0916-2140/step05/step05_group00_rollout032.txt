    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):

    def optimize_with_pso(initial_guess, num_particles=50, max_iter=100, inertia=0.9, cognitive=1.2, social=1.2):
        particles = np.random.rand(num_particles, n_points)
        particles = initial_guess + (particles - 0.5) * 2 * (initial_guess.max() - initial_guess.min())
        particles = np.clip(particles, 0, 1)
        velocities = np.random.rand(num_particles, n_points) * 0.01
        best_positions = particles.copy()
        best_fitness = np.array([objective(p) for p in particles])
        global_best_position = best_positions[np.argmin(best_fitness)]
        global_best_fitness = np.min(best_fitness)
        
        for _ in range(max_iter):
            for i in range(num_particles):
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                cognitive_term = cognitive * r1 * (best_positions[i] - particles[i])
                social_term = social * r2 * (global_best_position - particles[i])
                velocities[i] = inertia * velocities[i] + cognitive_term + social_term
                particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
                fitness = objective(particles[i])
                if fitness < best_fitness[i]:
                    best_fitness[i] = fitness
                    best_positions[i] = particles[i]
            global_best_position = best_positions[np.argmin(best_fitness)]
            global_best_fitness = np.min(best_fitness)
        
        return global_best_position, global_best_fitness
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_pso(initial_h)
    
    # If time allows, perform further optimization with adaptive parameters
    elapsed_time = 0

    while elapsed_time < budget_s - 100:
        num_particles = max(10, int(100 * (1 - elapsed_time / budget_s)))

        inertia = 0.9 * (1 - elapsed_time / budget_s)

        result = optimize_with_pso(perturbed_h, num_particles=num_particles, max_iter=max_iter, inertia=inertia, cognitive=cognitive, social=social)
        current_h, current_c5 = result
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
