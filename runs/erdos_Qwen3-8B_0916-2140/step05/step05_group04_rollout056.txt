import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization with different heuristics: using a combination of genetic algorithm and local search
    def genetic_optimize(population_size=50, generations=20, mutation_rate=0.1, crossover_rate=0.8):
        population = [np.clip(np.random.uniform(0, 1, n_points), 0, 1) for _ in range(population_size)]
        population = [ind for ind in population if np.sum(ind) * dx == 1]
        best_h = population[0]
        best_c5 = compute_overlap(best_h)
        
        for generation in range(generations):
            # Evaluate fitness
            fitness = [compute_overlap(ind) for ind in population]
            best_index = np.argmin(fitness)
            best_c5 = fitness[best_index]
            
            # Select parents
            parents = np.random.choice(population, size=population_size, p=fitness / np.sum(fitness))
            
            # Crossover
            new_population = []
            for _ in range(population_size // 2):
                parent1 = parents[np.random.randint(len(parents))]
                parent2 = parents[np.random.randint(len(parents))]
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < crossover_rate:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                new_population.append(child)
            
            # Mutation
            for i in range(len(new_population)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        new_population[i][j] = np.clip(new_population[i][j] + np.random.normal(0, 0.05), 0, 1)
            
            # Replace population
            population = new_population
            population = [np.clip(ind, 0, 1) for ind in population]
            population = [ind for ind in population if np.sum(ind) * dx == 1]
            
            # Update best
            if best_c5 > np.min(fitness):
                best_h = population[np.argmin(fitness)]
                best_c5 = np.min(fitness)
        
        return best_h, best_c5

    best_h, best_c5 = genetic_optimize()

    elapsed_time = 0
    start_time = time.time()
    perturbation_scale = 0.1

        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})

        if current_c5 < best_c5:

            best_c5 = current_c5
        perturbation_scale *= 0.95
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
