import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    def perform_local_search(initial_guess, max_iter=20, perturbation_size=0.015):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        for _ in range(max_iter):
            # Generate multiple perturbations
            perturbations = np.random.normal(0, perturbation_size, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            
            # Perform optimization
            result = opt.minimize(
                objective,
                perturbed_h,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 150}
            )
            
            current_h = result.x
            current_c5 = compute_overlap(current_h)
