    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with adaptive perturbations
    def local_search(h, num_iterations=100, perturbation_scale=0.05):
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        for _ in range(num_iterations):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            h = best_h
            perturbation_scale *= 0.95
        return best_h, best_c5

    best_h, best_c5 = local_search(initial_h)

    alternative_initial = np.zeros(n_points)
    for i in range(n_points):
        alternative_initial[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    alternative_initial = np.clip(alternative_initial, 0, 1)
    
    # Optimize from alternative initial guess
    alt_best_h, alt_best_c5 = local_search(alternative_initial)
    if alt_best_c5 < best_c5:
        best_h, best_c5 = alt_best_h, alt_best_c5
    
    # Consider other structured initial guesses
    structured_initial = np.zeros(n_points)
    for i in range(n_points):

        if current_c5 < best_c5:
