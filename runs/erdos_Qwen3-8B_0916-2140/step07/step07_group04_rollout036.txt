import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(h):
        return compute_overlap(h)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization with different strategies
    def optimize_with_perturbations(initial_guess, num_perturbations=5, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_perturbations):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_perturbations(initial_h)
    
    # Perform gradient descent with adaptive learning rate
    def gradient_descent(h, learning_rate=0.01, max_iter=100, tolerance=1e-6):
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for k in range(1, n_points):
                corr = np.correlate(h, 1 - h, mode="full")
                overlap = corr[k] * dx
                grad -= np.gradient(overlap, h)
            grad = grad / np.linalg.norm(grad)
            h = np.clip(h - learning_rate * grad, 0, 1)
            if np.abs(np.sum(h) - n_points / 2) < tolerance:
                break
        return h
    
    # Perform gradient descent with decreasing learning rate
    learning_rate = 0.1
    for _ in range(10):

        if current_c5 < best_c5:
