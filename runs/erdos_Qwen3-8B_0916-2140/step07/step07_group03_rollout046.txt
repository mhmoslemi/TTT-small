    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to generate a random perturbation
    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, len(h)), 0, 1)
    
    # Function to perform optimization with a given initial guess
    def optimize_guess(guess):
        result = opt.minimize(objective, guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 150})
        return result.x, result.fun
    
    # Function to generate multiple initial guesses with perturbations
    def generate_perturbed_guesses(base_guess, num_guesses, scale):
        guesses = []
        for _ in range(num_guesses):
            perturbed = perturb(base_guess, scale)
            guesses.append(perturbed)
        return guesses
    
    # Main optimization strategy
    def optimize():
        best_h = initial_h.copy()
        best_c5 = compute_overlap(best_h)
        elapsed_time = 0
        start_time = time.time()
        
        # First round: local search with multiple perturbations
        for scale in np.linspace(0.05, 0.001, 5):
            guesses = generate_perturbed_guesses(best_h, 10, scale)
            for guess in guesses:
                h, c5 = optimize_guess(guess)
                if c5 < best_c5:
                    best_h = h
                    best_c5 = c5
        
        # Second round: global search with decreasing scale
        while elapsed_time < budget_s - 100:
            perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)
            guesses = generate_perturbed_guesses(best_h, 5, perturbation_scale)
            for guess in guesses:
                h, c5 = optimize_guess(guess)
                if c5 < best_c5:
