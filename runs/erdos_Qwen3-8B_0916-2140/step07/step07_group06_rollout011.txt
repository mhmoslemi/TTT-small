import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial grid points
    n_points = 150
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial construction from initial_h_values

    if len(initial_h) != n_points:

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization with custom gradient descent and simulated annealing
    def optimize_with_gradient_descent(initial_guess, max_iterations=100, learning_rate=0.1, decay_rate=0.995):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)
        for iter in range(max_iterations):
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                grad[i] = (compute_overlap(h_plus) - compute_overlap(h_minus)) / (2e-6)
            
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
            
            learning_rate *= decay_rate
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_gradient_descent(initial_h)
    
    # Alternative initial guess: alternating pattern
    alternative_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            alternative_initial[i] = 1.0
        elif i % 3 == 1:
            alternative_initial[i] = 0.5
        else:
            alternative_initial[i] = 0.0
    alternative_initial = np.clip(alternative_initial, 0, 1)
    alternative_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(alternative_initial)), alternative_initial)
    
    # Optimize from the alternative initial guess
    alt_best_h, alt_best_c5 = optimize_with_gradient_descent(alternative_initial)
    if alt_best_c5 < best_c5:
        best_h, best_c5 = alt_best_h, alt_best_c5

    def simulated_annealing(initial_guess, max_iterations=100, temp=1.0, temp_decay=0.99):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)

        for _ in range(max_iterations):
            perturbations = np.random.normal(0, 0.05, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            current_c5 = compute_overlap(perturbed_h)
            delta_c5 = current_c5 - best_c5

                h = perturbed_h
                if current_c5 < best_c5:
                    best_h = h
                    best_c5 = current_c5
            current_temp *= temp_decay
        return best_h, best_c5
    
    # Run simulated annealing
    sa_best_h, sa_best_c5 = simulated_annealing(best_h)
    if sa_best_c5 < best_c5:
        best_h, best_c5 = sa_best_h, sa_best_c5
    
    # If time allows, perform further local search with adaptive perturbation scale
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
