    def compute_overlap(h):

        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):

    def constraint_sum(x):

        perturbed_h = np.clip(h + perturbations, 0, 1)
        return perturbed_h
    
    # Function to perform local optimization with adaptive settings
    def local_optimize(h):
        result = opt.minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x
    
    # Strategy: Use multiple starting points with different perturbations and select the best
