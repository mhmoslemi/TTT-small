import scipy.optimize as opt

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with adaptive perturbation
    def local_search(initial_guess, max_iter=50, perturbation_decay=0.95, perturbation_start=0.1):
        best_h = initial_guess.copy()
        best_c5 = compute_overlap(best_h)
        current_h = best_h
        elapsed_time = 0
        start_time = time.time()
        
        for _ in range(max_iter):
            if elapsed_time > budget_s - 100:
                break
            perturbation_scale = perturbation_start * (perturbation_decay ** _)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(current_h + perturbations, 0, 1)
            
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 100})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            
            elapsed_time = time.time() - start_time
        
        return best_h, best_c5

    best_h, best_c5 = local_search(initial_h)

    initial_points = [initial_h]
    for _ in range(3):
        perturbations = np.random.normal(0, 0.1, n_points)
        new_initial = np.clip(initial_h + perturbations, 0, 1)
        initial_points.append(new_initial)
    
    best_h = initial_h
    best_c5 = compute_overlap(initial_h)
