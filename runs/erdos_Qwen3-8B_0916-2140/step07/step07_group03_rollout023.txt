import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points and parameters
    n_points = 150
    dx = 2.0 / n_points

    # Initial construction from initial_h_values

    if len(initial_h) != n_points:

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a single optimization step
    def optimize_step(initial_guess):
        result = opt.minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        return result.x
    
    # Function to perform multiple optimization steps with adaptive perturbation
    def optimize_with_perturbations(initial_guess, num_steps=10, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        for step in range(num_steps):
            perturbation = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            current_h = optimize_step(perturbed_h)
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            perturbation_scale *= 0.95  # exponential decay of perturbation
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_perturbations(initial_h)
    
    # Perform local search with decreasing perturbation scale
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.01 * np.exp(-elapsed_time / 100)
        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_scale, n_points), 0, 1)
        current_h = optimize_step(perturbed_h)
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
