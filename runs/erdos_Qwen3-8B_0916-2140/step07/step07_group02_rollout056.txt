import scipy.optimize as opt

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds for the variables
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a single optimization
    def optimize_single(initial_guess):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x, result.fun
    
    # Function to generate multiple perturbed initial guesses
    def generate_perturbations(initial_guess, num_perturbations=5, perturbation_scale=0.1):
        perturbations = [np.random.normal(0, perturbation_scale, n_points) for _ in range(num_perturbations)]
        return [initial_guess + p for p in perturbations]
    
    # Main optimization loop
    best_h = initial_h
    best_c5 = compute_overlap(initial_h)
    
    # Define hyperparameters
    num_perturbations = 5
    num_iterations = 30
    perturbation_decay = 0.95
    initial_perturbation_scale = 0.1

    perturbation_scale = initial_perturbation_scale
    
    # Perform multiple iterations of perturbation and optimization
    for _ in range(num_iterations):

        perturbed_guesses = generate_perturbations(best_h, num_perturbations, perturbation_scale)
        # Evaluate all perturbed guesses
        current_best_h = best_h
        current_best_c5 = best_c5
        for guess in perturbed_guesses:
            # Clip the guess to [0, 1]
            guess = np.clip(guess, 0, 1)
            # Perform optimization
            result = optimize_single(guess)
            h_candidate, c5_candidate = result, compute_overlap(result)
            # Update best solution
            if c5_candidate < current_best_c5:
