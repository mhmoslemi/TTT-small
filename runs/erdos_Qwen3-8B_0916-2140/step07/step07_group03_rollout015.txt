import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a base grid size
    n_points = 150
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial grid construction based on the initial_h_values
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with decreasing perturbation
    def local_search(initial_guess, max_iter=100, perturbation_scale_start=0.05, perturbation_decay=0.95):
        best_h = initial_guess.copy()
        best_c5 = compute_overlap(best_h)
        elapsed_time = 0
        start_time = time.time()
        
        for _ in range(max_iter):
            perturbation_scale = perturbation_scale_start * (perturbation_decay ** _)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(best_h + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s - 100:
                break
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = local_search(initial_h)
    
    # Try with a different starting point: random binary construction
    def random_binary_initial(n_points):
        h = np.random.rand(n_points)
        h = np.cumsum(h) / np.sum(h) * n_points / 2
        return h
    
    best_h_binary, best_c5_binary = local_search(random_binary_initial(n_points))
    if best_c5_binary < best_c5:
        best_h, best_c5 = best_h_binary, best_c5_binary

        return h
    
    best_h_piecewise, best_c5_piecewise = local_search(piecewise_constant_initial(n_points))
    if best_c5_piecewise < best_c5:
        best_h, best_c5 = best_h_piecewise, best_c5_piecewise
    
    # Final local search with randomized perturbations
    def final_local_search(best_guess, max_perturbations=100, perturbation_scale_start=0.05, perturbation_decay=0.95):
        best_h = best_guess.copy()
        best_c5 = compute_overlap(best_h)
        elapsed_time = 0
        start_time = time.time()
        
        for _ in range(max_perturbations):
            perturbation_scale = perturbation_scale_start * (perturbation_decay ** _)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(best_h + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s - 100:
                break
        return best_h, best_c5
    
    best_h, best_c5 = final_local_search(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
