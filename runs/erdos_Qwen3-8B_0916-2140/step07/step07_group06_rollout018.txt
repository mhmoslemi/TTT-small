import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 150
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial grid points
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization function
    def optimize(initial_guess):
        result = opt.minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        return result.x, result.fun
    
    # --- Alternative Initial Construction ---
    # Alternative 1: Periodic 3-step pattern
    alt1_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            alt1_h[i] = 1.0
        elif i % 3 == 1:
            alt1_h[i] = 0.5
        else:
            alt1_h[i] = 0.0
    alt1_h = np.clip(alt1_h, 0, 1)
    
    # Alternative 2: Randomized 5-step pattern
    alt2_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            alt2_h[i] = 1.0
        elif i % 5 == 1:
            alt2_h[i] = 0.75
        elif i % 5 == 2:
            alt2_h[i] = 0.5
        elif i % 5 == 3:
            alt2_h[i] = 0.25
        else:
            alt2_h[i] = 0.0
    alt2_h = np.clip(alt2_h, 0, 1)
    
    # Alternative 3: Segment-based pattern
    alt3_h = np.zeros(n_points)
    segment_length = n_points // 5
    for i in range(5):
        alt3_h[i * segment_length : (i + 1) * segment_length] = 1.0 if i % 2 == 0 else 0.0
    alt3_h = np.clip(alt3_h, 0, 1)
    
    # Alternative 4: Randomized Sparse pattern
    alt4_h = np.zeros(n_points)
    for i in range(n_points):
        if np.random.rand() < 0.3:
            alt4_h[i] = 1.0
        elif np.random.rand() < 0.5:
            alt4_h[i] = 0.5
    alt4_h = np.clip(alt4_h, 0, 1)
    
    # Run optimization from each alternative
    best_h, best_c5 = initial_h, compute_overlap(initial_h)
    alt1_h, alt1_c5 = optimize(alt1_h)
    if alt1_c5 < best_c5:
        best_h, best_c5 = alt1_h, alt1_c5
    
    alt2_h, alt2_c5 = optimize(alt2_h)
    if alt2_c5 < best_c5:
        best_h, best_c5 = alt2_h, alt2_c5
    
    alt3_h, alt3_c5 = optimize(alt3_h)
    if alt3_c5 < best_c5:
        best_h, best_c5 = alt3_h, alt3_c5
    
    alt4_h, alt4_c5 = optimize(alt4_h)
    if alt4_c5 < best_c5:
        best_h, best_c5 = alt4_h, alt4_c5
    
    # Adaptive perturbation with decreasing scale
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
