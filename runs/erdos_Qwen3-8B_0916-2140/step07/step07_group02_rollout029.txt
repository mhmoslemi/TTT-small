        return np.max(corr[1:-1]) * dx

    def objective(x):

    def constraint_sum(x):

    def perturb(h, scale):
        perturbed = np.clip(h + np.random.normal(0, scale, h.shape), 0, 1)
        return perturbed
    
    # Define a function to perform a multi-start optimization with adaptive perturbation
    def multi_start_optimize(initial_guess, num_starts=5, maxiter=150, scale_start=0.05, scale_end=0.01):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)

        scale = scale_start

        for _ in range(num_starts):
            perturbed = perturb(initial_guess, scale)
            result = opt.minimize(objective, perturbed, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': maxiter})
