    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local optimization
    def optimize_local(h_initial, num_iterations=10, perturbation_scale=0.1):
        best_h = h_initial
        best_c5 = compute_overlap(best_h)
        for _ in range(num_iterations):
            # Create a perturbed version of the initial guess
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h_initial + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            perturbation_scale *= 0.9
        return best_h, best_c5
    
    # Perform an initial optimization
    best_h, best_c5 = optimize_local(initial_h)
    
    # Local search with fine-tuning
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        # Fine-tune with smaller perturbations
        perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)

        perturbed_h = np.clip(best_h + perturbations, 0, 1)

        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
