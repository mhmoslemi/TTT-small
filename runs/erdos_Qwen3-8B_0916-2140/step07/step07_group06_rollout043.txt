import scipy.optimize as opt
import time

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a local optimization from a starting guess
    def local_optimize(starting_guess, max_iter=150, tol=1e-10):
        result = opt.minimize(objective, starting_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tol, 'maxiter': max_iter})
        return result.x, result.fun
    
    # Function to generate a random perturbation of a guess
    def perturb_guess(guess, scale=0.05):
        perturbation = np.random.normal(0, scale, len(guess))
        perturbed = np.clip(guess + perturbation, 0, 1)
        return perturbed
    
    # Function to run a local search with multiple starting points
    def multi_start_search(initial_guess, num_starts=5, perturbation_scale=0.1):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_starts):
            perturbed = perturb_guess(initial_guess, perturbation_scale)
            current_h, current_c5 = local_optimize(perturbed)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with the initial guess
    best_h, best_c5 = multi_start_search(initial_h)
    
    # Generate a few more diverse initial guesses to explore different regions
    for _ in range(3):

        offset = np.random.uniform(0, 2)

        periodic_guess = np.clip(periodic_guess, 0, 1)
        periodic_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(periodic_guess)), periodic_guess)
