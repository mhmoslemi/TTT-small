    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Function to optimize from a given starting point
    def optimize_from_start(initial_guess, method='SLSQP'):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method=method, options={'ftol': 1e-10, 'maxiter': 150})
        return result.x, result.fun
    
    # Perturbation function
    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, len(h)), 0, 1)
    
    # Adaptive optimization with multiple starting points
    def adaptive_optimize(initial_guess, num_perturbations=5, perturbation_scales=[0.05, 0.02, 0.01, 0.005, 0.002]):

        best_c5 = compute_overlap(initial_guess)

            perturbed = perturb(initial_guess, scale)

            if c5 < best_c5:
                best_h = h
                best_c5 = c5
