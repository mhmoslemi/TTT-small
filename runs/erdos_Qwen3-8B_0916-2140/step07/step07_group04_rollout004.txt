import scipy.optimize as opt
import time

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform optimization with adaptive perturbations and cooling schedule
    def optimize_adaptive(initial_guess, max_iter=100, cooling_rate=0.99, perturbation_scale_start=0.1, perturbation_scale_end=0.01):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        current_h = initial_guess
        current_c5 = best_c5
        perturbation_scale = perturbation_scale_start
        for iter in range(max_iter):
            # Generate perturbation
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(current_h + perturbations, 0, 1)
            # Optimize with constraints
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 100})
            candidate_h = result.x
            candidate_c5 = compute_overlap(candidate_h)
            if candidate_c5 < current_c5:
                current_h = candidate_h
                current_c5 = candidate_c5
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            # Update perturbation scale
            perturbation_scale *= cooling_rate
        return best_h, best_c5
    
    # Perform adaptive optimization starting from initial guess
    best_h, best_c5 = optimize_adaptive(initial_h)
    
    # Refine further with local search
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        # Use a smaller perturbation scale for local refinement
        perturbations = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 100})

        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
