```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
import time
import random
import os

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # We'll start with 100 points and adjust if needed

    # Initial guess
    initial_h = initial_h_values
    initial_h = np.array(initial_h)
    initial_h = initial_h[:n_points] if n_points < len(initial_h) else initial_h[-n_points:]
    initial_h = np.interp(initial_h, (initial_h.min(), initial_h.max()), (0, 1))
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

    # Function to compute C5
    def compute_C5(h):
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        C5 = np.max(corr * dx)
        return C5

    # Function to optimize h with constraints
    def optimize_h(h0):
        nonlocal best_c5, best_h
        h = h0.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_C5(h)

    # Function to perturb h
    def perturb_h(h, perturbation_factor=0.05):
        h = h.copy()
        h += np.random.normal(0, perturbation_factor, h.shape)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Optimization loop
    for _ in range(100):
        current_h = perturb_h(initial_h)
        current_c5 = compute_C5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            print(f"New best C5: {best_c5}")
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 10:
            break

    print(f"Best C5 found: {best_c5}")
    return best_h, best_c5, n_points
```
