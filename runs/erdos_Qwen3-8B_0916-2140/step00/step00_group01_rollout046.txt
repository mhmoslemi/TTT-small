import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
from scipy.integrate import simps
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values():
    return [0.5] * 47  # Placeholder for the initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Start time tracking
    start_time = time.time()

    # Initial construction with 47 samples
    n_initial = 47
    h_initial = initial_h_values()
    dx_initial = 2.0 / n_initial

    # Compute initial C5 value
    c5_initial = evaluate_erdos_solution(h_initial)
    print(f"Initial C5 value: {c5_initial:.6f}")

    # Define a function to compute the overlap integral for a given h
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    # Define a function to compute the integral for all shifts
    def compute_all_correlations(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        correlation = np.correlate(h, 1 - h, mode="full")
        return correlation * dx

    # Define a function to compute the constraints
    def compute_constraints(h_values):
        n_points = len(h_values)
        total = np.sum(h_values) * (2.0 / n_points)
        return total - 1.0

    # Define the optimization problem as an optimization with constraints
    def optimize_func(h_values):
        c5 = compute_c5(h_values)
        return c5

    def compute_objective(h_values):
        return optimize_func(h_values)

    def compute_constraints_func(h_values):
        return compute_constraints(h_values)

    # Allow some flexibility to refine the initial solution
    for n_points in [47, 60, 80, 100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300, 320, 340, 360, 380, 400]:
        if time.time() - start_time > budget_s - 5:
            break

        print(f"Trying with n_points = {n_points}")

        # Initialize with the initial solution
        h = np.array(initial_h_values())
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)

        # Define bounds for each sample
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Define the objective function
        def objective(x):
            return compute_c5(x)

        # Define constraints
        def constraint(x):
            return np.sum(x) * (2.0 / n_points) - 1.0

        # Solve the constrained optimization
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 200}
        )

        if result.success:
            h_values = result.x
            c5 = result.fun
            print(f"New C5 value: {c5:.6f}")
            return h_values, c5, n_points

    # Return the best solution found
    print("No better solution found")
    return initial_h_values(), c5_initial, n_initial
