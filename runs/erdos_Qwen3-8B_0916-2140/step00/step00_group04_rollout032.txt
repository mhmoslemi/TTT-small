        nonlocal best_c5, best_h, elapsed
        for _ in range(100):
            if elapsed > budget_s:
                return
            h_perturbed = perturb(h)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5 = compute_overlap(h_perturbed)
            if c5 < best_c5:
