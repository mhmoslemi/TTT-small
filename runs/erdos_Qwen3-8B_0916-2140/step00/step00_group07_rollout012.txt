```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions from the environment
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Initial starting point
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h_values = initial_h
    c5 = evaluate_erdos_solution(h_values)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_values
        best_n_points = n_points

    # Try increasing n_points up to a maximum of 400
    for n_points in [100, 200, 300, 400]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate a random starting point
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Map to [0, 1]
        h = h * (2.0 / n_points)  # Ensure integral is 1
        h = h / np.sum(h) * n_points / 2  # Scale to make integral exactly 1

        # Optimization: minimize max overlap
        def objective(x):
            h = x.reshape(n_points)
            dx = 2.0 / n_points
            conv = signal.correlate(h, 1 - h, mode='full')
            overlap = np.max(conv) * dx
            return overlap

        def constraint(x):
            return np.sum(x) - n_points / 2

        # Use scipy.optimize to find a better solution
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points, constraints=[{'type': 'eq', 'fun': constraint, 'jac': lambda x: np.ones(n_points)}])
        if result.success:
            h_opt = result.x
            c5_val = result.fun
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt
                best_n_points = n_points

    # Return the best solution found
    return (best_h, best_c5, best_n_points)
```
