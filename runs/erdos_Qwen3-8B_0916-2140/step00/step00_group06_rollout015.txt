import numpy as np
import scipy.optimize as optimize
import time
import pandas as pd

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    def evaluate_erdos_solution(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values, dtype=np.float64)
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def objective(x, n_points, dx):
        h = x
        conv = np.correlate(h, 1.0 - h, mode="full")
        return np.max(conv) * dx
    
    def constraints(x, n_points, dx):
        return np.sum(x) - n_points / 2
    
    def grad_objective(x, n_points, dx):
        h = x
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_index = np.argmax(conv)
        grad = np.zeros_like(h)
        for i in range(n_points):
            for j in range(n_points):
                if i - j >= 0:
                    grad[i] -= conv[i - j]
                if i + j < n_points:
                    grad[i] -= conv[i + j]
        return grad * dx
    
    def grad_constraints(x, n_points):
        return np.ones_like(x)
    
    def make_initial_guess(n_points):
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                initial_h[i] = 1.0
            else:
                initial_h[i] = 0.0
        return initial_h
    
    def optimize_h(n_points, initial_h, max_iter=100, tol=1e-5):
        x0 = initial_h.copy()
        dx = 2.0 / n_points
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = optimize.minimize(
            fun=lambda x: objective(x, n_points, dx),
            x0=x0,
            bounds=bounds,
            method='L-BFGS-B',
            jac=lambda x: grad_objective(x, n_points, dx),
            constraints=optimize.BoundsConstraint(
                lb=np.zeros(n_points),
                ub=np.ones(n_points),
            ),
            options={'max_iter': max_iter, 'ftol': tol}
        )
        
        if result.success:
            return result.x
        else:
            return x0
    
    def evaluate_sequence(h, n_points, dx):
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def find_best_solution(n_points, seed_offset=0):
        best_h = None
        best_c5 = float('inf')
        
        for seed in range(seed_offset, seed_offset + 10):
            np.random.seed(seed)
            initial_h = make_initial_guess(n_points)
            h = optimize_h(n_points, initial_h)
            c5 = evaluate_sequence(h, n_points, 2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        return best_h, best_c5
    
    def run_optimization(n_points, initial_h, max_iterations=100, seed=42):
        h = initial_h.copy()
        best_h = h
        best_c5 = evaluate_sequence(h, n_points, 2.0 / n_points)
        
        for _ in range(max_iterations):
            new_h = optimize_h(n_points, h)
            new_c5 = evaluate_sequence(new_h, n_points, 2.0 / n_points)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                h = new_h
        
        return best_h, best_c5
    
    start_time = time.time()
    
    n_points = 1000
    initial_h = make_initial_guess(n_points)
    h, c5 = run_optimization(n_points, initial_h)
    
    if time.time() - start_time > budget_s - 10:
        return h.tolist(), c5, n_points
    
    return h.tolist(), c5, n_points
