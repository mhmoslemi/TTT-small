```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-4
    best_c5 = float('inf')
    best_h = None

    # Start with initial guess
    h = initial_h_values.copy()
    h = np.array(h[:n_points])
    h = np.round(h / np.max(h) * 1.0, 5)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Print progress
        if _ % 10 == 0:
            elapsed = time.time() - start_time
            remaining = max(0, budget_s - elapsed)
            print(f"Iteration {_}, Best C5: {best_c5:.6f}, Time elapsed: {elapsed:.2f}s, Remaining: {remaining:.2f}s")
        
        # Try perturbation for optimization
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Evaluate new solution
        corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        c5_perturbed = np.max(corr_perturbed) * dx
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        
        # Early stopping if improvement is minimal
        if best_c5 - c5 < improvement_threshold:
            break
    
    # Final evaluation
    evaluate_erdos_solution(best_h)
    return best_h, best_c5, n_points
```
