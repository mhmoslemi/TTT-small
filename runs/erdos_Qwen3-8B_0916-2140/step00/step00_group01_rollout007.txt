```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

# Pre-imported functions
from initial_h_values import initial_h_values
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 500
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Objective and constraints
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h = h_initial.copy()
    
    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[
            {'type': 'eq', 'fun': constraint_sum}
        ],
        options={'disp': False, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = result.fun
    best_n_points = n_points
    
    # Check if we can do better with fewer points
    for n in range(200, n_points, -1):
        dx = 2.0 / n
        h = np.linspace(0, 2, n)
        h = np.interp(h, np.linspace(0, 2, n_points), best_h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        if np.abs(sum_h - 1) < 1e-6:
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            best_c5 = c5
            best_n_points = n
            best_h = h
            break
    
    # Final check with best_n_points
    dx = 2.0 / best_n_points
    h = np.linspace(0, 2, best_n_points)
    h = np.interp(h, np.linspace(0, 2, n_points), best_h)
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h) * dx
    if np.abs(sum_h - 1) > 1e-6:
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        while np.abs(sum_h - 1) > 1e-6:
            h = np.clip(h, 0, 1)
            scale = 1 / sum_h
            h = h * scale
            sum_h = np.sum(h) * dx
    
    corr = np.correlate(h, 1 - h, mode="full")
    best_c5 = np.max(corr * dx)
    
    return (h, best_c5, best_n_points)
```
