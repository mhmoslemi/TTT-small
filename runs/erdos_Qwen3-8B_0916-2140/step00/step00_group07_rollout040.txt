        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def objective(h):
        return compute_c5(h)

    def optimize_h(h0):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h0
    
    # Start from initial solution
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    
    # Try several perturbations around the initial solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbation = np.clip(perturbation, -0.5, 0.5)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = optimize_h(perturbed_h)
