import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    while time.time() - start_time < budget_s - 10:
        # Start with initial_h_values or a random candidate
        h = initial_h_values.copy()
        if len(h) != n_points:
            h = np.random.rand(n_points)
        
        # Normalize to satisfy the integral constraint
        h = h / np.sum(h) * n_points / 2
        
        # Evaluate the current C5
        c5 = evaluate_erdos_solution(h)
        
        # Use a local search to improve the solution
        for _ in range(100):
            # Randomly select a segment to flip
            start_idx = np.random.randint(0, n_points - 10)
            end_idx = start_idx + 10
            h[start_idx:end_idx] = 1 - h[start_idx:end_idx]
            
            # Normalize again
            h = h / np.sum(h) * n_points / 2
            
            # Evaluate the new C5
            new_c5 = evaluate_erdos_solution(h)
            if new_c5 < c5:
                c5 = new_c5
                h = h.copy()
        
        # Store the best solution
        best_h = h
        best_c5 = c5
        
        # Decrease the number of points to speed up
        n_points = max(n_points - 10, 50)
    
    # Final evaluation
    final_h = best_h
    final_c5 = evaluate_erdos_solution(final_h)
    
    # Return the result
    return (final_h, final_c5, len(final_h))
