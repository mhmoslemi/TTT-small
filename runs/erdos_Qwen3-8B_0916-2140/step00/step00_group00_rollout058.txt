            total = np.sum(initial_guess) * (2.0 / n_points)

            initial_guess *= scale
            initial_guess = np.clip(initial_guess, 0.0, 1.0)
        
        # Optimize using a local solver
        result = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 100}
        )
        best_h = result.x
        best_c5 = result.fun
        return best_h, best_c5
    
    # Start with the initial solution
    best_h = h_initial
    best_c5 = evaluate_solution(best_h)
    best_n_points = len(best_h)
    
    # Try reducing n_points from the initial solution
    for n_points in [45, 30, 20, 10]:
        # Generate random initial guess satisfying the integral constraint
        initial_guess = np.random.rand(n_points)
        total = np.sum(initial_guess) * (2.0 / n_points)
        scale = (1.0 / total) if total != 0 else 1.0
        initial_guess *= scale
        initial_guess = np.clip(initial_guess, 0.0, 1.0)
        
        # Optimize
        start_time = time.time()
