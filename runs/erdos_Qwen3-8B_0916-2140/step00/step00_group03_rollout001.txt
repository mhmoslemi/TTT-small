```python
import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Starting with 100 points for speed
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values[:n_points])  # Use initial construction
    
    def objective(h):
        h = np.clip(h, 0, 1)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    def eval_termination(h):
        return time.time() - start_time > budget_s
    
    # Optimizer settings
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'disp': False, 'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # If we still have time, try a refined search
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.clip(best_h, 0, 1)
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        
        def refined_objective(h):
            h = np.clip(h, 0, 1)
            correlation = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(correlation) * dx
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            refined_objective,
            x0=np.clip(best_h, 0, 1),
            method='L-BFGS-B',
            bounds=bounds,
            options={'disp': False, 'ftol': 1e-8, 'maxiter': 50}
        )
        
        best_h = result.x
        best_c5 = refined_objective(best_h)
    
    # Ensure the integral constraint is met
    best_h = np.clip(best_h, 0, 1)
    integral = np.sum(best_h) * dx
    if abs(integral - 1) > 1e-6:
        best_h = np.clip(best_h, 0, 1)
        best_h = (best_h / integral) * 2
    
    final_c5 = objective(best_h)
    
    # Return the result
    return (best_h, final_c5, len(best_h))
```
