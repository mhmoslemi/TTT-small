import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import copy
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial parameters
    n_points = 100  # Start with 100 sample points
    max_iter = 500
    improvement_threshold = 1e-5
    time_budget = budget_s
    start_time = time.time()

    # Initial guess
    h = initial_h_values.copy()
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h_values = h.copy()
    best_c5 = evaluate_erdos_solution(h_values)

    def evaluate(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        shift = np.arange(-n_points + 1, n_points)
        overlap = np.zeros(len(shift))
        for k in shift:
            if k < 0:
                overlap[i] = np.sum(h[:n_points + k] * (1 - h[k:]))
            elif k > 0:
                overlap[i] = np.sum(h[:n_points - k] * (1 - h[k:]))
        return np.max(overlap)

    def optimize_h(h_start):
        h = h_start.copy()
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        bounds = [(0, 1) for _ in range(len(h))]
        result = optimize.minimize(lambda x: evaluate(x), h_start, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
        return result.x

    # Try adaptive refinement
    for _ in range(5):
        time_remaining = time_budget - (time.time() - start_time)
        if time_remaining <= 0:
            break

        # Try with more points
        n_points *= 2
        h_current = initial_h_values.copy()
        h_current = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_current)), h_current)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current.reshape(-1)

        # Optimize
        h_optimized = optimize_h(h_current)
        h_optimized = np.clip(h_optimized, 0, 1)
        c5 = evaluate(h_optimized)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_optimized.copy()

        # Check if we're improving
        if best_c5 - 0.3808 > 1e-5:
            continue
        else:
            break

    # Try further refinement
    n_points = 200
    h_current = initial_h_values.copy()
    h_current = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_current)), h_current)
    h_current = np.clip(h_current, 0, 1)
    h_current = h_current.reshape(-1)

    # Fine-tune
    h_optimized = optimize_h(h_current)
    h_optimized = np.clip(h_optimized, 0, 1)
    c5 = evaluate(h_optimized)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_optimized.copy()

    return best_h, best_c5, n_points
