import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import diags
from scipy.sparse.linalg import lgmres

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    
    # Initial guess based on equally spaced intervals
    for i in range(n_points):
        if i < n_points / 2:
            h_initial[i] = 1
        else:
            h_initial[i] = 0
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if sum_h != n_points / 2:
        h_initial = h_initial / sum_h * (n_points / 2)
    
    def compute_overlap(h):
        h = h / np.sum(h) * (n_points / 2)
        overlap = np.correlate(h, 1 - h, mode='full')
        return np.max(overlap) * dx
    
    def objective(h):
        return compute_overlap(h)
    
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    def constraint_func(h):
        return constraints(h)
    
    # Use a local optimizer with bounds
    bounds = Bounds(0, 1)
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Further refinement using a gradient-based method
    for _ in range(3):
        h_refined = best_h.copy()
        result = minimize(objective, h_refined, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-8, 'maxiter': 50})
        h_refined = result.x
        best_c5 = compute_overlap(h_refined)
        best_h = h_refined
    
    # Evaluate the best solution
    best_c5 = compute_overlap(best_h)
    return best_h, best_c5, n_points
