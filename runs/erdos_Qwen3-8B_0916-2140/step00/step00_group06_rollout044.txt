```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import LinearConstraint, Bounds
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial setup
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    h_initial = np.array(initial_h_values, dtype=np.float64)
    
    # Start with the initial solution
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Determine the resolution for the current run
    n_points = max(100, n_points_initial - 100)
    dx = 2.0 / n_points
    
    # Generate a random initial guess
    x = np.linspace(0, 2, n_points)
    h = np.random.uniform(0, 1, n_points)
    # Normalize h to satisfy the integral constraint
    h = h / np.sum(h) * (n_points / 2)
    
    # Define the objective function to minimize the maximum overlap integral
    def objective(h_vec):
        h = h_vec.copy()
        # Compute the overlap integral for each k
        max_overlap = 0.0
        for k in range(1, n_points):
            # Compute shifted h
            shifted_h = np.roll(h, k)
            # Compute the integrand: h(x) * (1 - h(x + k))
            integrand = h * (1 - shifted_h)
            # Integrate using the grid spacing dx
            overlap = np.sum(integrand) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Define the constraint: sum(h) * dx == 1
    constraint_sum = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    # Define bounds: 0 <= h[i] <= 1
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    # Define a solver with good settings for this problem
    solver_options = {
        'maxiter': 1000,
        'disp': False,
        'ftol': 1e-8
    }
    
    # Use the initial guess
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=[constraint_sum], options=solver_options)
    
    # Check if the result is valid
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = h
        best_c5 = objective(h)
    
    # Final evaluation
    final_c5 = best_c5
    final_h = best_h
    final_n_points = n_points
    
    return (final_h, final_c5, final_n_points)
```
