import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    target_sum = n_points / 2

    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Function to generate a new random h with sum(h) = target_sum
    def generate_random_h(n_points, target_sum):
        h = np.random.uniform(0, 1, n_points)
        h -= np.min(h)
        h /= np.max(h)
        h = h * (target_sum / np.sum(h)) + np.random.uniform(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        return h

    # Function to perform a local search starting from a given h
    def local_search(h, max_iter=100, learning_rate=0.01):
        h = np.copy(h)
        best_h = np.copy(h)
        best_c5 = evaluate_c5(h)
        for _ in range(max_iter):
            perturbation = np.random.normal(0, learning_rate, n_points)
            h = np.clip(h + perturbation, 0, 1)
            current_sum = np.sum(h)
            h = h * (target_sum / current_sum) if current_sum > 0 else h
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = current_c5
        return best_h, best_c5

    # Function to find the best solution using a combination of random restarts and local search
    def find_best_solution(n_restarts=5, max_local_iter=100):
        best_h = initial_h_values
        best_c5 = evaluate_c5(initial_h_values)
        for _ in range(n_restarts):
            h = generate_random_h(n_points, target_sum)
            h, current_c5 = local_search(h, max_local_iter)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        return best_h, best_c5

    # Check if we are within the time budget
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time < 10:
        return initial_h_values, evaluate_c5(initial_h_values), n_points

    # Use a more sophisticated approach: simulated annealing with a larger n_points
    n_points = 200
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        return evaluate_c5(h)

    def simulate_annealing(max_iter=1000, temp=1.0, cooling_rate=0.99):
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
        best_h = np.copy(h)
        best_c5 = evaluate_c5(h)
        for _ in range(max_iter):
            if temp < 0.001:
                break
            perturbation = np.random.normal(0, temp, n_points)
            new_h = np.clip(h + perturbation, 0, 1)
            new_sum = np.sum(new_h)
            new_h = new_h * (target_sum / new_sum) if new_sum > 0 else new_h
            current_c5 = evaluate_c5(new_h)
            if current_c5 < best_c5:
                best_h = np.copy(new_h)
                best_c5 = current_c5
            else:
                prob = np.exp((current_c5 - best_c5) / temp)
                if np.random.rand() < prob:
                    h = np.copy(new_h)
            temp *= cooling_rate
        return best_h, best_c5

    # Run simulated annealing
    best_h_sa, best_c5_sa = simulate_annealing()
    if best_c5_sa < best_c5:
        best_h = best_h_sa
        best_c5 = best_c5_sa

    # Use a smaller n_points for a faster evaluation
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Generate a new h with the optimal n_points
    best_h = np.random.uniform(0, 1, n_points)
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) > 0 else best_h
    best_c5 = evaluate_c5(best_h)

    # Get the best solution so far
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_points
