import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 200  # We'll use 200 points for balance between resolution and speed

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)

    def objective(h):
        nonlocal best_c5
        # Normalize h to ensure integral is 1
        h = h / np.sum(h)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Use a simple gradient descent approach with random restarts
    for _ in range(10):
        # Random perturbation to avoid local minima
        h = initial_h.copy()
        h += np.random.normal(0, 0.05, h.shape)
        h = np.clip(h, 0, 1)
        # Normalize to ensure integral is 1
        h = h / np.sum(h)

        # Simple gradient descent for optimization
        for step in range(200):
            # Compute current C5
            corr = signal.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * (2.0 / n_points)

            # Compute gradients for each point
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Compute contribution of h[i] to each correlation
                for k in range(n_points):
                    # For each shift k, compute how h[i] contributes to the correlation
                    # This is an approximation and may need refinement
                    if i - k >= 0 and i - k < n_points:
                        grad[i] += (1 - h[i - k]) * (2.0 / n_points)
                    if i + k < n_points:
                        grad[i] += (1 - h[i + k]) * (2.0 / n_points)

            # Update h with gradient descent step
            h += -0.01 * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h)

            # Early stopping if we're close to the best
            if current_c5 < best_c5 - 1e-6:
                break

        # Update best solution
        corr = signal.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * (2.0 / n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Check if we've used up the budget
        if time.time() - start_time > budget_s - 10:
            break

    # Final evaluation
    final_corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * (2.0 / n_points)
    return best_h, final_c5, n_points
