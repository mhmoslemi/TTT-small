import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time
from scipy.optimize import minimize
from scipy.sparse.linalg import lsqr

def run(seed=42, budget_s=600):
    # We use a small n_points for efficiency
    n_points = 200
    dx = 2.0 / n_points
    # Initial guess based on evenly spaced samples
    initial_h = np.zeros(n_points)
    # Split the interval [0, 2] into equal parts
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.5

    # Define the overlap integral function
    def compute_overlap(h):
        overlap = np.zeros(n_points)
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            overlap[k] = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)

    result = minimize(compute_overlap, initial_h, method='L-BFGS-B', bounds=(0, 1), options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
