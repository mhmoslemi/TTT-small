```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assume this is pre-loaded with initial construction

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the initial number of points and sample
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        return evaluate_erdos_solution(h)
    
    # Function to adjust h to satisfy the integral constraint
    def adjust_h(h):
        total = np.sum(h)
        scaling = (n_points_initial / 2.0) / total
        return h * scaling
    
    # Optimization function
    def objective(h):
        h = adjust_h(h)
        return compute_overlap(h)
    
    # Set initial guess
    h_guess = h_initial.copy()
    
    # Run L-BFGS optimization
    result = optimize.minimize(
        fun=objective,
        x0=h_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points_initial)],
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Extract best solution
    h_best = result.x
    c5_best = compute_overlap(h_best)
    
    # We can try a finer grid if the budget allows
    if budget_s > 300:
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 0.5 + 0.25  # Initial guess
        h = adjust_h(h)
        
        # Define the complete objective for the new grid
        def compute_overlap_new(h):
            h = adjust_h(h)
            return evaluate_erdos_solution(h)
        
        # Run another round of optimization on the new grid
        result = optimize.minimize(
            fun=compute_overlap_new,
            x0=h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        h_best = result.x
        c5_best = compute_overlap_new(h_best)
        n_points = len(h_best)
    
    return h_best, c5_best, len(h_best)
```
