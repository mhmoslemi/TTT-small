```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Preloaded initial construction

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Precompute all shifts
    shifts = np.unique(np.random.choice(n_points, size=100, replace=False)) % n_points

    # Optimization strategy:
    # 1. Iterative refinement of the current h
    # 2. Try perturbing the h values with small random changes
    # 3. Use gradient descent on the overlap integral

    for _ in range(300):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5:
            break

        # Try perturbation
        perturb = np.random.choice([0.05, -0.05, 0.1, -0.1], size=n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)

        # Ensure the integral condition is preserved
        scale = (n_points / 2) / np.sum(h)
        h *= scale

        # Evaluate the overlap integral
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Optional: Perform gradient descent on the current h
        # Simplified gradient descent approach
        for _ in range(10):
            # Evaluate all overlaps
            overlaps = np.zeros(n_points)
            for k in shifts:
                if k == 0:
                    continue
                shifted = np.roll(h, k)
                overlaps += np.abs(np.correlate(h, 1 - shifted, mode="full")) * (2.0 / n_points)

            # Gradient descent step
            grad = np.gradient(overlaps)
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h *= scale

            # Evaluate
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return best_h, best_c5, n_points
```
