import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.roll(h, shift=1, axis=0)
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return np.load("initial_h.npy")

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function: Minimize the maximum overlap
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraints: sum(h) == n_points / 2
    def constraint_func(h_flat):
        return np.sum(h_flat) - n_points / 2
    
    # Initial guess
    initial_guess = initial_h.copy()
    
    # Optimization setup
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_func}],
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = result.fun
    elapsed_time = time.time() - start_time
    
    print(f"Optimization complete. Best C5: {best_c5}, Time: {elapsed_time:.2f} seconds")
    
    return (best_h.tolist(), best_c5, n_points)
