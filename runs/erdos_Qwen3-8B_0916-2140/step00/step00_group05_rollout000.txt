        def objective(h_flat):
            h = h_flat.reshape(n)

            shifted = np.roll(h, shift=1, axis=0)
            overlap = np.correlate(h, 1 - shifted, mode="full")

        def constraint(h_flat):
            h = h_flat.reshape(n)

        bounds = Bounds(0, 1)

        result = minimize(objective, h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="L-BFGS-B", options={"maxiter": 1000})
