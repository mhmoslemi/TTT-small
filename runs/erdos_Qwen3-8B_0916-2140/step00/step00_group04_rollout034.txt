import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import scipy.sparse.linalg as splinalg
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25,
                                 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25,
                                 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25,
                                 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25, 0.25])
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    current_c5 = np.max(np.correlate(initial_h_values, 1 - initial_h_values, mode="full")) * dx
    best_h = initial_h_values.copy()
    best_c5 = current_c5
    best_n_points = n_points
    
    if n_points < 1000:
        try:
            from scipy.optimize import Bounds
            from scipy.optimize import minimize
            from scipy.optimize import TNC
            
            def objective(h):
                return np.max(np.correlate(h, 1 - h, mode="full")) * dx
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            def constraint_bounds(h):
                return np.all((0 <= h) & (h <= 1))
            
            bounds = Bounds([0.0] * n_points, [1.0] * n_points)
            
            def evaluate(h):
                if not constraint_bounds(h):
                    return np.inf
                return objective(h)
            
            result = minimize(evaluate, best_h, method='TNC', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
            
            if result.success:
                best_h = result.x
                best_c5 = result.fun
                best_n_points = n_points
                print(f"Found improved h with C₅ = {best_c5}")
            else:
                print("Optimization failed")
        except Exception as e:
            print(f"Optimization error: {e}")
    
    return best_h, best_c5, best_n_points
