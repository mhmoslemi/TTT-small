import numpy as np
import scipy.optimize as optimize

# Pre-imported functions
def evaluate_erdos_solution(h):
    """Evaluates the Erdős minimum overlap problem constant C₅."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def initial_h_values():
    """Returns an initial guess for the function h."""
    # Sample initial construction from literature or heuristic
    n_points = 95
    x = np.linspace(0, 2, n_points, endpoint=False)
    h = np.zeros(n_points)
    h[:n_points//3] = 1.0
    h[n_points//3 : 2*n_points//3] = 0.5
    h[2*n_points//3 :] = 0.0
    # Normalize to ensure ∫h dx = 1
    total = np.sum(h) * (2.0 / n_points)
    h /= total
    return h

def run(seed=42, budget_s=600):
    start_time = time.time()
    warnings.filterwarnings("ignore", category=RuntimeWarning)
    
    # Initial setup and parameters
    n_points = 100  # Selected as a balance between accuracy and speed
    dx = 2.0 / n_points
    n = n_points
    initial_h = initial_h_values()
    
    # Define the objective function
    def objective(h_flat):
        h = h_flat.reshape(n)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Define constraints
    def constraints(h_flat):
        h = h_flat.reshape(n)
        return np.sum(h) - (n / 2)
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n)]
    
    # Define the optimization problem
    def optimize_solution(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraints},
            options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        )
        return result.x
    
    # Run the optimization
    initial_guess = initial_h
    best_h = initial_guess
    best_c5 = evaluate_erdos_solution(initial_guess)
    
    print(f"Initial C₅: {best_c5}")
    print(f"Initial h: {initial_guess}")
    
    # Additional optimization attempts
    for _ in range(5):
        initial_guess = np.random.uniform(0, 1, n)
        # Ensure the integral constraint is roughly satisfied
        initial_guess = initial_guess / np.sum(initial_guess) * (n / 2)
        result = optimize_solution(initial_guess)
        current_c5 = evaluate_erdos_solution(result)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
            print(f"Found better C₅: {best_c5}")
    
    # Final check
    print(f"Best C₅ found: {best_c5}")
    print(f"Final h: {best_h}")
    
    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning best solution found.")
    
    return (best_h.tolist(), best_c5, n)
