```python
import numpy as np
import time
import scipy.optimize as opt
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
from scipy.sparse import diags
from scipy.sparse.linalg import lsqr
import multiprocessing as mp
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the number of points for the grid
    n_points = 100  # A balance between accuracy and speed
    x = np.linspace(0, 2, n_points)
    
    # Initial guess: copy initial_h_values and scale to fit the current n_points
    if len(initial_h_values) != n_points:
        # Interpolate initial_h_values to match n_points
        initial_h = np.interp(x, np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        initial_h = initial_h_values.copy()
    
    # Define the objective function: evaluate the maximum overlap integral
    def objective(h):
        h = np.clip(h, 0, 1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Define the constraints: sum(h) == n_points / 2
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess for optimization
    initial_guess = initial_h
    
    # Use L-BFGS-B for optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Check if the result is within the budget time
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded. Returning best solution found so far.")
        return result["x"], result["fun"], n_points
    
    # If optimization succeeded, return the best solution
    return result["x"], result["fun"], n_points
```
