import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial parameters
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=np.float64)
    total_area = np.sum(h) * dx
    if np.abs(total_area - 1.0) > 1e-8:
        h = h / total_area * 1.0

    def compute_C5(h):
        n = len(h)
        dx = 2.0 / n
        h_correlate = np.correlate(h, 1 - h, mode="full")
        overlap = h_correlate * dx
        return np.max(overlap)

    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()

    # Define a function to update the solution based on a small perturbation
    def update_solution(h):
        nonlocal best_C5, best_h
        perturb = np.random.normal(0, 0.01, size=len(h))
        new_h = np.clip(h + perturb, 0, 1)
        new_h = new_h / np.sum(new_h) * 1.0 * (2.0 / len(new_h))  # Ensure area is 1
        new_C5 = compute_C5(new_h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h.copy()

    # Main optimization loop
    for _ in range(1000):
        if time.time() - start_time > budget_s:
            break
        update_solution(best_h)

    # Final result
    return best_h, best_C5, len(best_h)
