import scipy.optimize as optimize

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        max_overlap = 0.0
        for k in range(1, n - 1):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Define the objective function for optimization
    def objective_function(h_values):
        h = np.array(h_values)
        if np.any(h < 0) or np.any(h > 1):
            return np.inf
        return compute_c5(h)
    
    # Define the constraint: sum(h) * dx = 1
    def constraint_sum(h_values):
        h = np.array(h_values)
        return np.sum(h) - n_points / 2
    
    # Function to create a random perturbation of h
    def perturb_h(h, perturbation_factor=0.1):
        perturbed = h.copy()
        for i in range(len(h)):
            perturbed[i] += np.random.uniform(-perturbation_factor, perturbation_factor)
            perturbed[i] = np.clip(perturbed[i], 0.0, 1.0)
        return perturbed
    
    # Function to generate a random initial guess
    def generate_random_initial():
        h = np.random.uniform(0, 1, n_points)
        h = h - np.min(h)
        h = h / np.max(h)
        return h
    
    # Initialize with the given initial_h_values
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    # Optimization parameters
    max_iterations = 100
    random_perturbations = 10
    local_optimization_attempts = 3
    
    elapsed_time = 0.0
    
    for _ in range(max_iterations):
        # Try random perturbations
        for _ in range(random_perturbations):
            perturbed = perturb_h(best_h)
            current_c5 = compute_c5(perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed
                elapsed_time += 0.1
                if elapsed_time > budget_s:
                    return best_h, best_c5, n_points
    
        # Try local optimization
        for _ in range(local_optimization_attempts):
            # Create a small perturbation around best_h
            perturbed = perturb_h(best_h, perturbation_factor=0.05)
            # Use scipy.optimize to minimize the objective
            bounds = [(0, 1) for _ in range(n_points)]
            result = optimize.minimize(
                objective_function,
                perturbed,
                bounds=bounds,
                method='L-BFGS-B'
            )
            
            if result.success:
                h_candidate = result.x
                current_c5 = compute_c5(h_candidate)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_candidate
                    elapsed_time += 0.5
                    if elapsed_time > budget_s:
                        return best_h, best_c5, n_points
    
        # If no improvement, try a new random initial guess
        new_h = generate_random_initial()
        current_c5 = compute_c5(new_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h
            elapsed_time += 0.5
            if elapsed_time > budget_s:
                return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
