import numpy as np
import scipy.optimize as opt

import matplotlib.pyplot as plt

# Pre-imported functions

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility

    # Precompute necessary values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    h = np.array(initial_h)
    
    # Define the overlap integral function
    def compute_c5(h_values):
        h = np.array(h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        
        # Compute the max value
        return np.max(corr)
    
    # Function to compute the gradient (we use finite differences for now)
    def compute_gradient(h_values):
        h = np.array(h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        
        # Compute the gradient of the max function
        # Find the index where the maximum occurs
        max_index = np.argmax(corr)
        # Compute the gradient for that index
        grad = np.zeros(n_points)
        
        # Compute the derivative of the correlation at the index
        # We assume h is in [0, 1], and the derivative is with respect to h[i]
        # The overlap integral is max over k of integral h(x)(1 - h(x+k)) dx
        # For a given k, the overlap is sum_{i} h[i] * (1 - h[(i - k) % n_points]) * dx
        # So the derivative with respect to h[i] is:
        # For each k, if the maximum is at k, then the contribution to the gradient is
        # sum_{k} [ (1 - h[(i - k) % n_points]) * dx * (indicator that the max is at k) ]
        # We approximate this by finding the k where the max occurs and then computing the derivative
        
        # Find the k where the max occurs
        # The max occurs at index max_index in the full convolution
        # The corresponding k is max_index - n_points // 2
        k = max_index - n_points // 2
        
        # Compute the impact of each h[i] on the overlap for that k
        for i in range(n_points):
            # For this k, the overlap is integral h(x)(1 - h(x+k)) dx
            # The contribution of h[i] to this overlap is
            # h[i] * (1 - h[(i - k) % n_points]) * dx
            # So the derivative of the overlap with respect to h[i] is
            # (1 - h[(i - k) % n_points]) * dx
            # But since we are maximizing the overlap, the gradient is negative of this
            # So the gradient for h[i] is - (1 - h[(i - k) % n_points]) * dx
            
            # Compute the position in the shifted array
            shifted_index = (i - k) % n_points
            grad[i] = - (1 - h[shifted_index]) * dx
        
        return grad
    
    # Define the objective function (to minimize the max overlap)
    def objective(h_values):
        return compute_c5(h_values)
    
    # Define the gradient function (for optimization)
    def gradient(h_values):
        return compute_gradient(h_values)
    
    # Define the constraint: sum(h) * dx == 1
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Set up the optimization problem
    # We use a simple gradient descent with momentum
    # We'll start from the initial h_values and optimize
    # We'll try different starting points to avoid local optima
    
    # Define a list of starting points (we'll try a few different perturbations)
    starting_points = []

    starting_points.append(np.array([0.1 for _ in range(n_points)]))
    starting_points.append(np.array([1.0 for _ in range(n_points)]))
    starting_points.append(np.array([0.0 for _ in range(n_points)]))
    
    best_c5 = float('inf')
    best_h = initial_h
    
    for start in starting_points:
        # Copy the starting point
        h = np.array(start)
        # Ensure the sum is correct
        h = h / np.sum(h) * (n_points / 2)
        
        # Parameters for optimization
        learning_rate = 0.1

            current_c5 = compute_c5(h)
            
            # Update the best value
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Compute the gradient
            grad = compute_gradient(h)
            
            # Update h
            h = h - learning_rate * grad
            
            # Enforce bounds [0, 1]
            h = np.clip(h, 0, 1)
            
            # Enforce sum constraint
            h = h / np.sum(h) * (n_points / 2)
            
            # Check for convergence
            if iteration % 100 == 0:
                print(f"Iteration {iteration}, current C5: {current_c5}")
        
        # Store the best result
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
    
    # Return the best result
    return (best_h.tolist(), best_c5, len(best_h))
