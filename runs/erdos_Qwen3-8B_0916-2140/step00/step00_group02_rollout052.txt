def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000  # Start with 1000 points for better precision
    dx = 2.0 / n_points
    initial_h = initial_h_values
    h = np.array(initial_h, dtype=np.float64)
    h = h[:n_points]
    h = h / np.sum(h) * (n_points / 2)  # Ensure sum(h) * dx = 1
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2
    cons = ({'type': 'eq', 'fun': constraint_sum})
    result = minimize(objective, h, method='L-BFGS-B', jac=None, constraints=cons, options={'maxiter': 100})
    h_best = result.x
    c5 = objective(h_best)
    return h_best, c5, n_points
