```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.linalg import norm

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Load initial guess from file

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration parameters
    n_points = 100  # Start with 100 points for faster evaluation
    max_iterations = 100
    tolerance = 1e-4
    
    # Initial guess from preloaded data
    h = initial_h_values[:n_points]
    if len(h) != n_points:
        h = np.concatenate((h[:n_points//2], np.zeros(n_points//2)))  # Trim or pad as needed
    
    # Ensure the integral constraint is satisfied
    integral = np.sum(h) * (2.0 / n_points)
    if abs(integral - 1.0) > 1e-6:
        h = np.clip(h, 0, 1)
        h = h * (1.0 / integral)  # Normalize
    
    # Objective function
    def objective(x):
        nonlocal h
        h = x
        integral = np.sum(h) * (2.0 / n_points)
        if abs(integral - 1.0) > 1e-6:
            return np.inf  # Penalize constraint violation
        
        # Evaluate C5
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Optimization bounds
    bounds = Bounds(0, 1)
    
    # Optimization settings
    result = minimize(
        objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iterations, 'ftol': tolerance, 'gtol': tolerance}
    )
    
    # Refine with local search
    for _ in range(3):
        if result.fun < 1e-6:
            break
        result = minimize(
            objective,
            x0=result.x,
            method='SLSQP',
            bounds=bounds,
            options={'maxiter': max_iterations // 2, 'ftol': tolerance, 'eps': 1e-8}
        )
    
    best_h = result.x
    c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution
    return best_h, c5, n_points
```
