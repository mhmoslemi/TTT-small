# Pre-imported functions

def initial_h_values():
    return [0.5] * 45  # Placeholder for initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Initial sample with 45 points
    n_points = 45
    dx = 2.0 / n_points
    h_initial = initial_h_values()
    h_initial = np.array(h_initial, dtype=np.float64)
    
    # Function to compute C5 for a given h
    def compute_C5(h):
        h = np.array(h, dtype=np.float64)
        overlaps = []
        for k in range(1, n_points // 2 + 1):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1.0 - shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)

    # Optimizer parameters
    max_iter = 100
    tol = 1e-5
    best_C5 = float('inf')
    best_h = h_initial.copy()

    # Try multiple starting points
    for _ in range(5):
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
        perturbation = np.clip(perturbation, -0.5, 0.5)
        h = h_initial + perturbation
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * n_points / 2.0  # Ensure sum(h) * dx = 1

        # Run optimization
        h_opt = opt.minimize(
            lambda x: compute_C5(x),
            x0=h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 50, 'ftol': 1e-5, 'gtol': 1e-5}
        ).x

        # Evaluate the result
        current_C5 = compute_C5(h_opt)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_opt

        # Check time
        elapsed = time.time() - start_time
        if elapsed > max_time:
            break

    # Refine the best solution with a local search
    h_refined = best_h.copy()
    for _ in range(20):
        # Generate perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbation = np.clip(perturbation, -0.2, 0.2)
        h = h_refined + perturbation
        h = np.clip(h, 0.0, 1.0)
        h = h / np.sum(h) * n_points / 2.0  # Ensure sum(h) * dx = 1

        # Evaluate
        current_C5 = compute_C5(h)

            best_C5 = current_C5

        # Check time
