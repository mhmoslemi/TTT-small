import numpy as np
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy import sparse
from scipy.sparse import linalg as spla
import time

# Predefined initial construction (n=45)
initial_h_values = np.array([0.6, 0.7, 0.8, 0.9, 1.0, 0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05,
                             0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05,
                             0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05, 0.05])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the initial number of points
    n_points = 45
    if n_points < 100:
        n_points = 100
    else:
        n_points = 200
    
    # Ensure the initial h_values are of size n_points
    if len(initial_h_values) < n_points:
        initial_h_values = np.pad(initial_h_values, (0, n_points - len(initial_h_values)), 'constant')
    elif len(initial_h_values) > n_points:
        initial_h_values = initial_h_values[:n_points]
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        dx = 2.0 / n_points
        c5 = 0.0
        for k in range(1, 5):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            c5 = max(c5, overlap)
        return c5
    
    # Objective function for optimization
    def objective(h):
        return evaluate_overlap(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h_values.copy()
    
    # Optimization
    start_time = time.time()
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints={"type": "eq", "fun": constraint_sum})
    
    # Extract the best solution and compute the overlap integral
    h_opt = result.x
    c5_bounds = evaluate_overlap(h_opt)
    
    # Return the result
    return (h_opt, c5_bounds, n_points)
