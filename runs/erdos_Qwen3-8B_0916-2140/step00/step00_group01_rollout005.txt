import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    
    # If initial_h_values is provided, use it as a starting point
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = np.interp(initial_h_values, [0, 1], [0, 1])
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = initial_h_values[:n_points]
    
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = np.interp(h, [0, 1], [0, 1])
        h = h[:n_points]
        h = np.concatenate([h, h[:n_points-1]])
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap[overlap.size // 2 : ] * dx
        return np.max(overlap)
    
    def objective(h):
        return evaluate_c5(h)
    
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)
    
    result = opt.minimize(objective, initial_h_values, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
    
    if result.success:
        best_h = result.x
        c5 = evaluate_c5(best_h)
        return (best_h, c5, n_points)
    else:
        return (initial_h_values, evaluate_c5(initial_h_values), n_points)
