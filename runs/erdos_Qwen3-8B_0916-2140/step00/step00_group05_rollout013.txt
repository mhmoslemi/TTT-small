import numpy as np
import scipy.optimize as opt
import time
from functools import lru_cache

def initial_h_values(n_points):
    return np.array([0.5] * n_points)

def evaluate_erdos_solution(h_values, n_points):
    h = h_values
    h_shifted = np.roll(h, shift=1)
    overlap = np.sum(h * (1 - h_shifted))
    return overlap

def run(seed=42, budget_s=600):
    n_points = 200
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    best_c5 = float('inf')
    best_h = initial_h

    def objective(h):
        nonlocal best_c5
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        overlap = []
        for k in range(1, n_points):
            shifted = np.roll(h, shift=k)
            overlap_val = np.sum(h * (1 - shifted)) * dx
            overlap.append(overlap_val)
        c5 = max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    start_time = time.time()
    while time.time() - start_time < budget_s:
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        result = opt.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            best_h = result.x
            best_c5 = result.fun
        else:
            best_h = initial_h

    return best_h, best_c5, n_points
