```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 1000  # Start with a moderate number of points

    # Initial guess based on the pre-defined initial_h_values
    initial_h = initial_h_values
    if len(initial_h) < n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    h = initial_h.copy()

    def objective(h):
        nonlocal best_c5
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Optimization using downhill simplex
    result = opt.minimize(
        objective,
        x0=h,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8},
        bounds=[(0, 1) for _ in range(n_points)]
    )

    # Refinement using finer grid
    if time.time() - start_time < budget_s * 0.7:
        n_points = 2000
        h_refined = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(result.x)), result.x)
        h_refined = np.clip(h_refined, 0, 1)
        dx = 2.0 / n_points
        corr = np.correlate(h_refined, 1 - h_refined, mode="full")
        c5 = np.max(corr) * dx
        best_c5 = c5
        best_h = h_refined

    # Final check with current best
    if best_h is None:
        best_h = result.x

    # Ensure the solution satisfies the integral constraint
    integral = np.sum(best_h) * (2.0 / len(best_h))
    if abs(integral - 1.0) > 1e-6:
        best_h = np.clip(best_h, 0, 1)
        integral = np.sum(best_h) * (2.0 / len(best_h))
        if abs(integral - 1.0) > 1e-6:
            # Adjust to meet the constraint
            scale = 1.0 / integral
            best_h = best_h * scale
            best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5, len(best_h)
```
