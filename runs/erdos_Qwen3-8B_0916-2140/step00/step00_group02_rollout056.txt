```python
import numpy as np
import scipy.optimize as optimize
import time
import math
from scipy.optimize import minimize
from scipy import linalg
import cvxpy as cp

initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = None

    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h])  # To handle wrap-around in convolution
        conv = np.correlate(h, 1 - h, mode='full')
        conv = conv[n_points//2 : n_points//2 + n_points]
        c5 = np.max(conv) * dx
        return c5

    def objective(h):
        return evaluate_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))

    def _run_optimization():
        nonlocal best_c5, best_h
        initial_h = initial_h_values[:n_points].copy()
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = (initial_h * n_points / 2).astype(float)
        initial_h = initial_h / (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h[:n_points]

        result = minimize(objective, initial_h, method='L-BFGS-B', bounds=(0, 1), options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100})
        if result.success:
            h = result.x
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

        for _ in range(10):
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h

        return best_h, best_c5

    start_time = time.time()
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        h, c5 = _run_optimization()
        print(f"Iteration: {_,} | C5: {c5:.6f}")
    
    return best_h, best_c5, n_points
```
