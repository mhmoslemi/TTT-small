```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
from typing import Tuple, List

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=600, n_points=100, max_iter=100, tolerance=1e-6):
    np.random.seed(seed)
    
    # Initialize with the given initial_h_values
    h = initial_h_values.copy()
    n = len(h)
    
    # Ensure h is within [0, 1]
    h = np.clip(h, 0, 1)
    
    # Ensure the integral condition is satisfied
    total = np.sum(h) * (2.0 / n)
    scaling = 1.0 / total if total > 0 else 1.0
    h = h * scaling
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / len(h)))
    
    # Objective function to minimize
    def objective(x):
        nonlocal h
        h = x.copy()
        h = np.clip(h, 0, 1)
        total = np.sum(h) * (2.0 / len(h))
        scaling = 1.0 / total if total > 0 else 1.0
        h = h * scaling
        return evaluate_overlap(h)
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(n)]
    
    # Perform optimization
    result = minimize(
        objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': tolerance, 'maxiter': max_iter}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * (2.0 / len(best_h))
    scaling = 1.0 / total if total > 0 else 1.0
    best_h = best_h * scaling
    
    c5_bound = evaluate_overlap(best_h)
    
    return (best_h, c5_bound, len(best_h))
```
