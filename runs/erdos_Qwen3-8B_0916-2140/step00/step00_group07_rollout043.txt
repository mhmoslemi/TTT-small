import numpy as np
import scipy.optimize as optimize
import time
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    
    def compute_overlap(h):
        h = h.reshape(-1)
        h_extended = np.concatenate([h, h, h])
        result = np.correlate(h_extended, 1 - h, mode="full")
        return result[n_points - 1 : 2 * n_points - 1] * dx
    
    def objective(h):
        overlaps = compute_overlap(h)
        return np.max(overlaps)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def plot_solution(h):
        import matplotlib.pyplot as plt
        plt.figure(figsize=(10, 5))
        plt.plot(np.linspace(0, 2, n_points), h, label='h(x)')
        plt.title("Step Function h(x) for Erdős Minimum Overlap Problem")
        plt.xlabel("x")
        plt.ylabel("h(x)")
        plt.legend()
        plt.grid(True)
        plt.show()
    
    def evaluate_erdos_solution(h):
        return objective(h)
    
    def optimize_h(h0):
        result = optimize.minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        return result
    
    def apply_shear(h, shear_amount):
        h_sheared = np.zeros_like(h)
        for i in range(n_points):
            idx = int(np.clip(i - shear_amount, 0, n_points - 1))
            h_sheared[i] = h[idx]
        return h_sheared
    
    def perturb_h(h, perturbation_strength=0.05):
        h_perturbed = h + np.random.uniform(-perturbation_strength, perturbation_strength, size=h.shape)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed
    
    def grid_search(h0, num_steps=5, num_perturbations=5):
        best_h = h0.copy()
        best_c5 = objective(h0)
        for _ in range(num_steps):
            perturbed_h = perturb_h(best_h)
            for _ in range(num_perturbations):
                h_candidate = apply_shear(perturbed_h, np.random.uniform(-0.5, 0.5))
                c5 = objective(h_candidate)
                if c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5
        return best_h, best_c5
    
    def optimize_with_grid_search(h0, num_steps=5, num_perturbations=5):
        best_h = h0.copy()
        best_c5 = objective(h0)
        for _ in range(num_steps):
            perturbed_h = perturb_h(best_h)
            for _ in range(num_perterturbations):
                h_candidate = apply_shear(perturbed_h, np.random.uniform(-0.5, 0.5))
                c5 = objective(h_candidate)
                if c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5
        return best_h, best_c5
    
    def optimize_with_adaptive_search(h0, max_iterations=100, learning_rate=0.1):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = objective(h)
        for _ in range(max_iterations):
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_temp = h.copy()
                h_temp[i] += 1e-5
                c5_temp = objective(h_temp)
                h_temp[i] -= 2 * 1e-5
                c5_temp -= objective(h_temp)
                grad[i] = c5_temp / (2 * 1e-5)
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            c5 = objective(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        return best_h, best_c5
    
    h0 = initial_h_values
    h_opt, best_c5 = optimize_with_grid_search(h0)
    
    h_opt, best_c5 = optimize_with_adaptive_search(h_opt)
    
    return h_opt, best_c5, n_points
