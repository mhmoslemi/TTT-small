def compute_overlap(h, dx):
    n = len(h)
    overlap = np.zeros(n - 1)
    for k in range(1, n - 1):
        sum_ = 0.0
        for i in range(n - k):
            sum_ += h[i] * (1 - h[i + k])
        overlap[k - 1] = sum_ * dx
    return np.max(overlap)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    max_budget = budget_s
    start_time = time.time()
    
    # Initial n_points from the initial_h_values
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    total = np.sum(h) * dx
    h = h * (1 / (total)) * (n_points / 2)
    
    # Minimize overlap using simulated annealing
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        dx = 2.0 / n_points
        overlap = compute_overlap(h, dx)
        return overlap

    def bounds_func(x):
        return (0.0, 1.0)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Simulated Annealing Parameters
    temp = 100.0
    temp_decay = 0.995
    iterations = 1000
    best_h = h.copy()
    best_value = objective(best_h)
    
    for _ in range(iterations):
        if time.time() - start_time > max_budget:
            break
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        new_h = np.clip(best_h + perturbation, 0.0, 1.0)
        new_value = objective(new_h)
        
        # Acceptance probability
        delta = new_value - best_value
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            best_h = new_h
            best_value = new_value
        
        temp *= temp_decay
    
    # Final evaluation
    final_h = best_h
    dx = 2.0 / n_points
    final_overlap = compute_overlap(final_h, dx)
    print(f"Best C5: {final_overlap}")
    
    return final_h, final_overlap, n_points
