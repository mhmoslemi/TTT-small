import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 300  # Reasonable trade-off between speed and accuracy
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start with a grid-based construction close to the initial_h_values
    h[:n_points//2] = 1.0  # First half is 1
    h[n_points//2:] = 0.0  # Second half is 0
    h = (h + np.random.normal(0, 0.05, n_points)) / 2  # Small perturbation
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    # Evaluate initial C5
    correlation = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlation) * dx
    
    # Set up optimization: minimize C5 by adjusting h
    def objective(x):
        nonlocal h
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Normalize
        correlation = signal.correlate(h, 1 - h, mode="full")
        return np.max(correlation) * dx  # Return the maximum overlap in terms of C5
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization with L-BFGS-B to minimize C5
    start_time = time.time()
    result = opt.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    # Best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Evaluate final C5
    final_correlation = signal.correlate(best_h, 1 - best_h, mode="full")
    c5_bound = np.max(final_correlation) * dx
    
    return best_h, c5_bound, n_points
