import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.integrate import simps
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 500  # Reduced from initial 70 for better performance
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial guess: Use a smooth wave-like function to start the optimization
    # This is a simple harmonic function that meets the integral constraint
    h_initial = np.sin(np.pi * x / 2) / 2 + 0.5
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = (h_initial - h_initial.min()) / (h_initial.max() - h_initial.min())
    
    # Ensure the integral constraint is satisfied
    integral = np.sum(h_initial) * dx
    scaling_factor = (1 / integral) if integral != 0 else 1
    h_initial = h_initial * scaling_factor
    
    # Optimization variables
    h = cp.Variable(n_points)
    
    # Objective function: minimize the maximum overlap integral
    # This is a non-convex problem, so we use a heuristic approach with multiple trials
    
    def objective(h_vals):
        # Compute the overlap integrals for shifts k = -1, 0, 1
        max_overlap = 0.0
        for k in range(1, 4):
            # Discretized shift
            shifted = np.roll(h_vals, k)
            overlap = np.sum(h_vals * (1 - shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Optimization using a gradient descent-like approach with simulated annealing
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    # Use a library with global optimization
    def optimize_h(h_vals):
        nonlocal best_c5, best_h
        for _ in range(100):
            # Random perturbation
            perturbation = np.random.normal(0, 0.1, n_points)
            perturbation = np.clip(perturbation, -0.5, 0.5)
            h_vals = np.clip(h_vals + perturbation, 0, 1)
            
            # Ensure integral constraint
            integral = np.sum(h_vals) * dx
            scaling_factor = (1 / integral) if integral != 0 else 1
            h_vals = h_vals * scaling_factor
            
            # Compute the overlap
            max_overlap = 0.0
            for k in range(1, 4):
                shifted = np.roll(h_vals, k)
                overlap = np.sum(h_vals * (1 - shifted)) * dx
                if overlap > max_overlap:
                    max_overlap = overlap
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h_vals.copy()
    
    # Run the heuristic optimization
    start_time = time.time()
    while time.time() - start_time < budget_s:
        optimize_h(best_h)
    
    # Return the best solution found
    return best_h, best_c5, n_points
