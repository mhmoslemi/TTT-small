```python
import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Function to compute the gradient of the overlap integral
    def compute_gradient(h):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Estimate the derivative using finite differences
            h_plus = h.copy()
            h_plus[i] += 1e-6
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-6)
        return grad
    
    # Simple gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    max_iterations = 1000
    max_time = budget_s - 10  # Reserve time for final evaluation
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > max_time:
            break
        
        grad = compute_gradient(h)
        h = h - learning_rate * grad
        
        # Enforce constraints: h ∈ [0, 1], ∑h = n_points / 2
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        print(f"Iteration {iteration+1}/{max_iterations}, C₅ = {best_c5:.8f}")
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    print(f"Final C₅ bound: {final_c5:.8f}")
    
    return best_h, final_c5, len(best_h)
```
