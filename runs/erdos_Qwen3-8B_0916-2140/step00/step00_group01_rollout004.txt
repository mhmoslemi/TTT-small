```python
import numpy as np
import scipy.optimize as optimize
import time
import math

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Start with 100 points for optimization
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:] = initial_h_values[:n_points]  # Start with initial construction
    
    # Ensure the integral condition is satisfied
    sum_h = np.sum(h)
    required_sum = n_points / 2
    if sum_h != required_sum:
        h = np.array([0.5 for _ in range(n_points)])
    
    def compute_C5(h):
        h = h.astype(float)
        h = np.concatenate([h, h])
        result = np.correlate(h, 1 - h, mode="full")
        result = result[1:-1]
        result = result * dx
        return np.max(result)
    
    def objective(h):
        return compute_C5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Set up bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'maxiter': 100, 'disp': False},
        tol=1e-8
    )
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    # If we have remaining time, try increasing the number of points
    if time.time() - start_time < budget_s:
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.zeros(n_points)
        best_h[:] = 0.5  # Start with a flat function
        
        def compute_C5_higher(h):
            h = h.astype(float)
            h = np.concatenate([h, h])
            result = np.correlate(h, 1 - h, mode="full")
            result = result[1:-1]
            result = result * dx
            return np.max(result)
        
        def objective_higher(h):
            return compute_C5_higher(h)
        
        def constraint_higher(h):
            return np.sum(h) - n_points / 2
        
        bounds_higher = [(0.0, 1.0) for _ in range(n_points)]
        
        result_higher = optimize.minimize(
            objective_higher,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds_higher,
            constraints=[{'type': 'eq', 'fun': constraint_higher}],
            options={'maxiter': 100, 'disp': False},
            tol=1e-8
        )
        
        best_h = result_higher.x
        best_C5 = compute_C5_higher(best_h)
    
    # Final check to ensure the integral condition is satisfied
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        best_h = np.array([0.5 for _ in range(n_points)])
    
    return (best_h, best_C5, n_points)
```
