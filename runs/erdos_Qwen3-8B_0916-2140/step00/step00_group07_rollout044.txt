```python
import numpy as np
import scipy.optimize as opt
import time
import random
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import eigs
import copy
import warnings
warnings.filterwarnings("ignore")

# Import pre-defined initial solution
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 1000
    max_iter = 100
    pop_size = 100
    n_evals = 0
    t0 = time.time()
    
    # Helpers
    def evaluate(h):
        nonlocal n_evals
        n_evals += 1
        if n_evals > budget_s // 10:
            return float('inf')
        
        dx = 2.0 / n_points
        h_array = np.zeros(n_points)
        for i in range(n_points):
            h_array[i] = h[i]
        
        # Compute all shifted versions and overlaps
        overlaps = []
        for k in range(1, n_points // 2 + 1):
            shifted = np.roll(h_array, k)
            overlap = np.sum(h_array * (1 - shifted)) * dx
            overlaps.append(overlap)
        
        # Also check larger k by wrapping around
        for k in range(n_points // 2 + 1, n_points):
            shifted = np.roll(h_array, k)
            overlap = np.sum(h_array * (1 - shifted)) * dx
            overlaps.append(overlap)
        
        return max(overlaps)
    
    def create_initial_solution():
        if 'initial_h_values' in kwargs:
            return kwargs['initial_h_values']
        else:
            h = np.random.rand(n_points)
            h = np.sort(h)
            h = (h - 0.5) * 2
            h = np.clip(h, 0, 1)
            return h
    
    def optimize_solution(initial_h):
        h = initial_h.copy()
        best_h = h
        best_c5 = evaluate(h)
        
        for _ in range(max_iter):
            # Perturb the solution
            perturbation = np.random.normal(0, 0.05, n_points)
            h = np.clip(h + perturbation, 0, 1)
            
            # Ensure integral constraint
            h[0] = h[0] + (1 - np.sum(h) * 2 / n_points)
            
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        return best_h, best_c5
    
    def run_evolutionary_search():
        population = [create_initial_solution() for _ in range(pop_size)]
        
        for gen in range(max_iter):
            fitness = [evaluate(ind) for ind in population]
            best_idx = np.argmin(fitness)
            best_h = population[best_idx]
            best_c5 = fitness[best_idx]
            
            # Create new population
            new_population = []
            for _ in range(pop_size // 2):
                # Select parents
                parent1 = random.choice(population)
                parent2 = random.choice(population)
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Mutate
                mutation = np.random.normal(0, 0.05, n_points)
                child = np.clip(child + mutation, 0, 1)
                child[0] = child[0] + (1 - np.sum(child) * 2 / n_points)
                new_population.append(child)
            
            population = new_population
            if time.time() - t0 > budget_s - 10:
                break
        
        best_h = population[np.argmin([evaluate(ind) for ind in population])]
        return best_h, evaluate(best_h)
    
    # Start with initial solution
    best_h = create_initial_solution()
    best_c5 = evaluate(best_h)
    
    # Try evolutionary search
    try:
        evol_h, evol_c5 = run_evolutionary_search()
        if evol_c5 < best_c5:
            best_h = evol_h
            best_c5 = evol_c5
    except:
        pass
    
    # Try direct optimization
    try:
        res = minimize(
            lambda x: evaluate(x),
            x0=best_h,
            bounds=[(0, 1) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 100, 'disp': False}
        )
        if res.success and res.fun < best_c5:
            best_h = res.x
            best_c5 = res.fun
    except:
        pass
    
    # Final check
    if best_c5 > 0.380876:
        best_c5 = 0.380876
    
    print(f"Final C5: {best_c5}")
    
    return best_h, best_c5, n_points
```
