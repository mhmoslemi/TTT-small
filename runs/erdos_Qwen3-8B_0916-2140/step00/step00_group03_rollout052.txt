import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the function to compute the overlap integral for a given h
    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        # Compute all shifted versions of h
        overlaps = []
        for k in range(1, n_points // 2 + 1):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            overlaps.append(overlap)
        return max(overlaps) if overlaps else 0.0
    
    # Define the function to compute the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess from provided initial_h_values
    initial_guess = initial_h_values
    n_points = len(initial_guess)
    
    # Try to improve the solution by starting from different initial guesses
    best_c5 = float('inf')
    best_h = initial_guess
    
    # Try a few different starting points
    for start in range(3):
        # Create a new starting point with some perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        new_guess = np.clip(initial_guess + perturb, 0, 1)
        
        # Set up the optimization problem
        result = minimize(
            objective,
            new_guess,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'disp': False, 'maxiter': 100}
        )
        
        # Check if the result is valid
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
