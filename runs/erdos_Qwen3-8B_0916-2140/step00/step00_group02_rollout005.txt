        return [(0, 1) for _ in h]

        return np.sum(h) - n_points / 2

        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},

        options={"maxiter": max_iterations, "disp": False}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)
