```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial solution
    n_points = 100  # We start with a moderate number of points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h / np.sum(h) * n_points / 2  # Normalize to meet the integral constraint
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_ones = np.ones_like(h)
        corr = np.correlate(h, h_ones - h, mode="full")
        overlap_integral = np.max(corr) * dx
        return overlap_integral
    
    # Optimization loop
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n_points
    
    # Try different n_points to find the best balance between accuracy and speed
    for n_points in [100, 200, 300, 400, 500, 600, 800, 1000]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate a random starting point
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
        
        # Optimization using L-BFGS-B
        def objective(x):
            h = x.reshape(n_points)
            return compute_c5(h)
        
        def constraint(x):
            h = x.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        constraints = [{'type': 'eq', 'fun': constraint}]
        result = optimize.minimize(objective, h, method='L-BFGS-B', constraints=constraints, options={'disp': False})
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.reshape(n_points)
                best_n = n_points
    
    # Final check with the initial solution (to avoid getting stuck in local optima)
    h_initial = np.array(initial_h_values[:best_n], dtype=np.float64)
    h_initial = h_initial / np.sum(h_initial) * best_n / 2
    current_c5_initial = compute_c5(h_initial)
    if current_c5_initial < best_c5:
        best_c5 = current_c5_initial
        best_h = h_initial
    
    # Evaluate best solution
    final_c5 = best_c5
    final_h = best_h
    
    return (final_h, final_c5, len(final_h))
```
