```python
import numpy as np
import scipy.optimize
import scipy.sparse
import scipy.sparse.linalg
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.interpolate import interp1d
from scipy.integrate import quad
from functools import lru_cache
import warnings
warnings.filterwarnings("ignore", category=RuntimeWarning)

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_time = budget_s
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points

    # Initial guess
    h = initial_h_values[:n_points]
    h = h[:n_points]  # Ensure it has the right length
    h = h / np.sum(h) * (n_points / 2)

    # Define the overlap integral as a function
    def overlap_integral(h):
        shifted_h = np.roll(h, shift=1)
        integral = np.sum(h * (1 - shifted_h)) * dx
        return integral

    # Objective function to minimize
    def objective(h):
        return np.max(np.array([overlap_integral(h), overlap_integral(np.roll(h, 2)), overlap_integral(np.roll(h, 3)), overlap_integral(np.roll(h, 4)), overlap_integral(np.roll(h, 5))]))

    # Bounds and constraints
    bounds = Bounds(0, 1)
    constraints = []

    # Try a few different shifts
    def evaluate_c5(h):
        values = []
        for shift in range(1, 6):
            shifted_h = np.roll(h, shift)
            integral = np.sum(h * (1 - shifted_h)) * dx
            values.append(integral)
        return np.max(values)

    def evaluate_c5_sequential(h):
        c5_values = []
        for shift in range(1, 6):
            shifted_h = np.roll(h, shift)
            integral = np.sum(h * (1 - shifted_h)) * dx
            c5_values.append(integral)
        return np.max(c5_values)

    def objective(h):
        return evaluate_c5_sequential(h)

    # Optimization
    result = scipy.optimize.minimize(
        fun=objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )

    best_h = result.x
    best_c5 = evaluate_c5_sequential(best_h)

    # Try a coarser grid with more samples for better resolution
    n_points = 200
    dx = 2.0 / n_points
    h = best_h

    # Interpolate to the new grid
    h_interp = interp1d(np.arange(n_points), h, kind='linear', fill_value="extrapolate")
    h_new = h_interp(np.arange(n_points))
    h_new = h_new / np.sum(h_new) * (n_points / 2)

    # Final optimization on the larger grid
    result = scipy.optimize.minimize(
        fun=evaluate_c5_sequential,
        x0=h_new,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 50, 'ftol': 1e-6, 'gt': 1e-6}
    )

    final_h = result.x
    final_c5 = evaluate_c5_sequential(final_h)

    # Return the result
    elapsed = time.time() - start_time
    if elapsed > max_time:
        print(f"Time limit exceeded ({max_time} seconds). Returning best so far.")
        return (h, result.fun, n_points)
    else:
        return (final_h, final_c5, n_points)
```
