import numpy as np
import time
import random

def run(seed=42, budget_s=600):

    np.random.seed(seed)

    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-8:
        raise ValueError("Initial h_values does not satisfy the integral constraint.")

    # Function to compute the overlap integral
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points):
            shifted_h[k:] = h[:n_points - k]
        overlap = np.zeros(n_points)
        for k in range(1, n_points):
            overlap[k] = np.sum(h * (1 - shifted_h[k:])) * dx
        return np.max(overlap)

    # Function to perturb h_values slightly
    def perturb_h(h_values, perturbation_strength=0.05):
        h = np.array(h_values)
        h = h + np.random.uniform(-perturbation_strength, perturbation_strength, size=h.shape)
        h = np.clip(h, 0.0, 1.0)
        return h.tolist()

    # Function to compute the sum of h_values and adjust to maintain the integral constraint
    def adjust_h(h_values):
        h = np.array(h_values)
        total = np.sum(h)
        target = len(h) / 2
        if np.abs(total - target) < 1e-8:
            return h.tolist()
        delta = (target - total) / len(h)
        h = h + delta
        h = np.clip(h, 0.0, 1.0)
        return h.tolist()

    # Initial solution
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    # Define the optimizers
    def optimize_with_sgd():
        nonlocal best_h, best_c5
        h = np.array(best_h)
        for _ in range(1000):
            # Perturbation
            h_perturbed = perturb_h(h)
            h_perturbed = adjust_h(h_perturbed)
            c5 = compute_c5(h_perturbed)
            if c5 < best_c5:
                best_h = h_perturbed
                best_c5 = c5
            h = np.array(h_perturbed)

    def optimize_with_grid_search():
        nonlocal best_h, best_c5
        h = np.array(best_h)
        for _ in range(100):
            # Random perturbation
            h_perturbed = perturb_h(h)
            h_perturbed = adjust_h(h_perturbed)
            c5 = compute_c5(h_perturbed)
            if c5 < best_c5:
                best_h = h_perturbed
                best_c5 = c5
            h = np.array(h_perturbed)

    start_time = time.time()
    optimize_with_sgd()
    optimize_with_grid_search()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # If we still have time, try more refined perturbations
        for _ in range(100):
            h = np.array(best_h)
            h_perturbed = perturb_h(h, perturbation_strength=0.01)
            h_perturbed = adjust_h(h_perturbed)
            c5 = compute_c5(h_perturbed)
            if c5 < best_c5:
                best_h = h_perturbed
                best_c5 = c5

    return best_h, best_c5, len(best_h)
