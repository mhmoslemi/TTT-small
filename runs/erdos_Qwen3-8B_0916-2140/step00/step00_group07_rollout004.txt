def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h, dtype=np.float64)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr * dx)
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with the initial construction
    h = initial_h_values.copy()
    
    n_points = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Define the objective and constraints
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Set up bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization parameters
    max_iterations = 100
    tolerance = 1e-5
    
    for _ in range(max_iterations):
        start_time = time.time()
        
        # Tune the initial guess using Sequential Least Squares Programming (SLSQP)
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='SLSQP',
            tol=tolerance,
            options={'maxiter': 50, 'ftol': tolerance, 'disp': False}
        )
        
        # Update the best solution
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            h = current_h.copy()
        else:
            break
        
        # Stop if budget is exceeded
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, n_points
