import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.integrate import simps

def run(seed=42, budget_s=600, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:-k]
        product = h * (1 - shifted)
        integral = simps(product, dx=dx)
        return integral

    def objective(h):
        return compute_c5(h)

    def bounds():
        return Bounds(0, 1)

    def constraints():
        return [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

    def evaluate_solution(h):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k:] = h[:-k]
        product = h * (1 - shifted)
        integral = simps(product, dx=dx)
        return integral

    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = evaluate_solution(best_h)

    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break

        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)

        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds(),
            constraints=constraints(),
            options={'disp': False, 'maxiter': 100}
        )

        if result.success:
            h_result = result.x
            current_c5 = evaluate_solution(h_result)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_result.copy()

    return best_h, best_c5, n_points
