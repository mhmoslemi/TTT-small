import numpy as np
import scipy.optimize as optimize
import time
import sys
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-8:
        initial_h = np.array([0.5 for _ in range(n_points)])
    
    # Function to evaluate the C_5 value
    def evaluate_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h = np.concatenate((h, h))  # To handle full convolution
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

    # Objective function for optimization
    def objective(h):
        return evaluate_c5(h)

    # Optimization using SciPy
    def optimize_c5(h_start, max_iter=50, tol=1e-6):
        h = h_start.copy()
        for _ in range(max_iter):
            # Compute current C_5 value
            c5_val = evaluate_c5(h)
            if c5_val < 1e-10:
                break
            # Perturb the function values to explore the space
            idx = np.random.choice(len(h), size=10, replace=False)
            for i in idx:
                delta = np.random.normal(0, 0.1)
                h[i] = np.clip(h[i] + delta, 0, 1)
            # Check for convergence
            if np.abs(objective(h) - c5_val) < tol:
                break
        return h, evaluate_c5(h)

    # Run the optimization
    start_time = time.time()
    h_opt, c5_bound = optimize_c5(initial_h)
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print(f"Time exceeded budget, returning best of initial guess")
        return initial_h, evaluate_c5(initial_h), len(initial_h)

    return h_opt, c5_bound, len(h_opt)
