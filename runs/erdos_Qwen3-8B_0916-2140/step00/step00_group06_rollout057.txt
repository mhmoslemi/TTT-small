```python
import numpy as np
import time
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.signal import correlate
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # A sample initial construction
    n = 95
    h = np.zeros(n)
    for i in range(n):
        if i < n // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h.tolist()

# Strategy:
# - Start with the initial_h_values as a base
# - Use a gradient-based optimization strategy with a custom loss function
# - Use a stepwise approach to minimize the maximum overlap integral
# - Regularly sample and evaluate to avoid local minima

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # A balance between resolution and performance

    # Initial guess
    h_initial = initial_h_values()
    while time.time() - start_time < budget_s:
        # Convert initial guess to numpy array
        h = np.array(h_initial, dtype=np.float64)
        # Define the constraint: sum(h) == n_points / 2
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}

        # Objective function: minimize the maximum overlap integral
        def objective(x):
            h = x
            corr = correlate(h, 1 - h, mode="full")
            return np.max(corr)

        # Bounds: each h[i] must be in [0, 1]
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Use L-BFGS-B optimizer
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'disp': False})
        h_opt = result.x

        # Evaluate the solution
        c5 = evaluate_erdos_solution(h_opt.tolist())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.tolist()

        # Introduce some randomness to avoid local optima
        perturb = np.random.normal(0, 0.05, n_points)
        h_initial = np.clip(h_opt + perturb, 0, 1).tolist()

        # Print progress
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        print(f"Elapsed: {elapsed:.2f}s / Remaining: {remaining:.2f}s | Current C₅: {best_c5:.6f}")

    return best_h, best_c5, n_points
```
