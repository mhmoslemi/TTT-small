import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds
from functools import partial

def run(seed=42, budget_s=600, **kwargs):
    # Set the random seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial h_values from the provided initial_h_values
    initial_h_values = kwargs.get('initial_h_values', np.load('initial_h.npy'))
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    sum_h_initial = np.sum(initial_h_values)
    
    # Define the function to compute the overlap integral
    def compute_overlap(h, dx):
        n = len(h)
        overlap = np.zeros(n - 1)
        for k in range(1, n):
            overlap_k = 0
            for i in range(n - k):
                overlap_k += h[i] * (1 - h[i + k])
            overlap[k - 1] = overlap_k * dx
        return np.max(overlap)
    
    def fitness(h, dx):
        return compute_overlap(h, dx)
    
    # Try different n_points values
    n_points_values = [50, 100, 200, 400]
    
    # Try each n_points value to find the best solution
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_values:
        # Calculate dx
        dx = 2.0 / n_points
        
        # Generate a random initial guess
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # scale to ensure sum(h) * dx = 1
        
        # Define bounds for the optimization variables
        bounds = Bounds([0.0] * n_points, [1.0] * n_points)
        
        # Define the function to minimize with constraints
        def objective(x):
            return fitness(x, dx)
        
        # Use a constrained optimization method to find the minimum of the overlap
        result = optimize.minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        # Extract the best solution found
        h_candidate = result.x
        c5_candidate = result.fun
        
        # Update the best solution
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n_points = n_points
        
        # Print progress
        print(f"n_points={n_points}, c5={best_c5}")
    
    # Return the best solution found
    return (best_h, best_c5, best_n_points)
