import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set the seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial sample sizes to try
    sample_sizes = [100, 200, 300, 400, 500]
    
    # Best result so far
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    # Try different sample sizes
    for n in sample_sizes:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Generate initial samples
        h = np.array(initial_h_values)
        h = np.interp(h, [0, len(h)-1], [0, 2])
        h = np.interp(h, [0, 2], [0, n])
        h = np.round(h).astype(int)
        h = np.minimum(np.maximum(h, 0), n)
        
        # If initial_h_values is not provided, initialize randomly
        if len(h) == 0:
            h = np.random.rand(n) * 2 - 1
            h = np.round(h).astype(int)
            h = np.minimum(np.maximum(h, 0), n)
        
        # Normalize to satisfy the integral constraint
        total = np.sum(h)
        scaling = n / 2 / total
        h = np.round(h * scaling).astype(int)
        h = np.minimum(np.maximum(h, 0), n)
        
        # Convert to float
        h = h.astype(float)
        
        # Define objective function: maximize the overlap integral
        def objective(x):
            h_new = x.copy()
            h_new = np.clip(h_new, 0, n)
            # Compute the overlap integral
            overlap = np.zeros(n + 1)
            for i in range(n):
                for k in range(1, n + 1):
                    start = i
                    end = i + k
                    if end > n:
                        break
                    overlap[k] += h_new[i] * (1 - h_new[end])
            return np.max(overlap) * (2.0 / n)
        
        # Define constraints
        def constraint(x):
            return np.sum(x) - n / 2
        
        # Use L-BFGS-B to minimize the objective
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, n)] * n, constraints=[{'type': 'eq', 'fun': constraint}])
        
        # Get the best solution
        best_h_current = result.x
        c5_current = result.fun
        
        # Check if this is better
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = best_h_current
            best_n = n
        
        # Print progress
        print(f"Sample size {n}: Best C5 = {best_c5:.6f}")
    
    # Final evaluation
    best_h = np.round(best_h).astype(float)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Return the result
    return (best_h, best_c5, len(best_h))
