        overlap = overlap * dx
        return np.max(overlap)

        nonlocal best_c5
        h = np.array(x, dtype=np.float64)

            best_c5 = c5
            best_h = h.copy()
        return c5

        history.append(current_c5)

            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent step
        h_prev = h.copy()
        h_new = h.copy()
        for i in range(n_points):
            if h[i] < 1:
                h_new[i] = min(1.0, h[i] + learning_rate)

                h_new[i] = max(0.0, h[i] - learning_rate)
