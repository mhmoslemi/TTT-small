```python
import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_n_points = 100  # Use a manageable number of points for speed

    # Define the initial h_values as a starting point
    initial_h_values = np.zeros(max_n_points)
    for i in range(max_n_points):
        initial_h_values[i] = 0.5 * (1 + np.cos(2 * np.pi * i / max_n_points))
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values[-1] = 0.5  # Adjust last point for integration
    initial_h_values = np.maximum(0, np.minimum(1, initial_h_values))

    # Function to compute the overlap integral
    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h_values = h
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Objective function for optimization
    def objective(h):
        return compute_c5(h)

    # Constraint: ∑h_i dx = 1 → ∑h_i = n_points / 2
    def constraint_func(h):
        return np.sum(h) - max_n_points / 2

    # Bounds for h: each element must be in [0, 1]
    bounds = Bounds(0, 1)

    # Perform optimization
    res = optimize.minimize(
        objective,
        initial_h_values,
        method="L-BFGS-B",
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_func},
        options={"maxiter": 1000, "ftol": 1e-8}
    )

    # Extract the best solution
    best_h = res.x
    best_c5 = compute_c5(best_h)
    n_points = len(best_h)

    return (best_h, best_c5, n_points)
```
