import numpy as np
import scipy.optimize
import time

from scipy.integrate import simps
from scipy.signal import correlate
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    c5_values = []
    for k in range(1, n_points):
        shifted = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted)) * dx
        c5_values.append(overlap)
    return max(c5_values)

def initial_h_values():
    return np.load('initial_h.npy')  # Placeholder for initial solution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    total = np.sum(initial_h) * dx
    if np.abs(total - 1.0) > 1e-6:
        initial_h = np.clip(initial_h, 0, 1)
    
    # Function to evaluate the C5 value
    def evaluate_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        c5_values = []
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            c5_values.append(overlap)
        return max(c5_values)
    
    # Objective function: minimize the maximum overlap value
    def objective(h):
        return evaluate_c5(h)
    
    # Constraints: sum(h) * dx == 1
    def constraint_sum(h):
        return np.sum(h) * (2.0 / len(h)) - 1.0
    
    # Equality constraint: sum(h) * dx == 1
    def constraint_sum_eq(h):
        return np.sum(h) - len(h) / 2.0
    
    # Optimization options
    options = {
        'maxiter': 1000,
        'ftol': 1e-6,
        'disp': False
    }
    
    # Start optimization from the initial guess
    result = minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(len(initial_h))],
        constraints=[{'type': 'eq', 'fun': constraint_sum_eq}],
        options=options
    )
    
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)
