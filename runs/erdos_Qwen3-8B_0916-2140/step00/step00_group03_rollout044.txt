    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the overlap integral
    def compute_c5(h):
        h = np.array(h)
        shifts = np.arange(-n_points + 1, n_points)
        max_overlap = 0.0
        for k in shifts:
            if k <= 0:
                shifted = np.roll(h, k)
            else:
                shifted = np.roll(h, -k)
            overlap = np.sum(h * (1 - shifted)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Objective function to minimize
    def objective(h):
        return compute_c5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    h = cp.Variable(n_points)
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    obj = cp.Minimize(compute_c5(h))
    
    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver='SCS', max_iter=1000)
    except:
        pass
    
    # Extract the optimal h
    h_values = h.value.flatten()
    
    # Compute the final C5
    c5_bound = compute_c5(h_values)
