import numpy as np
import scipy.optimize as opt
import time
import random
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define initial h from provided initial_h_values
    initial_h_values = np.array([0.0] * 47)
    for i in range(47):
        initial_h_values[i] = 0.5 * (1 + np.cos(np.pi * (i / 47 - 0.5)))
    initial_h_values = initial_h_values / np.sum(initial_h_values)
    initial_h_values = initial_h_values * 47 / 2

    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlap = np.correlate(h, 1 - h, mode='full')
        return np.max(overlap * dx)

    def objective_and_gradient(h_flat):
        h = h_flat.reshape(-1)
        return compute_overlap(h), np.zeros_like(h_flat)

    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_overlap(h)

    def constraints(h_flat):
        h = h_flat.reshape(-1)
        sum_h = np.sum(h)
        return sum_h - len(h) / 2

    n_points = 100
    h_flat = np.random.rand(n_points) * 2 - 1
    h_flat = h_flat * (1 / 2) * 2

    # Perform a grid search with random perturbations and use a local optimizer
    best_overlap = float('inf')
    best_h = h_flat
    for _ in range(20):
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h_flat = np.clip(h_flat + perturbation, 0, 1)
        h_flat = h_flat * (len(h_flat) / 2) / np.sum(h_flat)

        # Local optimization
        res = opt.minimize(objective, h_flat, method='L-BFGS-B', bounds=(0, 1), options={'maxiter': 100})
        if res.success:
            h_flat = res.x
            current_overlap = compute_overlap(h_flat)
            if current_overlap < best_overlap:
                best_overlap = current_overlap
                best_h = h_flat

    # Final refinement
    for _ in range(10):
        perturbation = np.random.rand(n_points) * 0.05 - 0.025
        h_flat = np.clip(h_flat + perturbation, 0, 1)
        h_flat = h_flat * (len(h_flat) / 2) / np.sum(h_flat)
        current_overlap = compute_overlap(h_flat)
        if current_overlap < best_overlap:
            best_overlap = current_overlap
            best_h = h_flat

    # Final check and return
    final_overlap = compute_overlap(best_h)
    return best_h.tolist(), final_overlap, len(best_h)
