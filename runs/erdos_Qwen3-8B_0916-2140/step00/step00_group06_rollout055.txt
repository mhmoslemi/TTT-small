```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Preloaded initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess from provided initial_h_values
    n_points = len(initial_h_values)
    h = initial_h_values.copy()

    def objective(x):
        h_ = x.copy()
        # Enforce bounds [0, 1]
        h_[h_ < 0] = 0
        h_[h_ > 1] = 1
        # Compute the maximum overlap integral
        corr = np.correlate(h_, 1 - h_, mode="full")
        max_overlap = np.max(corr) * (2.0 / n_points)
        return max_overlap

    def constraints(x):
        return np.sum(x) - n_points / 2

    # Initial guess from initial_h_values
    x0 = h.copy()

    # Bounds for each x_i (0 <= x_i <= 1)
    bounds = [(0, 1) for _ in range(n_points)]

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraints}],
        method='L-BFGS-B',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed = time.time() - start_time

    # If the time budget is not exhausted, perform a few local searches
    if elapsed < budget_s - 10:
        # Consider a few perturbations
        for _ in range(3):
            perturb = np.random.normal(0, 0.05, n_points)
            h_perturb = np.clip(best_h + perturb, 0, 1)
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb[np.abs(h_perturb) < 1e-8] = 0
            h_perturb = np.clip(h_perturb, 0, 1)
            current_c5 = evaluate_erdos_solution(h_perturb)
            if current_c5 < best_c5:
                best_h = h_perturb
                best_c5 = current_c5

    # Final check to ensure sum is correct
    best_h = np.clip(best_h, 0, 1)
    best_h = np.round(best_h, 5)
    best_h = np.clip(best_h, 0, 1)
    best_h = np.round(best_h, 5)
    best_h = np.clip(best_h, 0, 1)
    best_h = np.round(best_h, 5)

    return (best_h, best_c5, n_points)
```
