import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting with initial_h_values and a small n_points
    n_points_start = 250
    h_start = initial_h_values.copy()
    h_start = np.interp(np.linspace(0, 2, n_points_start), np.linspace(0, 2, len(h_start)), h_start)
    
    # Function to compute C5
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)
    
    # Function to compute the objective for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem
    def optimize_h(n_points):
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_start)), h_start)
        h = np.clip(h, 0, 1)
        return h
    
    # Try with n_points = 500
    n_points = 500
    h = optimize_h(n_points)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Optimization using simulated annealing
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Generate a new candidate h
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        h_new[idx] = np.random.rand() * 2 - 1
        h_new = np.clip(h_new, 0, 1)
        
        # Compute overlap and constraints
        c5_new = compute_c5(h_new)
        
        # Accept with probability based on temperature
        temp = max(0.1, (budget_s - (time.time() - start_time)) / budget_s)
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / temp):
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5, n_points
