```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import sys

# Initial construction from the problem
initial_h_values = np.load('initial_h.npy')  # We assume this is available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 95  # Starting with the initial n_points
    h = initial_h_values.copy()
    
    # Define the objective function
    def objective(h):
        dx = 2.0 / len(h)
        c5 = 0.0
        for k in range(1, len(h) - 1):
            corr = correlate(h, 1 - h, mode="full")
            c5 = max(c5, np.max(corr) * dx)
        return c5

    # Optimization settings
    max_iter = 100
    tol = 1e-5
    n_samples = 100  # Number of samples to consider for optimization

    # Extract a subset of the current h for optimization
    idxs = np.random.choice(len(h), n_samples, replace=False)
    h_opt = h[idxs].copy()
    best_c5 = objective(h)
    best_h = h.copy()
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random perturbation
        perturbation = np.random.uniform(-0.05, 0.05, size=len(h_opt))
        h_perturbed = np.clip(h_opt + perturbation, 0, 1)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_perturbed)
        scale = (2.0 / len(h_perturbed)) / total if total != 0 else 1.0
        h_perturbed = np.clip(h_perturbed * scale, 0, 1)
        
        # Evaluate the new C5
        c5_new = objective(h_perturbed)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            best_h[idxs] = h_perturbed
            print(f"Iteration {i+1}: New C5 = {best_c5:.8f}")
    
    # Final evaluation
    final_h = best_h
    final_dx = 2.0 / len(final_h)
    corr = correlate(final_h, 1 - final_h, mode="full")
    c5_final = np.max(corr) * final_dx
    return final_h, c5_final, len(final_h)
```
