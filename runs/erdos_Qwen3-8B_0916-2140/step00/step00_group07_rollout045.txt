import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode="full")
        return np.max(conv * dx)

    # Function to compute the overlap integral for a given window shift
    def compute_overlap(h, shift):
        h_shifted = np.roll(h, shift)
        return np.dot(h, 1 - h_shifted) * dx

    # Function to evaluate the maximum overlap
    def evaluate_max_overlap(h):
        max_overlap = 0.0
        for shift in range(1, n_points):
            overlap = compute_overlap(h, shift)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Objective function to minimize the maximum overlap
    def objective(h):
        return evaluate_max_overlap(h)

    # Constraints:
    # 1. All h_i in [0, 1]
    # 2. Sum of h_i = n_points / 2

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization function
    def optimize_h(initial_guess):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        objective_func = evaluate_max_overlap(h.value) if h.value is not None else 0.0
        prob = cp.Problem(cp.Minimize(objective_func), constraints)
        start_time = time.time()
        result = prob.solve(solver=cp.SCS, max_iter=1000)
        elapsed = time.time() - start_time
        if result is None:
            return initial_guess, 0.0, n_points
        return h.value, objective_func, n_points

    # Start with the initial guess
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_n = n_points

    # Try a few different starting points
    for _ in range(5):
        # Perturb the initial guess
        perturbed_h = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (np.sum(perturbed_h) / (n_points / 2))

        # Optimize
        h_opt, c5_opt, n_opt = optimize_h(perturbed_h)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_opt

    # Return the best solution
    return best_h, best_c5, best_n
