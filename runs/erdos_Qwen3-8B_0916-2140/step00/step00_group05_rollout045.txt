def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, int(n_points / 2) + 1):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h))
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess based on the provided initial_h_values
    initial_h = initial_h_values(70)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Convert initial guess to float
    initial_h = initial_h.astype(float)
    
    # Ensure the initial guess satisfies the integral constraint
    initial_h_sum = np.sum(initial_h)
    initial_h = initial_h * (n_points / 2) / initial_h_sum
    
    # Define the optimization variables
    n_eval = 100
    max_overlap = 0.0
    
    # Define the function to evaluate the overlap
    def objective(x):
        nonlocal max_overlap
        h = x
        h = h.astype(float)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute all overlaps
        max_overlap = 0.0
        for k in range(1, int(n_points / 2) + 1):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h))
            if overlap > max_overlap:
                max_overlap = overlap
        
        return max_overlap * (2.0 / n_points)
    
    # Use a local optimizer with line search
    result = opt.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',

        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
