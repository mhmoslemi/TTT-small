    def compute_c5(h):

    def objective(h):
        return max(compute_c5(h), compute_c5(np.roll(h, 1)), compute_c5(np.roll(h, 2)))

    bounds = [(0, 1) for _ in range(n_points)]

    result = optimize.differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=15,

        disp=False,
        callback=None,
