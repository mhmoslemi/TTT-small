import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 1000
    dx = 2.0 / n_points
    h = np.random.rand(n_points)
    h = np.minimum(np.maximum(h, 0), 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraints(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    def bounds(x):
        return np.concatenate((np.zeros(n_points), np.zeros(n_points)))
    
    def optimize_h(h_initial):
        x0 = h_initial.copy()
        res = minimize(objective, x0, method='L-BFGS-B', bounds=Bounds(0, 1))
        return res.x
    
    start_time = time.time()
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(initial_h_values)
    best_n_points = len(initial_h_values)
    
    try:
        # Try with initial guess
        h_initial = initial_h_values
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_opt = optimize_h(h_initial)
        c5 = evaluate_erdos_solution(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            best_n_points = len(h_opt)
    except:
        pass
    
    try:
        # Try with random initial guess
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_opt = optimize_h(h_initial)
        c5 = evaluate_erdos_solution(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            best_n_points = len(h_opt)
    except:
        pass
    
    if time.time() - start_time > budget_s - 10:
        return best_h, best_c5, best_n_points
