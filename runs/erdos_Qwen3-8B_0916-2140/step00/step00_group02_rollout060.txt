from scipy.optimize import LinearConstraint, minimize_scalar

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.array([0.8427, 0.8748, 0.8476, 0.8231, 0.8061, 0.8059, 0.8081, 0.8145, 0.8278, 0.8415, 0.8517,

                             0.8343, 0.8284, 0.8291, 0.8318, 0.8353, 0.8392, 0.8466, 0.8516])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting point: current solution
    n_start = len(initial_h_values)
    h = initial_h_values.copy()

    # Define the objective function to minimize the maximum overlap
    def objective(x):
        h_values = x
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap) * (2.0 / len(h_values))
        return c5

    # Constraints: sum of h must be n_start / 2 (to maintain average 0.5)
    def constraint_sum(x):
        return np.sum(x) - n_start / 2
    cons = {'type': 'eq', 'fun': constraint_sum}

    # Bounds: h in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_start)]

    # Initial guess
    x0 = h.copy()

    # Optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    
    # If optimization fails, fall back to initial solution
    if result.success:
        h = result.x
    else:
        h = initial_h_values

    # Compute the final C5 value
    c5 = evaluate_erdos_solution(h)
    
    return h, c5, len(h)
