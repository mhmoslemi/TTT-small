import numpy as np
import scipy.optimize as optimize
import time
import sys

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all shifted versions of h
    shifts = np.arange(-1, 1.5, 0.5)
    overlaps = []
    for shift in shifts:
        shifted_h = np.roll(h, int(round(shift * n_points / dx)))
        overlap = np.sum(h * (1 - shifted_h)) * dx
        overlaps.append(overlap)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Start with 200 points for a balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points) * 2 - 1  # A random initial guess
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    best_c5 = float('inf')
    best_h = initial_h_values.copy()

    for _ in range(100):  # 100 iterations of optimization
        def objective(x):
            nonlocal best_c5
            h = x.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_erdos_solution(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            return c5

        # Use L-BFGS-B to minimize the maximum overlap
        result = optimize.minimize(objective, best_h, method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-8})
        if time.time() - start_time > budget_s - 10:
            break
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        else:
            break

    # Final evaluation with the best solution
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return best_h, final_c5, n_points
