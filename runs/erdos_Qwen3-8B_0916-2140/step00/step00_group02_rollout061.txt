```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess with 45 samples
    n_points_initial = 45
    h_initial = initial_h_values
    h_initial = h_initial[:n_points_initial]
    
    # Try with smaller n_points to save time
    n_points = 100  # smaller than initial to save computation time
    dx = 2.0 / n_points
    
    # Define the overlap integral for a given h
    def compute_overlap(h):
        h = np.array(h)
        k_values = np.arange(-n_points + 1, n_points)
        overlap = []
        for k in k_values:
            if k == 0:
                continue
            h_shifted = np.roll(h, k)
            integral = np.sum(h * (1 - h_shifted)) * dx
            overlap.append(integral)
        return max(overlap)
    
    # Objective function to minimize (max overlap)
    def objective(h):
        return compute_overlap(h)
    
    # Constraints: sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = np.zeros(n_points)
    for i in range(n_points_initial):
        initial_guess[i] = h_initial[i]
    initial_guess = initial_guess[:n_points]
    
    # Optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[{"type": "eq", "fun": constraint_sum}],
        method="SLSQP"
    )
    
    if time.time() - start_time > budget_s - 10:
        return (result["x"], objective(result["x"]), n_points)
    
    # Try with even smaller n_points
    n_points = 50
    dx = 2.0 / n_points
    h_new = np.zeros(n_points)
    
    # Use the best solution from the first run as initial guess
    best_h = result["x"]
    for i in range(n_points):
        if i < len(best_h):
            h_new[i] = best_h[i]
    
    # Re-define compute_overlap for new n_points
    def compute_overlap_new(h):
        h = np.array(h)
        k_values = np.arange(-n_points + 1, n_points)
        overlap = []
        for k in k_values:
            if k == 0:
                continue
            h_shifted = np.roll(h, k)
            integral = np.sum(h * (1 - h_shifted)) * dx
            overlap.append(integral)
        return max(overlap)
    
    # New objective
    def objective_new(h):
        return compute_overlap_new(h)
    
    # New constraints
    def constraint_sum_new(h):
        return np.sum(h) - n_points / 2
    
    # New bounds
    bounds_new = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess_new = h_new
    
    # Optimize again with smaller n_points
    result_new = optimize.minimize(
        objective_new,
        initial_guess_new,
        bounds=bounds_new,
        constraints=[{"type": "eq", "fun": constraint_sum_new}],
        method="SLSQP"
    )
    
    # Final result
    best_h_final = result_new["x"]
    best_c5 = objective_new(best_h_final)
    
    return (best_h_final, best_c5, n_points)
```
