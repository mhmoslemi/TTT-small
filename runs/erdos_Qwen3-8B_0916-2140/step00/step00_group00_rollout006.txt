def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points to allow a good initial search
    total_mass = n_points / 2
    dx = 2.0 / n_points

    # Initial solution from the provided initial_h_values
    h_initial = initial_h_values[:n_points]  # Truncate to n_points
    h_initial = np.clip(h_initial, 0, 1)
    h_initial[0] = 0.0
    h_initial[-1] = 0.0
    h_initial = h_initial / h_initial.sum() * total_mass

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * total_mass
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def objective(x):
        h = np.clip(x, 0, 1)
        h = h / h.sum() * total_mass
        return compute_c5(h)

    def constraint(x):
        h = np.clip(x, 0, 1)
        h = h / h.sum() * total_mass
        return np.array([compute_c5(h)])

    # Optimization using scipy.optimize
    x0 = h_initial
    result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], options={'ftol': 1e-8, 'maxiter': 1000})
    best_h = result.x
    best_c5 = compute_c5(best_h)

    # If needed, refine the solution with more points
    n_points = 200
    dx = 2.0 / n_points
    total_mass = n_points / 2

    # Rebuild the initial solution with more points
    h_initial = np.zeros(n_points)
    h_initial[0] = 0.0
    h_initial[-1] = 0.0
    h_initial[1:-1] = 0.5
    h_initial = h_initial / h_initial.sum() * total_mass

    def compute_c5_extended(h):
        h = np.clip(h, 0, 1)
        h = h / h.sum() * total_mass
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def objective_extended(x):
        h = np.clip(x, 0, 1)
        h = h / h.sum() * total_mass
        return compute_c5_extended(h)

    x0 = h_initial
    result = opt.minimize(objective_extended, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], options={'ftol': 1e-8, 'maxiter': 1000})
    best_h = result.x
    best_c5 = compute_c5_extended(best_h)

    return (best_h, best_c5, n_points)
