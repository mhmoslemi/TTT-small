import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize, Bounds
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define number of points and sample initial h
    n_points = 100  # Reduce for faster optimization
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    
    # Ensure the integral condition
    h_sum = np.sum(h)
    required_sum = n_points / 2
    if h_sum != required_sum:
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale = required_sum / h_sum
        h *= scale
    
    # Define objective function to minimize max overlap
    def objective(x):
        nonlocal h
        h = x.copy()
        idx = np.abs(np.arange(n_points) - np.arange(n_points)) <= 1
        correlation = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(correlation) * dx
        return overlap
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with constraints
    result = minimize(
        objective,
        h,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False},
        tol=1e-6
    )
    
    # Return the best solution found
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
