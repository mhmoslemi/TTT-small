```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define initial h_values
    n_points = 100
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    
    # Define parameters
    max_iterations = 100
    learning_rate = 0.01
    momentum = 0.9
    tolerance = 1e-5
    
    # Function to compute the overlap integral C_5
    def compute_C5(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate((h, h))  # Duplicate to handle wrap-around
        result = np.correlate(h, 1 - h[:n_points], mode="full")
        result = result[n_points - 1 : n_points + 1]
        return np.max(result) * (2.0 / n_points)
    
    # Function to update h using gradient descent
    def update_h(h, learning_rate, momentum, tolerance):
        h_new = np.copy(h)
        gradient = np.zeros_like(h)
        
        # Compute overlap integrals for each shift
        shifts = np.arange(-n_points + 1, n_points)
        for shift in shifts:
            if shift == 0:
                continue
            # Compute overlap integral for this shift
            h_shifted = np.roll(h, shift)
            integral = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
            
            # Update gradient
            gradient += h * (1 - h_shifted) * (1 - 2 * integral)
        
        # Update h
        h_new = h + learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        
        # Check for convergence
        if np.max(np.abs(h_new - h)) < tolerance:
            return h_new, False
        return h_new, True
    
    # Optimization loop
    best_h = np.copy(h)
    best_C5 = compute_C5(h)
    start_time = time.time()
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        h, converged = update_h(h, learning_rate, momentum, tolerance)
        current_C5 = compute_C5(h)
        
        if current_C5 < best_C5:
            best_h = np.copy(h)
            best_C5 = current_C5
        
        if not converged:
            learning_rate *= 0.995
        
    # Final evaluation
    final_C5 = compute_C5(best_h)
    return best_h, final_C5, n_points
```
