import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions for evaluation and initial construction
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.roll(h, shift=1)
    overlap = np.correlate(h, 1 - shifted_h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return np.load("initial_h_values.npy")

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(initial_h)
    best_h = initial_h.copy()
    best_c5 = current_c5
    
    # Define a function to compute the overlap for a given h
    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h_array = np.array(h)
        overlap = np.correlate(h_array, 1 - h_array, mode="full")
        return np.max(overlap) * dx
    
    # Define a function to compute the integral constraint
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return integral_constraint(h)
    
    # Optimization parameters
    num_iterations = 50
    step_size = 0.1
    tolerance = 1e-4
    
    for _ in range(num_iterations):
        # Compute current C5
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            print(f"Iteration {_+1}: Best C5 = {best_c5:.6f}")
        
        # Compute gradient estimate (approximate)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += step_size
            h_minus = best_h.copy()
            h_minus[i] -= step_size
            grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * step_size)
        
        # Update h using gradient descent
        best_h = best_h - step_size * grad
        
        # Enforce bounds [0, 1]
        best_h = np.clip(best_h, 0, 1)
        
        # Enforce integral constraint
        if np.abs(integral_constraint(best_h)) > 1e-6:
            # Adjust to satisfy the integral constraint
            delta = (n_points / 2 - np.sum(best_h)) / n_points
            best_h += delta
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
