def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    n = n_points
    dx = 2.0 / n
    initial_h = initial_h_values[:n]
    h = initial_h.copy()
    
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx

    def constraint_sum(h):
        return np.sum(h) - n / 2

    bounds = [(0, 1) for _ in range(n)]
    
    def evaluate(solution):
        nonlocal h
        h = solution
        return evaluate_erdos_solution(h)
    
    def optimize_solution():
        result = optimize.minimize(
            lambda x: objective(x),
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        return result
    
    def perturb_solution(solution):
        perturbation = np.random.normal(0, 0.05, size=len(solution))
        perturbed = np.clip(solution + perturbation, 0, 1)
        return perturbed
    
    best_solution = h
    best_value = evaluate(best_solution)
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        if np.random.rand() < 0.2:
            h = perturb_solution(h)
        else:
            result = optimize_solution()
            h = result.x
        current_value = evaluate(h)
        
        if current_value < best_value:
            best_value = current_value
            best_solution = h
    
    return best_solution, best_value, n
