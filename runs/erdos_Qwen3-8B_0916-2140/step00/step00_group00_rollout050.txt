import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values
    h = np.array(initial_h[:n_points], dtype=np.float64)
    h /= np.sum(h) * dx
    h *= n_points / 2

    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, h])
        c5 = 0
        for k in range(1, n_points):
            overlap = np.sum(h[:n_points] * (1 - h[k:n_points + k]))
            c5 = max(c5, overlap * dx)
        return c5

    def objective(x):
        x = np.array(x, dtype=np.float64)
        x = np.concatenate([x, x])
        total = np.sum(x[:n_points]) * dx
        if abs(total - 1) > 1e-8:
            return np.inf
        return compute_c5(x)

    def gradient(x):
        x = np.array(x, dtype=np.float64)
        x = np.concatenate([x, x])
        grad = np.zeros_like(x)
        total = np.sum(x[:n_points]) * dx
        for i in range(n_points):
            for k in range(1, n_points):
                overlap = np.sum(x[:n_points] * (1 - x[k:n_points + k]))
                if i < n_points:
                    for j in range(n_points):
                        if j < n_points:
                            grad[i] += (1 - x[k + j]) * dx
                        if j + k < n_points:
                            grad[i] -= x[j] * dx
        return grad

    def evaluate(x):
        x = np.array(x, dtype=np.float64)
        x = np.concatenate([x, x])
        total = np.sum(x[:n_points]) * dx
        if abs(total - 1) > 1e-8:
            return np.inf
        c5 = 0
        for k in range(1, n_points):
            overlap = np.sum(x[:n_points] * (1 - x[k:n_points + k]))
            c5 = max(c5, overlap * dx)
        return c5

    def objective_and_grad(x):
        x = np.array(x, dtype=np.float64)
        x = np.concatenate([x, x])
        total = np.sum(x[:n_points]) * dx
        if abs(total - 1) > 1e0:
            return np.inf, np.zeros_like(x)
        c5 = 0
        for k in range(1, n_points):
            overlap = np.sum(x[:n_points] * (1 - x[k:n_points + k]))
            c5 = max(c5, overlap * dx)
        return c5, gradient(x)

    def optimize_h(h0):
        h = h0.copy()
        bounds = [(0, 1) for _ in range(2 * n_points)]
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            method='L-BFGS-B',
            jac=gradient,
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000}
        )
        if result.success:
            return result.x[:n_points]
        return h0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        h = optimize_h(best_h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    return best_h, best_c5, n_points
