```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.linalg import norm
from scipy.optimize import Bounds
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 500
    h_initial = initial_h_values[:n_points]
    h_initial = np.array(h_initial, dtype=np.float64)
    h_initial = np.concatenate((h_initial, h_initial[::-1]))  # Mirror to double length

    # Ensure the initial h has the correct sum
    required_sum = n_points / 2
    current_sum = np.sum(h_initial)
    h_initial = h_initial * (required_sum / current_sum)

    # Define the cost function to minimize: C5
    def objective(h):
        h = np.array(h, dtype=np.float64)
        n = len(h)
        c5 = np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)
        return c5

    # Define constraints: sum(h) == n / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimizer settings
    bounds = Bounds(0, 1)
    result = minimize(objective, h_initial, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method="SLSQP")

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    n_points_final = len(best_h)

    # Check if we exceed budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (np.array([]), 1.0, 0)

    return (best_h, best_c5, n_points_final)
```
