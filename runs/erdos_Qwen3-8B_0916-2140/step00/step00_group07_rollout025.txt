import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy import linalg

def run(seed=42, budget_s=600):
    # Initial construction from initial_h_values (n=80)
    n_points = 80
    initial_h = initial_h_values
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h) * dx
    assert np.isclose(initial_sum, 1), "Initial h does not satisfy the integral constraint"

    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        return np.max(corr)

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    # Start with the initial guess
    h0 = initial_h.copy()
    result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'maxiter': 100})

    # If the result is not good enough, try a local search with random perturbations
    best_h = result.x
    best_c5 = compute_c5(best_h)
    best_time = time.time()

    if time.time() - best_time > budget_s * 0.7:
        # Try a few perturbations
        for _ in range(3):
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            h_perturb = np.clip(best_h + perturb, 0, 1)
            h_perturb = np.array(h_perturb)
            h_perturb = h_perturb / np.sum(h_perturb) * n_points / 2
            current_c5 = compute_c5(h_perturb)
            if current_c5 < best_c5:
                best_h = h_perturb
                best_c5 = current_c5

    # Simple greedy improvement: try to reduce the maximum overlap
    for _ in range(5):
        # Find the index with the highest overlap
        overlaps = []
        for i in range(n_points):
            h = best_h.copy()
            h[i] = 0
            overlaps.append(compute_c5(h))
        best_index = np.argmax(overlaps)
        best_h[best_index] = 0
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h / np.sum(best_h) * n_points / 2
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
