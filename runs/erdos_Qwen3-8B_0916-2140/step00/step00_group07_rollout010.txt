```python
import numpy as np
import scipy.optimize as opt
import scipy.interpolate as interpolate
import time
import multiprocessing
from functools import partial
import logging
import sys
import os
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    max_iterations = 100
    learning_rate = 0.1
    temp = 1.0
    cooling_rate = 0.995
    early_stop = 5
    best_c5 = float('inf')
    best_h = None
    best_iter = 0

    # Initial setup
    initial_h = initial_h_values
    initial_h = np.array(initial_h[:n_points])
    h = initial_h.copy()
    h = h / np.sum(h) * n_points / 2

    # Function to compute C5
    def compute_c5(h):
        h = h.reshape(-1)
        h = np.concatenate((h, h))
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[n_points:]
        dx = 2.0 / n_points
        c5 = np.max(corr * dx)
        return c5

    # Optimization loop
    for iteration in range(max_iterations):
        start_time = time.time()
        if iteration > 0:
            h = h + learning_rate * np.random.randn(n_points) / np.sqrt(n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_iter = iteration

        if iteration % 10 == 0:
            print(f"Iteration {iteration}, Best C5: {best_c5:.8f}")

        # Cooling schedule
        temp *= cooling_rate
        if temp < 1e-6:
            break

        # Early stopping if no improvement for some iterations
        if iteration - best_iter > early_stop:
            break

    return best_h, best_c5, n_points
```
