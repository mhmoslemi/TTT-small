# Precompute the cross-correlation for the current h_values
def compute_c5(h_values, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    return np.max(corr) * dx

def optimize_h(n_points, initial_guess=None, max_iterations=100):
    # Initialize h_values
    if initial_guess is None:
        h = np.random.rand(n_points)
    else:
        h = initial_guess.copy()
    # Normalize h to satisfy ∫h dx = 1
    h /= np.sum(h) * 2.0 / n_points
    # Optimization parameters
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Objective function: maximize overlap integral for the worst shift
    def objective(x):
        x = np.array(x)
        x /= np.sum(x) * 2.0 / n_points
        corr = np.correlate(x, 1 - x, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Minimize the maximum overlap
    def minimize_objective(x):
        return -objective(x)
    
    result = optimize.minimize(
        minimize_objective,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iterations}
    )
    return result.x

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 400  # As per the problem statement, n_points < 1000
    h = initial_h_values.copy()

    best_h = h
    best_c5 = compute_c5(h, n_points)
    start_time = time.time()
    
    # Optimization parameters
    max_iterations = 100
    # Try multiple starting points
    for i in range(5):
        if time.time() - start_time > budget_s:
            break
        h = np.random.rand(n_points)
        h /= np.sum(h) * 2.0 / n_points
        result = optimize_h(n_points, initial_guess=h, max_iterations=max_iterations)
        current_c5 = compute_c5(result, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
    
    return best_h, best_c5, n_points
