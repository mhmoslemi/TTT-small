import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial sample point count
    n_points = 200
    
    # Initial guess based on the provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        h = np.array(h)
        shifted_h = np.zeros_like(h)
        for k in range(1, 100):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * (2.0 / n_points)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: each h[i] must be in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h0 = h_initial
    
    # Optimization
    start_time = time.time()
    
    # Run optimization
    result = optimize.minimize(
        objective,
        h0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the result is within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Time exceeded budget. Returning initial solution with C5 = {objective(h_initial)}")
        return (h_initial, objective(h_initial), n_points)
    
    return (best_h, best_c5, n_points)
