import numpy as np

import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial solution (n_points=95)
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    # Evaluate the initial solution
    def evaluate_erdos_solution(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        return np.max(conv * dx)
    
    initial_c5 = evaluate_erdos_solution(initial_h_values)
    print(f"Initial C₅ value: {initial_c5:.6f}")
    
    # Define a function to compute the maximum overlap
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        return np.max(conv * dx)
    
    # Define a function to update h with a small perturbation
    def perturb_h(h, perturbation_size):
        h_perturb = h.copy()
        for i in range(len(h)):
            if np.random.rand() < perturbation_size:
                h_perturb[i] = np.clip(h_perturb[i] + np.random.uniform(-0.1, 0.1), 0, 1)
        return h_perturb
    
    # Define the optimization routine
    def optimize_h(initial_h, max_iter=100, perturbation_size=0.1):
        h = initial_h.copy()
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        
        for iteration in range(max_iter):
            start_time = time.time()
            
            # Perturb the current solution
            h_perturb = perturb_h(h, perturbation_size)
            
            # Ensure the integral constraint is satisfied
            sum_h_perturb = np.sum(h_perturb)
            required_sum = n_points / 2
            adjustment = (required_sum - sum_h_perturb) / n_points
            h_perturb += adjustment
            
            # Clip values to [0, 1]
            h_perturb = np.clip(h_perturb, 0, 1)
            
            # Compute overlap
            current_c5 = compute_overlap(h_perturb)
            
            if current_c5 < best_c5:
                best_h = h_perturb.copy()
                best_c5 = current_c5
                
            # Log progress
            elapsed_time = time.time() - start_time
            print(f"Iteration {iteration + 1}: C₅ = {current_c5:.6f}, elapsed time: {elapsed_time:.2f} s")
            
            # If we've used up the budget, return the best solution
            if elapsed_time + 10 > budget_s:
                print("Time budget exceeded. Returning best solution found.")
                return best_h, best_c5
            
            h = h_perturb
    
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(initial_h_values, max_iter=100, perturbation_size=0.1)
    
    # Return the result
    n_points = len(best_h)
    return best_h, best_c5, n_points
