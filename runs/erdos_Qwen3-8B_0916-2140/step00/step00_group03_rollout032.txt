```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial parameters
    max_n_points = 200
    n_points = max_n_points
    max_iterations = 100
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = None
    
    # Start with the initial solution
    h = initial_h_values.copy()
    h = np.array(h)
    
    # Evaluate initial solution
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Optimization
    for iteration in range(max_iterations):
        start_time = time.time()
        if time.time() - start_time > budget_s:
            break
        
        # Random perturbation
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h + perturb, 0, 1)
        h_perturb = np.where(h_perturb <= 1, h_perturb, 1)
        h_perturb = np.where(h_perturb >= 0, h_perturb, 0)
        
        # Check normalization
        total = np.sum(h_perturb) * (2.0 / n_points)
        if abs(total - 1.0) > 1e-6:
            h_perturb = h_perturb * (1.0 / total) * (2.0 / n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h_perturb)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturb.copy()
        
        # Check if we can reduce n_points
        n_points = max(50, n_points - 50)
        h = np.zeros(n_points)
        h = np.where(h_perturb[:n_points] <= 1, h_perturb[:n_points], 1)
        h = np.where(h_perturb[:n_points] >= 0, h_perturb[:n_points], 0)
        total = np.sum(h) * (2.0 / n_points)
        if abs(total - 1.0) > 1e-6:
            h = h * (1.0 / total) * (2.0 / n_points)
        
        # Use a local optimizer
        def objective(x):
            h_vec = x
            # Normalize
            total = np.sum(h_vec) * (2.0 / n_points)
            if abs(total - 1.0) > 1e-6:
                h_vec = h_vec * (1.0 / total) * (2.0 / n_points)
            # Compute the max overlap integral
            c5_val = evaluate_erdos_solution(h_vec)
            return c5_val
        
        # Local optimization
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )
        h = result.x
        
        # Check if we can reduce n_points further
        n_points = max(50, n_points - 50)
        h = np.zeros(n_points)
        h = np.where(h <= 1, h, 1)
        h = np.where(h >= 0, h, 0)
        total = np.sum(h) * (2.0 / n_points)
        if abs(total - 1.0) > 1e-6:
            h = h * (1.0 / total) * (2.0 / n_points)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
```
