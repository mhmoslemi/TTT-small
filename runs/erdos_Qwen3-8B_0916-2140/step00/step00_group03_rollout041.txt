import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess from pre-defined initial_h_values
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if np.abs(sum_h * dx - 1.0) > 1e-8:
        h = h / (sum_h * dx) * 1.0

    # Function to compute C5
    def compute_C5(h):
        dx = 2.0 / len(h)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:len(h) - k]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Objective function for optimization
    def objective(x):
        h = x.copy()
        dx = 2.0 / len(h)
        sum_h = np.sum(h)
        if np.abs(sum_h * dx - 1.0) > 1e-8:
            h = h / (sum_h * dx) * 1.0
        return compute_C5(h)

    # Optimization using L-BFGS-B with restarts
    def optimize_with_restarts(h, max_restarts=5, max_iter=100):
        best_h = h.copy()
        best_c5 = compute_C5(best_h)
        for restart in range(max_restarts):
            try:
                result = optimize.minimize(
                    objective,
                    h,
                    method='L-BFGS-B',
                    bounds=[(0, 1) for _ in range(len(h))],
                    options={'maxiter': max_iter, 'disp': False}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                continue
        return best_h, best_c5

    # Run optimization
    optimized_h, final_c5 = optimize_with_restarts(h)

    # Ensure sum of h * dx is exactly 1
    dx = 2.0 / len(optimized_h)
    sum_h = np.sum(optimized_h)
    optimized_h = optimized_h / (sum_h * dx) * 1.0

    # Final computation
    final_c5 = compute_C5(optimized_h)

    # Output
    n_points = len(optimized_h)
    return optimized_h, final_c5, n_points
