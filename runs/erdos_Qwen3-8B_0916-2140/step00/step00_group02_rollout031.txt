import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with 100 points for better resolution
    dx = 2.0 / n_points
    target_C5 = 0.380876
    max_iter = 50
    step_size = 0.01
    tolerance = 1e-4
    best_C5 = float('inf')
    best_h = None
    
    # Initial guess
    h = copy.copy(initial_h_values)
    h = np.array(h[:n_points])
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points//2]])
        h = np.pad(h, (0, n_points), mode='constant', constant_values=0)
        conv = np.correlate(h, 1 - h, mode="full")
        C5 = np.max(conv) * dx
        return C5
    
    # Function to perform gradient descent
    def optimize_h(h, max_iter=50, step_size=0.01, tolerance=1e-4):
        h = h.copy()
        for i in range(max_iter):
            # Compute current C5
            current_C5 = compute_C5(h)
            
            # Compute gradients
            grads = np.zeros_like(h)
            for k in range(1, n_points):
                if k >= len(h):
                    break
                conv = np.correlate(h, 1 - h, mode="full")
                grads = np.zeros_like(h)
                for pos in range(len(h)):
                    for shift in range(len(h)):
                        if shift + pos >= len(h):
                            break
                        val = conv[pos + shift] * dx
                        grads[pos] += val
                grads = grads / (np.max(grads) + 1e-10)
            
            # Update h
            h = np.clip(h - grads * step_size, 0, 1)
            
            # Check for convergence
            if i > 0 and np.abs(current_C5 - compute_C5(h)) < tolerance:
                break
            
            if i % 10 == 0:
                print(f"Iter {i}: C5 = {compute_C5(h)}")
        
        return h
    
    # Run the optimization
    h = optimize_h(h, max_iter=50, step_size=0.01, tolerance=1e-4)
    
    # Evaluate the best solution
    final_C5 = compute_C5(h)
    
    # Return the result
    return h.tolist(), final_C5, n_points
