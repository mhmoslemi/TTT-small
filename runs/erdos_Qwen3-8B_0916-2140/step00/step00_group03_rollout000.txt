import numpy as np
import scipy.optimize as opt
import scipy.sparse as sparse
import time

# Preloaded functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder for actual initial values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define parameters
    max_n_points = 500  # Max number of samples
    n_points = 500  # Start with 500 samples
    max_iter = 100  # Max number of iterations
    tol = 1e-4  # Tolerance for convergence
    
    # Initial guess based on provided initial_h_values or random
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points) * 2 - 1  # Random initial guess in [-1, 1]
    
    # Ensure integral constraint
    total = np.sum(h) * (2.0 / n_points)
    h = h * (1 / total) * (2.0 / n_points)  # Scale to satisfy integral constraint
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    for _ in range(max_iter):
        # Objective function: minimize the maximum overlap integral
        def objective(x):
            nonlocal n_points
            h_new = x.copy()
            total = np.sum(h_new) * (2.0 / n_points)
            h_new = h_new * (1 / total) * (2.0 / n_points)
            c5 = evaluate_erdos_solution(h_new)
            return c5
        
        # Gradient descent step
        grad = np.gradient(h)  # Approximate gradient
        step_size = 0.1
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        total = np.sum(h_new) * (2.0 / n_points)
        h_new = h_new * (1 / total) * (2.0 / n_points)
        
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Early stopping if we're close to the target
        if best_c5 <= 0.380876 + 1e-5:
            break
        
        h = h_new
    
    end_time = time.time()
    
    # If budget is exceeded, return best solution
    if end_time - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
