import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 200  # Start with 200 points for balance between speed and accuracy
    dx = 2.0 / n_points

    # Initial guess from provided initial_h_values
    h_initial = initial_h_values
    if len(h_initial) != n_points:
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    best_h = h_initial.copy()

    # Function to compute overlap integral
    def compute_overlap(h):
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * dx)
        return overlap

    # Function to optimize h to minimize the max overlap
    def objective(h):
        return compute_overlap(h)

    # Gradient descent with momentum for optimization
    def optimize_h(h_initial):
        h = h_initial.copy()
        learning_rate = 0.1
        momentum = 0.9
        max_iter = 1000
        grad = np.zeros_like(h)
        for _ in range(max_iter):
            # Compute overlap and gradient
            corr = np.correlate(h, 1 - h, mode="full")
            corr = corr * dx
            max_corr = np.max(corr)
            # Compute gradient
            for i in range(n_points):
                for k in range(n_points):
                    if i + k < n_points:
                        corr_i_k = corr[i + k]
                        grad[i] += (1 - h[i + k]) * corr_i_k
                    if i - k >= 0:
                        corr_i_k = corr[i - k]
                        grad[i] -= h[i - k] * corr_i_k
            # Update h
            grad = np.clip(grad / np.abs(grad).max() if np.abs(grad).max() > 0 else grad, -1, 1)
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            # Ensure integral constraint
            h = h * (n_points / 2) / np.sum(h)
        return h

    # Run optimization
    h_opt = optimize_h(best_h)
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt

    # Try reducing n_points to speed up further steps
    n_points = 100
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    # Run gradient descent with momentum
    h_opt = optimize_h(h)
    best_c5 = compute_overlap(h_opt)
    best_h = h_opt

    # Final evaluation
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        return best_h, best_c5, n_points

    return best_h, best_c5, n_points
