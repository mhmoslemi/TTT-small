import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial guess from the provided initial_h_values
    initial_h_values = np.array([0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9,
                                 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9,
                                 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9,
                                 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9,
                                 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9, 0.9])
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    # Objective function
    def objective(h):
        h = h.reshape(-1)
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Constraints: sum of h * dx = 1
    def constraint_sum(h):
        h = h.reshape(-1)
        return np.sum(h) - n_points / 2

    # Bounds: h[i] ∈ [0, 1]
    bounds = Bounds(0, 1)

    # Gradient (approximate via finite differences)
    def grad_objective(h):
        h = h.reshape(-1)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * 1e-6)
        return grad

    # Initial guess
    initial_guess = initial_h_values.copy()

    # Optimization - use L-BFGS-B for efficiency
    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        jac=grad_objective,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000}
    )

    # If the optimization takes too long, we fallback to a simple grid search
    if time.time() - start_time > budget_s * 0.8:
        # Fallback to a grid search
        h_values = np.random.uniform(0, 1, n_points)
        h_values = h_values / np.sum(h_values) * n_points / 2

        # Compute C5
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        c5_bound = np.max(corr) * (2.0 / n_points)
    else:
        # Use the result from the L-BFGS-B optimization
        h_values = result.x
        c5_bound = result.fun

    return h_values, c5_bound, n_points
