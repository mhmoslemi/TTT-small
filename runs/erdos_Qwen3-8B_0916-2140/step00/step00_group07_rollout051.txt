def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess using the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def compute_C5(h):
        h = np.array(h)
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k < 0:
                shifted_h = np.roll(h, k)
            else:
                shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    def objective(h):
        return compute_C5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    # Use a local optimizer with a custom initial guess
    result = minimize(objective, x0=initial_h, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                      method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 1000})
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    # Refinement step with a grid search over small perturbations
    refinement_attempts = 3
    for _ in range(refinement_attempts):
        perturbations = np.random.uniform(-0.05, 0.05, size=n_points)
        refined_h = np.clip(best_h + perturbations, 0, 1)
        refined_C5 = compute_C5(refined_h)
        if refined_C5 < best_C5:
            best_C5 = refined_C5
            best_h = refined_h
    
    # Final refinement with a local optimizer
    final_result = minimize(objective, x0=best_h, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}],
                           method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 500})
    
    best_h = final_result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
