```python
import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import LinearConstraint
import pickle
import os

def run(seed=42, budget_s=600):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define the initial number of samples and initial h_values
    n_points_initial = 95
    initial_h_values = np.load("initial_h_values.npy")
    
    # Define the number of samples to use for the current run
    n_points = 500  # Total number of samples
    dx = 2.0 / n_points
    
    # Define the integral constraint
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the objective function to minimize the max overlap
    def objective(h):
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx
        return np.max(overlap)
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define constraints
    constraint = LinearConstraint(np.ones(n_points), lb=0.0, ub=n_points / 2)
    
    # Initial guess from the initial_h_values scaled to the current n_points
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), initial_h_values)
    
    # Define the optimization problem
    result = minimize(
        fun=objective,
        x0=initial_h,
        bounds=bounds,
        constraints=[constraint],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    # Return the result
    return (best_h, best_c5, n_points)
```
