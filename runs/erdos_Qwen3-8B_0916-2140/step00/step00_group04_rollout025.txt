import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.concatenate([h[-2:], h])
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return [0.5] * 42

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    h_initial = initial_h_values()
    h_initial = np.array(h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    def objective(h_vals):
        h = np.array(h_vals)
        shifted_h = np.concatenate([h[-2:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Optimization using Nelder-Mead
    result = opt.minimize(
        objective,
        x0=h_initial,
        method='Nelder-Mead',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-6, 'xatol': 1e-6, 'maxiter': 1000}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h.tolist(), best_c5, n_points
