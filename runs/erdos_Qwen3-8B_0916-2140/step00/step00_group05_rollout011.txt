    def compute_c5(h):

        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Only the relevant part of the correlation (length = n_points)
        corr = corr[n_points - 1:]
        # Multiply by dx and find the maximum
        c5 = np.max(corr * dx)
        return c5

    # Objective function for optimization
    def objective(h_vector):
        h = h_vector.reshape(-1)
        dx = 2.0 / n_points
        # Ensure h is within [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Only the relevant part of the correlation (length = n_points)

        # Multiply by dx and find the maximum
        c5 = np.max(corr * dx)
        return c5
