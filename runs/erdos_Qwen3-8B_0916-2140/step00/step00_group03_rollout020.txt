```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize.slsqp import minimize_slsqp
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve
import warnings

# Initial construction
initial_h_values = np.loadtxt("initial_h_values.txt")

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:n_points - k]
    overlap = np.abs(np.correlate(h, 1 - h, mode="full")) * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values

    n_points = len(best_h)
    dx = 2.0 / n_points
    n_shifts = n_points - 1
    max_shifts = 10  # We'll consider the top 10 shifts with highest overlap

    # Initial evaluation
    current_c5 = evaluate_erdos_solution(best_h)
    best_c5 = current_c5
    print(f"Initial C₅: {best_c5:.6f}")

    # Define a function to compute the overlap for a given shift
    def compute_overlap(h, shift):
        h = np.array(h)
        shifted = np.zeros_like(h)
        for i in range(shift, len(h)):
            shifted[i] = h[i - shift]
        overlap = np.abs(np.correlate(h, 1 - h, mode="full")) * dx
        return overlap[shift]

    # Function to optimize with constraints
    def objective(x):
        nonlocal best_c5, best_h
        h = x.reshape(-1)
        # Compute overlap for the first few shifts
        overlaps = []
        for shift in range(1, 11):  # Check first 10 shifts
            if shift >= len(h):
                break
            shifted = np.roll(h, shift)
            overlap = np.abs(np.correlate(h, 1 - shifted, mode="full")) * dx
            overlaps.append(overlap[0])
        return np.max(overlaps)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    constraint = LinearConstraint(np.ones(n_points), 0, 0, name="sum_constraint")
    bounds = Bounds(0, 1, name="bounds")

    # Start with the initial guess
    x0 = initial_h_values.copy()
    result = minimize_slsqp(objective, x0, constraints=[constraint], bounds=bounds, method="SLSQP", ftol=1e-8, maxiter=100)
    h = result.x
    current_c5 = evaluate_erdos_solution(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h
        print(f"Improved C₅: {best_c5:.6f} with {len(h)} points")

    # Simple random search
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.random.rand(n_points)
        h = (h * n_points / 2).astype(float)
        h = h / np.sum(h) * n_points / 2
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
            print(f"Improved C₅: {best_c5:.6f} with {len(h)} points")

    # Return the best solution
    return best_h, best_c5, len(best_h)
```
