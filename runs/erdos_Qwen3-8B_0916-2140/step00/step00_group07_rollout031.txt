import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
import seaborn as sns
import warnings
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
from scipy.linalg import norm
from jax import grad, jit, vmap
import jax.numpy as jnp
import jax.random as jr
import jax

# Predefined initial h_values
initial_h_values = np.load('initial_h_values.npy')  # Expected to have shape (n_points,)

# Define the run function
def run(seed=42, budget_s=600, **kwargs):
    # Set the random seed
    np.random.seed(seed)
    start_time = time.time()
    
    # Predefined initial h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # The overlap integral is computed as the max of the cross-correlation
    def compute_C5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Objective function to minimize
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = initial_h.copy()
    
    # Use a more efficient solver with bounds and constraints
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    # Extract the best solution
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    # Track time elapsed
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Exceeded time budget, returning best solution found so far")
    
    return best_h, best_C5, n_points
