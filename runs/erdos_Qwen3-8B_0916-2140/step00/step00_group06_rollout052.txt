import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 512  # Reasonable balance between resolution and computation
    dx = 2.0 / n_points
    max_iter = 100
    tol = 1e-5
    h = np.random.rand(n_points) * 2 - 1  # Initial guess
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint

    def compute_overlap(h):
        h_reshaped = np.concatenate([h, np.zeros(n_points)])
        shifted = np.roll(h_reshaped, np.arange(1, n_points + 1))
        overlap = np.sum(h_reshaped * (1 - shifted))
        return overlap * dx

    def compute_C5(h):
        overlaps = []
        for k in range(1, n_points):
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)

    def objective(h):
        return compute_C5(h)

    # Optimization with limited iterations
    for iteration in range(max_iter):
        start_time = time.time()
        if iteration == 0:
            # Initial run using scipy.optimize
            result = optimize.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 100, 'ftol': tol}
            )
            h = result.x
        else:
            # Run for a few iterations with different initializations
            # Start from previous solution with some noise
            h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            result = optimize.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 50, 'ftol': tol}
            )
            h = result.x
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

    # Final evaluation
    c5_bound = compute_C5(h)
    return h, c5_bound, n_points
