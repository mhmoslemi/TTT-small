        h = np.array(h, dtype=np.float64)
        # Compute the overlap integral for all k
        max_overlap = 0.0
        for k in range(1, n_points):
            # Shift by k samples
            shifted = np.roll(h, k)
            # Compute the integral of h(x)(1 - h(x+k))
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Bounds for h: each value must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    x0 = initial_h

    # Optimization settings
    max_iter = 100
    tol = 1e-8
    start_time = time.time()

    result = minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': tol}
    )

    best_h = result.x
    best_c5 = objective(best_h)
