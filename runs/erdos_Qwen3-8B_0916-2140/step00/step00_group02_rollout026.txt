```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # We'll use a moderate number of points for efficiency
    dx = 2.0 / n_points
    max_k = 2  # We'll consider shifts up to 2

    # Define initial guess
    h = np.array(initial_h_values[:n_points])
    h = np.interp(h, (0, 2), (0, 2))  # Ensure it maps to [0, 2]
    h = np.interp(h, (0, 2), (0, 1))  # Rescale to [0, 1]

    # Ensure the integral is exactly 1
    h = h * (n_points / 2) / np.sum(h)

    def compute_C5(h):
        h = h.reshape(-1)
        shifts = np.arange(0, max_k + dx, dx)
        overlaps = []
        for k in shifts:
            shifted_h = np.roll(h, int(k / dx))
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)

    def objective(h):
        return compute_C5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Use L-BFGS-B as it's efficient for constrained problems
    result = optimize.minimize(
        objective,
        x0=h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    best_h = result.x
    best_C5 = compute_C5(best_h)

    # Return the best solution found
    return (best_h, best_C5, n_points)
```
