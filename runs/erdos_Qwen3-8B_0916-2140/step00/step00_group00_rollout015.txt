    def compute_C5(h):
        h = np.array(h)
        shifts = np.arange(-1, 2)  # Shifts to consider for C5 calculation
        overlaps = []
        for k in shifts:
            if k == 0:
                continue  # Skip k=0 as it's not relevant
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h))
            overlaps.append(overlap)
        return np.max(overlaps) * (2.0 / n_points)
    
    # Define the optimization problem
    def objective(x):
        return compute_C5(x)
    
    def constraint(x):
        return np.sum(x) - n_points / 2

    x0 = initial_h.copy()
    
    # Optimization with bounds
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    # Perform optimization
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=({'type': 'eq', 'fun': constraint}))
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    # Return the result
