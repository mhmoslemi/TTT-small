import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Global constants
MAX_BUDGET = 600
MAX_ITERATIONS = 100
MAX_EVALS = 100
INITIAL_H = initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initialize the best solution
    best_h = np.array(INITIAL_H, dtype=float)
    best_c5 = evaluate_erdos_solution(best_h)
    best_n = len(best_h)
    
    # Define the evaluation function
    def evaluate(h):
        n = len(h)
        dx = 2.0 / n
        h = np.array(h, dtype=float)
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(corr) * dx)
        return c5
    
    # Define the optimization function
    def optimize_h(h):
        nonlocal best_h, best_c5, best_n
        
        # Optimization parameters
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.array(h)
        
        # Evaluate current solution
        current_c5 = evaluate(h)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n = n_points
        
        return current_c5
    
    # Optimization process
    start_time = time.time()
    
    # Initial evaluation
    initial_c5 = evaluate(best_h)
    best_c5 = initial_c5
    
    # Run optimization
    for _ in range(MAX_ITERATIONS):
        # Create a perturbed version of the current solution
        perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(perturbed_h) * 2.0 / len(perturbed_h)
        scale_factor = 1.0 / total if total != 0 else 1.0
        perturbed_h *= scale_factor
        
        # Evaluate the perturbed solution
        current_c5 = evaluate(perturbed_h)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = perturbed_h.copy()
            best_n = len(perturbed_h)
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    best_c5 = final_c5
    
    # Return the results
    return best_h, best_c5, best_n
