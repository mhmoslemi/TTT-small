```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import LinearConstraint, Bounds
from scipy.sparse import csc_matrix
from scipy.sparse.linalg import lsqr
import numba
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points to balance speed and accuracy
    
    # Initial function to evaluate C5 for a given h
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h])  # Mirror to handle circular shifts
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[n_points:]
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Initial guess
    initial_h = initial_h_values[:]
    initial_h = np.array(initial_h)
    initial_h = np.concatenate([initial_h, initial_h])
    initial_h = initial_h[:n_points]
    initial_c5 = evaluate_c5(initial_h)
    best_c5 = initial_c5
    best_h = initial_h.copy()
    
    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate random samples for initial guess
        h = np.random.rand(n_points) * 2.0 - 1.0
        h = np.clip(h, 0.0, 1.0)
        h = (h - np.mean(h)) / np.std(h)  # Normalize
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2)  # Adjust for integral constraint
        h = np.random.permutation(h)
        h = np.clip(h, 0.0, 1.0)
        h = h + np.random.randn(n_points) * 0.05
        h = np.clip(h, 0.0, 1.0)
        
        # Optimization using gradient descent
        h_opt = h.copy()
        for _ in range(100):
            h_opt = h_opt * (1.0 - 0.01) + 0.01 * np.random.randn(n_points)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            
            # Evaluate C5
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        
        # Try a few more restarts
        for _ in range(5):
            h_restart = np.random.rand(n_points)
            h_restart = np.clip(h_restart, 0.0, 1.0)
            h_restart = h_restart * (n_points / 2)
            h_restart = np.random.permutation(h_restart)
            h_restart = np.clip(h_restart, 0.0, 1.0)
            
            # Evaluate C5
            c5 = evaluate_c5(h_restart)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_restart.copy()
    
    # Final evaluation and return
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
```
