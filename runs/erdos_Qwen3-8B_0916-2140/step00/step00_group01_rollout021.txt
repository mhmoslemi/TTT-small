```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy import stats
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # A 47-point initial construction
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # starting with 100 points for balance between speed and quality
    h = np.copy(initial_h_values)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    time_start = time.time()

    # Try a few starting points with higher n_points to explore
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - time_start > budget_s - 10:
            break

        # Resample h to n_points
        x = np.linspace(0, 2, n_points)
        h_resampled = np.interp(x, np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h_resampled = np.clip(h_resampled, 0, 1)
        h_resampled = np.random.permutation(h_resampled)
        h_resampled = np.sort(h_resampled)

        # Define the objective function
        def objective(h_vec):
            h = h_vec.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5_val = np.max(overlap) * (2.0 / len(h))
            return c5_val

        # Define constraints
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2

        # Bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]

        # Initial guess
        h_initial = h_resampled

        # Run optimization
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        if result.success:
            h_opt = result.x
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                print(f"New best C₅: {best_c5} with n_points = {n_points}")

    # Final check with the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
```
