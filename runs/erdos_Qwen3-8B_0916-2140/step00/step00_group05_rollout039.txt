```python
import numpy as np
import time
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import sys
import os
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600):
    # Set random seed
    random.seed(seed)
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    max_iter = 100
    tolerance = 1e-4
    initial_h = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h[:n_points]  # Ensure we work with n_points samples
    
    # Ensure the initial_h has the correct total integral
    total_integral = np.sum(initial_h) * dx
    scaling_factor = 1.0 / total_integral if total_integral != 0 else 0.5
    initial_h = np.clip(initial_h * scaling_factor, 0, 1)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Define constraints: sum(h) * dx == 1, i.e., sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Set up bounds for each variable to be between 0 and 1
    bounds = Bounds(0, 1)
    
    # Use Sequential Least Squares Programming (SLSQP) for constrained optimization
    result = minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': max_iter, 'ftol': tolerance, 'disp': False}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = result.fun
    
    # Use a local search to refine the solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, size=len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        current_c5 = evaluate_erdos_solution(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
    
    # Early stopping if the time budget is exceeded
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)

start_time = time.time()
```
