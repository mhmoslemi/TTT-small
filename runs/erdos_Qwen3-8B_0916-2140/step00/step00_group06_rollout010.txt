    def compute_C5(h):
        h = np.array(h)

        corr = np.correlate(h, 1 - h, mode="full")
        # Only the relevant part of the correlation is the one that corresponds to the overlap
        # The overlap is between 0 and 2, which maps to the central part of the correlation

        # Compute the maximum overlap integral
        max_overlap = np.max(relevant_corr * dx)
        return max_overlap

    def objective(h):
        return compute_C5(h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = optimize.minimize(
