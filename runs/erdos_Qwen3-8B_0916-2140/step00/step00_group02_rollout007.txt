```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction from initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Function to compute the loss for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Sequential Least Squares Programming (SLSQP) for optimization
    result = optimize.minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Get the best solution found
    best_h = result.x
    best_c5 = compute_c5(best_h)
    best_n_points = len(best_h)
    
    # If we have not improved significantly, try a smaller n_points
    if best_c5 > 0.380876:
        # Try a smaller n_points (e.g., 100)
        n_points_new = 100
        dx_new = 2.0 / n_points_new
        # Initial guess for smaller n_points
        h_new = np.zeros(n_points_new)
        # Fill with 0.5 to satisfy the integral constraint
        h_new[:int(n_points_new / 2)] = 0.5
        # Add random noise to break symmetry
        h_new += np.random.uniform(-0.1, 0.1, n_points_new)
        h_new = np.clip(h_new, 0, 1)
        # Enforce the integral constraint
        total = np.sum(h_new) * dx_new
        h_new = h_new * (1 / total) * (n_points_new / 2)
        
        # Optimize again with smaller n_points
        result = optimize.minimize(
            objective,
            h_new,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        best_h = result.x
        best_c5 = compute_c5(best_h)
        best_n_points = len(best_h)
    
    # Check if we have exceeded the time budget
    if time.time() - start_time > budget_s:
        print("Time budget exceeded. Returning best solution found.")
    
    return best_h, best_c5, best_n_points
```
