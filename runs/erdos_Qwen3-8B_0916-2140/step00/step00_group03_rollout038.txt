import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from scipy.integrate import trapz
from scipy.signal import correlate
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define initial h_values from initial_h_values
    try:
        from initial_h_values import initial_h_values
    except:
        initial_h_values = np.random.rand(70) * 2 - 1
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = (initial_h_values * 100).astype(int) / 100
        initial_h_values = np.clip(initial_h_values, 0, 1)
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if sum_h != n_points / 2:
        initial_h_values = (initial_h_values * (n_points / 2) / sum_h)
    
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_array = np.array(h)
        shifted_h = np.array([h_array[i:] for i in range(n)])
        max_overlap = 0
        for k in range(1, n):
            if k >= n:
                break
            overlap = np.sum(h_array[:n - k] * (1 - h_array[k:]))
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess from initial_h_values
    initial_guess = initial_h_values
    
    # Setup optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Refine with local search
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, size=len(best_h))
        perturb = np.clip(perturb, -0.1, 0.1)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = (new_h * n_points / 2) / np.sum(new_h)
        c5 = compute_c5(new_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h
    
    # Final refinement
    for _ in range(10):
        perturb = np.random.normal(0, 0.02, size=len(best_h))
        perturb = np.clip(perturb, -0.05, 0.05)
        new_h = np.clip(best_h + perturb, 0, 1)
        new_h = (new_h * n_points / 2) / np.sum(new_h)
        c5 = compute_c5(new_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h
    
    # Final check and return
    best_h = np.clip(best_h, 0, 1)
    best_h = (best_h * n_points / 2) / np.sum(best_h)
    
    return best_h, best_c5, len(best_h)
