```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for balance between accuracy and speed

    def objective(h):
        nonlocal best_c5
        h = np.array(h)
        dx = 2.0 / len(h)
        shifted_h = np.zeros_like(h)
        for k in range(len(h)):
            shifted_h[k:] = h[:len(h) - k]
        kernel = 1 - h
        correlation = np.correlate(h, kernel, mode="full")
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Initial guess
    initial_h = initial_h_values
    best_h = np.array(initial_h)
    best_c5 = objective(best_h)

    # Try reducing n_points to speed up
    for n_points in [50, 40, 30, 20, 10]:
        # Generate initial guess for smaller n_points
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 if i % 3 == 0 else 0.25 if i % 2 == 0 else 0.75
        h = h / np.sum(h) * n_points / 2

        # Optimize
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Final check with original n_points
    if best_c5 > 0.38:
        h = np.array(initial_h)
        best_c5 = objective(h)

    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print(f"Time budget exceeded. Returning best found solution with C5 = {best_c5}")
    return best_h, best_c5, len(best_h)
```
