import numpy as np
import scipy.optimize as opt
import time
import multiprocessing
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    n_points = 256  # Reasonable number of points for initial exploration
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-4

    # Initial guess based on provided initial_h_values
    h = np.array(initial_h_values)
    h = np.interp(h, [0, len(h)-1], [0, 2])  # Map to [0, 2]
    h = np.interp(h, [0, 2], [0, 1])  # Map to [0, 1]
    h = np.interp(h, [0, 2], [0, 1])  # Map to [0, 1]
    h = np.interp(h, [0, 2], [0, 1])  # Map to [0, 1]
    h = np.interp(h, [0, 2], [0, 1])  # Map to [0, 1]
    h = np.interp(h, [0, 2], [0, 1])  # Map to [0, 1]

    # Ensure the sum of h is n_points / 2
    total = np.sum(h)
    h = h * (n_points / 2) / total

    # Optimization function to minimize the maximum overlap
    def objective(h):
        h = np.clip(h, 0, 1)
        shifted = np.roll(h, shift=1)
        overlap = np.maximum(h - shifted, 0)
        return np.max(overlap)

    # Initial objective value
    initial_obj = objective(h)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', tol=tolerance, options={'maxiter': max_iterations})
    best_h = result.x

    # Ensure the sum of h is n_points / 2
    total = np.sum(best_h)
    best_h = best_h * (n_points / 2) / total

    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h)

    return best_h, c5_bound, n_points
