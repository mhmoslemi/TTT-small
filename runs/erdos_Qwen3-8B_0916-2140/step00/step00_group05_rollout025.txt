import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction with n_points = 70
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if sum_h != n_points / 2:
        initial_h_values = (initial_h_values / sum_h) * (n_points / 2)

        return np.max(corr) * dx
    
    def objective(h):
        return compute_C5(h)
    
    def constraints(h):
        return np.sum(h) - n_points / 2
    
    def bounds(h):
        return [(0, 1) for _ in h]
    
    def optimize_solution(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            bounds=bounds(h_initial),
            constraints={
                'type': 'eq',
                'fun': constraints,
                'jac': lambda h: np.ones_like(h)
            },
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
            tol=1e-8
        )
        return result.x, result.fun
    
    best_h = initial_h_values.copy()
    best_c5 = compute_C5(best_h)
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        h_initial = np.clip(np.random.uniform(0, 1, n_points), 0, 1)
        h_initial = (h_initial * n_points / 2) / np.sum(h_initial)
        
        result = optimize_solution(h_initial)
        current_c5 = result[1]
        if current_c5 < best_c5:
            best_h = result[0]
            best_c5 = current_c5
    
    # Further refinement with small perturbations
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = (h_perturbed * n_points / 2) / np.sum(h_perturbed)
        
        result = optimize_solution(h_perturbed)
        current_c5 = result[1]
        if current_c5 < best_c5:
            best_h = result[0]
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
