import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
from scipy.optimize import minimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points for a balance between speed and accuracy
    dx = 2.0 / n_points
    
    # Initial guess: use the provided initial_h_values and scale to match integral constraint
    h_initial = initial_h_values
    h_initial = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.linspace(0, 2, len(h_initial)))), h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.linspace(0, 2, len(h_initial)))), h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.linspace(0, 2, len(h_initial)))), h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.linspace(0, 2, len(h_initial)))), h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.linspace(0, 2, len(h_initial)))), h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(np.linspace(0, 2, len(h_initial)))), h_initial)
    
    # Rescale to meet the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h = h_initial.copy()
    
    # Optimization parameters
    max_iterations = 100
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    start_time = time.time()
    for _ in range(max_iterations):
        # Compute the current C5 value
        shifted_h = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted_h[k:] = h[:len(h) - k]
        overlap = np.sum(h * (1 - shifted_h)) * dx
        current_c5 = np.max(overlap)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Try perturbing the current solution
        perturbation = np.random.normal(0, 0.05, len(h))
        perturbation = np.clip(perturbation, -0.1, 0.1)
        h = np.clip(h + perturbation, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
