```python
import numpy as np
import scipy.optimize as opt
import time
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Example initial construction

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with moderate resolution
    
    # Define cost function to minimize
    def cost(h):
        nonlocal best_c5
        if len(h) != n_points:
            return float('inf')  # Invalid length
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Initial trial with provided initial_h_values
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
        h = np.clip(h, 0, 1)
        if np.sum(h) * (2.0 / n_points) != 1:
            h = np.random.uniform(0, 1, n_points)
            np.random.shuffle(h)
            np.sum(h) = n_points / 2
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try reducing n_points
    for n in range(50, 20, -1):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.random.uniform(0, 1, n)
        h = np.random.permutation(h)
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        h = np.random.permutation(h)
        np.sum(h) = n / 2
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try local optimization
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.random.uniform(0, 1, n_points)
        h = np.random.permutation(h)
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        h = np.random.permutation(h)
        np.sum(h) = n_points / 2
        # Simple hill climbing
        for i in range(n_points):
            for delta in [-0.05, 0.05]:
                h_new = h.copy()
                h_new[i] += delta
                h_new = np.clip(h_new, 0, 1)
                np.sum(h_new) = n_points / 2
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    
    # Final check with best_h
    final_h = best_h
    final_c5 = evaluate_erdos_solution(final_h)
    return final_h, final_c5, len(final_h)
```
