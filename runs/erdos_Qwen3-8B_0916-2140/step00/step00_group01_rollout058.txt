def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    shifted = np.roll(h, shift=1)
    overlap = np.correlate(h, 1 - shifted, mode="full")
    max_overlap = np.max(overlap)
    return max_overlap * dx
