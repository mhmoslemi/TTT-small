```python
import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    # Pre-defined initial solution
    initial_h_values = np.load('initial_h_values.npy')  # Load precomputed initial solution
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()

    def compute_overlap(h):
        h = np.array(h)
        shift = np.arange(-n_points + 1, n_points)
        overlap = np.zeros_like(shift)
        for k in range(1, len(shift)):
            shifted_h = np.roll(h, k)
            overlap[k] = np.dot(h, 1 - shifted_h) * dx
        return np.max(overlap)

    def objective(h):
        return compute_overlap(h)

    # Bounds for h: between 0 and 1
    bounds = Bounds(0, 1)

    # Define constraints:
    # Constraint: sum(h) * dx == 1 => sum(h) == n_points / 2
    def constraint_func(h):
        return np.dot(h, np.ones_like(h)) - n_points / 2

    constraint = LinearConstraint(np.ones(n_points), 0, 0, name='integral_constraint')

    # Initial guess
    h_initial = h_initial.copy()

    # Use L-BFGS-B optimizer with bounds and constraints
    start_time = time.time()
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time limit exceeded, returning initial solution")
        return (initial_h_values, compute_overlap(initial_h_values), n_points)

    best_h = result.x
    best_c5 = compute_overlap(best_h)
    return (best_h, best_c5, n_points)
```
