def run(seed=42, budget_s=600, **kwargs):
    n_points = 1000
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = np.resize(h, n_points)
    h = np.clip(h, 0, 1)
    
    def calculate_c5(h):
        h = h.reshape(-1)
        h_expanded = np.zeros((n_points, n_points), dtype=float)
        for i in range(n_points):
            h_expanded[i, :] = h
        corr = np.zeros(n_points, dtype=float)
        for k in range(1, n_points):
            corr[k] = np.sum(h_expanded[:n_points - k, :n_points - k] * (1 - h_expanded[k:, :n_points - k])) * dx
        return np.max(corr)
    
    def objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        return calculate_c5(h)
    
    def constraint(x):
        h = x.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    x0 = h.copy()
    
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        jac=None,
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6},
        callback=lambda x: print(f"Current C₅: {calculate_c5(x.reshape(n_points))}")
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_c5 = calculate_c5(best_h)
    
    return best_h, best_c5, n_points
