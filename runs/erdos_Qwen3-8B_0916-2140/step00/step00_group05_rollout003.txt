import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points // 2 + 1):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def initial_h_values(n_points):
    return np.random.rand(n_points) * 2 - 1

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from a simple construction
    n_points = 256  # Reasonable size for quick optimization
    h = initial_h_values(n_points)
    
    # Evaluate initial solution
    max_overlap = evaluate_erdos_solution(h)
    current_c5 = max_overlap
    best_h = h.copy()
    best_c5 = current_c5
    
    # Local search loop
    for step in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Perturb current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h = np.clip(best_h + perturbation, 0, 1)
        
        # Evaluate the new solution
        max_overlap = evaluate_erdos_solution(h)
        
        # Accept if it's better
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        print(f"Step {step + 1}: Current C5 = {best_c5:.6f}")
    
    return best_h, best_c5, n_points
