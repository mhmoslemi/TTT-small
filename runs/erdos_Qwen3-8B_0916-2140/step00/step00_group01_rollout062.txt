import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > overlap_result:
            overlap_result = overlap
    return overlap_result

def initial_h_values():
    return [0.5] * 47

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 47
    
    # Initial guess
    h = initial_h_values()
    
    # Evaluate initial solution
    best_c5 = evaluate_erdos_solution(h)
    best_h = copy.copy(h)
    
    # Try increasing n_points
    for n_points in [50, 100, 150, 200, 250, 300, 350, 400, 450, 500, 550]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Generate random initial guess
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = (h * 2 - 1) / 2  # Map to [0, 1]
        target_sum = n_points / 2
        h = h / np.sum(h) * target_sum
        
        # Optimization
        def objective(x):
            nonlocal best_c5
            h_vals = x
            c5 = evaluate_erdos_solution(h_vals)
            if c5 < best_c5:
                best_c5 = c5
                best_h = copy.copy(h_vals)
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B for faster convergence
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'disp': False}
        )
        
        h = result.x
        best_c5 = evaluate_erdos_solution(h)
        best_h = copy.copy(h)
    
    return best_h, best_c5, len(best_h)
