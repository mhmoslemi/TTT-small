def run(seed=42, budget_s=600, **kwargs):

    # Define parameters
    max_n_points = 450  # max number of samples
    n_points = 150  # starting with 150 samples
    n_evaluations = 100  # number of evaluations to run in parallel
    max_iter = 200  # maximum number of iterations
    C5_target = 0.380876

    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # truncate to n_points

    def objective(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        h = np.repeat(h, 2)  # repeat to cover [0, 2]
        h = h[:n_points]
        h = h[:n_points]
        h = h[:n_points]
        h = h[:n_points]
        t = np.linspace(0, 2, n_points)
        # Correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Normalize by dx
        dx = 2.0 / n_points
        c5 = np.max(corr * dx)
        return c5

    def merit_function(h):
        return objective(h)

    def is_better(h1, h2):
        c5_1 = objective(h1)
        c5_2 = objective(h2)
        return c5_1 < c5_2

    def optimize_single(h0, max_iter=200):
        h = h0.copy()
        for _ in range(max_iter):
            # Perturb h
            perturbation = np.random.normal(0, 0.05, size=len(h))
            h = np.clip(h + perturbation, 0, 1)
            # Compute the objective
            c5 = objective(h)
            if c5 < 0.38:
                return h, c5
        return h, objective(h)

    def evaluate_candidate(h):
        return h, objective(h)

    # Start with the initial guess
    best_h = h
    best_c5 = objective(h)

    start_time = time.time()

    # Try a few perturbations
    for _ in range(10):
        perturbation = np.random.normal(0, 0.1, size=n_points)
        h_candidate = np.clip(h + perturbation, 0, 1)
        c5 = objective(h_candidate)
        if c5 < best_c5:
            best_h = h_candidate
            best_c5 = c5
        if time.time() - start_time > budget_s - 5:
            break

    # Try an optimization step
    h_opt, c5_opt = optimize_single(best_h)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    # Try another perturbation
    perturbation = np.random.normal(0, 0.1, size=n_points)
    h_candidate = np.clip(best_h + perturbation, 0, 1)
    c5 = objective(h_candidate)
    if c5 < best_c5:
        best_h = h_candidate
        best_c5 = c5

    # If time is still left, try a local search
    if time.time() - start_time < budget_s - 10:
        h_opt, c5_opt = optimize_single(best_h, max_iter=100)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final optimization
    h_opt, c5_opt = optimize_single(best_h, max_iter=100)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    # Return the best solution
    return best_h, best_c5, n_points
