```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
from scipy.optimize import minimize
from scipy.linalg import norm
import time

# Predefined initial h_values
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600):
    # Set random seed
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 1000  # sufficient for performance and accuracy
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Objective function
    def objective(h):
        # Compute the overlap integral for all k
        h = h.reshape(-1)
        integrals = []
        for k in range(1, 100):  # Check for relevant k values
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            integrals.append(overlap)
        return max(integrals)
    
    # Initial guess based on the initial_h_values
    initial_guess = np.interp(np.linspace(0, 2, n_points), x, initial_h_values)
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform optimization
    start_time = time.time()
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the solution meets the integral constraint
    integral = np.sum(best_h) * dx
    if abs(integral - 1) > 1e-6:
        # If the constraint is violated, adjust using Lagrange multipliers
        # Scale the solution to satisfy the constraint
        scaling_factor = 1 / integral
        best_h = best_h * scaling_factor
    
    # Final check
    final_c5 = objective(best_h)
    
    # Return the best solution
    return best_h, final_c5, n_points
```
