```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize
from scipy import linalg
from functools import lru_cache

# Pre-imported functions
def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values)
    h = np.concatenate((h, h))  # double the array to handle periodicity
    h = h[:2 * n_points]
    h = h / np.max(h)  # normalize for correlation
    corr = np.correlate(h, 1 - h, mode="full")
    corr = corr[corr.size // 2 :]
    c5 = np.max(corr) * (2.0 / n_points)
    return c5

def initial_h_values(n_points=70):
    # A simple initial construction
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    max_n_points = 500
    n_points = 100  # Start with a small number of points
    best_c5 = float('inf')
    best_h = None
    
    # Function to evaluate a candidate h
    def evaluate_candidate(h_vals):
        nonlocal best_c5
        h = np.array(h_vals)
        n = len(h)
        if n != n_points:
            raise ValueError("Number of points must match")
        
        # Check constraints
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        if np.sum(h) * (2.0 / n_points) != 1.0:
            return float('inf')
        
        # Compute correlation
        h = np.concatenate((h, h))
        h = h[:2 * n_points]
        h = h / np.max(h)
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr[corr.size // 2 :]
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Optimization function
    def objective_func(h_vals):
        return evaluate_candidate(h_vals)
    
    # Function to generate random h
    def generate_random_h(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (2.0 / n_points)  # scale to ensure integral is 1
        return h
    
    # Run optimization
    def optimize_h(n_points):
        result = minimize(objective_func, x0=initial_h_values(n_points), method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x
    
    # Try several random starts
    for _ in range(10):
        h_vals = generate_random_h(n_points)
        result = optimize_h(n_points)
        c5 = evaluate_candidate(result)
        if c5 < best_c5:
            best_c5 = c5
            best_h = result
    
    # Try more points if time remains
    if time.time() - start_time < budget_s - 10:
        n_points = min(500, max_n_points)
        best_h = None
        best_c5 = float('inf')
        
        # Try several random starts with more points
        for _ in range(5):
            h_vals = generate_random_h(n_points)
            result = optimize_h(n_points)
            c5 = evaluate_candidate(result)
            if c5 < best_c5:
                best_c5 = c5
                best_h = result
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_candidate(best_h)
    else:
        final_c5 = float('inf')
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        print("Approaching time limit")
    
    return best_h, final_c5, len(best_h)
```
