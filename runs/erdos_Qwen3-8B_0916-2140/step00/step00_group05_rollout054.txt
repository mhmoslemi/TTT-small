    def compute_c5(h):

        one_minus_h_fft = np.fft.fft(1 - h)

        corr = np.real(corr_fft)
        max_overlap = np.max(corr * dx)
        return max_overlap

    def objective(h):

    def optimize_h(initial_guess):

        result = optimize.minimize(
            fun=objective,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            tol=1e-6,
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        
        if result.success:
            return result.x
        else:
            return h

    best_h = initial_h
