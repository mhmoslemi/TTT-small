import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import LinearOperator, lsqr
from scipy.linalg import eigvals
from scipy.optimize import Bounds
import sys
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the initial sample points
    n_points = 1000
    h_initial = initial_h_values
    if len(h_initial) != n_points:
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Define the integral bounds for the overlap function
    def overlap_integral(h):
        dx = 2.0 / n_points
        h = np.array(h)
        h = np.concatenate((h, h[:n_points//2]))  # For circular convolution
        corr = np.correlate(h, 1 - h[:n_points], mode='full')
        corr = corr[n_points-1:]
        return np.max(corr) * dx
    
    # Define the objective function to minimize the overlap integral
    def objective(h):
        return overlap_integral(h)
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a local optimization method with initial guess
    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
    
    # Extract the best solution
    best_h = result.x
    
    # Compute the final C5 value
    final_c5 = overlap_integral(best_h)
    
    # Return the best solution, the C5 value, and the number of points
    return (best_h, final_c5, n_points)
