import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 if i % 5 == 0 else 0.0
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 0.5 if i % 4 == 0 else 0.0
    initial_h_values = (np.sin(np.linspace(0, np.pi, n_points)) + 1.0) / 2.0
    initial_h_values = np.clip(initial_h_values, 0.0, 1.0)
    initial_h_values = initial_h_values * (n_points / 2.0) / np.sum(initial_h_values)
    h = initial_h_values

    def compute_c5(h):
        h = h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    max_iterations = 100
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(max_iterations):
        start_time = time.time()
        if it % 10 == 0:
            print(f"Iteration {it} / {max_iterations} - Current best C5: {best_c5:.6f}")

        # Define bounds for h
        bounds = Bounds(0.0, 1.0)

        # Objective function to minimize
        def objective(x):
            x = x.copy()
            x = np.clip(x, 0.0, 1.0)
            x = x * (n_points / 2.0) / np.sum(x)
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx

        # Constraint: sum(h) == n_points / 2.0
        def constraint(x):
            return np.sum(x) - n_points / 2.0

        cons = ({'type': 'eq', 'fun': constraint})

        # Initial guess
        x0 = h.copy()

        # Optimization
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 50})

        if result.success:
            h = result.x
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            print("Optimization failed. Trying another initial guess.")

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
