import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time
import copy

initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 45
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    h = h[:n_points]
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)
    h = h[:n_points]

    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlap = np.zeros(n)
        for k in range(1, n // 2 + 1):
            shifted = np.roll(h, k)
            overlap[k] = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)

    def objective(x):
        x = np.clip(x, 0, 1)
        return compute_overlap(x)

    def bounds_func(x):
        return np.clip(x, 0, 1)

    # Use a simple gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    max_iterations = 1000
    best_h = h.copy()
    best_c5 = compute_overlap(h)

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute current overlap
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(1, len(h) // 2 + 1):
            shifted = np.roll(h, k)
            grad += np.gradient(h * (1 - shifted), dx) * dx

        # Update h with momentum
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

    return best_h, best_c5, len(best_h)
