import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Precompute all shifts
    n_shifts = 5
    shifts = np.linspace(-1.0, 1.0, n_shifts)
    
    def compute_c5(h):
        c5 = 0.0
        for shift in shifts:
            h_shifted = np.roll(h, int(shift * n_points / dx))
            integral = np.sum(h * (1 - h_shifted)) * dx
            if integral > c5:
                c5 = integral
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    # Set bounds
    bounds = Bounds(0, 1)
    
    # Run optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'max_iter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check time
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print(f"Time exceeded budget. Final C5: {compute_c5(initial_h)}")
        return (initial_h, compute_c5(initial_h), n_points)
    
    # Final result
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
