```python
import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Initial parameters
    n_points = 200  # Start with a moderate number of points
    n_max = 400     # Maximum number of points allowed
    max_t = budget_s  # Time budget in seconds
    best_c5 = float('inf')
    best_h = None
    best_n = n_points

    # Initial sample
    h = initial_h_values[:n_points]
    h = np.array(h, dtype=np.float64)
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate and track best solution
    def update_best(h, c5, n):
        nonlocal best_c5, best_h, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Optimization setup
    def optimize_solution(h, n_points, max_iter=100):
        h_flat = h.flatten()
        n = len(h_flat)
        dx = 2.0 / n
        current_c5 = compute_c5(h)
        update_best(h, current_c5, n)
        best_c5_so_far = current_c5
        
        for iter in range(max_iter):
            # Evaluate all shifted overlaps
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            c5 = max_corr * dx
            if c5 < best_c5_so_far:
                best_c5_so_far = c5
                update_best(h, c5, n)
            
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.01, size=n)
            h = np.clip(h + perturbation, 0, 1)
            
            # Ensure integral constraint
            total = np.sum(h) * dx
            scale_factor = 1.0 / total if total != 0 else 1.0
            h *= scale_factor
        
        return best_c5_so_far, best_h, best_n
    
    # Start optimization
    start_time = time.time()
    best_c5, best_h, best_n = optimize_solution(h, n_points)
    
    # If time remains, increase n_points and re-optimize
    while time.time() - start_time < max_t - 1:
        n_points = min(n_points * 2, n_max)
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (2.0 / n_points)
        best_c5, best_h, best_n = optimize_solution(h, n_points)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
```
