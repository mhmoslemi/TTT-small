```python
import numpy as np
import scipy.optimize as optimize
import time
import multiprocessing
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda x: np.max(np.correlate(x, 1 - x, mode="full")) * (2.0 / len(x))
initial_h_values = np.load('initial_h_values.npy')  # Preloaded initial construction


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for faster evaluation
    best_c5 = float('inf')
    best_h = None

    def objective(h_values):
        nonlocal best_c5, best_h
        overlap = evaluate_erdos_solution(h_values)
        if overlap < best_c5:
            best_c5 = overlap
            best_h = h_values.copy()
        return -overlap  # Minimize overlap by maximizing -overlap

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Initial guess using initial_h_values
    initial_guess = initial_h_values[:n_points]  # Use first n_points samples

    # Solve using scipy.optimize
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[
            {'type': 'eq', 'fun': constraint_sum}
        ],
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    if result.success:
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)

    # Try increasing n_points for better accuracy if budget allows
    if time.time() - start_time < budget_s * 0.5:
        n_points = 200
        new_initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        initial_guess = new_initial_guess

        def objective_high_res(h_values):
            nonlocal best_c5, best_h
            overlap = evaluate_erdos_solution(h_values)
            if overlap < best_c5:
                best_c5 = overlap
                best_h = h_values.copy()
            return -overlap

        def constraint_sum_high_res(h_values):
            return np.sum(h_values) - n_points / 2

        result_high_res = optimize.minimize(
            fun=objective_high_res,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': constraint_sum_high_res}
            ],
            options={'ftol': 1e-8, 'maxiter': 100}
        )

        if result_high_res.success:
            best_h = result_high_res.x
            best_c5 = evaluate_erdos_solution(best_h)

    # Final check to ensure we are under time budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Try one more round with even higher n_points if time allows
        n_points = 300
        new_initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        initial_guess = new_initial_guess

        def objective_final(h_values):
            nonlocal best_c5, best_h
            overlap = evaluate_erdos_solution(h_values)
            if overlap < best_c5:
                best_c5 = overlap
                best_h = h_values.copy()
            return -overlap

        def constraint_sum_final(h_values):
            return np.sum(h_values) - n_points / 2

        result_final = optimize.minimize(
            fun=objective_final,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': constraint_sum_final}
            ],
            options={'ftol': 1e-8, 'maxiter': 100}
        )

        if result_final.success:
            best_h = result_final.x
            best_c5 = evaluate_erdos_solution(best_h)

    return (best_h, best_c5, n_points)
```
