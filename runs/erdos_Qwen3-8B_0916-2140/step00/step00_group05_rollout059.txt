def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # Initial construction (from the initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Function to compute the overlap integral for a given h
    def compute_c5(h_values):
        h = np.array(h_values)
        if len(h) != n_points:
            raise ValueError(f"Expected {n_points} points, got {len(h)}")
        # Compute the overlap integral for all k
        corr = correlate(h, 1 - h, mode="full")
        # Find the maximum value of the integral (scaled by dx)
        max_corr = np.max(corr) * dx
        return max_corr

    # Define the objective function to minimize, given h_values
    def objective(h_values):
        return compute_c5(h_values)

    # Constraints: sum of h_values must be exactly n_points / 2
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Initial guess based on the initial_h_values
    initial_guess = initial_h.copy()

    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimize using the L-BFGS-B algorithm
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="L-BFGS-B",
        options={"maxiter": 1000, "disp": False}
    )

    # Extract the optimal h_values
    best_h_values = result.x
    best_c5 = compute_c5(best_h_values)
    elapsed = time.time() - start_time

    # If we ran out of time, return the best solution found so far
    if elapsed > max_time:
        print(f"Time exceeded. Returning best solution found so far.")
        return (best_h_values, best_c5, n_points)

    return (best_h_values, best_c5, n_points)
