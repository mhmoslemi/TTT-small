import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set the random seed
    random.seed(seed)
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100  # Start with 100 points as a balance between speed and accuracy
    
    # Initial guess from the provided initial_h_values
    h_initial = initial_h_values
    h_initial = h_initial[:n_points]
    
    # Ensure the integral is 1
    integral = np.sum(h_initial) * (2.0 / n_points)
    if integral == 0:
        h_initial = np.random.rand(n_points)
    else:
        h_initial = h_initial * (1.0 / integral)
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        h = np.clip(h, 0, 1)
        # Compute the overlap integrals for all k in [0, 2]
        overlaps = []
        for k in np.arange(0, 2.0 + 1e-8, 0.1):
            overlap = 0.0
            for i in range(n_points):
                x = i * (2.0 / n_points)
                if x + k < 2.0:
                    j = int((x + k) * n_points / 2.0)
                    if j < 0 or j >= n_points:
                        continue
                    overlap += h[i] * (1.0 - h[j])
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Constraints: sum(h) * dx = 1
    def constraint(h):
        return np.sum(h) * (2.0 / n_points) - 1.0
    
    # Setup the optimization problem
    x0 = h_initial.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use the SLSQP algorithm for constrained optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 500}
    )
    
    # Extract the best solution
    h_values = result.x
    c5_bound = objective(h_values)
    
    # Return the result
    return h_values, c5_bound, n_points
