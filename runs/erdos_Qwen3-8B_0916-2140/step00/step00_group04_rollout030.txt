import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    random.seed(seed)
    
    # Precompute parameters
    n_points = 100  # We'll use 100 samples for a balance between accuracy and speed
    dx = 2.0 / n_points
    
    # Initialize h as a step function with values in [0, 1]
    h = np.zeros(n_points)
    
    # Start with a simple construction where the first half is 0 and the second is 1
    h[:n_points//2] = 0.0
    h[n_points//2:] = 1.0
    
    # Define function to compute the overlap integral
    def compute_c5(h):
        # Normalize h to be in [0, 1] and ensure the integral is 1
        h_norm = h / np.sum(h) * n_points / 2
        # Compute all overlaps
        c5 = 0.0
        for k in range(1, n_points):
            if k >= n_points:
                continue
            # Compute the overlap integral for this shift
            overlap = np.sum(h_norm[:n_points - k] * (1 - h_norm[k:]))
            c5 = max(c5, overlap * dx)
        return c5
    
    # Define the objective function for optimization
    def objective(x):
        h = x.reshape(n_points)
        return compute_c5(h)
    
    # Define the constraints: sum of h must be n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Use differential evolution for global optimization
    result = optimize.differential_evolution(
        objective,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=cons,
        maxiter=100,
        popsize=20,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Ensure the best_c5 is within the budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
