```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import matplotlib.pyplot as plt
from numba import njit, prange
import multiprocessing as mp

# Predefined initial h_values for reference
initial_h_values = np.load('initial_h_values.npy')

def objective_function(h_values, dx):
    h = h_values.copy()
    max_overlap = 0.0
    for k in range(1, len(h) - 1):
        overlap = np.sum(h[:len(h) - k] * (1 - h[k:])) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def evaluate_erdos_solution(h):
    return objective_function(h, 2.0 / len(h))

# Function to create a step function from samples
def create_step_function(h_samples):
    n_points = len(h_samples)
    dx = 2.0 / n_points
    return h_samples, dx

# Optimization strategy
def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    def objective(h):
        nonlocal best_c5, best_h, best_n_points
        h = h.astype(np.float64)
        n_points = len(h)
        dx = 2.0 / n_points
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
        return c5

    # Start with initial_h_values
    initial_n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    result = opt.minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1)] * initial_n_points)
    best_c5 = result.fun
    best_h = result.x.copy()
    best_n_points = initial_n_points

    # Try reducing the number of points to improve speed
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying n_points = {n_points}")
        h = np.random.uniform(0, 1, n_points)
        # Ensure the integral is exactly 1
        h = h / np.sum(h) * (n_points / 2)
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        c5 = result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = result.x.copy()
            best_n_points = n_points

    print(f"Best C₅ bound: {best_c5}")
    return best_h, best_c5, best_n_points
```
