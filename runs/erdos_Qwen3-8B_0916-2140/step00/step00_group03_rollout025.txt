```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time

def run(seed=42, budget_s=600, **kwargs):
    # Pre-defined initial solution
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def compute_c5(h):
        h = h.reshape(-1)
        h = np.repeat(h, 2)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def bounds_func(x):
        return [(0, 1) for _ in x]

    def optimize_with_bounds(h0, seed):
        np.random.seed(seed)
        h = h0.copy()
        h = h.astype(float)
        max_iter = 100
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_i = h[i]
                for k in range(n_points):
                    if k == i:
                        continue
                    h_k = h[k]
                    grad[i] += (1 - h_k) if h_i == 1 else 0
                    grad[k] += (1 - h_i) if h_k == 1 else 0
            grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else np.zeros_like(grad)
            h -= 0.1 * grad
            h = np.clip(h, 0, 1)
            if np.abs(np.sum(h) - n_points / 2) < 1e-6:
                break
        return h

    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        h = optimize_with_bounds(best_h, seed)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        seed += 1

    return best_h, best_c5, n_points
```
