```python
import numpy as np
import scipy.optimize as opt
import time
import multiprocessing
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Example initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    max_iter = 200
    learning_rate = 0.1
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = None

    # Initial guess
    h = initial_h_values[:n_points].copy()
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != n_points / 2:
        h = (h * n_points / 2) / sum_h
    h = np.clip(h, 0, 1)

    def objective(x):
        nonlocal best_c5
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = (h * n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)

    def gradient_descent_step(x, learning_rate, tolerance):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = (h * n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return h

    for _ in range(max_iter):
        h = gradient_descent_step(h, learning_rate, tolerance)
        if np.abs(np.sum(h) - n_points / 2) < tolerance:
            break

    return best_h, best_c5, n_points
```
