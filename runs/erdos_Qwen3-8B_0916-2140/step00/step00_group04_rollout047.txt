```python
import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.concatenate([h[-2:], h])
    correlate = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlate) * dx
    return c5

def initial_h_values():
    return [0.5 if i % 3 == 0 else 0.25 for i in range(42)]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess
    initial_h = initial_h_values()
    n_points = len(initial_h)

    # Optimization parameters
    max_iter = 100
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = initial_h.copy()

    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            print(f"Iteration {iteration+1}: C5 = {best_c5:.6f}")

        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)

        # Evaluate perturbed solution
        perturbed_c5 = evaluate_erdos_solution(perturbed_h)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h.copy()

    # Return the best solution found
    return best_h, best_c5, n_points
```
