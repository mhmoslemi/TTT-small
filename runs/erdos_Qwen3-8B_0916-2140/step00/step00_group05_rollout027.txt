```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import matplotlib.pyplot as plt
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    h = initial_h_values(n_points)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    start_time = time.time()

    # Optimization parameters
    max_iter = 100
    learning_rate = 0.01
    tolerance = 1e-4

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute current C5
        current_c5 = evaluate_erdos_solution(h)

        # Compute all shifts (k) and their overlaps
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k < 0:
                shifted_h = np.roll(h, k)
            else:
                shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * (2.0 / n_points)
            overlaps.append(overlap)

        # Gradient step to minimize the maximum overlap
        max_overlap = np.max(overlaps)
        indices = np.argsort(overlaps)
        max_index = indices[-1]
        k = shifts[max_index]

        # Compute derivative of overlap with respect to h
        if k < 0:
            shifted_h = np.roll(h, k)
            derivative = - (1 - shifted_h) * (2.0 / n_points)
        else:
            shifted_h = np.roll(h, k)
            derivative = - (1 - shifted_h) * (2.0 / n_points)

        # Update h using gradient descent
        h = h - learning_rate * derivative

        # Enforce constraints: h ∈ [0, 1], ∑h = n_points / 2
        h = np.clip(h, 0, 1)
        h = h - np.min(h)
        h = h / np.max(h)
        h = h * (n_points / 2) / np.sum(h)

        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

        # Check for convergence
        if abs(new_c5 - current_c5) < tolerance:
            break

    return best_h, best_c5, n_points
```
