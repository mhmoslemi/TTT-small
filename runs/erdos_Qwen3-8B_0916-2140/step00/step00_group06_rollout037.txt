```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    max_n_points = 500
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Start with the initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    dx = 2.0 / n_points_initial
    sum_h = np.sum(h_initial) * dx
    if abs(sum_h - 1.0) > 1e-6:
        h_initial = h_initial / (sum_h / 1.0)
    
    # Evaluate initial solution
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        h = h / (np.sum(h) * dx)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    best_c5 = evaluate_c5(h_initial)
    best_h = h_initial.copy()
    
    # Try reducing the number of points to speed up the optimization
    for n_points in range(50, max_n_points + 1, 50):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate a new random h with n_points samples
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * 2.0 / n_points)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Try a few more points with more structured samples
    for n_points in range(100, max_n_points + 1, 100):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Generate a piecewise constant function with n_points samples
        h = np.zeros(n_points)
        current_value = 0
        for i in range(n_points):
            if i % 5 == 0:
                current_value = np.random.rand()
            h[i] = current_value
        
        # Normalize to ensure integral is 1
        dx = 2.0 / n_points
        sum_h = np.sum(h) * dx
        h = h / (sum_h / 1.0)
        
        # Evaluate
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, len(best_h)
```
