import numpy as np
import scipy.optimize as optimize
import time
from scipy import stats

from scipy.optimize import minimize
import sys

# Pre-imported functions
def evaluate_erdos_solution(h_values, dx):
    n = len(h_values)
    overlap = np.zeros(n - 1)
    for k in range(1, n - 1):
        shifted = np.roll(h_values, k)
        product = h_values * (1 - shifted)
        overlap[k] = np.sum(product) * dx
    return np.max(overlap)

def initial_h_values(n):
    return np.random.rand(n) * 2 - 1

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    max_points = 256
    initial_n = 128
    n_points = initial_n
    
    # Initial guess based on the provided initial_h_values
    h_initial = initial_h_values(n_points)
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    scale_factor = target_sum / sum_h
    h_initial = np.clip(h_initial * scale_factor, 0, 1)
    sum_h = np.sum(h_initial)
    
    def objective_function(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale_factor = (n_points / 2) / sum_h
        h_scaled = h * scale_factor
        overlap = np.zeros(n_points - 1)
        for k in range(1, n_points - 1):
            shifted = np.roll(h_scaled, k)
            product = h_scaled * (1 - shifted)
            overlap[k] = np.sum(product) * (2.0 / n_points)
        return np.max(overlap)
    
    # Define bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    result = minimize(
        objective_function,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check if we can try a larger n_points
    if time.time() - start_time < budget_s * 0.8:
        n_points = min(256, max_points)
        h_initial = np.random.rand(n_points) * 2 - 1
        h_initial = np.clip(h_initial, 0, 1)
        sum_h = np.sum(h_initial)
        scale_factor = (n_points / 2) / sum_h
        h_initial = np.clip(h_initial * scale_factor, 0, 1)
        sum_h = np.sum(h_initial)
        
        def objective_function_large(h):
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale_factor = (n_points / 2) / sum_h
            h_scaled = h * scale_factor
            overlap = np.zeros(n_points - 1)
            for k in range(1, n_points - 1):
                shifted = np.roll(h_scaled, k)
                product = h_scaled * (1 - shifted)
                overlap[k] = np.sum(product) * (2.0 / n_points)
            return np.max(overlap)
        
        result = minimize(
            objective_function_large,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ft': 1e-8}
        )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    scale_factor = (n_points / 2) / sum_h
    best_h = np.clip(best_h * scale_factor, 0, 1)
    
    c5_bound = evaluate_erdos_solution(best_h, 2.0 / n_points)
    return best_h, c5_bound, n_points
