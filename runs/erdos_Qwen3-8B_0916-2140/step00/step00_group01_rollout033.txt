import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define initial h_values, if available
    initial_h_values = np.load("initial_h_values.npy")
    n_points = len(initial_h_values)
    
    # Define the overlap integral function
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        c5_values = []
        
        for k in range(1, n):
            overlap = np.sum(h[:n - k] * (1 - h[k:n]))
            c5_values.append(overlap * dx)
        
        return max(c5_values)
    
    # Define the objective function to minimize
    def objective(h):
        return compute_c5(h)
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Initial guess
    initial_h_values = np.load("initial_h_values.npy")
    
    # Perform optimization
    result = minimize(objective, initial_h_values, method='SLSQP', constraints=cons)
    
    # Extract best solution
    best_h = result.x
    c5_bound = compute_c5(best_h)
    
    # Return the best solution
    return (best_h, c5_bound, len(best_h))
