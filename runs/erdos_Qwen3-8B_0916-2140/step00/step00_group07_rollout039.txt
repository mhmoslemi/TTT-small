import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with a moderate number of points for optimization
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    
    # Refine the initial guess using interpolation from the initial_h_values
    def refine_initial_h(initial_h, n_refine):
        refined_h = np.zeros(n_refine)
        for i in range(n_refine):
            pos = i * 2.0 / n_refine
            for j in range(len(initial_h)):
                if abs(pos - j * dx) < 1e-6:
                    refined_h[i] = initial_h[j]
                    break
        return refined_h

    # Define the overlap integral
    def compute_c5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)

    # Define the objective function for optimization
    def objective(h, dx):
        return compute_c5(h)

    # Define constraints for optimization
    def constraint_sum(h, dx):
        return np.sum(h) * dx - 1.0

    # Use a smoother initial guess if initial_h_values is provided
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) > 0:
        initial_h = refine_initial_h(initial_h_values, n_points)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (2.0 / n_points)  # Ensure sum(h) * dx ≈ 1
        
    # Optimize
    result = minimize(objective, initial_h, method='SLSQP', bounds=Bounds(0, 1), constraints=[{'type': 'eq', 'fun': constraint_sum, 'args': (dx,)}])
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
