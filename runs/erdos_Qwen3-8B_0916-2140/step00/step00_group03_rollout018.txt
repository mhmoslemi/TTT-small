def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = h[np.roll(np.arange(n_points), i)]
    
    overlap = np.zeros(n_points)
    for k in range(n_points):
        overlap[k] = np.sum(h * (1 - shifted_h[:, k]))
    
    return np.max(overlap * dx)

def compute_c5_bound(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = h[np.roll(np.arange(n_points), i)]
    
    overlap = np.zeros(n_points)
    for k in range(n_points):
        overlap[k] = np.sum(h * (1 - shifted_h[:, k]))
    
    return np.max(overlap * dx)

def run(seed=42, budget_s=600):

        initial_h = initial_h / initial_sum * (n_points / 2)
    
    # Define objective function
    def objective(x):
        h = np.array(x).clip(0, 1)
        h_sum = np.sum(h)
        if np.abs(h_sum - n_points / 2) > 1e-6:
            return np.inf
        c5 = compute_c5_bound(h)
        return c5
    
    # Define constraints
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Optimization settings
    options = {'disp': False, 'ftol': 1e-8, 'maxiter': 100}
    
    # Run optimization
    result = minimize(objective, initial_h, method='SLSQP', constraints=cons, options=options)
    
    # Best solution found
