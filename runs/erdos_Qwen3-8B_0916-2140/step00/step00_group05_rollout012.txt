```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time
import functools

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    return c5

def initial_h_values():
    return np.load("initial_h.npy")

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_iter = 50
    n_points = 100  # Start with a medium-sized grid
    dx = 2.0 / n_points
    h = initial_h_values()
    h = h[:n_points]  # Ensure it's the correct length
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Step 1: Coarse grid optimization
    for iter in range(5):
        time_start = time.time()
        # Use a simple local optimization strategy
        h = np.clip(h + np.random.normal(0, 0.1, size=n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - time_start > budget_s / 2:
            break
    
    # Step 2: Refinement on finer grid
    n_points = 200
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1  # Initial guess
    h = h * (n_points / 2) / np.sum(h)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    for iter in range(20):
        time_start = time.time()
        # Use gradient descent with random perturbations
        h = np.clip(h + np.random.normal(0, 0.05, size=n_points), 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if time.time() - time_start > budget_s / 2:
            break
    
    return best_h, best_c5, n_points
```
