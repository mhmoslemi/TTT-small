import scipy.optimize as optimize
import time
import copy

    h = copy.deepcopy(initial_h)
    
    def evaluate_c5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        h = np.array(h_values)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    def objective(h_values):
        return evaluate_c5(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    def constraint_bounds(h_values):
        return np.all((h_values >= 0) & (h_values <= 1))
    
    # Initial evaluation
    initial_c5 = evaluate_c5(h)
    best_c5 = initial_c5
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 100
    time_budget = budget_s
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_budget:
            break
        
        # Use scipy's optimization solver
        result = optimize.minimize(
            fun=objective,
            x0=h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': constraint_bounds}
            ],
            method='SLSQP'
        )
        
        if result.success:
            h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            print("Optimization failed.")
            break
    
    return best_h, best_c5, n_points
