import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute cross-correlation
    corr = signal.correlate(h, 1 - h, mode="full")
    # Extract the relevant part of the correlation
    # The relevant part is the middle part where the overlap occurs
    # Since the overlap is between h and h shifted by k, the relevant indices are around the middle
    # We take the middle 2 * n_points - 1 elements
    # For the overlap integral, we take the max of the correlation values * dx
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n_points):
    # Initial construction with n_points samples
    # A simple piecewise constant function that alternates between 0 and 1
    # This is a rough approximation to the optimal function
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Parameters for the optimization
    n_points = 100  # Start with a moderate number of points
    max_iterations = 100
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    # Initial guess
    h_initial = initial_h_values(n_points)
    best_h = h_initial.copy()

    # Function to evaluate the overlap integral
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        # The relevant part is the middle part of the correlation
        # Extract the middle 2 * n_points - 1 elements
        corr_center = corr[n_points - 1 : 2 * n_points - 1]
        # Compute the max value
        c5 = np.max(corr_center) * (2.0 / n_points)
        return c5

    # Function to check the constraints
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

    # Optimization settings
    # Use a local optimizer to refine the initial guess
    # We will use the SLSQP method from scipy.optimize

    # Set a seed for reproducibility
    np.random.seed(seed)

    # Run the optimization
    for _ in range(max_iterations):
        start_time = time.time()
        result = optimize.minimize(
            objective,
            x0=best_h,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{"type": "eq", "fun": constraint}],
            method="SLSQP"
        )
        if time.time() - start_time > budget_s:
            break
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
        else:
            break

    # Further refinement with a smaller n_points
    n_points = 200
    best_h = np.zeros(n_points)
    # Re-initialize with a refined initial guess
    for i in range(n_points):
        if i % 5 == 0:
            best_h[i] = 1.0
        elif i % 5 == 1:
            best_h[i] = 0.5
        else:
            best_h[i] = 0.0

    # Run the optimization again with the new n_points
    for _ in range(max_iterations):

        result = optimize.minimize(
            objective,
            x0=best_h,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints=[{"type": "eq", "fun": constraint}],
            method="SLSQP"
        )
        if time.time() - start_time > budget_s:
            break
        if result.success:
            best_h = result.x
            best_c5 = objective(best_h)
        else:
            break

    return best_h, best_c5, n_points
