import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100  # Starting with 100 points for efficiency
    dx = 2.0 / n_points

    # Initialize h with a simple construction
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1
        else:
            h[i] = 0
    h = np.array(h, dtype=np.float64)
    h = h / (n_points / 2)  # Scale to match the integral constraint

    # Function to compute the overlap integral
    def compute_c5(h):
        shifted_h = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        return overlap

    # Function to compute the maximum overlap
    def compute_max_overlap(h):
        max_overlap = 0
        for k in range(1, n_points):
            shifted_h = np.roll(h, shift=k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Optimization function
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_max_overlap(h)

    # Constraint: total integral equals 1
    def constraint(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1.0

    # Bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    initial_guess = h.copy()

    # Setup optimization with constraints
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    h_opt = result.x
    c5 = compute_max_overlap(h_opt)
    return h_opt, c5, n_points
