```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Assume this is provided

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    max_iterations = 100  # Limit iterations to keep within time budget
    tolerance = 1e-6  # Optimization tolerance

    def objective(h):
        # Compute the overlap integral for all k
        conv = signal.correlate(h, 1 - h, mode='full')
        c5_val = np.max(conv) * dx
        return c5_val

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Initial guess
    h_init = np.copy(initial_h_values[:n_points])
    h_init = np.interp(h_init, (h_init.min(), h_init.max()), (0, 1))  # Normalize
    h_init = np.clip(h_init, 0, 1)
    h_init = np.interp(h_init, (h_init.min(), h_init.max()), (0, 1))  # Ensure in [0, 1]
    h_init = np.round(h_init, 5)  # Reduce precision

    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization
    result = minimize(
        fun=objective,
        x0=h_init,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iterations, 'ftol': tolerance}
    )

    best_h = result.x
    best_c5 = result.fun

    # Check if we can reduce n_points further
    for n in range(50, n_points, -1):
        dx = 2.0 / n
        h = np.interp(best_h, np.linspace(0, 2, n_points), np.linspace(0, 2, n))
        h = np.clip(h, 0, 1)
        h = np.round(h, 5)
        conv = signal.correlate(h, 1 - h, mode='full')
        c5_val = np.max(conv) * dx
        if c5_val < best_c5:
            best_h = h
            best_c5 = c5_val
            n_points = n
            dx = 2.0 / n_points

    return best_h, best_c5, n_points
```
