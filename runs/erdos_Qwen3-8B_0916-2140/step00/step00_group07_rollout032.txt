```python
import numpy as np
import scipy.optimize as optimize
import time
import sys
import random
import copy
import os

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100

    # Define some helper functions
    def compute_overlap(h):
        h_array = np.array(h)
        h_shifted = np.concatenate((h_array[-n_points//2:], h_array[:n_points//2]))
        overlap = np.correlate(h_array, 1 - h_shifted, mode="full")
        return np.max(overlap * (2.0 / n_points))

    def objective_function(h):
        return compute_overlap(h)

    def initial_guess(n_points):
        if n_points == 100:
            initial_h = initial_h_values
            initial_h = [initial_h[i] for i in range(n_points)]
            return initial_h
        else:
            return [np.random.uniform(0, 1) for _ in range(n_points)]

    # Optimization function
    def optimize_h(n_points, max_iter=10):
        h_initial = initial_guess(n_points)
        result = optimize.minimize(objective_function, x0=h_initial, bounds=[(0, 1) for _ in range(n_points)])
        return result.x

    # Main loop
    for n_points in [100, 200, 300, 400, 500, 600, 800, 1000]:
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying with n_points={n_points}")
        h = optimize_h(n_points)
        c5 = compute_overlap(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            print(f"Found better solution with C₅={best_c5:.6f}")
        time.sleep(0.1)

    print(f"Final best C₅={best_c5:.6f}")
    return best_h, best_c5, n_points
```
