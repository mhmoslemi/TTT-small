import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Initial guess from given initial_h_values scaled to [0, 2]
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    if 'initial_h_values' in kwargs:
        initial_h_values = np.array(kwargs['initial_h_values'])
    
    # Function to optimize
    def objective(h):
        return compute_c5(h)
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iterations = 100
    tolerance = 1e-5
    
    # Start optimization
    start_time = time.time()
    result = opt.minimize(objective, initial_h_values, bounds=bounds, method='L-BFGS-B', options={'maxiter': max_iterations, 'ftol': tolerance})
    elapsed = time.time() - start_time
    
    # If the optimization is not successful, return the initial guess
    if not result.success:
        return initial_h_values, compute_c5(initial_h_values), n_points
    
    best_h = result.x
    best_c5 = result.fun
    
    # Return the best solution found
    return best_h, best_c5, n_points
