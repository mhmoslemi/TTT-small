import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1-h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assume this is a preloaded initial guess

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Define the initial guess with n_points
    n_points = 100  # Initial n_points
    h = initial_h_values[:n_points]
    if len(h) != n_points:
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = (h - np.min(h)) / (np.max(h) - np.min(h)) if np.max(h) > np.min(h) else np.zeros(n_points)
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * 2 - 1
        h = np.clip(h, 0, 1)
        h = (h - np.min(h)) / (np.max(h) - np.min(h)) if np.max(h) > np.min(h) else np.zeros(n_points)
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * 2 - 1
        h = np.clip(h, 0, 1)
        h = (h - np.min(h)) / (np.max(h) - np.min(h)) if np.max(h) > np.min(h) else np.zeros(n_points)
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = (h - np.min(h)) / (np.max(h) - np.min(h)) if np.max(h) > np.min(h) else np.zeros(n_points)
    total = np.sum(h)
    h = h * (1.0 / total) * 2.0 if total > 0 else np.full(n_points, 0.5)

    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Define the objective function and constraints
    def objective(x):
        nonlocal best_c5
        h = x.copy()
        total = np.sum(h)
        if total == 0:
            return best_c5 + 1
        h = h * (2.0 / total)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        return current_c5

    # Define bounds and constraints
    def constraint(x):
        return np.sum(x) - n_points / 2

    # Set up the optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B'
    )

    # Update best_h and best_c5 with the result
    best_h = result.x
    total = np.sum(best_h)
    best_h = best_h * (2.0 / total) if total > 0 else np.full(n_points, 0.5)
    best_c5 = evaluate_erdos_solution(best_h)

    # Try reducing n_points
    if best_c5 < 0.381:
        n_points = int(np.sqrt(n_points))
        h = best_h[:n_points]
        total = np.sum(h)
        h = h * (2.0 / total) if total > 0 else np.full(n_points, 0.5)
        best_c5 = evaluate_erdos_solution(h)

    # Final evaluation
    total = np.sum(best_h)
    best_h = best_h * (2.0 / total) if total > 0 else np.full(n_points, 0.5)
    final_c5 = evaluate_erdos_solution(best_h)

    if time.time() - start_time > budget_s - 10:
        print("Time budget exceeded, returning best solution so far.")
        return best_h, final_c5, n_points

    # Explore with a larger n_points
    n_points = 1000
    h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = (h - np.min(h)) / (np.max(h) - np.min(h)) if np.max(h) > np.min(h) else np.zeros(n_points)
    h = h * 2 - 1
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    h = h * (2.0 / total) if total > 0 else np.full(n_points, 0.5)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    result = optimize.minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B'
    )

    best_h = result.x
    total = np.sum(best_h)
    best_h = best_h * (2.0 / total) if total > 0 else np.full(n_points, 0.5)
    best_c5 = evaluate_erdos_solution(best_h)

    # Final evaluation
    total = np.sum(best_h)
    best_h = best_h * (2.0 / total) if total > 0 else np.full(n_points, 0.5)
    final_c5 = evaluate_erdos_solution(best_h)

    if time.time() - start_time > budget_s - 10:
        print("Time budget exceeded, returning best solution so far.")
        return best_h, final_c5, n_points

    return best_h, final_c5, n_points
