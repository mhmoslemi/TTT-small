import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the number of points (adjust for performance)
    n_points = 100  # A balance between performance and accuracy
    
    # Initialize h_values with a base pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5 if i % 3 == 0 else 0.3 if i % 3 == 1 else 0.7
    
    # Define the overlap integral function
    def compute_c5(h):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define the objective function to minimize C5
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_c5(h)
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up the optimization problem
    result = optimize.minimize(
        objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Ensure the sum condition is satisfied (dx = 2.0 / n_points, sum(h) * dx = 1)
    # Adjust h_values to ensure the sum condition is met
    total_sum = np.sum(best_h)
    dx = 2.0 / n_points
    adjustment_factor = (1.0 / dx) / total_sum if total_sum != 0 else 1.0
    best_h = best_h * adjustment_factor
    
    return best_h, best_c5, n_points
