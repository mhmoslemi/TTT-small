```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import random
import cvxpy as cp

# Pre-imported functions
from initial_h_values import initial_h_values
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points
    
    # Evaluate initial guess
    h = initial_h_values
    h = np.array(h[:n_points])
    if len(h) < n_points:
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    
    # Try increasing n_points up to 500
    for n in range(100, 501, 100):
        if time.time() - start_time > budget_s - 10:
            break
        print(f"Trying with {n} points...")
        
        # Generate random initial guess
        h = np.random.uniform(0, 1, n)
        # Normalize to satisfy ∫h(x)dx = 1
        h = h / (np.sum(h) / n)
        
        # Optimize using L-BFGS-B
        def objective(x):
            nonlocal best_c5
            h = x.reshape(-1)
            # Compute 1 - h
            one_minus_h = 1 - h
            # Compute convolution
            conv = signal.correlate(h, one_minus_h, mode='full')
            # Compute maximum value
            c5_candidate = np.max(conv) * (2 / n)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
            return c5_candidate
        
        # Set bounds for optimization
        bounds = [(0, 1) for _ in range(n)]
        
        # Run L-BFGS-B
        result = opt.minimize(
            objective,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        # Update best if needed
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
    
    # If still not better, try some heuristic adjustments
    if best_c5 > 0.380876:
        print("Trying heuristic adjustments...")
        h = best_h
        # Create a more balanced histogram
        h = np.round(h, 5)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n
        h = np.round(h, 5)
        h = np.clip(h, 0, 1)
        
        # Compute final c5
        one_minus_h = 1 - h
        conv = signal.correlate(h, one_minus_h, mode='full')
        c5_final = np.max(conv) * (2 / len(h))
        
        # Final check
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h
    
    # Convert best_h to list and return
    best_h = best_h.tolist()
    return best_h, best_c5, len(best_h)
```
