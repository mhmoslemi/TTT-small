        shift_values = np.linspace(-2, 2, 1000)

        for k in shift_values:
            shifted_h = np.roll(h, int(k * len(h) / 2))
            overlap = np.sum(h * (1 - shifted_h)) * dx
            c5_values.append(overlap)

    result = optimize.minimize(
        objective,
        initial_guess,
        method='COBYLA',
        bounds=[(0, 1) for _ in range(len(initial_guess))],
        constraints=[{'type': 'eq', 'fun': constraint}]
    )
