```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time
import matplotlib.pyplot as plt

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return np.load('initial_h_values.npy').tolist()

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial construction
    initial_h = initial_h_values()
    n_points = len(initial_h)
    
    # Optimization parameters
    max_iter = 100
    tolerance = 1e-6
    
    def objective(h):
        return evaluate_erdos_solution(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.all((h >= 0) & (h <= 1))
    
    # Optimization
    result = minimize(
        objective,
        initial_h,
        method='SLSQP',
        constraints=[
            {'type': 'eq', 'fun': constraint_sum},
        ],
        bounds=[(0, 1) for _ in range(n_points)],
        tol=tolerance,
        options={'maxiter': max_iter, 'ftol': tolerance, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    # If time remaining, try random perturbations
    remaining_time = budget_s - elapsed_time
    if remaining_time > 0:
        perturbations = np.random.rand(n_points)
        for _ in range(5):
            perturbed_h = best_h * (1 + 0.05 * (perturbations - 0.5))
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            c5 = evaluate_erdos_solution(perturbed_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = perturbed_h
    
    return best_h, best_c5, len(best_h)
```
