import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time

def run(seed=42, budget_s=600):
    # Pre-defined initial construction
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    assert np.isclose(sum_h * dx, 1), "Initial h_values does not satisfy the integral constraint."

    # Define the objective function for a given h
    def compute_overlap(h):
        n = len(h)
        overlap = np.zeros(n)
        for k in range(1, n):
            if k >= n:
                break
            overlap[k] = np.dot(h[:n - k], 1 - h[k:])
        return np.max(overlap) * dx

    # Define the constraint function (integral constraint)
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Define the bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h_initial = initial_h_values.copy()

    # Optimization settings
    max_iter = 100
    tol = 1e-5

    best_h = h_initial
    best_c5 = compute_overlap(h_initial)

    # Strategy: Use simulated annealing with bounds
    start_time = time.time()
    for _ in range(max_iter):
        # Create a perturbed version
        h_pert = h_initial.copy()
        idx = np.random.randint(n_points)
        h_pert[idx] = np.clip(h_pert[idx] + np.random.normal(0, 0.1), 0.0, 1.0)

        # Check constraint
        if not np.isclose(np.sum(h_pert) * dx, 1, atol=1e-6):
            continue

        c5 = compute_overlap(h_pert)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pert

        # Decrease temperature
        temp = max(0.1, 1.0 / (max_iter / 2))

        # Accept with probability
        if np.random.rand() < np.exp((best_c5 - c5) / temp):
            h_initial = h_pert

        # Check time
        if time.time() - start_time > budget_s - 1:
            break

    # Return the best solution
    return best_h, best_c5, n_points
