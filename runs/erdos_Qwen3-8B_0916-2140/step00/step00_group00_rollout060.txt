import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if np.isclose(sum_h, n_points / 2):
        pass
    else:
        initial_h = initial_h * (n_points / 2) / sum_h

    def evaluate_erdos(h, n_points, dx):
        h = np.array(h)
        h = np.concatenate((h, h[:n_points - n_points // 2]))
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps) * dx
        return c5

    def cost_function(h):
        h = np.array(h)
        h = np.concatenate((h, h[:n_points - n_points // 2]))
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps) * dx
        return c5

    def optimize_h(h0, budget_s, n_points):
        start_time = time.time()
        best_h = h0.copy()
        best_c5 = evaluate_erdos(best_h, n_points, dx)
        time_remaining = budget_s - (time.time() - start_time)

        def objective(h):
            return cost_function(h)

        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]

        for _ in range(100):
            if time_remaining < 10:
                break
            try:
                result = optimize.minimize(
                    objective,
                    h0,
                    bounds=bounds,
                    constraints={"type": "eq", "fun": constraint_sum},
                    method="SLSQP",
                    options={"maxiter": 100, "ftol": 1e-8}
                )
                if result.success:
                    h0 = result.x
                    current_c5 = evaluate_erdos(h0, n_points, dx)
                    if current_c5 < best_c5:
                        best_h = h0.copy()
                        best_c5 = current_c5
                time_remaining = budget_s - (time.time() - start_time)
            except:
                break

        return best_h, best_c5

    best_h, best_c5 = optimize_h(initial_h, budget_s, n_points)
    return best_h, best_c5, n_points
