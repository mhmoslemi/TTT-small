```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points and default h_values
    n_points = 100
    h = np.random.rand(n_points) * 2 - 1  # Initial guess
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    # Define objective function with bounds and constraints
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_erdos_solution(h)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=[n_points / 2], ub=[n_points / 2])
    
    # Initial guess
    initial_guess = h.flatten()
    
    # Start timing
    start_time = time.time()
    
    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check if optimization ran within budget
    if time.time() - start_time > budget_s:
        print("Optimization exceeded time budget; returning initial construction")
        return (initial_h_values, evaluate_erdos_solution(initial_h_values), len(initial_h_values))
    
    # Extract best solution
    best_h = result.x.reshape(n_points)
    best_c5 = result.fun
    return (best_h, best_c5, n_points)
```
