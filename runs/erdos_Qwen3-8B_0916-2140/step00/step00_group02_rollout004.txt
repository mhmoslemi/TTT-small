def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial construction (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    if not np.isclose(sum_h, n_points / 2):
        initial_h = initial_h * (n_points / 2) / sum_h
    
    # Helper function to compute the overlap integral
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_reshaped = np.zeros(2 * n)
        h_reshaped[:n] = h
        h_reshaped[n:] = h
        corr = np.correlate(h_reshaped, 1 - h_reshaped[:n], mode="full")
        return np.max(corr * dx)
    
    # Optimization function to minimize C₅
    def cost_function(h_flat):
        h = h_flat.copy()
        h = h / (2.0 / n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    # Initial guess
    h_flat = initial_h.copy()
    h_flat = h_flat / (2.0 / n_points)
    h_flat = np.clip(h_flat, 0, 1)
    h_flat = h_flat * (n_points / 2) / np.sum(h_flat)
    
    # Optimization
    result = optimize.minimize(
        fun=cost_function,
        x0=h_flat,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_h = best_h / (2.0 / n_points)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    c5 = compute_c5(best_h)
    n_points = len(best_h)
    
    return (best_h, c5, n_points)
