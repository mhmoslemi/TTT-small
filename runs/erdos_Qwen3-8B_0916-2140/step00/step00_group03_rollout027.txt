import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import sys
import warnings

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points - 1):
        shifted_h[k:] = h[:n_points - k]
    shifted_h = np.concatenate([shifted_h[1:], shifted_h[0]])
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(np.abs(corr)) * dx
    return c5

def initial_h_values(n_points):
    if n_points == 70:
        return [0.5 if i % 7 == 0 else 0.0 for i in range(n_points)]
    else:
        return np.random.uniform(0, 1, n_points)

def run(seed=42, budget_s=600, **kwargs):
    n_points = 100
    max_iter = 100
    tol = 1e-5
    best_c5 = np.inf
    best_h = None
    best_n = n_points

    # Initialize with the provided initial construction
    h_initial = initial_h_values(n_points)
    h_initial = np.array(h_initial, dtype=np.float64)
    if np.sum(h_initial) * 2.0 / n_points != 1.0:
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = (h_initial - np.min(h_initial)) / (np.max(h_initial) - np.min(h_initial))
        h_initial = (h_initial - 0.5) * 2.0 + 0.5
    h_initial = h_initial * 2.0 / n_points
    h_initial = np.clip(h_initial, 0, 1)

    start_time = time.time()
    elapsed = 0

    # Define the objective function
    def objective(x):
        nonlocal best_c5, best_h, best_n, n_points
        h = x.copy()
        if np.sum(h) * 2.0 / n_points != 1.0:
            return np.inf
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(corr)) * 2.0 / n_points
        return c5

    # Define constraints
    def constraint(x):
        return np.sum(x) * 2.0 / n_points - 1.0

    # Bounds
    bounds = Bounds(0, 1)

    # Initial guess
    x0 = h_initial

    # Optimization settings
    options = {
        'maxiter': max_iter,
        'ftol': tol,
        'disp': False
    }

    # Perform optimization
    result = minimize(objective, x0, bounds=bounds, constraints=LinearConstraint(np.ones(n_points), 1.0, 1.0), method='SLSQP', options=options)

    best_h = result.x
    best_c5 = result.fun
    best_n = n_points

    # Check if we can reduce n_points further
    for n_points in [50, 40, 30, 20, 10]:
        h_initial = initial_h_values(n_points)
        if np.sum(h_initial) * 2.0 / n_points != 1.0:
            h_initial = np.random.uniform(0, 1, n_points)
            h_initial = (h_initial - np.min(h_initial)) / (np.max(h_initial) - np.min(h_initial))
            h_initial = (h_initial - 0.5) * 2.0 + 0.5
        h_initial = h_initial * 2.0 / n_points
        h_initial = np.clip(h_initial, 0, 1)
        x0 = h_initial
        result = minimize(objective, x0, bounds=bounds, constraints=LinearConstraint(np.ones(n_points), 1.0, 1.0), method='SLSQP', options=options)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            best_n = n_points

    # Check if we can increase n_points further
    for n_points in [150, 200]:
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = (h_initial - np.min(h_initial)) / (np.max(h_initial) - np.min(h_initial))
        h_initial = (h_initial - 0.5) * 2.0 + 0.5
        h_initial = h_initial * 2.0 / n_points
        h_initial = np.clip(h_initial, 0, 1)
        x0 = h_initial
        result = minimize(objective, x0, bounds=bounds, constraints=LinearConstraint(np.ones(n_points), 1.0, 1.0), method='SLSQP', options=options)
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            best_n = n_points

    # Ensure we stop before the time limit
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h.tolist(), best_c5, best_n)

    return (best_h.tolist(), best_c5, best_n)
