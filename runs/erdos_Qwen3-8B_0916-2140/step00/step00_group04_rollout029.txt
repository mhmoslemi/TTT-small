import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from scipy.optimize import minimize

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

n_points = 50
dx = 2.0 / n_points
initial_sum = n_points / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided values
    h = initial_h_values[:n_points].copy()
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-6:
        h = np.array([0.5] * n_points)
    
    def objective(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def grad_objective(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        corr = np.correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if i - k >= 0:
                    grad[i] -= (1 - h[k]) * np.where(i - k == max_idx, 1, 0)
                if i + k < n_points:
                    grad[i] -= (1 - h[k]) * np.where(i + k == max_idx, 1, 0)
        return grad
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    def constraint_bounds(h_values):
        return np.all((0 <= h_values) & (h_values <= 1))
    
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        jac=grad_objective,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    if best_c5 < 0.380876:
        print(f"Improved bound found: {best_c5}")
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Time budget exceeded: {elapsed_time:.2f} seconds")
    
    return (best_h, best_c5, n_points)
