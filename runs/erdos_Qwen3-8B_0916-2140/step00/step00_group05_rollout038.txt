import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import minres
from scipy.optimize import LinearConstraint
from itertools import product
import sys
import os
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy') if os.path.exists('initial_h_values.npy') else np.random.rand(70)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with a moderate number of points
    h = np.random.rand(n_points)
    h = h / (2.0 / n_points)  # Normalize to ensure the integral is 1
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-5
    eval_freq = 10
    best_c5 = float('inf')
    best_h = h.copy()
    best_n = n_points
    
    for iteration in range(max_iter):
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_n = n_points
        
        # Check if we should increase the number of points
        if iteration % eval_freq == 0 and n_points < 1000:
            n_points += 50
            h = np.random.rand(n_points)
            h = h / (2.0 / n_points)
            h = np.clip(h, 0, 1)
            print(f"INCREASED POINTS TO {n_points} (Iteration {iteration})")
        
        # Simple gradient descent-like update
        shifts = np.linspace(-1.0, 1.0, 50)
        overlap = np.zeros_like(h)
        for shift in shifts:
            shifted_h = np.roll(h, int(shift * n_points / 2))
            overlap += np.abs(h - shifted_h)
        grad = -overlap / (2.0 / n_points)
        
        # Update h with a small step
        h += grad * 0.1
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h * (2.0 / n_points) / np.mean(h)
        
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final C5: {final_c5}")
    print(f"Best C5: {best_c5}")
    print(f"Best n_points: {best_n}")
    
    return best_h, final_c5, best_n
