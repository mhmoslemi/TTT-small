```python
import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time
import sys
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define the target C₅ value
    target_c5 = 0.380876
    n_points = 100  # Starting with 100 points, can be increased if needed
    best_c5 = float('inf')
    best_h = None
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        h_array = np.array(h)
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initial guess
    h_initial = initial_h_values[:]
    h_initial = h_initial[:n_points]
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    
    # Define a function to optimize
    def objective(x):
        nonlocal best_c5, best_h
        h = x.copy()
        h[h < 0] = 0
        h[h > 1] = 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Define bounds for each sample
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B
    result = optimize.minimize(
        objective,
        x0=initial_h_values[:n_points],
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # After optimization, we also try a few random perturbations to avoid local minima
    for _ in range(10):
        perturbed = np.array(best_h)
        perturbed += np.random.uniform(-0.1, 0.1, perturbed.shape)
        perturbed = np.clip(perturbed, 0, 1)
        c5_val = compute_c5(perturbed)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = perturbed
    
    # Final check: if best_c5 is below target, we can increase n_points for better approximation
    if best_c5 < target_c5:
        n_points = 200
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
