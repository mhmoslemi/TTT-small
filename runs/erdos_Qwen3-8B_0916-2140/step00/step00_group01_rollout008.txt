import numpy as np
import scipy.optimize as opt
import time
import random

# Pre-imported functions
initial_h_values = np.array([...])  # Assuming this is a valid starting point

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    c5 = max_corr * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    n_points = len(best_h)
    dx = 2.0 / n_points
    initial_c5 = evaluate_erdos_solution(best_h)
    best_c5 = initial_c5
    
    # Define optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    momentum = 0.9
    tolerance = 1e-5
    
    # Define objective function
    def objective(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5
    
    # Gradient descent with momentum
    def optimize_h(h_initial):
        h = h_initial.copy()
        v = np.zeros_like(h)
        for _ in range(max_iterations):
            grad = np.zeros_like(h)
            for i in range(n_points):
                for k in range(n_points):
                    if i + k < n_points:
                        val = np.correlate(h, 1 - h, mode="full")[i + k]
                    else:
                        val = np.correlate(h, 1 - h, mode="full")[i + k - n_points]
                    grad[i] += (val - np.min(val)) / n_points
            v = momentum * v - learning_rate * grad
            h += v
            h = np.clip(h, 0, 1)
            current_c5 = objective(h)
            if current_c5 < best_c5:
                nonlocal best_c5
                best_c5 = current_c5
                best_h = h.copy()
            if time.time() - start_time > budget_s - 1:
                return best_h, best_c5
    
    # Run optimization
    optimized_h, optimized_c5 = optimize_h(best_h)
    
    # Final check to ensure best_h is valid
    final_c5 = evaluate_erdos_solution(optimized_h)
    return optimized_h, final_c5, len(optimized_h)
