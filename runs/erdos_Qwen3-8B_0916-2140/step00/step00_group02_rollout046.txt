def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default configuration
    n_points = 100
    max_iter = 100
    tol = 1e-4
    best_c5 = float('inf')
    best_h = None
    
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
    
    def update_h(h):
        nonlocal best_c5, best_h
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        return current_c5
    
    # Start from initial construction
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h *= 2.0 / n_points  # Ensure integral is approximately 1
        h = np.clip(h, 0, 1)  # Ensure h stays in [0,1]
    
    best_c5 = update_h(h)
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Attempt to find better h by perturbing
        h_perturbed = h + 0.01 * np.random.randn(len(h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed *= 2.0 / len(h_perturbed)  # Ensure integral is exactly 1
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        current_c5 = update_h(h_perturbed)
        
        # Optionally, use convex optimization for better results
        # This is a placeholder for more advanced optimization
        # Here we use a simple gradient descent approach for demonstration
        
        # Simple gradient descent step
        h_grad = np.zeros_like(h)
        for shift in range(1, len(h) - 1):
            overlap = np.correlate(h, 1 - h, mode="full")
            grad = np.gradient(overlap, axis=0)
            h_grad[shift] = grad[shift]
        h = h - 0.1 * h_grad
        h = np.clip(h, 0, 1)
        h *= 2.0 / len(h)
        h = np.clip(h, 0, 1)
        
        current_c5 = update_h(h)
    
    return best_h, best_c5, n_points
