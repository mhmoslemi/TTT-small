```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize, DifferentialEvolution
import time

# Pre-imported functions
def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1 - h, mode="full")
    return np.max(result) * dx

def initial_h_values():
    return np.load('initial_h.npy')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from preloaded data
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Objective function to minimize: max overlap integral
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Budget-based optimization strategy
    # Start with Differential Evolution to escape local optima
    result_de = DifferentialEvolution(objective, bounds=[(0, 1)] * n_points, strategy='best1bin', maxiter=100, popsize=20, mutation=0.9, recombination=0.5, tol=1e-6)
    best_h_de = result_de.x
    best_c5_de = objective(best_h_de)
    print(f"Differential Evolution result: {best_c5_de}")
    
    # Refine using L-BFGS with initial guess from DE
    result_lbfgs = minimize(objective, x0=best_h_de, method='L-BFGS-B', bounds=[(0, 1)] * n_points, tol=1e-6, max_iter=100)
    best_h_lbfgs = result_lbfgs.x
    best_c5_lbfgs = objective(best_h_lbfgs)
    print(f"L-BFGS-B result: {best_c5_lbfgs}")
    
    # Further refinement using Nelder-Mead
    result_nm = minimize(objective, x0=best_h_lbfgs, method='Nelder-Mead', bounds=[(0, 1)] * n_points, tol=1e-6, max_iter=50)
    best_h_nm = result_nm.x
    best_c5_nm = objective(best_h_nm)
    print(f"Nelder-Mead result: {best_c5_nm}")
    
    # Final refinement with high-precision Nelder-Mead
    result_nm_final = minimize(objective, x0=best_h_nm, method='Nelder-Mead', bounds=[(0, 1)] * n_points, tol=1e-8, max_iter=100)
    best_h_final = result_nm_final.x
    best_c5_final = objective(best_h_final)
    
    # Format result
    n_points = len(best_h_final)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        print("Time budget exceeded, returning the best result so far.")
    
    return (best_h_final, best_c5_final, n_points)
```
