import numpy as np
import scipy.optimize as opt

# Pre-imported functions

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Starting with 100 points
    max_iter = 100
    tol = 1e-4
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate((h, h))  # To handle wrap-around
        shifts = np.arange(-n_points + 1, n_points)
        max_overlap = 0.0
        for k in shifts:
            if k == 0:
                continue
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Function to evaluate the objective for optimization
    def objective(h):
        return compute_C5(h)
    
    # Function to generate an initial guess
    def generate_initial_guess():
        h = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        return h
    
    # Function to perform gradient descent
    def gradient_descent(initial_guess, max_iter, tol):
        h = initial_guess.copy()
        for _ in range(max_iter):
            # Compute the C5 value
            c5 = compute_C5(h)
            
            # Compute the gradient manually (approximate using finite differences)
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                c5_plus = compute_C5(h_plus)
                grad[i] = (c5_plus - c5) / 1e-6
            
            # Update the solution
            h = h - 0.01 * grad
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            total = np.sum(h) * dx
            h = h * (1 / dx) / (np.sum(h) / dx)
            
            # Check for convergence
            if np.abs(np.sum(grad)) < tol:
                break
        return h
    
    # Generate initial guess
    h_initial = generate_initial_guess()
    
    # Run gradient descent
    h_opt = gradient_descent(h_initial, max_iter, tol)
    
    # Evaluate the result
    c5 = compute_C5(h_opt)
    
    # Check if we need to increase n_points
    if c5 > 0.380876:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = generate_initial_guess()
        h_opt = gradient_descent(h_initial, max_iter, tol)
        c5 = compute_C5(h_opt)
    
    # Final evaluation
    final_c5 = compute_C5(h_opt)
    return h_opt, final_c5, n_points
