import numpy as np
import time

from functools import lru_cache

# Predefined initial solution (70 samples)
initial_h_values = np.array([...])  # Replace with actual initial values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Helper: Compute the overlap integral for a given h
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap * dx)
    
    # Initial guess from the provided initial_h_values
    h_initial = initial_h_values
    n_points = len(h_initial)
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-5
    best_c5 = compute_c5(h_initial)
    best_h = h_initial.copy()
    
    # Try a few perturbations to avoid local minima
    for perturbation in range(3):
        # Perturb the initial guess
        h_perturbed = h_initial.copy()
        idx = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        h_perturbed[idx] = np.random.uniform(0, 1, size=len(idx))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Try different sample sizes (smaller for faster evaluation)
        for n in [50, 100, 200]:
            # Resample h to n_points samples
            h_resampled = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, n_points), h_perturbed)
            h_resampled = np.clip(h_resampled, 0, 1)
            
            # Check if the integral condition is approximately satisfied
            if np.abs(np.sum(h_resampled) - n / 2) > 1e-3:
                continue
            
            # Evaluate C5
            start_time = time.time()
            c5 = compute_c5(h_resampled)
            elapsed_time = time.time() - start_time
            
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_resampled
            
            # Early stopping if time is running out
            if elapsed_time > budget_s * 0.8:
                break
    
    # Final optimization with CVXPY for finer adjustments
    n_points = 200
    h = cp.Variable(n_points)
    dx = 2.0 / n_points
    objective = cp.Minimize(cp.Max(cp.sum([h[i] * (1 - h[(i + k) % n_points]) * dx for k in range(1, 5)])))
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    problem = cp.Problem(objective, constraints)
    start_time = time.time()
    try:
        problem.solve(solver=cp.SCS, max_iter=100, eps=1e-5)
    except:
        pass
    
    # If CVXPY didn't solve, use the best solution so far
    if best_c5 is not None:
        final_h = best_h
    else:
        final_h = h.value if h is not None else initial_h_values
    
    # Final evaluation
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
