```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.interpolate import interp1d
import time

# Pre-imported functions
initial_h_values = np.load("initial_h.npy")

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points to balance speed and accuracy
    n_points_list = [100, 200, 300]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        start_time = time.time()
        
        # Generate initial samples
        if n_points == 100:
            h = initial_h_values[:100]
        else:
            h = np.random.rand(n_points) * 2 - 1  # Random initial guess
        
        # Ensure sum(h) == n_points / 2
        h = h / np.sum(h) * (n_points / 2)
        
        # Define the function to minimize
        def objective(x):
            nonlocal n_points
            h_vals = x
            # Ensure h_vals are within [0, 1]
            h_vals = np.clip(h_vals, 0, 1)
            
            # Compute the overlap integral for all k
            c5_values = []
            for k in range(1, n_points):
                # Interpolate h
                x_vals = np.linspace(0, 2, n_points)
                h_interp = interp1d(x_vals, h_vals, kind='linear', fill_value='extrapolate')
                
                # Compute 1 - h(x + k)
                shifted_x = np.linspace(k / n_points, (k + 1) / n_points, n_points)
                shifted_h = h_interp(shifted_x)
                shifted_h = np.clip(shifted_h, 0, 1)
                term = h_interp(x_vals) * (1 - shifted_h)
                integral = np.sum(term) * (2 / n_points)
                c5_values.append(integral)
            
            # Compute max_c5
            current_c5 = max(c5_values)
            return current_c5
        
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run optimization
        result = minimize(
            objective,
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        
        # Extract the best solution
        best_h_candidate = result.x
        
        # Compute the final C₅ value
        h_vals = np.clip(best_h_candidate, 0, 1)
        x_vals = np.linspace(0, 2, n_points)
        
        # Compute the overlap integral for all k
        c5_values = []
        for k in range(1, n_points):
            # Interpolate h
            h_interp = interp1d(x_vals, h_vals, kind='linear', fill_value='extrapolate')
            
            # Compute 1 - h(x + k)
            shifted_x = np.linspace(k / n_points, (k + 1) / n_points, n_points)
            shifted_h = h_interp(shifted_x)
            shifted_h = np.clip(shifted_h, 0, 1)
            term = h_interp(x_vals) * (1 - shifted_h)
            integral = np.sum(term) * (2 / n_points)
            c5_values.append(integral)
        
        final_c5 = max(c5_values)
        
        # Update best solution
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = best_h_candidate
            best_n = n_points
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, best_n
```
