# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.concatenate((h[-n_points+1:], h))
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values():
    return [0.5] * 45

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction from initial_h_values
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    if sum_h != n_points / 2:
        initial_h = [0.5] * n_points

    best_c5 = evaluate_erdos_solution(initial_h)
    best_h = initial_h.copy()

    # Try reducing n_points for faster evaluation
    for n in range(40, 10, -1):
        n_points = n
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Initialize with a simple pattern to avoid getting stuck in local optima
        for i in range(n_points):
            h[i] = 0.5 if i % 3 == 0 else 0.25
        # Normalize to ensure ∫h(x)dx = 1
        h = h / (np.sum(h) * dx)
        h = np.clip(h, 0, 1)
        c5_val = evaluate_erdos_solution(h.tolist())
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.tolist()
        if time.time() - start_time > budget_s - 10:
            break

    # Advanced optimization
    if time.time() - start_time < budget_s - 10:
        for n in range(60, 30, -1):

            dx = 2.0 / n_points
            h = np.zeros(n_points)
            # Try a sinusoidal-like pattern to spread out the function
            for i in range(n_points):
                h[i] = 0.5 * (1 + np.sin(2 * np.pi * i / n_points))
            # Normalize to ensure ∫h(x)dx = 1
            h = h / (np.sum(h) * dx)
            h = np.clip(h, 0, 1)
            c5_val = evaluate_erdos_solution(h.tolist())
            if c5_val < best_c5:
                best_c5 = c5_val

            if time.time() - start_time > budget_s - 10:
                break

    # Use a more sophisticated optimization approach
    if time.time() - start_time < budget_s - 10:
        n_points = 60
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Try to place the function in a way that it avoids overlap with shifted versions
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
        h = h / (np.sum(h) * dx)
        h = np.clip(h, 0, 1)
        c5_val = evaluate_erdos_solution(h.tolist())
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.tolist()

    # Use a more advanced numerical optimization
    if time.time() - start_time < budget_s - 10:

        dx = 2.0 / n_points
        h = np.zeros(n_points)

        for i in range(n_points):

                h[i] = 0.75
            elif i % 4 == 0:
                h[i] = 0.25
        h = h / (np.sum(h) * dx)
        h = np.clip(h, 0, 1)
        c5_val = evaluate_erdos_solution(h.tolist())
        if c5_val < best_c5:

    return (best_h, best_c5, len(best_h))
