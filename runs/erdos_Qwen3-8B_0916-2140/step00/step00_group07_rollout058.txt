import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with n_points = 80 from initial_h_values
    n_points_initial = 80
    initial_h = initial_h_values[:n_points_initial]
    
    # We'll try several n_points starting from 80 and increasing by 20
    n_points_list = [80, 100, 120, 140, 160, 180, 200]
    
    best_c5 = float('inf')
    best_h = None
    best_n = n_points_initial
    
    for n_points in n_points_list:
        if n_points < 80:
            continue
        
        # Ensure the initial_h is scaled to match the current n_points
        if n_points_initial != n_points:
            # Interpolate the initial_h to the new n_points
            x = np.linspace(0, 2, n_points_initial)
            X = np.linspace(0, 2, n_points)
            h_interpolated = np.interp(X, x, initial_h)
            h = h_interpolated.copy()
        else:
            h = initial_h.copy()
        
        # Ensure the integral condition: sum(h) == n_points / 2
        h_sum = np.sum(h)
        scale = (n_points / 2) / h_sum
        h = h * scale
        
        # Use simulated annealing to optimize the function
        start_time = time.time()
        best_c5_current = float('inf')
        best_h_current = h.copy()
        
        def objective(h_flat):
            h = h_flat.reshape((n_points,))
            # Check constraints
            if np.any(h < 0) or np.any(h > 1):
                return float('inf')
            
            # Compute the overlap integral for all k in [0, 1]
            max_c5 = 0
            for k in np.linspace(0, 1, 100):
                # Wrap the array to handle circularity
                h_shifted = np.roll(h, int(k * n_points))
                overlap = np.sum(h * (1 - h_shifted)) * (2 / n_points)
                if overlap > max_c5:
                    max_c5 = overlap
            return max_c5
        
        # Simulated annealing parameters
        temp = 1.0
        cooling_rate = 0.995
        max_iterations = 500
        best_h_flat = h.copy().flatten()
        best_c5_current = objective(best_h_flat)
        
        for _ in range(max_iterations):
            # Generate a random perturbation
            perturbation = np.random.normal(0, 0.1, n_points)
            new_h_flat = best_h_flat + perturbation
            new_h_flat = np.clip(new_h_flat, 0, 1)
            
            # Compute the new objective
            new_c5 = objective(new_h_flat)
            
            # Accept the new solution with probability
            if new_c5 < best_c5_current or np.random.rand() < np.exp((best_c5_current - new_c5) / temp):
                best_h_flat = new_h_flat
                best_c5_current = new_c5
            
            temp *= cooling_rate
        
        # Convert back to h values
        best_h_current = best_h_flat.reshape(n_points)
        elapsed_time = time.time() - start_time
        
        if elapsed_time > budget_s:
            break
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n = n_points
            
        print(f"n_points={n_points}, C5={best_c5_current:.6f}")
    
    # Return the best solution found
    return (best_h, best_c5, best_n)
