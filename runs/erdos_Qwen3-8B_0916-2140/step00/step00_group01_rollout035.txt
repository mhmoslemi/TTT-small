def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Define initial_h_values
    if 'initial_h_values' in globals():
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(47)
    
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - n_points // 2]])  # Extend for full convolution
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Objective function to minimize
    def obj_func(x):
        x = np.array(x)
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Optimization settings
    options = {'disp': False, 'maxiter': 100, 'ftol': 1e-8}
    
    # Perform the optimization
    start_time = time.time()
    best_c5 = float('inf')
    best_h = x0
    
    # Run optimization
    result = minimize(obj_func, x0, method='SLSQP', constraints=cons, options=options)
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we can perform additional iterations if time allows
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        # Try perturbing the solution slightly
        for _ in range(3):
            perturb = np.random.normal(0, 0.05, n_points)
            perturb = np.clip(perturb, -0.1, 0.1)
            x_perturb = np.clip(best_h + perturb, 0, 1)
            x_perturb = np.clip(x_perturb, 0, 1)
            x_perturb = x_perturb / np.sum(x_perturb) * n_points / 2
            current_c5 = compute_c5(x_perturb)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = x_perturb
    
    return (best_h, best_c5, len(best_h))
