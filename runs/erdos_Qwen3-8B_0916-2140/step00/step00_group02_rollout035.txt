import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points for faster evaluation
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h[:n_points]  # Ensure it's within the n_points
    h /= np.sum(h) * dx  # Normalize to make sure the integral is 1
    h = np.clip(h, 0, 1)  # Ensure values are in [0, 1]

    # Function to compute C5 given h
    def compute_c5(h):
        h = h.astype(np.float64)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Function to minimize the maximum overlap
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        return c5

    # Bounds for all h[i] in [0, 1]
    bounds = Bounds(0, 1)

    # Initial guess
    initial_guess = h.copy()

    # Run optimization with L-BFGS-B to find a better solution
    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'disp': False}
    )
    elapsed = time.time() - start_time

    best_h = result.x
    best_c5 = compute_c5(best_h)

    # If we have time, try increasing n_points
    if elapsed < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Discretize the best_h from the previous step
        for i in range(n_points):
            h[i] = np.interp(i * dx, np.linspace(0, 2, len(best_h)), best_h)
        h = np.clip(h, 0, 1)
        h /= np.sum(h) * dx
        h = np.clip(h, 0, 1)

        def compute_c5_with_new_n(h):
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5

        def objective_with_new_n(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            c5 = compute_c5_with_new_n(h)
            return c5

        bounds_new = Bounds(0, 1)
        initial_guess_new = h.copy()

        result_new = optimize.minimize(
            objective_with_new_n,
            initial_guess_new,
            method='L-BFGS-B',
            bounds=bounds_new,
            options={'maxiter': 50, 'disp': False}
        )
        best_h_new = result_new.x
        best_c5_new = compute_c5_with_new_n(best_h_new)

        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new

    # Return the best solution found
    return (best_h, best_c5, len(best_h))
