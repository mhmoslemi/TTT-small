import numpy as np
import scipy.optimize as sco
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Initial guess from provided initial_h_values
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    h_initial = h_initial / np.sum(h_initial) * n_points_initial / 2

    # We will try several starting points with different n_points
    n_points_options = [100, 200, 300, 400, 500]

    for n_points in n_points_options:
        if time.time() - start_time > budget_s - 10:
            break

        print(f"\nTrying n_points = {n_points}...")

        # Define grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)

        def objective(h):
            nonlocal best_c5, best_h, best_n_points
            h = np.array(h, dtype=np.float64)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if np.abs(sum_h - n_points / 2) > 1e-8:
                return float('inf')

            # Compute 1 - h(x + k)
            c5 = 0
            for k in range(1, n_points):
                if k >= n_points:
                    break
                h_shifted = np.roll(h, k)
                overlap = np.sum(h * (1 - h_shifted))
                c5 = max(c5, overlap)

            # Normalize by dx
            c5 *= dx
            return c5

        # Optimization
        initial_guess = np.random.rand(n_points) * 2 - 1
        result = sco.minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(0, 1)] * n_points, options={'maxiter': 100})
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n_points = n_points

    print(f"\nBest C5 found: {best_c5}")
    return (best_h, best_c5, best_n_points)
