    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Function to compute the overlap integral
    def compute_c5(h):

        h = np.concatenate([h, h])  # To handle wrap-around
        result = np.correlate(h, 1 - h, mode="full")
        return np.max(result) * dx

    # Objective function to minimize
    def objective(h):
        return compute_c5(h)

    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimize using gradient descent with constraints
    def optimize_h(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"maxiter": 1000, "ftol": 1e-8}
        )
        if result.success:
            return result.x
        else:
            return initial_guess

    # Run optimization
    best_h = initial_h
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    # Simple greedy descent to explore
    for _ in range(50):
        current_h = best_h
        current_c5 = best_c5

        # Try perturbing a single point
        for i in range(n_points):
            h = copy.copy(current_h)
            h[i] = np.clip(h[i] + np.random.uniform(-0.1, 0.1), 0, 1)
            c5 = compute_c5(h)
            if c5 < current_c5:
                best_h = h
                best_c5 = c5
                break

        # Try restarting with new initial guess
        new_h = np.random.uniform(0, 1, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = optimize_h(new_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
