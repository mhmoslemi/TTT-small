import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
import time
from functools import lru_cache

def run(seed=42, budget_s=600, **kwargs):

    np.random.seed(seed)

    def integrate_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlap = np.zeros(n)
        for i in range(n):
            for k in range(1, n):
                if i + k < n:
                    overlap[i] += h[i] * (1 - h[i + k])
                if i - k >= 0:
                    overlap[i] += h[i] * (1 - h[i - k])
        return np.max(overlap * dx)

    def build_matrix(n_points):
        n = n_points
        A = np.zeros((n, n))
        for i in range(n):
            for k in range(1, n):
                if i + k < n:
                    A[i, i + k] += 1
                if i - k >= 0:
                    A[i, i - k] += 1
        return sparse.csr_matrix(A)

    def evaluate_cost(h, n_points):
        dx = 2.0 / n_points
        overlap = np.zeros(n_points)
        for i in range(n_points):
            for k in range(1, n_points):
                if i + k < n_points:
                    overlap[i] += h[i] * (1 - h[i + k])
                if i - k >= 0:
                    overlap[i] += h[i] * (1 - h[i - k])
        return np.max(overlap * dx)

    def cvxpy_minimize_overlap(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        problem = optimize.OptimizeResult()
        problem.x = h
        problem.fun = evaluate_cost(h, n_points)
        return problem

    def compute_initial_solution(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points / 2:
                h[i] = 1
            else:
                h[i] = 0
        return h

    def build_optimization_problem(n_points):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return h

    def optimize_h(n_points, max_iterations=100):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        for _ in range(max_iterations):
            overlap = np.zeros(n_points)
            for i in range(n_points):
                for k in range(1, n_points):
                    if i + k < n_points:
                        overlap[i] += h[i] * (1 - h[i + k])
                    if i - k >= 0:
                        overlap[i] += h[i] * (1 - h[i - k])
            max_overlap = np.max(overlap * 2.0 / n_points)
            if max_overlap < 0.380876:
                break
            for i in range(n_points):
                h[i] = max(0, min(1, h[i] - 0.1))
        return h

    def fine_tune_solution(h, n_points):
        h = h.copy()
        for i in range(n_points):
            for k in range(1, n_points):
                if i + k < n_points:
                    h[i] = max(0, min(1, h[i] - 0.01))
                if i - k >= 0:
                    h[i] = max(0, min(1, h[i] - 0.01))
        return h

    def get_initial_solution():
        h = np.zeros(47)
        for i in range(47):
            if i < 23:
                h[i] = 1
            else:
                h[i] = 0
        return h

    n_points = 1000
    h = get_initial_solution()
    for i in range(n_points):
        h[i] = max(0, min(1, h[i]))
    h = h / np.sum(h) * (n_points / 2)

    start_time = time.time()
    for _ in range(10):
        h = optimize_h(n_points)
        if time.time() - start_time > budget_s - 1:
            break
    h = fine_tune_solution(h, n_points)
    c5_bound = evaluate_cost(h, n_points)
    return h, c5_bound, n_points
