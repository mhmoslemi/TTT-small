import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100  # Starting with 100 points for balance between speed and quality
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initialize with a step function to avoid local optima
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 0.5
        else:
            h[i] = 0.5

    h += np.random.normal(0, 0.05, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1

    best_c5 = np.inf
    best_h = h.copy()
    best_n_points = n_points

    # Optimization strategy: Use a combination of grid search and local optimization
    # Try increasing n_points in steps up to 500 if time allows

    for n_points in [100, 200, 300, 400, 500]:
        start_time = time.time()
        if start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        
        # Start with a good initial guess (from previous step)
        if n_points == 100:
            h = best_h.copy()
        else:
            # Resample from the best solution so far
            h = np.random.choice(best_h, size=n_points, replace=True)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        # Define the optimization problem using CVXPY
        x = cp.Variable(n_points)
        constraints = [x >= 0, x <= 1, cp.sum(x) == n_points / 2]

        # Objective: minimize the maximum overlap integral
        # This is a semi-infinite optimization problem and approximated through sampling

        # Sample a set of k to evaluate the overlap integral for
        k_samples = np.linspace(-1, 1, 100)  # Sample shifts near 0
        objective = cp.Minimize(cp.Max([cp.sum(x[i] * (1 - x[(i + int(k * n_points)) % n_points])) for k in k_samples]))

        try:
            prob.solve(solver=cp.SCS, eps=1e-3, max_iter=100, time_limit=100)
        except:
            pass

        if prob.status == 'optimal':

            c5 = np.max([np.sum(current_h[i] * (1 - current_h[(i + int(k * n_points)) % n_points])) for k in k_samples]) * dx

                best_c5 = c5
                best_h = current_h.copy()
                best_n_points = n_points
            print(f"n_points={n_points}, c5={c5}")

            # Fallback to previous best if needed

    print(f"Best solution found with n_points={best_n_points}, c5={best_c5}")
    return (best_h, best_c5, best_n_points)
