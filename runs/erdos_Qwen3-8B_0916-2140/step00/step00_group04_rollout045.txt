import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values (n=42 samples)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define the function to compute the max overlap integral
    def compute_C5(h):
        h = np.array(h)
        shifts = np.arange(-n_points + 1, n_points)
        max_overlap = 0
        for k in shifts:
            if k == 0:
                continue
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Objective function to minimize (max overlap integral)
    def objective(h):
        return compute_C5(h)
    
    # Constraints: sum(h) * dx = 1 => sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess from the initial_h_values
    initial_guess = initial_h
    
    # Optimization: use a local optimizer with restarts
    def optimize_with_restarts():
        best_c5 = float('inf')
        best_h = initial_guess.copy()
        
        for _ in range(10):  # Restart limit
            start_time = time.time()
            result = minimize(
                objective,
                initial_guess,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_sum},
                method='L-BFGS-B',
                options={'maxiter': 100, 'disp': False}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
            else:
                # If optimization fails, try a random perturbation
                perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
                initial_guess = np.clip(best_h + perturbation, 0, 1)
            
            elapsed_time = time.time() - start_time
            if elapsed_time > budget_s / 2:
                break
        
        return best_h, best_c5
    
    best_h, best_c5 = optimize_with_restarts()
    return best_h, best_c5, n_points
