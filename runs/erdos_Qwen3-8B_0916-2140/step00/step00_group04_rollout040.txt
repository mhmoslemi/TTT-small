import numpy as np
import scipy.optimize as optimize
import time
import functools

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    # Define the objective function to minimize
    def objective(h):
        nonlocal n_points
        h = np.array(h, dtype=np.float64)
        # Rescale to [0, 2]
        x = np.linspace(0, 2, n_points)
        # Compute the correlation
        result = np.correlate(h, 1 - h, mode='full')
        # Take the middle part of the correlation (around the peak)
        center = len(result) // 2
        result = result[center - 200: center + 200]
        # Multiply by dx
        dx = 2.0 / n_points
        result = result * dx
        # Find the maximum value
        c5 = np.max(result)
        return c5
    
    # Define the constraint: sum(h) * dx = 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to evaluate and update best solution
    def evaluate_and_update(h_candidate):
        nonlocal best_c5, best_h
        c5 = objective(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
    
    # Function to optimize with bounds
    def optimize_with_bounds(h_initial):
        result = optimize.minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP'
        )
        if result.success:
            evaluate_and_update(result.x)
        return result
    
    # Start from initial solution
    optimize_with_bounds(best_h)
    
    # Try a few perturbations
    for _ in range(10):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Rescale to ensure the constraint is satisfied
        scale_factor = (n_points / 2) / np.sum(perturbed_h)
        perturbed_h = perturbed_h * scale_factor
        optimize_with_bounds(perturbed_h)
    
    # Try reducing the number of points
    for n in range(n_points_initial - 1, 10, -1):
        n_points = n
        x = np.linspace(0, 2, n)
        # Resample best_h to n points
        resampled_h = np.interp(x, np.linspace(0, 2, len(best_h)), best_h)
        # Optimize with new n
        optimize_with_bounds(resampled_h)
    
    # Try with even fewer points
    for n in range(10, 5, -1):
        x = np.linspace(0, 2, n)
        resampled_h = np.interp(x, np.linspace(0, 2, len(best_h)), best_h)
        # Optimize with new n
        optimize_with_bounds(resampled_h)
    
    return best_h, best_c5, n_points
