import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from functools import partial

# Pre-imported functions
initial_h_values = np.array([0.407, 0.407, 0.399, 0.399, 0.399, 0.399, 0.399, 0.399, 0.399, 0.399, 0.399, 0.399,
                             0.399, 0.399, 0.399, 0.399, 0.399, 0.399, 0.399, 0  # truncated for clarity
                             ])  # example initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define initial points
    n_points = 200  # chosen for a balance between accuracy and speed
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initialize h_values from the initial_h_values (truncated and scaled)
    initial_h = initial_h_values[:n_points // 2] if len(initial_h_values) > n_points // 2 else initial_h_values
    initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), mode='constant', constant_values=0.5)
    initial_h = initial_h[:n_points]
    
    # Ensure sum(h) * dx == 1
    initial_h = np.clip(initial_h, 0, 1)
    initial_sum = np.sum(initial_h) * dx
    scale_factor = 1.0 / initial_sum
    initial_h = np.clip(initial_h * scale_factor, 0, 1)
    
    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, h])  # double the array for full convolution
        conv = signal.correlate(h, 1 - h[:n_points], mode='full')
        overlap = conv[n_points-1:] * dx
        return np.max(overlap)
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Use a local optimizer with a good initial guess
    result = optimize.minimize(
        fun=objective,
        x0=initial_h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol}
    )
    
    h_values = result.x
    c5_bound = result.fun
    
    return h_values, c5_bound, n_points
