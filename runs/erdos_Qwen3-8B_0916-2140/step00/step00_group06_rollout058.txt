    n_points = len(h_initial)
    dx = 2.0 / n_points

    if np.isclose(sum_h, n_points / 2):
        pass
    else:
        h_initial = h_initial / (sum_h / (n_points / 2))
    
    def evaluate_c5(h):
        h = np.array(h)
        h = np.append(h, h[0])  # To handle wrap-around
        cor = np.correlate(h, 1 - h, mode="full")
        cor = cor[len(cor) // 2 : ]  # Keep the relevant part
        c5 = np.max(cor * dx)
        return c5
    
    def objective(x, dx, n_points):
        return evaluate_c5(x)
    
    def constraints(x, n_points):
        return np.sum(x) - n_points / 2
    
    def bounds(n_points):
        return Bounds(0, 1)
    
    def optimize_h(h_initial, n_points, dx, max_iterations=100, tolerance=1e-6):
        result = minimize(
            fun=lambda x: objective(x, dx, n_points),
            x0=h_initial,
            bounds=bounds(n_points),
            method='L-BFGS-B',
            options={'maxiter': max_iterations, 'ftol': tolerance}
        )
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
        return best_h, best_c5
    
    start_time = time.time()
    
    best_h = h_initial
    best_c5 = evaluate_c5(h_initial)
    
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        
        h_opt, c5_opt = optimize_h(best_h, n_points, dx)
        
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

            best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)
