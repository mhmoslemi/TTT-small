```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial sample size
    n_points = 100  # A reasonable balance between speed and accuracy
    
    # Initial guess based on the provided initial_h_values
    initial_guess = np.array(initial_h_values[:n_points])
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        return evaluate_erdos_solution(h)
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: each h[i] ∈ [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization setup
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Extract the best solution found
    best_h = result.x
    c5_bound = evaluate_erdos_solution(best_h)
    elapsed_time = time.time() - start_time
    
    # Ensure we return within the time budget
    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning best solution found so far")
    
    return (best_h, c5_bound, n_points)
```
