```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.interpolate import interp1d
from scipy.signal import correlate
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Precompute initial solution
    initial_h = initial_h_values
    n_points = len(initial_h)
    initial_c5 = evaluate_erdos_solution(initial_h)
    
    # Create a list of candidate starting points
    starting_points = [initial_h]
    
    # Function to generate random perturbation
    def perturb(h, magnitude=0.1):
        h_copy = h.copy()
        for i in range(len(h_copy)):
            h_copy[i] += np.random.uniform(-magnitude, magnitude)
            h_copy[i] = np.clip(h_copy[i], 0.0, 1.0)
        return h_copy
    
    # Try a few random perturbations as starting points
    for _ in range(3):
        starting_points.append(perturb(initial_h))
    
    best_c5 = initial_c5
    best_h = initial_h.copy()
    
    # Function to evaluate a solution
    def objective(x, n_points):
        h = x
        # Ensure h is in [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Ensure integral constraint is satisfied
        integral = np.sum(h) * (2.0 / n_points)
        # Adjust h to satisfy the integral constraint
        if integral < 1.0:
            diff = 1.0 - integral
            idxs = np.argsort(h)
            for i in idxs:
                if h[i] < 1.0:
                    h[i] += diff
                    diff = 0.0
                    break
        elif integral > 1.0:
            diff = integral - 1.0
            idxs = np.argsort(h)
            for i in idxs:
                if h[i] > 0.0:
                    h[i] -= diff
                    diff = 0.0
                    break
        # Compute C5
        c5_val = evaluate_erdos_solution(h)
        return c5_val
    
    # Optimize each starting point
    for start in starting_points:
        # Set up bounds
        bounds = Bounds(0.0, 1.0)
        
        # Optimization parameters
        options = {"maxiter": 50, "disp": False}
        
        # Run optimization
        result = minimize(objective, start, bounds=bounds, method="L-BFGS-B", options=options)
        
        # Check if result is better
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final evaluation
    final_c5 = best_c5
    final_h = best_h
    n_points = len(final_h)
    
    # Output the result
    return (final_h, final_c5, n_points)
```
