import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess
    n_points = 200
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        dx = 2.0 / n_points
        shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    shifted[i, k] = h[i]
                else:
                    shifted[i, k] = 0
        overlap = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    overlap[k] += shifted[i, k] * (1 - h[i + k])
        c5 = np.max(overlap * dx)
        return c5

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        dx = 2.0 / n_points
        h = h[:n_points]
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        overlap = np.zeros(n_points)
        for k in range(n_points):
            for i in range(n_points):
                if i + k < n_points:
                    overlap[k] += h[i] * (1 - h[i + k])
        c5 = np.max(overlap * dx)
        return c5

    def optimize_h(h_initial):
        result = minimize(evaluate, h_initial, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-8})
        return result.x

    best_h = optimize_h(h_initial)
    best_c5 = evaluate(best_h)
    return best_h, best_c5, n_points
