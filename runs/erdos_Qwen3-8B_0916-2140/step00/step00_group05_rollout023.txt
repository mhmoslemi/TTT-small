# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.roll(h, 1)
    overlap = np.correlate(h, 1 - h_shifted, mode="full")
    c5 = np.max(overlap) * dx
    return c5

def initial_h_values(n_points):
    # Initial construction: 70 samples
    # This is a smooth, piecewise linear function with 70 points
    x = np.linspace(0, 2, n_points)
    h = np.zeros_like(x)
    for i in range(n_points):
        if x[i] < 1:
            h[i] = x[i] / 1
        else:
            h[i] = 1 - (x[i] - 1) / 1
    return h.tolist()

def run(seed=42, budget_s=600, **kwargs):

    # Define variables
    max_budget = budget_s
    start_time = time.time()

    # Define initial parameters
    n_points_initial = 70
    h_initial = initial_h_values(n_points_initial)

    # Define a function to evaluate the overlap for a given h_values
    def evaluate_overlap(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        if n_points < 2:
            return float('inf')
        h_shifted = np.roll(h, 1)
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        return np.max(overlap) * dx

    # Define a function to generate a random h_values
    def generate_random_h(n_points):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        return h.tolist()

    # Define a function to perform a local search around a given solution
    def local_search(h_values, n_points, max_steps=50, step_size=0.05):
        h = np.array(h_values)
        for _ in range(max_steps):
            # Generate a random perturbation
            perturbation = np.random.normal(0, step_size, n_points)
            h_perturbed = np.clip(h + perturbation, 0, 1)
            # Evaluate the overlap
            c5 = evaluate_overlap(h_perturbed.tolist())
            if c5 < evaluate_overlap(h_values):
                h = h_perturbed
        return h.tolist()

    # Define a function to perform a random search over a range of sample sizes
    def random_search(max_n_points, seed):
        np.random.seed(seed)
        best_c5 = float('inf')
        best_h = None
        for n_points in range(10, max_n_points + 1):
            if (time.time() - start_time) > budget_s:
                break
            # Generate a random h_values
            h = generate_random_h(n_points)
            # Normalize the integral to 1
            total = np.sum(h) * (2.0 / n_points)
            h = np.array(h) / (total / (2.0 / n_points))
            # Evaluate the overlap
            c5 = evaluate_overlap(h.tolist())
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.tolist()
        return best_h, best_c5

    # Consider a range of sample sizes
    max_n_points = 100
    best_h = None
    best_c5 = float('inf')

    # Start with the initial solution
    h = initial_h_values(n_points_initial)
    total = np.sum(h) * (2.0 / n_points_initial)
    h = np.array(h) / (total / (2.0 / n_points_initial))
    c5 = evaluate_overlap(h.tolist())
    best_h = h.tolist()
    best_c5 = c5

    # Perform a local search around the initial solution
    h_local = local_search(h.tolist(), n_points_initial)
    c5_local = evaluate_overlap(h_local)
    if c5_local < best_c5:
        best_h = h_local
        best_c5 = c5_local

    # Perform a random search over a range of sample sizes
    # Start from n_points = 50 to 100
    for n_points in range(50, max_n_points + 1):
        if (time.time() - start_time) > budget_s:
            break
        h = generate_random_h(n_points)
        total = np.sum(h) * (2.0 / n_points)
        h = np.array(h) / (total / (2.0 / n_points))
        c5 = evaluate_overlap(h.tolist())
        if c5 < best_c5:
            best_h = h.tolist()
            best_c5 = c5

    # Perform a more structured search by adjusting the sample size
    for n_points in range(50, max_n_points + 1, 20):
        if (time.time() - start_time) > budget_s:
            break
        # Start from a previous solution
        start_h = best_h
        # Generate a new solution
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Normalize the integral
        total = np.sum(h) * (2.0 / n_points)
        h = np.array(h) / (total / (2.0 / n_points))
        c5 = evaluate_overlap(h.tolist())
        if c5 < best_c5:
            best_h = h.tolist()
            best_c5 = c5

    # Final check: local search
    best_h = local_search(best_h, len(best_h))
    final_c5 = evaluate_overlap(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Check for over-budget
    if (time.time() - start_time) > budget_s - 10:
        return (best_h, best_c5, len(best_h))

    # Return the best solution found
    return (best_h, best_c5, len(best_h))
