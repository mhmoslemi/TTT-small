```python
import numpy as np
import matplotlib.pyplot as plt
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize
import os
import sys
import random
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 45  # Start with the initial size
    max_budget = budget_s
    
    # Initial values
    h = initial_h_values
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_n_points = n_points
    
    # Try with smaller n_points
    for n_points in [45, 30, 20, 15, 10]:
        # Create a grid
        x = np.linspace(0, 2, n_points)
        dx = 2.0 / n_points
        
        # Define the initial guess
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else h
        
        # Define the objective
        def objective(h):
            # Compute the overlap integral for all shifts
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Define the constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        def constraint_fun(h):
            return constraint(h)
        
        # Define bounds: 0 <= h[i] <= 1
        bounds = Bounds(0, 1)
        
        # Linear constraint: sum(h) == n_points / 2
        cons = LinearConstraint(np.ones(n_points), n_points / 2, n_points / 2)
        
        # Optimization
        result = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            h_opt = result.x
            c5 = evaluate_erdos_solution(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                best_n_points = n_points
        
        # Print progress
        print(f"Checked n_points={n_points}, best C5={best_c5}")
    
    # Return the best result
    return best_h, best_c5, best_n_points
```
