import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as spla
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100
    h = initial_h_values[:n_points]
    
    # Convert to array
    h = np.array(h, dtype=np.float64)
    
    # Ensure sum of h is n_points / 2
    total = np.sum(h)
    scale = (n_points / 2) / total if total != 0 else 1.0
    h *= scale
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute all overlaps
    def evaluate_all_overlaps(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        overlaps = corr * dx
        return overlaps
    
    # Optimization strategy
    def optimize_h(h, max_iter=100, tol=1e-5):
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        
        for _ in range(max_iter):
            # Compute all overlaps
            overlaps = evaluate_all_overlaps(best_h)
            
            # Find the maximum overlap
            max_overlap = np.max(overlaps)
            
            # Compute gradient (approximate)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                delta = (evaluate_all_overlaps(h_plus) - evaluate_all_overlaps(h_minus)) / (2 * 1e-5)
                grad[i] = np.sum(delta)
            
            # Project gradient to the feasible set
            grad = np.clip(grad, -1e5, 1e5)
            
            # Update h with gradient descent
            h -= 0.1 * grad
            
            # Ensure h is within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Ensure sum is correct
            total = np.sum(h)
            if total != 0:
                h *= (n_points / 2) / total
            
            # Compute current c5
            current_c5 = compute_c5(h)
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
            
            if abs(best_c5 - current_c5) < tol:
                break
        
        return best_h, best_c5
    
    # Run optimization
    start_time = time.time()
    best_h, best_c5 = optimize_h(h)
    
    # Check if we can do better with more points
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h *= (n_points / 2) / total
        
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        
        for _ in range(50):
            # Compute overlaps
            overlaps = evaluate_all_overlaps(best_h)
            max_overlap = np.max(overlaps)
            
            # Find the maximum overlap
            max_index = np.argmax(overlaps)
            shift = max_index - n_points // 2
            if shift < 0:
                shift = 0
            if shift > n_points:
                shift = n_points
            
            # Adjust the shifted region
            shift_region = np.arange(max_index - shift, max_index + shift + 1)
            shift_region = np.clip(shift_region, 0, n_points)
            shift_region = np.unique(shift_region)
            
            # Modify the shifted region to reduce overlap
            for i in shift_region:
                if best_h[i] > 0.5:
                    best_h[i] -= 0.1
                elif best_h[i] < 0.5:
                    best_h[i] += 0.1
            
            # Ensure h is within [0, 1]
            best_h = np.clip(best_h, 0, 1)
            
            # Ensure sum is correct
            total = np.sum(best_h)
            if total != 0:
                best_h *= (n_points / 2) / total
            
            # Compute current c5
            current_c5 = compute_c5(best_h)
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
        
        if compute_c5(best_h) < best_c5:
            best_h = best_h
    
    # Final check
    if time.time() - start_time < budget_s - 10:
        n_points = 300
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h *= (n_points / 2) / total
        
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        
        for _ in range(50):
            # Compute overlaps
            overlaps = evaluate_all_overlaps(best_h)
            max_overlap = np.max(overlaps)
            
            # Find the maximum overlap
            max_index = np.argmax(overlaps)
            shift = max_index - n_points // 2
            if shift < 0:
                shift = 0
            if shift > n_points:
                shift = n_points
            
            # Adjust the shifted region
            shift_region = np.arange(max_index - shift, max_index + shift + 1)
            shift_region = np.clip(shift_region, 0, n_points)
            shift_region = np.unique(shift_region)
            
            # Modify the shifted region to reduce overlap
            for i in shift_region:
                if best_h[i] > 0.5:
                    best_h[i] -= 0.1
                elif best_h[i] < 0.5:
                    best_h[i] += 0.1
            
            # Ensure h is within [0, 1]
            best_h = np.clip(best_h, 0, 1)
            
            # Ensure sum is correct
            total = np.sum(best_h)
            if total != 0:
                best_h *= (n_points / 2) / total
            
            # Compute current c5
            current_c5 = compute_c5(best_h)
            
            # Update best solution
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    # Ensure we are within time budget
    if time.time() - start_time < budget_s - 10:
        h = best_h.copy()
        return h.tolist(), best_c5, len(h)
    else:
        return best_h.tolist(), best_c5, len(best_h)
