import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Initial construction
    initial_h_values = np.load("initial_h_values.npy")
    n_points_initial = len(initial_h_values)
    
    # We start with a smaller set of samples for faster optimization
    n_points = 100  # Reasonable size for fast evaluation
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # We start with a simple symmetric construction: flat in the middle, rising and falling
    for i in range(n_points):
        x = i * dx
        if 0.5 - 0.25 <= x <= 0.5 + 0.25:
            h[i] = 1.0
        elif 0.75 - 0.25 <= x <= 0.75 + 0.25:
            h[i] = 0.0
        else:
            h[i] = 0.5
    
    # Normalize to ensure the integral is 1
    h *= n_points / 2
    h = np.clip(h, 0, 1)
    
    def objective(h):
        # Compute the overlap integral for all k
        # This is equivalent to computing the cross-correlation of h and 1 - h
        # Use a sliding window to compute the integral for all possible k
        c5_values = []
        for k in range(-n_points + 1, n_points):
            # Ensure that the overlap is within [0,2]
            if k < 0:
                # Left shift
                overlap = h[:n_points + k] * (1 - h[k:n_points])
            elif k > 0:
                # Right shift
                overlap = h[k:n_points] * (1 - h[:n_points - k])
            else:
                # No shift
                overlap = h * (1 - h)
            c5_values.append(np.sum(overlap) * dx)
        return np.max(c5_values)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
    best_h = result.x
    
    # Make sure the values are within [0,1]
    best_h = np.clip(best_h, 0, 1)
    
    # Compute the final C5 value
    c5 = objective(best_h)
    
    # Return the best solution
    return (best_h, c5, n_points)
