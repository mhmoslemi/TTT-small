def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the convolution (correlation)
    result = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part and scale
    max_overlap = np.max(result * dx)
    return max_overlap

def initial_h_values(n_points):
    # Sample initial construction
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

    # Start optimization
    start_time = time.time()
    
    for _ in range(max_iter):

        current_c5 = np.max(result * dx)

            perturbation = 1e-6
