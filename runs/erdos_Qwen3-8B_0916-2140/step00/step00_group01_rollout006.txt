import numpy as np
import scipy.optimize as opt
import time
import random
from scipy.optimize import minimize
from scipy.integrate import trapz
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    
    # Initial guess from known construction (modified)
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h[i] = 0.0
        elif x < 1.0:
            initial_h[i] = 0.5
        elif x < 1.5:
            initial_h[i] = 0.75
        else:
            initial_h[i] = 1.0
    
    # Enforce integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Define function to compute C5
    def compute_C5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        shift = np.arange(-n_points+1, n_points)
        max_overlap = 0.0
        for k in shift:
            if k < -n_points + 1 or k > n_points - 1:
                continue
            shifted_h = np.roll(h, k)
            integrand = h * (1 - shifted_h)
            overlap = trapz(integrand, dx=dx)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Define loss function and constraints
    def objective(h):
        return compute_C5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Perform optimization
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], constraints=cons, options={'disp': False, 'maxiter': 100})
    
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    return best_h, best_C5, n_points
