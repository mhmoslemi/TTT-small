```python
import numpy as np
import scipy.optimize as opt
import time
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
from scipy.optimize import LinearConstraint, Bounds
from scipy.optimize import minimize

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    n_points = 128
    dx = 2.0 / n_points
    max_iter = 100
    tolerance = 1e-6

    # Initial guess
    h_initial = initial_h_values[:]
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)

    def objective(h):
        nonlocal n_points, dx
        h = np.clip(h, 0, 1)
        h = h.reshape(-1)
        h_fft = np.fft.fft(h)
        h_fft_shifted = np.fft.fftshift(h_fft)
        conv = np.fft.ifft(np.conj(h_fft_shifted) * h_fft_shifted).real
        conv = conv[n_points - 1 : 2 * n_points - 1]
        overlap = conv * dx
        c5 = np.max(overlap)
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    h_initial = np.clip(h_initial, 0, 1)

    # Optimization
    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': max_iter, 'ftol': tolerance, 'disp': False}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)

    # Evaluate the best solution
    c5 = evaluate_erdos_solution(best_h)
    return best_h, c5, n_points
```
