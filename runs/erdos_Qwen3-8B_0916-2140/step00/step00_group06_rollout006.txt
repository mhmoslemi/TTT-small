import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    
    # Initial guess from provided initial_h_values
    h = np.array(initial_h_values)
    h = h[:n_points]  # Truncate if necessary
    if len(h) < n_points:
        h = np.pad(h, (0, n_points - len(h)), mode='constant', constant_values=0.5)
    if len(h) > n_points:
        h = h[:n_points]
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        dx = 2.0 / len(h_values)
        h = h_values
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find maximum value
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to optimize
    def optimize_h(h_initial):
        # Prepare initial guess
        h = h_initial.copy()
        # Set up optimization problem
        def objective(x):
            h = x.copy()
            # Ensure h is within [0, 1]
            h = np.clip(h, 0, 1)
            # Compute overlap
            return compute_c5(h)
        
        def constraint(x):
            h = x.copy()
            # Ensure integral constraint
            return np.sum(h) - n_points / 2
        
        # Use L-BFGS-B to optimize
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        
        # Return the best solution found
        return result.x
    
    # Run optimization
    start_time = time.time()
    best_h = h
    best_c5 = compute_c5(h)
    
    # Run optimization with initial guess
    optimized_h = optimize_h(h)
    best_c5 = compute_c5(optimized_h)
    best_h = optimized_h
    
    # Refine further with more iterations
    for _ in range(3):
        optimized_h = optimize_h(best_h)
        current_c5 = compute_c5(optimized_h)
        if current_c5 < best_c5:
            best_h = optimized_h
            best_c5 = current_c5
        if time.time() - start_time > budget_s - 10:
            break
    
    # Ensure the solution is within the [0, 1] range
    best_h = np.clip(best_h, 0, 1)
    
    # Check the integral constraint
    integral = np.sum(best_h) * (2.0 / n_points)
    if abs(integral - 1.0) > 1e-6:
        best_h = np.clip(best_h, 0, 1)
    
    return best_h, best_c5, n_points
