import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    def objective(h_values):
        h = h_values.copy()
        h = h / np.sum(h) * (n_points / 2)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def evaluate(h):
        h = h.copy()
        h = h / np.sum(h) * (n_points / 2)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    def optimize_solution(h_initial):
        bounds = Bounds(0, 1)
        result = optimize.minimize(
            evaluate,
            h_initial,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        return result.x
    
    h_opt = optimize_solution(h)
    c5 = evaluate(h_opt)
    
    # Reduce to 100 points
    h_values = h_opt[:n_points]
    
    return h_values, c5, n_points
