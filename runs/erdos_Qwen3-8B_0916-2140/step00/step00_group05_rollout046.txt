import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
import copy

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial guess from provided initial_h_values
    initial_h_values = kwargs.get('initial_h_values', np.random.rand(70) * 2 - 1)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    n_points = len(initial_h_values)

    # We'll use a smaller n_points for faster optimization
    n_points = 200  # Reasonable balance between speed and accuracy
    dx = 2.0 / n_points
    h_values = np.random.rand(n_points) * 2 - 1
    h_values = np.clip(h_values, 0, 1)
    total = np.sum(h_values) * dx
    scale = 1.0 / total
    h_values *= scale

    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h_fft = np.fft.fft(h)
        one_minus_h = 1 - h
        one_minus_h_fft = np.fft.fft(one_minus_h)
        overlap_fft = h_fft * np.conj(one_minus_h_fft)
        overlap = np.fft.ifft(overlap_fft).real
        c5 = np.max(overlap) * dx
        return c5

    def objective(x):
        return evaluate_c5(x)

    def constraints(x):
        return np.sum(x) - n_points / 2

    def bounds(x):
        return np.clip(x, 0, 1)

    result = minimize(objective, h_values, method='L-BFGS-B', bounds=Bounds(0, 1), tol=1e-10, options={'maxiters': 1000})
    h_opt = result.x

    # If necessary, perform a local search from the best result
    for _ in range(5):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h_opt + perturb, 0, 1)
        total_perturb = np.sum(h_perturb) * dx
        scale_perturb = 1.0 / total_perturb
        h_perturb *= scale_perturb
        c5_perturb = evaluate_c5(h_perturb)
        if c5_perturb < evaluate_c5(h_opt):
            h_opt = h_perturb

    best_c5 = evaluate_c5(h_opt)
    return h_opt, best_c5, n_points
