import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial point
    n_points = 45
    h = initial_h_values
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    
    # Precompute the shifted functions
    def shift_function(h, k):
        return np.roll(h, k)
    
    # Precompute all shifted (1 - h) functions
    shifted_ones_minus_h = []
    for k in range(1, n_points + 1):
        shifted = shift_function(h, k)
        shifted_ones_minus_h.append(1 - shifted)
    
    # Function to compute the overlap integral for a given h
    def compute_C5(h):
        dx = 2.0 / len(h)
        max_overlap = 0.0
        for k in range(1, len(h) + 1):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Gradient descent with momentum
    def gradient_descent(num_iterations=1000, learning_rate=0.01, momentum=0.9, tolerance=1e-4):
        h = np.copy(initial_h_values)
        best_h = np.copy(h)
        best_C5 = compute_C5(h)
        
        for iteration in range(num_iterations):
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = np.copy(h)
            
            # Compute the gradient (simplified version)
            grad = np.zeros_like(h)
            for k in range(1, len(h) + 1):
                shifted = np.roll(h, k)
                overlap = np.sum(h * (1 - shifted))
                grad -= np.sum((1 - shifted) * np.roll(h, -k)) * (2.0 / len(h))
            
            # Update h using gradient descent with momentum
            grad_norm = np.linalg.norm(grad)
            if grad_norm > tolerance:
                grad /= grad_norm
                h -= learning_rate * grad
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0.0, 1.0)
        
        return best_h, best_C5
    
    # Perform a local search
    best_h, best_C5 = gradient_descent(num_iterations=1000, learning_rate=0.01, momentum=0.9)
    
    # Try a few random perturbations to escape local optima
    for _ in range(5):
        perturbation = np.random.normal(0, 0.05, len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        current_C5 = compute_C5(perturbed_h)
        if current_C5 < best_C5:
            best_h = np.copy(perturbed_h)
            best_C5 = current_C5
    
    # Ensure the total integral is exactly 1
    total_integral = np.sum(best_h) * (2.0 / len(best_h))
    if total_integral != 1.0:
        best_h = np.copy(initial_h_values)
        best_C5 = compute_C5(best_h)
    
    # Return the result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (initial_h_values, compute_C5(initial_h_values), n_points)
    else:
        return (best_h, best_C5, n_points)
