import scipy.interpolate as interpolate

def run(seed=42, budget_s=600):
    # Predefined initial construction
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Helper: Compute C5
    def compute_c5(h):
        h = np.array(h)
        shifted = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - shifted, mode='full')
        return np.max(overlap * dx)

    # Helper: Compute all overlaps for a given h
    def compute_all_overlaps(h):
        h = np.array(h)
        overlaps = []
        for k in range(1, len(h) - 1):
            shifted = np.roll(h, shift=k)
            overlap = np.correlate(h, 1 - shifted, mode='full')
            overlaps.append(overlap * dx)
        return overlaps

    # Function to optimize: minimize max overlap
    def objective(h):
        overlaps = compute_all_overlaps(h)
        return np.max(overlaps)

    # Optimize starting from initial_h
    h0 = initial_h.copy()
    result = optimize.minimize(
        objective,
        h0,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    # Check if optimization succeeded
    if result.success:
        h_opt = result.x
        c5 = compute_c5(h_opt)
        return (h_opt, c5, n_points)
    else:
        # Fallback to initial_h if optimization fails
        c5 = compute_c5(initial_h)
        return (initial_h, c5, len(initial_h))
