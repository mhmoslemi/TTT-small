```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import partial
import time

# Pre-imported functions
initial_h_values = np.load('initial_h.npy')  # Example initial construction

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h_initial = initial_h_values
    n_points = len(h_initial)
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    assert np.isclose(sum_h * dx, 1.0), "Initial h does not satisfy the integral constraint"
    
    # Objective function
    def objective(h):
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, h[:n_points//2]])  # Duplicate to handle circular shifts
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = []
        for k in shifts:
            if k < 0:
                shifted_h = h[-k:]
            else:
                shifted_h = h[:k]
            overlap = np.sum(h[:n_points] * (1 - shifted_h[:n_points])) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    # Bounds for h: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # If optimization didn't improve, fallback to initial
    if result.success:
        h_opt = result.x
    else:
        h_opt = h_initial
    
    # Final evaluation
    h_opt = np.array(h_opt, dtype=np.float64)
    h_opt = np.concatenate([h_opt, h_opt[:n_points//2]])  # Duplicate to handle shifts
    shifts = np.arange(-n_points + 1, n_points)
    overlaps = []
    for k in shifts:
        if k < 0:
            shifted_h = h_opt[-k:]
        else:
            shifted_h = h_opt[:k]
        overlap = np.sum(h_opt[:n_points] * (1 - shifted_h[:n_points])) * dx
        overlaps.append(overlap)
    c5_bound = max(overlaps)
    
    # Return best solution
    return h_opt[:n_points], c5_bound, n_points
```
