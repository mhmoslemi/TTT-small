import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

def initial_h_values():
    return np.array([0.5]*70)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Objective function: minimize the maximum overlap
    def objective(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Constraint: sum of h must be n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h.tolist()
    
    # Optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-6, "maxiter": 1000}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    return best_h, best_c5, len(best_h)
