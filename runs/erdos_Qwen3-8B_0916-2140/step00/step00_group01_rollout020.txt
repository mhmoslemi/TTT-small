```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.optimize import Bounds, minimize, minimize_scalar

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the correlation (sliding window)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral is max over k of (integral of h(x)(1 - h(x + k)) dx)
    # We take the maximum value and multiply by dx
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    return np.load('initial_h.npy').tolist()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Start with a moderate number of samples
    initial_h = initial_h_values()
    initial_h = initial_h[:n_points]
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h_initial = np.array(initial_h)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Bounds
    bounds = Bounds(0, 1)
    
    # Optimization
    result = minimize(objective, h_initial, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
    
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we can reduce n_points further
    if time.time() - start_time < budget_s - 10:
        n_points = 100
        h_initial = np.array([0.5] * n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        def objective_small(h_flat):
            h = h_flat.reshape(n_points)
            c5 = evaluate_erdos_solution(h)
            return c5
        
        result_small = minimize(objective_small, h_initial, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
        
        best_h = result_small.x
        best_c5 = result_small.fun
    
    elapsed_time = time.time() - start_time
    print(f"Completed in {elapsed_time:.2f} seconds")
    
    return best_h.tolist(), best_c5, n_points
```
