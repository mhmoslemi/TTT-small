    initial_h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(initial_h)), initial_h)

    initial_h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(initial_h)), initial_h)

    def objective(h):
        h = np.array(h).reshape(n_points)
        h = np.clip(h, 0, 1)
        h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(h)), h)
        h = h[:n_points]
        shift = np.arange(1, n_points)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    def perturb(h, scale=0.1):
        h = np.array(h)
        h += np.random.normal(0, scale, h.shape)
        h = np.clip(h, 0, 1)
        return h

    def evaluate(h):
        return objective(h)

    def de_optimizer(h0, max_iter=100, scale=0.1, perturb_scale=0.1):
        h = h0.copy()
        best_h = h.copy()
        best_c5 = evaluate(h)
        time_start = time.time()
        for _ in range(max_iter):
            if time.time() - time_start > budget_s:
                break
            h = perturb(h, scale)
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        return best_h, best_c5

    # Initial guess
    h_init = initial_h

    # Run differential evolution
    best_h, best_c5 = de_optimizer(h_init, max_iter=100, scale=0.1, perturb_scale=0.1)

    # Further refinement using gradient descent
    def loss(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.interp(np.linspace(0, 2, n_points), np.sort(np.unique(h)), h)
        h = h[:n_points]
        shift = np.arange(1, n_points)
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    h = best_h.copy()
    for _ in range(50):
        if time.time() - time_start > budget_s:
            break
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            grad[i] = (loss(h_plus) - loss(h_minus)) / (2 * 1e-5)
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)

    best_c5 = loss(h)
    best_h = h

    return best_h, best_c5, n_points
