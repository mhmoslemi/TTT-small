```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from functools import partial
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assuming the initial construction is available as a numpy array

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    max_time = budget_s
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values
    n_points = len(best_h)
    
    # Function to evaluate the overlap integral
    def evaluate(h):
        nonlocal best_c5
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Optimization function: minimize the maximum overlap integral
    def objective(h_flat):
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_flat)), h_flat)
        return evaluate(h)
    
    # Initial guess: Use the initial construction
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    # Define bounds: each h[i] must be between 0 and 1
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use a simple optimization method with early stopping
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'disp': False},
        tol=1e-6
    )
    
    # Check if the result is better than the initial guess
    if result.success:
        h_result = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(result.x)), result.x)
        c5_result = evaluate(h_result)
        if c5_result < best_c5:
            best_c5 = c5_result
            best_h = h_result
    
    # If no better solution found, return the initial guess
    if best_c5 == float('inf'):
        best_c5 = evaluate(initial_h_values)
    
    # Ensure the best_h is of length n_points
    best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    
    print(f"Best C5: {best_c5}")
    print(f"Best h shape: {len(best_h)}")
    return (best_h, best_c5, len(best_h))
```
