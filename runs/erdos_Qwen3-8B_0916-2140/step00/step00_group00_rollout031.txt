def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points = 45  # Initial number of points
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    h = h[:n_points]
    
    # Initial objective: compute C_5
    def compute_C5(h):
        h = np.array(h)
        x = np.linspace(0, 2, n_points)
        shifted = np.zeros_like(h)
        for k in range(n_points):
            shifted[k:] = h[:n_points - k]
        shifted[:k] = 0
        overlap = np.sum(h * (1 - shifted)) * dx
        return np.max(overlap)
    
    # Objective function to minimize
    def objective(h):
        return compute_C5(h)
    
    # Equality constraint: sum(h) = n_points / 2
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Gradient of the objective function
    def objective_grad(h):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_i = h[i]
            for k in range(len(h)):
                if k <= i:
                    overlap_k = np.sum(h * (1 - np.roll(h, k))) * dx
                    grad[i] += (1 - np.roll(h, k)[i]) * dx
        return grad
    
    # Gradient of the constraint
    def constraint_grad(h):
        return np.ones_like(h)
    
    # Use the initial point
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Use Nelder-Mead for optimization
    result = optimize.minimize(objective, h, method='Nelder-Mead', bounds=[(0, 1) for _ in range(n_points)], constraints=cons)

    # Evaluate the solution for the given C_5
    evaluated_C5 = evaluate_erdos_solution(best_h)
