import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 80
    h_initial = initial_h_values
    dx = 2.0 / n_points
    max_iterations = 100
    step_size = 0.01
    best_c5 = float('inf')
    
    def evaluate(h):
        nonlocal best_c5
        h = np.array(h, dtype=np.float64)
        h = np.concatenate([h, h])  # Mirror to handle full convolution
        shifted = np.zeros_like(h)
        for k in range(n_points):
            shifted[k] = 1 - h[k]
        corr = np.correlate(h, shifted, mode="full")
        c5 = np.max(corr * dx)
        if c5 < best_c5:
            best_c5 = c5
        return c5
    
    # Optimization
    start_time = time.time()
    
    # Initial evaluation
    h = h_initial.copy()
    current_c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simple gradient ascent with line search
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current c5
        current_c5 = evaluate(h)
        
        # Compute gradient approximation
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += step_size
            c5_plus = evaluate(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= step_size
            c5_minus = evaluate(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * step_size)
        
        # Update h using gradient ascent
        h = np.clip(h + grad * step_size, 0, 1)
        
        # Ensure integral constraint
        h_sum = np.sum(h)
        if h_sum > n_points / 2:
            h = np.clip(h, 0, 1)
            h = h - (h_sum - n_points / 2) / n_points
        elif h_sum < n_points / 2:
            h = np.clip(h, 0, 1)
            h = h + (n_points / 2 - h_sum) / n_points
        
        # Update best solution
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
