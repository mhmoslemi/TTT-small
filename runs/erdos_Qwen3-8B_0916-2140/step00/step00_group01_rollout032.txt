```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Define initial parameters
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Define function to compute C5
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        h_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(h_corr) * dx
        return c5
    
    # Define function to compute the bound for a given h
    def objective_function(h):
        return compute_c5(h)
    
    # Define function to compute the constraint
    def constraint_function(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    
    # Start with the initial guess
    h_current = initial_h
    
    # Best solution so far
    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()
    
    # Optimization loop
    for _ in range(max_iter):
        start_time = time.time()
        
        # Use BFGS with bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            objective_function,
            h_current,
            bounds=bounds,
            method='L-BFGS-B',
            options={'gtol': tolerance, 'maxfun': 1000}
        )
        
        if result.success:
            h_current = result.x
            current_c5 = compute_c5(h_current)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()
        
        # Check remaining time
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
