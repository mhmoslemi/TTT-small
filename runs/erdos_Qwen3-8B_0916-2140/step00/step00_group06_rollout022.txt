import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial configuration from the pre-imported global variable
    initial_h_values = np.load('initial_h_values.npy')  # Mocked as per instructions
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.all(h >= 0) and np.all(h <= 1)
    
    # Optimization method
    def optimize_h(h0):
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h0,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x
    
    # Run optimization
    start_time = time.time()
    h_opt = optimize_h(initial_h_values)
    elapsed_time = time.time() - start_time
    
    # Calculate the final C5 value
    c5 = objective(h_opt)
    
    return h_opt, c5, len(h_opt)
