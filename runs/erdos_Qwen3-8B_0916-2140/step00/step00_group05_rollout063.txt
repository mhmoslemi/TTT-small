```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all overlaps
    overlaps = []
    for k in range(1, n_points):
        shifted = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted)) * dx
        overlaps.append(overlap)
    return np.max(overlaps)

def initial_h_values():
    # Initial construction with 70 samples
    return np.load('initial_h.npy')

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial sample with 70 points
    n_points_initial = 70
    h_initial = initial_h_values()
    best_c5 = evaluate_erdos_solution(h_initial, n_points_initial)
    best_h = h_initial.copy()
    best_n_points = n_points_initial
    
    # Try smaller n_points
    for n_points in [30, 50, 60, 70, 80, 90, 100, 110, 120, 130, 140, 150]:
        if n_points <= 10:
            continue
        
        # Generate random h with sum(h) = n_points / 2
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
    
    # Now perform optimization on the best n_points
    n_points = best_n_points
    h = best_h.copy()
    
    # Refine using optimization
    # Objective: minimize max overlap
    # Constraints: sum(h) = n_points / 2, 0 <= h[i] <= 1
    
    # Create variable
    x = cp.Variable(n_points)
    # Objective: minimize the maximum overlap
    # To handle max, we can use an auxiliary variable and constraints
    
    # Define the objective
    max_overlap = cp.Variable()
    objective = cp.Minimize(max_overlap)
    constraints = []
    
    # Add constraints for each k
    for k in range(1, n_points):
        shifted = cp.reshape(cp.roll(x, k), (n_points, 1))
        overlap = cp.sum(x * (1 - shifted))
        constraints.append(overlap <= max_overlap)
    
    # Add constraint for sum(h) = n_points / 2
    constraints.append(cp.sum(x) == n_points / 2)
    # Add bounds
    constraints.append(x >= 0)
    constraints.append(x <= 1)
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver='SCS', eps=1e-4, verbose=False)
        if problem.status == 'optimal':
            h = x.value
            best_c5 = evaluate_erdos_solution(h, n_points)
            best_h = h.copy()
            best_n_points = n_points
    except:
        pass
    
    # Make sure the values are within [0, 1]
    h = np.clip(best_h, 0, 1)
    # Ensure sum is correct
    h = h / np.sum(h) * (best_n_points / 2)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(h, best_n_points)
    
    # Ensure we don't exceed the budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (h.tolist(), final_c5, best_n_points)
    
    return (h.tolist(), final_c5, best_n_points)
```
