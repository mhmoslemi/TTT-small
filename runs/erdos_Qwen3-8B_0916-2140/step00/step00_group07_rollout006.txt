import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    total = np.sum(h) * (2.0 / n_points)
    h = h * (1.0 / total) * (2.0 / n_points)
    h = np.clip(h, 0, 1)
    
    def compute_c5(h, n_points):
        dx = 2.0 / n_points
        h_reshaped = np.concatenate([h, h])
        shifted = np.zeros((n_points, n_points), dtype=np.float32)
        for i in range(n_points):
            shifted[i, :] = h_reshaped[i:i + n_points]
        overlap = np.sum(shifted * (1 - shifted), axis=1)
        overlap = np.max(overlap) * dx
        return overlap
    
    def objective(h_values):
        return compute_c5(h_values, n_points)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    def objective_gradient(h_values):
        grad = np.zeros_like(h_values)
        dx = 2.0 / n_points
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    val = h_values[i] * (1 - h_values[i + k]) * dx
                    grad[i] += val
                    grad[i + k] -= val * dx
        return grad
    
    def constraint_sum_gradient(h_values):
        return np.ones_like(h_values)
    
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        jac=objective_gradient,
        constraints=[{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_gradient}],
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h, n_points)
    return (best_h, best_c5, n_points)
