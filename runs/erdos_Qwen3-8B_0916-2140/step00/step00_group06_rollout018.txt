import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.integrate import simps

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Define parameters
    max_n_points = 1000  # Maximum number of points to consider
    n_points = 100  # Starting with 100 samples
    target_c5 = 0.380876
    
    # Initial guess: use the provided initial_h_values
    h_initial = initial_h_values[:n_points]
    h_initial = h_initial[:n_points]  # Ensure it's of length n_points
    total_integral = np.sum(h_initial) * (2.0 / n_points)
    
    # Define function to evaluate C5
    def evaluate_objective(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        total_integral = np.sum(h) * (2.0 / len(h))
        if not np.isclose(total_integral, 1.0):
            return np.inf  # Penalize deviation from the integral constraint
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    # Define bounds for each sample
    bounds = Bounds(0, 1)
    
    # Optimization
    result = minimize(
        evaluate_objective,
        x0=h_initial,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Evaluate the best solution found
    best_h = result.x
    best_c5 = evaluate_objective(best_h)
    
    # Check if we can try a larger n_points for better resolution
    if time.time() - start_time < budget_s * 0.5:
        n_points = 200
        best_h = np.clip(best_h, 0, 1)
        h_initial = best_h[:n_points]
        h_initial = np.random.permutation(h_initial)
        
        result = minimize(
            evaluate_objective,
            x0=h_initial,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        best_h = result.x
        best_c5 = evaluate_objective(best_h)
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
