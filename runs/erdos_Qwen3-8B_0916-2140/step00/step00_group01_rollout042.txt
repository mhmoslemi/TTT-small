import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000  # Starting with 1000 points for sufficient resolution
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    initial_h_values[::2] = 0.0
    initial_h_values[1::2] = 1.0
    h = initial_h_values.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    best_n = n_points
    
    def compute_c5(h):
        h = h.reshape(-1)
        h = np.concatenate((h, h))  # Double the array to handle wrap-around
        overlap = np.correlate(h, 1 - h[:n_points], mode='full')
        return np.max(overlap * dx)
    
    def objective(x):
        return compute_c5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Perform a grid search with local optimization
    for _ in range(10):
        start_time = time.time()
        # Initial guess
        h = initial_h_values.copy()
        # Optimization
        res = minimize(objective, h, method='L-BFGS-B', bounds=(0, 1), options={'maxiter': 100})
        c5 = res.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = res.x.copy()
            best_n = n_points
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s * 0.8:
            break
    
    # Try reducing n_points for speed
    for n in [500, 300, 200, 150, 100, 50]:
        dx = 2.0 / n
        h = np.random.rand(n) * 2 - 1  # Initial guess
        h = (h + 1) / 2  # Normalize to [0, 1]
        h = h * (n / 2)  # Ensure sum is n / 2
        h = h / np.sum(h)
        h = h * 2  # Map to [0, 2]
        h = h / 2  # Normalize to [0, 1]
        h = h[:n]
        # Optimization
        res = minimize(objective, h, method='L-BFGS-B', bounds=(0, 1), options={'maxiter': 100})
        c5 = res.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = res.x.copy()
            best_n = n
    
    return best_h, best_c5, best_n
