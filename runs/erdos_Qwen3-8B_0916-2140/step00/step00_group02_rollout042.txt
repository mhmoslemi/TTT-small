import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):

    initial_h = initial_h_values
    
    # Parameters
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_c5 = 0.380876  # Target to beat
    
    # Define the function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        # Compute all shifted versions of h
        c5_values = []
        for k in range(1, n_points - 1):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            c5_values.append(overlap)
        return max(c5_values)
    
    # Define an objective to minimize the maximum overlap
    def objective(x):
        return compute_c5(x)
    
    # Define bounds for the optimization
    bounds = Bounds(0, 1)
    
    # Define constraints: sum(h) == n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_sum_func(x):
        return constraint_sum(x)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=optimize.LinearConstraint(
            np.ones(n_points),
            lb=0,
            ub=n_points / 2
        ),
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Ensure we stay within budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        return (best_h, best_c5, len(best_h))
    
    # Try a few more iterations with different starting points
    for _ in range(3):
        # Add small random noise to the current solution
        perturbation = np.random.uniform(-0.05, 0.05, size=len(best_h))
        new_h = np.clip(best_h + perturbation, 0, 1)
        
        # Check if the sum constraint is satisfied
        if np.abs(np.sum(new_h) - n_points / 2) > 1e-6:
            new_h = new_h - (np.sum(new_h) - n_points / 2) / len(new_h)
        
        # Optimize from the new starting point
        result = optimize.minimize(
            objective,
            new_h,
            bounds=bounds,
            constraints=optimize.LinearConstraint(
                np.ones(len(new_h)),
                lb=0,
                ub=len(new_h) / 2
            ),
            method='SLSQP',

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
