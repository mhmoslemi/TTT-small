```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 400  # Starting with 400 points for balance between speed and accuracy
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Initialize with a piecewise constant function with 4 segments
    for i in range(n_points):
        if i < n_points / 4:
            initial_h_values[i] = 0.0
        elif i < n_points / 2:
            initial_h_values[i] = 0.5
        elif i < 3 * n_points / 4:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.5
    
    # Optimize the function
    def objective(h):
        h = h.reshape(n_points)
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use L-BFGS-B optimizer
    result = optimize.minimize(
        fun=objective,
        x0=initial_h_values,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Print and return results
    print(f"Best C₅ bound: {best_c5}")
    return best_h, best_c5, n_points
```
