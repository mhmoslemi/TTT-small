def compute_overlap_integral(h, dx):
    n = len(h)
    overlap = np.correlate(h, 1.0 - h, mode="full")
    # Only consider positive lags (since negative lags are symmetric)
    max_overlap = np.max(overlap[n//2:])
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):

    n_points = n_points_initial
