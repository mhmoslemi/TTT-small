```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.integrate import simps
from scipy.special import gamma, comb
from scipy.interpolate import interp1d
import time

# Import pre-defined initial h_values
initial_h_values = np.load("initial_h.npy")

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points), h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided h_values
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    
    # Define objective function to minimize
    def objective(x, n_points):
        dx = 2.0 / n_points
        h = x
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol}
    )
    
    # Extract the best solution found
    best_h = result.x
    c5_bound = result.fun
    
    # Check if the solution is valid (sum(h) * dx = 1)
    dx = 2.0 / n_points
    sum_h = np.sum(best_h) * dx
    if abs(sum_h - 1) > 1e-8:
        print("Warning: Total integral does not match 1")
    
    return best_h, c5_bound, n_points
```
