import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points to balance speed and accuracy
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Initial guess
    h_initial = initial_h.copy()

    # Optimization with constraints
    h = cp.Variable(n_points)
    obj = cp.Minimize(objective(h))
    cons = [cp.sum(h) == n_points / 2]
    prob = cp.Problem(obj, cons)
    
    start_time = time.time()
    try:
        result = prob.solve(solver='SCS', eps=1e-5, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        if result is None:
            raise RuntimeError("Optimization failed")
        h_values = h.value
        c5_bound = objective(h_values)
    except:
        h_values = h_initial
        c5_bound = objective(h_initial)
    
    # Ensure h_values are within [0, 1]
    h_values = np.clip(h_values, 0, 1)
    
    return h_values, c5_bound, n_points
