```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100
    target_c5 = 0.380876
    max_iterations = 100
    tol = 1e-4
    best_c5 = np.inf
    best_h = None
    
    # Initial guess
    h = np.copy(initial_h_values[:n_points])
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_evals = 1000
    evals_counter = 0
    
    # Define objective with constraints
    def objective(x):
        nonlocal evals_counter
        if evals_counter >= max_evals:
            return best_c5
        evals_counter += 1
        
        h = x.copy()
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if np.abs(h_sum - n_points / 2) > 1e-6:
            return np.inf
        
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use L-BFGS-B for optimization
    result = opt.minimize(
        objective,
        h,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    # Check if result is better
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Perform a few iterations of local search
    for _ in range(5):
        # Generate small perturbations
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbation = np.clip(perturbation, -0.5, 0.5)
        new_h = np.clip(best_h + perturbation, 0, 1)
        
        # Enforce integral constraint
        total = np.sum(new_h)
        scale = (n_points / 2) / total if total != 0 else 1.0
        new_h = np.clip(new_h * scale, 0, 1)
        
        # Evaluate
        c5 = evaluate_erdos_solution(new_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h
    
    return best_h, best_c5, n_points
```
