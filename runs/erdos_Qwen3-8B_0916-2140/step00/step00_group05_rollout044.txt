import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.stats import beta
from scipy.sparse import diags
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000  # Start with 1000 points for accuracy
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)

    def compute_C5(h):
        h = np.concatenate([h, h[:n_points//2]])
        shifts = np.arange(-n_points//2, n_points//2 + 1)
        max_overlap = 0
        for k in shifts:
            if k == 0:
                continue
            shifted = np.roll(h, k)
            overlap = np.sum(h[:n_points] * (1 - shifted[:n_points]))
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx

    def objective(h_values):
        return compute_C5(h_values)

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess from provided initial_h_values
    if 'initial_h_values' in kwargs:
        h = kwargs['initial_h_values']
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h * (n_points / 2) / np.sum(h)

    # Optimize
    result = minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )

    if result.success:
        best_h = result.x
    else:
        best_h = h

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    c5 = compute_C5(best_h)
    return best_h, c5, n_points
