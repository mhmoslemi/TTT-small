import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initial guess based on current construction
    if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n_points:
        h = np.array(initial_h_values)
    else:
        # Default: uniform distribution
        h = np.full(n_points, 1.0 / n_points) * n_points / 2

    # Ensure the sum is exactly n_points / 2
    h = h / np.sum(h) * (n_points / 2)

    # Define the overlap integral
    def compute_c5(h):
        h = np.concatenate([h, h])  # To handle wrap-around
        h_shifted = np.roll(h, shift=1)
        overlap = np.sum(h * (1 - h_shifted))
        return overlap * dx

    # Define objective function to minimize max overlap
    def objective(h):
        max_overlap = 0.0
        for k in range(1, n_points):
            h_shifted = np.roll(h, shift=k)
            overlap = np.sum(h * (1 - h_shifted))
            max_overlap = max(max_overlap, overlap * dx)
        return max_overlap

    # Optimization method: minimize the maximum overlap
    def optimize_h(h0):
        result = optimize.minimize(
            fun=objective,
            x0=h0,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        return result.x

    # Run optimization
    start_time = time.time()
    best_h = h
    best_c5 = compute_c5(h)

    # Try multiple starting points
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        h = np.random.uniform(0.0, 1.0, size=n_points)
        h = h / np.sum(h) * (n_points / 2)
        h_opt = optimize_h(h)
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
