import numpy as np
import scipy.optimize as optimize
import copy
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 1000
    max_time = budget_s
    start_time = time.time()
    
    # Initial guess from provided construction
    initial_h = copy.deepcopy(initial_h_values)
    initial_h = np.array(initial_h)
    
    # Discretize the function
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=True)
    
    def evaluate_c5(h_candidate):
        h = np.array(h_candidate, dtype=np.float64)
        if len(h) != n_points:
            raise ValueError("h must have length n_points")
        # Compute the overlap integral
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(np.abs(cross_corr)) * dx
        return c5
    
    def objective(h_candidate):
        return evaluate_c5(h_candidate)
    
    def constraint_sum(h_candidate):
        return np.sum(h_candidate) - n_points / 2
    
    # Define bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial solution
    h_initial = initial_h
    
    # Define the optimization problem
    def optimize_solution():
        result = optimize.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
        return result.x
    
    # Run the optimization
    best_h = optimize_solution()
    best_c5 = evaluate_c5(best_h)
    
    # Refine the solution with a local optimizer
    for _ in range(5):
        result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}])
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
    
    # Final evaluation
    final_c5 = best_c5
    return best_h, final_c5, n_points
