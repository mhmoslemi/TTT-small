```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Define objective function and constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    # Constraint: sum(h) == n_points / 2
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Optimization using SLSQP
    result = optimize.minimize(
        fun=objective,
        x0=initial_h,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Check if we've exceeded the time budget
    if time.time() - start_time > budget_s - 10:
        return (initial_h, objective(initial_h), n_points)
    
    # Evaluate the result
    best_h = result.x
    best_c5 = objective(best_h)
    best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
```
