import numpy as np
import scipy.optimize as optimize
import time
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    h_initial = initial_h_values
    n_points = len(h_initial)
    
    # If initial construction is not provided, we can generate a random one
    if h_initial is None:
        n_points = 100
        h_initial = np.random.rand(n_points)
        h_initial = (h_initial - h_initial.min()) / (h_initial.max() - h_initial.min())
    
    # Compute initial C5 value
    dx = 2.0 / n_points
    h_initial = np.clip(h_initial, 0, 1)
    sum_h = np.sum(h_initial)
    if abs(sum_h - n_points / 2) > 1e-12:
        h_initial = (h_initial * (n_points / 2)) / sum_h
    
    # Compute initial C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - len(h) / 2) > 1e-12:
            h = (h * (len(h) / 2)) / sum_h
        n = len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    initial_c5 = compute_c5(h_initial)
    best_c5 = initial_c5
    best_h = h_initial.copy()
    best_n_points = n_points
    
    # Define the objective function for optimization
    def objective(x, n_points):
        dx = 2.0 / n_points
        h = x
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-12:
            h = (h * (n_points / 2)) / sum_h
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Define the constraint function
    def constraint(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Define the bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define the optimization parameters
    max_iterations = 100
    tolerance = 1e-6
    time_limit = budget_s
    
    # Start the optimization
    start_time = time.time()
    for _ in range(10):
        if time.time() - start_time > time_limit:
            break
        
        # Define the optimization problem
        x0 = best_h.copy()
        res = optimize.minimize(objective, x0, args=(n_points,),
                               bounds=bounds,
                               method='L-BFGS-B',
                               jac=None,
                               options={'maxiter': 100, 'ftol': 1e-8})
        
        if res.success:
            best_h = res.x
            best_c5 = objective(best_h, n_points)
            best_n_points = n_points
            print(f"New best C5: {best_c5}")
            
            # Try to reduce n_points
            if n_points > 10:
                n_points = int(n_points / 2)
                best_h = (best_h * n_points) / len(best_h)
                best_h = np.clip(best_h, 0, 1)
                best_h = (best_h * (n_points / 2)) / np.sum(best_h)
                best_h = np.clip(best_h, 0, 1)
                
        else:
            print("Optimization failed.")
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    print(f"Final C5: {final_c5}")
    
    return (best_h, final_c5, best_n_points)
