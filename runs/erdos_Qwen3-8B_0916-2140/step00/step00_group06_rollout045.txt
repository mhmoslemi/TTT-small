import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    # Set random seed for reproducibility
    np.random.seed(seed)

    # Initial point
    n_points = 100  # Start with 100 points for faster initial computation
    h = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values
    h = np.array(h, dtype=np.float64)

    # Ensure the integral constraint is satisfied
    integral = np.sum(h) * (2.0 / n_points)
    if integral != 1.0:
        adjustment = (1.0 - integral) / n_points
        h += adjustment
        h = np.clip(h, 0.0, 1.0)

    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_shifted = np.concatenate((h[-1:], h[:-1]))  # Shifted version for correlation
        correlation = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    def fitness(x):
        h = x.reshape(-1)
        h = np.clip(h, 0.0, 1.0)
        c5 = compute_c5(h)
        return c5

    def objective(x):
        return fitness(x)

    def constraint(x):
        return np.sum(x) - (n_points / 2)

    def bounds(x):
        return np.array([0.0, 1.0])

    # Optimization using L-BFGS-B with initial guess
    initial_guess = h.copy()
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B',
        options={'gtol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    # If the current solution is not good enough, refine with more points
    if best_c5 > 0.380876:
        n_points = 200
        best_h = np.linspace(0, 2, n_points)
        best_h = np.random.rand(n_points)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = (best_h - np.min(best_h)) / (np.max(best_h) - np.min(best_h))  # Normalize to [0, 1]
        best_h = np.clip(best_h, 0.0, 1.0)
        integral = np.sum(best_h) * (2.0 / n_points)
        adjustment = (1.0 - integral) / n_points
        best_h += adjustment
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    # Final check to ensure the integral constraint is satisfied
    integral = np.sum(best_h) * (2.0 / n_points)
    if np.abs(integral - 1.0) > 1e-6:
        adjustment = (1.0 - integral) / n_points
        best_h += adjustment
        best_h = np.clip(best_h, 0.0, 1.0)

    # Final compute of C5
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
