        # The relevant part is the middle part of the correlation
        idx = len(corr) // 2
        c5 = np.max(corr[idx - 3:idx + 4]) * (2.0 / n_points)
        return c5

    def optimize_h(h):
        # Compute the overlap integral and return the value
        return compute_c5(h)
