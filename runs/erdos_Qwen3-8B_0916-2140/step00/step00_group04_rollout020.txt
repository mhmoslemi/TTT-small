    dx = 2.0 / n_points
    
    # Compute current C5
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    current_c5 = compute_c5(initial_h)
    best_h = initial_h.copy()
    best_c5 = current_c5
    
    # Try with smaller n_points
    for n in [256, 512, 1024, 2048, 4096]:
        if n < n_points:
            continue
        
        # Generate random initial guess
        h = np.random.rand(n)
        # Normalize to satisfy integral constraint
        h = h / (np.sum(h) / (n / 2))
        
        # Optimization: minimize max overlap
        def objective(x, n, dx):
            x = np.array(x)
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx
        
        # Bounds
        bounds = [(0, 1) for _ in range(n)]
        
        # Use scipy.optimize to find minimum
        result = optimize.minimize(
            lambda x: objective(x, n, dx),
            x0=h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n
        
    # Ensure best solution is with highest n_points
    if best_n is not None:
        return (best_h, best_c5, best_n)
    else:
        return (initial_h, current_c5, n_points)
