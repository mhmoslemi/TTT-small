import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d
from scipy.ndimage import convolve1d
from functools import lru_cache

initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 1000  # Start with 1000 points for good precision
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points] if n_points <= len(initial_h_values) else np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_func = interp1d(np.linspace(0, 2, n_points), h, kind='linear', bounds_error=False, fill_value=0)
        overlaps = []
        for k in range(1, 5):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h_func(np.linspace(0, 2, n_points)) * (1 - h_func(np.linspace(0, 2, n_points) + k)) * dx)
            overlaps.append(overlap)
        return max(overlaps)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, jac=None, options={'maxiter': 100})
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    print(f"Best C₅: {best_c5}")
    
    # Reduce n_points to 500 for faster evaluation and final answer
    n_points = 500
    dx = 2.0 / n_points
    best_h = best_h[:n_points]
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
