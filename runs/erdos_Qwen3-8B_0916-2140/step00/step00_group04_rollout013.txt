```python
import numpy as np
import scipy.optimize as opt
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 samples for computational efficiency

    # Function to evaluate the C5 value
    def evaluate_C5(h):
        h = np.array(h)
        kernel = 1 - h
        conv = np.convolve(h, kernel, mode='full')
        dx = 2.0 / len(h)
        c5 = np.max(conv) * dx
        return c5

    # Function to generate an initial guess with uniform distribution
    def generate_uniform_guess(n):
        return np.random.uniform(0, 1, n)

    # Function to generate an initial guess with fixed density
    def generate_fixed_density_guess(n):
        return np.full(n, 0.5)

    # Explored starting points
    start_points = [
        initial_h_values,
        generate_uniform_guess(100),
        generate_fixed_density_guess(100),
    ]

    # Try each starting point
    for start_point in start_points:
        if len(start_point) > n_points:
            start_point = start_point[:n_points]
        h_start = start_point.copy()
        h_start = np.clip(h_start, 0, 1)
        sum_h = np.sum(h_start)
        scaling = (n_points / 2.0) / sum_h if sum_h > 0 else 1.0
        h_start = np.clip(h_start * scaling, 0, 1)

        # Define bounds for the optimization
        bounds = [(0, 1) for _ in range(n_points)]

        # Optimization method with regularization
        def objective(h_flat):
            h = h_flat.reshape(-1)
            sum_h = np.sum(h)
            if sum_h == 0:
                return float('inf')
            scaling = (n_points / 2.0) / sum_h
            scaled_h = np.clip(h * scaling, 0, 1)
            kernel = 1 - scaled_h
            conv = np.convolve(scaled_h, kernel, mode='full')
            dx = 2.0 / n_points
            c5 = np.max(conv) * dx
            return c5

        def constraint_sum(h_flat):
            return np.sum(h_flat) - n_points / 2.0

        # Optimization with constraints
        result = opt.minimize(
            objective,
            x0=h_start,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method='L-BFGS-B',
            options={"disp": False, "maxiter": 100}
        )

        # Check if the optimization was successful
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            scaling = (n_points / 2.0) / sum_h
            scaled_h = np.clip(h_opt * scaling, 0, 1)
            c5 = evaluate_C5(scaled_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = scaled_h
        else:
            print("Optimization failed.")

        # Early stopping if time is exhausted
        if time.time() - start_time > budget_s - 10:
            break

    # Final check with a denser grid if needed
    if best_c5 > 0.380876:
        n_points = 200
        best_h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(best_h)
        scaling = (n_points / 2.0) / sum_h
        best_h = np.clip(best_h * scaling, 0, 1)
        best_c5 = evaluate_C5(best_h)

    return best_h, best_c5, n_points
```
