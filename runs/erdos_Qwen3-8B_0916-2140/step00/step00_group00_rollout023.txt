def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = h.reshape(-1)
    h_corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(h_corr) * dx
    return c5

def evaluate_erdos_solution(h):
    return compute_c5(h)

def run(seed=42, budget_s=600):
    random.seed(seed)
    np.random.seed(seed)
    
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # since sum(h) * dx = 1
    
    # Gaussian smoothing kernel
    def gaussian_kernel(n_points, sigma=0.1):
        x = np.linspace(-1, 1, n_points)
        return np.exp(-x**2 / (2 * sigma**2)) / np.sqrt(2 * np.pi)
    
    # Gradient descent with momentum for optimization
    def optimize_h(h, n_iter=100, lr=0.01, momentum=0.9):
        h = h.copy()
        n_points = len(h)
        dx = 2.0 / n_points
        target_sum = n_points / 2
        loss_history = []
        
        # Initialize velocities
        velocities = np.zeros(n_points)
        
        for _ in range(n_iter):
            # Compute the overlap integral for all shifts
            h_corr = signal.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(h_corr) * dx
            loss = max_overlap
            
            # Compute gradients
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Contribution to loss from all shifts
                for shift in range(1, n_points):
                    if i - shift >= 0:
                        grad[i] += (1 - h[i - shift]) * dx
                    if i + shift < n_points:
                        grad[i] += (1 - h[i + shift]) * dx
                grad[i] -= h[i] * dx
                grad[i] += (1 - h[i]) * dx
            
            # Update velocities and weights
            velocities = momentum * velocities - lr * grad
            h += velocities
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0.0, 1.0)
            
            # Ensure sum(h) * dx = 1
            h_sum = np.sum(h)
            if h_sum != 0:
                h = h / h_sum * target_sum
            
            loss_history.append(loss)
        
        return h, loss_history
    
    # Run optimization
    start_time = time.time()
    n_iter = 200
    best_c5 = np.inf
    best_h = h
    
    # Try multiple initializations with Gaussian smoothing
    for _ in range(3):
        # Apply Gaussian smoothing to the initial h
        kernel = gaussian_kernel(n_points, sigma=0.1)
        h_smooth = np.convolve(h, kernel, mode='same')
        h_smooth = np.clip(h_smooth, 0.0, 1.0)
        h_smooth = h_smooth / np.sum(h_smooth) * target_sum
        
        # Run optimization
        optimized_h, _ = optimize_h(h_smooth, n_iter=n_iter)
        current_c5 = compute_c5(optimized_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
