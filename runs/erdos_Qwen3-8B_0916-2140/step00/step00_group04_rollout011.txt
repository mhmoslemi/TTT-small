import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h.npy')
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    
    def obj_func(h, dx, n_points, sum_h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - sum_h) > 1e-6:
            return float('inf')
        
        def overlap_integral(h, dx):
            h_shifted = np.zeros_like(h)
            for k in range(1, n_points):
                h_shifted[k:] = h[:n_points - k]
            overlap = np.correlate(h, 1 - h_shifted, mode='full')
            return np.max(overlap) * dx
        
        return overlap_integral(h, dx)
    
    def evaluate_erdos_solution(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-6:
            return float('inf')
        
        def overlap_integral(h, dx):
            h_shifted = np.zeros_like(h)
            for k in range(1, n_points):
                h_shifted[k:] = h[:n_points - k]
            overlap = np.correlate(h, 1 - h_shifted, mode='full')
            return np.max(overlap) * dx
        
        return overlap_integral(h, dx)
    
    def optimize_h(initial_guess, n_points, dx, sum_h, max_iter=100, tol=1e-6):
        result = minimize(
            fun=obj_func,
            x0=initial_guess,
            args=(dx, n_points, sum_h),
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': max_iter, 'ftol': tol}
        )
        return result
    
    def refine_solution(h, n_points, dx, sum_h, max_refinement=5):
        for _ in range(max_refinement):
            result = optimize_h(h, n_points, dx, sum_h)
            if result.success:
                h = result.x
            else:
                break
        return h
    
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    
    if n_points < 1000:
        refined_h = refine_solution(best_h, n_points, dx, sum_h)
        best_c5 = evaluate_erdos_solution(refined_h)
    
    return (best_h, best_c5, len(best_h))
