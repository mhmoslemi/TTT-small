import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import lsqr
import time

def initial_h_values(n_points=70):
    # A known initial construction for h(x) with 70 samples
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.3 <= x <= 0.7:
            h[i] = 1.0
        elif 0.2 <= x <= 0.5 or 0.8 <= x <= 1.1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    return h

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all shifted (1 - h[x + k]) for k in [0, 1, ..., n_points - 1]
    overlap = np.zeros(n_points)
    for k in range(n_points):
        shifted = np.roll(h, k)
        overlap[k] = np.sum(h * (1 - shifted)) * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with 100 points for better accuracy
    h = initial_h_values(n_points)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Helper function to evaluate the C5 value for a given h
    def objective(h):
        return evaluate_erdos_solution(h)

    # Optimization loop with simulated annealing
    for trial in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        # Random perturbation
        h = best_h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] = np.random.rand()
        # Ensure the integral constraint
        total = np.sum(h) * 2.0 / n_points
        h = h * (1.0 / total) * (n_points / 2.0)
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Reduce to 50 points for faster evaluation and final result
    n_points = 50
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        if 0.3 <= x <= 0.7:
            h[i] = 1.0
        elif 0.2 <= x <= 0.5 or 0.8 <= x <= 1.1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    # Final evaluation
    c5 = evaluate_erdos_solution(h)
    return (h, c5, n_points)
