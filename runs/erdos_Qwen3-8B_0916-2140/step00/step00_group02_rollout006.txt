import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C
from sklearn.preprocessing import StandardScaler
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    n_points = 45
    h_initial = initial_h_values
    dx = 2.0 / n_points
    sum_h = np.sum(h_initial)
    h_initial = h_initial * (n_points / 2) / sum_h
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        shifted_h = np.concatenate([h[-2:], h])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to optimize h
    def objective(h):
        return compute_c5(h)
    
    # Function to check constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial point
    h0 = h_initial.copy()
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with constraints
    result = minimize(objective, h0, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], method='SLSQP')
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Refine with more points if time allows
    if budget_s > 100:
        n_points = 100
        dx = 2.0 / n_points
        h_refined = np.random.rand(n_points) * 0.5
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Optimize refined h
        result_refined = minimize(objective, h_refined, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint_sum}], method='SLSQP')
        best_h_refined = result_refined.x
        best_c5_refined = compute_c5(best_h_refined)
        
        best_h = best_h_refined
        best_c5 = best_c5_refined
    
    return best_h, best_c5, len(best_h)
