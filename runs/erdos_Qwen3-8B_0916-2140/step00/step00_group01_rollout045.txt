import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import matplotlib.pyplot as plt

# Initial construction from initial_h_values
initial_h_values = np.loadtxt('initial_h_values.txt')  # Assume this is a preloaded file

n_points = len(initial_h_values)
dx = 2.0 / n_points

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    h = h_values
    # Compute all shifted versions of 1 - h
    all_shifts = []
    for k in range(1, n):
        shifted_h = np.roll(h, k)
        all_shifts.append(1 - shifted_h)
    # Compute cross-correlation
    corr = np.zeros(n)
    for i in range(n):
        corr[i] = np.sum(h[i:] * (1 - h[:n - i]))
    corr = np.concatenate([corr[n - 1:], corr[:n - 1]])
    corr = corr[1:]  # Exclude self-correlation
    return np.max(corr * dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start from initial_h_values
    h = initial_h_values.copy()
    
    # Refine with a few steps of iterative optimization
    for _ in range(5):
        # Compute overlaps for all k
        n = len(h)
        overlaps = np.zeros(n - 1)
        for k in range(1, n):
            shifted_h = np.roll(h, k)
            overlaps[k - 1] = np.sum(h * (1 - shifted_h))
        overlaps = overlaps * dx
        max_overlap = np.max(overlaps)
        
        # Find the k that gives the worst overlap
        worst_k = np.argmax(overlaps) + 1
        
        # Perturb the worst_k position
        if worst_k < n:
            # Perturb the value at the current position
            perturb = np.random.uniform(-0.1, 0.1)
            h[worst_k] = np.clip(h[worst_k] + perturb, 0, 1)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1) > 1e-6:
            h = initial_h_values.copy()
            h = np.clip(h + np.random.normal(0, 0.05, h.shape), 0, 1)
            total = np.sum(h) * dx
            if abs(total - 1) > 1e-6:
                raise ValueError("Failed to maintain integral constraint")
    
    c5_bound = evaluate_erdos_solution(h)
    
    return h, c5_bound, len(h)
