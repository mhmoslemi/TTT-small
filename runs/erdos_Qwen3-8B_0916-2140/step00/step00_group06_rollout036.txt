```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 1000  # Starting with 1000 points
    max_iter = 100
    step_size = 0.01
    convergence_threshold = 1e-5
    n_candidates = 10
    n_ensemble = 10
    n_tuning = 50
    phase_shifts = np.linspace(-1.0, 1.0, 50)
    
    # Initial guess
    h = initial_h_values.copy()
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    
    # Function to evaluate C5
    def evaluate_C5(h):
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h.reshape(-1)
        h_fft = np.fft.fft(h)
        overlap = np.abs(np.fft.ifft(h_fft * np.conj(h_fft[n_points//2:]))) * 2
        return np.max(overlap)
    
    # Function to perturb h
    def perturb_h(h, step_size):
        h = h.copy()
        indices = np.random.choice(len(h), size=int(len(h)*0.1), replace=False)
        h[indices] += np.random.uniform(-step_size, step_size, size=len(indices))
        h = np.clip(h, 0, 1)
        return h
    
    # Function to find optimal shifts
    def find_optimal_shifts(h):
        shifts = []
        for shift in phase_shifts:
            shifted_h = np.roll(h, int(shift * n_points))
            shifted_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(shifted_h)), shifted_h)
            shifts.append((shift, shifted_h))
        return shifts
    
    # Main optimization loop
    best_h = h
    best_C5 = evaluate_C5(h)
    
    for iteration in range(max_iter):
        print(f"Iteration {iteration + 1}/{max_iter}")
        
        # Generate candidate shifts
        shift_candidates = find_optimal_shifts(best_h)
        
        # Evaluate each candidate
        evaluations = []
        for shift, shifted_h in shift_candidates:
            evaluations.append((shift, shifted_h))
        
        # Find the best shift
        best_shift, best_shifted_h = min(evaluations, key=lambda x: evaluate_C5(x[1]))
        
        # Update best_h
        best_h = best_shifted_h
        best_C5 = evaluate_C5(best_h)
        
        # Check for convergence
        if best_C5 < convergence_threshold:
            print("Converged to C5 < 1e-5")
            break
        
        # Generate perturbations
        perturbations = [perturb_h(best_h, step_size) for _ in range(n_candidates)]
        
        # Evaluate perturbations
        perturbation_scores = []
        for perturbation in perturbations:
            score = evaluate_C5(perturbation)
            perturbation_scores.append(score)
        
        # Choose the best perturbation
        best_perturbation = perturbations[np.argmin(perturbation_scores)]
        
        # Update best_h
        best_h = best_perturbation
        best_C5 = evaluate_C5(best_h)
        
        # Check if we've exceeded the time budget
        if time.time() - start_time > budget_s - 10:
            print("Time budget exceeded, returning current best")
            break
    
    print(f"Best C5: {best_C5}")
    print(f"Time elapsed: {time.time() - start_time:.2f}s")
    return (best_h, best_C5, n_points)
```
