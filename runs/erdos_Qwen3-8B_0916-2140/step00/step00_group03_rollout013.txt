import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=600, n_points=200):
    np.random.seed(seed)
    
    # Initial h_values from the given initial_h_values
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    if n_points_initial != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), initial_h)
    
    def compute_overlap(h):
        h = h.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        return np.max(overlap * dx)
    
    def objective(h):
        return compute_overlap(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Constraints
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = optimize.minimize(
        objective,
        h0,
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    h_best = result.x
    c5 = compute_overlap(h_best)
    
    return h_best, c5, n_points
