```python
import numpy as np
import scipy.optimize as optimize
import time
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
from sklearn.gaussian_process import GaussianProcessRegressor
from scipy.optimize import minimize
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n_points=80):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points / 3:
            h[i] = 1.0
        elif n_points / 3 <= i < 2 * n_points / 3:
            h[i] = 0.5
        else:
            h[i] = 0.0
    return h

def objective_function(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def constraint_sum(h_values, n_points):
    return np.sum(h_values) - n_points / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial guess: use the initial construction
    n_points = 100  # Using 100 points for better accuracy while staying under time budget
    initial_h = initial_h_values(n_points)
    initial_h = initial_h + np.random.normal(0, 0.05, n_points)
    initial_h = np.clip(initial_h, 0, 1)

    # Objective function to minimize: C5
    def objective(x):
        return objective_function(x, n_points)

    # Constraint: sum of h must be n_points / 2
    def constraint(x):
        return constraint_sum(x, n_points)

    # Bounds: each h[i] ∈ [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization using constrained optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint}, method="SLSQP", options={"maxiter": 100, "ftol": 1e-8, "disp": False})

    # Evaluate best solution
    best_h = result.x
    best_c5 = objective(best_h)

    # Return the best solution and C5 value
    return (best_h.tolist(), best_c5, n_points)
```
