    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Helper function to compute C5
    def compute_C5(h_values):
        h = np.array(h_values)
        n = len(h)
        shift = np.roll(h, 1)
        shifted = np.roll(h, -1)
        # Compute the overlap integrals for k = -1, 0, 1
        overlap_1 = np.sum(h * (1 - shifted)) * dx
        overlap_0 = np.sum(h * (1 - h)) * dx
        overlap_neg1 = np.sum(h * (1 - shift)) * dx
        return max(overlap_1, overlap_0, overlap_neg1)
    
    # Objective function for optimization
    def objective(h_values):
        return compute_C5(h_values)
    
    # Constraints
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds for each h[i] (0 <= h[i] <= 1)
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    # Starting point (initial_h_values)
    x0 = initial_h.copy()
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    time_limit = budget_s
    
    start_time = time.time()
    best_solution = x0
    best_C5 = compute_C5(x0)
    
    # Use L-BFGS-B optimizer with constraints
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='L-BFGS-B',
        options={'maxiter': max_iter, 'ftol': tolerance, 'gtol': tolerance}
    )
    
    # Update best solution
    if result.success:
        best_solution = result.x
        best_C5 = compute_C5(best_solution)
    
    # If no improvement, try a few perturbations
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(best_solution + perturbation, 0, 1)
        current_C5 = compute_C5(perturbed)
        if current_C5 < best_C5:
            best_solution = perturbed
            best_C5 = current_C5
    
    # Ensure the solution meets the sum constraint
    sum_h = np.sum(best_solution)
    if abs(sum_h - n_points / 2) > tolerance:
        # Normalize if necessary
        scale = (n_points / 2) / sum_h
        best_solution = np.clip(best_solution * scale, 0, 1)
    
    # Final check
    final_C5 = compute_C5(best_solution)
