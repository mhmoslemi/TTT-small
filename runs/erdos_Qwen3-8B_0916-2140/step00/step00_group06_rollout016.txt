import numpy as np
import scipy.optimize as optimize
import random
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    start_time = time.time()
    
    # Initial guess from provided initial_h_values
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral for a given h
    def evaluate_overlap(h):
        h = np.array(h)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        conv = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(conv) * dx
        return overlap
    
    # Optimization objective: minimize the maximum overlap
    def objective(h):
        return evaluate_overlap(h)
    
    # Optimization with bounds and constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h_initial = np.array(initial_h_values)
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        fun=objective,
        x0=h_initial,
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Ensure we are within the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        return (best_h, best_c5, n_points)
    
    # If not, try a few more random perturbations to escape local minima
    for _ in range(5):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        c5_perturb = evaluate_overlap(h_perturb)
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb
    
    return (best_h, best_c5, n_points)
