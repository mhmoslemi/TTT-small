import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    n_points = 200
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    h = np.random.permutation(h)
    
    # Parameters
    max_iterations = 100
    learning_rate = 0.1
    tolerance = 1e-5
    time_limit = budget_s
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > time_limit:
            break
        
        # Compute current C5
        c5 = 0
        for k in range(1, n_points):
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = np.max(corr * (2.0 / n_points))
            if overlap > c5:
                c5 = overlap
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(1, n_points):
                h_i = h[i]
                h_i_plus_k = h[(i + k) % n_points]
                h_i_minus_k = h[(i - k) % n_points]
                
                # Compute contribution to gradient
                contribution = 0
                if h_i_plus_k > 0:
                    contribution -= (1 - h_i_plus_k) / n_points
                if h_i_minus_k > 0:
                    contribution += (1 - h_i_minus_k) / n_points
                
                grad[i] += contribution
        
        # Update h
        h = np.clip(h - learning_rate * grad, 0, 1)
        
        # Normalize h
        total = np.sum(h) * (2.0 / n_points)
        h = h * (1 / total) * (2.0 / n_points)
    
    # Final C5
    final_c5 = 0
    for k in range(1, n_points):
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * (2.0 / n_points))
        if overlap > final_c5:
            final_c5 = overlap
    
    return h, final_c5, n_points
