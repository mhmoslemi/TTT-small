        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = copy.copy(h)
            h_plus[i] += epsilon
            cost_plus = compute_overlap(h_plus)
            
            h_minus = copy.copy(h)
            h_minus[i] -= epsilon
            cost_minus = compute_overlap(h_minus)
            
            grad[i] = (cost_plus - cost_minus) / (2 * epsilon)
        return grad

        return compute_cost(h)

        c5 = compute_overlap(h)
        return (h, c5, n_points)

    h = np.array(initial_h)

    learning_rate = 0.01
