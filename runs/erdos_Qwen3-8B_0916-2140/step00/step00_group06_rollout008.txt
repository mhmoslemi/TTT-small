import numpy as np
import scipy.optimize as opt
import time
import matplotlib.pyplot as plt

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial construction
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 256  # Start with a moderate number of points
    max_iter = 1000  # Maximum number of iterations
    learning_rate = 0.01
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = np.copy(initial_h_values[:n_points])
    
    # Ensure the initial h satisfies the integral condition
    h = np.copy(best_h)
    total = np.sum(h)
    h /= total  # Normalize to ensure ∫h dx = 1
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
        
        # Compute the overlap integral for each shift
        shifts = np.arange(-len(h) + 1, len(h))
        overlap = np.zeros_like(shifts)
        
        for i, shift in enumerate(shifts):
            overlap[i] = np.sum(h * (1 - h[shift:] if shift >= 0 else 1 - h[:shift]))
        
        # Compute gradient (direction of maximum overlap)
        grad = np.zeros(len(h))
        for i in range(len(h)):
            for shift in shifts:
                if shift >= 0:
                    val = h[i] * (1 - h[shift:])
                    val -= h[i] * (1 - h[shift:])
                    grad[i] += val
                else:
                    val = h[i] * (1 - h[:shift])
                    val -= h[i] * (1 - h[:shift])
                    grad[i] += val
        
        # Update h with gradient descent
        h -= learning_rate * grad
        
        # Ensure h stays within [0, 1]
        h = np.clip(h, 0, 1)
        
        # Ensure ∫h dx = 1
        h /= np.sum(h)
    
    # Final check
    h = np.copy(best_h)
    total = np.sum(h)
    h /= total
    
    corr = np.correlate(h, 1 - h, mode="full")
    final_c5 = np.max(corr) * (2.0 / len(h))
    
    return (h, final_c5, len(h))
