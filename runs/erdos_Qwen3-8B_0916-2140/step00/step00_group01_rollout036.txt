        for i in range(n_points):
            if i < n_points // 2:

                h[i] = 1.0
        initial_h_values = h
    
    def compute_C5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        h_shifted = np.roll(h, shift=1)
        overlap = np.correlate(h, 1 - h_shifted, mode='full')
        return np.max(overlap) * dx
    
    def objective_func(h):
        return compute_C5(h)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h_values.copy()
    
    # Optimization settings
    maxiter = 100
    tol = 1e-6
    
    # Perform the optimization
    result = minimize(
        fun=objective_func,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'maxiter': maxiter, 'ftol': tol, 'disp': False}
    )
    
    # Extract the best solution
    best_h = result.x
    best_C5 = compute_C5(best_h)
    
    # Return the best solution, its C5 value, and the number of points
    return best_h, best_C5, len(best_h)
