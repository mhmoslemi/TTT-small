import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize


def run(seed=42, budget_s=600, **kwargs):
    # Parameters
    n_points = 80  # Starting with 80 points as per initial construction
    dx = 2.0 / n_points
    max_iter = 100
    tol = 1e-4

    # Initial construction from given initial_h_values
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

    # Ensure initial_h_values is in the correct range and has correct sum
    initial_h_values = np.clip(initial_h_values, 0, 1)
    # Rescale to ensure sum(h) == n_points / 2
    initial_h_values = initial_h_values / np.sum(initial_h_values) * (n_points / 2)

    # Objective function to minimize the maximum overlap integral
    def objective(h):
        h = np.array(h)
        # Compute the overlap integrals for all k
        overlaps = []
        for k in range(1, n_points - 1):
            # Mirror h to handle circular convolution
            h_shifted = np.concatenate([h[-k:], h[:-k]])
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)

    # Constraints: sum(h) == n_points / 2 and 0 <= h[i] <= 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization using Sequential Least Squares Programming (SLSQP)
    result = minimize(
        objective,
        initial_h_values,
        method="SLSQP",
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        tol=tol,
        options={"maxiter": max_iter, "ftol": tol, "disp": False}
    )

    # Extract the best solution found
    best_h = result.x
    best_c5 = objective(best_h)

    # Return the best solution
    return best_h, best_c5, n_points
