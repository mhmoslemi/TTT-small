import numpy as np
import scipy.optimize
import time
import copy

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    initial_h = initial_h_values
    h = np.array(initial_h, dtype=np.float64)
    
    def compute_C5(h):
        h = np.array(h)
        h = np.concatenate([h, h])
        shifted = np.array([h[i:] for i in range(len(h))])
        overlaps = np.sum(h[:, np.newaxis] * (1 - shifted), axis=1)
        return np.max(overlaps) * (2.0 / len(h))
    
    def objective(x):
        x = np.array(x, dtype=np.float64)
        return compute_C5(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    def constraint_bounds(x):
        return np.all((0 <= x) & (x <= 1))
    
    def optimize_h(h_init):
        bounds = [(0.0, 1.0) for _ in range(len(h_init))]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        result = scipy.optimize.minimize(
            objective,
            h_init,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        if not result.success:
            return h_init
        return result.x
    
    best_h = copy.copy(initial_h)
    best_c5 = compute_C5(best_h)
    
    for _ in range(10):
        start_time = time.time()
        h_opt = optimize_h(best_h)
        if not np.all((0.0 <= h_opt) & (h_opt <= 1.0)):
            continue
        c5 = compute_C5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.9:
            break
    
    return best_h, best_c5, len(best_h)
