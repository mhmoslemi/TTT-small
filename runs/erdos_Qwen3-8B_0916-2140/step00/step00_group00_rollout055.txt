def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    # Compute the overlap integral for all k
    overlaps = []
    for k in range(1, n_points):
        shifted = np.roll(h, k)
        product = h * (1 - shifted)
        overlap = np.sum(product) * dx
        overlaps.append(overlap)
    # Compute max overlap
    max_overlap = np.max(overlaps)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    # Start time
    start_time = time.time()
    
    # Define initial parameters
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize h_values to a basic function that approximates the known solution
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0 / (n_points // 2)
        else:
            h[i] = 0.0
    
    # Try to improve the initial solution using a local optimization method
    def objective(x):
        h = np.array(x, dtype=np.float64)
        # Compute the overlap integral for all k
        overlaps = []
        for k in range(1, n_points):
            shifted = np.roll(h, k)
            product = h * (1 - shifted)
            overlap = np.sum(product) * dx
            overlaps.append(overlap)
        # Compute max overlap
        max_overlap = np.max(overlaps)
        return max_overlap
    
    def constraints(x):
        # Constraint: sum(h) * dx = 1
        return np.sum(x) * dx - 1.0
    
    # Bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use the initial guess to start optimization
    x0 = h.copy()
    
    # Run optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="L-BFGS-B",
        options={"maxiter": 100}
    )
    
    # Extract the best solution
    best_h = result.x
    
    # Compute the final C₅ value
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Check if the solution meets the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If time is up, return the initial solution
        return (initial_h_values, evaluate_erdos_solution(initial_h_values), len(initial_h_values))
    
    # Return the best solution found
    return (best_h, final_c5, n_points)
