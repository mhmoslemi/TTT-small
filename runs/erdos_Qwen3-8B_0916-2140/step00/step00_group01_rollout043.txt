# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all shifts k in [0, 2] to avoid edge effects, but focus on k in [0, 1]
    # We evaluate for k in [0, 1] with step size 0.1
    k_values = np.arange(0, 1.01, 0.01)
    max_overlap = 0.0
    for k in k_values:
        # Compute the overlap integral
        # h(x) * (1 - h(x + k)) dx
        overlap = 0.0
        for i in range(n_points):
            x = i * dx
            x_shifted = x + k
            if x_shifted >= 2.0:
                continue
            idx = int(np.clip(x_shifted * n_points, 0, n_points - 1))
            overlap += h_values[i] * (1 - h_values[idx]) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap
