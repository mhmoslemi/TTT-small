import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import time
import sys
import random

# Initial construction (80 samples)
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Precompute useful constants
    n_points_initial = initial_h_values.shape[0]
    dx = 2.0 / n_points_initial
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        n = len(h)
        overlap = np.zeros(n)
        for i in range(n):
            for j in range(n):
                if i + j < n:
                    overlap[i] += h[i] * (1 - h[j])
        return np.max(overlap) * dx
    
    # Function to compute the overlap between h and 1 - h with a shift
    def compute_overlap_fft(h):
        n = len(h)
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        shifted_h_fft = np.zeros_like(h_fft)
        for k in range(n):
            shifted_h_fft[k] = h_fft[(n - k) % n]
        corr = np.fft.ifft(h_fft * np.conj(one_minus_h_fft)).real
        return np.max(corr) * dx
    
    # Initial guess from provided initial_h_values
    n_points = n_points_initial
    h = initial_h_values.copy()
    
    # Function to evaluate current solution
    def evaluate_solution(h):
        return compute_overlap_fft(h)
    
    # Optimize with local search
    def optimize_local_search(h, max_iter=100, learning_rate=0.01, tolerance=1e-4):
        for _ in range(max_iter):
            current_c5 = evaluate_solution(h)
            if current_c5 < tolerance:
                break
            best_shift = None
            best_h = h.copy()
            best_c5 = current_c5
            for shift in range(n_points):
                h_shifted = np.roll(h, shift)
                new_c5 = compute_overlap_fft(h_shifted)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_shift = shift
                    best_h = h_shifted
            if best_c5 < current_c5:
                h = best_h
            else:
                break
        return h, evaluate_solution(h)
    
    # Run local search
    h, c5_bound = optimize_local_search(h)
    
    # Evaluate current solution
    final_c5 = evaluate_solution(h)
    
    # Extract the final h_values
    h_values = h
    
    # Return the result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (h_values, final_c5, n_points)
    
    return (h_values, final_c5, n_points)
