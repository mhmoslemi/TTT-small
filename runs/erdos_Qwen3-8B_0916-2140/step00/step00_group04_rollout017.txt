import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from scipy.sparse import linalg as sparse_linalg
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
import pickle

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):

    # Set random seed
    np.random.seed(seed)

    # Initial guess
    n_points = 100
    h = initial_h_values()[:n_points]
    dx = 2.0 / n_points

    # Precompute the shift indices for k in [0, 1, ..., floor(2 - 0.0001)]
    max_k = int(np.ceil(2.0 - 1e-4))  # Max shift to keep within [0, 2]
    k_shifts = np.arange(-max_k, max_k + 1)

    # Function to compute overlap integral for a given h
    def compute_overlap(h, dx, k_shifts):
        max_k = len(k_shifts) - 1
        overlaps = []
        for k in k_shifts:
            # Compute (1 - h(x + k))
            shifted_h = np.roll(h, k)
            shifted_h = np.clip(shifted_h, 0, 1)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)

    # Objective function to minimize
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return compute_overlap(h, dx, k_shifts)

    # Constraints
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    def constraint_bounds(h_flat):
        return np.all(0 <= h_flat) and np.all(h_flat <= 1)

    # Set up bounds for optimization
    bounds = Bounds(0, 1)

    # Initial guess
    initial_guess = h

    # Optimization settings
    max_iterations = 100
    tol = 1e-5

    # Optimization using L-BFGS-B
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': max_iterations, 'ftol': tol}
    )

    best_h = result.x
    best_c5 = compute_overlap(best_h, dx, k_shifts)

    # If not improved, perform a grid search around the initial guess
    if best_c5 > 0.380876:
        # Fine-tune around best_h with small perturbations
        for perturbation in np.linspace(-0.1, 0.1, 21):
            perturbed_h = best_h + perturbation * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            c5_candidate = compute_overlap(perturbed_h, dx, k_shifts)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = perturbed_h

    # Final result
    return best_h, best_c5, n_points
