import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial h_values (70 samples)
    initial_h_values = np.array([0.6685, 0.7566, 0.8236, 0.8744, 0.9148, 0.9453, 0.9675, 0.9827, 0.9918, 0.9965,
                                0.9982, 0.9971, 0.9933, 0.9868, 0.9776, 0.9658, 0.9513, 0.9339, 0.9137, 0.8907,
                                0.8651, 0.8376, 0.8083, 0.7775, 0.7454, 0.7122, 0.6782, 0.6438, 0.6089, 0.5737,
                                0.5383, 0.5033, 0.4684, 0.4335, 0.3991, 0.3653, 0.3322, 0.3001, 0.2692, 0.2394,
                                0.2108, 0.1831, 0.1565, 0.1309, 0.1064, 0.0831, 0.0613, 0.0408, 0.0214, 0.0031,
                                0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000,
                                0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000,
                                0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000])
    
    n_points = 70
    dx = 2.0 / n_points
    
    def evaluate_erdos_solution(h):
        h = np.array(h)
        h = np.concatenate((h, h))
        result = np.correlate(h, 1 - h, mode='full')
        overlap = result * dx
        return np.max(overlap)
    
    # Use initial guess
    h = initial_h_values.copy()
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-4
    
    for _ in range(max_iter):
        start_time = time.time()
        h = h.copy()
        
        # Compute current overlap
        current_c5 = evaluate_erdos_solution(h)
        
        # Set up optimization problem
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            x >= 0,
            x <= 1
        ]
        obj = cp.Maximize(cp.sum(x * (1 - x[np.roll(range(n_points), 1)])) * dx)
        
        # Solve the problem
        problem = cp.Problem(obj, constraints)
        try:
            problem.solve(solver=cp.OSQP, eps_abs=1e-6, eps_rel=1e-6, max_iter=1000)
            if problem.status == 'optimal':
                h = x.value
            else:
                h = np.random.uniform(0, 1, n_points)
                h = h / np.sum(h) * (n_points / 2)
        except:
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * (n_points / 2)
        
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s:
            break
    
    c5_bound = evaluate_erdos_solution(h)
    return h, c5_bound, n_points
