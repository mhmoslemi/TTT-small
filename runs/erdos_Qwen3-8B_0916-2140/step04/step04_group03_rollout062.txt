import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the initial grid and sample points
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    initial_h = np.zeros(n_points)
    
    # Initial construction: piecewise constant with alternating high and low values
    for i in range(n_points):
        if i % 2 == 0:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.7
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization setup
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial optimization
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try several random perturbations to escape local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Try a coarser grid (n=100) for lower resolution but faster optimization
    n_points = 100
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Interpolate best_h from finer grid to coarser grid
    fine_x = np.linspace(0, 2, 200)
    fine_h = np.interp(fine_x, x, best_h)
    coarse_h = np.interp(x, fine_x, fine_h)
    coarse_h = np.clip(coarse_h, 0, 1)
    
    # Optimization on coarser grid
    initial_h = coarse_h
    
    def compute_overlap_new(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective_new(x):
        return compute_overlap_new(x)
    
    def constraint_sum_new(x):
        return np.sum(x) - n_points / 2
    
    bounds_new = [(0, 1) for _ in range(n_points)]
    cons_new = ({'type': 'eq', 'fun': constraint_sum_new})
    
    result_new = minimize(objective_new, initial_h, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h_new = result_new.x
    best_c5_new = compute_overlap_new(best_h_new)
    
    return (best_h_new.tolist(), best_c5_new, n_points)
