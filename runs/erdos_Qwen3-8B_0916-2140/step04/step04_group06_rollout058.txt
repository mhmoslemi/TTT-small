import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()  # Load initial construction
    
    # Start with a moderately high resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Define the overlap function
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Define objective function
    def objective(x):
        return compute_overlap(x)

    # Define constraint: sum of h must be n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Initial guess: interpolate initial_h_values onto the current grid
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Optimization using L-BFGS-B for better convergence
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 200})
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Try several random perturbations to escape local optima
    for _ in range(10):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 200})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)

    # Try a higher resolution (n=300)
    n_points = 300
    dx = 2.0 / n_points
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    initial_h = np.clip(initial_h, 0, 1)

    def compute_overlap_higher(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective_higher(x):
        return compute_overlap_higher(x)

    def constraint_sum_higher(x):
        return np.sum(x) - n_points / 2

    bounds_higher = [(0, 1) for _ in range(n_points)]
    cons_higher = ({'type': 'eq', 'fun': constraint_sum_higher})

    start_time = time.time()
    result_higher = minimize(objective_higher, initial_h, bounds=bounds_higher, constraints=cons_higher, method='L-BFGS-B', options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 200})
    time_used = time.time() - start_time
    if time_used < budget_s - 10:
        best_h_higher = result_higher.x
        best_c5_higher = compute_overlap_higher(best_h_higher)
        return (best_h_higher.tolist(), best_c5_higher, n_points)
    else:
        return (best_h.tolist(), best_c5, n_points)
