import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define a function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * 2.0 / len(h)
    
    # Define a function to compute the integral constraint
    def constraint_sum(x):
        return np.sum(x) - len(x) / 2
    
    # Load initial_h_values (n=300)
    initial_h_values = np.load('initial_h_values.npy')
    n_initial = len(initial_h_values)
    
    # Function to optimize
    def objective(x):
        return compute_overlap(x)
    
    # Initial optimization with initial_h_values
    bounds = [(0, 1) for _ in range(n_initial)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    result = minimize(objective, initial_h_values, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try several perturbations to escape local optima
    for _ in range(10):
        perturbation = np.random.uniform(-0.05, 0.05, len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Refine with finer grids (n=400, 500, 600)
    n_points = [400, 500, 600]
    start_time = time.time()
    
    for n in n_points:
        if time.time() - start_time > budget_s - 10:
            break
        dx = 2.0 / n
        x = np.linspace(0, 2, n)
        x_prev = np.linspace(0, 2, len(best_h))
        f = interp1d(x_prev, best_h, kind='linear', fill_value="extrapolate")
        initial_h = f(x)
        initial_h = np.clip(initial_h, 0, 1)
        
        bounds_new = [(0, 1) for _ in range(n)]
        cons_new = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2})
        
        result_new = minimize(objective, initial_h, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        best_h_new = result_new.x
        best_c5_new = compute_overlap(best_h_new)
        
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    return (best_h.tolist(), best_c5, len(best_h))
