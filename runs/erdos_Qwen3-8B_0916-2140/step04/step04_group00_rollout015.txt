import scipy.optimize as opt
import time

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Optimization function with improved settings
    def optimize_h(initial_guess):
        result = opt.minimize(lambda x: compute_overlap(x), initial_guess, bounds=[(0, 1) for _ in range(n_points)], method='L-BFGS-B', options={'ftol': 1e-10, 'maxiter': 200})
        return result.x
    
    # Function to generate perturbations
    def perturb(h, magnitude=0.05):
        pert = np.random.normal(0, magnitude, len(h))
        return np.clip(h + pert, 0, 1)
    
    # Phase 1: Initial optimization with perturbations
    best_h = optimize_h(initial_h)
    best_c5 = compute_overlap(best_h)
    best_time = time.time()
    
    # Phase 2: Iterative improvement with smart perturbations
    for _ in range(10):
        if time.time() - best_time > budget_s - 10:
            break
        perturbed_h = perturb(best_h, magnitude=0.03)
        current_h = optimize_h(perturbed_h)
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        print(f"Current best C5: {best_c5:.6f}")
    
    # Phase 3: Local search with fine grid around best solution
    if time.time() - best_time < budget_s - 10:

                current_c5 = compute_overlap(current_h)
