import numpy as np
import scipy.optimize as opt

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):

    # Function to compute the overlap integral
    def compute_overlap(h, dx):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x, dx):
        return compute_overlap(x, dx)
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Optimization function with early stopping
    def optimize_with_early_stop(initial_h, n_points, dx, max_iter=150, tolerance=1e-8):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)})
        result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', 
                          options={'ftol': tolerance, 'maxiter': max_iter})
        return result.x
    
    # Start with initial grid
    n_points_initial = len(initial_h)
    dx_initial = 2.0 / n_points_initial
    best_h = initial_h
    best_c5 = compute_overlap(best_h, dx_initial)
    
    # Try optimization with initial grid
    result = optimize_with_early_stop(initial_h, n_points_initial, dx_initial)
    best_h = result
    best_c5 = compute_overlap(best_h, dx_initial)
    
    # Try several random perturbations to escape local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, n_points_initial)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = optimize_with_early_stop(perturbed_h, n_points_initial, dx_initial)
        current_c5 = compute_overlap(result, dx_initial)
        if current_c5 < best_c5:
            best_h = result
            best_c5 = current_c5
    
    # Try a finer grid (n=400) with interpolated initial construction
    n_points = 400
    dx = 2.0 / n_points
    initial_h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), best_h)
    initial_h_fine = np.clip(initial_h_fine, 0, 1)
    
    result_fine = optimize_with_early_stop(initial_h_fine, n_points, dx)
    best_c5_fine = compute_overlap(result_fine, dx)

    if best_c5_fine < best_c5_coarse:

        best_c5_final = best_c5_coarse
