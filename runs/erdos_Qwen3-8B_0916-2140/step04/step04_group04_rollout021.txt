import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    initial_h_values[n_points // 4 : 3 * n_points // 4] = 1.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    def optimize_with_adaptive_perturbations(x0, max_perturb=0.1, n_perturb=20, n_iter=100, tol=1e-8):
        best_h = x0.copy()
        best_c5 = compute_overlap(best_h)
        
        for _ in range(n_perturb):
            perturb = np.random.uniform(-max_perturb, max_perturb, n_points)
            x_perturb = np.clip(best_h + perturb, 0, 1)
            
            result = minimize(objective, x_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tol, 'maxiter': n_iter})
            
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        return best_h, best_c5
    
    def optimize_with_sweep(x0, n_sweeps=5, alpha=0.9, tol=1e-8):
        h = x0.copy()
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        
        for _ in range(n_sweeps):
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            h = np.clip(h + alpha * perturb, 0, 1)
            result = minimize(objective, h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tol, 'maxiter': 100})
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        return best_h, best_c5
    
    x0 = initial_h_values.copy()
    best_h = x0.copy()
    best_c5 = compute_overlap(best_h)
    
    for _ in range(3):
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
        x0 = best_h.copy()
    
    best_h, best_c5 = optimize_with_adaptive_perturbations(best_h, max_perturb=0.05, n_perturb=20, n_iter=100, tol=1e-9)
    best_h, best_c5 = optimize_with_sweep(best_h, n_sweeps=5, alpha=0.9, tol=1e-9)
    
    return (best_h.tolist(), best_c5, n_points)
