import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a fine grid (n=200) as initial guess
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial guess: flat function with high density on [0,1]
    h_initial = np.zeros(n_points)
    h_initial[:int(n_points / 2)] = 1.0
    h_initial = np.array(h_initial, dtype=np.float64)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum(h) = n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization setup
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial optimization
    result = minimize(objective, h_initial, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try several random perturbations to escape local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Second round fine-tuning with more iterations
    result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try another random perturbation
    perturbation = np.random.uniform(-0.05, 0.05, n_points)
    perturbed_h = np.clip(best_h + perturbation, 0, 1)
    result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    if compute_overlap(result.x) < best_c5:
        best_h = result.x
        best_c5 = compute_overlap(best_h)
    
    # Try a coarser grid (n=100) for higher-level adjustments
    n_points_coarse = 100
    dx_coarse = 2.0 / n_points_coarse
    x_coarse = np.linspace(0, 2, n_points_coarse)
    
    # Interpolate the best_h from fine grid to coarse grid
    h_coarse = np.interp(x_coarse, x, best_h)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Function to compute the overlap integral for coarse grid
    def compute_overlap_coarse(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx_coarse
    
    # Objective function for coarse grid
    def objective_coarse(x):
        return compute_overlap_coarse(x)
    
    # Constraint: sum(h) = n_points_coarse / 2
    def constraint_sum_coarse(x):
        return np.sum(x) - n_points_coarse / 2
    
    # Define bounds for coarse grid
    bounds_coarse = [(0, 1) for _ in range(n_points_coarse)]
    
    # Optimization setup for coarse grid
    cons_coarse = ({'type': 'eq', 'fun': constraint_sum_coarse})
    
    # Initial optimization for coarse grid
    result_coarse = minimize(objective_coarse, h_coarse, bounds=bounds_coarse, constraints=cons_coarse, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h_coarse = result_coarse.x
    best_c5_coarse = compute_overlap_coarse(best_h_coarse)
    
    # Interpolate coarse solution back to fine grid
    h_final = np.interp(x, x_coarse, best_h_coarse)
    h_final = np.clip(h_final, 0, 1)
    
    # Final overlap computation
    best_c5_final = compute_overlap(h_final)
    
    # Return the best solution found
    return (h_final.tolist(), best_c5_final, n_points)
