```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Load initial construction
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * 2.0 / len(h)
    
    def constraint_sum(x):
        return np.sum(x) - len(x) / 2
    
    def objective(x):
        return compute_overlap(x)
    
    def perturb_vector(x, strength=0.05):
        return np.clip(x + np.random.uniform(-strength, strength, len(x)), 0, 1)
    
    def optimize_with_perturbations(initial_h, n_points, max_iter=200):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': max_iter})
        best_h = result.x
        best_c5 = compute_overlap(best_h)
        
        for _ in range(10):
            perturbed_h = perturb_vector(best_h)
            result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': max_iter})
            if compute_overlap(result.x) < best_c5:
                best_h = result.x
                best_c5 = compute_overlap(best_h)
        
        return best_h, best_c5
    
    start_time = time.time()
    
    # Initial optimization with initial_h_values
    n_points = len(initial_h_values)
    best_h, best_c5 = optimize_with_perturbations(initial_h_values, n_points)
    
    # Refine with finer grids
    for n_points in [200, 300, 400]:
        if time.time() - start_time > budget_s - 10:
            break
        x = np.linspace(0, 2, n_points)
        x_prev = np.linspace(0, 2, len(best_h))
        initial_h = np.interp(x, x_prev, best_h)
        initial_h = np.clip(initial_h, 0, 1)
        
        best_h_new, best_c5_new = optimize_with_perturbations(initial_h, n_points)
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    return (best_h.tolist(), best_c5, len(best_h))
```
