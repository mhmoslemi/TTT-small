    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint function
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization function with improved settings
    def optimize_h(initial_guess):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        return result.x
    
    # Function to generate perturbations
    def generate_perturbations(h, perturbation_range=0.05):
        perturbations = np.random.normal(0, perturbation_range, n_points)
        return np.clip(h + perturbations, 0, 1)
    
    # Function to perform local search with multiple perturbations
    def local_search(h, num_perturbations=10):
        best_h = h
        best_c5 = compute_overlap(h)
        for _ in range(num_perturbations):
            perturbed_h = generate_perturbations(h)
            current_h = optimize_h(perturbed_h)
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5

    def fine_tuning(h, step=0.05):
