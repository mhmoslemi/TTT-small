import scipy.optimize as opt

    one_minus_h = 1 - initial_h
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, one_minus_h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint function
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization with improved settings
    def optimize_h(initial_guess):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        return result.x
    
    # Initial optimization
    best_h = optimize_h(initial_h)
    best_c5 = compute_overlap(best_h)
    
    # Local search with guided perturbations
    if budget_s > 300:
        for _ in range(5):

            current_h = optimize_h(perturbed_h)
