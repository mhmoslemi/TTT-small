import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Using 200 points for better resolution
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    def optimize_with_perturbations(x0, n_perturb=20, n_iter=50, tol=1e-8):
        best_h = x0.copy()
        best_c5 = compute_overlap(best_h)
        
        for _ in range(n_perturb):
            # Perturb the current best solution
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            x_perturb = np.clip(best_h + perturb, 0, 1)
            
            # Perform local optimization with adaptive tolerance
            result = minimize(objective, x_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tol, 'maxiter': n_iter})
            
            # Update best solution
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        return best_h, best_c5
    
    # First optimization step with SLSQP
    x0 = initial_h_values.copy()
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)
    
    # Perform multi-start optimization with perturbations and adaptive step size
    best_h, best_c5 = optimize_with_perturbations(best_h, n_perturb=20, n_iter=200, tol=1e-9)
    
    return (best_h.tolist(), best_c5, n_points)
