import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Using 100 points for better performance
    dx = 2.0 / n_points

    # Define an initial guess with a piecewise linear function
    def initial_h_values(x):
        if 0 <= x < 0.5:
            return 0.0
        elif 0.5 <= x < 1.0:
            return 1.0
        elif 1.0 <= x < 1.5:
            return 0.0
        else:
            return 1.0

    # Generate the initial guess
    x_vals = np.linspace(0, 2, n_points)
    initial_h = np.array([initial_h_values(x) for x in x_vals])

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Optimization parameters
    max_iter = 100
    tol = 1e-8
    n_perturb = 20

    # Initial optimization with SLSQP
    x0 = initial_h.copy()
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tol, 'maxiter': max_iter})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)

    # Multi-start optimization with perturbations
    def optimize_with_perturbations(x0, n_perturb, n_iter, tol):
        best_h = x0.copy()
        best_c5 = compute_overlap(best_h)
        for _ in range(n_perturb):
            # Perturb the current solution
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            x_perturb = np.clip(best_h + perturb, 0, 1)
            # Local optimization
            result = minimize(objective, x_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tol, 'maxiter': n_iter})
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        return best_h, best_c5

    best_h, best_c5 = optimize_with_perturbations(best_h, n_perturb, max_iter, tol)

    return (best_h.tolist(), best_c5, n_points)
