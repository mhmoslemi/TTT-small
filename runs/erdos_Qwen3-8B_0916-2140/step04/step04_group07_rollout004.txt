import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define initial grid points and start with a coarse grid
    n_points = 100
    dx = 2.0 / n_points
    x_vals = np.linspace(0, 2, n_points)
    
    # Initialize with a simple initial construction
    initial_h = np.zeros(n_points)
    initial_h[:n_points//2] = 1.0
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Perform optimization with initial guess
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try several perturbations to avoid local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Try a finer grid with more points (n=200) to find a better solution
    n_points = 200
    dx = 2.0 / n_points
    x_vals = np.linspace(0, 2, n_points)
    
    # Interpolate best_h to the new grid
    initial_h_new = np.interp(x_vals, np.linspace(0, 2, len(best_h)), best_h)
    initial_h_new = np.clip(initial_h_new, 0, 1)
    
    def compute_overlap_new(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective_new(x):
        return compute_overlap_new(x)
    
    def constraint_sum_new(x):
        return np.sum(x) - n_points / 2
    
    bounds_new = [(0, 1) for _ in range(n_points)]
    cons_new = ({'type': 'eq', 'fun': constraint_sum_new})
    
    result_new = minimize(objective_new, initial_h_new, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h_new = result_new.x
    best_c5 = compute_overlap_new(best_h_new)
    
    # Final refinement with a different initial guess
    initial_h_refine = np.zeros(n_points)
    initial_h_refine[:n_points//3] = 1.0
    initial_h_refine[n_points//3:n_points//2] = 0.5
    initial_h_refine = np.clip(initial_h_refine, 0, 1)
    
    result_refine = minimize(objective_new, initial_h_refine, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h_refine = result_refine.x
    best_c5_refine = compute_overlap_new(best_h_refine)
    
    if best_c5_refine < best_c5:
        best_h_new = best_h_refine
        best_c5 = best_c5_refine
    
    return (best_h_new.tolist(), best_c5, n_points)
