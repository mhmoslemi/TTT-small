import scipy.optimize as opt
import time
from scipy.optimize import Bounds

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Optimizer settings
    method = 'L-BFGS-B'
    ftol = 1e-10
    maxiter = 150
    perturbation_factor = 0.05
    
    # Optimization function
    def optimize_h(initial_guess):
        result = opt.minimize(
            compute_overlap,
            initial_guess,
            bounds=Bounds(0, 1),
            method=method,
            options={'ftol': ftol, 'maxiter': maxiter}
        )
        return result.x
    
    # Initial optimization
    best_h = optimize_h(initial_h)
    best_c5 = compute_overlap(best_h)
    
    # Local search with different perturbation strategies
    elapsed_time = 0
    start_time = time.time()
    
    # Perturbation strategy 1: Random perturbations
    for _ in range(10):
        perturbations = np.random.normal(0, perturbation_factor, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        current_h = optimize_h(perturbed_h)
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
