import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):

    # Initial grid points

    dx = 2.0 / n_points

    # Initial construction from initial_h_values

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Optimization function
    def optimize_h(initial_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        result = opt.minimize(
            fun=compute_overlap,
            x0=initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        return result.x
    
    # Local search with perturbations
    def local_search(h, iterations=5, perturbation=0.05):
        best_h = h
        best_c5 = compute_overlap(h)
        for _ in range(iterations):
            perturbations = np.random.normal(0, perturbation, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            current_h = optimize_h(perturbed_h)
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Global search with random starting points
    def global_search(num_starts=5):
        best_h = np.random.rand(n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = np.random.rand(n_points) * 2 - 1
        best_h = np.clip(best_h, 0, 1)
        best_c5 = compute_overlap(best_h)
        
        for _ in range(num_starts):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = np.random.rand(n_points) * 2 - 1
            h = np.clip(h, 0, 1)
            current_h = optimize_h(h)
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5

    start_time = time.time()

    best_h = initial_h

    best_h, best_c5 = local_search(best_h)

                if current_c5 < best_c5:

                    best_c5 = current_c5
