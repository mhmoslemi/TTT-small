import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial grid points
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial construction from initial_h_values
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Optimization settings
    def optimize_h(initial_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        result = opt.minimize(
            compute_overlap, 
            initial_guess, 
            bounds=bounds, 
            constraints=cons, 
            method='SLSQP', 
            options={'ftol': 1e-10, 'maxiter': 150, 'disp': False}
        )
        return result.x
    
    # Local search with perturbations
    def perturb_h(h, perturbation_scale=0.05):
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(h + perturbations, 0, 1)
        return perturbed_h
    
    # Main optimization loop
    best_h = optimize_h(initial_h)
    best_c5 = compute_overlap(best_h)
    elapsed_time = 0
    
    # Budget for local search and fine-tuning
    local_search_budget = 0.6 * budget_s
    fine_tuning_budget = 0.4 * budget_s
    
    # Local search with perturbations
    if elapsed_time < local_search_budget:
        for _ in range(10):
            perturbed_h = perturb_h(best_h, 0.05)
            current_h = optimize_h(perturbed_h)
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            elapsed_time += 10
    
    # Fine-tuning with grid search
    if elapsed_time < fine_tuning_budget:
        for i in range(n_points):
            for delta in [-0.05, 0.05]:
                perturbed_h = np.copy(best_h)
                perturbed_h[i] += delta
                perturbed_h = np.clip(perturbed_h, 0, 1)
                current_h = optimize_h(perturbed_h)
                current_c5 = compute_overlap(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            elapsed_time += 5
    
    # Final check
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
