import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial grid points
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial construction from initial_h_values
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Optimization settings
    def optimize_h(initial_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        
        result = opt.minimize(
            compute_overlap,
            initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        return result.x
    
    # Initial optimization
    best_h = optimize_h(initial_h)
    best_c5 = compute_overlap(best_h)
    
    # Local search with perturbations
    if budget_s > 300:
        for _ in range(5):
            perturbations = np.random.normal(0, 0.05, n_points)
            perturbed_h = np.clip(best_h + perturbations, 0, 1)
            current_h = optimize_h(perturbed_h)
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    
    # Fine-tune with gradient descent
    if budget_s > 400:
        def objective(h):
            return compute_overlap(h)
        
        def gradient(h):
            perturb = 1e-6
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = np.copy(h)
                h_plus[i] += perturb
                h_minus = np.copy(h)
                h_minus[i] -= perturb
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * perturb)
            return grad
        
        learning_rate = 0.01
        for _ in range(100):
            grad = gradient(best_h)
            best_h = np.clip(best_h - learning_rate * grad, 0, 1)
            current_c5 = compute_overlap(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
