import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    def initial_h_values(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    n_points_base = 200
    dx_base = 2.0 / n_points_base
    
    # Initial construction
    h_initial = initial_h_values(n_points_base)
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points_base)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    result = minimize(objective, h_initial, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try several random perturbations to escape local optima
    for _ in range(10):
        perturbation = np.random.uniform(-0.05, 0.05, n_points_base)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Try higher resolution
    n_points = 300
    dx = 2.0 / n_points
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_base), best_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    bounds_new = [(0, 1) for _ in range(n_points)]
    cons_new = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})

        h = np.array(x, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective_new(x):
        return compute_overlap_new(x)
    
    result_new = minimize(objective_new, initial_h, bounds=bounds_new, constraints=cons_new, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h_new = result_new.x
    best_c5_new = compute_overlap_new(best_h_new)
    
    # Try even higher resolution
    n_points_high = 400
    dx_high = 2.0 / n_points_high
    initial_h_high = np.interp(np.linspace(0, 2, n_points_high), np.linspace(0, 2, n_points), best_h_new)
    initial_h_high = np.clip(initial_h_high, 0, 1)
    
    bounds_high = [(0, 1) for _ in range(n_points_high)]
    cons_high = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points_high / 2})

        h = np.array(x, dtype=np.float64)

    best_c5_high = compute_overlap_high(best_h_high)
