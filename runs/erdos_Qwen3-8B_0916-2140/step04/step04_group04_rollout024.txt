            # Perturb the current best solution
            perturb = np.random.normal(0, 0.05, n_points)

            # Starting point for optimization
            x_start = x_perturb.copy()
            
            # Set up optimization
            result = minimize(
                objective,
                x_start,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'ftol': tol, 'maxiter': n_iters, 'disp': False}
            )

            current_c5 = compute_overlap(result.x)
