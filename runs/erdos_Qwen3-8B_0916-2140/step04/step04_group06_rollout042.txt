    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Try multiple perturbations to escape local optima
    for _ in range(15):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)

        result = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)

    # Try higher resolution with adaptive grid refinement
    n_points = 400
