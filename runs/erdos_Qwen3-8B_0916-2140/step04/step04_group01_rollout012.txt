    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Define a method to perturb the current guess
    def perturb_guess(guess, perturbation_scale=0.05):
        perturbation = np.random.normal(0, perturbation_scale, n_points)
        new_guess = np.clip(guess + perturbation, 0, 1)
        return new_guess
    
    # Define a method to find a better guess
    def find_better_guess(guess):
        perturbations = np.random.normal(0, 0.05, n_points)
        new_guess = np.clip(guess + perturbations, 0, 1)
        result = opt.minimize(objective, new_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x
    
    # Start with initial guess
    best_h = initial_h
    best_c5 = compute_overlap(initial_h)
    
    # Perform multiple local searches
    num_searches = 10
    search_time = 0
    start_time = time.time()
    
    for _ in range(num_searches):
        if search_time >= budget_s - 100:
            break
        current_guess = find_better_guess(best_h)
        current_c5 = compute_overlap(current_guess)
        if current_c5 < best_c5:
            best_h = current_guess
            best_c5 = current_c5
        search_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
