    def evaluate_c5(h, n_points):
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Function to generate initial samples
    def generate_initial_samples(n_points, seed):
        np.random.seed(seed)

        for i in range(n_points):
            h[i] = np.random.rand()
        # Normalize to have integral 1
        h = h / (np.sum(h) / n_points)
        return h
    
    # Function to perform coordinate descent
    def coordinate_descent(h, n_points, max_iter=1000, tolerance=1e-5):
        h = h.copy()
        dx = 2.0 / n_points
        for _ in range(max_iter):
            for i in range(n_points):
                # Compute current value
                val = h[i]
                # Compute the effect of setting h[i] to 0 and 1
                h0 = h.copy()
                h0[i] = 0.0
                c0 = evaluate_c5(h0, n_points)
                
                h1 = h.copy()
                h1[i] = 1.0
                c1 = evaluate_c5(h1, n_points)
                
                # Choose the value that minimizes the overlap
                if c0 < c1:
                    h[i] = 0.0
                elif c1 < c0:
                    h[i] = 1.0
                else:
                    # Keep the value if no improvement
                    pass
            # Check for convergence
            if np.abs(evaluate_c5(h, n_points) - best_c5) < tolerance:
                break
        return h
    
    # Function to perform local search
    def local_search(h, n_points, max_steps=100, step_size=0.1):
        h = h.copy()

        for _ in range(max_steps):
            # Compute current value
            current_c5 = evaluate_c5(h, n_points)

            i = np.random.randint(0, n_points)

                h = h_trial
