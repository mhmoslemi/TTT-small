import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize_scalar

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 45
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    
    # Create a more diverse initial guess that alternates between high and low values
    for i in range(n_points):
        if i % 3 == 0:
            initial_h_values[i] = 1.0
        elif i % 3 == 1:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.0
    
    # Ensure the initial guess satisfies the sum constraint
    total_sum = np.sum(initial_h_values)
    scaling_factor = (n_points / 2) / total_sum
    initial_h_values = initial_h_values * scaling_factor
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h_values.copy()
    
    # Define a wrapper to perform the optimization
    def optimize_solution(x0):
        result = minimize_scalar(objective, bounds=(0, 1), method='bounded', x0=x0)
        return result
    
    # Use a more sophisticated approach with multiple starting points
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Try multiple starting points with slight perturbations
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        x0 = initial_h_values + perturbation
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (n_points / 2) / np.sum(x0)
        
        result = optimize.minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum})
        
        if result.success:
            curr_c5 = result.fun
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = result.x
    
    # Return the best solution found
    return best_h, best_c5, n_points
