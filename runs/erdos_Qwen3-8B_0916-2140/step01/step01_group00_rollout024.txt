```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=70):
    # Initial construction (sample with 70 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 150  # Use more points for better precision
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraints(h):
        # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
        return np.array([np.sum(h) - n_points / 2])

    # Define bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization setup with multiple restarts for better exploration
    best_c5 = float('inf')
    best_h = initial_h
    start_time = time.time()

    for _ in range(3):
        # Start with a slightly perturbed initial guess
        perturbed_h = initial_h + np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        
        result = minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 10:
            break

    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
```
