from time import time
from concurrent.futures import ThreadPoolExecutor

    if n_points < 100:
        n_points = 100

    # Define objective function for optimization
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points]
        h = h / np.sum(h) * n_points / 2
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * (2.0 / n_points)
        return c5

    # Constraint: sum of h values must equal n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h value
    bounds = Bounds(0, 1)

    # Constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0

    start_time = time()

    def run_optimization():
        nonlocal evals
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = objective(best_h)

    # Define a function to evaluate a given h
    def evaluate_h(h):
        return objective(h)

    # Use ThreadPoolExecutor for parallel evaluation
    def parallel_evaluate(h_list):
        with ThreadPoolExecutor() as executor:
            results = list(executor.map(evaluate_h, h_list))
        return results

    # Generate multiple initial guesses around the current best
    def perturb_h(h, perturbation_factor=0.1):
        h = h.copy()
        for i in range(len(h)):
            h[i] += np.random.uniform(-perturbation_factor, perturbation_factor)
            h[i] = np.clip(h[i], 0, 1)
        return h

    # Create a list of perturbed initial guesses
    num_perturbations = 10
    perturbed_guesses = [perturb_h(best_h) for _ in range(num_perturbations)]

    # Evaluate all perturbed guesses in parallel
    perturbed_results = parallel_evaluate(perturbed_guesses)

    # Find the best among the perturbed guesses
    for i in range(num_perturbations):
        if perturbed_results[i] < best_c5:
            best_h = perturbed_guesses[i]
            best_c5 = perturbed_results[i]

    # Final evaluation
    final_c5 = best_c5
    final_h = best_h

    for _ in range(5):
        if time() - start_time > budget_s - 1:

        new_h = run_optimization()
        new_c5 = objective(new_h)
        if new_c5 < best_c5:

            best_c5 = new_c5
