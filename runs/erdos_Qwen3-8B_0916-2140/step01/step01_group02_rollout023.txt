import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points to find best balance between accuracy and speed
    n_points_list = [100, 200, 400]
    
    best_c5 = float('inf')
    best_h = np.array([])
    best_n = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        
        # Use initial guess from provided initial_h_values if available
        if hasattr(globals(), 'initial_h_values'):
            initial_h = initial_h_values
            # Ensure initial_h has the same length as n_points
            if len(initial_h) != n_points:
                initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
            h_values = initial_h[:n_points]
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        # Define objective function
        def objective(x):
            h = x
            # Compute correlation with 1 - h
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        # Define constraint
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization with more iterations and tighter tolerances
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 200, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        current_h = result.x
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n = n_points
    
    return best_h, best_c5, best_n
