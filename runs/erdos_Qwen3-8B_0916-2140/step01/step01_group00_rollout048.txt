import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=70):
    # Initial construction (sample with 70 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a smaller number of points to allow more aggressive optimization
    n_points = 50
    dx = 2.0 / n_points
    
    # Initial construction with fewer points
    initial_h = initial_h_values(n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])
    
    # Define bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization setup
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 200}
    )
    
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # Try with a different initial guess
        initial_h = np.random.rand(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 200}
        )
        
        if result.success:
            best_h = result.x
            c5 = evaluate_erdos_solution(best_h)
            return (best_h.tolist(), c5, n_points)
        else:
            # Fallback to initial_h_values with n_points=50
            c5 = evaluate_erdos_solution(initial_h)
            return (initial_h.tolist(), c5, n_points)
