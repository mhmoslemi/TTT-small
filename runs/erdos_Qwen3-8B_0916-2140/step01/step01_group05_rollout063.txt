import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize, DifferentialEvolution, TNC
from scipy.optimize.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different numbers of points
    n_points_list = [20, 30, 40, 50, 60, 70, 80, 90, 100]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    # Initial construction from initial_h_values (n=45)
    initial_h_values = np.array([0.0] * 45)
    # Example initial_h_values (replace with actual values if provided)
    # For demonstration, let's create a simple initial guess
    for i in range(45):
        initial_h_values[i] = 0.5 if i % 3 == 0 else 0.0
    initial_h_values = initial_h_values / (np.sum(initial_h_values) / 45)
    
    # Evaluate initial guess
    dx = 2.0 / 45
    corr = np.correlate(initial_h_values, 1 - initial_h_values, mode="full")
    c5_initial = np.max(corr * dx)
    best_c5 = c5_initial
    best_h = initial_h_values
    best_n_points = 45
    
    # Explore different n_points
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Create initial guess for this n_points
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 0.5 if i % 3 == 0 else 0.0
        initial_h = initial_h / (np.sum(initial_h) / n_points)
        
        # Define the objective function to minimize
        def objective(h):
            h = h.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        # Define the constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        # Use differential evolution for global optimization
        result = minimize(
            objective,
            x0=initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
    
    return best_h, best_c5, best_n_points
