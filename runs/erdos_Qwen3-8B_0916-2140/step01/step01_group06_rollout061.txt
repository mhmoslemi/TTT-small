import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Increasing to 200 for better resolution
    dx = 2.0 / n_points
    # Start with a simple step function
    initial_h_values = np.zeros(n_points)
    initial_h_values[:n_points//2] = 1.0
    initial_h_values[n_points//2:] = 0.0
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        # Compute the correlation with 1 - h
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    x0 = initial_h_values.copy()

    # Perform a local optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Try a few more local optimizations with different initializations to avoid local optima
    for _ in range(3):
        perturbation = np.random.normal(0, 0.05, n_points)
        x0 = np.clip(best_h + perturbation, 0, 1)
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200})
        best_h = result.x
        best_c5 = compute_overlap(best_h)

    return (best_h.tolist(), best_c5, n_points)
