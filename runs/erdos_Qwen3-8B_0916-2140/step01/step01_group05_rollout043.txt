    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Define a more sophisticated optimization routine
    def optimize_h(h_initial):
        result = minimize(
            fun=objective,
            x0=h_initial,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
        )
        return result
    
    # Perform a local optimization
    result = optimize_h(x0)
    best_h = result.x
    best_c5 = result.fun
