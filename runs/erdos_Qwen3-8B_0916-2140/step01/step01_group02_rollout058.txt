```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points for better performance
    n_points_list = [200, 300, 400]
    
    best_c5 = float('inf')
    best_h = np.zeros(1)
    best_n = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        
        # Start with a reasonable initial guess based on known constructions
        if n_points == 200:
            for i in range(n_points):
                if i % 5 == 0:
                    h_values[i] = 1.0
                elif i % 5 == 1:
                    h_values[i] = 0.5
                elif i % 5 == 2:
                    h_values[i] = 0.0
                elif i % 5 == 3:
                    h_values[i] = 0.5
                elif i % 5 == 4:
                    h_values[i] = 1.0
        elif n_points == 300:
            for i in range(n_points):
                if i % 6 == 0:
                    h_values[i] = 1.0
                elif i % 6 == 1:
                    h_values[i] = 0.5
                elif i % 6 == 2:
                    h_values[i] = 0.0
                elif i % 6 == 3:
                    h_values[i] = 0.5
                elif i % 6 == 4:
                    h_values[i] = 1.0
                elif i % 6 == 5:
                    h_values[i] = 0.0
        elif n_points == 400:
            for i in range(n_points):
                if i % 7 == 0:
                    h_values[i] = 1.0
                elif i % 7 == 1:
                    h_values[i] = 0.5
                elif i % 7 == 2:
                    h_values[i] = 0.0
                elif i % 7 == 3:
                    h_values[i] = 0.5
                elif i % 7 == 4:
                    h_values[i] = 1.0
                elif i % 7 == 5:
                    h_values[i] = 0.0
                elif i % 7 == 6:
                    h_values[i] = 0.5
        
        h_values = np.clip(h_values, 0, 1)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        start_time = time.time()
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        elapsed_time = time.time() - start_time
        
        if elapsed_time > budget_s:
            continue
        
        current_h = result.x
        current_c5 = objective(current_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n = n_points
    
    return best_h, best_c5, best_n
```
