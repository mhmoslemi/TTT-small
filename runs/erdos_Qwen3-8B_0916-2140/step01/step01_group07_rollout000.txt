    def compute_c5(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)

        c5 = np.max(result) * dx
        return c5

    # Optimization function that minimizes the maximum overlap
    def objective(h):
        return compute_c5(h)

    # Constraint function for the integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds for the function values
    bounds = Bounds(0, 1)

    # Constraint for the integral
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0

    start_time = time()

    # Function to perform optimization
    def run_optimization():
        nonlocal evals
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x
