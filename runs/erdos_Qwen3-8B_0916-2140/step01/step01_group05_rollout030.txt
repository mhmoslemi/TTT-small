import numpy as np

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    # Set the random seed for reproducibility
    np.random.seed(seed)
    
    # Start with a base number of points
    n_points = 120  # Start with more points to allow finer adjustments
    dx = 2.0 / n_points
    
    # Initial guess based on some structured pattern (like alternating values)
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            initial_h[i] = 0.5
        elif i % 3 == 1:
            initial_h[i] = 0.75
        else:
            initial_h[i] = 0.25
    
    # Normalize to ensure the integral is 1
    initial_h = initial_h / (np.sum(initial_h) / n_points)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint for the integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for h
    bounds = Bounds(0, 1)
    
    # Set up the optimization problem
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP"
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
