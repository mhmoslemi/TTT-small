import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a higher number of points for better resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial h_values from the given initial_h_values
    # Initialize h with some periodic structure to reduce overlaps
    initial_h = np.array([0.0] * n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    initial_h = initial_h / (np.sum(initial_h) / n_points)  # Ensure sum is n_points / 2
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Optimization with more advanced settings
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method="L-BFGS-B", options={"ftol": 1e-8, "maxiter": 500})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
