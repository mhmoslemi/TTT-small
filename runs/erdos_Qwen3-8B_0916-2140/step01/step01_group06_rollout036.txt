```python
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse.linalg import lsqr
from functools import partial
import random

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points for better accuracy
    dx = 2.0 / n_points

    # Define initial guess based on a simple piecewise function
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            initial_h[i] = 1.0
        elif i < 2 * n_points // 3:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    x0 = initial_h.copy()

    # Perform a local optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Try a few random perturbations to escape local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_c5 = compute_overlap(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5

    return (best_h.tolist(), best_c5, n_points)
```
