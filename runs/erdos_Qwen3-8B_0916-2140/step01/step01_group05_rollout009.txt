import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, DifferentialEvolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    n_points_options = [100, 150, 200]
    best_c5 = float('inf')
    best_h = np.array([])
    best_n = 0
    
    for n_points in n_points_options:
        dx = 2.0 / n_points
        
        # Initial guess based on a simple construction
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0:
                initial_h[i] = 1.0
            else:
                initial_h[i] = 0.0
        
        # Normalize to ensure the integral is 1
        initial_h = initial_h / (np.sum(initial_h) / n_points)
        
        # Define the objective function to minimize
        def objective(h):
            h = h.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        # Define the constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        # Initial guess
        x0 = initial_h.copy()
        
        # Set up the optimization problem
        result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', tol=1e-6)
        
        # Check if the result is valid and better than the current best
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            best_n = n_points
        
        # Also try a different starting point using a random construction
        random_initial_h = np.random.rand(n_points)
        random_initial_h = random_initial_h / (np.sum(random_initial_h) / n_points)
        
        result = minimize(objective, random_initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', tol=1e-6)
        
        if result.success and result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x
            best_n = n_points
    
    # If no valid result found, return the initial guess
    if best_h.size == 0:
        best_h = np.array([0.0] * 100)
        best_h = best_h / (np.sum(best_h) / 100)
        best_c5 = 0.5
    
    return best_h, best_c5, best_n
