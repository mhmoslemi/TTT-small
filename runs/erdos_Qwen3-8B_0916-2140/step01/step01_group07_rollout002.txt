    def evaluate_overlap(h):

        # Compute the correlation

        max_overlap = np.max(result) * dx

    def objective(h):
        return evaluate_overlap(h)

    bounds = Bounds(0, 1)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    max_iter = 100
    tol = 1e-6

    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization function
    def optimize_once(h):
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    # Run optimization with multiple restarts
    start_time = time()
