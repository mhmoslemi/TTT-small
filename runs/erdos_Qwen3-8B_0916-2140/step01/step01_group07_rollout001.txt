```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
from scipy.optimize import Bounds, LinearConstraint, minimize

def run(seed=42, budget_s=600):
    # Initial construction
    initial_h_values = np.load('initial_h_values.npy')
    n_initial = len(initial_h_values)
    n_points = 200
    dx = 2.0 / n_points

    # Define the initial guess
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_initial), initial_h_values)

    # Define the objective function
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points]
        h = h / np.sum(h) * n_points / 2
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.all(0 <= h) and np.all(h <= 1)

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0

    start_time = time.time()

    # Define the optimization function
    def optimize():
        nonlocal evals
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'eps': 1e-8, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = objective(best_h)

    # Run multiple optimizations with different starting points
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        new_h = optimize()
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Final optimization to refine the solution
    if time.time() - start_time < budget_s - 1:
        final_h = best_h.copy()
        final_h = final_h / np.sum(final_h) * n_points / 2
        final_c5 = objective(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
```
