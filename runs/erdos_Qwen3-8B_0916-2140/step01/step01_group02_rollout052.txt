```python
import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    
    # Initial guess: more refined piecewise constant function with 10 intervals
    for i in range(n_points):
        if i % 10 == 0:
            h_values[i] = 1.0
        elif i % 10 == 1:
            h_values[i] = 0.75
        elif i % 10 == 2:
            h_values[i] = 0.5
        elif i % 10 == 3:
            h_values[i] = 0.25
        elif i % 10 == 4:
            h_values[i] = 0.0
        elif i % 10 == 5:
            h_values[i] = 0.25
        elif i % 10 == 6:
            h_values[i] = 0.5
        elif i % 10 == 7:
            h_values[i] = 0.75
        elif i % 10 == 8:
            h_values[i] = 1.0
        elif i % 10 == 9:
            h_values[i] = 0.75
    h_values = np.clip(h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with multiple restarts for better exploration
    best_c5 = float('inf')
    best_h = h_values
    
    for _ in range(3):
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
            h_values = result.x
        else:
            print("Optimization failed, using previous best")
            break
    
    return best_h, best_c5, n_points
```
