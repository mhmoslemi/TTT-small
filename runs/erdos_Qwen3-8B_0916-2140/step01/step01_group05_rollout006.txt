import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, DifferentialEvolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Define a better initial guess by using a wave-like pattern
    initial_h = np.sin(np.linspace(0, np.pi, n_points))
    initial_h = (initial_h - np.min(initial_h)) / (np.max(initial_h) - np.min(initial_h))
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Use differential evolution for global optimization
    result = minimize(
        objective, 
        x0=initial_h, 
        bounds=bounds, 
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP"
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
