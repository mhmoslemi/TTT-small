import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.3
        elif i < 40:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between performance and precision
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraints(h):
        # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
        return np.array([np.sum(h) - n_points / 2])

    # Define bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Run optimization with multiple initial guesses
    best_result = None
    best_c5 = float('inf')

    # Try multiple initializations to avoid local optima
    for _ in range(3):
        # Generate a new random initial guess
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x)
            if current_c5 < best_c5:
                best_result = result.x
                best_c5 = current_c5
    
    if best_result is not None:
        return (best_result.tolist(), best_c5, n_points)
    else:
        # If optimization fails, use initial_h and compute manually
        c5 = evaluate_erdos_solution(initial_h)
        return (initial_h.tolist(), c5, n_points)
