    def make_alternative_initials(n_points):

        return initial_h

    # Objective function to minimize the max overlap
    def objective(h):
        h = np.array(h)

        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)

    max_iter = 100
    tol = 1e-6

    def run_optimization(initial_guess):

    for initial in alternative_initials:
