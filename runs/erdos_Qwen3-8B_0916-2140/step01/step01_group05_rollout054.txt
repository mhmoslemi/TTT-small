import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a higher number of points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial guess using a periodic construction
    def initial_guess(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            # Define a periodic piecewise function with alternating flat regions and peaks
            if 0 <= x < 0.5:
                h[i] = 0.5
            elif 0.5 <= x < 1.0:
                h[i] = 1.0
            elif 1.0 <= x < 1.5:
                h[i] = 0.5
            elif 1.5 <= x < 2.0:
                h[i] = 0.0
        # Normalize to ensure the integral is 1
        total = np.sum(h) * dx
        h = h / (total / dx)
        return h
    
    initial_h = initial_guess(n_points)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Set up the optimization problem
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP')
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
