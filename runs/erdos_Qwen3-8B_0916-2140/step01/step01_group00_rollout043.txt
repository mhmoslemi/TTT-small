```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 30:
            h[i] = 0.3
        elif i < 60:
            h[i] = 0.6
        else:
            h[i] = 0.9
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between performance and precision
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    # Function to evaluate the overlap integral
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Function to enforce the constraint sum(h) == n_points / 2
    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])

    # Bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimization with SLSQP
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    # Check if optimization was successful
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # If optimization fails, try a different starting point, e.g., based on clustering
        h = initial_h.copy()
        # Apply PCA to the initial_h to get a different direction
        pca = PCA(n_components=1)
        pca.fit(h.reshape(-1, 1))
        principal_component = pca.transform(h.reshape(-1, 1))
        # Use the principal component as a new starting point
        h = np.clip(principal_component + np.random.uniform(-0.5, 0.5, size=n_points), 0, 1)
        h = h.reshape(n_points)
        # Enforce the constraint: sum(h) == n_points / 2
        h = h / np.sum(h) * n_points / 2
        # Re-evaluate
        c5 = evaluate_erdos_solution(h)
        return (h.tolist(), c5, n_points)
```
