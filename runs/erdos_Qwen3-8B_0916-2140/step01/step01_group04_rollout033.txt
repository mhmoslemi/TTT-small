```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                0.5, 0.5, 0.5, 0.5, 0.5], dtype=np.float64)
    
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    def objective(h, n_points, dx, k_values):
        overlap = 0
        for k in k_values:
            integral = np.sum(h[:n_points - k] * (1 - h[k:]))
            overlap = max(overlap, integral)
        return overlap * dx
    
    def optimize_step_function(n_points, initial_guess=None):
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = lambda h: np.sum(h) - n_points / 2
        h0 = initial_guess if initial_guess is not None else np.full(n_points, 0.5)
        
        result = opt.minimize(
            lambda h: objective(h, n_points, 2.0 / n_points, range(1, n_points)),
            h0,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    best_h = initial_h_values
    best_c5 = compute_overlap(initial_h_values)
    best_n_points = len(initial_h_values)
    
    def generate_k_values(n_points):
        return [k for k in range(1, n_points) if k <= 10]
    
    start_time = time.time()
    
    n_points_values = [45, 90, 135, 180, 225]
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            h_opt, c5_opt = optimize_step_function(n_points, initial_guess=initial_h_values)
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
```
