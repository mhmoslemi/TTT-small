```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    def optimize_h(n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        result = minimize(objective, x0=initial_h_values[:n_points], bounds=bounds, constraints=[constraint])
        return result.x, result.fun
    
    best_h = initial_h_values
    best_c5 = compute_overlap(initial_h_values)
    best_n_points = len(initial_h_values)
    
    n_points_values = [45, 90, 135]
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        try:
            h_opt, c5_opt = optimize_h(n_points)
            print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
```
