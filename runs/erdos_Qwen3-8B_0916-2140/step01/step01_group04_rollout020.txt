    def compute_overlap(h, n_points):

        return np.max(overlaps) * dx

    def optimize_step_function(n_points, initial_guess=None):
        # Define the step function as a vector h of size n_points
        # with h[i] in [0, 1] and sum(h) = n_points / 2
        
        # Define bounds (0 <= h[i] <= 1) for all i
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: maximize the overlap integral
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        # Constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Perform optimization with different strategies to avoid local optima
        if initial_guess is None:
            initial_guess = np.full(n_points, 0.5)
        
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}], method='SLSQP')
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    # Strategy: Start with a few different initial points and try to refine
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3]

        perturbed = initial_guess.copy()
        for i in range(len(perturbed)):
            perturbed[i] += np.random.uniform(-perturbation_factor, perturbation_factor)
            perturbed[i] = np.clip(perturbed[i], 0, 1)
