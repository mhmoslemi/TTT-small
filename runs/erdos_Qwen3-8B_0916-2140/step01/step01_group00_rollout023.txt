```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Start with 100 points for balance between performance and precision
    n_points = 100
    dx = 2.0 / n_points
    
    # Define initial construction
    initial_h = initial_h_values(n_points)
    
    # Define a more aggressive starting point with gradient descent
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraints(h):
        # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
        return np.array([np.sum(h) - n_points / 2])

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # First optimization: use SLSQP with more iterations
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 200}
    )

    # If successful, use this result as a starting point for a second optimization
    if result.success:
        best_h = result.x
        # Use best_h as a starting point for a second optimization with more aggressive parameters
        result2 = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="L-BFGS-B",
            options={"maxiter": 100, "ftol": 1e-8, "gtol": 1e-8}
        )
        # Choose the better of the two results
        if result2.success:
            best_h = result2.x
        else:
            best_h = result.x
    else:
        best_h = initial_h

    c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), c5, n_points)
```
