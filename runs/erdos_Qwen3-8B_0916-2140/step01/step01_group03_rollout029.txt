import scipy.optimize as optimize
import time

    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()

    for _ in range(5):

        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand - (np.sum(h_rand) - n_points / 2) / n_points
        
        # Use scipy's optimization to find a better solution
        result = optimize.minimize(
            objective,
            h_rand,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tolerance, 'maxiter': 100}
        )
