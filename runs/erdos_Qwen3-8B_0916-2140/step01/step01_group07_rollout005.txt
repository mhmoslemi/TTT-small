from time import time

def run(seed=42, budget_s=600):
    # Define initial solution with n_points = 100
    n_points = 100

    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2

    # Define the objective function
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / np.sum(h) * n_points / 2
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Function to optimize
    def optimize_with_bounds(h_initial):
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6, 'disp': False}
        )
        return result.x

    # Add a few different starting points
    starting_points = [initial_h]
    for i in range(5):
        starting_points.append(initial_h + np.random.uniform(-0.1, 0.1, n_points))
        starting_points.append(initial_h * np.random.uniform(0.5, 1.5, n_points))
        starting_points.append(initial_h * np.random.uniform(0.9, 1.1, n_points))
    
    best_h = initial_h
    best_c5 = objective(initial_h)

    start_time = time()

    for h_initial in starting_points:
        if time() - start_time > budget_s - 1:
            break
        h_initial = h_initial / np.sum(h_initial) * n_points / 2
        new_h = optimize_with_bounds(h_initial)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    return (best_h, best_c5, len(best_h))
