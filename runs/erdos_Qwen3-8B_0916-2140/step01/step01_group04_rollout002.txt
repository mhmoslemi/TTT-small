    def compute_c5(h):

        return np.max(overlaps) * dx

    def optimize_step_function(n_points, initial_guess=None):
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        if initial_guess is None:
            initial_guess = np.full(n_points, 0.5)

        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")

    def perturb_solution(h, perturb_strength=0.1):
        perturbed = h.copy()
        delta = np.random.normal(0, perturb_strength, size=len(h))
        perturbed += delta
        perturbed = np.clip(perturbed, 0, 1)
        return perturbed
    
    # Function to perform simulated annealing
    def simulated_annealing(n_points, initial_h, max_iterations=100, temp=1.0, cooling_rate=0.995):
        current_h = initial_h.copy()
        current_c5 = compute_c5(current_h)

        best_c5 = current_c5
        iterations = 0

            perturbed_h = perturb_solution(current_h, perturb_strength=0.1)

                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
            
            temp *= cooling_rate
            iterations += 1
        
        return best_h, best_c5
