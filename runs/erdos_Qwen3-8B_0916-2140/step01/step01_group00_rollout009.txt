```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import Bounds
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.3
        elif i < 40:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        elif i < 80:
            h[i] = 0.3
        else:
            h[i] = 0.6
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_h = initial_h_values(n_points)
    
    # Optimization setup
    result = minimize(
        evaluate,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 500}
    )
    
    if result.success:
        best_h = result.x
        c5 = evaluate(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # Fallback to initial guess if optimization fails
        c5 = evaluate(initial_h)
        return (initial_h.tolist(), c5, n_points)
```
