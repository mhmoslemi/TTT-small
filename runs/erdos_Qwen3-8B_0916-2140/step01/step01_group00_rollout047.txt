```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=70):
    # Initial construction (sample with 70 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a smaller number of points for faster optimization
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    
    # Generate initial guess from the initial_h_values function
    h_initial = initial_h_values(n_points)
    
    # Define objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define constraints
    def constraint(h):
        return np.sum(h) - n_points / 2  # ensure sum(h) = n_points / 2
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 100
    tol = 1e-8
    time_limit = budget_s
    
    start_time = time.time()
    
    # Run SLSQP optimization
    result = minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": tol, "maxiter": max_iter}
    )
    
    # If optimization succeeds, return the result
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    
    # If optimization fails, fall back to initial guess
    else:
        c5 = evaluate_erdos_solution(h_initial)
        return (h_initial.tolist(), c5, n_points)
```
