import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
import matplotlib.pyplot as plt
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # start with 100 points, can scale up if needed
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    local_opt = minimize(objective, initial_h_values, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = local_opt.x
    best_c5 = compute_overlap(best_h)
    
    # Try a global optimization starting from multiple points
    def global_optimize():
        def objective_global(x):
            return compute_overlap(x)
        
        def constraint_global(x):
            return np.sum(x) - n_points / 2
        
        bounds_global = [(0, 1) for _ in range(n_points)]
        cons_global = ({'type': 'eq', 'fun': constraint_global})
        
        def fitness(x):
            return compute_overlap(x)
        
        result = differential_evolution(fitness, bounds_global, constraints=cons_global, strategy='best1bin', popsize=10, maxiter=50, tol=1e-8, mutation=(0.5, 1), recombination=0.9)
        best_global = result.x
        best_c5_global = compute_overlap(best_global)
        return best_global, best_c5_global
    
    global_best_h, global_best_c5 = global_optimize()
    
    if global_best_c5 < best_c5:
        best_h = global_best_h
        best_c5 = global_best_c5
    
    # Try a hybrid approach with a more refined search
    def refine_search(x0):
        def objective_refined(x):
            return compute_overlap(x)
        
        def constraint_refined(x):
            return np.sum(x) - n_points / 2
        
        bounds_refined = [(0, 1) for _ in range(n_points)]
        cons_refined = ({'type': 'eq', 'fun': constraint_refined})
        
        result = minimize(objective_refined, x0, bounds=bounds_refined, constraints=cons_refined, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 50})
        return result.x, compute_overlap(result.x)
    
    refined_h, refined_c5 = refine_search(best_h)
    if refined_c5 < best_c5:
        best_h = refined_h
        best_c5 = refined_c5
    
    return (best_h.tolist(), best_c5, n_points)
