import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points values to balance speed and accuracy
    n_points_list = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Initial guess based on a simple periodic construction
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = 0.5 * (1 + np.cos(2 * np.pi * i / n_points))
        initial_h = initial_h / (np.sum(initial_h) / n_points)
        
        # Define the objective function to minimize
        def objective(h):
            h = h.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        # Define the constraints
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Bounds for h
        bounds = Bounds(0, 1)
        
        # Initial guess
        x0 = initial_h.copy()
        
        # Use L-BFGS-B for faster convergence
        result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='L-BFGS-B', tol=1e-5, options={'maxiter': 200})
        
        # Extract the best solution
        current_h = result.x
        current_c5 = result.fun
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n_points = n_points
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
