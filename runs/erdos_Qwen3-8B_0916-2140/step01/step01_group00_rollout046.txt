```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=70):
    # Initial construction (sample with 70 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points for a good balance between computation and precision
    n_points_start = 70
    n_points = n_points_start
    
    # Define initial guess using the initial_h_values function
    initial_h = initial_h_values(n_points)
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization setup
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 150}
    )
    
    # If the optimization is successful, we return the best solution
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # If optimization fails, fall back to the initial guess
        c5 = evaluate_erdos_solution(initial_h)
        return (initial_h.tolist(), c5, n_points)
```
