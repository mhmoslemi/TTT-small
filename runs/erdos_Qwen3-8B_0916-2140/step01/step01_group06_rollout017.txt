import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Start with higher resolution for better exploration
    dx = 2.0 / n_points

    def initial_h_values(x):
        return np.interp(x, [0, 2], [0, 1])  # Placeholder for initial construction

    # Load initial h_values if available
    initial_h = np.array(initial_h_values(np.linspace(0, 2, n_points)), dtype=np.float64)
    initial_h = np.clip(initial_h, 0, 1)

    # Define the objective function to minimize max overlap
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Use a scaled gradient descent with momentum for better convergence
    def gradient_descent_optimizer(initial_guess, max_iter=1000, learning_rate=0.01, momentum=0.9, tolerance=1e-6):
        h = initial_guess.copy()
        h_prev = h.copy()
        h_prev_prev = h.copy()
        for _ in range(max_iter):
            # Compute overlap
            overlap = compute_overlap(h)
            # Compute gradient of overlap with respect to h
            # We'll approximate the gradient numerically
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                overlap_plus = compute_overlap(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                overlap_minus = compute_overlap(h_minus)
                grad[i] = (overlap_plus - overlap_minus) / (2 * 1e-6)
            # Update h using gradient descent with momentum
            h_prev_prev = h_prev
            h_prev = h
            h = h - learning_rate * grad
            # Apply momentum
            h = h_prev + momentum * (h_prev - h_prev_prev)
            # Clip h to [0, 1]
            h = np.clip(h, 0, 1)
            # Check for convergence
            if np.max(np.abs(grad)) < tolerance:
                break
        return h

    # Run gradient descent from the initial guess
    best_h = gradient_descent_optimizer(initial_h.copy())
    best_c5 = compute_overlap(best_h)

    # Optionally perform a local optimization to refine the solution
    def objective(x):
        return compute_overlap(x)

    result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)

    return (best_h.tolist(), best_c5, n_points)
