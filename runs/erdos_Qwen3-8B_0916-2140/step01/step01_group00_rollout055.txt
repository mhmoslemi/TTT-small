```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.5
        elif i < 40:
            h[i] = 0.8
        elif i < 60:
            h[i] = 0.2
        else:
            h[i] = 0.5
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with 100 points
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial_h_values as a starting point
    initial_h = initial_h_values(n_points)
    
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraints(h):
        # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
        return np.array([np.sum(h) - n_points / 2])

    # Define bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # First optimization step with SLSQP
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # Try a grid search with perturbations
        best_c5 = float('inf')
        best_h = initial_h
        for perturbation in [0.01, 0.02, 0.05, 0.1]:
            perturbed_h = np.array([h + np.random.normal(0, perturbation) for h in best_h])
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            
            corr = np.correlate(perturbed_h, 1 - perturbed_h, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = perturbed_h
        return (best_h.tolist(), best_c5, n_points)
```
