```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.3
        elif i < 40:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points values
    n_points_list = [100, 150, 200]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 100
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        initial_h = initial_h_values(n_points)
        
        def objective(h):
            # Compute the overlap integral for all k
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraints(h):
            # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
            return np.array([np.sum(h) - n_points / 2])
        
        # Define bounds for h: each h[i] ∈ [0, 1]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization setup
        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        # Check if we have a valid solution
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n = n_points
        
        # If optimization fails, use initial_h and compute manually
        else:
            current_c5 = evaluate_erdos_solution(initial_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = initial_h
                best_n = n_points
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, best_n)
```
