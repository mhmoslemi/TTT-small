import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a higher number of points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess from a more refined version of the known construction
    # Constructing a better initial guess through symmetry and cluster-based optimization
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5
    
    # Create a more refined initial guess with clusters
    cluster_size = 10
    cluster_centers = np.linspace(0, 2, 10)
    cluster_weights = np.ones(10) / 10.0
    
    for i in range(n_points):
        x = i * dx
        dist = np.abs(cluster_centers - x)
        weights = np.exp(-dist / 0.1)
        weights /= np.sum(weights)
        initial_h[i] = np.dot(weights, cluster_weights)
    
    # Ensure sum is n_points / 2
    initial_h = initial_h / (np.sum(initial_h) / n_points) * (n_points / 2)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Use a more robust optimization algorithm
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method="SLSQP", options={"maxiter": 200})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    return best_h, best_c5, n_points
