```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.3
        elif i < 40:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a lower number of points for faster exploration
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction
    initial_h = initial_h_values(n_points)
    
    # Define bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraints(h):
        # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
        return np.array([np.sum(h) - n_points / 2])
    
    # Optimization setup with SLSQP
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 150}
    )
    
    # If optimization is successful, use the result
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # If optimization fails, try a different initial guess
        # Try a random perturbation of the initial guess
        perturbed_h = initial_h + 0.1 * (np.random.rand(n_points) - 0.5)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        
        result = minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        
        if result.success:
            best_h = result.x
            c5 = evaluate_erdos_solution(best_h)
            return (best_h.tolist(), c5, n_points)
        else:
            # If optimization still fails, return the initial guess
            c5 = evaluate_erdos_solution(initial_h)
            return (initial_h.tolist(), c5, n_points)
```
