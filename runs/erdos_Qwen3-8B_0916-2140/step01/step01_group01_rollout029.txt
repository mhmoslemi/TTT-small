    np.random.seed(seed)
    
    # Start with a smaller n_points to save time
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize the function with a step function
    step_func = np.zeros(n_points)
    step_func[:n_points // 2] = 0.0
    step_func[n_points // 2:] = 1.0
    step_func = step_func - step_func[n_points // 2] / 2
    
    # Define the initial guess as the step function
    initial_guess = step_func
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Optimization function with constraints
    def objective(h):
        return compute_c5(h)
    
    # Constraints: sum of h * dx = 1 --> sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # First, try SLSQP
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 1000, 'eps': 1e-6, 'ftol': 1e-8}
    )
