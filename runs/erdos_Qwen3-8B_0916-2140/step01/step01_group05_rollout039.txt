```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Starting with n_points = 100 for better resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess based on initial_h_values (assuming it's defined in the global scope)
    # Reconstruct initial_h_values for n_points=100, extrapolating or interpolating if necessary
    # For simplicity, we'll use a flat initial guess with a small random perturbation
    initial_h = np.random.rand(n_points)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint for the integral
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-5
    method = 'L-BFGS-B'
    
    # Run optimization
    start_time = time.time()
    result = minimize(objective, x0, method=method, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, tol=tol, maxiter=max_iter)
    
    # Check if we have enough time left to run a local search
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        # Perform a local search with perturbations
        perturbation = 0.05
        perturbed_h = result.x + np.random.uniform(-perturbation, perturbation, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Re-evaluate the perturbed solution
        result = minimize(objective, perturbed_h, method=method, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, tol=tol, maxiter=max_iter)
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
```
