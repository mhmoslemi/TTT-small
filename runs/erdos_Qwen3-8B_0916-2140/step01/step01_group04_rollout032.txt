        return np.max(overlaps) * dx

    print(f"Initial C5 value: {c5_initial:.6f}")
    
    # Step 2: Define a function to optimize the step function h
    def optimize_step_function(n_points, initial_guess=None):
        # Define the step function as a vector h of size n_points
        # with h[i] in [0, 1] and sum(h) = n_points / 2
        
        # Define bounds (0 <= h[i] <= 1) for all i
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: maximize the overlap integral
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        # Constraint: sum(h) == n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        # Constraint: make the function more smooth (optional)
        def constraint_smoothness(h):
            return np.sum(np.abs(np.gradient(h))) - 100
        
        # Set up optimization
        if initial_guess is None:
            initial_guess = np.random.rand(n_points)
            initial_guess = np.clip(initial_guess, 0, 1)
            initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
        
        # Constraints
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        # Optionally add smoothness constraint
        # cons.append({'type': 'ineq', 'fun': lambda h: -constraint_smoothness(h)})
        
        # Perform optimization
        result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='L-BFGS-B')
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    # Step 3: Try different n_points (starting with initial n_points_initial)
    best_h = h_initial
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            h_opt, c5_opt = optimize_step_function(n_points)
