import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with 45 points
    n_points_initial = 45
    initial_h_values = np.array([0.0] * n_points_initial)
    for i in range(n_points_initial):
        initial_h_values[i] = 1.0 if i < n_points_initial // 2 else 0.0
    h_initial = initial_h_values.copy()
    
    # Step 1: Start with initial construction and evaluate it
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    c5_initial = compute_overlap(h_initial)
    print(f"Initial C5 value: {c5_initial:.6f}")
    
    # Define a function to optimize the step function h
    def optimize_step_function(n_points, initial_guess=None):
        # Define bounds (0 <= h[i] <= 1) for all i
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: maximize the overlap integral
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        # Constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Initial guess
        if initial_guess is None:
            initial_guess = np.array([0.5] * n_points)
        
        # Perform optimization
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    # Step 2: Try different n_points (starting with initial n_points_initial)
    best_h = h_initial
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            # Start with a guess based on the initial construction
            initial_guess = np.array([0.5] * n_points)
            
            # If n_points is a multiple of initial n_points_initial, use the initial construction as a guess
            if n_points % n_points_initial == 0:
                scale_factor = n_points // n_points_initial
                initial_guess = np.array([0.5] * n_points)
                for i in range(n_points_initial):
                    initial_guess[i * scale_factor : (i + 1) * scale_factor] = h_initial[i]
            
            h_opt, c5_opt = optimize_step_function(n_points, initial_guess)
            print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
