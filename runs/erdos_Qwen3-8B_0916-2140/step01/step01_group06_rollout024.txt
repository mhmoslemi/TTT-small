import numpy as np

from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Starting with 200 points for better accuracy
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0  # Example initial values
                                for _ in range(n_points)])
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        h_reshaped = np.reshape(h, (1, n_points))
        k_values = np.arange(-n_points + 1, n_points)
        corr = np.zeros(len(k_values))
        for i, k in enumerate(k_values):
            if k < 0:
                shifted = np.roll(h, k)
            else:
                shifted = np.roll(h, k)
            corr[i] = np.sum(h * (1 - shifted)) * dx
        return np.max(corr)
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    
    x0 = initial_h.copy()
    
    # Perform a local optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Optionally scale up to more points if needed
    # We'll keep it at 200 for performance
    
    return (best_h.tolist(), best_c5, n_points)
