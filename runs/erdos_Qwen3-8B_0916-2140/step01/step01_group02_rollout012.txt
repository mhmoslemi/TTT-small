import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Increased resolution for better optimization
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    
    # Start with a more refined initial guess: piecewise constant with 5 intervals
    for i in range(n_points):
        if i % 5 == 0:
            h_values[i] = 1.0
        elif i % 5 == 1:
            h_values[i] = 0.5
        elif i % 5 == 2:
            h_values[i] = 0.0
        elif i % 5 == 3:
            h_values[i] = 0.5
        elif i % 5 == 4:
            h_values[i] = 1.0
    h_values = np.clip(h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with adaptive step size and multiple restarts for better exploration
    best_c5 = float('inf')
    best_h = h_values
    
    # Run optimization with multiple restarts
    for restart in range(3):
        initial_guess = h_values + np.random.uniform(-0.1, 0.1, n_points)
        initial_guess = np.clip(initial_guess, 0, 1)
        initial_guess = initial_guess * (1.0 / (np.sum(initial_guess) * dx))
        
        result = minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8, "disp": False},
        )
        
        current_c5 = objective(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, n_points
