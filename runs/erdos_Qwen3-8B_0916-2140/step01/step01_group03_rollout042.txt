import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to generate random initial points
    def generate_random_initial(n):
        return np.random.uniform(0, 1, n)
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    time_budget = budget_s
    start_time = time.time()
    
    # Use initial guess
    h0 = initial_h
    
    # Initial best solution
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Optimization loop
    for _ in range(max_iter):
        # Use scipy's optimization to find a better solution
        result = optimize.minimize(
            objective,
            h0,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tolerance, 'maxiter': 100}
        )
        
        if result.success:
            current_h = result.x
            current_C5 = compute_C5(current_h)
