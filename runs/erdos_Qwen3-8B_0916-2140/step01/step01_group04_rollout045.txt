    def optimize_step_function(n_points):
        # Define the step function as a vector h of size n_points
        # with h[i] in [0, 1] and sum(h) = n_points / 2

        # Bounds for all h[i] in [0, 1]
        bounds = [(0, 1) for _ in range(n_points)]

        # Objective function: minimize the maximum overlap integral
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx

        # Constraint: sum of h must be n_points / 2
        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        # Optimization using SLSQP
        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints=[{"type": "eq", "fun": constraint_sum}],
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 1000}
        )

        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")

    # Try different n_points values, starting from a smaller set
    n_points_values = [45, 60, 90, 120, 150, 180, 210, 240, 270, 300]

    best_c5 = c5_initial
    best_n_points = n_points_initial
