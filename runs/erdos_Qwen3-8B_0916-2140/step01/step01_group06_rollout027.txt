import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse.linalg import lsqr
from functools import partial
from scipy.optimize import Bounds, minimize, differential_evolution

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # starting with 100 points, can scale up if needed
    dx = 2.0 / n_points

    # Initial guess based on a periodic construction
    def initial_h_values(x):
        x = np.array(x)
        x = np.sort(x)
        return np.sin(np.pi * (x - 0.5)) / 2 + 0.5

    initial_h = np.array(initial_h_values(np.linspace(0, 2, n_points)), dtype=np.float64)
    initial_h = np.clip(initial_h, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    # Use differential evolution for global optimization
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=({'type': 'eq', 'fun': constraint_sum}),
        poltergeist=True,
        maxiter=100,
        popsize=30,
        mutation=(0.5, 1),
        recombination=0.7,
        tol=1e-8,
        disp=False,
        mutation_func=None,
        seed=seed,
        timeout=budget_s / 2
    )

    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Optionally refine with local optimization
    def local_objective(x):
        return compute_overlap(x)

    local_bounds = [(0, 1) for _ in range(n_points)]
    local_result = minimize(
        fun=local_objective,
        x0=best_h,
        bounds=local_bounds,
        constraints=({'type': 'eq', 'fun': constraint_sum}),
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 50}
    )

    best_h = local_result.x
    best_c5 = compute_overlap(best_h)

    return (best_h.tolist(), best_c5, n_points)
