import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with 45 points
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                0.5, 0.5, 0.5], dtype=np.float64)
    
    # Define a function to compute the overlap integral
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    # Define a function to optimize the step function h
    def optimize_step_function(n_points, initial_guess):
        # Define bounds (0 <= h[i] <= 1) for all i
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: minimize the maximum overlap integral
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        # Constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Perform optimization
        result = minimize(objective, initial_guess, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    # Starting with the initial construction
    best_h = initial_h_values
    best_c5 = compute_overlap(best_h)
    best_n_points = len(best_h)
    
    # Try increasing the number of points in steps
    start_time = time.time()
    
    # Initial n_points_sequence: [45, 90, 135, 180]
    n_points_sequence = [45, 90, 135, 180]
    
    for n_points in n_points_sequence:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Get an initial guess by resampling the best_h so far
        if n_points > len(best_h):
            # Resample the current best_h to n_points
            resampling_ratio = n_points / len(best_h)
            resampled_h = np.kaiser(len(best_h), 5) * np.kaiser(len(best_h), 5)
            resampled_h = resampled_h / np.max(resampled_h)
            resampled_h = np.repeat(resampled_h, int(resampling_ratio))
            resampled_h = resampled_h[:n_points]
            resampled_h = np.clip(resampled_h, 0, 1)
            resampled_h = resampled_h * (n_points / 2.0) / np.sum(resampled_h)
        else:
            # Resample the current best_h to n_points
            resampling_ratio = n_points / len(best_h)
            resampled_h = np.kaiser(len(best_h), 5) * np.kaiser(len(best_h), 5)
            resampled_h = resampled_h / np.max(resampled_h)
            resampled_h = np.repeat(resampled_h, int(resampling_ratio))
            resampled_h = resampled_h[:n_points]
            resampled_h = np.clip(resampled_h, 0, 1)
            resampled_h = resampled_h * (n_points / 2.0) / np.sum(resampled_h)
        
        try:
            h_opt, c5_opt = optimize_step_function(n_points, resampled_h)
            print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
