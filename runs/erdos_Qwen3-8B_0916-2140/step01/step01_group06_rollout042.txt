```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Using 200 points for better resolution
    dx = 2.0 / n_points
    initial_h_values = np.load('initial_h_values.npy')  # Starting from previous construction
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    
    x0 = initial_h.copy()
    
    # Perform optimization using L-BFGS-B for better performance
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 200})
    
    # Perform a few steps of gradient descent to refine the solution
    for _ in range(10):
        grad = np.gradient(result.fun, x0, method='central')
        x0 = np.clip(x0 - 0.01 * grad, 0, 1)
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 1})
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
```
