import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points, can scale up if needed
    dx = 2.0 / n_points
    
    # Initial guess using the provided initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Load provided initial values
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Use differential evolution to avoid local optima
    result = differential_evolution(
        fun=objective,
        bounds=bounds,
        constraints=cons,
        popsize=20,
        maxiter=100,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-6,
        polish=True,
        disp=False,
        vectorized=True,
        mutation_radius=0.5,
        workers=1
    )
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Optionally try a local optimization to refine the solution
    def local_objective(x):
        return compute_overlap(x)
    
    result_local = minimize(
        fun=local_objective,
        x0=best_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result_local.x
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
