```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
from scipy.optimize import Bounds, LinearConstraint, minimize
from time import time

# Pre-imported functions
from evaluate_erdos_solution import initial_h_values

def run(seed=42, budget_s=600):
    # Set parameters
    n_points = 200
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)

    # Define the optimization function
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points]
        h = h / np.sum(h) * n_points / 2
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0

    start_time = time()

    def run_optimization(starting_point):
        nonlocal evals
        result = minimize(
            objective,
            starting_point,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = objective(best_h)

    # Run multiple optimizations with different starting points
    for _ in range(5):
        if time() - start_time > budget_s - 1:
            break
        new_h = run_optimization(best_h)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Run final optimization with best_h
    final_h = best_h
    final_c5 = best_c5

    return (final_h, final_c5, len(final_h))
```
