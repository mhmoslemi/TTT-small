import numpy as np
import scipy.optimize as opt
import time
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for better resolution
    dx = 2.0 / n_points
    
    # Load initial construction from initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Ensure the file exists
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Use a global optimization approach with random restarts and adaptive step sizes
    best_h = x0
    best_c5 = compute_overlap(x0)
    best_time = 0
    
    for _ in range(5):  # Run with multiple restarts to avoid local optima
        start_time = time.time()
        
        # Use a modified version of the SLSQP solver with adaptive step sizes and bounds
        result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', 
                             options={'ftol': 1e-8, 'maxiter': 100, 'eps': 1e-6})
        
        if result.success:
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            x0 = current_h  # Use the improved solution as the next initial guess
        else:
            x0 = np.random.uniform(0, 1, n_points)
        
        elapsed_time = time.time() - start_time
        best_time = max(best_time, elapsed_time)
        
        if best_time > budget_s:
            break
    
    # Final evaluation
    final_c5 = compute_overlap(best_h)
    return (best_h.tolist(), final_c5, n_points)
