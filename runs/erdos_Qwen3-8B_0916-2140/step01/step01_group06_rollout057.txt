import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Using more points for better resolution
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)  # Start with uniform distribution
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess: uniform distribution
    x0 = np.array([0.5] * n_points)
    
    # Use a more robust optimizer with multiple restarts
    best_c5 = float('inf')
    best_h = np.array([0.5] * n_points)
    
    # Try multiple restarts with different initial guesses
    for _ in range(5):
        # Use random perturbation as initial guess
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        x0 = np.clip(0.5 + perturbation, 0, 1)
        
        # Optimize
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 200})
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return (best_h.tolist(), best_c5, n_points)
