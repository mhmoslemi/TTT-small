    # Define the optimization function
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])

        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.all(0 <= h) and np.all(h <= 1)

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0

    start_time = time()

    def run_optimization():
        nonlocal evals
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = objective(best_h)

    # Cluster-based search for better solutions
    def cluster_search(n_clusters):

        h_cluster = np.reshape(best_h, (1, -1))

        return centers

    for clusters in [5, 10, 20]:

            break

        cluster_centers = cluster_search(clusters)
        initial_guess = cluster_centers

        initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

        new_h = run_optimization()
