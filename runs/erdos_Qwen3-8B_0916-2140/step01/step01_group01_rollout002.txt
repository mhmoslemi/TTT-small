        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)

        # Function to compute the overlap integral
        def compute_c5(h):
            overlap = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(overlap) * dx
            return c5

        def constraint_func(h):
            return np.sum(h) - n_points / 2

        def objective(h):
            return compute_c5(h)

        # Set up the optimization
