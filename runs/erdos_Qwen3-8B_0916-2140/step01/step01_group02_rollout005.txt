```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points values to find the best balance between accuracy and speed
    n_candidates = [100, 200, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = 100
    
    for n_points in n_candidates:
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        
        # Start with a more refined initial guess
        if n_points == 100:
            # Initial guess similar to the previous one but with more structure
            for i in range(n_points):
                if i % 5 == 0:
                    h_values[i] = 1.0
                elif i % 5 == 1:
                    h_values[i] = 0.5
                elif i % 5 == 2:
                    h_values[i] = 0.0
                elif i % 5 == 3:
                    h_values[i] = 0.5
                elif i % 5 == 4:
                    h_values[i] = 1.0
        elif n_points == 200:
            # Break the interval [0, 2] into 20 equal segments, and assign values based on segment
            for i in range(n_points):
                segment = i // 10
                if segment % 2 == 0:
                    h_values[i] = 1.0
                else:
                    h_values[i] = 0.5
        elif n_points == 300:
            # Another structure with more variation
            for i in range(n_points):
                if i % 10 == 0:
                    h_values[i] = 1.0
                elif i % 10 == 1 or i % 10 == 2:
                    h_values[i] = 0.75
                elif i % 10 == 3 or i % 10 == 4:
                    h_values[i] = 0.5
                elif i % 10 == 5 or i % 10 == 6:
                    h_values[i] = 0.25
                else:
                    h_values[i] = 0.0
        
        h_values = np.clip(h_values, 0, 1)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        current_c5 = objective(result.x)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
            best_n = n_points
    
    return best_h, best_c5, best_n
```
