import scipy.optimize as optimize

    def compute_C5(h):

        c5 = np.max(result * dx)
        return c5

    def objective(h):
        return compute_C5(h)

    # Initial guess
    h0 = initial_h.copy()

    for _ in range(10):

        sum_perturbed = np.sum(h_perturbed)

        h_perturbed = np.clip(h_perturbed * scaling_factor, 0, 1)
        
        # Run local optimization
        result = optimize.minimize(
            objective,

            bounds=[(0, 1) for _ in range(n_points)],
