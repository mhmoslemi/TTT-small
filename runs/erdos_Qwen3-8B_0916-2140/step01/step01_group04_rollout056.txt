```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Preloaded initial construction
    n_points_initial = len(initial_h_values)
    
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    def objective(h):
        return compute_overlap(h)
    
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    def optimize_with_constraints(n_points, initial_guess=None):
        bounds = [(0, 1) for _ in range(n_points)]
        result = opt.minimize(
            objective,
            x0=initial_guess if initial_guess is not None else np.random.rand(n_points),
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    best_h = initial_h_values
    best_c5 = compute_overlap(initial_h_values)
    best_n_points = n_points_initial
    
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3]
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            h_opt, c5_opt = optimize_with_constraints(n_points, initial_guess=best_h)
            print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
```
