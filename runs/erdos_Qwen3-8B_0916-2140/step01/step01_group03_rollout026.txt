import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Start with initial guess
    best_h = initial_h
    best_C5 = compute_C5(best_h)
    
    # Try a few different starting points to avoid local optima
    for _ in range(3):
        # Perturb initial guess
        perturbation = np.random.normal(0, 0.1, size=n_points)
        perturbed_h = np.clip(initial_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Run optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        
        if result.success:
            current_h = result.x
            current_C5 = compute_C5(current_h)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
                print(f"New best C5: {best_C5}")
        else:
            print("Optimization failed for perturbed guess.")
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
