import scipy.optimize as optimize
import time

    np.random.seed(seed)

    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Use a more aggressive optimization approach with multiple restarts and parameter sweeps
    best_C5 = float('inf')
    best_h = initial_h
    
    # Define a set of parameter sweeps for different optimization settings
    param_sweeps = [
        {'method': 'SLSQP', 'ftol': 1e-8, 'maxiter': 200},
        {'method': 'L-BFGS-B', 'ftol': 1e-8, 'maxiter': 200},
        {'method': 'Nelder-Mead', 'ftol': 1e-8, 'maxiter': 200},
        {'method': 'SLSQP', 'ftol': 1e-8, 'maxiter': 200, 'eps': 1e-8},
    ]
    
    # Try different starting points based on the initial_h_values
    starting_points = [initial_h]
    
    # Add some perturbations to the initial points to avoid local optima
    for perturbation in [0.05, 0.1, 0.15]:
        perturbed = np.array(initial_h) + np.random.normal(0, perturbation, len(initial_h))
        perturbed = np.clip(perturbed, 0, 1)
        starting_points.append(perturbed)
    
    # Optimization loop with multiple starting points and parameter sweeps
    for start_h in starting_points:
        for params in param_sweeps:
            try:
                result = optimize.minimize(
                    objective,
                    start_h,
                    bounds=[(0, 1) for _ in range(n_points)],
                    constraints=[{'type': 'eq', 'fun': constraint}],
                    method=params['method'],
                    options={
                        'ftol': params.get('ftol', 1e-8),
                        'maxiter': params.get('maxiter', 200),
                        'eps': params.get('eps', 1e-8)
                    }
                )
                if result.success:
                    current_C5 = result.fun
                    if current_C5 < best_C5:
                        best_C5 = current_C5
                        best_h = result.x
                        print(f"New best C5: {best_C5}")
            except:
                pass
    
    # Check time budget
    if time.time() - start_time > time_budget:
        print(f"Time budget exceeded. Returning best found solution.")
