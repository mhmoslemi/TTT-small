```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
import scipy.signal as signal
from scipy.optimize import Bounds, LinearConstraint, minimize
from time import time

# Pre-imported initial_h_values for starting point
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=600):
    # Define the initial solution with n_points
    n_points = 200
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2

    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points]
        h = h / np.sum(h) * n_points / 2
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Optimization function
    def objective(h):
        return compute_c5(h)

    # Bounds for the variables
    bounds = Bounds(0, 1)

    # Initial guess
    initial_guess = initial_h.copy()

    # Optimization settings
    max_iter = 100
    tol = 1e-5

    start_time = time()

    def run_optimization():
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = compute_c5(best_h)

    # Run multiple optimizations from different starting points
    for _ in range(5):
        if time() - start_time > budget_s - 1:
            break
        new_h = run_optimization()
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    final_h = best_h
    final_c5 = best_c5
    return (final_h, final_c5, len(final_h))
```
