            x = np.array(x)

            result = np.correlate(x, 1 - x, mode="full")
            return np.max(result) * dx
        
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        result = minimize(
