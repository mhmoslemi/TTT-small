```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    # Initial construction with 45 points
    initial_h_values = np.array([0.0] * 45)
    for i in range(45):
        if i % 3 == 0:
            initial_h_values[i] = 1.0
        elif i % 3 == 1:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 0.0
    initial_h_values = initial_h_values.astype(float)
    
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    c5_initial = compute_overlap(initial_h_values)
    print(f"Initial C5 value: {c5_initial:.6f}")
    
    def optimize_step_function(n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        result = opt.minimize(objective, initial_h_values, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    best_h = initial_h_values
    best_c5 = c5_initial
    best_n_points = 45
    
    n_points_values = [45, 90, 135, 180]
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            h_opt, c5_opt = optimize_step_function(n_points)
            print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
```
