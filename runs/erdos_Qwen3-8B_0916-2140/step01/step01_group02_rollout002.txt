```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Improve initial guess: piecewise constant with more intervals
    for i in range(n_points):
        if i % 5 == 0:
            initial_h_values[i] = 1.0
        elif i % 5 == 1:
            initial_h_values[i] = 0.75
        elif i % 5 == 2:
            initial_h_values[i] = 0.5
        elif i % 5 == 3:
            initial_h_values[i] = 0.25
        elif i % 5 == 4:
            initial_h_values[i] = 0.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(initial_h_values) * dx
    scaling = 1.0 / total
    initial_h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with more iterations and tighter tolerances
    result = minimize(
        objective,
        initial_h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 200, "ftol": 1e-9, "eps": 1e-9},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
```
