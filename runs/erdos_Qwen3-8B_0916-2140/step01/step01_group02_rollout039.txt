import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define a range of n_points to try, starting from lower values for speed
    n_points_values = [100, 200, 300]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100
    
    for n_points in n_points_values:
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        
        # Initial guess: piecewise constant function with more intervals
        for i in range(n_points):
            # Create a more structured initial guess with periodicity and randomness
            if i % 5 == 0:
                h_values[i] = 1.0
            elif i % 5 == 1:
                h_values[i] = 0.5
            elif i % 5 == 2:
                h_values[i] = 0.0
            elif i % 5 == 3:
                h_values[i] = 0.5
            elif i % 5 == 4:
                h_values[i] = 1.0
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization with more aggressive parameters
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 150, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
    
    return best_h, best_c5, best_n_points
