import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
from scipy.optimize import Bounds, LinearConstraint, minimize
from time import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial construction parameters
    n_start = 100
    n_points = 200  # We'll use twice as many points for better resolution
    dx = 2.0 / n_points
    
    # Initial H-values based on a piecewise constant function with low overlap
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4:
            initial_h_values[i] = 0.0
        elif i < n_points // 2:
            initial_h_values[i] = 0.5
        elif i < 3 * n_points // 4:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.5
    
    # Define the optimization function
    def objective(h):
        h = np.array(h).astype(float)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the constraint for the integral of h(x) = 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds for the function values
    bounds = Bounds(0, 1)

    # Define the constraint for the integral
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess based on the initial construction
    initial_guess = initial_h_values

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0
    best_h = initial_guess
    best_c5 = objective(best_h)

    start_time = time()

    def run_optimization():
        nonlocal evals
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    for _ in range(10):
        if time() - start_time > budget_s - 1:
            break
        new_h = run_optimization()
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Return the best solution found
    return (best_h, best_c5, len(best_h))
