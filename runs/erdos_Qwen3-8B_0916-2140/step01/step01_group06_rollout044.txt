import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, DifferentialEvolution
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can scale up if needed
    dx = 2.0 / n_points
    initial_h_values = [0.0] * n_points
    for i in range(n_points):
        initial_h_values[i] = 0.5 * (1 - np.sin(np.pi * i / n_points))
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Start with a local optimization
    result_local = minimize(objective, initial_h, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 50})
    best_h = result_local.x
    best_c5 = compute_overlap(best_h)
    
    # If not improved enough, try a global optimization
    if best_c5 > 0.38:
        result_global = DifferentialEvolution(objective, bounds=bounds, constraints={'type': 'eq', 'fun': constraint_sum}, maxiter=50, popsize=20, strategy='best1bin', tol=1e-5, mutation=(0.5, 1), recombination=0.9)
        best_h = result_global.x
        best_c5 = compute_overlap(best_h)
    
    # Optionally scale to higher n_points if budget allows
    # For now, keep at 100 points for performance
    
    return (best_h.tolist(), best_c5, n_points)
