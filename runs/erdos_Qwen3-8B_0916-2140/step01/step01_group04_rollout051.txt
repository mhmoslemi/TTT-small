import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction with 45 points
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points_initial = len(initial_h_values)
    
    # Step 1: Evaluate the initial construction
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    c5_initial = compute_overlap(initial_h_values)
    print(f"Initial C5 value: {c5_initial:.6f}")
    
    # Step 2: Define a function to optimize the step function h
    def optimize_step_function(n_points):
        # Define the step function as a vector h of size n_points
        # with h[i] in [0, 1] and sum(h) = n_points / 2
        
        # Define bounds (0 <= h[i] <= 1) for all i
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: maximize the overlap integral
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        # Constraint: sum(h) == n_points / 2
        constraint_sum = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        
        # Perform optimization
        result = minimize(objective, initial_h_values, bounds=bounds, constraints=constraint_sum, method='SLSQP')
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    # Step 3: Try different n_points (start with higher n_points to capture more structure)
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]
    
    best_h = initial_h_values
    best_c5 = c5_initial
    best_n_points = n_points_initial
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            h_opt, c5_opt = optimize_step_function(n_points)
            print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
            
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    print(f"Best C5 value found: {best_c5:.6f} with {best_n_points} points")
    
    return best_h, best_c5, best_n_points
