```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution
from functools import partial
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import lsqr
from scipy.signal import correlate

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # starting with 100 points
    
    # Define initial construction (n=100 samples)
    def initial_h_values(x):
        if 0 <= x < 0.5:
            return 0.0
        elif 0.5 <= x < 1.0:
            return 1.0
        elif 1.0 <= x < 1.5:
            return 0.0
        elif 1.5 <= x < 2.0:
            return 1.0
    
    x_vals = np.linspace(0, 2, n_points)
    initial_h = np.array([initial_h_values(x) for x in x_vals])
    initial_h = np.clip(initial_h, 0, 1)
    
    # Use differential evolution for global optimization
    def objective(h):
        h = np.array(h, dtype=np.float64)
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * (2.0 / n_points)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Create a fixed initial population using the initial_h
    initial_pop = np.array([initial_h + np.random.uniform(-0.1, 0.1) for _ in range(20)])
    initial_pop = np.clip(initial_pop, 0, 1)
    
    # Run differential evolution with a budget of 600 seconds
    start_time = time.time()
    result = differential_evolution(
        objective,
        bounds,
        popsize=20,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=int((budget_s - (time.time() - start_time)) / 0.1),
        tol=1e-6,
        disp=False,
        polish=True,
        vectorsize=n_points,
        initial_population=initial_pop
    )
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
```
