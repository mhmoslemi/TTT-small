import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points values
    n_points_list = [256, 512, 1024]
    
    best_c5 = float('inf')
    best_h = np.array([])
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        scaling = 1.0 / total
        h *= scaling
        
        # Define the objective function
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        # Define the constraints
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': constraint})
        
        # Optimization
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                print(f"Best C5 for n_points={n_points}: {best_c5}")
    
    return best_h, best_c5, len(best_h)
