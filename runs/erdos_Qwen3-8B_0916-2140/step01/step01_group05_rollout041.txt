import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a base number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Define initial guess based on a uniform pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            initial_h[i] = 1.0
        elif i % 5 == 1 or i % 5 == 2:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    
    # Normalize to ensure the integral is 1
    initial_h = initial_h / (np.sum(initial_h) / n_points)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Set up the optimization problem
    result = optimize.minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum})
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
