import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Initial construction
    initial_h = initial_h_values(n_points)
    initial_c5 = evaluate_erdos_solution(initial_h)

    # Try with different n_points to find optimal balance
    best_h = initial_h
    best_c5 = initial_c5
    best_n_points = n_points

    # Try with n_points = 50, 75, 100, 125, 150
    for n_points in [50, 75, 100, 125, 150]:
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < 10:
                h[i] = 0.3
            elif i < 20:
                h[i] = 0.6
            elif i < 30:
                h[i] = 0.9
            elif i < 40:
                h[i] = 0.3
            elif i < 50:
                h[i] = 0.6
            elif i < 60:
                h[i] = 0.9
            else:
                h[i] = 0.3
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
            best_n_points = n_points

    # Try with different optimization approaches
    for n_points in [100, 150, 200]:
        dx = 2.0 / n_points
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        initial_h = np.random.uniform(0, 1, n_points)
        # Ensure sum(h) is approximately n_points / 2
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)

        def objective(h):
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        def constraints(h):
            return np.array([np.sum(h) - n_points / 2])

        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )

        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                best_n_points = n_points

    return (best_h.tolist(), best_c5, best_n_points)
