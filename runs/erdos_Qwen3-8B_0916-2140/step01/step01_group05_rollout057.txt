import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess: start with a flat h and then refine
    initial_h = np.full(n_points, 0.5)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h
    
    # Optimization settings
    options = {
        'maxiter': 200,
        'ftol': 1e-8,
        'disp': False
    }
    
    # First optimization: local search
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='SLSQP', options=options)
    
    # If the result is good, refine with a global search
    if result.success:
        # Use the result as a starting point for a global search
        x0 = result.x
        result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='L-BFGS-B', options=options)
    
    # If the result is still good, try a different starting point
    if result.success:
        # Create a perturbed version of the result as a new starting point
        perturb = np.random.normal(0, 0.1, size=n_points)
        x0 = result.x + perturb
        x0 = np.clip(x0, 0, 1)
        result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='L-BFGS-B', options=options)
    
    # Ensure the result is valid
    if not result.success:
        result = minimize(objective, initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='L-BFGS-B', options=options)
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
