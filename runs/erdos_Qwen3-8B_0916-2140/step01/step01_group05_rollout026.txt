```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess based on a more evenly distributed pattern
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            initial_h[i] = 1.0
        elif i % 2 == 0:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.0
    initial_h = initial_h / (np.sum(initial_h) / n_points)  # Ensure sum is n_points / 2
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Optimization parameters
    maxiter = 100
    tol = 1e-6
    options = {'maxiter': maxiter, 'tol': tol}
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Perform the optimization
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, options=options)
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
```
