import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # initial n_points
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0]*n_points)
    # Define a good initial guess based on known patterns
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h_values[i] = 0.0
        elif x < 1.0:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    
    x0 = initial_h_values.copy()
    
    # Local optimization with SLSQP
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try to improve with a more sophisticated optimizer
    # Use a COBYLA method for bounds-constrained optimization
    result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='COBYLA', options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Optionally, try to scale up n_points to 200 and re-optimise
    n_points = 200
    dx = 2.0 / n_points
    # Interpolate the best_h to n_points
    best_h_interpolated = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    best_h_interpolated = np.clip(best_h_interpolated, 0, 1)
    
    x0 = best_h_interpolated.copy()
    
    # Local optimization with SLSQP
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try to improve with a more sophisticated optimizer
    # Use a COBYLA method for bounds-constrained optimization
    result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='COBYLA', options={'maxiter': 100})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
