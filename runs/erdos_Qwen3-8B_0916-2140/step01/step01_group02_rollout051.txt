```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.load("initial_h_values.npy")
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    h_values = initial_h.copy()
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = minimize(
        objective,
        h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="L-BFGS-B",
        options={"maxiter": 1000, "ftol": 1e-8, "eps": 1e-8},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
```
