import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, LinearConstraint
from functools import lru_cache

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.3
        elif i < 40:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between performance and precision
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Define constraints: sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial optimization with SLSQP
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    # If optimization succeeded, return the result
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # Try a second optimization with different initial guess
        new_initial_h = np.random.uniform(0, 1, n_points)
        new_initial_h = new_initial_h * (n_points / 2) / np.sum(new_initial_h)
        result = minimize(
            objective,
            new_initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_sum},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        if result.success:
            best_h = result.x
            c5 = evaluate_erdos_solution(best_h)
            return (best_h.tolist(), c5, n_points)
        else:
            # Fallback to initial_h
            c5 = evaluate_erdos_solution(initial_h)
            return (initial_h.tolist(), c5, n_points)
