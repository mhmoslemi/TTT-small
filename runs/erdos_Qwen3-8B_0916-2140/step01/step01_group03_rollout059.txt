import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    time_budget = budget_s
    start_time = time.time()
    
    # Optimization loop
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Try different starting points and optimization strategies
    strategies = [
        ('SLSQP', {'ftol': tolerance, 'maxiter': 100}),
        ('Nelder-Mead', {}),
        ('L-BFGS-B', {'bounds': [(0, 1) for _ in range(n_points)]})
    ]
    
    for strategy, params in strategies:
        print(f"Trying optimization strategy: {strategy}")
        
        result = optimize.minimize(
            objective,
            h0,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=strategy,
            options=params
        )
        
        if result.success:
            current_h = result.x
            current_C5 = compute_C5(current_h)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
                h0 = current_h
                print(f"New best C5: {best_C5}")
        else:
            print(f"Optimization with {strategy} failed.")
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
