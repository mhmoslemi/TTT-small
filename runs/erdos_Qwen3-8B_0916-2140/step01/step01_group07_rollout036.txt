import numpy as np
import scipy.optimize as opt

import time

def run(seed=42, budget_s=600):
    # Define the initial solution with 100 samples
    initial_h = initial_h_values.copy()
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h[:n_points]
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2

    def compute_c5(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])
        h = h[:n_points]
        h = h / np.sum(h) * n_points / 2
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def optimize_h():
        n_points = 100
        dx = 2.0 / n_points
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]

        def objective(h):
            h = np.array(h)
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = np.concatenate([h, h[:n_points//2]])
            h = h[:n_points]
            h = h / np.sum(h) * n_points / 2
            result = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(result) * dx
            return c5

        result = opt.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8, 'disp': False}
        )
        return result.x

    start_time = time.time()
    best_h = initial_h
    best_c5 = compute_c5(initial_h)
    best_c5 = best_c5

    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h()
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

    return (best_h, best_c5, len(best_h))
