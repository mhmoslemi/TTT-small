import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # higher resolution for better optimization
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    
    # Start with a simple baseline construction: uniform h(x) = 0.5
    initial_h = np.array([0.5 for _ in range(n_points)])
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial guess: simple step function
    x0 = np.array([0.5 for _ in range(n_points)])
    
    # Use a more advanced optimization method with fewer iterations
    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='L-BFGS-B', options={'ftol': 1e-8, 'maxiter': 1000})
    
    # Refine using a local optimization
    result = opt.minimize(objective, result.x, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
