import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial n_points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial_h_values is provided in the global scope
    # We'll start with a structured guess based on known patterns
    
    # Define initial_h_values as a structured guess
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 0.5 if i < n_points // 2 else 0.0
    initial_h = initial_h / (np.sum(initial_h) / n_points)  # Ensure sum is n_points / 2
    
    # Objective function to minimize the max overlap
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Constraint: sum of h is n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Use L-BFGS-B for faster convergence
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method="L-BFGS-B",
        tol=1e-10,
        options={"max_iter": 1000, "ftol": 1e-12}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Simple check: ensure the solution is valid
    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = np.clip(best_h, 0, 1)
        best_c5 = objective(best_h)
    
    # Ensure the constraint is satisfied
    constraint_val = np.sum(best_h) - n_points / 2
    if abs(constraint_val) > 1e-10:
        best_h = best_h / (np.sum(best_h) / n_points)
    
    # Final evaluation
    final_c5 = objective(best_h)
    
    return best_h, final_c5, n_points
