        result = minimize(

            h0,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6, 'disp': False}
        )

    best_c5 = objective(best_h)

    start_time = time()
    for _ in range(3):
        if time() - start_time > budget_s - 1:
            break
