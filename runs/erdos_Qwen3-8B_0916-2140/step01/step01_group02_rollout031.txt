import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    
    # Use the initial construction provided
    for i in range(n_points):
        if i % 5 == 0:
            initial_h_values[i] = 1.0
        elif i % 5 == 1:
            initial_h_values[i] = 0.5
        elif i % 5 == 2:
            initial_h_values[i] = 0.0
        elif i % 5 == 3:
            initial_h_values[i] = 0.5
        elif i % 5 == 4:
            initial_h_values[i] = 1.0
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(initial_h_values) * dx
    scaling = 1.0 / total
    initial_h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = LinearConstraint(np.ones(n_points), 1.0 / dx, 1.0 / dx)
    
    # Optimization with more aggressive settings
    result = minimize(
        objective,
        initial_h_values,
        bounds=bounds,
        constraints=constraint,
        method="SLSQP",
        options={"maxiter": 200, "ftol": 1e-8, "eps": 1e-8, "disp": False},
        callback=lambda x: None  # Suppress intermediate output
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
