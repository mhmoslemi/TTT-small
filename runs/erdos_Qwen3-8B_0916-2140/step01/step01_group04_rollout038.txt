    def objective(h, n_points):
        dx = 2.0 / n_points
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    def constraint(h):
        return np.sum(h) - len(h) / 2
    
    def optimize_step_function(n_points, initial_guess=None):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_func(h):
            return objective(h, n_points)
        
        # Use a more robust optimization method with bounds
        result = minimize(
            fun=objective_func,
            x0=initial_guess if initial_guess is not None else np.random.uniform(0, 1, n_points),
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    # Step 3: Try different n_points starting from initial
    best_h = h_initial
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]
    
    start_time = time.time()
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break

            # Try with initial guess and without initial guess
            for initial_guess in [None, best_h]:
                try:
                    h_opt, c5_opt = optimize_step_function(n_points, initial_guess)
                    print(f"Optimized C5 value for {n_points} points: {c5_opt:.6f}")
                    
                    if c5_opt < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt
                        best_n_points = n_points
