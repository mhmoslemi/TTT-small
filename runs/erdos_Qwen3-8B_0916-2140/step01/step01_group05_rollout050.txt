import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import cg
from itertools import product

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a reasonable number of points
    n_points = 45
    
    # Define the initial guess based on the provided initial_h_values
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        initial_h_values[i] = initial_h_values[i]
    initial_h_values = initial_h_values / (np.sum(initial_h_values) / n_points)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * (2.0 / n_points))
        return c5
    
    # Define the constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Gradient descent with momentum to avoid local minima
    def gradient_descent(h, learning_rate=0.1, momentum=0.9, max_iter=1000, tol=1e-6):
        h = h.copy()
        velocity = np.zeros_like(h)
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * (2.0 / n_points))
            
            # Compute gradient
            for i in range(n_points):
                delta = np.zeros_like(h)
                delta[i] = 1.0
                corr_delta = np.correlate(h + delta, 1 - h, mode="full") - np.correlate(h, 1 - h, mode="full")
                grad[i] = np.max(corr_delta * (2.0 / n_points))
            
            # Update velocity and h
            velocity = momentum * velocity - learning_rate * grad
            h += velocity
            
            # Clip to bounds
            h = np.clip(h, 0, 1)
            
            # Check for convergence
            if np.max(np.abs(grad)) < tol:
                break
        return h
    
    # Perform a series of gradient descent steps starting from the initial guess
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    best_n_points = n_points
    
    # Try increasing the number of points up to 100, in steps of 10
    for n in range(50, 101, 10):
        dx = 2.0 / n
        h = np.random.rand(n) * 2 - 1  # Initial guess with random values
        h = np.clip(h, 0, 1)
        h = h * (n / 2)  # Ensure the sum is n / 2
        
        # Perform gradient descent
        h = gradient_descent(h)
        
        # Compute the current C5 value
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        
        # Keep the best solution
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n
    
    return best_h, best_c5, best_n_points
