import numpy as np

from scipy.optimize import Bounds, LinearConstraint, minimize
from time import time

def run(seed=42, budget_s=600):
    # Initial guess: based on the provided initial_h_values
    initial_h_values = evaluate_erdos_solution()
    n_initial = len(initial_h_values)
    n_points = 200  # More points for better resolution
    dx = 2.0 / n_points

    # Interpolate the initial_h_values to n_points
    x_initial = np.linspace(0, 2, n_initial)
    x_new = np.linspace(0, 2, n_points)
    initial_h = np.interp(x_new, x_initial, initial_h_values)

    # Define the objective function
    def objective(h):
        h = np.array(h)
        h = np.interp(x_new, np.linspace(0, 2, len(h)), h)
        h = np.concatenate([h, h[:n_points//2]])  # Repeat to handle full correlation
        h = h[:n_points]
        # Normalize to satisfy the integral constraint
        h = h / np.sum(h) * n_points / 2
        # Compute the overlap integral
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Constraint: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds: h in [0, 1]
    bounds = Bounds(0, 1)

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0

    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Define the optimization function
    def optimize_with_seed(seed):
        np.random.seed(seed)
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = objective(best_h)

    start_time = time()

    for _ in range(10):
        if time() - start_time > budget_s - 1:
            break
        # Perform optimization with a new seed
        new_h = optimize_with_seed(seed + _)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    final_h = best_h
    final_c5 = best_c5
    return (final_h, final_c5, len(final_h))
