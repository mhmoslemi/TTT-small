```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try different n_points values to find a good balance between accuracy and speed
    n_points_list = [100, 200, 300]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        
        # Start with a more balanced initial guess based on a checkerboard pattern
        for i in range(n_points):
            if i % 5 == 0:
                h_values[i] = 1.0
            elif i % 5 == 1:
                h_values[i] = 0.5
            elif i % 5 == 2:
                h_values[i] = 0.0
            elif i % 5 == 3:
                h_values[i] = 0.5
            elif i % 5 == 4:
                h_values[i] = 1.0
        h_values = np.clip(h_values, 0, 1)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h_values) * dx
        scaling = 1.0 / total
        h_values *= scaling
        
        def objective(x):
            h = x
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        def constraint(x):
            return np.sum(x) * dx - 1.0
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        start_time = time.time()
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 50, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        elapsed_time = time.time() - start_time
        
        if elapsed_time > budget_s:
            print(f"Time exceeded budget for n_points={n_points}, skipping...")
            continue
        
        best_h_current = result.x
        best_c5_current = objective(best_h_current)
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
```
