import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Helper function to create a random initial guess with specified sum
    def create_random_initial(n_points):
        h = np.random.uniform(0, 1, n_points)
        scale = (n_points / 2) / np.sum(h)
        return h * scale
    
    # We will try multiple starting points to avoid local optima
    n_starting_points = 5
    best_C5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    for start_idx in range(n_starting_points):
        # Create a random starting point
        h0 = create_random_initial(n_points)
        
        # Optimization settings
        max_iter = 100
        tolerance = 1e-6
        time_remaining = budget_s - (time.time() - start_time)
        
        # Optimization loop
        current_h = h0
        current_C5 = compute_C5(current_h)
        
        for _ in range(max_iter):
            # Use scipy's optimization to find a better solution
            result = minimize(
                objective,
                current_h,
                bounds=Bounds([0, 1], [0, 1]),
                constraints=[{'type': 'eq', 'fun': constraint}],
                method='SLSQP',
                options={'ftol': tolerance, 'maxiter': 100, 'disp': False}
            )
            
            if result.success:
                current_h = result.x
                current_C5 = compute_C5(current_h)
                
                if current_C5 < best_C5:
                    best_C5 = current_C5
                    best_h = current_h
                    print(f"New best C5: {best_C5}")
            else:
                print("Optimization failed.")
                break
            
            # Check time budget
            if time.time() - start_time > time_remaining:
                print(f"Time budget exceeded for starting point {start_idx}. Returning best found solution.")
                break
        
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
    
    return best_h, best_C5, len(best_h)
