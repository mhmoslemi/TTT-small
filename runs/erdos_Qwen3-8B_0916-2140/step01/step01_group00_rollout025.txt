import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, differential_evolution
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=70):
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a base number of points
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    # Define the objective function
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Set bounds for h: each h[i] ∈ [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization setup using differential evolution
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    # Post-process the result to ensure constraint is met
    best_h = result.x
    if np.abs(constraint_sum(best_h)) > 1e-6:
        # Adjust to meet the constraint
        scaling = (n_points / 2) / np.sum(best_h)
        best_h = np.clip(best_h * scaling, 0, 1)
    
    # Evaluate the best solution
    c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), c5, n_points)
