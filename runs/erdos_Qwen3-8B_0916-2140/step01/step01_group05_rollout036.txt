import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100  # Increased from 45 for better resolution
    dx = 2.0 / n_points
    
    # Use initial_h_values as a starting point
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        initial_h_values[i] = initial_h_values[i]
    initial_h_values = initial_h_values / (np.sum(initial_h_values) / n_points)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h_values.copy()
    
    # Use a more sophisticated optimization method with a budget
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_sum},
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
