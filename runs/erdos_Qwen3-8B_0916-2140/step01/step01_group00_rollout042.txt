```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction with 100 points
    h = np.zeros(n)
    for i in range(n):
        if i < 30:
            h[i] = 0.3
        elif i < 60:
            h[i] = 0.6
        else:
            h[i] = 0.9
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 150  # Increase to 150 points for better precision
    
    initial_h = initial_h_values(n_points)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization with multiple restarts to avoid local minima
    best_result = None
    best_c5 = float('inf')
    time_start = time.time()
    
    for _ in range(3):  # 3 restarts
        if time.time() - time_start > budget_s - 10:
            break
        initial_guess = initial_h + np.random.uniform(-0.1, 0.1, n_points)
        initial_guess = np.clip(initial_guess, 0.0, 1.0)
        
        result = minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        if result.success:
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_result = result.x
    
    if best_result is not None:
        best_h = best_result
        c5 = best_c5
        return (best_h.tolist(), c5, n_points)
    else:
        c5 = evaluate_erdos_solution(initial_h)
        return (initial_h.tolist(), c5, n_points)
```
