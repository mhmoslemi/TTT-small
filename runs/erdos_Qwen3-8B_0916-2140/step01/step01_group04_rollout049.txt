def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5], dtype=np.float64)
    
    def compute_overlap(h):
        n = len(h)
        dx = 2.0 / n
        overlaps = np.correlate(h, 1 - h, mode="full")
        return np.max(overlaps) * dx
    
    def optimize_with_random_search(n_points, initial_h, max_iterations=100):
        best_h = initial_h.copy()
        best_c5 = compute_overlap(best_h)
        
        for _ in range(max_iterations):
            perturbations = np.random.normal(0, 0.1, n_points)
            h = initial_h + perturbations
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-5:
                continue
            c5 = compute_overlap(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
        return best_h, best_c5
    
    def optimize_with_simulated_annealing(n_points, initial_h, temp=1.0, cooling_rate=0.995, iterations=1000):
        h = initial_h.copy()
        current_c5 = compute_overlap(h)
        best_h = h.copy()
        best_c5 = current_c5
        
        for _ in range(iterations):
            perturbation = np.random.normal(0, 0.1, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            sum_h = np.sum(h_perturbed)
            if abs(sum_h - n_points / 2) > 1e-5:
                continue
            new_c5 = compute_overlap(h_perturbed)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h = h_perturbed
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
            temp *= cooling_rate
        return best_h, best_c5
    
    def optimize_with_gradient_descent(n_points, initial_h, learning_rate=0.01, max_epochs=1000, tolerance=1e-5):
        h = initial_h.copy()
        for _ in range(max_epochs):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlaps)
            c5 = max_overlap * dx
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i < n_points:
                    for j in range(n_points):
                        if j < n_points:
                            overlap = np.correlate(h, 1 - h, mode="full")[n_points - 1 + j]
                            derivative = (overlap - overlap[i]) / (1e-8)
                            grad[i] += derivative
            grad = grad / (n_points * dx)
            
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-5:
                continue
            
            if np.abs(np.sum(grad)) < tolerance:
                break
        return h, compute_overlap(h)

        scaling_factor = (n_points / 2) / sum_h
        h = h * scaling_factor
        return h
    
    start_time = time.time()

            initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), mode='constant')
        elif len(initial_h) > n_points:
            initial_h = initial_h[:n_points]
        try:
            h_opt, c5_opt = optimize_with_random_search(n_points, initial_h)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    for n_points in n_points_values:
        if time.time() - start_time > budget_s - 10:
            break
        initial_h = random_initial(n_points)
        try:
            h_opt, c5_opt = optimize_with_simulated_annealing(n_points, initial_h)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n_points = n_points
        except:
            print(f"Optimization failed for n={n_points}, skipping.")
    
    for n_points in n_points_values:
