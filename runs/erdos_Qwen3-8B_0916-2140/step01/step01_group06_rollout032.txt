import numpy as np

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Starting with 200 points for better resolution
    dx = 2.0 / n_points

    # Initial construction: Inspired by an optimized piecewise constant function
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h_values[i] = 0.0
        elif 0.5 <= x < 1.5:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0

    # Function to compute the overlap integral
    def compute_overlap(h_values):
        h = np.array(h_values, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr[1:-1]) * dx
        return max_overlap

    # Optimization bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Use a hybrid approach: first with a global optimization method, then local refinement
    def objective(x):
        return compute_overlap(x)

    # Global optimization with differential evolution
    result_global = minimize(
        objective, 
        initial_h_values, 
        bounds=bounds, 
        constraints={'type': 'eq', 'fun': constraint_sum}, 
        method='L-BFGS-B', 
        options={'ftol': 1e-8, 'max_iter': 100, 'disp': False}
    )

    best_h = result_global.x
    best_c5 = compute_overlap(best_h)

    # Local optimization with SLSQP to refine the solution
    result_local = minimize(
        objective, 
        best_h, 
        bounds=bounds, 
        constraints={'type': 'eq', 'fun': constraint_sum}, 
        method='SLSQP', 
        options={'ftol': 1e-8, 'max_iter': 100, 'disp': False}
    )

    best_h = result_local.x
    best_c5 = compute_overlap(best_h)

    # Final check and return
    return (best_h.tolist(), best_c5, n_points)
