import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
from functools import lru_cache
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    # Initial construction (sample with 100 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 0.3
        elif i < 40:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between performance and precision

    # Define initial construction
    initial_h = initial_h_values(n_points)

    # First optimization step: fine-tune the initial construction
    def objective(h):
        # Compute the overlap integral for all k
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5

    def constraints(h):
        # Ensure sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
        return np.array([np.sum(h) - n_points / 2])

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    best_h = result.x
    c5 = evaluate_erdos_solution(best_h)
    print(f"Initial optimization c5: {c5}")

    # Second optimization step: use a more refined approach with random perturbations
    n_points_refined = 100
    dx = 2.0 / n_points_refined
    h_refined = best_h.copy()

    # Function to evaluate the cost
    def cost_func(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Define a new initial construction that better distributes values
    def refined_initial_h(n=100):
        h = np.zeros(n)
        for i in range(n):
            if i < 20:
                h[i] = 0.3
            elif i < 35:
                h[i] = 0.6
            elif i < 50:
                h[i] = 0.9
            elif i < 70:
                h[i] = 0.6
            else:
                h[i] = 0.3
        return h

    refined_initial = refined_initial_h(n_points_refined)
    h_refined = refined_initial.copy()

    result_refined = minimize(
        cost_func,
        h_refined,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    best_h_refined = result_refined.x
    c5_refined = evaluate_erdos_solution(best_h_refined)
    print(f"Refined optimization c5: {c5_refined}")

    # Final step: use a more robust optimization with more iterations
    result_final = minimize(
        cost_func,
        best_h_refined,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="L-BFGS-B",
        options={"ftol": 1e-8, "maxiter": 200}
    )

    best_h_final = result_final.x
    c5_final = evaluate_erdos_solution(best_h_final)
    print(f"Final optimization c5: {c5_final}")

    return (best_h_final.tolist(), c5_final, n_points_refined)
