```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 1000  # Increased n_points for higher precision
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    
    # Initial guess: use the provided initial_h_values
    try:
        initial_h_values = np.load("initial_h_values.npy")
        h_values = initial_h_values[:n_points]
    except:
        # Fallback to a simple alternating pattern
        for i in range(n_points):
            if i % 5 == 0:
                h_values[i] = 1.0
            elif i % 5 == 1:
                h_values[i] = 0.5
            elif i % 5 == 2:
                h_values[i] = 0.0
            elif i % 5 == 3:
                h_values[i] = 0.5
            elif i % 5 == 4:
                h_values[i] = 1.0
    
    h_values = np.clip(h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with more iterations and tighter tolerances
    result = minimize(
        objective,
        h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 150, "ftol": 1e-10, "eps": 1e-8},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
```
