```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h = np.load("initial_h_values.npy")

    # Rescale initial_h to meet the integral constraint
    initial_h_scaled = initial_h / np.sum(initial_h) * n_points / 2
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)

    # Define objective function
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Define constraint
    def constraint(x):
        return np.sum(x) * dx - 1.0

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = minimize(
        objective,
        initial_h_scaled,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )

    best_h = result.x
    best_c5 = objective(best_h)

    return best_h, best_c5, n_points
```
