import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import Bounds
from scipy.optimize import differential_evolution
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for better resolution
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    
    # Assume initial_h_values is defined in the global scope and pre-loaded
    # For this example, we'll use a simple random initialization
    initial_h = np.random.uniform(0, 1, n_points)
    initial_h = initial_h / (np.sum(initial_h) / n_points)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using differential evolution
    result = differential_evolution(
        objective,
        bounds,
        constraints=[LinearConstraint(np.ones(n_points), 0, n_points / 2)],
        maxiter=500,
        popsize=30,
        tolerance=1e-5,
        mutation=(0.5, 1),
        seed=seed,
        polish=False,
        timebudget=budget_s
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
