```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=100):
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    # Start with fewer points to allow more aggressive optimization
    n_points = 80
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    # Initial refinement - smooth the initial construction to avoid spikes
    def smooth_h(h, alpha=0.5):
        h_smooth = np.zeros_like(h)
        for i in range(len(h)):
            if i == 0:
                h_smooth[i] = h[i]
            elif i == len(h) - 1:
                h_smooth[i] = h[i]
            else:
                h_smooth[i] = alpha * h[i] + (1 - alpha) * (h[i-1] + h[i+1]) / 2
        return h_smooth
    
    initial_h = smooth_h(initial_h)
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraints(h):
        return np.array([np.sum(h) - n_points / 2])
    
    # Set up bounds and optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 150}
    )
    
    if result.success:
        best_h = result.x
        c5 = evaluate_erdos_solution(best_h)
        return (best_h.tolist(), c5, n_points)
    else:
        # Fallback to initial_h and evaluate
        c5 = evaluate_erdos_solution(initial_h)
        return (initial_h.tolist(), c5, n_points)
```
