        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)

        current_C5 = compute_C5(h0)
        
        # Compute gradient of the objective
        def gradient(h):
            h = np.array(h)
            result = np.correlate(h, 1 - h, mode="full")
            gradient = np.zeros_like(h)
            
            # Compute the gradient of the max operation
            max_index = np.argmax(result * dx)
            if max_index < len(result) // 2:
                # The maximum is in the middle of the correlation
                # Compute the derivative of the correlation with respect to h
                for i in range(len(h)):
                    # The derivative of the correlation at position i
                    # is the sum over k of (1 - h[i + k]) where i + k is within bounds
                    # This is O(n^2), but we only need it for the max index
                    if max_index - i >= 0 and max_index - i < len(h):
                        grad = 1 - h[max_index - i]
                    else:
                        grad = 0
                    gradient[i] = grad
            return gradient

        # Update h0 using a gradient descent step with momentum
        step_size = 0.1

        # Ensure h0 is within bounds
        h0 = np.clip(h0, 0, 1)
        
        # Enforce the constraint
        constraint_value = np.sum(h0) - n_points / 2
        if constraint_value != 0:

        current_C5 = compute_C5(h0)
        if current_C5 < best_C5:

            best_h = h0

    )
