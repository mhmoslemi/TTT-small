    def objective(h):

    # Objective function to maximize the minimum overlap (for verification)

        h = np.array(h)

    # Best solution tracker
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Optimization loop with multiple restarts and adaptive steps
    for restart in range(3):
        # Initial guess with random variation
        h0 = np.array(initial_h)
        h0 += np.random.uniform(-0.1, 0.1, size=h0.shape)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0) if np.sum(h0) != 0 else h0

        method = 'SLSQP'
        maxiter = 100
        ftol = 1e-6
        
        # Run optimization for this restart
        result = optimize.minimize(
            objective,
            h0,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            method=method,
            options={'ftol': ftol, 'maxiter': maxiter}
        )

                print(f"Found a potentially better solution with max overlap: {max_overlap}")
