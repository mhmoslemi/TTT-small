```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
from scipy.optimize import Bounds, LinearConstraint, minimize
from time import time

def run(seed=42, budget_s=600):
    # Define initial solution
    n_points = 100
    initial_h_values = np.concatenate([
        np.linspace(0, 1, n_points // 2 + 1)[::-1],
        np.linspace(1, 0, n_points // 2 + 1)
    ])
    initial_h_values = initial_h_values[:n_points]
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    def run_optimization(start_h):
        initial_guess = start_h.copy()
        initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8, 'disp': False}
        )
        return result.x

    # Start with initial_h_values
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)

    start_time = time()

    # Try multiple starting points
    for _ in range(3):
        if time() - start_time > budget_s - 1:
            break
        # Perturb initial_h_values to explore different regions
        perturbed_h = initial_h_values.copy()
        perturbed_h += np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * n_points / 2

        new_h = run_optimization(perturbed_h)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Final optimization with best_h as starting point
    if time() - start_time < budget_s - 1:
        final_h = run_optimization(best_h)
        final_c5 = objective(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return (best_h, best_c5, len(best_h))
```
