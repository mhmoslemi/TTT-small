```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    initial_h = initial_h + np.random.normal(0, 0.05, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.sum(initial_h) * (n_points / 2)

    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5

    def objective(h):
        return compute_C5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    def optimize_with_bounds(h, bounds):
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x

    def get_bounds(h):
        return [(0, 1) for _ in range(n_points)]

    best_h = initial_h
    best_C5 = compute_C5(best_h)

    for _ in range(10):
        current_h = optimize_with_bounds(best_h, get_bounds(best_h))
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h

    return best_h, best_C5, n_points
```
