import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of points, starting with 100 for better resolution
    n_points = 100
    dx = 2.0 / n_points
    
    # Load initial_h_values (assuming they are available in the global scope)
    # For the purpose of this example, we'll use a simple uniform distribution as initial guess
    # This is a placeholder; in practice, initial_h_values would be provided
    initial_h = np.full(n_points, 0.5)
    
    # Ensure the initial guess satisfies the constraint sum(h) = n_points / 2
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Define the objective function to minimize
    def objective(h):
        h = h.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Define the constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Optimization settings
    options = {
        'maxiter': 100,
        'ftol': 1e-8,
        'disp': False
    }
    
    # Run the optimization
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "eq", "fun": constraint_sum}, method='L-BFGS-B', options=options)
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Return the solution
    return best_h, best_c5, n_points
