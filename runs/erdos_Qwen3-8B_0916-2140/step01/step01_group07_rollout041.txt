import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

def run(seed=42, budget_s=600):
    # Initial construction from initial_h_values
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    # Define the optimization function
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Function to generate a random initialization
    def random_h(n_points):
        return np.random.uniform(0, 1, n_points)

    # Optimization settings
    max_iter = 100
    tol = 1e-6
    evals = 0
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    def run_optimization(guess):
        nonlocal evals
        result = minimize(
            objective,
            guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    # Start with the initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2
    best_h = initial_guess
    best_c5 = objective(best_h)

    # Run multiple optimizations with different initializations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        guess = random_h(n_points)
        guess = guess / np.sum(guess) * n_points / 2
        new_h = run_optimization(guess)
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h

    return (best_h, best_c5, len(best_h))
