        initial_x = np.linspace(0, 2, len(initial_h_values))

        def compute_c5(h):
            overlap = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(overlap) * dx
            return c5
        
        # Optimization function with constraints
        def objective(h):
            return compute_c5(h)
        
        # Constraints: sum of h * dx = 1 --> sum(h) = n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        # Bounds: 0 <= h[i] <= 1
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the optimization problem
        result = minimize(
            objective,
            initial_guess,
            method='SLSQP',
