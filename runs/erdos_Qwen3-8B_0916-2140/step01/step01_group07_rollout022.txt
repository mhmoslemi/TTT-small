```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
from scipy.optimize import Bounds, LinearConstraint, minimize
from time import time

def run(seed=42, budget_s=600):
    # Define the initial solution from initial_h_values
    initial_h_values = np.loadtxt('initial_h_values.txt')
    n_initial = len(initial_h_values)
    n_points = 200  # Higher resolution for better optimization
    dx = 2.0 / n_points

    # Interpolate initial_h_values to the new n_points
    x_initial = np.linspace(0, 2, n_initial)
    x_new = np.linspace(0, 2, n_points)
    initial_h = np.interp(x_new, x_initial, initial_h_values)

    # Define the optimization function
    def objective(h):
        h = np.array(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Constraint functions
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_bounds(h):
        return np.all(0 <= h) and np.all(h <= 1)

    bounds = Bounds(0, 1)
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    # Initial guess
    initial_guess = initial_h.copy()
    initial_guess = initial_guess / np.sum(initial_guess) * n_points / 2

    # Optimization settings
    max_iter = 200
    tol = 1e-6
    evals = 0

    start_time = time()

    def run_optimization():
        nonlocal evals
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': tol, 'gtol': tol, 'disp': False}
        )
        return result.x

    best_h = initial_guess
    best_c5 = objective(best_h)

    # Run multiple optimizations with different initializations
    for _ in range(5):
        if time() - start_time > budget_s - 1:
            break
        new_h = run_optimization()
        new_c5 = objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5

    # Use a more aggressive optimization to refine the solution
    refined_h = best_h.copy()
    refined_h = refined_h / np.sum(refined_h) * n_points / 2
    refined_result = np.correlate(refined_h, 1 - refined_h, mode="full")
    refined_c5 = np.max(refined_result) * dx

    final_h = refined_h
    final_c5 = refined_c5
    return (final_h, final_c5, len(final_h))
```
