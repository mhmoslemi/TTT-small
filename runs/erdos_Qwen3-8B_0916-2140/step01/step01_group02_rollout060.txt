import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a better initial guess based on known constructions
    n_points = 200
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    
    # Use a more refined initial guess inspired by optimal constructions
    for i in range(n_points):
        x = i * dx
        # Use a piecewise linear function with periodicity
        if x < 0.5:
            initial_h[i] = 1.0
        elif x < 1.0:
            initial_h[i] = 0.5
        elif x < 1.5:
            initial_h[i] = 0.0
        elif x < 2.0:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 1.0
    
    # Ensure the integral constraint is satisfied
    total = np.sum(initial_h) * dx
    scaling = 1.0 / total
    initial_h *= scaling
    
    def objective(x):
        h = x
        # Compute the overlap integral for all k
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # We'll use a more efficient optimization procedure
    # We will perform multiple rounds of optimization with different starting points
    # to avoid local optima
    
    # First optimization round
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="L-BFGS-B",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Second optimization round to refine the solution
    result = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="COBYLA",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Third optimization round with a different method
    result = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Return the best solution found
    return best_h, best_c5, n_points
