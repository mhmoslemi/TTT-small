```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values(n=70):
    # Initial construction (sample with 70 points)
    h = np.zeros(n)
    for i in range(n):
        if i < 10:
            h[i] = 0.3
        elif i < 20:
            h[i] = 0.6
        elif i < 30:
            h[i] = 0.9
        elif i < 40:
            h[i] = 0.3
        elif i < 50:
            h[i] = 0.6
        elif i < 60:
            h[i] = 0.9
        else:
            h[i] = 0.3
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a moderate number of points for balance
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial construction
    initial_h = initial_h_values(n_points)
    
    # Function to evaluate the overlap integral
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) * dx == 1 (equivalently sum(h) == n_points / 2)
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] is [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization setup
    result = minimize(
        objective,
        initial_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint_func},
        method="SLSQP",
        options={"ftol": 1e-10, "maxiter": 200}
    )
    
    # If optimization fails, try a different initial guess
    if not result.success:
        # Use a random perturbation of the initial guess
        perturbed_h = initial_h + np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        result = minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint_func},
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 200}
        )
    
    # If still not successful, fallback to initial_h
    if not result.success:
        c5 = evaluate_erdos_solution(initial_h)
        return (initial_h.tolist(), c5, n_points)
    
    best_h = result.x
    c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), c5, n_points)
```
