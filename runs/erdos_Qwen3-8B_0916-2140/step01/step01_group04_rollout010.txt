        return np.max(overlaps) * dx

        bounds = [(0, 1) for _ in range(n_points)]
        dx = 2.0 / n_points
        sum_h = n_points / 2
        
        def objective(x):
            return np.max(np.correlate(x, 1 - x, mode="full")) * dx
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - sum_h}]
        
        result = opt.minimize(objective, h_initial, bounds=bounds, constraints=constraints, method='SLSQP')
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    best_h = h_initial
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    # Evaluate initial solution
    print(f"Initial C5: {best_c5:.6f} with {best_n_points} points")
    
    # Try 2x, 3x, and 4x the initial n_points
    target_n_points = [n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]
    
    start_time = time.time()
    
    for n_points in target_n_points:
        if time.time() - start_time > budget_s - 10:
            break
        
        try:
            h_opt, c5_opt = optimize_h(n_points)
