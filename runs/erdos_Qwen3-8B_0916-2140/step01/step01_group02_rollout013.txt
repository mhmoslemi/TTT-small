```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Try with different n_points to explore various scales
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess: piecewise constant function with varying intervals
    h_values = np.zeros(n_points)
    
    # Initial guess based on a more refined pattern
    for i in range(n_points):
        pos = i / n_points
        if 0.0 <= pos < 0.2:
            h_values[i] = 1.0
        elif 0.2 <= pos < 0.4:
            h_values[i] = 0.5
        elif 0.4 <= pos < 0.6:
            h_values[i] = 0.0
        elif 0.6 <= pos < 0.8:
            h_values[i] = 0.5
        elif 0.8 <= pos <= 1.0:
            h_values[i] = 1.0
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with more iterations and tighter tolerances
    result = minimize(
        objective,
        h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 150, "ftol": 1e-9, "eps": 1e-9},
        callback=lambda x: print(f"Current C5: {objective(x)}")
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    return best_h, best_c5, n_points
```
