import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 500  # Using 500 points for better resolution
    dx = 2.0 / n_points

    # Initial construction based on a piecewise constant function
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h_values[i] = 0.0
        elif x < 1.0:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Initial guess
    x0 = initial_h_values.copy()

    # Optimization parameters
    max_iter = 100
    tolerance = 1e-8
    time_limit = budget_s

    start_time = time.time()

    # Perform a local optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': tolerance, 'maxiter': max_iter})

    best_h = result.x
    best_c5 = compute_overlap(best_h)

    # Optionally perform multiple restarts with different initial guesses
    # This is commented out to keep the code efficient, but can be uncommented for better exploration

    #     current_c5 = compute_overlap(result.x)
    #     if current_c5 < best_c0:
    #         best_h = result.x
    #         best_c5 = current_c5

    # Ensure we stop before the time limit
    elapsed_time = time.time() - start_time
    if elapsed_time > time_limit:
        print("Time limit exceeded, returning best solution so far")
    
    return (best_h.tolist(), best_c5, n_points)
