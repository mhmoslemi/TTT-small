import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to generate a new initial guess with perturbations
    def perturb_initial_guess(h, perturbation_factor=0.05):
        h_perturbed = h.copy()
        for i in range(len(h_perturbed)):
            h_perturbed[i] += np.random.uniform(-perturbation_factor, perturbation_factor)
            h_perturbed[i] = np.clip(h_perturbed[i], 0, 1)
        return h_perturbed
    
    # Initial guess
    h0 = initial_h
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    time_budget = budget_s
    start_time = time.time()
    
    # Function to run optimization with a given initial guess
    def optimize_with_guess(initial_guess):
        result = optimize.minimize(
            objective,
            initial_guess,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tolerance, 'maxiter': 100}
        )
        if result.success:
            return result.x, compute_C5(result.x)
        else:
            return initial_guess, compute_C5(initial_guess)
    
    # Best solution found so far
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Run optimization with multiple initial guesses
    for _ in range(5):
        # Perturb the initial guess
        h_perturbed = perturb_initial_guess(h0)
        current_h, current_C5 = optimize_with_guess(h_perturbed)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            h0 = current_h
            print(f"New best C5: {best_C5}")
    
    # Check time budget
    if time.time() - start_time > time_budget:
        print(f"Time budget exceeded. Returning best found solution.")
    
    return best_h, best_C5, len(best_h)
