import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    
    # Initial guess: piecewise constant function with 20 intervals
    for i in range(n_points):
        if i % 20 == 0:
            h_values[i] = 1.0
        elif i % 20 == 1:
            h_values[i] = 0.75
        elif i % 20 == 2:
            h_values[i] = 0.5
        elif i % 20 == 3:
            h_values[i] = 0.25
        elif i % 20 == 4:
            h_values[i] = 0.0
        elif i % 20 == 5:
            h_values[i] = 0.25
        elif i % 20 == 6:
            h_values[i] = 0.5
        elif i % 20 == 7:
            h_values[i] = 0.75
        elif i % 20 == 8:
            h_values[i] = 1.0
        elif i % 20 == 9:
            h_values[i] = 0.75
        elif i % 20 == 10:
            h_values[i] = 0.5
        elif i % 20 == 11:
            h_values[i] = 0.25
        elif i % 20 == 12:
            h_values[i] = 0.0
        elif i % 20 == 13:
            h_values[i] = 0.25
        elif i % 20 == 14:
            h_values[i] = 0.5
        elif i % 20 == 15:
            h_values[i] = 0.75
        elif i % 20 == 16:
            h_values[i] = 1.0
        elif i % 20 == 17:
            h_values[i] = 0.75
        elif i % 20 == 18:
            h_values[i] = 0.5
        elif i % 20 == 19:
            h_values[i] = 0.25
    
    h_values = np.clip(h_values, 0, 1)
    
    # Ensure the integral constraint is satisfied
    total = np.sum(h_values) * dx
    scaling = 1.0 / total
    h_values *= scaling
    
    def objective(x):
        h = x
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(x):
        return np.sum(x) * dx - 1.0
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with multiple restarts
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    for restart in range(5):
        result = minimize(
            objective,
            h_values,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8},
            callback=lambda x: print(f"Current C5: {objective(x)}")
        )
        
        if result.success:
            current_c5 = objective(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
