    def optimize_with_params(n_points, initial_guess=None):
        # Define the step function as a vector h of size n_points
        # with h[i] in [0, 1] and sum(h) = n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function: minimize the maximum overlap
        def objective(h):
            dx = 2.0 / n_points
            overlaps = np.correlate(h, 1 - h, mode="full")
            return np.max(overlaps) * dx
        
        # Constraint: sum(h) == n_points / 2
        constraint_sum = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        
        # Optimization parameters
        options = {'maxiter': 1000, 'disp': False}
        
        # Initial guess
        if initial_guess is None:
            initial_guess = np.random.rand(n_points)
            initial_guess = (initial_guess * (n_points / 2)) / np.sum(initial_guess)
        
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=[constraint_sum], options=options)
        
        if result.success:
            return result.x, result.fun
        else:
            raise ValueError("Optimization failed.")
    
    best_h = initial_h
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    # Try a range of point counts
    n_points_values = [n_points_initial, n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]
