def run(seed=42, budget_s=600, **kwargs):

    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Define a function to generate a new candidate solution
    def generate_candidate(h):
        # Perturb current solution slightly
        perturb = np.random.normal(0, 0.05, size=len(h))
        perturb = np.clip(perturb, -0.1, 0.1)
        new_h = np.clip(h + perturb, 0, 1)
        return new_h
    
    # Optimization loop
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # We'll run multiple iterations of local search and simulated annealing
    max_iter = 100
    temp = 1.0
    temp_decay = 0.995
    time_remaining = time_budget
    
    for _ in range(max_iter):
        # Try local search with perturbation
        current_h = generate_candidate(best_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f"New best C5: {best_C5}")
        
        # Simulated annealing step
        if np.random.rand() < 0.3:
            new_h = generate_candidate(best_h)
            new_C5 = compute_C5(new_h)
            delta = new_C5 - best_C5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_h = new_h
                best_C5 = new_C5
                print(f"Annealing step: New best C5: {best_C5}")
        
        # Update temperature
        temp *= temp_decay
        
        # Check time budget
        if time.time() - start_time > time_remaining:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
