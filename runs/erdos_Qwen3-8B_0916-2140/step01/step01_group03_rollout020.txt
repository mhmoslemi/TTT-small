import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Function to compute the maximum overlap integral for all shifts
    def compute_max_overlap(h):
        h = np.array(h)
        shifts = np.arange(0, 2, dx)
        max_overlap = 0.0
        for k in shifts:
            if k == 0:
                continue
            overlap = np.sum(h * (1 - h[k:])) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        return compute_max_overlap(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-6
    time_budget = budget_s
    start_time = time.time()
    
    # Optimization loop
    best_h = h0
    best_C5 = compute_max_overlap(h0)
    
    # Try different starting points
    for _ in range(3):
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h0 = np.clip(h0 + perturbation, 0, 1)
        
        # Optimization
        result = optimize.minimize(
            objective,
            h0,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': tolerance, 'maxiter': 100}
        )
        
        if result.success:
            current_h = result.x
            current_C5 = compute_max_overlap(current_h)
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
                print(f"New best C5: {best_C5}")
        else:
            print("Optimization failed.")
            break
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
