    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    def minimize_with_restart(x0, max_iter=150, restarts=5):
        best_x = x0
        best_val = compute_overlap(x0)
        for _ in range(restarts):
            perturbation = np.random.uniform(-0.05, 0.05, n_points)
            perturbed_x = np.clip(x0 + perturbation, 0, 1)
            result = opt.minimize(objective, perturbed_x, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': max_iter})

                best_val = compute_overlap(result.x)
                best_x = result.x
        return best_x, best_val

    # First optimization with initial point
    best_x, best_val = minimize_with_restart(initial_h)

    # Try fine-tuning with different starting points
    for _ in range(3):
