import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting from initial_h_values, but we'll try different starting points
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to perform a more aggressive perturbation
    def perturb(h):
        h = h.copy()
        # Add random noise to perturb the current solution
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a more thorough local search
    def local_search(h):
        best_h = h
        best_C5 = compute_C5(h)
        for _ in range(5):
            # Perturb the current solution
            perturbed_h = perturb(h)
            # Optimize from the perturbed point
            current_h = optimize_step(perturbed_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
            h = current_h
        return best_h
    
    # Function to perform a more strategic search with different starting points
    def strategic_search(h):
        best_h = h
        best_C5 = compute_C5(h)
        
        # Try a few perturbations and optimize from there
        for _ in range(3):
            # Random perturbation
            perturbed_h = h.copy()
            noise = np.random.normal(0, 0.05, size=len(h))
            perturbed_h += noise
            perturbed_h = np.clip(perturbed_h, 0, 1)
            
            # Optimize from the perturbed point
            current_h = optimize_step(perturbed_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
        
        # Try a more structured perturbation
        structured_perturb = np.array([0.5 if i % 5 == 0 else 0 for i in range(n_points)])
        perturbed_h = h.copy()
        perturbed_h = np.clip(perturbed_h + structured_perturb, 0, 1)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
        
        return best_h
    
    # Perform the main optimization
    best_h = h0
    best_C5 = compute_C5(h0)
    best_n_points = n_points
    
    # Try different starting points to avoid local optima
    for _ in range(3):
        # Use a different starting point
        h_start = np.random.uniform(0, 1, size=n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Perform a full optimization from this starting point
        h_current = h_start
        for _ in range(20):
            h_current = local_search(h_current)
            current_C5 = compute_C5(h_current)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h_current
                best_n_points = n_points
            # Check time budget
            if time.time() - start_time > time_budget:
                print(f"Time budget exceeded. Returning best found solution.")
                break
    
    return best_h, best_C5, best_n_points
