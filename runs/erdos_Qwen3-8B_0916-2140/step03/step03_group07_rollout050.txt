    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint function
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Helper function for optimization
    def optimize_with_perturbations(h0, max_iter=150, perturbation_scale=0.05):
        result = opt.minimize(objective, h0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': max_iter})
        best_h = result.x
        best_c5 = compute_overlap(best_h)
        
        # Perturb and re-optimize
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': max_iter})
        best_h = result.x
        best_c5 = compute_overlap(best_h)
        
        return best_h, best_c5
