        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial optimization with SLSQP
    def optimize_with_slsqp(start_h):
        result = minimize(objective, start_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        return result.x, result.fun
    
    # Use differential evolution to find better initial points
    def objective_for_de(x):
        h = np.array(x, dtype=np.float64)
        return compute_overlap(h)
    
    def constraint_sum_for_de(x):
        return np.sum(x) - n_points / 2
    
    cons_de = ({'type': 'eq', 'fun': constraint_sum_for_de})
    
    de_bounds = [(0, 1) for _ in range(n_points)]
    
    result_de = differential_evolution(objective_for_de, de_bounds, constraints=cons_de, strategy='best1bin', popsize=20, maxiter=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, callback=None, disp=False)
    best_de_h = result_de.x
    best_de_c5 = compute_overlap(best_de_h)
    
    # Now perform SLSQP optimization from the DE result
    best_slsqp_h, best_slsqp_c5 = optimize_with_slsqp(best_de_h)

    if budget_s > 300:
        # Create a perturbed version of the best_slsqp_h to avoid local minima

        result_perturbed = optimize_with_slsqp(perturbed_h)
