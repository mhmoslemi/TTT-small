    def compute_overlap(h, n_points, dx):

        return np.max(corr[1:-1]) * dx

    def optimize_solution(h_initial, n_points, dx, budget_s):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(x):
            return compute_overlap(x, n_points, dx)

            return np.sum(x) - n_points / 2
        
        cons = ({'type': 'eq', 'fun': constraint_sum})

            objective,

            bounds=bounds, 
            constraints=cons, 
            method='SLSQP', 
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        
        best_h = result.x
        best_c5 = compute_overlap(best_h, n_points, dx)
        return best_h, best_c5

    best_h = initial_h
    best_c5 = compute_overlap(best_h, initial_n_points, 2.0 / initial_n_points)

    if budget_s > 300:
        perturb = np.random.uniform(-0.1, 0.1, initial_n_points)
