import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a coarser grid for faster exploration
    n_points = 100
    dx = 2.0 / n_points
    
    # Define the initial guess as a piecewise constant function
    # Inspired by the known construction but with slight variations
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if x <= 0.5:
            initial_h[i] = 0.5
        elif x <= 1.0:
            initial_h[i] = 0.75
        elif x <= 1.5:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 0.25
    
    # Define the overlap function
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    # Define the objective function
    def objective(x):
        return compute_overlap(x)
    
    # Define the constraint function
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up the optimization problem
    cons = ({'type': 'eq', 'fun': constraint_sum})
    result = opt.minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Try a few random perturbations to escape local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        if compute_overlap(result.x) < best_c5:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Refine with a finer grid
    n_points = 200
    dx = 2.0 / n_points
    initial_h_refined = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
    initial_h_refined = np.clip(initial_h_refined, 0, 1)
    
    # Define the overlap function for the refined grid
    def compute_overlap_refined(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Define the objective function for the refined grid
    def objective_refined(x):
        return compute_overlap_refined(x)
    
    # Define the constraint function for the refined grid
    def constraint_sum_refined(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds for the refined grid
    bounds_refined = [(0, 1) for _ in range(n_points)]
    
    # Set up the refined optimization problem
    cons_refined = ({'type': 'eq', 'fun': constraint_sum_refined})
    result_refined = opt.minimize(objective_refined, initial_h_refined, bounds=bounds_refined, constraints=cons_refined, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    
    best_h_refined = result_refined.x
    best_c5_refined = compute_overlap_refined(best_h_refined)
    
    return (best_h_refined.tolist(), best_c5_refined, n_points)
