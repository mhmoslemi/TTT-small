import scipy.optimize as optimize
import time

    def compute_C5(h):

        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)

        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()

    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100, 'disp': False}
        )
        return result.x if result.success else h
    
    # Function to perform a more aggressive perturbation
    def perturb(h):
        h = h.copy()
        # Add random noise to perturb the current solution
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        return h

    def local_search(h):
        best_h = h

            perturbed_h = perturb(h)
            # Optimize from the perturbed point
            current_h = optimize_step(perturbed_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
