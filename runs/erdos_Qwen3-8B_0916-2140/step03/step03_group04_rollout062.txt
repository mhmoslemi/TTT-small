import scipy.optimize as opt
from scipy.optimize import Bounds, minimize
import time

def run(seed=42, budget_s=600):

    n_points_initial = len(initial_h)
    n_points = 800  # Double the points for better resolution
    
    # Convert initial_h to n_points samples
    x_initial = np.linspace(0, 2, n_points_initial)
    x_new = np.linspace(0, 2, n_points)
    initial_h_new = np.interp(x_new, x_initial, initial_h)
    initial_h_new = np.clip(initial_h_new, 0, 1)
    
    dx = 2.0 / n_points
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization with SLSQP
    result = minimize(objective, initial_h_new, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # If budget allows, perform a second optimization with a different starting point
    if budget_s > 300:
        # Create a perturbed version of the best_h to avoid local minima
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        
        # Second optimization
        result_perturbed = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        best_h_perturbed = result_perturbed.x
        best_c5_perturbed = compute_overlap(best_h_perturbed)

        if best_c5_perturbed < best_c5:

    if budget_s > 400:
