    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function for optimization
    def objective(x):
        return compute_overlap(x)
    
    # Constraint function for total integral
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization
    result = opt.minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Define a perturbation function to explore the search space
    def perturb(h, sigma=0.05):
        perturbed = np.clip(h + np.random.normal(0, sigma, h.shape), 0, 1)
        return perturbed
    
    # Perform multiple perturbation-based optimizations
    for _ in range(3):
        if budget_s <= time.time():
            break
        perturbed = perturb(best_h)
        result = opt.minimize(objective, perturbed, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
