    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Function to apply a perturbation
    def perturb(h, magnitude=0.05):
        perturbations = np.random.normal(0, magnitude, len(h))
        return np.clip(h + perturbations, 0, 1)
    
    # Function to perform a local search
    def local_search(h, max_iter=100, perturb_magnitude=0.05):
        current_h = h.copy()
        best_h = current_h
        best_c5 = compute_overlap(current_h)
        
        for _ in range(max_iter):
            perturbed_h = perturb(current_h, perturb_magnitude)
            c5 = compute_overlap(perturbed_h)
            
            if c5 < best_c5:
                best_h = perturbed_h
                best_c5 = c5
                current_h = best_h
            
            perturb_magnitude *= 0.99  # Decrease perturbation magnitude over time
        
        return best_h, best_c5
    
    # Function to perform a global search with multiple starting points
    def global_search(initial_h, num_starts=5, max_iter=100, perturb_magnitude=0.05):
        best_h = initial_h.copy()
        best_c5 = compute_overlap(best_h)
        
        for _ in range(num_starts):
            current_h = perturb(initial_h, perturb_magnitude)
            current_h = np.clip(current_h, 0, 1)
            perturbed_h, c5 = local_search(current_h, max_iter, perturb_magnitude)
            
            if c5 < best_c5:
                best_h = perturbed_h
                best_c5 = c5
        
        return best_h, best_c5
    
    # Perform a global search
    best_h, best_c5 = global_search(initial_h)
