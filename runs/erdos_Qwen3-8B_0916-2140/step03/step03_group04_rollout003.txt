    n_points = 400
    dx = 2.0 / n_points
    
    # We'll start from the initial_h_values provided (n=400)

        corr = np.correlate(h, 1 - h, mode="full")

    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Refine using a different solver with more iterations
    if budget_s > 300:
        result_refined = minimize(objective, best_h, bounds=bounds, constraints=cons, method='COBYLA', options={'ftol': 1e-10, 'maxiter': 500})

        best_c5_refined = compute_overlap(best_h_refined)

            best_h = best_h_refined
