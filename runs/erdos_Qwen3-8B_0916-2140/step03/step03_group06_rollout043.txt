import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points

    # Start with initial guess
    initial_h_values = np.array([0.0] * n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0

    initial_h = initial_h_values.copy()

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Initial optimization with SLSQP
    x0 = initial_h.copy()
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 150})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)

    # Perform a grid search with perturbations
    n_starts = 20
    for _ in range(n_starts):
        perturb = np.random.normal(0, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)
        result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 150})
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Optional refinement with a more robust optimizer
    final_result = minimize(objective, best_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = final_result.x
    best_c5 = compute_overlap(best_h)

    return (best_h.tolist(), best_c5, n_points)
