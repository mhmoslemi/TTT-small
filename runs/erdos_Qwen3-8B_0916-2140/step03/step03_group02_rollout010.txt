import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=Bounds(0, 1),
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to create a random perturbation
    def perturb(h):
        perturbation = np.random.normal(0, 0.05, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    # Function to apply a small step in the direction of the gradient
    def gradient_step(h):
        h = np.array(h)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * 1e-4)
        step_size = 0.1
        return np.clip(h - step_size * grad, 0, 1)
    
    # Main optimization loop
    best_h = h0
    best_C5 = compute_C5(h0)
    iterations = 0
    max_iterations = 100
    warm_start = True
    
    while time.time() - start_time < time_budget and iterations < max_iterations:
        # Try different perturbation strategies
        if np.random.rand() < 0.5:
            # Random perturbation
            perturbed_h = perturb(best_h)
        else:
            # Gradient step
            perturbed_h = gradient_step(best_h)
        
        # Optimization
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f'New best C5: {best_C5}')
        
        iterations += 1
    
    return best_h, best_C5, len(best_h)
