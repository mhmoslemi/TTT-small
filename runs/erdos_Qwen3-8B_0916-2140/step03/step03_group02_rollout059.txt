import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Function to evaluate the quality of a solution
    def evaluate(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    best_h = h0
    best_C5 = evaluate(h0)
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step using Sequential Least Squares Programming (SLSQP)
    def optimize_step(h):
        result = optimize.minimize(
            evaluate,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to create a random perturbation
    def perturb(h):
        perturbation = np.random.normal(0, 0.05, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    # Main optimization loop with multiple restarts
    max_restarts = 10
    restarts = 0
    while time.time() - start_time < time_budget and restarts < max_restarts:
        # Generate a new random initial guess
        h_initial = np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimize from this initial guess
        current_h = optimize_step(h_initial)
        current_C5 = evaluate(current_h)
        
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f'New best C5: {best_C5}')
        
        restarts += 1
    
    return best_h, best_C5, len(best_h)
