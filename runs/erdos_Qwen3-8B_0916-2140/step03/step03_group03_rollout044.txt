    def compute_overlap(h, dx):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function to minimize
    def objective(x, dx):
        return compute_overlap(x, dx)
    
    # Constraint function for integral
    def constraint_sum(x, n_points, dx):

    # Function to perform optimization with constraints
    def optimize_h(n_points, dx, initial_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = ({'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points, dx)})
        result = minimize(objective, initial_guess, args=(dx,), bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x, result.fun

    n_points_list = [100, 150, 200, 250]

    for n_points in n_points_list:

        x_initial = np.linspace(0, 2, len(initial_h))

        # Run optimization
        start_time = time.time()
        result = optimize_h(n_points, dx, initial_h_new)

        if elapsed_time < budget_s:
