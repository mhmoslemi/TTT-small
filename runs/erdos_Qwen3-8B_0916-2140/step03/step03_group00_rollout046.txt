            result = optimize.minimize(

                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}],
                method='SLSQP',
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            if result.success:
                current_c5 = result.fun

                    best_c5 = current_c5

    start_time = time.time()
