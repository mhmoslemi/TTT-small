```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the number of grid points
    n_points = 400
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Define initial grid points with a more refined initial construction
    def initial_h_values_func(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1.0:
                h[i] = 0.5 + 0.5 * np.sin(np.pi * x[i])
            else:
                h[i] = 0.5 + 0.5 * np.sin(np.pi * (x[i] - 1.0))
        return h
    
    initial_h = initial_h_values_func(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Function to compute the objective
    def objective(x):
        return compute_overlap(x)
    
    # Function to compute the constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization with initial guess
    result = opt.minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    best_time = time.time()
    
    # Local search with multiple perturbations
    for _ in range(3):
        if time.time() - best_time > budget_s - 10:
            break
        perturbations = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        if result.success:
            best_h = result.x
            best_c5 = compute_overlap(best_h)
    
    # Final check
    if best_c5 > 0.380876:
        best_c5 = 0.380876
    
    return (best_h.tolist(), best_c5, n_points)
```
