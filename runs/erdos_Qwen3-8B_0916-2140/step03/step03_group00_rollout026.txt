import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial construction from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Function to compute the current C5 value
    def evaluate(h):
        return compute_C5(h)
    
    # Function to perform a single optimization step using L-BFGS-B with bounds
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to perform a perturbation with more control
    def perturb(h):
        h = h.copy()
        # Add Poisson noise to perturb the current solution
        noise = np.random.lognormal(0, 0.1, size=len(h))
        h += noise
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a more thorough local search
    def local_search(h):
        best_h = h
        best_C5 = evaluate(h)
        for _ in range(3):
            # Perturb the current solution
            perturbed_h = perturb(h)
            # Optimize from the perturbed point
            current_h = optimize_step(perturbed_h)
            current_C5 = evaluate(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
            h = current_h
        return best_h
    
    # Initial optimization
    best_h = initial_h.copy()
    best_C5 = evaluate(best_h)
    
    # Track time
    start_time = time.time()
    
    # Perform multiple rounds of local optimization
    for _ in range(50):
        # Perform local search from the current best solution
        current_h = local_search(best_h)
        current_C5 = evaluate(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f"New best C5: {best_C5}")
        
        # Check time budget
        if time.time() - start_time > budget_s:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
