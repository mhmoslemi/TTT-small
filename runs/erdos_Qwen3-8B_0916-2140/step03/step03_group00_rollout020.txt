import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step using L-BFGS-B
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to perturb the current solution
    def perturb(h):
        h = h.copy()
        noise = np.random.normal(0, 0.05, size=len(h))
        h += noise
        h = np.clip(h, 0, 1)
        return h
    
    # Function to perform a more thorough local search
    def local_search(h):
        best_h = h
        best_C5 = compute_C5(h)
        for _ in range(5):
            # Perturb the current solution
            perturbed_h = perturb(h)
            # Optimize from the perturbed point
            current_h = optimize_step(perturbed_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
            h = current_h
        return best_h
    
    # Function to perform a more aggressive restart
    def restart_search(h):
        best_h = h
        best_C5 = compute_C5(h)
        for _ in range(3):
            # Random restart
            random_h = np.random.uniform(0, 1, size=n_points)
            # Enforce integral constraint
            random_h = random_h * (n_points / 2) / np.sum(random_h)
            current_h = optimize_step(random_h)
            current_C5 = compute_C5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h
        return best_h
    
    # Initial optimization
    best_h = h0
    best_C5 = compute_C5(h0)
    
    # Perform multiple rounds of optimization and restarts
    for _ in range(30):
        # Perform local search from the current best solution
        current_h = local_search(best_h)
        current_C5 = compute_C5(current_h)
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f"New best C5: {best_C5}")
        
        # Perform a restart
        restarted_h = restart_search(best_h)
        restarted_C5 = compute_C5(restarted_h)
        
        if restarted_C5 < best_C5:
            best_C5 = restarted_C5
            best_h = restarted_h
            print(f"New best C5 after restart: {best_C5}")
        
        # Check time budget
        if time.time() - start_time > time_budget:
            print(f"Time budget exceeded. Returning best found solution.")
            break
    
    return best_h, best_C5, len(best_h)
