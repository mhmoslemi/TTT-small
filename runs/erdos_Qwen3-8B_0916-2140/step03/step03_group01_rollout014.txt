import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Using 200 points for better resolution
    dx = 2.0 / n_points

    # Initial guess based on known patterns for the Erdős problem
    initial_h_values = np.array([0.0]*n_points)
    for i in range(n_points):
        x = i * dx
        if x < 0.5:
            initial_h_values[i] = 0.0
        elif x < 1.0:
            initial_h_values[i] = 0.5
        else:
            initial_h_values[i] = 1.0

    initial_h = initial_h_values.copy()

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # First optimization step with SLSQP
    x0 = initial_h.copy()
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)

    # Refinement with adaptive perturbation and multi-start
    n_starts = 20
    for _ in range(n_starts):
        # Random perturbation
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)

        # Optimizing with reduced tolerance and more iterations
        result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 200})
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Final refinement with genetic algorithm for global search
    from scipy.optimize import differential_evolution

    def objective_de(x):
        return compute_overlap(x)

    result_de = differential_evolution(objective_de, bounds, strategy='best1bin', maxiter=50, popsize=20, tol=1e-8, mutation=(0.5, 1), recombination=0.9)
    final_h = result_de.x
    final_c5 = compute_overlap(final_h)

    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h

    return (best_h.tolist(), best_c5, n_points)
