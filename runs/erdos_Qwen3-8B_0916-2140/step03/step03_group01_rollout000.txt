import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Using 200 points for better resolution
    dx = 2.0 / n_points
    
    # Define initial construction
    def initial_h_values():
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1:
                h[i] = 0.5 * (1 - np.cos(np.pi * x[i]))
            else:
                h[i] = 0.5 * (1 + np.cos(np.pi * x[i]))
        return h
    
    initial_h = initial_h_values()
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Start with initial guess
    x0 = initial_h.copy()
    
    # First optimization step with SLSQP
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)
    
    # Multi-start optimization with adaptive refinement and different initial guesses
    n_starts = 10
    best_c5 = float('inf')
    
    for _ in range(n_starts):
        # Perturb the current best guess with different patterns
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)
        
        # Use a different optimization method for diversity
        if np.random.rand() < 0.5:
            result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method='Nelder-Mead', options={'ftol': 1e-8, 'maxiter': 150})
        else:
            result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 200})
        
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    # Optionally refine with a higher-resolution interpolation
    # Here, we keep n_points = 200 for performance
    
    return (best_h.tolist(), best_c5, n_points)
