import scipy.optimize as opt

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint function
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # First optimization
    result = opt.minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # If time allows, perform a local search with perturbed initial points
    if budget_s > 300:
        perturbations = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        best_h = result.x
        best_c5 = compute_overlap(best_h)
    
    # Try a different initial construction: step function with alternating regions
    def create_alternating_h(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 0.5
            elif i % 3 == 1:
                h[i] = 0.75
            else:
                h[i] = 0.25
        return h
    
    alt_h = create_alternating_h(n_points)
    result = opt.minimize(objective, alt_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
    alt_best_h = result.x
    alt_best_c5 = compute_overlap(alt_best_h)
    
    if alt_best_c5 < best_c5:
        best_h = alt_best_h
        best_c5 = alt_best_c5
    
    # Try a different initial construction: piecewise linear with periodicity
    def create_periodic_h(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            val = 0.5 + 0.25 * np.sin(2 * np.pi * i / n_points)

        return h

    result = opt.minimize(objective, per_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})

    per_best_c5 = compute_overlap(per_best_h)
    
    if per_best_c5 < best_c5:
        best_h = per_best_h
        best_c5 = per_best_c5
    
    return (best_h.tolist(), best_c5, n_points)
