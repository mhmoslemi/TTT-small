import numpy as np
import scipy.optimize as opt

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 200  # Resolution for the step function
    dx = 2.0 / n_points
    
    # Load initial construction
    initial_h_values = np.array(initial_h_values, dtype=np.float64)
    
    # If initial_h has different length, interpolate
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h = np.clip(initial_h_values, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial guess
    x0 = initial_h.copy()
    
    # Use different optimization methods to avoid local optima
    def optimize_with(method, x0, **kwargs):
        try:
            result = minimize(objective, x0, bounds=bounds, constraints=cons, method=method, **kwargs)
            return result.x
        except:
            return x0
    
    # First optimization with SLSQP
    opt_result = optimize_with('SLSQP', x0, options={'ftol': 1e-8, 'maxiter': 150})
    best_h = opt_result
    best_c5 = compute_overlap(best_h)
    
    # Try a different method for refinement
    opt_result = optimize_with('COBYLA', best_h, options={'ftol': 1e-8, 'maxiter': 150})
    best_h = opt_result
    best_c5 = compute_overlap(best_h)
    
    # Try a constrained optimization with different parameters
    opt_result = optimize_with('Nelder-Mead', best_h, options={'ftol': 1e-8, 'maxiter': 150})
    best_h = opt_result
    best_c5 = compute_overlap(best_h)
    
    # Multi-start optimization with perturbations
    n_starts = 10
    best_c5 = float('inf')
    
    for _ in range(n_starts):
        # Perturb the initial guess slightly
        perturb = np.random.normal(0, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)
        
        # Perform a local optimization
        result = optimize_with('SLSQP', x0_perturb, options={'ftol': 1e-8, 'maxiter': 150})
        current_c5 = compute_overlap(result)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
    
    # Optional refinement with a more robust optimizer
    # We'll perform a final optimization with tighter tolerances
    final_result = optimize_with('SLSQP', best_h, options={'ftol': 1e-10, 'maxiter': 200})
    best_h = final_result
    best_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
