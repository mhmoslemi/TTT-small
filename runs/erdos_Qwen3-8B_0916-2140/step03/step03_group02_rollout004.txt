import numpy as np
import scipy.optimize as optimize
import time
import matplotlib.pyplot as plt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Function to perform a single optimization step
    def optimize_step(h):
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to create a random perturbation
    def perturb(h):
        perturbation = np.random.normal(0, 0.05, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    # Function to create a structured perturbation
    def structured_perturb(h):
        perturbation = np.sin(np.linspace(0, 2 * np.pi, len(h))) * 0.05
        return np.clip(h + perturbation, 0, 1)
    
    # Main optimization loop
    best_h = h0
    best_C5 = compute_C5(h0)
    iterations = 0
    max_iterations = 1000
    
    while time.time() - start_time < time_budget and iterations < max_iterations:
        # Try both random and structured perturbations
        perturbed_h = perturb(best_h)
        current_h = optimize_step(perturbed_h)
        current_C5 = compute_C5(current_h)
        
        # Compare with structured perturbation
        structured_perturbed_h = structured_perturb(best_h)
        structured_current_h = optimize_step(structured_perturbed_h)
        structured_current_C5 = compute_C5(structured_current_h)
        
        # Choose the better solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = current_h
            print(f'Random perturbation: New best C5: {best_C5}')
        
        if structured_current_C5 < best_C5:
            best_C5 = structured_current_C5
            best_h = structured_current_h
            print(f'Structured perturbation: New best C5: {best_C5}')
        
        iterations += 1
    
    return best_h, best_C5, len(best_h)
