import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import Bounds, minimize, differential_evolution
from scipy.integrate import trapz

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with a reasonable n_points
    n_points = 200
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)  # Placeholder for initial values
    
    # Define the initial guess from the given code (if it's available in the environment)
    if hasattr(evaluate_erdos_solution, 'initial_h_values'):
        initial_h_values = np.array(evaluate_erdos_solution.initial_h_values, dtype=np.float64)
    
    # Ensure the initial_h_values is of the correct length
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    # Clip to [0, 1]
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Main optimization routine
    def optimize_with_multiple_start(x0, n_starts=10, method='SLSQP', **kwargs):
        best_c5 = float('inf')
        best_h = x0.copy()
        
        for _ in range(n_starts):
            # Perturb the initial guess
            perturb = np.random.uniform(-0.05, 0.05, n_points)
            x0_perturb = np.clip(x0 + perturb, 0, 1)
            
            # Perform a local optimization with adaptive tolerance
            result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method=method, **kwargs)
            current_c5 = compute_overlap(result.x)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
        
        return best_h, best_c5
    
    # First optimization with SLSQP
    x0 = initial_h_values.copy()
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)
    
    # Multi-start optimization with adaptive step size
    best_h, best_c5 = optimize_with_multiple_start(best_h, n_starts=10, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 200})
    
    # Coarse optimization with differential evolution to escape local optima
    def coarse_objective(x):
        return compute_overlap(x)
    
    def coarse_constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons_coarse = ({'type': 'eq', 'fun': coarse_constraint_sum})
    
    # Coarse optimization with differential evolution
    result = differential_evolution(coarse_objective, bounds=bounds, constraints=cons_coarse, strategy='best1bin', popsize=20, maxiter=50, tol=1e-6, mutation=(0.5, 1), recombination=0.8, polish=True,disp=False)
    coarse_h = result.x
    coarse_c5 = compute_overlap(coarse_h)
    
    # Compare and pick the best solution
    if coarse_c5 < best_c5:
        best_h = coarse_h
        best_c5 = coarse_c5
    
    return (best_h.tolist(), best_c5, n_points)
