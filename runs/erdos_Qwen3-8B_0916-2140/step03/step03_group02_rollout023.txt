import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial configuration from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_C5(h):
        h = np.array(h)
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result * dx)
        return c5
    
    # Objective function to minimize C5
    def objective(h):
        return compute_C5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Initial guess
    h0 = initial_h.copy()
    
    # Optimization settings
    time_budget = budget_s
    start_time = time.time()
    
    # Define bounds for the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to perform a single optimization step with a specified method
    def optimize_step(h, method='SLSQP'):
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method=method,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x if result.success else h
    
    # Function to create a random perturbation
    def perturb(h):
        perturbation = np.random.normal(0, 0.05, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    # Main optimization loop with adaptive cooling
    best_h = h0
    best_C5 = compute_C5(h0)
    iterations = 0
    max_iterations = 100
    cooling_rate = 0.995
    temp = 1.0
    
    while time.time() - start_time < time_budget and iterations < max_iterations:
        # Perturb and optimize
        perturbed_h = perturb(best_h)
        current_h = optimize_step(perturbed_h, method='Nelder-Mead')
        current_C5 = compute_C5(current_h)
        
        # Adaptive cooling
        if current_C5 < best_C5:
            temp *= cooling_rate
            best_C5 = current_C5
            best_h = current_h
            print(f'New best C5: {best_C5}')
        else:
            temp *= 1.005
        
        iterations += 1
    
    return best_h, best_C5, len(best_h)
