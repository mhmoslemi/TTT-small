import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial
from scipy.optimize import Bounds, minimize
from sklearn.decomposition import PCA
from sklearn.cluster import KMeans

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Function to evaluate the overlap
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Function to compute the constraint
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    # Function to generate initial guess based on a pattern
    def generate_initial_guess(n_points):
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        for i in range(n_points):
            if x[i] < 1:
                h[i] = 0.5
            else:
                h[i] = 1.0
        return h
    
    # Function to refine a guess using PCA and clustering
    def refine_guess(h, n_refine=10):
        x = np.linspace(0, 2, n_points)
        h_refined = h.copy()
        for _ in range(n_refine):
            # Apply PCA to reduce dimensionality
            pca = PCA(n_components=10)
            pca.fit(h_refined.reshape(-1, 1))
            h_pca = pca.transform(h_refined.reshape(-1, 1))
            
            # Cluster the PCA components
            kmeans = KMeans(n_clusters=5, random_state=42)
            labels = kmeans.fit_predict(h_pca)
            
            # Reconstruct the guess using the cluster centroids
            centroids = kmeans.cluster_centers_
            h_reconstructed = np.zeros_like(h_refined)
            for i in range(n_points):
                h_reconstructed[i] = centroids[labels[i]]
            h_refined = h_reconstructed
        return h_refined

    # Main function
    n_points = 200
    dx = 2.0 / n_points
    
    # Generate initial guess
    initial_h = generate_initial_guess(n_points)
    
    # Refine the initial guess
    initial_h = refine_guess(initial_h)
    
    # Define the objective function
    def objective(x):
        return compute_overlap(x)
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform optimization with adaptive step size
    def optimize_guess(x0):
        result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 200})
        return result.x, result.fun
    
    # Start from the initial guess
    best_h = initial_h
    best_c5 = compute_overlap(best_h)
    
    # Perform multiple restarts with perturbations
    n_starts = 10
    for _ in range(n_starts):
        # Perturb the current best guess
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)
        
        # Optimize from the perturbed guess
        h_opt, c5_opt = optimize_guess(x0_perturb)
        
        # Update best guess if found a better solution
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Return the best guess, its C5 value, and the number of points
    return (best_h.tolist(), best_c5, n_points)
