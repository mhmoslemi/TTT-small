import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from functools import partial

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 800  # Using 800 points for better resolution
    dx = 2.0 / n_points

    # Initialize with a good starting point
    def initial_h_values(n):
        x = np.linspace(0, 2, n)
        h = np.zeros(n)
        for i in range(n):
            if x[i] < 1:
                h[i] = 0.5 if x[i] < 0.5 else 1.0
            else:
                h[i] = 0.5 if x[i] < 1.5 else 1.0
        return h

    initial_h = initial_h_values(n_points)
    initial_h = np.clip(initial_h, 0, 1)

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Start with initial guess
    x0 = initial_h.copy()

    # First optimization step with SLSQP
    opt_result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 100})
    best_h = opt_result.x
    best_c5 = compute_overlap(best_h)

    # Multi-start with perturbations and adaptive step size
    n_starts = 15
    best_c5 = float('inf')

    for _ in range(n_starts):
        # Perturb the initial guess slightly
        perturb = np.random.uniform(-0.05, 0.05, n_points)
        x0_perturb = np.clip(best_h + perturb, 0, 1)

        # Perform a local optimization with adaptive tolerance
        result = minimize(objective, x0_perturb, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 200})
        current_c5 = compute_overlap(result.x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    # Optionally perform a gradient descent refinement
    def objective_grad(x):
        h = np.array(x, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr[1:-1]) * dx
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points and i + k >= 0:
                    grad[i] += (1 - h[k]) * (corr[i + k + 1] - corr[i + k]) * dx
                if i - k >= 0 and i - k < n_points:
                    grad[i] += (1 - h[k]) * (corr[i - k + 1] - corr[i - k]) * dx
        return max_corr, grad

    # Gradient descent refinement
    learning_rate = 0.1
    n_iterations = 50
    for _ in range(n_iterations):
        _, grad = objective_grad(best_h)
        best_h = np.clip(best_h - learning_rate * grad, 0, 1)
        current_c5 = compute_overlap(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5

    # Final result
    return (best_h.tolist(), best_c5, n_points)
