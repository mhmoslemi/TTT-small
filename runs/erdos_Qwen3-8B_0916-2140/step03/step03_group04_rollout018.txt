def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define initial n_points, starting with a large grid
    n_points = 1000
    dx = 2.0 / n_points
    
    # We'll start from the initial_h_values provided (n=400)
    initial_h = np.array(initial_h_values, dtype=np.float64)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    def objective(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Initial optimization with SLSQP
    result = minimize(objective, initial_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    
    # Define a perturbation function
    def perturb_h(h, pert_scale=0.05):
        pert = np.random.normal(0, pert_scale, h.shape)
        return np.clip(h + pert, 0, 1)
    
    # Perform multiple perturbations and optimizations
    num_perturbations = 5
    best_c5_value = best_c5
    best_h_value = best_h
    
    for _ in range(num_perturbations):
        perturbed_h = perturb_h(best_h)
        result_perturbed = minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        current_c5 = compute_overlap(result_perturbed.x)
        if current_c5 < best_c5_value:
            best_c5_value = current_c5
            best_h_value = result_perturbed.x
    
    # If budget allows, perform a random restart optimization
    if budget_s > 300:
        # Create a randomized starting point
        random_start = np.random.uniform(0, 1, n_points)
        result_random = minimize(objective, random_start, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
        current_c5 = compute_overlap(result_random.x)
        if current_c5 < best_c5_value:
            best_c5_value = current_c5
            best_h_value = result_random.x
    
    # If budget allows, perform a grid-based exploration
    if budget_s > 400:
        # Define a smaller grid for exploration
        grid_size = 10
        for shift in np.linspace(-1, 1, grid_size):
            shifted_h = np.roll(best_h_value, int(shift * n_points / 2))
            result_shifted = minimize(objective, shifted_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 200})
            current_c5 = compute_overlap(result_shifted.x)
            if current_c5 < best_c5_value:
                best_c5_value = current_c5
                best_h_value = result_shifted.x
    
    return (best_h_value.tolist(), best_c5_value, n_points)
