import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization function with adaptive perturbations and gradient-based descent
    def optimize_with_gradient_descent(initial_guess, num_steps=500, step_size=0.01, tolerance=1e-5):
        best_h = initial_guess.copy()
        best_c5 = compute_overlap(best_h)
        for step in range(num_steps):
            # Compute the gradient using finite differences
            grad = np.zeros_like(best_h)
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += 1e-6
                h_minus = best_h.copy()
                h_minus[i] -= 1e-6
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * 1e-6)
            
            # Update h using gradient descent
            best_h = np.clip(best_h - step_size * grad, 0, 1)
            
            # Compute current objective
            current_c5 = compute_overlap(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
            if abs(current_c5 - best_c5) < tolerance:
                break
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_gradient_descent(initial_h)
    
    # Perform further refinement with smaller steps and adaptive perturbations
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.01 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})

            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
