    def compute_overlap(h):

        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimizer parameters
    opt_method = 'SLSQP'
    opt_options = {'ftol': 1e-9, 'maxiter': 150}
    
    # Function to perform local search with adaptive perturbations
    def local_search(starting_h, perturbation_scale=0.05, max_iter=100):
        h = starting_h.copy()
        for _ in range(max_iter):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            h_perturbed = np.clip(h + perturbations, 0, 1)
            result = opt.minimize(objective, h_perturbed, bounds=bounds, constraints=cons, method=opt_method, options=opt_options)
            h = result.x
            if result.success:
                perturbation_scale *= 0.95  # Decrease perturbation scale
            else:
                perturbation_scale *= 1.1  # Increase perturbation scale
        return h

    def generate_ensembles(initial_guess, num_ensembles=5):
        ensembles = []

            perturbations = np.random.normal(0, 0.1, n_points)
