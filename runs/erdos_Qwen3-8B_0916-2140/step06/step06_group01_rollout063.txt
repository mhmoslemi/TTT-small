import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):

    # Define grid points

    dx = 2.0 / n_points

    # Initial construction from initial_h_values

    if len(initial_h) != n_points:

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local optimization
    def local_optimize(initial_guess, num_iterations=10, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        
        for _ in range(num_iterations):
            # Generate perturbations
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', 
                                 options={'ftol': 1e-10, 'maxiter': 150})
            
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Adaptive perturbation scale and multi-start optimization
    best_h = initial_h
    best_c5 = compute_overlap(best_h)
    
    # Run multi-start optimization with decreasing perturbation scale
    perturbation_scale = 0.05
    num_perturbations = 10
    start_time = time.time()
    
    for _ in range(num_perturbations):
        # Perturb initial guess
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(initial_h + perturbations, 0, 1)
        
        # Run local optimization
        local_h, local_c5 = local_optimize(perturbed_h)
        
        if local_c5 < best_c5:
            best_h = local_h
            best_c5 = local_c5
        
        # Decrease perturbation scale
        perturbation_scale *= 0.95
    
    # Local refinement with smaller perturbation scale
    perturbation_scale = 0.01
    elapsed_time = time.time() - start_time

        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        
        # Run local optimization
        local_h, local_c5 = local_optimize(perturbed_h)
        
        if local_c5 < best_c5:
            best_h = local_h
            best_c5 = local_c5
        
        perturbation_scale *= 0.95
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
