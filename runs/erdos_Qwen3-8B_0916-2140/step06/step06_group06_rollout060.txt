    def compute_overlap(h):

        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    def local_search(start_h, max_iter=100, perturbation_scale=0.1):
        current_h = start_h.copy()
        best_h = current_h
        best_c5 = compute_overlap(current_h)
        
        for _ in range(max_iter):
            # Generate perturbation
            perturbation = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(current_h + perturbation, 0, 1)
            
            # Perform optimization
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons,

            # Update best solution
            current_h = result.x
            current_c5 = compute_overlap(current_h)
