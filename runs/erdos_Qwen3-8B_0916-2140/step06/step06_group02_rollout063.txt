import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization function with adaptive perturbations
    def optimize_with_perturbations(initial_guess, num_perturbations=5, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_perturbations):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_perturbations(initial_h)
    
    # Further improve with gradient-based descent
    def optimize_with_descent(initial_guess, max_iter=100, learning_rate=0.01, tolerance=1e-5):

            grad = np.zeros_like(h)

            corr = np.correlate(h, 1 - h, mode="full")
            max_index = np.argmax(corr[1:-1])
            corr_vals = corr[1:-1]
            # Forward difference for gradient
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] = min(1, h_plus[i] + 1e-6)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")

                h_minus = h.copy()

                corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")

            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            # Check for convergence

            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
