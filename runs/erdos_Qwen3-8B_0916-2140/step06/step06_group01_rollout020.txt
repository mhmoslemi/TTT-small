import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with adaptive perturbations
    def adaptive_local_search(initial_guess, max_iter=100, perturbation_scale=0.05, decay_factor=0.99):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)
        current_scale = perturbation_scale
        
        for _ in range(max_iter):
            perturbations = np.random.normal(0, current_scale, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            
            result = opt.minimize(
                objective,
                perturbed_h,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 50}
            )
            
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            
            h = current_h
            current_scale *= decay_factor
        
        return best_h, best_c5
    
    # Perform initial search with perturbations
    best_h, best_c5 = adaptive_local_search(initial_h)
    
    # Perform multi-start local search with decreasing perturbation scales
    elapsed_time = 0
    start_time = time.time()
    perturbation_scale = 0.05
    
    while elapsed_time < budget_s - 100:
        # Generate a random perturbation with current scale
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)

            method='SLSQP',
