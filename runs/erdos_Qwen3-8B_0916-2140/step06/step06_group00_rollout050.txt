```python
import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define a grid with n_points samples
    n_points = 300
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initialize with a piecewise-constant function
    initial_h = np.zeros(n_points)
    initial_h[0:n_points//3] = 1.0
    initial_h[n_points//3:2*n_points//3] = 0.0
    initial_h[2*n_points//3:] = 1.0
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization function with adaptive perturbations
    def optimize_with_perturbations(initial_guess, num_perturbations=5, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_perturbations):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_perturbations(initial_h)
    
    # If time allows, perform further local search with adaptive perturbation scale
    start_time = time.time()
    while time.time() - start_time < budget_s - 100:
        # Reduce perturbation scale over time
        perturbation_scale = 0.05 * np.exp(-time.time() + start_time)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
```
