    def compute_overlap(h):

        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]

        best_h = h_initial
        best_c5 = compute_overlap(h_initial)

            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h_initial + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': ftol, 'maxiter': maxiter})

            if current_c5 < best_c5:
