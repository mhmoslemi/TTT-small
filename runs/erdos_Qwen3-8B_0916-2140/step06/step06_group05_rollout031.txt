    def compute_overlap(h):

    def objective(x):

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    # Function to perform a local search with a given perturbation scale
    def local_search(guess, perturbation_scale, max_iter=150):
        result = opt.minimize(
            objective, 
            guess, 
            bounds=bounds, 
            constraints=cons, 
            method='SLSQP', 
            options={'ftol': 1e-10, 'maxiter': max_iter}
        )
        return result.x, result.fun
