    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimizer parameters
    maxiter = 100
    ftol = 1e-10
    perturbation_scale = 0.05
    num_perturbations = 5
    num_local_search_steps = 10
    
    # Function to perform local search with perturbations
    def local_search(h_initial, perturbation_scale, num_perturbations, maxiter, ftol):
        best_h = h_initial
        best_c5 = compute_overlap(h_initial)
        for _ in range(num_perturbations):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h_initial + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': ftol, 'maxiter': maxiter})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Perform initial local search
    best_h, best_c5 = local_search(initial_h, perturbation_scale, num_perturbations, maxiter, ftol)
    
    # Adaptive local search with decreasing perturbation scale
    elapsed_time = 0

    while elapsed_time < budget_s - 100:
