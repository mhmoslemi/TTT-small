import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define parameters

    dx = 2.0 / n_points

    # Initial construction from initial_h_values

    if len(initial_h) != n_points:

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Starting with the initial guess
    best_h = initial_h
    best_c5 = compute_overlap(best_h)
    
    # Optimization parameters
    def optimize_with_multiple_scales(initial_guess, scales):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        for scale in scales:
            perturbations = np.random.normal(0, scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Refinement with local search
    def refine_solution(h, max_iterations, scale_factor):
        for _ in range(max_iterations):
            perturbation = np.random.normal(0, scale_factor, n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < compute_overlap(h):
                h = current_h
        return h
    
    # Run optimization
    best_h, best_c5 = optimize_with_multiple_scales(initial_h, [0.05, 0.02, 0.01])
    best_h = refine_solution(best_h, 20, 0.005)
    best_c5 = compute_overlap(best_h)
    
    # Additional refinement with decreasing scales
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.01 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
