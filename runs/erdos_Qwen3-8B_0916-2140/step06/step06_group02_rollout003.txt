    def optimize_local(initial_guess):
        result = opt.minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        return result.x
    
    # Function to perform a global optimization with simulated annealing
    def optimize_global(initial_guess):

            return compute_overlap(x)

            return np.sum(x) - n_points / 2

            objective_simulated_annealing,
            initial_guess,
            bounds=bounds,

            options={'ftol': 1e-10, 'maxiter': 150}
        )
