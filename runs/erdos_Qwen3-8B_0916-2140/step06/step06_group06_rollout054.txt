import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):

    # Initial grid points

    dx = 2.0 / n_points

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with adaptive perturbation
    def adaptive_search(initial_guess, max_iter=50, perturbation_scale_start=0.05, perturbation_scale_end=0.1):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)
        for iter in range(max_iter):
            # Adaptive perturbation scale
            perturbation_scale = perturbation_scale_start + (perturbation_scale_end - perturbation_scale_start) * (iter / max_iter)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                h = current_h
            # Early stopping if improvement slows down
            if iter > 10 and np.abs(best_c5 - compute_overlap(h)) < 1e-8:
                break
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = adaptive_search(initial_h)
    
    # Perform multi-start local search with different perturbation scales
    start_time = time.time()
    elapsed_time = 0
    while elapsed_time < budget_s - 100:
        # Use different perturbation scales for each start
        perturbation_scales = np.linspace(0.01, 0.1, 5)
        for scale in perturbation_scales:
            perturbations = np.random.normal(0, scale, n_points)
            perturbed_h = np.clip(best_h + perturbations, 0, 1)

            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        elapsed_time = time.time() - start_time
