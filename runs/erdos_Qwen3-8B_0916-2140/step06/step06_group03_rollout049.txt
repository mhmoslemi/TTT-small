    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform optimization with multiple perturbations
    def optimize_with_perturbations(initial_guess, scales):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        for scale in scales:
            perturbations = np.random.normal(0, scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Run optimization with multiple perturbation scales
    best_h, best_c5 = optimize_with_perturbations(initial_h, [0.05, 0.02, 0.01])

    elapsed_time = 0
