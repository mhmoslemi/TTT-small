    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Define a function to generate a random perturbation
    def perturb(h, scale=0.05):
        perturbation = np.random.normal(0, scale, len(h))
        return np.clip(h + perturbation, 0, 1)
    
    # Define a function to perform a local search with adaptive perturbation
    def local_search(h, max_iter=100, perturbation_scale=0.05):
        h = np.array(h, dtype=np.float64)
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        for _ in range(max_iter):
            perturbed_h = perturb(best_h, perturbation_scale)
            result = opt.minimize(
                objective,
                perturbed_h,
                bounds=bounds,
                constraints=cons,
                method='SLSQP',
                options={'ftol': 1e-10, 'maxiter': 150}
            )
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            perturbation_scale *= 0.9995  # Decrease perturbation scale over time
        return best_h, best_c5
    
    # Perform a multi-start local search with adaptive perturbation
    best_h = initial_h.copy()
    best_c5 = compute_overlap(initial_h)
    
    # Run local search from the initial guess
    best_h, best_c5 = local_search(initial_h)
    
    # If time allows, run additional local searches with different perturbations
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        # Generate a new perturbation and perform a local search
        perturbation = np.random.normal(0, 0.05, n_points)
