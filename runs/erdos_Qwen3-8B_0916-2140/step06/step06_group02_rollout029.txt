import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Define the grid
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial grid points from initial_h_values
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimization with a different strategy: using a genetic algorithm to explore diverse solutions
    def genetic_optimize(population_size=30, generations=50, mutation_rate=0.1, crossover_rate=0.8):
        def fitness(h):
            return compute_overlap(h)
        
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h for h in population if np.sum(h) * dx == 1]
        
        best_h = min(population, key=fitness)
        best_c5 = fitness(best_h)
        
        for _ in range(generations):
            # Evaluate fitness
            fitnesses = [fitness(h) for h in population]
            population.sort(key=lambda x: fitnesses[population.index(x)])
            
            # Select top half
            selected = population[:len(population)//2]
            
            # Create new population
            new_population = []
            for _ in range(len(selected)):
                # Crossover
                if np.random.rand() < crossover_rate:
                    a, b = np.random.choice(selected, 2, replace=False)
                    child = np.zeros(n_points)
                    for i in range(n_points):
                        if np.random.rand() < 0.5:
                            child[i] = a[i]
                        else:
                            child[i] = b[i]
                    new_population.append(child)
                else:
                    new_population.append(np.random.uniform(0, 1, n_points))
            
            # Apply mutation
            for h in new_population:
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        h[i] += np.random.uniform(-0.1, 0.1)
                        h[i] = np.clip(h[i], 0, 1)
                # Enforce constraint
                h_sum = np.sum(h)
                if h_sum != n_points / 2:
                    h = np.random.uniform(0, 1, n_points)
                    h_sum = np.sum(h)
                    if h_sum != n_points / 2:
                        h = np.random.uniform(0, 1, n_points)
                        h = np.clip(h, 0, 1)
                        h_sum = np.sum(h)
                        h = h - (h_sum - n_points / 2) / n_points
                        h = np.clip(h, 0, 1)
            
            population = new_population
            # Update best
            for h in population:
                if fitness(h) < best_c5:
                    best_h = h
                    best_c5 = fitness(h)
        
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = genetic_optimize()
    
    # Local search with SLSQP
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
