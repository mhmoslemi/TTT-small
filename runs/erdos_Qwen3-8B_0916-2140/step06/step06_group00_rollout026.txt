import scipy.optimize as opt
import time
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

def run(seed=42, budget_s=600):

    dx = 2.0 / n_points

    if len(initial_h) != n_points:

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Optimizer with adaptive perturbation and piecewise optimization
    def optimize_with_perturbations(initial_guess, num_perturbations=5, perturbation_scale=0.05):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_perturbations):
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Adaptive perturbation and local search
    def adaptive_optimize(initial_guess, max_iter=100, perturbation_decay=0.95):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)
        for i in range(max_iter):
            perturbation_scale = 0.05 * (perturbation_decay ** i)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                h = best_h
        return best_h, best_c5
    
    # Initial optimization
    best_h, best_c5 = optimize_with_perturbations(initial_h)
    
    # Further local search with adaptive perturbation
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})

        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    # Advanced search using K-means clustering to perturb regions of interest
    def kmeans_perturbation(h, n_clusters=5, perturbation_scale=0.05):
        clusterer = KMeans(n_clusters=n_clusters, n_init=10)
        clusterer.fit(h.reshape(-1, 1))
        labels = clusterer.labels_
        for i in range(n_clusters):
            cluster_indices = np.where(labels == i)[0]
            cluster_mean = np.mean(h[cluster_indices])
            cluster_std = np.std(h[cluster_indices])
            perturbations = np.random.normal(0, perturbation_scale * cluster_std, len(cluster_indices))
            h[cluster_indices] += perturbations
            h[cluster_indices] = np.clip(h[cluster_indices], 0, 1)

        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
