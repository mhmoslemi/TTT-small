import scipy.optimize as opt
import time

    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a local search with adaptive perturbation
    def local_search(h, max_iter=50, perturbation_scale=0.05):
        h = np.copy(h)
        for _ in range(max_iter):
            perturbation = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < compute_overlap(h):
                h = current_h
            perturbation_scale *= 0.99
        return h
    
    # Perform a multi-start optimization with adaptive perturbation
    best_h = initial_h
    best_c5 = compute_overlap(best_h)
    for _ in range(5):
        perturbation = np.random.normal(0, 0.1, n_points)
        perturbed_h = np.clip(initial_h + perturbation, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        best_h = local_search(best_h)
    
    # Final refinement
    best_h = local_search(best_h, max_iter=100, perturbation_scale=0.05)
    best_c5 = compute_overlap(best_h)
    
    # If time allows, perform further refinements
    start_time = time.time()

        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
