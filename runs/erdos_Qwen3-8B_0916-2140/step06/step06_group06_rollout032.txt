    def compute_overlap(h):

    # Objective function
    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local optimization with adaptive step size
    def local_optimize(initial_guess, max_iter=50, step_size=0.1, tol=1e-6):
        x = initial_guess.copy()
        for _ in range(max_iter):
            grad = np.zeros_like(x)
            for k in range(1, n_points - 1):
                h = np.zeros_like(x)
                h[k] = 1
                h = np.correlate(h, 1 - x, mode="full")
                grad[k] = np.max(h[1:-1]) * dx
            grad = grad - np.mean(grad)
            x = np.clip(x - step_size * grad, 0, 1)
            if np.abs(np.sum(x) - n_points / 2) < tol:
                break
        return x
