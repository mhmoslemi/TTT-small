    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform a single optimization step
    def optimize_step(current_h, perturbation_scale=0.1, maxiter=100):
        result = opt.minimize(
            objective, 
            current_h, 
            bounds=bounds, 
            constraints=cons, 
            method='SLSQP', 
            options={'ftol': 1e-10, 'maxiter': maxiter}
        )
        return result.x
    
    # Function to perform a local search with perturbations
    def local_search(initial_guess, num_steps=10, perturbation_scale=0.1):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        for _ in range(num_steps):
            perturbation = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbation, 0, 1)
            result = optimize_step(perturbed_h, perturbation_scale / 10, 100)
            current_c5 = compute_overlap(result)
            if current_c5 < best_c5:
                best_h = result
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = local_search(initial_h)
    
    # Perform multiple rounds of local search with decreasing perturbation scale
    elapsed_time = 0
    start_time = time.time()
    perturbation_scale = 0.1
    max_iter = 10
    while elapsed_time < budget_s - 100:
        best_h, best_c5 = local_search(best_h, num_steps=max_iter, perturbation_scale=perturbation_scale)
