    def compute_overlap(h):

        return np.max(corr[1:-1]) * dx

    def objective(x):
        return compute_overlap(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    cons = ({'type': 'eq', 'fun': constraint_sum})

    def adaptive_local_search(start_h, max_iter=200, perturb_scale_start=0.05, perturb_scale_end=0.01):
        h = np.array(start_h, dtype=np.float64)
        best_h = h.copy()
        best_c5 = compute_overlap(h)
        for _ in range(max_iter):
            perturbation_scale = perturb_scale_start * (perturb_scale_end / perturb_scale_start) ** (_ / max_iter)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(h + perturbations, 0, 1)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 100})
