import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):

    # Initial grid points

    dx = 2.0 / n_points

    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to generate random perturbations
    def perturb(h, scale):
        return np.clip(h + np.random.normal(0, scale, len(h)), 0, 1)
    
    # Optimization function with adaptive scaling
    def optimize_with_adaptive_scaling(initial_guess, max_iter=100, scale_decay=0.95, scale_initial=0.1):
        best_h = initial_guess
        best_c5 = compute_overlap(best_h)
        current_h = initial_guess
        scale = scale_initial
        for iter in range(max_iter):
            perturbed_h = perturb(current_h, scale)
            result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            scale *= scale_decay
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_adaptive_scaling(initial_h)
    
    # Further refinement with multiple restarts
    num_restarts = 5
    for _ in range(num_restarts):
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
    
    # If time allows, perform further refinement
    elapsed_time = 0
    start_time = time.time()

        perturbed_h = np.clip(best_h + np.random.normal(0, perturbation_scale, n_points), 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
