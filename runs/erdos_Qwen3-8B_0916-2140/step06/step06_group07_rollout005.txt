import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initial grid points
    n_points = 200
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    
    # Initial construction from initial_h_values
    initial_h = np.array(initial_h_values, dtype=np.float64)
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform gradient descent with adaptive learning rate
    def optimize_with_adaptive_gradient_descent(initial_guess, num_steps=100, learning_rate=0.1, decay_rate=0.99):
        h = initial_guess.copy()
        best_h = h
        best_c5 = compute_overlap(h)
        for step in range(num_steps):
            grad = np.zeros_like(h)
            for k in range(1, n_points - 1):
                # Compute derivative for each shift k
                # We approximate the derivative using finite differences
                h_shift = np.roll(h, k)
                # Compute the overlap integral for h and h_shift
                corr = np.correlate(h, 1 - h_shift, mode="full")
                overlap = np.max(corr[1:-1]) * dx
                # Compute gradient with respect to h[j]
                for j in range(n_points):
                    h_shift_j = np.roll(h, k)
                    # Perturb h[j] and recompute overlap
                    h_perturbed = h.copy()
                    h_perturbed[j] += 1e-6
                    corr_perturbed = np.correlate(h_perturbed, 1 - h_shift_j, mode="full")
                    overlap_perturbed = np.max(corr_perturbed[1:-1]) * dx
                    grad[j] = (overlap_perturbed - overlap) / 1e-6
            # Update h using gradient descent
            learning_rate_current = learning_rate * (decay_rate ** step)
            h -= learning_rate_current * grad
            h = np.clip(h, 0, 1)
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
        return best_h, best_c5
    
    # Start with initial guess
    best_h, best_c5 = optimize_with_adaptive_gradient_descent(initial_h)
    
    # If time allows, perform further local search with adaptive perturbation scale
    elapsed_time = 0
    start_time = time.time()
    while elapsed_time < budget_s - 100:
        perturbation_scale = 0.05 * np.exp(-elapsed_time / 100)
        perturbations = np.random.normal(0, perturbation_scale, n_points)
        perturbed_h = np.clip(best_h + perturbations, 0, 1)
        result = opt.minimize(objective, perturbed_h, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        current_h = result.x
        current_c5 = compute_overlap(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed_time = time.time() - start_time
    
    return (best_h.tolist(), best_c5, n_points)
