    # Function to compute the overlap integral
    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local search with adaptive perturbations
    def adaptive_local_search(initial_guess, max_iter=50, perturbation_scale=0.2, scale_factor=0.98):
        best_h = initial_guess.copy()
        best_c5 = compute_overlap(best_h)
        current_h = best_h
        for _ in range(max_iter):
            # Generate perturbations
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(current_h + perturbations, 0, 1)
            
            # Ensure the constraint is satisfied
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            
            # Perform optimization
            result = opt.minimize(
                objective, 
                perturbed_h, 
                bounds=bounds, 
                constraints=cons, 
                method='SLSQP', 
                options={'ftol': 1e-8, 'maxiter': 100}
            )
            
            # Update best solution
            current_h = result.x
            current_c5 = compute_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            
            # Decrease perturbation scale
            perturbation_scale *= scale_factor
        
        return best_h, best_c5
    
    # Initial optimization with adaptive perturbations
    best_h, best_c5 = adaptive_local_search(initial_h)
    
    # Further refinement with smaller perturbations
    elapsed_time = 0
    start_time = time.time()
    perturbation_scale = 0.1

        )
