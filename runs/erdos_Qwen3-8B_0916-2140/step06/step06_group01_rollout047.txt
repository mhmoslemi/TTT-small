import scipy.optimize as opt
import time

    def compute_overlap(h):
        h = np.array(h, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr[1:-1]) * dx
    
    # Objective function
    def objective(x):
        return compute_overlap(x)
    
    # Constraint: sum of h equals n_points / 2
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Function to perform local optimization
    def local_optimize(initial_guess):
        result = opt.minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 150})
        return result.x, result.fun
    
    # Function to perform perturbation-based optimization
    def perturb_optimize(initial_guess, num_perturbations=10):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        
        for i in range(num_perturbations):
            # Add perturbations with different scales
            perturbation_scale = 0.01 * (1 - i / num_perturbations)
            perturbations = np.random.normal(0, perturbation_scale, n_points)
            perturbed_h = np.clip(initial_guess + perturbations, 0, 1)
            result = local_optimize(perturbed_h)
            current_h, current_c5 = result
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Function to perform grid-based optimization
    def grid_search(initial_guess, num_points=10):
        best_h = initial_guess
        best_c5 = compute_overlap(initial_guess)
        
        # Generate grid points around initial guess
        grid_points = np.linspace(0, 1, num_points)

                perturbations = np.random.normal(0, 0.05, n_points)
                perturbed_h = np.clip(initial_guess + perturbations, 0, 1)

        elapsed_time = time.time() - start_time
