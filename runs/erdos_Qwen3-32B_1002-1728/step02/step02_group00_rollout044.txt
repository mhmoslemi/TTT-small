        return np.max(corr) * dx

    num_workers = 5
    np.random.seed(seed)
    initial_guesses = []
    for _ in range(num_workers):
        noise = np.random.normal(0, 0.02, n_points)
        perturbed = np.clip(h_init + noise, 0, 1)
        adjusted = adjust_to_sum_vec(perturbed, target_sum)
        initial_guesses.append(adjusted)

    best_h = h_init.copy()
    best_c5 = objective(best_h)

    for i in range(num_workers):
        if time.time() - start_time > budget_s * 0.95:
            break
        x0 = initial_guesses[i]
        result = minimize(
            fun=objective,
            x0=x0,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'tol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
