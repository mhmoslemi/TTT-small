import time

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    start_time = time.time()
    K = 64
    dx = 2.0 / K
    target_sum = 32  # K//2 = 32

    # Initialize binary h
    h = np.zeros(K, dtype=float)
    h[::2] = 1.0  # Alternate 1 and 0

    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    # Simulated annealing parameters
    initial_T = 10.0
    cooling_rate = 0.995
    T = initial_T

    np.random.seed(seed)
    while time.time() - start_time < budget_s * 0.99:
        # Propose a move
        ones = np.where(current_h == 1.0)[0]
        zeros = np.where(current_h == 0.0)[0]
        if len(ones) * len(zeros) == 0:
            break  # No moves possible
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # Swap i and j
        new_h = current_h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]

        # Compute new c5
        new_c5 = compute_c5(new_h)

        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5

        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

        T *= cooling_rate

    assert np.all(0.0 <= best_h) and np.all(best_h <= 1.0), "Values out of bounds"
    sum_h = np.sum(best_h)
    assert abs(sum_h - target_sum) < 1e-6, "Integral constraint violated"

    return (best_h.tolist(), best_c5, K)
