import numpy as np
import time
from scipy.optimize import Bounds, LinearConstraint, minimize

def compute_c5(h, dx, n_points):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def generate_periodic(p, n_points):
    pattern = np.zeros(n_points)
    for i in range(n_points):
        if i % p < p // 2:
            pattern[i] = 1.0
    return pattern

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    n = len(h)
    if abs(current_sum - target_sum) < 1e-8:
        return h
    scale = target_sum / current_sum
    h_scaled = h * scale
    h_scaled = np.clip(h_scaled, 0.0, 1.0)
    diff = target_sum - np.sum(h_scaled)
    if abs(diff) > 1e-8:
        h_scaled[0] += diff
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 30
    tournament_size = 3
    mutation_rate = 0.1
    mutation_std = 0.05
    max_no_change_gen = 50

    def population_initializer():
        pop = [project_h(h_initial, target_sum)]
        for p in [2, 4, 6, 8, 10]:
            pattern = generate_periodic(p, n_points)
            pop.append(project_h(pattern, target_sum))
        while len(pop) < pop_size:
            h = np.random.rand(n_points)
            h = project_h(h, target_sum)
            pop.append(h)
        return np.array(pop)

    population = population_initializer()
    best_c5 = float('inf')
    best_h = h_initial.copy()
    last_best_c5 = best_c5
    start_time = time.time()
    no_progress_gen = 0

    while time.time() - start_time < budget_s - 5:
