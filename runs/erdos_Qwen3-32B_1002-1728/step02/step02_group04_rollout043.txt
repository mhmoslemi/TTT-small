import time

def compute_c5_overlaps(h, dx):
    n_points = len(h)
    ones_minus_h = 1 - h
    overlaps = np.zeros(n_points)
    for k in range(n_points):
        shifted = np.roll(ones_minus_h, -k)
        overlaps[k] = np.dot(h, shifted) * dx
    return overlaps

def compute_gradient(h, k_min, dx):
    n_points = len(h)
    grad = np.zeros(n_points)
    for i in range(n_points):
        ipk = (i + k_min) % n_points
        imk = (i - k_min) % n_points
        grad[i] = (1 - h[ipk]) - h[imk]
    grad *= dx
    return grad

def compute_linear_minimizer(grad, required_sum):
    sorted_indices = np.argsort(grad)
    n_points = len(grad)
    m = int(required_sum)
    remaining = required_sum - m
    v = np.zeros(n_points)
    v[sorted_indices[:m]] = 1.0
    if m < n_points:
        v[sorted_indices[m]] = remaining
    return v

def ensure_feasible(h, required_sum):
    # Ensures sum and clips to [0,1]
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    if np.abs(current_sum - required_sum) > 1e-6:
        h = h * (required_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h
