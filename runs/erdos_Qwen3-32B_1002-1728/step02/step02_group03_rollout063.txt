            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            h = h_candidate
        else:
            # Reduce step size
            alpha *= 0.8
            if alpha < 1e-4:
                h = best_h
                alpha = 0.01

    # Final evaluation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_final)
    return best_h, final_c5, n_points
