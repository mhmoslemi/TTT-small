        k_offset = j_max - (n_points - 1)

        gradient = np.zeros(n_points)
        for x in range(n_points):
            offset = x + k_offset
            if 0 <= offset < n_points:
                gradient[x] += (1 - h[offset]) * dx
                gradient[offset] -= h[x] * dx

        h_var = cp.Variable(n_points)
        objective = cp.Minimize(gradient @ h_var)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status != cp.OPTIMAL:
            continue

        h = h_var.value
        h = np.clip(h, 0, 1)

    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    return best_h, final_c5, n_points
