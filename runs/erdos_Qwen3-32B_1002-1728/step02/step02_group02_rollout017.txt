import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def constraint_func(h_vec):
        h = h_vec.reshape(n_points)
        return h.sum() - target_sum

    cons = [{'type': 'eq', 'fun': constraint_func}]
    bounds = tuple([(0.0, 1.0) for _ in range(n_points)])

    start_time = time.time()
    maxiter = 10000
    options = {'maxiter': maxiter, 'ftol': 1e-10, 'disp': False}
    
    result = minimize(
        objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options=options
    )

    elapsed = time.time() - start_time
    remaining = max(0, budget_s - elapsed)

    # Final projection to feasible region
    h_opt = np.clip(result.x, 0, 1)
    h_sum = h_opt.sum()
    if not np.isclose(h_sum, target_sum):
        h_opt = np.clip(h_opt * (target_sum / h_sum), 0, 1)

    corr_values = correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5 = np.max(corr_values)
    return h_opt, c5, n_points
