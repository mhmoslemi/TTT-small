import numpy as np

from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def _project(p):
        clamped = np.clip(p, 0, 1)
        clamped_sum = clamped.sum()
        if clamped_sum == 0:
            return np.ones_like(clamped) * 0.5
        temp = clamped * (target_sum / clamped_sum)
        return np.clip(temp, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population = []
    population_size = 50
    # Initialize population
    for _ in range(5):
        pert = 0.1 * np.random.randn(n_points)
        new_h = _project(h_initial + pert)
        population.append(new_h)
    for _ in range(10):
        rand_h = _project(np.random.rand(n_points))
        population.append(rand_h)
    for _ in range(35):
        grad_h = _project(h_initial + 0.05 * np.random.randn(n_points))
        population.append(grad_h)
    # Evaluate initial scores
    scores = [compute_c5(h) for h in population]
    best_idx = np.argmin(scores)
    best_h, best_c5 = population[best_idx].copy(), scores[best_idx]

    start_time = time.time()

        new_children = []
        for _ in range(5):  # Generate 5 children per iteration
            # Tournament selection

            contestant_indices = np.random.choice(population_size, tournament_size, replace=False)
            contestant_scores = [scores[i] for i in contestant_indices]
            best_indices = np.argsort(contestant_scores)
            p1_idx = contestant_indices[best_indices[0]]
            p2_idx = contestant_indices[best_indices[1]]
            p1 = population[p1_idx].copy()
            p2 = population[p2_idx].copy()
            # Crossover
