import numpy as np
import time
import numpy.fft

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def scale_to_sum(h, target):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return 0.5 * np.ones_like(h)
        h_scaled = h * (target / s)
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        return h_scaled

    def evaluate_particle(h_vec):
        other = 1.0 - h_vec
        h_fft = numpy.fft.fft(h_vec)
        other_fft = numpy.fft.fft(other)
        corr_fft = h_fft * numpy.conj(other_fft)
        corr = numpy.fft.ifft(corr_fft).real
        return corr * dx

    def evaluate_particles(particles):
        scores = np.zeros(len(particles))
        for i in range(len(particles)):
            corr = evaluate_particle(particles[i])
            scores[i] = np.max(corr)
        return scores

    n_particles = 50
    max_iter = 1000
    omega = 0.7
    phi_c = 1.5
    phi_s = 1.5
    velocity_clamp = 0.1
    local_search_top = 5
    local_search_steps = 3

    particles = [scale_to_sum(h_initial + np.random.normal(0, 0.05, n_points), target_sum) for _ in range(n_particles)]
    velocities = [np.random.uniform(-velocity_clamp, velocity_clamp, n_points) for _ in range(n_particles)]
    particles = np.array(particles)
    velocities = np.array(velocities)

    pbest_positions = particles.copy()
    pbest_scores = evaluate_particles(particles)

    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s - 5:
        iteration += 1

        for i in range(n_particles):
            current = particles[i]
            vel = velocities[i]

            r_cog = np.random.rand()
            r_soc = np.random.rand()
            cognitive = phi_c * r_cog * (pbest_positions[i] - current)
            social = phi_s * r_soc * (gbest_position - current)

            new_vel = omega * vel + cognitive + social
            new_vel = np.clip(new_vel, -velocity_clamp, velocity_clamp)

            new_pos = np.clip(new_pos, 0.0, 1.0)
            new_pos = scale_to_sum(new_pos, target_sum)

            particles[i] = new_pos
            velocities[i] = new_vel

        scores = evaluate_particles(particles)

        for i in range(n_particles):
            if scores[i] < pbest_scores[i]:
                pbest_scores[i] = scores[i]
                pbest_positions[i] = particles[i].copy()

        curr_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[curr_gbest_idx] < gbest_score:
            gbest_score = pbest_scores[curr_gbest_idx]
            gbest_position = pbest_positions[curr_gbest_idx].copy()

        if time.time() - start_time < budget_s - 10:

                h = particles[idx].copy()
                for _ in range(local_search_steps):
                    epsilon = 1e-3

    final_c5 = np.max(corr)
