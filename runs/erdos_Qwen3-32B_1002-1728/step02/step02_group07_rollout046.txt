            lambda x: objective_block(x, blocks, n_points, dx),
            x0=x0_block,
            method='SLSQP',
            bounds=bounds,
            constraints=lin_constraint,
            options={'maxiter': 50, 'ftol': 1e-6}
        )

        if result.success:
            x_opt = result.x
            h_new = np.zeros(n_points)
            for i, bl in enumerate(blocks):
                h_new[bl] = x_opt[i]
            h_new = np.clip(h_new, 0, 1)
            h_new = adjust_sum(h_new, target_sum)

            corr_values = np.correlate(h_new, 1 - h_new, mode='full') * dx
            curr_c5 = np.max(corr_values)

            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_new.copy()
        h_current = h_new.copy() + 0.02 * np.random.randn(n_points)
        h_current = np.clip(h_current, 0, 1)
        h_current = adjust_sum(h_current, target_sum)

    return best_h, best_c5, n_points
