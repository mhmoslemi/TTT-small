        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_all = [(0.0, 1.0) for _ in range(n_points)]

    # Continue optimization if time remains
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_all,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            improved_h = result.x
            improved_c5 = result.fun
            if improved_c5 < best_c5:
                best_h = improved_h
                best_c5 = improved_c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
