    ones_idx = np.random.choice(n_points, size=target_sum, replace=False)
    h[ones_idx] = 1.0
    
    # Objective function
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Simulated Annealing parameters
    temp = 1.0
    cooling_rate = 0.995
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    while time.time() - start_time < budget_s - 1:
        # Propose a move: swap one 0 <-> 1
        zero_idx = np.where(h == 0)[0]
        one_idx = np.where(h == 1)[0]
        if len(zero_idx) == 0 or len(one_idx) == 0:
            break
        
        # Randomly select positions to swap
        i = np.random.choice(zero_idx)
        j = np.random.choice(one_idx)
        
        # Create new candidate
        new_h = h.copy()
        new_h[i] = 1.0
        new_h[j] = 0.0
        
        # Evaluate candidate
        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5
        
        # Acceptance criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h[:] = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h[:] = h
                best_c5 = current_c5
        
        # Cool down
        temp *= cooling_rate
    
    # Final constraints
    assert np.isclose(np.sum(best_h), target_sum), "Integral constraint violated"
    assert all(0 <= val <= 1 for val in best_h), "Values out of bounds"
    assert np.allclose(best_h, np.round(best_h)), "Non-binary values found"
    
    return (best_h.tolist(), best_c5, n_points)
