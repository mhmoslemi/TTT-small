    bounds = Bounds(0.0, 1.0)

            x *= target_sum / s

        return x

    # Differential evolution with sum penalty
    if time.time() - start_time < budget_s - 100:
        def de_obj(x):
            x = x.reshape(-1)
            return np.max(np.correlate(x, 1 - x, 'full') * dx)
        
        de_constraints = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
        result_de = differential_evolution(
            de_obj,

            seed=seed,
            maxiter=100,

            workers=1,

    if time.time() - start_time < budget_s - 50:
