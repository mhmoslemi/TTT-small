import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    alpha = 0.01  # Learning rate
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_indices = np.where(corr == current_c5)[0]
        k_values = list(set([idx - (n_points - 1) for idx in max_indices]))

        gradient = np.zeros(n_points)
        for k in k_values:
            if k == 0:
                gradient += (1 - 2 * h) * dx
                continue
            if k > 0:
                max_i = n_points - k
                if max_i <= 0:
                    continue  # No valid indices
                valid_i = np.arange(0, max_i)
                j = valid_i + k
                grad_i = (1 - h[j]) * dx
                grad_j = -h[valid_i] * dx
                gradient[valid_i] += grad_i
                gradient[j] += grad_j
            else:
                m = -k
                valid_i = np.arange(m, n_points)
                if len(valid_i) == 0:
                    continue  # No valid indices
                j = valid_i - m
                grad_i = (1 - h[j]) * dx
                grad_j = -h[valid_i] * dx
                gradient[valid_i] += grad_i
                gradient[j] += grad_j

        # Update h
        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s) > 1e-10:
            h *= target_sum / s
        else:
            h[:] = 0.0
            h[:int(target_sum)] = 1.0

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    alpha = 0.005  # Learning rate
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    while time.time() - start_time < budget_s:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_indices = np.where(corr == current_c5)[0]
        k_values = list(set([idx - (n_points - 1) for idx in max_indices]))

        gradient = np.zeros(n_points)
        for k in k_values:
            if k == 0:
                gradient += (1 - 2 * h) * dx
                continue
            if k > 0:
                max_i = n_points - k
                if max_i <= 0:
                    continue
                valid_i = np.arange(0, max_i)
                j = valid_i + k
                grad_i = (1 - h[j]) * dx
                grad_j = -h[valid_i] * dx
                gradient[valid_i] += grad_i
                gradient[j] += grad_j
            else:
                m = -k
                valid_i = np.arange(m, n_points)
                if len(valid_i) == 0:
                    continue
                j = valid_i - m
                grad_i = (1 - h[j]) * dx
                grad_j = -h[valid_i] * dx
                gradient[valid_i] += grad_i
                gradient[j] += grad_j

        h -= alpha * gradient
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s) > 1e-10:
            h *= target_sum / s

    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    return best_h, final_c5, n_points
