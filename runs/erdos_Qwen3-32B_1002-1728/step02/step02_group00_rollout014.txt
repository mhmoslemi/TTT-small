import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    assert abs(np.sum(h_init) - target_sum) < 1e-6, "Initial h does not meet integral constraint"

    h = cp.Variable(n_points)
    t = cp.Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]

    best_h = h_init.copy()
    best_c5 = None

    iteration = 0
    max_iterations = 100
    while time.time() - start_time < budget_s * 0.95 and iteration < max_iterations:
        objective = cp.Minimize(t)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.CVXOPT, verbose=False)

        if problem.status != cp.OPTIMAL:
            break

        current_h = h.value
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_corr = np.max(corr)
        current_c5 = max_corr * dx

        if best_c5 is None or current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

        max_index = np.argmax(corr)
        k_current = max_index - (n_points - 1)
        k_val = abs(int(k_current))

        if k_val == 0:
            break

        if k_val < n_points:
            sum_expr = 0.0
            for i in range(n_points - k_val):
                sum_expr += h[i] * (1 - h[i + k_val])
            constraints.append(sum_expr <= t)

        iteration += 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
