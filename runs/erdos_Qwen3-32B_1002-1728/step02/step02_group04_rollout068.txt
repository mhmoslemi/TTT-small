    k_worst = np.argmax(corr[:len(h)])
    worst_c5 = np.max(corr[:len(h)]) * dx
    return k_worst, worst_c5

def objective_for_k(h_vec, k, n_points):
    h = h_vec.reshape(-1)
    shifted = np.roll(h, -k)
    return -np.sum(h * shifted)

def optimize_for_k(h_initial, k, n_points, required_sum, dx):
    def objective(x):
        return objective_for_k(x, k, n_points)
    x0 = h_initial.copy()
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - required_sum})
    bounds = Bounds([0.]*n_points, [1.]*n_points)
    res = minimize(objective, x0=x0, method='SLSQP', constraints=cons, bounds=bounds, options={'maxiter': 100, 'ftol': 1e-6})
    if not res.success:
        return h_initial.copy()
    optimized_h = res.x
    optimized_h = np.clip(optimized_h, 0, 1)
    h_sum = np.sum(optimized_h)
    if not np.isclose(h_sum, required_sum, atol=1e-5):
        optimized_h = optimized_h * (required_sum / h_sum)
        optimized_h = np.clip(optimized_h, 0, 1)
    return optimized_h

def project_to_feasible(h, required_sum):
    h = np.clip(h, 0, 1).copy()
    current_sum = np.sum(h)
    if np.isclose(current_sum, required_sum):
        return h
    delta = required_sum - current_sum
    if delta > 0:
        indices = np.where(h < 1)[0]
        if len(indices) > 0:
            add_per = delta / len(indices)
            h[indices] += add_per
            h = np.clip(h, 0, 1)
    else:
        delta = -delta
        indices = np.where(h > 0)[0]
        if len(indices) > 0:
            subtract_per = delta / len(indices)
            h[indices] -= subtract_per
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2
    h = np.clip(h_initial, 0, 1).copy()
    h_sum = np.sum(h)
    if not np.isclose(h_sum, required_sum):
        h = h * (required_sum / h_sum)
        h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    start_time = time.time()
    perturbation_strength = 0.05
    iteration = 0
    max_iterations = 2000
    no_improvement_threshold = 50
    no_improvement_counter = 0
    np.random.seed(seed)
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        k_worst, _ = find_worst_k(best_h, dx)
        optimized_h = optimize_for_k(best_h, k_worst, n_points, required_sum, dx)
        mi = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        optimized_h[mi] += np.random.normal(0, perturbation_strength, len(mi))
        optimized_h = project_to_feasible(optimized_h, required_sum)
        new_c5 = compute_c5(optimized_h, dx)
        if new_c5 < best_c5:
            best_h[:] = optimized_h.copy()
            best_c5 = new_c5
            no_improvement_counter = 0
            perturbation_strength = max(0.01, perturbation_strength * 0.9)
        else:
            no_improvement_counter += 1
            if no_improvement_counter >= no_improvement_threshold:
                perturbation_strength = min(0.1, perturbation_strength * 1.5)
                no_improvement_counter = 0
                mi = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
                best_h[mi] += np.random.normal(0, perturbation_strength, len(mi))
                best_h = project_to_feasible(best_h, required_sum)
                new_c5 = compute_c5(best_h, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                else:
                    optimized_h = best_h.copy()
        if time.time() - start_time > budget_s - 1:
            break
    return best_h, best_c5, n_points
