import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum(h):
        current_sum = np.sum(h)
        offset = current_sum - target_sum
        h[0] -= offset
        h[0] = np.clip(h[0], 0.0, 1.0)
        return h

    n_particles = 100
    particles = []

    for _ in range(50):
        h = h_initial.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale_factor = target_sum / current_sum if current_sum != 0 else 1.0
        h *= scale_factor
        h = np.clip(h, 0, 1)
        h = adjust_sum(h)
        particles.append(h.copy())

    for _ in range(50):
        h = np.random.uniform(0.0, 1.0, size=n_points)
        current_sum = np.sum(h)
        scale_factor = target_sum / current_sum if current_sum != 0 else 1.0
        h *= scale_factor
        h = np.clip(h, 0, 1)
        h = adjust_sum(h)
        particles.append(h.copy())

    particles = np.array(particles)
    velocities = np.zeros_like(particles)

    p_best_positions = particles.copy()
    p_best_values = np.array([compute_c5(p) for p in particles])
    g_best_idx = np.argmin(p_best_values)
    g_best_value = p_best_values[g_best_idx]
    g_best_position = particles[g_best_idx].copy()

    c1, c2 = 2.0, 2.0
    w = 0.9
