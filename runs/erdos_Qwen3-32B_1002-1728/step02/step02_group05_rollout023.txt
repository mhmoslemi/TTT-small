        x0 = best_h.copy()
        x0 += np.random.normal(0, 0.1, n_points)  # Larger perturbation
        x0 = np.clip(x0, 0, 1)
        offset = np.sum(x0) - n_points/2
        x0[0] -= offset
        x0[0] = max(0.0, min(1.0, x0[0]))
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[sum_constraint],
            options={'maxiter': 150, 'ftol': 1e-5, 'disp': False}
        )
        if result.success:
            curr_h = np.clip(result.x, 0, 1)
        else:
            curr_h = x0.copy()
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset
        curr_h[0] = max(0.0, min(1.0, curr_h[0]))
        corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_val)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h[:] = curr_h.copy()

    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
