import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Binarize the initial h to generate a structured starting point
    target_count = int(n_points / 2)
    sorted_indices = np.argsort(h_init)[::-1]
    h_binary = np.zeros(n_points)
    h_binary[sorted_indices[:target_count]] = 1.0
    h_binary = h_binary * (target_sum / np.sum(h_binary))

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        regularization = 0.1 * np.sum(h * (1.0 - h))  # Encourage binary
        return max_corr + regularization

    result = minimize(
        fun=objective,
        x0=h_binary,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Iterative refinement with perturbations
    iteration = 0
    MAX_ITER = 50
    while time.time() - start_time < budget_s * 0.95 and iteration < MAX_ITER:
        rs = np.random.RandomState(seed + int(time.time()))
        perturbed = best_h + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = (perturbed / np.sum(perturbed)) * target_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    # Final bound validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
