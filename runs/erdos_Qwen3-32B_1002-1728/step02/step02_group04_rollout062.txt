import numpy as np
import time

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    valid_corr = corr[:len(h)]
    return np.max(valid_corr) * dx

    n_points = len(h)
    dx = 2.0 / n_points

    if h_sum <= 1e-10:
        h = np.zeros(n_points)

        h *= scaling
        h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    np.random.seed(seed)
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 5:
        iteration += 1
        # Compute all correlations to find worst k
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[:n_points]
        max_k_idx = np.argmax(valid_corr)
        max_corr = valid_corr[max_k_idx]
        current_c5 = max_corr * dx

        k = max_k_idx
        grad = np.zeros(n_points)
        valid_i = np.arange(n_points - k)
        j = valid_i + k
        grad[valid_i] += (1 - h[j])
        grad[j] -= h[valid_i]

        # Solve linear subproblem for d
        sorted_indices = np.argsort(grad)
        d = np.zeros(n_points)
        total = 0.0
        for i in range(n_points):
            idx = sorted_indices[i]
            if total + 1.0 <= required_sum:
                d[idx] = 1.0
                total += 1.0
            else:
                remaining = required_sum - total
                d[idx] = remaining

        gamma = 2.0 / (iteration + 2)
        h_new = h + gamma * (d - h)
        h_new = np.clip(h_new, 0, 1)

        current_c5_new = compute_c5(h, dx)
