def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def tournament_selection(population, fitness, tournament_size=3):
    indices = np.random.choice(len(fitness), tournament_size)
    best_idx = indices[np.argmin(fitness[indices])]
    return population[best_idx].copy()

def repair(h, target_sum):

    current_sum = h.sum()
    h[0] += (target_sum - current_sum)
    h = np.clip(h, 0.0, 1.0)
    return h
