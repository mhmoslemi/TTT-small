        g = np.zeros_like(h)
        if valid_length > 0:
            g[:valid_length] = 1 - h[k_shift:]
        h = h - alpha * g
        h = project_h(h, required_sum)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h[:] = h.copy()
            best_c5 = current_c5
            alpha *= 0.95
        if iteration % decay_interval == 0:
            alpha *= 0.95
        if np.random.rand() < 0.05 or iteration % perturb_interval == 0:
            perturb = np.random.normal(0, 0.05, n_points)
            h += perturb
            h = project_h(h, required_sum)
        if time.time() - start_time > budget_s - 1:
            break
    return best_h, best_c5, n_points
