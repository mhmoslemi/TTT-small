        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, size=2)
            child = (parent1 + parent2) / 2.0
            child = make_valid(child, target_sum, n_points)

            noise = np.random.normal(0, 0.05, size=n_points)
            child_mutation = child + noise
            child_mutation = make_valid(child_mutation, target_sum, n_points)
            new_population.append(child_mutation)

        population = new_population

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
