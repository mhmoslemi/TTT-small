        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c5_current = np.max(corr)
        max_indices = np.where(np.isclose(corr, max_c5_current, rtol=1e-5))[0]
        gradients = []

        for idx in max_indices:
            lag = idx - (n_points - 1)
            m = abs(lag)
            if m == 0:
                # Gradient for k=0 (sum h[i]*(1 - h[i]) dx)
                grad = (1.0 - 2.0 * h) * dx
                gradients.append(grad)

                for i in range(count):
                    grad_i = (1.0 - h[i + m]) * dx
                    grad[i] += grad_i
                    grad[i + m] -= h[i] * dx
                gradients.append(grad)

        if gradients:

            if norm > 1e-8:
                h -= alpha * total_grad / norm
