    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def _project(p):
        clamped = np.clip(p, 0, 1)
        clamped_sum = clamped.sum()
        if clamped_sum == 0:
            return np.ones_like(clamped) * 0.5
        temp = clamped * (target_sum / clamped_sum)
        return np.clip(temp, 0, 1)

    current_h = h_initial.copy()
    current_score = compute_c5(current_h)
    best_h = current_h.copy()
    best_score = current_score

    T_initial = 1.0
    alpha = 0.999
    max_initial_step = 0.2

    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        new_h = current_h.copy()

        i, j = np.random.choice(n_points, 2, replace=False)
        step_size = max_initial_step * T_initial
        delta = np.random.uniform(-step_size, step_size)
        new_h[i] += delta
        new_h[j] -= delta

        new_h = _project(new_h)
        new_score = compute_c5(new_h)

        delta_score = new_score - current_score

        if delta_score < 0 or np.random.rand() < np.exp(-delta_score / max(T_initial, 1e-8)):
            current_h = new_h.copy()
            current_score = new_score
            if new_score < best_score:
                best_h = current_h.copy()
                best_score = new_score

        T_initial *= alpha

    return best_h, best_score, n_points
