import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    np.random.seed(seed)

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_candidate():
        h = np.random.rand(n_points) + 0.1
        total = h.sum()
        h = (h / total) * target_sum
        h = np.clip(h, 0, 1)
        return h

    pop_size = 25
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        population.append(generate_candidate())

    best_c5 = np.inf
    best_h = h_init.copy()
    gen_idx = 0

    while time.time() - start_time < budget_s - 60:
        scores = []
        for h in population:
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            scores.append((c5, h))
        scores.sort()
        elites = [s[1] for s in scores[:max(1, int(0.25 * pop_size))]]

        new_population = elites[:]
        for _ in range(pop_size - len(elites)):
            r = np.random.rand()
            if r < 0.3:  # mutate elite
                parent = np.random.choice(elites)
                h = parent.copy()
                h += np.random.randn(n_points) * 0.025
                h = np.clip(h, 0, 1.0)
                total = h.sum()
                h = (h / total) * target_sum if total > 1e-6 else h
                h = np.clip(h, 0, 1.0)
                new_population.append(h)
            else:  # crossover
                p1 = np.random.choice(elites)
                p2 = np.random.choice(elites)
                alpha = np.random.rand() * 0.5 + 0.25

                h = (h / total) * target_sum if total > 1e-6 else h

                new_population.append(h)
        population = new_population
