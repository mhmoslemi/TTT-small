        worst_idx = np.argmax(corr)

        # Project to feasible region

        # Decay alpha

            grad[m] = 1 - h[i_plus] - h[i_minus]
        # Update h with gradient descent
        h = h - alpha * grad
        # Project to feasible region
        h = ensure_feasible(h, n_points, required_sum)
        # Perturb occasionally
        if iteration % perturb_interval == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = ensure_feasible(h, n_points, required_sum)
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Decay alpha
        alpha *= alpha_decay_rate
        # Check time
        if time.time() - start_time > budget_s - 10:
            break
    return best_h, best_c5, n_points
