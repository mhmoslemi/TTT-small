                new_h = current_h.copy()
                new_h[j] = down_val
                adjust_sum(new_h, target_sum)
                down_c5 = compute_c5(new_h)
                if down_c5 < orig_c5:
                    current_h = new_h.copy()
                    improved = True
            if not improved:
                break
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

    return best_h, best_c5, n_points
