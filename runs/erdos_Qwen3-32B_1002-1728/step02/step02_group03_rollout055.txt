def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.inf
    max_iter = 10000  # or based on time
    alpha = 0.01  # learning rate
    num_top_shifts = 5  # number of top shifts to consider

    start_time = time.time()
    while time.time() - start_time < budget_s - 0.1:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5_current = np.max(corr)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()

        # Find the indices of top shifts
        max_vals = np.sort(corr)
        top_k_vals = max_vals[-num_top_shifts:]
        top_k_indices = np.argwhere(corr >= top_k_vals[0]).flatten()

        # Compute shifts in steps for each of these indices
        shifts_in_steps = np.array([idx - (n_points - 1) for idx in top_k_indices])

        # Compute gradient for all top shifts
        grad = np.zeros_like(h)
        for k in shifts_in_steps:
            for i in range(n_points):
                forward_index = i + k
                if 0 <= forward_index < n_points:
                    grad[i] += (1 - h[forward_index])
                reverse_index = i - k
                if 0 <= reverse_index < n_points:
                    grad[i] += -h[reverse_index]

        # Normalize gradient
        # Compute the magnitude and normalize
        grad_norm = np.sqrt(np.sum(grad**2))
        if grad_norm > 0:
            grad /= grad_norm

        # Update h
        h -= alpha * grad

        # Clip h to [0,1]
        np.clip(h, 0, 1, out=h)

        # Rescale to maintain sum
        s = np.sum(h)
        if abs(s - target_sum) > 1e-10:  # only if sum not sufficient
            h *= target_sum / s

    # Final check
    corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_final)
    return best_h, final_c5, n_points

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    alpha = 0.005
    num_top_shifts = 5

    while time.time() - start_time < budget_s - 0.1:
        # Compute current overlaps
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Find top shifts contributing to max overlap
        sorted_corr = np.sort(corr)
        threshold = sorted_corr[-num_top_shifts]
        top_indices = np.argwhere(corr >= threshold).flatten()
        shifts_in_steps = [idx - (n_points - 1) for idx in top_indices]

        # Accumulate gradients
        grad = np.zeros(n_points, dtype=np.float64)
        for k in shifts_in_steps:
            for i in range(n_points):
                forward_idx = i + k
                if 0 <= forward_idx < n_points:
                    grad[i] += (1 - h[forward_idx])
                reverse_idx = i - k
                if 0 <= reverse_idx < n_points:
                    grad[i] += -h[reverse_idx]

        # Normalize gradient
        if np.sum(grad**2) > 1e-12:
            grad = (alpha * grad) / np.sqrt(np.sum(grad**2))

        # Apply update, clip, and rescale
        h -= grad
        np.clip(h, 0, 1, out=h)
        s = np.sum(h)
        h *= target_sum / max(s, 1e-6)

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
