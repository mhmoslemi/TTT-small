            corr = np.correlate(candidate_h, 1 - candidate_h, mode='full') * dx
            current_c5 = np.max(corr)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h[:] = candidate_h[:]
                high = mid
            else:
                low = mid
        else:
            low = mid
    
    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
