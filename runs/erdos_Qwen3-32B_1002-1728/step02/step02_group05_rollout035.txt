            curr_h = get_adjusted_h(result.x)
            corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            curr_c5 = np.max(corr_val)
            if (best_c5 is None or curr_c5 < best_c5) and np.all(curr_h <= 1.1):
                best_c5 = curr_c5
                best_h[:] = curr_h[:]

    while time.time() - start_time < budget_s - 5 and best_c5 > 0.3809:
        if not initial_guesses:
            break
        x0 = best_h.copy()
        x0 += np.random.normal(0, 0.05, n_points)
        x0 = np.clip(x0, 0, 1)
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[LinearConstraint(np.ones(n_points), target_sum, target_sum)],
            options={'maxiter': 50}
        )
        if result.success:
            curr_h = get_adjusted_h(result.x)
            corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            curr_c5 = np.max(corr_val)
            if curr_c5 < best_c5 and np.all(curr_h <= 1.1):
                best_c5 = curr_c5
                best_h[:] = curr_h[:]

    corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr_final), n_points
