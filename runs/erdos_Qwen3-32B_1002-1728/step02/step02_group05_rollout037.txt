import time

def project_h(h, target_sum):
    h = np.array(h, copy=True)

    if abs(curr_sum - target_sum) < 1e-8:
        return h
    delta = curr_sum - target_sum
    if delta > 0:  # Reduce values

        adjust = delta / len(valid)
        h[valid] -= adjust
    else:
        valid = np.where(h < 1 - 1e-6)[0]
        adjust = -delta / len(valid)
        h[valid] += adjust
    h = np.clip(h, 0.0, 1.0)
    return h
