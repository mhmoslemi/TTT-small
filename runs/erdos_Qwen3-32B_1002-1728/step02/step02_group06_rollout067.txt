from scipy.optimize import differential_evolution, Bounds, minimize, LinearConstraint
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    maxiter_de = 100

    def scale_and_clip(h, target_sum):
        h = np.ascontiguousarray(h)
        total = np.sum(h)
        if total == 0:
            h[0] = 1.0
            total = 1.0
        scale = target_sum / total
        h_clipped = np.clip(h * scale, 0.0, 1.0)
        return h_clipped

    def objective_de(h_flat):
        h = h_flat.reshape(-1)
        h_constrained = scale_and_clip(h, target_sum)
        corr = np.correlate(h_constrained, 1 - h_constrained, mode='full') * dx
        penalty = (np.sum(h_constrained) - target_sum) ** 2 * 1e5
        return np.max(corr) + penalty

    def objective_slsqp(h_flat):
        h = h_flat.reshape(-1)
        h_clipped = np.clip(h, 0.0, 1.0)
