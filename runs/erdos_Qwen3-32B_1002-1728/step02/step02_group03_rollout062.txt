import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = int(budget_s * 0.9)
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()

    for it in range(max_iter):
        # Compute current overlap
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_clipped.copy()

        # Exit early if time expires
        if time.time() - start_time > budget_s - 2:
            break

        # Find maximum overlap index and corresponding lag
        m_max = np.argmax(corr)
        k_samples = m_max - (n_points - 1)

        # Compute gradient for the dominant shift
        grad = np.zeros(n_points)
        for m in range(n_points):
            m_plus_k = m + k_samples
            if 0 <= m_plus_k < n_points:
                term1 = 1.0 - h_clipped[m_plus_k]
            else:
                term1 = 1.0
            m_minus_k = m - k_samples
            if 0 <= m_minus_k < n_points:
                term2 = -h_clipped[m_minus_k]
            else:
                term2 = 0.0
            grad[m] = term1 + term2

        # Adaptive learning rate
        alpha = 0.01 * (0.997 ** it)
        h_update = h_clipped - alpha * grad
        h_update = np.clip(h_update, 0, 1)

        # Project to feasible set (sum constraint)
        s = np.sum(h_update)
        if abs(s) > 1e-10:
            h_update *= target_sum / s
        else:
            # Fallback: uniform distribution
            h_update = np.full(n_points, 0.5)

        h = h_update.copy()

    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
