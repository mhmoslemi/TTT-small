    h = initial_h_values.copy()

    # Parameters
    alpha = 0.01
    max_alpha = 0.1
    min_alpha = 0.001
    alpha_decay = 0.995
    patience = 50
    no_improvement = 0
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full') * dx
        # Find maximum overlap for valid shifts (0 to n-1)
        relevant_idx_start = n_points - 1
        relevant_corr = corr[relevant_idx_start: relevant_idx_start + n_points]

            no_improvement = 0

        else:
            no_improvement += 1
            if no_improvement > patience:
                # Perturb
                h = best_h + np.random.normal(0, 0.01, n_points)
                no_improvement = 0
                alpha = max(alpha * alpha_decay, min_alpha)
        # Early exit if converged
        if no_improvement > patience * 2:
            break

        k_max = np.argmax(relevant_corr)
        # Compute gradient for k_max
        k = k_max
        # Compute gradient
        def compute_gradient(h, k, n):

def compute_gradient(h, k, n_points):
