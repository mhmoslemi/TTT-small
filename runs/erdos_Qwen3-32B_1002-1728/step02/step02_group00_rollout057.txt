    start_time = time.time()
    K = 32
    dx = 2.0 / K
    target_sum = K // 2  # 16
    h = np.zeros(K)
    h[::2] = 1.0  # Alternating 1 and 0
    assert np.sum(h) == target_sum, "Initial h sum is incorrect"

    def compute_c5(h_vec):
        kh = h_vec
        k1h = 1 - kh
        corr = np.correlate(kh, k1h, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    np.random.seed(seed + 123)

    # Annealing parameters
    initial_T = 100.0
    cooling_rate = 0.9995
    T = initial_T
    max_iterations = 500000  # Adjust based on budget

    while time.time() - start_time < budget_s * 0.95:
        # Generate a move
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]  # Swap
        new_c5 = compute_c5(h_new)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # Accept the move
            h = h_new.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        # Cool the temperature
        T *= cooling_rate
        if T < 1e-6:
            T = 1e-6

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum violation"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, K)
