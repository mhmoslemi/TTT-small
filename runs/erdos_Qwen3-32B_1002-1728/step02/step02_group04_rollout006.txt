    cross_corr = np.correlate(h, 1 - h, mode='full')

        h *= scaling
        h = np.clip(h, 0, 1)

    n_points = len(h)

    required_sum = n_points / 2.0

    decay_steps = 100

    iteration = 0

    while time.time() - start_time < budget_s - 1:

        cross_corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(cross_corr)

        shifted_sub = np.arange(n_points) - k

        term2 = np.where(mask_sub, h[shifted_sub.astype(int)], 0.0)

        grad = (term1 - term2) * dx

        # Update h
        h += alpha * grad

        # Project to feasible set
        h = project_to_feasible(h, required_sum)

        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

            alpha *= alpha_decay
