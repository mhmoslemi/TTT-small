    target_sum = n_points / 2.0

    h = h_init.copy()
    best_c5 = np.inf
    best_h = h_init.copy()

    max_iter = 3  # Limit iterations to respect time budget
    for iter in range(max_iter):
        # Continuous relaxation with cvxpy
        h_cont = cp.Variable(n_points)
        constraints = [0 <= h_cont, h_cont <= 1, cp.sum(h_cont) == target_sum]

        # Compute overlap expressions for all shifts
        exprs = []
        for k in range(n_points):
            terms = [h_cont[i] * (1 - h_cont[i + k]) for i in range(n_points - k)]
            expr = cp.sum(terms) * dx
            exprs.append(expr)
        for s in range(1, n_points):
            terms = [h_cont[i] * (1 - h_cont[i - s]) for i in range(s, n_points)]
            expr = cp.sum(terms) * dx
            exprs.append(expr)

        # Objective
        obj = cp.Minimize(cp.Max(exprs))
        prob = cp.Problem(obj, constraints)

        # Solve with ECOS (free solver)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, maximum_iterations=150)
        except:
            # Fallback to CVXOPT if ECOS is not available
            prob.solve(solver=cp.CVXOPT, verbose=False)

        h_opt = np.array(h_cont.value).flatten()

        # Binarize h_opt
        sorted_indices = np.argsort(-h_opt)
        cumsum = np.cumsum(h_opt[sorted_indices])
        idx = np.argmax(cumsum >= target_sum)
        threshold = h_opt[sorted_indices[idx]]
        h_bin = (h_opt >= threshold).astype(float)

        # Adjust sum if necessary
        h_bin_sum = np.sum(h_bin)
        if h_bin_sum < target_sum:
            # Find the smallest value >= threshold with h_bin=0 and set to 1
            candidates = (h_opt >= threshold) & (h_bin == 0)
            if np.any(candidates):
                idx_adj = np.argmin(h_opt[candidates])
                h_bin[sorted_indices[idx_adj]] = 1.0
        elif h_bin_sum > target_sum:
            # Find the largest value <= threshold with h_bin=1 and set to 0
            candidates = (h_opt <= threshold) & (h_bin == 1)
            if np.any(candidates):
                idx_adj = np.argmax(h_opt[candidates])
                h_bin[sorted_indices[idx_adj]] = 0.0

        # Check and update best solution
        corr = np.correlate(h_bin, 1 - h_bin, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_bin.copy()

        h = best_h.copy()

        if time.time() - start_time > budget_s * 0.98:
            break

    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    return (best_h.tolist(), final_c5, n_points)
