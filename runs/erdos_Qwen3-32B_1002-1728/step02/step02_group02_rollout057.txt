import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_flat):
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    bounds = Bounds(0.0, 1.0)
    A_eq = np.ones((1, n_points))
    linear_constraint = LinearConstraint(A_eq, lb=target_sum, ub=target_sum)

    start_time = time.time()
    best_result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=[linear_constraint],
        options={'maxiter': 5000, 'ftol': 1e-9, 'disp': False}
    )

    while time.time() - start_time < budget_s - 0.1:
        iter_remain = int((budget_s - (time.time() - start_time)) // 0.1)
        if iter_remain < 1:
            break
        result = minimize(
            fun=objective,
            x0=best_result.x,
            method='SLSQP',
            bounds=bounds,
            constraints=[linear_constraint],
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        if result.success and result.fun < best_result.fun:
            best_result = result

    optimized_h = np.clip(best_result.x, 0, 1)
    sum_h = np.sum(optimized_h)
    if abs(sum_h - target_sum) > 1e-6:
        scale = target_sum / sum_h
        optimized_h = np.clip(optimized_h * scale, 0, 1)

    final_corr = np.correlate(optimized_h, 1 - optimized_h, mode='full') * dx
    return optimized_h, np.max(final_corr), n_points
