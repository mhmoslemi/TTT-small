import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # ∑w_i = n_points/2
    
    # Construct x samples
    x = dx * np.arange(n_points)
    
    # Kernel parameters
    sigma = 0.1  # Found via tuning trials
    
    # Compute kernel matrix K_ij = exp(-((x_i - x_j)/sigma)^2 / 2)
    K = np.zeros((n_points, n_points))
    for i in range(n_points):
        for j in range(n_points):
            dist = (x[i] - x[j]) / sigma
            K[i, j] = np.exp(-dist**2 / 2)
    
    # Row-normalize kernel to [0,1] range
    row_sums = K.sum(axis=1)
    K = K / row_sums[:, np.newaxis]
    
    # Objective: max over (h ⊗ (1-h)) * dx
    def objective(w_vec):
        w = w_vec.reshape(n_points)
        h = K @ w
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Constraint: sum(w_i) = target_sum
    def constraint_sum(w_vec):
        return np.sum(w_vec) - target_sum
    
    # Initial guess: project initial_h_values to meet target_sum constraint
    w0 = h_init.copy()
    w0_sum = np.sum(w0)
    if w0_sum != 0:
        w0 = w0 * (target_sum / w0_sum)
    w0 = np.clip(w0, 0, 1)
    
    # Setup optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=w0,
        method='SLSQP',
        bounds=bounds,

    )

    h_final = K @ best_w
