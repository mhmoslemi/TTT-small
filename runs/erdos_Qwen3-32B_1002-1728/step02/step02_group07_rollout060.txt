import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 1.0 - 1e-6)
        if can_incr.any():
            total_available = (1.0 - h_adj[can_incr]).sum()
            if total_available > 1e-9:
                h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_available
    else:
        D = -D
        can_decr = (h_adj > 1e-6)
        if can_decr.any():
            total_available = h_adj[can_decr].sum()
            if total_available > 1e-9:
                h_adj[can_decr] -= D * h_adj[can_decr] / total_available
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    h_initial = adjust_sum(h_initial, target_sum)
    
    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return np.max(corr)
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5
    
    np.random.seed(seed)
    start_time = time.time()
    T_initial = max(0.1, 10.0 * current_c5)
    
    while time.time() - start_time < budget_s:
        t = int((time.time() - start_time) / 10) + 1
        T = max(0.01, T_initial / np.log(1 + t))
        pert_scale = max(0.005, 0.18 * T)
        
        candidate_h = current_h + pert_scale * np.random.randn(n_points)
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
