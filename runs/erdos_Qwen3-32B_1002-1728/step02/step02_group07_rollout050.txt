import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = cp.Variable(n_points)
    t = cp.Variable()

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]

    for k in range(-(n_points - 1), n_points):
        if k >= 0:
            start_i = 0
            end_i = n_points - 1 - k
        else:
            start_i = -k
            end_i = n_points - 1
        if start_i > end_i:
            continue
        i_list = list(range(start_i, end_i + 1))
        if not i_list:
            continue
        indices = np.array(i_list)
        terms = h[indices] * h[indices + k]
        sum_ij = cp.sum(terms)
        constraints.append(1 - dx * sum_ij <= t)

    objective = cp.Minimize(t)
    problem = cp.Problem(objective, constraints)

    start_time = time.time()
    try:
        problem.solve(solver=cp.ECOS, max_iters=200, verbose=False)
    except Exception as e:
        pass

    h_sol = h.value
    h_sol = np.clip(h_sol, 0, 1)
    current_sum = np.sum(h_sol)
    if not np.isclose(current_sum, target_sum):
        scale = target_sum / current_sum
        h_sol = h_sol * scale
        h_sol = np.clip(h_sol, 0, 1)

    corr_values = np.correlate(h_sol, 1 - h_sol, mode='full') * dx
    c5_bound = np.max(corr_values)
