    def compute_c5(h):
        h = np.asarray(h)
        c = correlate(h, 1.0 - h, mode='full')
        return np.max(c) * dx

    # Repair function to maintain constraints
    def repair(h):

        current_sum = np.sum(h)
        delta = target_sum - current_sum
        if abs(delta) > 1e-6:

            h[0] = np.clip(h[0], 0.0, 1.0)

    current_h = repair(h_initial)
    best_h = current_h.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # Simulated Annealing parameters
    initial_temperature = 0.1
    cooling_rate = 0.999
    T = initial_temperature
