        grad[i] = (1 - h[j]) - h[i_minus_k]
    return grad

def project(h, required_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if abs(current_sum - required_sum) < 1e-10:
        return h_clipped
    # Scale to required sum
    h_clipped *= required_sum / current_sum
    h_clipped = np.clip(h_clipped, 0, 1)
    residual = required_sum - np.sum(h_clipped)
    h_clipped[0] += residual
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2

    h = project(h_initial, required_sum)
    best_h = h.copy()
    worst_k, worst_value = compute_worst_k_and_value(h)
    best_c5 = worst_value * dx

    np.random.seed(seed)
    
    initial_lr = 0.01
    decay_rate = 0.95
    perturb_interval = 200
    patience = 50
    start_time = time.time()
    lr = initial_lr
    iter_count = 0
    last_improvement = 0

    while time.time() - start_time < budget_s - 5:
        iter_count += 1
        worst_k, current_value = compute_worst_k_and_value(h)
        current_c5 = current_value * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count

        # Early stopping
        if iter_count - last_improvement > patience:
            break

        # Compute gradient
        grad = compute_gradient(h, worst_k, n_points)
        
        # Update with learning rate
        h = project(h - lr * grad, required_sum)
        
        # Adjust learning rate
        if iter_count % 100 == 0:
            lr *= decay_rate
            lr = max(lr, 1e-6)
        
        # Perturbation
        if iter_count % perturb_interval == 0:
            perturbation = np.random.normal(0, 0.05, n_points)
            h += perturbation
            h = project(h, required_sum)

        # Break early if time is tight
        if time.time() - start_time + 3 > budget_s:
            break

    return best_h, best_c5, n_points
