def run(seed=42, budget_s=1100):

    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    start_time = time.time()
    best_c5 = np.inf
    h_best = h.copy()
    solver = cp.SCIP
    max_iter = 5000
    no_improve = 0
    tolerance = 1e-7

    for iter_idx in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break

        # Compute current worst overlap
        corr = correlate(h, 1 - h, mode='full')
        relevant_corr = corr[n_points - 1:2 * n_points - 1]
        current_c5 = np.max(relevant_corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            h_best = h.copy()
            no_improve = 0
        else:

            if no_improve >= 15:

        objective = cp.Minimize(objective_expr)
