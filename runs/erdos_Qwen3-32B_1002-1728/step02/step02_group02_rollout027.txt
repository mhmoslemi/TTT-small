        return np.max(corr) * dx

    current_c5 = compute_c5(h_current)
    best_c5 = current_c5
    best_h = h_current.copy()

    while time.time() - start_time < budget_s - 0.1:
        h_new = h_current.copy()
        perturb_count = np.random.randint(2, 5)
        indices = np.random.choice(n_points, size=perturb_count, replace=False)
        delta = (np.random.rand(perturb_count) - 0.5) * 0.4

        new_c5 = compute_c5(h_new)
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0:
