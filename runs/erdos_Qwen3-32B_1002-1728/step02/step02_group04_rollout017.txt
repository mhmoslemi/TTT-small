import numpy as np
import time
import cvxpy as cp

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, c5_list, tournament_size=10):
    indices = np.random.choice(len(population), size=tournament_size, replace=False)
    best_idx = indices[np.argmin([c5_list[i] for i in indices])]
    return population[best_idx].copy()

def ensure_feasible(candidate, n_points, required_sum):
    candidate = np.clip(candidate, 0, 1)
    current_sum = candidate.sum()
    if current_sum < 1e-10:
        candidate.fill(required_sum / n_points)
    else:
        candidate *= required_sum / current_sum
        candidate = np.clip(candidate, 0, 1)
        excess = candidate.sum() - required_sum
        if abs(excess) > 1e-8 and abs(excess) < 1e-6:
            candidate -= excess / n_points
            candidate = np.clip(candidate, 0, 1)
    return candidate

            child = ensure_feasible(child, n_points, required_sum)
