    mutation_rate = 0.03
    mutation_std = 0.08
    tournament_size = 5
    elitism_ratio = 0.15
    grad_steps = 30
    grad_lr = 0.01

    def scale_to_sum(h):
        s = np.sum(h)
        if s < 1e-10:
            return h
        scale = target_sum / s
        return np.clip(h * scale, 0, 1)

    def compute_c5_and_k(h):
        n = len(h)
        dx = 2.0 / n
        max_c5 = -np.inf
        best_k = 0
        for k in range(-n+1, n):
            overlap = 0.0
            for j in range(n):
                if 0 <= j < n and 0 <= j + k < n:
                    overlap += h[j] * (1 - h[j + k])
            if overlap > max_c5:
                max_c5 = overlap
                best_k = k
        return max_c5 * dx, best_k

    def compute_gradient(h, k, dx):
        n = len(h)
        gradient = np.zeros(n)
        for i in range(n):
            term1 = 0.0
            if 0 <= i + k < n:
                term1 = (1.0 - h[i + k])
            term2 = 0.0
            if 0 <= i - k < n:
                term2 = -h[i - k]
            gradient[i] = (term1 + term2) * dx
        return gradient

    def refine_with_gradient(h, dx, max_steps=grad_steps, learning_rate=grad_lr, rescale=True):
        learning_rate = grad_lr
        for step in range(max_steps):
            max_c5, best_k = compute_c5_and_k(h)
            grad = compute_gradient(h, best_k, dx)
            direction = -grad
            h_new = h + learning_rate * direction
            h_new = np.clip(h_new, 0, 1)
            if rescale:
                h_new = scale_to_sum(h_new)
            new_c5, _ = compute_c5_and_k(h_new)
            if new_c5 < max_c5:
                h = h_new.copy()
            else:
                learning_rate *= 0.95
                if learning_rate < 1e-5:
                    break
                h_new = h + learning_rate * direction
                h_new = np.clip(h_new, 0, 1)
                if rescale:
                    h_new = scale_to_sum(h_new)
                new_c5, _ = compute_c5_and_k(h_new)
                if new_c5 < max_c5:
                    h = h_new.copy()
                else:
                    break
        return h

    def evaluate(candidates):
        c5_values = []
        best_ks = []
        for h in candidates:
            c5, bk = compute_c5_and_k(h)
            c5_values.append(c5)
            best_ks.append(bk)
        return np.array(c5_values), np.array(best_ks)

    def mutate(h):
        h = np.copy(h)
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, mutation_std, size=np.sum(mask))
        h = np.clip(h, 0, 1)
        h = scale_to_sum(h)
        return h

    def crossover(parent1, parent2):
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        child = scale_to_sum(child)
        return child

    def tournament_select(candidates, scores, size):
        indices = np.random.choice(len(candidates), size)
        winner_idx = indices[np.argmin(scores[indices])]
        return candidates[winner_idx]

    population = [h_initial]
    for _ in range(pop_size - 1):
        mut = h_initial + np.random.normal(0, 0.05, n_points)
        mut = scale_to_sum(np.clip(mut, 0, 1))
        population.append(mut)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = None
    start_time = time.time()

    while time.time() - start_time < budget_s - 1:
        scores, best_ks = evaluate(population)
        curr_best_idx = np.argmin(scores)
        if best_c5 is None or scores[curr_best_idx] < best_c5:
            best_c5 = scores[curr_best_idx]
            best_k = best_ks[curr_best_idx]

        # Refine best candidate
        refined = refine_with_gradient(population[curr_best_idx], dx, max_steps=grad_steps, learning_rate=grad_lr)
        refined_c5, _ = compute_c5_and_k(refined)
        if refined_c5 < best_c5:
            best_h = refined.copy()
            best_c5 = refined_c5

        # Evolutionary steps
        elm_size = int(elitism_ratio * pop_size)
        elitists = population[np.argsort(scores)[:elm_size]].copy()

        new_population = elitists.copy()
        while len(new_population) < pop_size:
            parent1 = tournament_select(population, scores, tournament_size)
            parent2 = tournament_select(population, scores, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population = np.vstack([new_population, child])

        population = new_population[:pop_size]

    corr_values = compute_c5_and_k(best_h)[0]
    final_c5 = corr_values
    # Final refinement
    best_h = refine_with_gradient(best_h, dx, max_steps=10*grad_steps, learning_rate=grad_lr)
    final_c5, _ = compute_c5_and_k(best_h)
    return best_h, final_c5, n_points
