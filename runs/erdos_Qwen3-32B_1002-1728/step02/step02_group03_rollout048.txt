        k = abs(k_shift)

        if k == 0:
            continue
        
        # Compute gradient for this k
        grad = np.zeros(n_points)
        for i in range(k, n_points):
            j = i - k
            grad[i] += (1.0 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Gradient descent step
        h = h - alpha * grad
        
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)

        current_sum = np.sum(h)
        if abs(current_sum) < 1e-10:

    # Final evaluation
    corr = np.correlate(h, 1 - h, mode='full') * dx
    c5_bound = np.max(corr)
