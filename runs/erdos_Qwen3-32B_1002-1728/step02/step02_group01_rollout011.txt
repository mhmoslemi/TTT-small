import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    h = np.clip(h, 0, 1)

    if total != 0:
        h = (h / total) * target_sum

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    np.random.seed(seed)

    # Simulated annealing parameters
    T = 0.1
    alpha = 0.99
    min_T = 1e-6

    while time.time() - start_time < budget_s and T > min_T:
        # Generate perturbed solution
        perturbed = best_h + np.random.normal(0, T * 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)

            perturbed = (perturbed / total_p) * target_sum

        corr = np.correlate(perturbed, 1 - perturbed, mode='full')
        new_c5 = np.max(corr) * dx

        # Acceptance criterion
        delta_c5 = new_c5 - best_c5
        if delta_c5 <= 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h = perturbed.copy()
            best_c5 = new_c5

        # Cool down
        T *= alpha

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
