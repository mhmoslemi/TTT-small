def run(seed=42, budget_s=1100, **kwargs):

    def project(h, required_sum):
        h_proj = np.clip(h, 0.0, 1.0)
        s = h_proj.sum()
        if np.allclose(s, required_sum):
            return h_proj
        delta = required_sum - s
        # Adjust h_proj to add delta
        for i in range(n_points):
            if delta <= 1e-10:
                break
            if h_proj[i] < 1.0:
                add = min(1.0 - h_proj[i], delta)
                h_proj[i] += add
                delta -= add
        return h_proj

    h = project(h, required_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    np.random.seed(seed)
    start_time = time.time()
    alpha = 0.01
    decay = 0.99
    patience = 0
    max_patience = 50  # number of iterations without improvement before perturbing
    iter_count = 0

    while time.time() - start_time < budget_s - 10:
        iter_count += 1
        # Compute current overlaps
        corr = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(corr)
        k = k_max
        # Compute gradient for k
        h_forward = np.roll(h, shift=k)
        h_backward = np.roll(h, shift=-k)
        grad = (1 - h_forward - h_backward) * dx
        # Update h
        h = h - alpha * grad
        # Project back
        h = project(h, required_sum)
        # Compute new C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                # Perturb
                h = best_h + np.random.normal(0, 0.05, n_points)
                h = project(h, required_sum)
                patience = 0
        # Decay alpha
        alpha *= decay
        # Occasionally reset alpha to initial if very small
        if alpha < 1e-4:
            alpha = 0.01
    return best_h, best_c5, n_points
