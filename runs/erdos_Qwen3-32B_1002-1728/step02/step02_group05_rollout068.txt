        current_sum = np.sum(h)
        if current_sum != 0:
            h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        offset = np.sum(h) - target_sum
        h[0] -= offset
        h[0] = max(0.0, min(1.0, h[0]))
        particles[i] = h

    for i in range(num_particles):
        fitness = compute_c5(particles[i], n_points, dx)
        if fitness < personal_best_fitnesses[i]:
            personal_best_fitnesses[i] = fitness
            personal_best_positions[i] = particles[i].copy()
        if fitness < global_best_fitness:
            global_best_fitness = fitness
            global_best_position = particles[i].copy()

    c1 = 2.0
    c2 = 2.0
    w0 = 0.9
    w_min = 0.4
    start_time = time.time()

    while True:
        current_time = time.time()
        if current_time - start_time > budget_s - 5:
            break

        w = max(w_min, w0 - (w0 - w_min) * (current_time - start_time) / budget_s)
        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel_cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            vel_social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            target_sum = n_points / 2.0
            current_sum = np.sum(particles[i])
            offset = current_sum - target_sum
            particles[i][0] -= offset
            particles[i][0] = max(0.0, min(1.0, particles[i][0]))

        for i in range(num_particles):
            fitness = compute_c5(particles[i], n_points, dx)
            if fitness < personal_best_fitnesses[i]:
                personal_best_fitnesses[i] = fitness
                personal_best_positions[i] = particles[i].copy()
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best_position = particles[i].copy()

    corr_values = scipy.signal.correlate(global_best_position, 1 - global_best_position, mode='full') * dx
    final_c5 = np.max(corr_values)
    return global_best_position, final_c5, n_points
