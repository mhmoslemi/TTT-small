import numpy as np
import time

def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
    return pattern

def create_random_candidate(n_points):
    h = np.random.rand(n_points)
    h /= h.sum()
    h *= n_points / 2.0
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2.0
    h[0] += target_sum - h.sum()
    h = np.clip(h, 0, 1)
    return h

def mutate(h, sigma=0.05):
    h = h.copy()
    h += np.random.normal(0, sigma, len(h))
    h = np.clip(h, 0, 1)
    target_sum = len(h) / 2.0
    h[0] += target_sum - h.sum()
    h = np.clip(h, 0, 1)
    return h

def crossover(h1, h2):
    h1 = h1.copy()
    h2 = h2.copy()
    alpha = np.random.uniform(0.3, 0.7)
    h_child = alpha * h1 + (1 - alpha) * h2
    h_child = np.clip(h_child, 0, 1)
    target_sum = len(h_child) / 2.0
    h_child[0] += target_sum - h_child.sum()
    h_child = np.clip(h_child, 0, 1)
    return h_child

def tournament_selection(population, fitness, size=3):
    indices = np.random.choice(len(population), size, replace=False)
    best_index = indices[np.argmin(fitness[indices])]
    return population[best_index].copy()
