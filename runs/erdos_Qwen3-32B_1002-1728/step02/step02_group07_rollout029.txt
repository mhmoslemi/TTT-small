def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.99999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible

import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.99999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-6)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def create_intervals(n_points, m):
    intervals = []
    points_per_interval = n_points // m
    remainder = n_points % m
    start = 0
    for i in range(m):
        size = points_per_interval + (1 if i < remainder else 0)
        end = start + size
        intervals.append((start, end))
        start = end
    return intervals

def get_partitioned_values(h_full, intervals):
    h_part = []
    for start, end in intervals:
        avg = np.mean(h_full[start:end])
        h_part.append(avg)
    return np.array(h_part)

def assemble_h(intervals, h_part, n_points):
    h_full = np.zeros(n_points)
    for i, (start, end) in enumerate(intervals):
        h_full[start:end] = h_part[i]
    return h_full

def objective_partition(x, intervals, n_points, dx):
    x = x.reshape(-1)
    h_full = assemble_h(intervals, x, n_points)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    return np.max(corr)
