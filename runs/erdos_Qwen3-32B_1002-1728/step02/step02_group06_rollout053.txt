import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h_initial.copy()
    # Initial evaluation
    corr_init = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr_init) * dx
    current_h = best_h.copy()

    T = 0.1  # Initial temperature
    alpha = 0.999  # Cooling rate
    start_time = time.time()

    while time.time() - start_time < budget_s - 10 and T > 1e-5:
        # Perturbation with adaptive noise scale
        noise = np.random.normal(0, 1, n_points) * T * 0.1
        h_perturbed = current_h + noise
        # Repair: clip then adjust sum
        h_perturbed = np.clip(h_perturbed, 0, 1)
        sum_perturbed = np.sum(h_perturbed)
        diff = target_sum - sum_perturbed
        h_perturbed[0] += diff
        h_perturbed = np.clip(h_perturbed, 0, 1)

        # Evaluate candidate
        corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(corr) * dx

        # Acceptance criterion
        if new_c5 < best_c5 or np.random.rand() < math.exp((best_c5 - new_c5) / T):
            current_h[:] = h_perturbed
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h[:] = current_h

        T *= alpha

    # Final evaluation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
