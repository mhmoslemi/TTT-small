import cvxpy as cp
from collections import defaultdict

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Precompute all (i,j) pairs with i < j
    all_pairs = [(i, j) for i in range(n_points) for j in range(i+1, n_points)]
    k_to_pair_indices = defaultdict(list)
    for idx, (i, j) in enumerate(all_pairs):
        k = j - i
        k_to_pair_indices[k].append(idx)

    valid_ks = sorted(k_to_pair_indices.keys())

    # Precompute initial C5 value
    corr = np.correlate(h_initial, 1 - h_initial, mode='full') * dx
    c5_initial = np.max(corr)

    best_h = h_initial.copy()
    best_c5 = c5_initial
    start_time = time.time()

    low = 0.38  # Target lower bound
    high = 0.381  # Initial upper bound guess
    max_binary_steps = 40  # Limit binary search to stay within budget

    for iter in range(max_binary_steps):
        if time.time() - start_time > budget_s - 30:
            break
        mid = (low + high) / 2

        # Construct LP with cvxpy
        h = cp.Variable(n_points)
        y = cp.Variable(len(all_pairs))

        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2.0
        ]

        for idx, (i, j) in enumerate(all_pairs):
            constraints += [
                y[idx] >= h[i] + h[j] - 1,
                y[idx] >= 0
            ]

        for k in valid_ks:
