import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def _project(p):
        clamped = np.clip(p, 0, 1)
        clamped_sum = clamped.sum()
        if clamped_sum == 0:
            return np.ones_like(clamped) * 0.5
        scale = target_sum / clamped_sum
        scaled = clamped * scale
        return np.clip(scaled, 0, 1)

    def evaluate(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = []
    # Generate initial population
    for _ in range(5):  # 10% perturbations of initial_h
        h = h_initial.copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        population.append(_project(h))
    for _ in range(10):  # 20% random solutions
        h = np.random.rand(n_points)
        population.append(_project(h))
    for _ in range(35):  # 70% slide mutants of initial_h
        h = h_initial.copy()
        start = np.random.randint(0, n_points - 2)
        end = np.random.randint(start + 2, min(start + 10, n_points))
        delta = np.random.uniform(-0.1, 0.1)
        h[start:end] += delta
        population.append(_project(h))

    population = [p.copy() for p in population]
    scores = np.array([evaluate(ind) for ind in population])
    best_idx = np.argmin(scores)
    best_individual = population[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s - 0.1:
        num_offspring = 10
        children = []
        for _ in range(num_offspring):
            # Tournament selection
            candidates1 = np.random.choice(population_size, 3, replace=False)
            parent1_idx = candidates1[np.argmin(scores[candidates1])]
            candidates2 = np.random.choice(population_size, 3, replace=False)
            parent2_idx = candidates2[np.argmin(scores[candidates2])]
            parent1 = population[parent1_idx].copy()
            parent2 = population[parent2_idx].copy()

            # Crossover
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, parent1, parent2)
            child = _project(child)

            if mutation_type < 0.1:  # Jitter mutation
                noise = np.random.normal(0, 0.05, n_points)
                child += noise

            elif mutation_type < 0.3:  # Invert mutation

                end = np.random.randint(start + 2, min(start + 10, n_points))
                child[start:end] = 1 - child[start:end]

            else:  # Slide mutation

                end = np.random.randint(start + 2, min(start + 10, n_points))
                delta = np.random.uniform(-0.1, 0.1)

        # Combine and select top 50

    corr_values = correlate(best_individual, 1 - best_individual, mode='full') * dx
    return best_individual, np.max(corr_values), n_points
