import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    np.random.seed(seed)
    
    start_time = time.time()
    best_c = 0.38085  # Current upper bound
    best_h = h_initial.copy()
    
    # Binary search for minimal C
    low = 0.0
    high = best_c
    for _ in range(80):  # Max iterations within time budget
        if time.time() - start_time > budget_s - 20:
            break
        mid = (low + high) / 2
        
        # Create variables and constraints for current mid
        h = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h) == n_points / 2.0,
            h <= 1,
            h >= 0
        ]
        
        # For each lag k in 1, 2, ..., n_points-1
        for k in range(1, n_points):  
            j_array = (np.arange(n_points) + k) % n_points
            u_k = cp.Variable(n_points, nonneg=True)
            # Constraints to approximate h[i] * h[j]
            constraints.extend([
                u_k <= h, 
                u_k <= h[j_array],
                u_k >= h + h[j_array] - 1
            ])
            # Ensure 1 - C <= sum(u_k) * dx → sum(u_k) >= (1 - C) / dx
            constraints.append(cp.sum(u_k) >= (1 - mid) * n_points / 2.0)
        
        # Check feasibility
        problem = cp.Problem(cp.Minimize(0), constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False, max_iters=150)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                best_c = mid
                best_h = h.value.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid  # Skip error, adjust lower bound
    
    # Post-process to ensure sum constraint
    offset = np.sum(best_h) - n_points / 2
    best_h[0] = max(0.0, min(1.0, best_h[0] - offset))
    
    # Final evaluation
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_vals)
    return best_h, final_c5, n_points
