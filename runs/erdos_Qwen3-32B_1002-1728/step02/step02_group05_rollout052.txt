        G = np.fft.fft(1 - h, fft_len)
        corr = np.fft.ifft(H * G).real
        return np.max(corr) * dx

    num_particles = 100
    particles_position = np.zeros((num_particles, n_points))
    for i in range(50):
        particles_position[i] = h_initial.copy()
    for i in range(50, num_particles):
        h = np.random.uniform(0, 1, n_points)
        h_clamped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clamped)
        if current_sum > 1e-8:
            scale = (n_points / 2.0) / current_sum
            h_clamped *= scale
            h_clamped = np.clip(h_clamped, 0, 1)
            offset = (n_points / 2.0 - np.sum(h_clamped))
            h_clamped[0] += offset
            h_clamped[0] = np.clip(h_clamped[0], 0, 1)
        particles_position[i] = h_clamped

    particles_velocity = np.zeros((num_particles, n_points))
    pbest_position = particles_position.copy()
    pbest_c5 = np.array([compute_c5(p) for p in particles_position])
    gbest_index = np.argmin(pbest_c5)
    gbest_position = particles_position[gbest_index].copy()
    best_c5 = pbest_c5[gbest_index]
    start_time = time.time()

    c1, c2 = 2.0, 2.0
    w_min, w_max = 0.4, 0.9

    while time.time() - start_time < budget_s - 1:
        current_time = time.time() - start_time
        time_f = min(1.0, current_time / budget_s)
        w = w_max - (w_max - w_min) * time_f

        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (pbest_position[i] - particles_position[i])
            social = c2 * r2 * (gbest_position - particles_position[i])
            particles_velocity[i] = w * particles_velocity[i] + cognitive + social

            new_pos = particles_position[i] + particles_velocity[i]
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = np.sum(new_pos)
            if abs(current_sum - n_points / 2) > 1e-8:
                if current_sum > 1e-8:
                    scale = (n_points / 2.0) / current_sum
                    new_pos = new_pos * scale
                    new_pos = np.clip(new_pos, 0, 1)
                offset = (n_points / 2.0 - np.sum(new_pos))
                new_pos[0] += offset
                new_pos[0] = np.clip(new_pos[0], 0, 1)
            particles_position[i] = new_pos

            current_c5 = compute_c5(new_pos)
            if current_c5 < pbest_c5[i]:
                pbest_c5[i] = current_c5
                pbest_position[i] = new_pos.copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                gbest_position = new_pos.copy()

    final_c5 = compute_c5(gbest_position)
    return gbest_position, final_c5, n_points
