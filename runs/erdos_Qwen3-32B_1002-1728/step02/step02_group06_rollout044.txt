def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def generate_random_binary(n_points):
    target_sum = n_points / 2
    ones_needed = int(target_sum)
    vector = np.zeros(n_points)
    idx = np.random.choice(n_points, ones_needed, replace=False)
    vector[idx] = 1.0
    if target_sum - ones_needed > 0:
        idx = np.random.choice(n_points, 1)
        vector[idx[0]] = 0.5
    return vector

    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
