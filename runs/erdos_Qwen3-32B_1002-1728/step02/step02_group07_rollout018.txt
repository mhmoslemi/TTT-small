import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        h = h.reshape(-1)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()
    
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()
    perturb_strength = 0.05
    n_no_imp = 0
    
    while time.time() - start_time < budget_s - 10:
        # Perturb and adjust
        x0 = best_h.copy() + perturb_strength * np.random.randn(n_points)
        x0 = np.clip(x0, 0, 1)
        x0 = adjust_sum(x0, target_sum)
        
        # Local optimization
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        
        curr_h = np.clip(result.x, 0, 1)
        curr_h = adjust_sum(curr_h, target_sum)
        
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
            perturb_strength = max(0.005, perturb_strength * 0.8)
            n_no_imp = 0
        else:
            n_no_imp += 1
            if n_no_imp >= 5:
                perturb_strength = min(1.0, perturb_strength * 1.5)
                n_no_imp = 0
    
    return best_h, best_c5, n_points
