import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    population_size = 40
    offspring_per = 3
    mutation_strength = 0.15
    start_time = time.time()
    
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = np.random.uniform(0.0, 1.0, n_points)
        h = adjust_sum(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    while time.time() - start_time < budget_s - 5:
        time_remaining = budget_s - (time.time() - start_time)
        
        # Evaluate all population members
        c5_values = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            c5_values.append(c5)
        
        # Track best solution
        for h, c5 in zip(population, c5_values):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Select top 25%
        scores = np.array(c5_values)
        top_indices = np.argsort(scores)[:max(1, int(0.25 * len(population)))]
        top_individuals = [population[i] for i in top_indices]
        
        # Generate offspring
        offspring = []
        for h in top_individuals:
            for _ in range(offspring_per):
                # Mutate with noise and adjust
                h_mut = h.copy()
                noise = mutation_strength * np.random.randn(n_points)
                h_mut += noise
                h_mut = np.clip(h_mut, 0.0, 1.0)
                h_mut = adjust_sum(h_mut, target_sum)
                offspring.append(h_mut)
        
        # Combine and select new population
        combined = top_individuals + offspring
        if len(combined) > 2 * population_size:
            # Re-evaluate to avoid stale values
            combined_scores = [np.max(np.correlate(h, 1 - h, mode='full')) * dx for h in combined]
            indices = np.argsort(combined_scores)
            population = [combined[i] for i in indices[:population_size]]
        else:
            population = combined[:]
        
        # Decay mutation strength
        if time.time() - start_time > budget_s - 150:
            mutation_strength *= 0.95  # Closer to end
    
    return best_h, best_c5, n_points
