import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    alpha = 0.01
    max_iterations = int(budget_s * 0.95)  # Assume each iteration takes ~1e-3s
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    for iteration in range(max_iterations):
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Find the k with maximum overlap
        max_idx = np.argmax(corr)
        k = max_idx - (n_points -1)  # compute the actual shift k

        # Compute gradient for this k
        grad = np.ones(n_points)
        j = np.arange(n_points) + k
        valid = np.bitwise_and(j >=0, j <n_points)
        grad[valid] = 1 - h[j[valid]]
        # Apply gradient step
        h -= alpha * grad
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Project to feasible sum
        sum_h = np.sum(h)
        if abs(sum_h - target_sum) > 1e-8:
            scale = target_sum / sum_h
            h = h * scale
            h = np.clip(h, 0, 1)  # clip again after scaling to ensure [0,1]
        # Periodically check if we need to adjust alpha
        if iteration % 100 == 0:
            alpha *= 0.95  # decay learning rate

        # Check time
        if time.time() - start_time > budget_s - 1:
            break

    # Finally, check one more time
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    return best_h, final_c5, n_points
