import time

def run(seed=42, budget_s=1100, **kwargs):

        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    best_c5 = None
    best_h = h_initial.copy()

    initial_guesses = [h_initial.copy()]
    periods = [3,4,5,6,7,8,9]

            h = np.zeros(n_points)
            indices = ((np.arange(n_points) - phase) % period) == 0
            h[indices] = 1.0
            target_sum = n_points / 2.0
            current_sum = np.sum(h)
            if current_sum > 1e-8:
                scale = target_sum / current_sum
                h = h * scale
                h = np.clip(h, 0, 1)
                offset = target_sum - np.sum(h)
                h[0] += offset
