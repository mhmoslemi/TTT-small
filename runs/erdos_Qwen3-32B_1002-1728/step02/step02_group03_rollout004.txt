            score = evaluate_c5(particles[i])
            if score < pbest_scores[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest_position = particles[i].copy()
                    gbest_score = score

        if iteration % 100 == 0:
            top_indices = np.argsort(pbest_scores)[:3]
            for idx in top_indices:
                h = pbest_positions[idx].copy()
                for _ in range(5):
                    grad = np.zeros(n_points)
                    for j in range(n_points):
                        eps = 0.01
                        h_up = h.copy()
                        h_up[j] += eps
                        h_up = scale_to_sum(h_up)
                        du = evaluate_c5(h_up)
                        
                        h_down = h.copy()
                        h_down[j] -= eps
                        h_down = scale_to_sum(h_down)
                        dd = evaluate_c5(h_down)
                        
                        grad[j] = (du - dd) / (2 * eps)
                    h = h - 0.005 * grad
                    h = np.clip(h, 0, 1)
                    h = scale_to_sum(h)
                    if evaluate_c5(h) < gbest_score:
                        gbest_position = h.copy()
                        gbest_score = evaluate_c5(h)

    final_corr = np.correlate(gbest_position, 1 - gbest_position, mode='full') * dx
    final_c5 = np.max(final_corr)
    return gbest_position, final_c5, n_points
