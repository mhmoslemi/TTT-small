import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def generate_periodic_pattern(p):
        pattern = np.zeros(n_points)
        half_p = p // 2
        for i in range(n_points):
            if i % p < half_p:
                pattern[i] = 1.0
        return pattern

    p_list = [2, 4, 6, 8, 10]
    periodic_patterns = [generate_periodic_pattern(p) for p in p_list]

    def repair_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if abs(current_sum - target_sum) < 1e-6:
            return h_clipped
        delta = target_sum - current_sum
        h_clipped[0] += delta
        return np.clip(h_clipped, 0, 1)

    def random_individual():
        h = np.random.uniform(0, 1, n_points)
        return repair_h(h)

    population = [h_initial] + periodic_patterns
    while len(population) < 50:
        population.append(random_individual())
    population = [repair_h(ind) for ind in population]

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    while True:
        current_time = time.time()
        if current_time - start_time > budget_s - 0.1:
            break

        c5_list = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            c5_list.append(c5)

        current_best_idx = np.argmin(c5_list)
        if c5_list[current_best_idx] < best_c5:
            best_c5 = c5_list[current_best_idx]
            best_h = population[current_best_idx].copy()

        sorted_indices = np.argsort(c5_list)
        sorted_population = [population[i] for i in sorted_indices]

        tournament_size = 3
        num_parents = 10  # top 20% parents
        num_children = 10  # replace 20% of population
        parents = []
        for _ in range(num_parents):
            contestants = np.random.choice(len(population), tournament_size, replace=False)
            contestant_c5 = [c5_list[i] for i in contestants]
            winner_idx = contestants[np.argmin(contestant_c5)]
            parents.append(population[winner_idx])

        children = []
        for _ in range(num_children):
            p1, p2 = parents[np.random.choice(len(parents))], parents[np.random.choice(len(parents))]
            alpha = 0.3 + 0.4 * np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
            if np.random.rand() < 0.1:
                child += np.random.normal(0, 0.05, n_points)
            child = repair_h(child)
            children.append(child)

        best_individual = sorted_population[0]
        population = [best_individual]
        population.extend(sorted_population[1 : len(sorted_population) - num_children])
        population.extend(children)

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
