import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = tuple((0.0, 1.0) for _ in range(n_points))
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum},)

    start_time = time.time()
    options = {
        'maxiter': 500,
        'ftol': 1e-8,
        'disp': False
    }

    result = minimize(
        objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    optimized_h = result.x.copy()
    if not np.all(0 <= optimized_h) or not np.all(optimized_h <= 1):
        optimized_h = np.clip(optimized_h, 0.0, 1.0)
    current_sum = optimized_h.sum()
    if not np.isclose(current_sum, target_sum, rtol=1e-6):
        scale = target_sum / current_sum
        optimized_h = np.clip(optimized_h * scale, 0.0, 1.0)

    corr_values = np.correlate(optimized_h, 1 - optimized_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return optimized_h, c5_bound, n_points
