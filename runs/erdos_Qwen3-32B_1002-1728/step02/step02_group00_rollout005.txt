    best_c5 = compute_c5(h)
    best_h = h.copy()

    T = 1.0  # Initial temperature
    cooling_rate = 0.995

    while time.time() - start_time < budget_s:
        indices_1 = np.where(h == 1)[0]
        indices_0 = np.where(h == 0)[0]
        if len(indices_1) == 0 or len(indices_0) == 0:
            break
        i = np.random.choice(indices_1)
        j = np.random.choice(indices_0)

        h_candidate = h.copy()
        h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]
        c5_candidate = compute_c5(h_candidate)
        delta = c5_candidate - best_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate.copy()
            best_c5 = min(best_c5, c5_candidate)
            best_h = h.copy()

        T *= cooling_rate

        # Termination check
        if time.time() - start_time > budget_s * 0.95:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-8, "Constraint violation"
    assert np.all((best_h == 0) | (best_h == 1)), "Invalid values"
    return (best_h.tolist(), best_c5, K)
