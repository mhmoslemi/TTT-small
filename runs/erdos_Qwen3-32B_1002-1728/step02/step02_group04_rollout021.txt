def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    current_sum = np.sum(h)
    if current_sum > 1e-10:
        scaling = required_sum / current_sum
        h = h * scaling
        h = np.clip(h, 0, 1)
