    step_size_initial = 0.2
    perturb_points = 5

    T = initial_T

    while time.time() - start_time < budget_s - 0.1:
        current_step_size = step_size_initial * (T / initial_T)

        indices = np.random.choice(n_points, perturb_points, replace=False)
        perturbation = np.zeros(n_points)
        for idx in indices:
            perturbation[idx] = np.random.uniform(-current_step_size, current_step_size)
        h_proposed = h_current + perturbation
        h_proposed = _project(h_proposed, target_sum, n_points)

        new_score = compute_score(h_proposed, dx)

        if new_score < current_score:
            accept = True
        else:
            delta = new_score - current_score
            prob = np.exp(-delta / T)
            accept = np.random.rand() < prob

        if accept:
            h_current = h_proposed.copy()
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best_h = h_current.copy()

        T *= alpha

    corr_values = correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr_values), n_points
