        else:
            curr_h = x0
            curr_c5 = objective(curr_h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h[:] = curr_h[:]

    # Final validation
    best_h = adjust_sum(best_h)
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
