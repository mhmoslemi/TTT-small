                             phi_c * r1 * (personal_best[i] - positions[i]) +
                             phi_s * r2 * (global_best - positions[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            positions[i] += velocities[i]
            
            # Project to feasible region
            positions[i] = np.clip(positions[i], 0, 1)
            pos_sum = np.sum(positions[i])
            if pos_sum != 0:
                positions[i] = positions[i] * (target_sum / pos_sum)
            positions[i] = np.clip(positions[i], 0, 1)

        # Evaluate and update bests
        new_scores = np.array([evaluate(p) for p in positions])
        for i in range(n_particles):
            if new_scores[i] < personal_best_scores[i]:
                personal_best_scores[i] = new_scores[i]
                personal_best[i] = positions[i].copy()
                if new_scores[i] < best_c5:
                    global_best = positions[i].copy()
                    best_c5 = new_scores[i]

    # Local gradient descent refinement
    current = global_best.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        has_improved = False
        for i in range(n_points):
            for delta in [0.01, -0.01]:
                trial = current.copy()
                trial[i] += delta
                trial = np.clip(trial, 0, 1)
                pos_sum = np.sum(trial)
                if pos_sum != 0:
                    trial = trial * (target_sum / pos_sum)
                trial = np.clip(trial, 0, 1)
                score = evaluate(trial)
                if score < best_c5:
                    current = trial
                    best_c5 = score
                    has_improved = True
                    break
        if not has_improved:
            break

    final_corr = np.correlate(current, 1 - current, 'full') * dx
    final_c5 = np.max(final_corr)
    return current, final_c5, n_points
