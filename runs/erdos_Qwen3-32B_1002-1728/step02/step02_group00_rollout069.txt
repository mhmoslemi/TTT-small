            min_dists = [min(abs(p - x) for x in positions) for p in available]
            best_idx = np.argmax(min_dists)
            best_pos = available.pop(best_idx)
            positions.append(best_pos)
        h = np.zeros(n)
        for pos in positions:
            h[pos] = 1
        return h

    h_greedy = generate_greedy_h(n_points)
    candidates = [h_init, h_greedy]
    best_h = h_init.copy()
    best_c5 = np.inf

    for h_start in candidates:
        h = h_start.copy()
        bounds = [(0.0, 1.0)] * n_points
        constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        
        def objective(x):
            h = x.reshape(n_points)
            corr = np.correlate(h, 1-h, 'full')
            return np.max(corr) * dx
        
        res = minimize(
            objective, h, method='SLSQP', bounds=bounds,
            constraints=constraints, options={'maxiter': 150, 'ftol': 1e-8}
        )
        if res.success:
            curr_h = res.x.reshape(n_points)
            if np.all((curr_h >= -1e-8) & (curr_h <= 1.0001)):
                corr = np.correlate(curr_h, 1 - curr_h, 'full')
                val = np.max(corr) * dx
                if val < best_c5:
                    best_c5 = val
                    best_h = curr_h

        while time.time() - start_time < budget_s * 0.9:  # Leave 10% for final checks
            rs = np.random.RandomState(seed + int(time.time()))
            perturb = rs.normal(0.0, 0.05, n_points)
            perturbed = np.clip(h + perturb, 0, 1)
            perturbed *= target_sum / np.sum(perturbed)
            res_perturb = minimize(
                objective, perturbed, method='SLSQP', bounds=bounds,
                constraints=constraints, options={'maxiter': 100}
            )
            if res_perturb.success:
                h_new = res_perturb.x.reshape(n_points)
                if np.all((h_new >= -1e-8) & (h_new <= 1.0001)) and abs(np.sum(h_new) - target_sum) < 1e-6:
                    corr_new = np.correlate(h_new, 1 - h_new, 'full')
                    val_new = np.max(corr_new) * dx
                    if val_new < best_c5:
                        best_c5, best_h = val_new, h_new

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    return (best_h.tolist(), best_c5, n_points)
