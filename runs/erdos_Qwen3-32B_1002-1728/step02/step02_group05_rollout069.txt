                pbest_scores[i] = score
                pbest_positions[i] = particles[i].copy()

        current_best_idx = np.argmin(pbest_scores)
        if pbest_scores[current_best_idx] < best_c5:
            best_c5 = pbest_scores[current_best_idx]
            best_h = pbest_positions[current_best_idx].copy()
            gbest_idx = current_best_idx
            gbest_position = pbest_positions[current_best_idx].copy()
            gbest_score = pbest_scores[current_best_idx]

        if iteration % 100 == 0 and iteration > 200:
            no_improvement = abs(best_c5 - pbest_scores[gbest_idx]) < 1e-6
            if no_improvement:
                for _ in range(n_particles // 5):
                    idx = np.random.randint(n_particles)
                    h = np.random.rand(n_points)
                    h = adjust_sum(h)
                    particles[idx] = h.copy()
                    velocities[idx] = np.random.rand(n_points) * 0.1 - 0.05
                    pbest_positions[idx] = h.copy()
                    pbest_scores[idx] = objective(h)

    best_h = adjust_sum(best_h)
    c5_final = objective(best_h)
    return best_h, c5_final, n_points
