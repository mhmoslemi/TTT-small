    return np.max(corr) * global_dx

def project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    return np.clip(clamped * (target_sum / clamped_sum), 0, 1)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Set global configuration for constraint/objective
    global global_target_sum, global_dx, global_n_points
    global_target_sum = target_sum
    global_dx = dx
    global_n_points = n_points

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    options = {'maxiter': 10000, 'ftol': 1e-8, 'disp': False}

    start_time = time.time()
    result = minimize(
        objective_function, 
        h_initial, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        options=options
    )

    h_opt = project(
        result.x if result.success else h_initial.copy(), 
        target_sum, 
        n_points
    )

    # Compute final solution within budget
    corr_values = correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr_values)
    return h_opt, c5_bound, n_points
