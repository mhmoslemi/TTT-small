import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    current_h = h_initial.copy()
    best_h = current_h.copy()
    
    def compute_c5(h):
        corr = correlate(h, 1.0 - h, mode='full') * dx
        return np.max(corr)

    def perturb(h, n_points):
        new_h = h.copy()
        i, j = np.random.choice(n_points, size=2, replace=False)
        # Move value from i to j or vice versa
        max_move = min(new_h[i], 1.0 - new_h[j])
        if max_move > 1e-10 and np.random.rand() < 0.5:
            delta = np.random.uniform(0, max_move)
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_move_rev = min(new_h[j], 1.0 - new_h[i])
            if max_move_rev > 1e-10:
                delta = np.random.uniform(0, max_move_rev)
                new_h[i] += delta
                new_h[j] -= delta
        return new_h

    # Initial evaluation
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # SA parameters
    temp = 1.0
    alpha = 0.9995

                best_c5 = current_c5
