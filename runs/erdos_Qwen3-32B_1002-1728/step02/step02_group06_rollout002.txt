import numpy as np
import time

def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-8:
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    T = 1.0
    alpha = 0.999
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        noise = np.random.normal(0, 0.1 * T, size=n_points)
        new_h = current_h + noise
        new_h = repair_h(new_h, target_sum)
        new_c5 = compute_c5(new_h, dx)

        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            current_h[:] = new_h
            current_c5 = new_c5

        if current_c5 < best_c5:
            best_h[:] = current_h
            best_c5 = current_c5

        T *= alpha

    corr_best = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_best)
    return best_h, c5_bound, n_points
