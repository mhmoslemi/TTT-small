    h0 = _project_sum(h0, target_sum, n_points)

    def objective(h):
        h_vec = h.reshape(-1)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    start_time = time.time()
    best_result = h0.copy()

    def callback(xk, state=None):
        nonlocal best_result
        best_result = xk.copy()
        if time.time() - start_time >= budget_s - 0.1:
            return True
        return False

    minimize(
        objective,
        x0=h0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        callback=callback,
        options={'maxiter': 3000, 'ftol': 1e-6, 'disp': False}
    )

    h_opt = _project_sum(best_result, target_sum, n_points)
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx

    return h_opt, c5_bound, n_points
