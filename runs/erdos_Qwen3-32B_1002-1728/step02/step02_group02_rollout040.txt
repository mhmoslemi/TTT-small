import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def sum_constraint(h_vec):
        return h_vec.sum() - target_sum

    # Setup optimization
    cons = {'type': 'eq', 'fun': sum_constraint}
    bounds = [(0, 1)] * n_points

    # Configure SLSQP with safe iteration cap
    maxiter = int(min(10000, budget_s * 10))  # Conservative iteration estimate
    options = {
        'maxiter': maxiter,
        'ftol': 1e-8,
        'disp': False,
        'iprint': 0
    }

    start_time = time.time()
    result = minimize(
        objective, 
        h_initial, 
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options=options
    )

    # Postprocessing to ensure strict constraint satisfaction
    optimized_h = np.clip(result.x, 0, 1)
    h_sum = optimized_h.sum()
    if not np.isclose(h_sum, target_sum, atol=1e-6):
        scaling = target_sum / h_sum
        optimized_h = np.clip(optimized_h * scaling, 0, 1)

    # Final evaluation
    corr_values = correlate(optimized_h, 1 - optimized_h, mode='full') * dx
    final_c5 = np.max(corr_values)

    return optimized_h, final_c5, n_points
