import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def optimize_for_kw(h_current, n_points, shift_k, dx):

        h = h_vec.reshape(-1)
        max_i = n_points - shift_k
        sum_hh = 0.0
        for i in range(max_i):
            sum_hh += h[i] * h[i + shift_k]
        return -sum_hh  # Maximize sum_hh by minimizing -sum_hh

    h_flat = h_current.copy()

        options={'maxiter': 100, 'disp': False}
    )
    if result.success:
        optimized_h = result.x.reshape(-1)
