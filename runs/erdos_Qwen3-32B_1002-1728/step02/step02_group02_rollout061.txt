import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start_time = time.time()
    max_iter = 1000

    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute current max overlap and corresponding lag
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        k_lag = max_corr_idx - (n_points - 1)

        # Compute gradient for the worst lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_lag
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] += -h[i] * dx

        # Formulate convex optimization problem
        h_new = cp.Variable(n_points)
        constraints = [0 <= h_new, h_new <= 1, cp.sum(h_new) == target_sum]
        current_value = corr[max_corr_idx] * dx
        obj = cp.Minimize(current_value + grad @ (h_new - h))
        problem = cp.Problem(obj, constraints)

        # Solve the convex relaxation
        problem.solve(solver=cp.ECOS, verbose=False, max_iters=100)
        if problem.status != cp.OPTIMAL:
            continue

        # Update h with the solution
        new_h_vals = h_new.value
        if new_h_vals is None:
            continue
        new_h_vals = np.clip(new_h_vals, 0, 1)
        h[:] = new_h_vals

        # Check for improvement
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
