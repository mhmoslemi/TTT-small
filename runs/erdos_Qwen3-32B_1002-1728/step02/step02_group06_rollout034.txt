def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern
