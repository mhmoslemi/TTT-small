import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Objective function: max correlation scaled by dx
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)  # Ensure bounds
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: sum(h) == target_sum
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(0.0, 1.0)

    # Optimization options
    options = {
        'maxiter': 10000,
        'ftol': 1e-8,
        'disp': False,
        'eps': 1e-6
    }

    # Time tracking
    start_time = time.time()

    # Run optimization with time guard
    result = minimize(
        objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    # Enforce hard constraints after optimization
    h_opt = np.clip(result.x, 0, 1)
    h_sum = np.sum(h_opt)
    if not np.isclose(h_sum, target_sum, atol=1e-6):
        scale = target_sum / h_sum
        h_opt *= scale
        h_opt = np.clip(h_opt, 0, 1)  # Final safeguard

    # Final evaluation of C5
    corr_values = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr_values)

    # Ensure time budget not exceeded
    remaining_time = max(budget_s - (time.time() - start_time), 0.1)
    time.sleep(remaining_time)

    return h_opt, c5_bound, n_points
