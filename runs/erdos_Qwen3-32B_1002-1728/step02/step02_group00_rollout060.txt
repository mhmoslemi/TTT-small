    n_points_global = n_points

    num_workers = 5
    initial_guesses = [h_init.copy()]
    rs = np.random.RandomState(seed)
    for _ in range(num_workers - 1):
        pert = rs.normal(0, 0.05, n_points)
        perturbed = h_init + pert
        perturbed = np.clip(perturbed, 0, 1)
        current_sum_pert = np.sum(perturbed)
        if current_sum_pert > 1e-6:
            perturbed = perturbed * (target_sum / current_sum_pert)
        initial_guesses.append(perturbed)
    
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    for i, guess in enumerate(initial_guesses):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = minimize(
            fun=objective_with_penalty,
            x0=guess,
            method='COBYLA',
            bounds=bounds,
            options={'maxiter': 200, 'tol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points).copy()
            current_sum_h = np.sum(h_candidate)
            if abs(current_sum_h - target_sum) > 1e-5:
                factor = target_sum / current_sum_h
                h_candidate = np.clip(h_candidate * factor, 0, 1)
            h_candidate = np.clip(h_candidate, 0, 1)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.95:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
