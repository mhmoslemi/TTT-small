import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def _project(p):
        clamped = np.clip(p, 0, 1)
        clamped_sum = clamped.sum()
        if clamped_sum == 0:
            return np.ones_like(clamped) * 0.5
        return np.clip(clamped * (target_sum / clamped_sum), 0, 1)

    def crossover(p1, p2):
        n = len(p1)
        split_points = np.sort(np.random.choice(n, size=5, replace=False))
        split_points = np.insert(split_points, 0, 0)
        split_points = np.append(split_points, n)
        child1 = np.zeros(n)
        child2 = np.zeros(n)
        for i in range(len(split_points) - 1):
            start, end = split_points[i], split_points[i+1]
            if i % 2 == 0:
                child1[start:end] = p1[start:end]
                child2[start:end] = p2[start:end]
            else:
                child1[start:end] = p2[start:end]
                child2[start:end] = p1[start:end]
        return child1, child2

    def mutate(individual):
        mutation = np.random.rand()
        n = len(individual)
        if mutation < 0.1:  # Jitter
            jitter = 0.05 * np.random.randn(n)
            return _project(np.clip(individual + jitter, 0, 1))
        elif mutation < 0.3:  # Invert
            a, b = np.sort(np.random.randint(0, n, 2))
            individual[a:b] = 1 - individual[a:b]
            return _project(individual)
        else:  # Slide
            a, b = np.sort(np.random.randint(0, n, 2))
            delta = np.random.uniform(-0.1, 0.1, size=b - a)
            individual[a:b] = np.clip(individual[a:b] + delta, 0, 1)
            return _project(individual)

    population = [h_initial.copy()]
    for _ in range(49):  # 50 members
        h = np.random.rand(n_points)
        population.append(_project(h))
    fitness = np.array([compute_c5(h) for h in population])
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = np.min(fitness)

    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        # Tournament selection
        parents = []
        for _ in range(2):
            t = np.random.choice(len(population), 5, replace=False)
            winner = t[np.argmin(fitness[t])]
            parents.append(population[winner])
        # Crossover
        child1, child2 = crossover(*parents)
        # Mutation
        child1, child2 = mutate(child1), mutate(child2)
        # Projection
        child1, child2 = _project(child1), _project(child2)
        # Evaluate
        c1, c2 = compute_c5(child1), compute_c5(child2)
        # Replace worst
        worst = np.argsort(fitness)[-2:]
        for i, idx in enumerate(worst):
            population[idx] = child1 if i == 0 else child2
            fitness[idx] = c1 if i == 0 else c2
        # Best update
        current_best = np.argmin(fitness)
        if fitness[current_best] < best_c5:
            best_h, best_c5 = population[current_best].copy(), fitness[current_best]

    return best_h, best_c5, n_points
