        next_sum = np.sum(next_initial)
        if next_sum < 1e-6 or next_sum > n_points:
            # Skip invalid
            continue
        next_initial = next_initial * (target_sum / next_sum)
        result = minimize(
            fun=objective,
            x0=next_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                    # also check binary version
                    h_bin = binary_h(h_candidate, target_sum)
                    result_bin = minimize(
                        fun=objective,
                        x0=h_bin,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 100, 'ftol': 1e-8}
                    )
                    if result_bin.success:
                        corr_bin = np.correlate(result_bin.x, 1 - result_bin.x, mode='full')
                        c5_bin = np.max(corr_bin) * dx
                        if c5_bin < best_c5:
                            best_h = result_bin.x
                            best_c5 = c5_bin
        if time.time() - start_time > budget_s * 0.95:
            break

    # Final binary check for best h
    h_binary_save = binary_h(best_h, target_sum)
    result_binary = minimize(
        fun=objective,
        x0=h_binary_save,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result_binary.success:
        h_final_binary = result_binary.x
        corr_bin = np.correlate(h_final_binary, 1 - h_final_binary, mode='full')
        c5_bin = np.max(corr_bin) * dx
        if c5_bin < best_c5:
            best_h = h_final_binary
            best_c5 = c5_bin

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)

By incorporating these structured initializations and the binary checks, the code should have a better chance of finding a lower C5.

import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Generate structured initial guesses
    even = np.zeros(n_points)
    even[np.arange(0, n_points, 2)] = 1.0
    odd = np.zeros(n_points)
    odd[np.arange(1, n_points, 2)] = 1.0
    checkerboard = np.kron([1, 0], np.ones(n_points//2))
    checkerboard = checkerboard[:n_points]
    rs = np.random.RandomState(seed)
    indices = rs.permutation(n_points)
    random1 = np.zeros(n_points)
    random1[indices[:n_points//2]] = 1.0
    additional_initials = [even, odd, checkerboard, random1]

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Binary thresholding helper
    def binary_h(h_vec, target_sum_bin):
        n_points = len(h_vec)
        h_sorted = np.argsort(h_vec)
        n_ones = int(target_sum_bin)
        h_binary = np.zeros(n_points)
        h_binary[h_sorted[-n_ones:]] = 1.0
        return h_binary

    result_init = minimize(
        fun=objective,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    best_h = result_init.x.reshape(n_points)
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    # Binarize initial solution
    h_binary = binary_h(best_h, target_sum)
    result_bin = minimize(
        fun=objective,
        x0=h_binary,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result_bin.success:
        h_candidate = result_bin.x.reshape(n_points)
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate

    all_initials = additional_initials.copy()
    while time.time() - start_time < budget_s * 0.90:
        if all_initials:
            next_initial = all_initials.pop(0)
        else:
            perturbed = best_h + rs.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if abs(perturbed_sum - target_sum) > 1e-6:
                perturbed = (perturbed / perturbed_sum) * target_sum
            next_initial = perturbed

        result = minimize(
            fun=objective,
            x0=next_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr_candidate) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                h_bin = binary_h(h_candidate, target_sum)
                result_binary = minimize(
                    fun=objective,
                    x0=h_bin,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 80, 'ftol': 1e-8}
                )
                if result_binary.success:
                    corr_binary = np.correlate(result_binary.x, 1 - result_binary.x, mode='full')
                    c5_binary = np.max(corr_binary) * dx

                        best_h = result_binary.x
                        best_c5 = c5_binary

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
