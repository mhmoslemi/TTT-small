import numpy as np
import time
from numpy import correlate

def perturb(h):
    h = h.copy()
    for _ in range(3):  # Apply 3 perturbations
        n_points = len(h)
        i, j = np.random.choice(n_points, size=2, replace=False)
        direction = np.random.choice([0, 1])
        if direction == 0:
            delta_max = min(h[i], 1 - h[j])
            if delta_max <= 0:
                continue
            delta = delta_max * abs(np.random.randn()) * 0.15
            delta = np.clip(delta, 0, delta_max)
            h[i] -= delta
            h[j] += delta
        else:
            delta_max = min(h[j], 1 - h[i])
            if delta_max <= 0:
                continue
            delta = delta_max * abs(np.random.randn()) * 0.15
            delta = np.clip(delta, 0, delta_max)
            h[j] -= delta
            h[i] += delta
    return h

def compute_c5(h, dx):
    kernel = 1 - h
    corr = correlate(h, kernel, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial_h_values is valid
    assert np.isclose(h_initial.sum(), target_sum), "Initial h sum invalid"

    best_h = h_initial.copy()
    best_score = compute_c5(best_h, dx)
    current_h = h_initial.copy()
    current_score = best_score

    T_initial = 1.5  # Higher initial temperature for better exploration
    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        candidate_h = perturb(current_h)
        candidate_score = compute_c5(candidate_h, dx)
        delta = candidate_score - current_score

        time_elapsed = time.time() - start_time
        T = max(1e-6, T_initial * (1.0 - (time_elapsed / budget_s) ** 0.5))  # Nonlinear cooling schedule

        accept_prob = np.exp(-delta / T) if delta > 0 else 1.0
        if np.random.rand() < accept_prob:
            current_h = candidate_h
            current_score = candidate_score
            if current_score < best_score:
                best_h = current_h.copy()
                best_score = current_score

        if time.time() - start_time >= budget_s - 0.1:
            break

    return best_h, best_score, n_points
