import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        return np.max(correlate(h, 1 - h, mode='full')) * dx

    current_h = h_initial.copy()
    best_h = current_h.copy()
    current_energy = compute_energy(current_h)
    best_energy = current_energy

    T = 1.0
    T_final = 1e-3
    alpha = 0.9999

    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        candidate_h = current_h.copy()
        num_changes = np.random.randint(2, 6)
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        deltas = np.random.uniform(-0.2, 0.2, size=num_changes)
        candidate_h[indices] += deltas
        candidate_h = _project(candidate_h, target_sum, n_points)
        
        candidate_energy = compute_energy(candidate_h)
        delta = candidate_energy - current_energy

        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            current_h = candidate_h.copy()
            current_energy = candidate_energy
            if candidate_energy < best_energy:
                best_h = current_h.copy()
                best_energy = candidate_energy

        if T > T_final:
            T *= alpha

    return best_h, best_energy, n_points

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum <= 1e-8:
        return np.ones_like(clamped) * 0.5
    temp = clamped * (target_sum / clamped_sum)
    return np.clip(temp, 0, 1)
