            t_candidate -= 1e-6
            h_binary = (h_continuous >= t_candidate).astype(float)

        h_continuous = h_binary  # Use binary solution for next iteration

        if time.time() - start_time > budget_s * 0.95:
            break

    # Final evaluation
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), final_c5, n_points)
