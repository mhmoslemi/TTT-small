import numpy as np
from scipy.optimize import minimize
from scipy.ndimage import gaussian_filter1d
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    kernel_sigma = 1.2  # Smooth perturbations with adaptive kernel
    noise_std = 0.03    # Optimized noise scale
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:  # Safety buffer
        n_attempt += 1
        # Generate smoothed perturbation
        smoothed_best = gaussian_filter1d(best_h, sigma=kernel_sigma, mode='wrap')
        # Perturb with noise plus deterministic shift
        perturbed = smoothed_best + np.random.normal(0, noise_std, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Enforce constraint with dynamic scaling
        total = np.sum(perturbed)
        if total > 1e-6:
            perturbed = perturbed * (target_sum / total)
        perturbed = np.clip(perturbed, 0, 1)

        # Advanced optimizer settings with tighter coupling constraint
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-8, 'disp': False}
        )

        if result.status == 0:  # Check for proper convergence
            new_h = result.x.reshape(n_points)
            new_c5 = objective(new_h)
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    # Final post-processing constraints check
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
