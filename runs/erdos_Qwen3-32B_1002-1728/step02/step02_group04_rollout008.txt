def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def get_gradient(h, k, n_points):
    gradient = np.ones_like(h)
    if k < 0:
        # Left shift by |k|
        shift = -k
        # For left shift, the gradient is 1 - h[i + shift], but only if i + shift < n_points
        # So, for i in [0, n_points - shift), gradient[i] = 1 - h[i + shift]
        # For i >= n_points - shift, gradient[i] = 1
        if shift <= n_points:
            gradient[:n_points - shift] = 1.0 - h[shift:]
    elif k > 0:
        # Right shift by k
        if k <= n_points:
            gradient[k:] = 1.0 - h[:n_points - k]
    return gradient

def solve_linear_subproblem(gradient, n_points, required_sum):
    sorted_indices = np.argsort(gradient)
    s = np.zeros_like(gradient)
    remaining = required_sum
    for idx in sorted_indices:
        if remaining <= 0:
            break
        if remaining >= 1:
            s[idx] = 1.0
            remaining -= 1.0
        else:
            s[idx] = remaining
            remaining = 0
    # In case due to floating point issues, adjust
    s[sorted_indices[:int(required_sum)]] = 1.0
    if required_sum - int(required_sum) > 1e-10:
        s[sorted_indices[int(required_sum)]] = required_sum - int(required_sum)
    return s
