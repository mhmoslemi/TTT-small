import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    alpha = 0.01  # Step size

    while time.time() - start_time < budget_s - 1:
        # Compute overlaps and identify worst shift
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        relevant = overlaps[n_points:]
        if len(relevant) == 0:
            break  # Avoid empty array case
        max_index = np.argmax(relevant)
        k_shift = max_index + 1  # Corresponding sample shift

        # Compute gradient for k_shift
        shifted_neg = np.roll(h, -k_shift)
        shifted_pos = np.roll(h, k_shift)
        grad = dx * ((1.0 - shifted_neg) - shifted_pos)

        # Update step
        h = h - alpha * grad

        # Clip to bounds
        h = np.clip(h, 0, 1)

        # Project to feasible sum
        s = np.sum(h)
        if abs(s) > 1e-10:
            scale = target_sum / s
            h = h * scale
            h = np.clip(h, 0, 1)  # Re-clip after scaling

        # Track best solution
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Adaptive step size (optional safeguard)
        if current_c5 > best_c5:
            alpha *= 0.5  # Halve step size if no improvement

    return best_h, best_c5, n_points
