import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Generate pairs (i, i + k) for all valid k and i
    pairs = []
    for k in range(1, n_points):
        for i in range(n_points - k):
            pairs.append((i, i + k))
    num_pairs = len(pairs)

    h = cp.Variable(n_points)
    m = cp.Variable(num_pairs)
    constraints = []

    # McCormick constraints for each product term m[i] >= h[i1] * h[i2]
    for idx, (i1, i2) in enumerate(pairs):
        constraints.extend([
            m[idx] >= 0,
            m[idx] >= h[i1] + h[i2] - 1,
            m[idx] <= h[i1],
            m[idx] <= h[i2]
        ])

    # Group m by lag k
    lag_to_indices = {}
    for idx, (i1, i2) in enumerate(pairs):
        k = i2 - i1
        if k not in lag_to_indices:
            lag_to_indices[k] = []
        lag_to_indices[k].append(idx)

    # Define objective: minimize t where t >= (n_points/2 - sum(m for pairs of lag k)) for all k
    t = cp.Variable()
    for k in lag_to_indices:
        indices = lag_to_indices[k]
        sum_m = cp.sum(m[indices])
        constraints.append(t >= (n_points / 2) - sum_m)

    # Add constraints for h
    constraints.extend([
        cp.sum(h) == n_points / 2,
        h <= 1,
        h >= 0
    ])

    objective = cp.Minimize(t)
    problem = cp.Problem(objective, constraints)

    start_time = time.time()
