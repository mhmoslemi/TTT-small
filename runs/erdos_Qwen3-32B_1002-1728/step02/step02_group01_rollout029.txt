import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Validate initial state
    h[...] = np.clip(h, 0, 1)
    total = h.sum()
    if total > 1e-9:
        h *= target_sum / total
        h[...] = np.clip(h, 0, 1)

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        return max(np.correlate(h, 1 - h, mode='full')) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Annealing parameters
    temp = 0.15  # Initial temperature
    cool_rate = 0.99
    steps_per_temp = 40

    rng = np.random.Generator(np.random.PCG64DXSM(seed))
    while time.time() - start_time < budget_s - 30:
        for _ in range(steps_per_temp):
            # Apply temperature-dependent perturbation
            noise = rng.normal(0, temp * 0.2, n_points)
            new_h = best_h + noise

            # Enforce bounds and integral constraint
            new_h = np.clip(new_h, 0, 1)
            total = new_h.sum()
            if 1e-9 < total < n_points - 1e-9:
                scale = target_sum / total
                new_h *= scale
                new_h = np.clip(new_h, 0, 1)

            # Metropolis acceptance
            current_c5 = compute_c5(new_h)
            if current_c5 < best_c5 or np.exp((best_c5 - current_c5) / max(temp, 1e-9)) > rng.random():
                best_h[...] = new_h
                best_c5 = current_c5

        temp *= cool_rate
        if temp < 1e-6:
            break

    # Final constraint check
    if abs(best_h.sum() - target_sum) > 1e-6:
        best_h *= target_sum / best_h.sum()
    best_h = np.clip(best_h, 0, 1)
    
    return (best_h.tolist(), best_c5, n_points)
