import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    pop_size = 25
    mutation_strength = 0.05
    elitism_rate = 0.2
    
    # Initialize population
    population = [h_initial.copy()]
    while len(population) < pop_size:
        base = h_initial.copy() if np.random.rand() < 0.5 else population[np.random.randint(len(population))].copy()
        perturb = np.random.normal(0, mutation_strength, size=n_points)
        candidate = np.clip(base + perturb, 0, 1)
        total = np.sum(candidate)
        if total != 0:
            candidate = (candidate / total) * target_sum
        candidate = np.clip(candidate, 0, 1)
        population.append(candidate)
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    def evaluate(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Initial evaluation
    for h in population:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
    
    while time.time() - start_time < budget_s - 5:
        # Evaluate population
        evaluated = []
        for h in population:
            c5 = evaluate(h)
            evaluated.append((h.copy(), c5))
            if c5 < best_c5:
                best_c5 = c5
                best_h[:] = h.copy()
        
        # Select elites
        evaluated.sort(key=lambda x: x[1])
        n_elites = max(1, int(pop_size * elitism_rate))
        elites = [x[0] for x in evaluated[:n_elites]]
        
        new_population = elites.copy()
        parents = elites
        
        while len(new_population) < pop_size:
            # Simple selection and crossover
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            cx_point = np.random.rand(n_points)
            child = np.where(cx_point > 0.5, p1, p2)
            
            # Mutation with normalization
            mut = np.random.normal(0, mutation_strength, n_points)
            child_py = np.clip(child + mut, 0, 1)
            total = np.sum(child_py)
            if total != 0:
                child_py = (child_py / total) * target_sum

            break
    
    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
