import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)

    n_points = len(h_initial)

    target_sum = n_points / 2.0

    def repair_h(p):
        clamped = np.clip(p, 0, 1)
        s = clamped.sum()
        if s == 0:
            return np.ones_like(clamped) * 0.5
        return np.clip(clamped * (target_sum / s), 0, 1)

    population_size = 50
    population = []
    
    # 10% from initial with small perturbation
    for _ in range(5):
        p = np.random.normal(0, 0.05, size=n_points)
        population.append(repair_h(h_initial + p))
    
    # 20% totally random
    for _ in range(10):
        p = np.random.rand(n_points)
        population.append(repair_h(p))
    
    # 70% from gradient-based mutations
    for _ in range(35):
        p = h_initial.copy()
        idx = np.random.choice(n_points, size=np.random.randint(5,15), replace=False)
        p[idx] += np.random.normal(0, 0.1, size=len(idx))
        population.append(repair_h(p))
    
    # Evaluate initial population
    scores = [np.max(correlate(h, 1-h, 'full')) * dx for h in population]
    best_idx = np.argmin(scores)

    start_time = time.time()
    stagnation_count = 0
    target_stagnation = 5
    
    while time.time() - start_time < budget_s - 0.1:
        # Tournament selection
        tournament_size = 5
        parents = []

           选手 = np.random.choice(population_size, tournament_size)
            contest = [scores[i] for i in 选手]

            选手2 = 选手.copy()
            选手2[np.argmin(contest)] = -1
            winner2 = 选手2[np.argmin([scores[i] for i in 选手2 if i != -1])]
            parents.extend([population[winner1].copy(), population[winner2].copy()])
        
        # Crossover
        offspring = []
        for i in range(0, 10, 2):  # 5-point recombination
            p1, p2 = parents[i], parents[i+1]
            if np.random.random() < 0.8:
                cx_points = np.sort(np.random.choice(n_points-1, 4, replace=False)+1)
                segments = np.split(np.concatenate(([0], cx_points, [n_points])), 5)
                child = np.zeros_like(p1)
                for j, (start, end) in enumerate(segments):
                    if j % 2 == 0:
                        child[start:end] = p1[start:end]

                        child[start:end] = p2[start:end]
                offspring.append(repair_h(child))
