def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = 32.0

    final_temp = 0.001
    alpha = 0.995  # cooling rate
    num_iterations = 1000

    temp = initial_temp

                break
