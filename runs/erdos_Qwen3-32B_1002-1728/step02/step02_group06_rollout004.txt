import time

def generate_random_binary(n, target_sum):
    h = np.zeros(n)
    indices = np.random.permutation(n)
    count = int(np.floor(target_sum))
    h[indices[:count]] = 1.0
    if target_sum - count > 1e-8:
        h[indices[count]] = target_sum - count
    return h

def compute_gradient(h, s, n):
    grad = np.zeros(n)
    for i in range(n):
        j = i + s
        if 0 <= j < n:
            grad[i] += (1 - h[j])
            grad[j] -= h[i]
    return grad
