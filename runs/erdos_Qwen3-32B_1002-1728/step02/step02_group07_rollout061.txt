import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = h_initial.copy()
    current_c5 = objective(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    T_initial = current_c5 * 1.8  # Adjusted initial temperature scale
    start_time = time.time()

    rng = np.random.Generator(np.random.PCG64(seed))
    max_iterations = 100000  # Fallback in case time tracking fails
    iteration = 0

    while time.time() - start_time < budget_s - 20 and iteration < max_iterations:
        iteration += 1
        noise = rng.normal(0, 1, n_points)

        # Compute temperature based on elapsed time
        elapsed = time.time() - start_time
        if elapsed < 1e-9:
            T = T_initial
        else:
            T = T_initial / np.log(1 + elapsed)

        # Perturbation with temperature-aware mutation strength
        mutation_strength = 0.15 * T / T_initial
        candidate_h = current_h + mutation_strength * noise
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_h = adjust_sum(candidate_h, target_sum)

        candidate_c5 = objective(candidate_h)

        if delta <= 0 or rng.rand() < np.exp(-delta / T):

                best_h = current_h.copy()

    best_h = adjust_sum(best_h, target_sum)
    final_c5 = objective(best_h)
