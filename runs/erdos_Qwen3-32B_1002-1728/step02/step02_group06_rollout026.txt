            coords = np.random.permutation(n_points)
            h_prev = h_current.copy()
            for j in coords:
                orig = h_prev[j]
                candidates = [orig - 0.05, orig, orig + 0.05]
                best_candidate = h_prev.copy()
                best_c5_candidate = float('inf')
                for val in candidates:
                    h_test = h_prev.copy()
                    h_test[j] = val
                    h_test = np.clip(h_test, 0, 1)
                    h_test[0] += target_sum - np.sum(h_test)
                    h_test = np.clip(h_test, 0, 1)
                    corr = np.correlate(h_test, 1 - h_test, mode='full') * dx
                    c5 = np.max(corr)
                    if c5 < best_c5_candidate:
                        best_c5_candidate, best_candidate = c5, h_test.copy()
                if best_c5_candidate < float('inf'):
                    h_current = best_candidate.copy()
                    corr_current = np.correlate(h_current, 1 - h_current, mode='full') * dx
                    c5_current = np.max(corr_current)
                    if c5_current < best_c5:
                        best_c5 = c5_current
                        best_h = h_current.copy()
                    improved = True
            if not improved:
                break
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
