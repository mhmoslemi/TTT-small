    D = target_sum - current_sum

        can_incr = h_adj < 0.9999
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = h_adj > 1e-4
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def objective(h, n_points, dx, target_sum):
    h = h.reshape(-1).copy()
    h[-1] = 2 * target_sum * dx / dx - np.sum(h[:-1])
    h_clipped = np.clip(h, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
    return float(np.max(corr))

def perturbation(x):
    x += 0.05 * np.random.randn(x.size)
    return np.clip(x, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h_initial.copy()
    best_c5 = float(np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx))

    np.random.seed(seed)
    start_time = time.time()

    def wrapped_objective(x):
        return objective(x, n_points, dx, target_sum)

    while time.time() - start_time < budget_s:
        # Basin hopping step to explore the space
        bh_res = basinhopping(wrapped_objective, best_h, niter=1, T=0.5, 
                              stepsize=0.1, minimizer_kwargs={"bounds": Bounds(0,1), 
                                                              "constraints": LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)},
                              take_step=perturbation)
        candidate_h = bh_res.x
        candidate_h = adjust_sum(candidate_h, target_sum)
        candidate_c5 = np.max(np.correlate(candidate_h, 1 - candidate_h, mode='full') * dx)

        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            print(f"New best C5: {best_c5:.6f}")

        # Local optimization with SLSQP
        res = minimize(
            lambda x: np.max(np.correlate(adjust_sum(x, target_sum), 1 - adjust_sum(x, target_sum), mode='full') * dx),
            x0=best_h, 
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum),
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        refined_h = res.x
        refined_h = adjust_sum(refined_h, target_sum)
        refined_c5 = np.max(np.correlate(refined_h, 1 - refined_h, mode='full') * dx)

        if refined_c5 < best_c5:
            best_h = refined_h.copy()
            best_c5 = refined_c5
            print(f"Refined best C5: {best_c5:.6f}")

    return best_h, best_c5, n_points
