import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def generate_random_individual(n_points, target_sum):
    for _ in range(5):
        h = np.random.rand(n_points)
        h_adj = adjust_sum(h, target_sum)
        if abs(h_adj.sum() - target_sum) < 1e-4:
            return h_adj
    h = np.random.rand(n_points)
    return adjust_sum(h, target_sum)

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

    best_h = h_initial.copy()
