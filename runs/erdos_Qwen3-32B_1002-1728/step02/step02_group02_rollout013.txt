import numpy as np
import time
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_sum = h.sum()
    assert np.all((h >= -1e-8) & (h <= 1.0 + 1e-8))
    assert abs(h_sum - target_sum) < 1e-8

    def compute_C5(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5

    initial_T = 1.0
    alpha = 0.999
    T = initial_T

    start_time = time.time()
    max_time = budget_s

    while time.time() - start_time < max_time - 0.1:
        h_candidate = h.copy()
        found = False
        for _ in range(100):
            i, j = np.random.choice(n_points, 2, replace=False)
            h_i = h_candidate[i]
            h_j = h_candidate[j]
            lower = max(-h_i, h_j - 1.0)
            upper = min(1.0 - h_i, h_j)
            if lower > upper:
                continue
            delta = lower + (upper - lower) * np.random.rand()
            h_new = h_candidate.copy()
            h_new[i] += delta
            h_new[j] -= delta
            found = True
            break
        if not found:
            continue

        new_C5 = compute_C5(h_new)
        delta_C5 = new_C5 - current_C5

        if delta_C5 < 0:
            h[:] = h_new
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_h[:] = h_new
                best_C5 = new_C5
        else:
            if np.random.rand() < np.exp(-delta_C5 / T):
                h[:] = h_new
                current_C5 = new_C5

        T *= alpha

    best_h_sum = best_h.sum()
    if abs(best_h_sum - target_sum) > 1e-8:
        best_h = np.clip(best_h, 0, 1)
        best_h_sum = best_h.sum()
        factor = target_sum / best_h_sum
        best_h *= factor
        best_h = np.clip(best_h, 0, 1)

    final_C5 = compute_C5(best_h)
    return best_h, final_C5, n_points
