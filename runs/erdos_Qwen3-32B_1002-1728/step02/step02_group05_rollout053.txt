            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint.copy(),
            options={'maxiter': 150}
        )
        curr_h = np.clip(result.x.copy(), 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset
        curr_h[0] = max(0.0, min(1.0, curr_h[0]))
        corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_val)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h[:] = curr_h[:]

    # Final check
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
