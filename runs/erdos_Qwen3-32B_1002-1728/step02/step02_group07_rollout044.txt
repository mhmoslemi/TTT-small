import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    np.random.seed(seed)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_current = h_initial.copy()
    current_c5 = compute_c5(h_current)
    best_c5 = current_c5
    best_h = h_current.copy()

    T_initial = max(1e-4, 10 * current_c5)
    t = 1

    start_time = time.time()
    budget_seconds = budget_s - 10

    while time.time() - start_time < budget_seconds:
        try:
            T = T_initial / np.log(1 + t ** 0.5)
        except:
            T = 1e-6

        noise = np.random.randn(n_points) * T * 0.1
        h_candidate = h_current + noise
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = adjust_sum(h_candidate, target_sum)

        new_c5 = compute_c5(h_candidate)
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = h_candidate.copy()
            current_c5 = new_c5

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h_candidate.copy()

        t += 1

    return best_h, best_c5, n_points
