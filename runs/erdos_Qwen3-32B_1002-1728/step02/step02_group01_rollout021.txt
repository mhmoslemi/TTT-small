import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:  # Stop early to avoid timeout
        n_attempt += 1
        rnd = np.random.RandomState(seed + n_attempt)
        prob = rnd.rand()
        
        if prob < 0.33:
            # Gaussian noise perturbation
            perturbed = best_h + rnd.normal(0, 0.05, n_points)
        elif prob < 0.66:
            # Boundary jitter
            perturbed = best_h.copy()
            mask = (rnd.rand(n_points) < 0.15)
            perturbed[mask] = rnd.choice([0., 1.], size=np.sum(mask))
        else:
            # Block perturbation
            block_size = rnd.randint(3, 11)
            start = rnd.randint(0, n_points - block_size)
            perturbed = best_h.copy()
            perturbed[start:start+block_size] = rnd.rand(block_size) * 0.5 + 0.25
        
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        rnd = np.random.RandomState(seed + n_attempt)
        prob = rnd.rand()
        
        if prob < 0.33:
            perturbed = best_h + rnd.normal(0, 0.05 if n_attempt < 1000 else 0.02, n_points)
        elif prob < 0.66:
            perturbed = best_h.copy()
            mask = (rnd.rand(n_points) < 0.15)
            perturbed[mask] = rnd.choice([0., 1.], size=np.sum(mask), p=[0.5, 0.5])
        else:
            block_size = max(3, min(11, int(n_points * 0.1)))
            start = rnd.randint(0, n_points - block_size)
            perturbed = best_h.copy()
            noise_level = 0.25 * (1 - n_attempt / 1000)  # Annealing noise
            perturbed[start:start+block_size] = np.clip(
                perturbed[start:start+block_size] + rnd.normal(0, noise_level, block_size),
                0, 1
            )

        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 80, 'ftol': 1e-9, 'disp': False}
        )

        if result.success:
            new_h = result.x
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
