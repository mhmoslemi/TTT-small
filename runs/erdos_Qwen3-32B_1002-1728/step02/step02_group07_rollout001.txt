import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy().reshape(-1)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(-1)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    best_h = h_initial.copy()
    best_c5 = objective(best_h)
    perturbation_strength = 0.02
    min_strength = 0.001
    max_strength = 0.1

    start_time = time.time()
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 20:
        prev_best = best_c5

        x0 = best_h.copy() + perturbation_strength * np.random.randn(n_points)
        x0 = np.clip(x0, 0, 1)
        current_sum_x0 = x0.sum()
        if current_sum_x0 != 0:
            x0 = x0 * (target_sum / current_sum_x0)
        else:
            x0.fill(target_sum / n_points)

        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=Bounds(0.0, 1.0),
            constraints=LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum),
            options={'maxiter': 100, 'ftol': 1e-6}
        )

        curr_h = result.x.copy()
        curr_h = np.clip(curr_h, 0, 1)
        current_sum = curr_h.sum()
        if current_sum != 0:
            curr_h = curr_h * (target_sum / current_sum)

        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
            perturbation_strength = min(perturbation_strength * 1.1, max_strength)
        else:
            perturbation_strength = max(perturbation_strength * 0.9, min_strength)

        if prev_best == curc_c5:
            perturbation_strength = max(perturbation_strength * 1.05, min_strength)

    return best_h, best_c5, n_points
