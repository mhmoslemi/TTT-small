        sum_after_clip = np.sum(h_scaled)
        if sum_after_clip > 1e-6:
            h_scaled = h_scaled * (target_sum / sum_after_clip)
            h_scaled = np.clip(h_scaled, 0, 1)
        population.append(h_scaled)

    best_h = h.copy()
    best_c5 = np.inf

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:  # Safety margin
        # Evaluate all population members
        fitness_list = []
        for candidate in population:
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            max_corr = np.max(corr)
            c5 = max_corr * dx
            fitness_list.append(c5)

        # Update best solution
        current_best_idx = np.argmin(fitness_list)
        if fitness_list[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness_list[current_best_idx]

        # Select elites and generate next population
        sorted_indices = np.argsort(fitness_list)
        sorted_pop = [population[i] for i in sorted_indices]
        elites_count = max(1, int(pop_size * elites_pct))
        elites = sorted_pop[:elites_count]
        new_pop = elites[:]

        while len(new_pop) < pop_size:
            # Parent selection (from elites and top half of population)
            parent1 = sorted_pop[np.random.randint(len(sorted_pop))]
            parent2 = sorted_pop[np.random.randint(len(sorted_pop))]
            # Crossover
            child = parent1 * crossover_alpha + parent2 * (1 - crossover_alpha)
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child > 1e-6:
                child = child * (target_sum / sum_child)
                child = np.clip(child, 0, 1)

            # Mutation
            if np.random.rand() < mutation_prob:
                noise = np.random.normal(0, mutation_rate, n_points)
                child += noise
                child = np.clip(child, 0, 1)
                sum_child = np.sum(child)
                if sum_child > 1e-6:
                    child = child * (target_sum / sum_child)
                    child = np.clip(child, 0, 1)

            new_pop.append(child)

        # Replace population
        population = new_pop

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
