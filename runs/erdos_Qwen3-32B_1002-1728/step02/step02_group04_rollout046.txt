import time

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def compute_gradient(h, k, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        j = i + k
