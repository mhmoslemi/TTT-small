def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    n_particles = 100
    particles_position = np.zeros((n_particles, n_points))

    # Initialize first half with initial_h_values perturbed
    for i in range(n_particles // 2):
        h = h_initial.copy()
        h += np.random.normal(0, 0.1, h.shape)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != 0:
            scale = target_sum / sum_h
            h *= scale
            h = np.clip(h, 0, 1)
        delta = target_sum - np.sum(h)
        h[0] += delta
        h[0] = max(0.0, min(1.0, h[0]))
        particles_position[i] = h.copy()

    # Initialize second half with random h values
    for i in range(n_particles // 2, n_particles):
        h = np.random.rand(n_points)
        sum_h = np.sum(h)
        if sum_h < 1e-9:
            h = np.ones_like(h)
            sum_h = n_points
        scale = target_sum / sum_h
        h *= scale
        h = np.clip(h, 0, 1)
        delta = target_sum - np.sum(h)
        h[0] += delta
        h[0] = max(0.0, min(1.0, h[0]))
        particles_position[i] = h.copy()

    particles_velocity = np.random.rand(n_particles, n_points) * 0.1 - 0.05  # Initial velocities

    def compute_fitness(h):
        g = 1 - h
        corr = scipy.signal.fftconvolve(h, g[::-1], mode='full')
        return np.max(corr) * dx

    pbest_position = particles_position.copy()
    pbest_fitness = np.array([compute_fitness(p) for p in particles_position])
    gbest_idx = np.argmin(pbest_fitness)
    gbest_fitness = pbest_fitness[gbest_idx]
    gbest_position = pbest_position[gbest_idx].copy()

    w_max, w_min = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    start_time = time.time()

    while time.time() - start_time < budget_s:

        w = w_max - (w_max - w_min) * (elapsed / budget_s)

        # Update velocities and positions
