import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h_vec):
        h = h_vec.reshape(-1)
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()
    
    def adjust_sum(h, target_sum):
        h_adj = h.copy()
        current_sum = h_adj.sum()
        if abs(current_sum - target_sum) < 1e-9:
            return h_adj
        D = target_sum - current_sum
        if D > 0:
            can_incr = (h_adj < 0.9999)
            if not can_incr.any():
                return h_adj
            total_possible = (1.0 - h_adj[can_incr]).sum()
            h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
        else:
            D = -D
            can_decr = (h_adj > 1e-4)
            if not can_decr.any():
                return h_adj
            total_possible = h_adj[can_decr].sum()
            h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
        return h_adj

    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()
    noise_scale = 0.02
    min_noise = 0.005
    max_noise = 0.05

    while time.time() - start_time < budget_s - 20:
        # Perturb with adaptive noise
        perturbation = noise_scale * np.random.randn(n_points)
        x0 = best_h.copy() + perturbation
        x0 = np.clip(x0, 0, 1)
        x0 = adjust_sum(x0, target_sum)

        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 150, 'ftol': 1e-7}
        )
        
        curr_h = np.clip(result.x, 0, 1)
        curr_h = adjust_sum(curr_h, target_sum)
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
            noise_scale = min(max_noise, noise_scale * 1.1)
        else:
            noise_scale = max(min_noise, noise_scale * 0.9)
    
    return best_h, best_c5, n_points
