    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    np.random.seed(seed)

    def project_h(h):
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if abs(total) < 1e-10:
            return h
        scaling = required_sum / total
        h *= scaling
        h = np.clip(h, 0.0, 1.0)
        return h

    h = project_h(h_initial).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start_time = time.time()
    iteration = 0

    alpha = 0.01
    decay_interval = 100
    perturb_interval = 200

    while time.time() - start_time < budget_s - 10:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        k_samples = max_idx - (n_points - 1)
        m = abs(k_samples)
        
        if m >= n_points:
            gradient = np.zeros_like(h)
        else:
            if k_samples >= 0:
                term1 = np.zeros_like(h)
                term1[0 : n_points - m] = 1.0 - h[m:]
                term2 = np.zeros_like(h)
                term2[m:] = -h[0 : n_points - m]
            else:
                m_pos = m
                term1 = np.zeros_like(h)
                term1[m_pos:] = 1.0 - h[0 : n_points - m_pos]
                term2 = np.zeros_like(h)
                term2[0 : n_points - m_pos] = -h[m_pos:]
            gradient = (term1 + term2) * dx
        
        h = h - alpha * gradient
        h = project_h(h)

        iteration += 1
        if iteration % perturb_interval == 0:
            h = project_h(h + np.random.normal(0, 0.05, size=n_points))

        if iteration % decay_interval == 0:
            alpha *= 0.95

        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
