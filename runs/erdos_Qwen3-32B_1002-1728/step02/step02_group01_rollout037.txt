import numpy as np
import time

def evaluate(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def mutate(parent, target_sum):
    noisy = parent + np.random.normal(0, 0.05, size=parent.shape)
    clipped = np.clip(noisy, 0, 1)
    current_sum = clipped.sum()
    if current_sum == 0:
        return parent.copy()
    scaled = clipped * (target_sum / current_sum)
    return np.clip(scaled, 0, 1)

def generate_random_h(n_points, target_sum):
    h = np.random.rand(n_points)
    total = h.sum()
    if total == 0:
        return np.zeros(n_points)
    h = (h / total) * target_sum
    h = np.clip(h, 0, 1)
    # Adjust sum after clipping
    adjusted_sum = h.sum()
    delta = target_sum - adjusted_sum
    mid_mask = (h > 1e-6) & (h < 1 - 1e-6)
    if np.any(mid_mask) and delta != 0:
        h[mid_mask] += delta / np.count_nonzero(mid_mask)
    return np.clip(h, 0, 1)
