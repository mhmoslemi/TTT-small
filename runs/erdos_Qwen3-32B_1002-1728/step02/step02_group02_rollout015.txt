            end = min(start + np.random.randint(2, 11), n_points)
            h2 = h.copy()
            h2[start:end] = np.clip(h2[start:end] + np.random.uniform(-0.2, 0.2, end - start), 0, 1)
            return _project(h2)

    while time.time() - start_time < budget_s - 0.1:
        # Selection
        candidates = tournament_select(3)
        candidates_scores = [compute_c5(c) for c in candidates]
        indices = np.argsort(candidates_scores)
        p1, p2 = candidates[indices[0]], candidates[indices[1]]

        # Crossover and mutate
        child = mutate(crossover(p1, p2))

        # Evaluate
        child_score = compute_c5(child)

        # Replacement
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_c5:
                best_h = child
                best_c5 = child_score

        # Convergence check
        if time.time() - start_time > budget_s - 5:
            break

    return best_h, best_c5, n_points
