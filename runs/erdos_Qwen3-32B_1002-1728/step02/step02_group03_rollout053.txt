import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    step_size = 0.01
    best_h = h.copy()
    best_c5 = np.inf
    patience = 0
    decay_rate = 0.95

    while time.time() - start_time < budget_s - 1:
        # Compute overlaps and identify critical shifts
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c5 = np.max(corr)
        max_indices = np.where(np.isclose(corr, max_c5, atol=1e-8))[0]
        if len(max_indices) > 5:
            max_indices = max_indices[:5]  # Limit to top 5 shifts

        # Compute gradient from critical shifts
        grad = np.zeros(n_points)
        for idx in max_indices:
            k = idx - (n_points - 1)
            for i in range(n_points):
                if 0 <= i + k < n_points:
                    grad[i] += (1 - h[i + k]) * dx

        # Update h with gradient descent
        h -= step_size * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        
        if abs(current_sum) > 1e-10:
            h *= (target_sum / current_sum)
            h = np.clip(h, 0, 1)  # Final clip after rescaling

        # Track improvement
        corr_current = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
            # Adaptive step size
            if best_c5 < 0.3809:
                step_size *= 0.99
        else:
            patience += 1
            if patience > 100:
                step_size *= decay_rate  # Reduce step size if no improvement
                patience = 0

    return best_h, best_c5, n_points
