    target_sum = n_points / 2
    pop_size = 50
    generations = 2000
    mutation_rate = 0.05
    mutation_std = 0.1
    tournament_size = 5
    elitism_ratio = 0.2
    grad_steps = 100
    grad_step_size = 0.01

    def evaluate(candidates):
        c5_values = []
        for h in candidates:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5_values.append(np.max(corr))
        return np.array(c5_values)

    def scale_to_sum(h):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return h
        scale = target_sum / s
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    def mutate(h):
        h = np.copy(h)
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, mutation_std, size=np.sum(mask))
        h = np.clip(h, 0, 1)
        h = scale_to_sum(h)
        return h

    def crossover(parent1, parent2):
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        child = scale_to_sum(child)
        return child

    def tournament_select(candidates, scores, size):
        indices = np.random.choice(len(candidates), size)
        winner_idx = indices[np.argmin(scores[indices])]
        return candidates[winner_idx]

    population = [h_initial]
    for _ in range(pop_size - 1):
        mut = h_initial + np.random.normal(0, 0.05, n_points)
        mut = scale_to_sum(np.clip(mut, 0, 1))
        population.append(mut)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = np.inf
    start_time = time.time()

    while time.time() - start_time < budget_s - 1:
        scores = evaluate(population)
        curr_best_idx = np.argmin(scores)
        if scores[curr_best_idx] < best_c5:
            best_c5 = scores[curr_best_idx]
            best_h = population[curr_best_idx].copy()

        elitism_size = int(elitism_ratio * pop_size)
        elitists = population[np.argsort(scores)[:elitism_size]].copy()

        new_population = elitists.copy()
        while len(new_population) < pop_size:
            parent1 = tournament_select(population, scores, tournament_size)
            parent2 = tournament_select(population, scores, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population = np.vstack([new_population, child])

        population = new_population[:pop_size]

    # Gradient refinement
    current_h = best_h.copy()
    current_c5 = best_c5
    step = 1.0
    while time.time() - start_time < budget_s - 1 and step <= grad_steps:
        # Compute gradient on current max-k index
        corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
        max_index = np.argmax(corr)
        k_shift = max_index - (n_points - 1)
        if k_shift <= 0 or k_shift >= n_points:
            break

        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] * (1 - h[i + k_shift]) for i + k_shift in domain
            if i + k_shift < n_points:
                grad[i] += (1 - current_h[i + k_shift]) * dx
            # Contribution from h[i - k_shift] * (1 - h[i]) for i - k_shift in domain
            if i - k_shift >= 0:
                grad[i] -= current_h[i - k_shift] * dx

        # Remove mean to maintain sum constraint
        grad -= grad.mean()
        # Update with decreasing step size
        current_h = current_h - grad_step_size * grad
        current_h = np.clip(current_h, 0, 1)
        current_h = scale_to_sum(current_h)

        # Recompute objective
        corr = np.correlate(current_h, 1 - current_h, mode='full') * dx
        new_c5 = np.max(corr)
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            grad_step_size *= 0.9
        step += 1

    corr_values = np.correlate(current_h, 1 - current_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return current_h, final_c5, n_points
