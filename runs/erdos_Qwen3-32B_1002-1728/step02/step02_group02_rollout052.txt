        options={'maxiter': 100000}
    )
    
    best_h = _project(result.x, target_sum, n_points)
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_value = np.max(corr_values)
    return best_h, c5_value, n_points
