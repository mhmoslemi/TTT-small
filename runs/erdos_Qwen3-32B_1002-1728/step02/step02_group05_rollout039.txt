    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    def generate_halton_guess(seed_val, n_points):
        sampler = Halton(d=1, seed=seed_val)
        samples = sampler.random(n=n_points)[:, 0]
        h = (samples < 0.5).astype(float)
        offset = (n_points / 2) - np.sum(h)
        h += offset / n_points
        h = np.clip(h, 0.0, 1.0)
        return h

    def objective_with_penalty(h):
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-2:
            return 1e6  # Hard constraint violation penalty
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    for period in [3,4,5,6,7]:

            h[indices] = 1.0
            factor = (n_points / 2) / np.sum(h)
            h *= factor
            h = np.clip(h, 0.0, 1.0)
