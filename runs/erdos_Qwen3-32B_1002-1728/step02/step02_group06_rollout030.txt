import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 50
    max_generations = 10000
    tournament_size = 3
    crossover_rate = 0.7
    mutation_rate = 0.1
    mutation_scale = 0.05

    def generate_random():
        h = np.random.rand(n_points)
        h = h / h.sum() * target_sum
        return np.clip(h, 0, 1)

    def generate_pattern(p):
        pattern = (np.arange(n_points) % p) < (p // 2)
        h = np.random.permutation(pattern.astype(float))
        h = h / h.sum() * target_sum
        return np.clip(h, 0, 1)

    pop = []
    pop.append(h_init.copy())
    pop.extend(generate_pattern(p) for p in [2, 4, 6, 8, 10])
    for _ in range(pop_size - len(pop)):
        pop.append(generate_random())

    def evaluate(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if abs(current_sum) < 1e-10:
            return np.zeros_like(h)
        h *= target_sum / current_sum
        h = np.clip(h, 0, 1)
        adjustment = target_sum - h.sum()
        h[0] += adjustment
        return np.clip(h, 0, 1)

    best_h = np.array(pop[0])
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break

        pop = [repair(h) for h in pop]
        scores = np.array([evaluate(h) for h in pop])
        sorted_indices = np.argsort(scores)
        pop = [pop[i] for i in sorted_indices]
        parents = pop[:pop_size // 5]

        children = []
        for _ in range(pop_size // 5):
            if np.random.rand() < crossover_rate:
                a, b = np.random.choice(len(parents), 2)
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * parents[a] + (1 - alpha) * parents[b]
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, mutation_scale, n_points)
                    child += noise
                children.append(repair(child))

        weak = pop[-len(children):]
        for i, child in enumerate(children):
            weak[i] = child

        current_best = evaluate(pop[0])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = pop[0].copy()

    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(final_corr), n_points
