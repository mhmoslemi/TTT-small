    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        if np.isnan(current_c5).any():
            break

        # Find worst shift
        s_max = np.argmax(corr)
        k = s_max - (n_points - 1)

        # Compute gradient for k
        grad = np.zeros_like(h)
        if k >= 0:
            valid = np.arange(n_points - k)
            grad[valid] += (1 - h[valid + k]) * dx
            grad[valid + k] -= h[valid] * dx
        else:
            m = -k
            valid = np.arange(m, n_points)
            grad[valid - m] += (1 - h[valid]) * dx
            grad[valid] -= h[valid - m] * dx

        # Update h with gradient
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        s = h.sum()
        h = h * (target_sum / s)
        h = np.clip(h, 0, 1)
        alpha *= decay_rate

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr), n_points
