import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0
    perturbation_strength = 0.1

    while time.time() - start_time < budget_s - 60:  # Stop early to avoid timeout
        n_attempt += 1
        # Choose perturbation type randomly
        perturb_type = np.random.choice(['gaussian', 'boundary', 'block'])

        # Generate perturbed candidate based on the perturbation type
        if perturb_type == 'gaussian':
            perturbed = best_h + np.random.normal(0, perturbation_strength, n_points)
        elif perturb_type == 'boundary':
            jitter_mask = np.random.rand(n_points) < 0.2  # 20% chance to perturb each point
            directions = np.random.choice([-1, 1], size=n_points)
            perturbed = best_h.copy()
            perturbed[jitter_mask] += directions[jitter_mask] * perturbation_strength
        elif perturb_type == 'block':
            start = np.random.randint(0, n_points - 1)
            end = np.random.randint(start + 1, n_points)
            block_size = end - start
            delta = np.random.normal(0, perturbation_strength * block_size)
            perturbed = best_h.copy()
            perturbed[start:end] += delta / block_size  # Adjust values in the block
        else:
            # This shouldn't happen
            perturbed = best_h.copy()

        # Clip to [0, 1]
        perturbed = np.clip(perturbed, 0, 1)
        # Scale to meet the sum constraint
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        # Clip again due to scaling (numerical precision)
        perturbed = np.clip(perturbed, 0, 1)

        # Optimization with increased maxiter and tighter tolerance
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
        )

        if result.success:
            new_h = result.x.reshape(n_points)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            # Check constraints and update best solution
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    # Decrease perturbation strength after improvement
                    perturbation_strength *= 0.95

            perturbation_strength *= 0.98

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h_func = h_vec.reshape(n_points)
        corr = np.correlate(h_func, 1 - h_func, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0
    perturbation_strength = 0.1

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        # Randomly choose perturbation type
        perturb_type = np.random.choice(['gaussian', 'boundary', 'block'])

        # Generate perturbed candidate
        if perturb_type == 'gaussian':
            perturbed = best_h + np.random.normal(0, perturbation_strength, n_points)
        elif perturb_type == 'boundary':
            jitter_mask = np.random.rand(n_points) < 0.2
            directions = np.random.choice([-1, 1], size=n_points)
            perturbed = best_h.copy()
            perturbed[jitter_mask] += directions[jitter_mask] * perturbation_strength
        elif perturb_type == 'block':
            start = np.random.randint(0, n_points - 1)
            end = np.random.randint(start + 1, n_points)
            block_size = end - start
            delta = np.random.normal(0, perturbation_strength * block_size)
            perturbed = best_h.copy()
            perturbed[start:end] += delta / block_size

        # Project to feasible region
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        # Optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
        )

        if result.success:
            new_h = result.x.reshape(n_points)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            # Validate constraints
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    # Adaptive learning rate after improvement
                    perturbation_strength *= 0.95

        # Periodic temperature decay
        if n_attempt % 50 == 0:
            perturbation_strength *= 0.98
        perturbation_strength = max(0.01, perturbation_strength)

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
