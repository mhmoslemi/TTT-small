import time
import cvxpy as cp

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

        k_worst_index = np.argmax(corr)
        k_shift = k_worst_index - (n_points - 1)

        valid_pairs = []
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                valid_pairs.append((i, j))

        h_var = cp.Variable(n_points)
        objective = cp.Minimize(sum(h_var[i] * h_var[j] for i, j in valid_pairs))
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == required_sum]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.OSQP, verbose=False)

        if prob.status != cp.OPTIMAL:
            continue

        h_qpsol = np.clip(np.array(h_var.value).flatten(), 0, 1)
