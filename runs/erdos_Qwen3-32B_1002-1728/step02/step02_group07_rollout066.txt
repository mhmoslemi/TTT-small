            h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
        return h_adj

    def objective(h):
        h = h.reshape(-1)
        h_adj = adjust_sum(h, target_sum)
        corr_values = np.correlate(h_adj, 1 - h_adj, mode='full') * dx
        return np.max(corr_values)

    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()
    
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    while time.time() - start_time < budget_s - 20:
        result = minimize(
            objective,
            x0=x0,
            method='Nelder-Mead',
            bounds=bounds,
            options={'maxiter': 300, 'xatol': 1e-6}
        )
        
        curr_h = result.x
        curr_h_adj = adjust_sum(curr_h, target_sum)
        curr_c5 = np.max(np.correlate(curr_h_adj, 1 - curr_h_adj, mode='full') * dx)
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h_adj.copy()
        
        x0 = best_h.copy() + 0.02 * np.random.randn(n_points)
        x0 = np.clip(x0, 0, 1)
    
    return best_h, best_c5, n_points
