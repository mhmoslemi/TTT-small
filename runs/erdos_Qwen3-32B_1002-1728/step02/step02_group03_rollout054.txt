        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find max indices
        max_val = np.max(corr_full)
        max_mask = np.isclose(corr_full, max_val)
        max_indices = np.where(max_mask)[0]
        max_indices = max_indices[-top_k_shifts:] if len(max_indices) > top_k_shifts else max_indices

        total_grad = np.zeros(n_points)
        for m in max_indices:
            shift_samples = m - (n_points - 1)
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + shift_samples
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
                    grad[j] -= h[i]
            total_grad += grad

        total_grad *= dx
        velocity = momentum * velocity - alpha * total_grad
        h += velocity

        # Enforce constraints
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total) > 1e-8:
            h *= target_sum / total
            h = np.clip(h, 0, 1)

    # Validate final best_h
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
