        grad = np.zeros(n_points)
        for k in k_shifts:
            g = np.zeros(n_points)
            for j in range(n_points):
                if 0 <= j + k < n_points:
                    g[j] += (1 - h[j + k])
                if 0 <= j - k < n_points:
                    g[j] -= h[j - k]
            grad += g
        
        if len(k_shifts) > 0:
            grad /= len(k_shifts)
            h -= alpha * grad
        else:
            h = best_h.copy()
        
        # Project onto constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (target_sum / h_sum)
            h = np.clip(h, 0, 1)
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
            alpha = 0.02
        else:
            no_improve += 1
            if no_improve > patience:
                alpha *= 0.95

    return best_h, best_c5, n_points
