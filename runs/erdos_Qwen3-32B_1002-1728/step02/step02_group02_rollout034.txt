def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = np.sum(clamped)
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    scaling_factor = target_sum / clamped_sum
    adjusted = clamped * scaling_factor
    adjusted = np.clip(adjusted, 0, 1)
    delta = target_sum - np.sum(adjusted)
    if abs(delta) > 1e-6:
        mask = (adjusted > 0) & (adjusted < 1)
        if np.any(mask):
            adjusted[mask] += delta / np.sum(mask)
            adjusted = np.clip(adjusted, 0, 1)
    return adjusted

def _crossover(parent1, parent2, n_points):
    points = np.sort(np.random.choice(n_points, size=4, replace=False))
    segments = []
    segments.append( (0, points[0]) )
    segments.append( (points[0], points[1]) )
    segments.append( (points[1], points[2]) )
    segments.append( (points[2], points[3]) )
    segments.append( (points[3], n_points) )
    off1 = np.zeros(parent1.shape)
    off2 = np.zeros(parent1.shape)
    for i, (start, end) in enumerate(segments):
        if i % 2 == 0:
            off1[start:end] = parent1[start:end]
            off2[start:end] = parent2[start:end]
        else:
            off1[start:end] = parent2[start:end]
            off2[start:end] = parent1[start:end]
    return off1, off2

def jitter_mutation(p, sigma=0.05):
    p += np.random.normal(0, sigma, size=len(p))
    return p

def invert_segment_mutation(p, max_len=10):
    start = np.random.randint(0, len(p))
    length = np.random.randint(1, min(max_len, len(p) - start))
    p[start:start+length] = 1 - p[start:start+length]
    return p

def slide_mutation(p):
    start = np.random.randint(0, len(p))
    length = np.random.randint(1, min(10, len(p) - start))
    shift = np.random.uniform(-0.1, 0.1)
    p[start:start+length] += shift
    return p
