        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            new_h_half = result.x.reshape(half_n)
            new_corr = np.correlate(create_full_h(new_h_half), 1 - create_full_h(new_h_half), mode='full')
            new_c5 = np.max(new_corr) * dx
            if (abs(np.sum(new_h_half) - target_sum_half) < 1e-6 and
                np.all((new_h_half >= -1e-6) & (new_h_half <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h_half = new_h_half
                    best_c5 = new_c5

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
