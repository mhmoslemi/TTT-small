import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    cons = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)

    def objective(h_vec):
        h = h_vec.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    rs = np.random.RandomState(seed)
    initializations = [h_init.copy()]
    for _ in range(4):
        perturbed = h_init.copy() + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)

            perturbed = perturbed / perturbed.sum() * target_sum
        perturbed = np.nan_to_num(perturbed, nan=0.0)
        initializations.append(perturbed)

    best_h = h_init.copy()

    for init in initializations:
        if time.time() - start_time > budget_s * 0.98:
            break
        result = minimize(
            fun=objective,
            x0=init,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'tol': 1e-8}
        )
        if result.success:
            h_candidate = np.clip(result.x, 0, 1)
            curr_sum = h_candidate.sum()
            if abs(curr_sum - target_sum) < 1e-6:
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')

                    best_h = h_candidate.copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
