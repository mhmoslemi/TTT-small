            corr = np.correlate(h_grid, 1 - h_grid, mode='full')
            overlaps = corr * dx_grid
            return np.log(np.sum(np.exp(alpha * overlaps))) / alpha

        # Run optimization
        result = minimize(
            fun=objective_log_sum,
            x0=w_perturbed,
            method='SLSQP',
            bounds=bounds_w,
            constraints=cons_w,
            options={'maxiter': 30, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            w_opt = result.x
            # Compute actual overlap to check improvement
            h_grid_opt = np.zeros(m)
            for i in range(n_points):
                diff = x_grid - x_centers[i]
                kernel = np.exp(-0.5 * (diff / sigma)**2)
                h_grid_opt += w_opt[i] * kernel
            corr_opt = np.correlate(h_grid_opt, 1 - h_grid_opt, mode='full')
            overlaps_opt = corr_opt * dx_grid
            current_c5_opt = np.max(overlaps_opt)

            # Project to discrete step function and validate constraint
            h_projected = np.zeros(n_points)
            for i in range(n_points):
                diff_centers = x_centers - x_centers[i]
                kernel_vals = np.exp(-0.5 * (diff_centers / sigma)**2)
                h_projected[i] = np.dot(w_opt, kernel_vals)
            h_projected = np.clip(h_projected, 0, 1)
            total = np.sum(h_projected)
            if abs(total - target_sum) > 1e-6:
                h_projected = h_projected / total * target_sum
                h_projected = np.clip(h_projected, 0, 1)

            # Evaluate C5 of the projected solution
            corr_projected = np.correlate(h_projected, 1 - h_projected, mode='full')
            c5_projected = np.max(corr_projected) * dx

            if c5_projected < best_c5:
                best_w = w_opt.copy()
                best_h = h_projected.copy()
                best_c5 = c5_projected
                print(f"Improved C5: {best_c5:.9f}")

    # Final solution with constraint validation
    h_final = best_h
    assert abs(np.sum(h_final) - target_sum) < 1e-6, "Integral violation"
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6)), "Bounds violation"

    return (h_final.tolist(), best_c5, n_points)
