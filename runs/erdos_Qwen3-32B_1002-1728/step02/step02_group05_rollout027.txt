    low, high = 0.0, 0.38086  # Just above target bound

    while time.time() - start_time < budget_s - 40:
        mid = (low + high) / 2
        h_var = cp.Variable(n_points, lb=0, ub=1)
        constraints = [cp.sum(h_var) == n_points / 2]

        # Formulate constraints for all k
        for k in range(1, n_points):
            m = n_points - k
            terms = [(h_var[i] - h_var[i - k]) * dx for i in range(k, n_points)]
            lhs = cp.sum(terms)
            constraints.append(lhs + m * dx <= 2 * mid)

        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)

        if prob.status in {cp.OPTIMAL, cp.FEASIBLE}:
            # Extract and validate solution
            curr_h = h_var.value.clip(0, 1)
            corr = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            actual_c5 = np.max(corr)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h[:] = curr_h[:]
            high = mid  # Try tighter bound
        else:
            low = mid  # Infeasible; relax lower bound

        # Small random restart
        if np.random.rand() < 0.25 and time.time() - start_time < budget_s - 40:
