def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def adjust_sum(h, target_sum):
    current_sum = h.sum()
    if np.isclose(current_sum, target_sum):

    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np

def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def adjust_sum(h, target_sum):
    h = h.copy()
    current_sum = h.sum()
    delta = target_sum - current_sum
    if np.isclose(delta, 0):
        return h

    for i in range(len(h)):
        current_val = h[i]
        if delta > 0:
            possible_increase = 1.0 - current_val
            if possible_increase > 0:
                added = min(delta, possible_increase)
                h[i] += added
                delta -= added
        elif delta < 0:
            possible_decrease = current_val
            if possible_decrease > 0:
                subtracted = min(-delta, possible_decrease)
                h[i] -= subtracted
                delta += subtracted
        if np.isclose(delta, 0):
            break
    h = np.clip(h, 0.0, 1.0)
