                    new_val = np.clip(new_val, 0.0, 1.0)
                    new_h = h.copy()
                    new_h[j] = new_val
                    new_h = repair(new_h)
                    new_c5 = compute_c5(new_h)
                    if new_c5 < best_new_c5:
                        best_new_c5 = new_c5
                        best_new_h = new_h.copy()
                if best_new_c5 < current_c5:
                    h = best_new_h.copy()
                    current_c5 = best_new_c5
                    made_progress = True
                if time.time() - start_cycle > 1.0:
                    break
            if not made_progress:
                break
            if best_c5 is None or current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
