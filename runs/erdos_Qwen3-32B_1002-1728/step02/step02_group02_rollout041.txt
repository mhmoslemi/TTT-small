import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h_current = h_initial.copy()
    best_h = h_current.copy()
    best_c5 = np.max(correlate(h_current, 1 - h_current, mode='full') * dx)

    start_time = time.time()
    max_iterations = 10000

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute current correlation and find worst lag
        corr = correlate(h_current, 1 - h_current, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        if max_corr * dx <= best_c5:
            best_c5 = max_corr * dx
            best_h = h_current.copy()
            # Early exit if sufficient improvement
            if best_c5 <= 0.380876:
                break
        lag_worst = max_idx - (n_points - 1)

        # Collect valid (i, j) pairs for the worst lag
        valid_indices = []
        for i in range(n_points):
            j = i + lag_worst
            if 0 <= j < n_points:
                valid_indices.append((i, j))
        if not valid_indices:
            continue  # Skip empty indices

        # Build linear objective coefficients
        coefficients = np.zeros(n_points)
        h_flat = h_current.flatten()
        for i, j in valid_indices:
            coefficients[i] += (1.0 - h_flat[j]) * dx
            coefficients[j] -= h_flat[i] * dx

        # Solve convex subproblem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(coefficients, h_var)))
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
                h_current = np.clip(h_var.value, 0.0, 1.0)
                h_current = _project_sum(h_current, target_sum)
                
                # Update best solution if improved
                curr_c5 = np.max(correlate(h_current, 1 - h_current, mode='full') * dx)
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_h = h_current.copy()
        except Exception:
            continue
    
    return best_h, best_c5, n_points

def _project_sum(p, target_sum):
    clamped = np.clip(p, 0, 1)
    s = np.sum(clamped)
    if np.abs(s) < 1e-9:
        return np.ones_like(clamped) * 0.5
    return (target_sum / s) * clamped
