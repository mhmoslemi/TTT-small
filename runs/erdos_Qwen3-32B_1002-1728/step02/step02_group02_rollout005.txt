import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr * dx)

    def project(h):
        clamped = np.clip(h, 0, 1)
        total = clamped.sum()
        if total == 0:
            return 0.5 * np.ones(n_points)
        return np.clip(clamped * (target_sum / total), 0, 1.0)

    # Ensure initial solution is feasible
    h_initial[:] = project(h_initial)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    start_time = time.time()
    result = minimize(
        objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={
            'maxiter': 10000,
            'ftol': 1e-8,
            'disp': False
        }
    )

    h_opt = result.x if result.success else h_initial
    h_opt = project(h_opt)

    corr = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr)
    return h_opt, c5_bound, n_points
