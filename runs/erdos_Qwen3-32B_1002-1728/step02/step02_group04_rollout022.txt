    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Hyperparameters
    alpha = 0.01
    alpha_decay = 0.995
    perturbation_interval = 200
    noise_sigma = 0.05
    max_iter = 100000
    start_time = time.time()

    # Main loop
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute full correlation
        corr = np.correlate(h, 1 - h, mode='full')
        k_full = np.argmax(corr)

        # Compute the continuous shift m
        m = k_full - n_points + 1

        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            j_plus = (i + m) % n_points
            j_minus = (i - m) % n_points
            grad[i] = 1.0 - h[j_plus] - h[j_minus]
        grad *= dx

        # Gradient step
        h -= alpha * grad

        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum > 1e-10:
            scale = required_sum / current_sum
            h *= scale
            # Clip again to ensure no overflow
            h = np.clip(h, 0, 1)

        # Check C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            alpha = max(alpha * alpha_decay, 1e-4)  # decay learning rate

        # Perturbation every so often

                h = np.clip(h, 0, 1)

        scaling = required_sum / h_sum

    perturb_freq = 200

    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        corr = np.correlate(h, 1 - h, mode='full')
        k_full = np.argmax(corr)
        m = k_full - n_points + 1
