    population.extend([perturb_block(best_h, target_sum, rs) for _ in range(5)])
    
    while time.time() - start_time < budget_s - 30:
        if not population:
            break
        idx = rs.randint(len(population))
        candidate = population.pop(idx)
        perturbation_type = rs.randint(3)
        if perturbation_type == 0:
            h_pert = perturb_gaussian(candidate, 0.05 * (0.9 ** (len(population))), target_sum, rs)
        elif perturbation_type == 1:
            h_pert = perturb_boundary(candidate, target_sum, rs)
        else:
            h_pert = perturb_block(candidate, target_sum, rs)
        
        result = minimize(
            fun=objective,
            x0=h_pert.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 120, 'ftol': 1e-10, 'disp': False}
        )
        
        if result.success:
            new_h = np.clip(result.x.reshape(n_points), 0, 1)
            total = np.sum(new_h)
            if total != 0:
                new_h = (new_h / total) * target_sum
            new_h = np.clip(new_h, 0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
            population.extend([best_h.copy(), perturb_gaussian(best_h, 0.03, target_sum, rs), perturb_block(best_h, target_sum, rs)])
        if len(population) > 50:
            population.pop(rs.randint(len(population)))
    
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    if total != 0:
        best_h = (best_h / total) * target_sum
    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
