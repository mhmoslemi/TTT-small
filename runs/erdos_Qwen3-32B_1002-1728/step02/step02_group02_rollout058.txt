import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]

    start_time = time.time()

    class BudgetExpired(Exception):
        pass

    def callback(xk, state):
        if time.time() - start_time > budget_s - 0.1:
            raise BudgetExpired

    options = {
        'maxiter': 10_000,
        'ftol': 1e-10,
        'disp': False,
        'callback': callback
    }

    result = None
    try:
        result = opt.minimize(
            objective,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options=options
        )
    except BudgetExpired:
        pass  # Use partial result

    # Post-process solution
    h_opt = (result.x if result is not None else h_initial).copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    current_sum = h_opt.sum()

    if not np.isclose(current_sum, target_sum, atol=1e-6):
        scaling = target_sum / current_sum
        h_opt = np.clip(h_opt * scaling, 0.0, 1.0)

    corr = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    c5_bound = np.max(corr)

    return h_opt, c5_bound, n_points
