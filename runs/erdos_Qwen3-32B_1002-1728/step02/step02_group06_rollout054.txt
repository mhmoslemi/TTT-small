                delta = target_sum - current_sum_scaled
                scaled[0] += delta
                scaled[0] = max(0.0, min(1.0, scaled[0]))
            h_vec = scaled
        return h_vec

    current_h = h.copy()
    best_h = current_h.copy()
    current_c5 = compute_c5(current_h)
    best_c5 = current_c5

    initial_temperature = 0.1
    alpha = 0.99999
    scale_perturbation = 0.1
    T = initial_temperature

    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s:
        perturbation = np.random.normal(0, scale_perturbation * T, n_points)
        new_h = current_h + perturbation
        new_h = repair(new_h, target_sum)

        new_c5 = compute_c5(new_h)
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_h = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

        T *= alpha
        iteration += 1

        if T < 1e-8:
            break

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
