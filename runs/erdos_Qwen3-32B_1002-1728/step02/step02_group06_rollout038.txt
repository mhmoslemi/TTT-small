    # Adjust one element to 0.5 if n_high is less than required sum
    sum_h = np.sum(h)
    delta = n_half - sum_h
    if delta != 0.0:  # for odd n_points
        h[indices[0]] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, lag, dx):
    n = len(h)
    gradient = np.zeros(n)
    for m in range(n):
        # Compute term1: 1 - h[m + lag] if m + lag is in [0, n-1]
        m_plus = m + lag
        term1 = 0.0
        if 0 <= m_plus < n:
            term1 = 1.0 - h[m_plus]
        # Compute term2: h[m - lag] if m - lag is in [0, n-1]
        m_minus = m - lag
        term2 = 0.0
        if 0 <= m_minus < n:
            term2 = h[m_minus]
        gradient[m] = (term1 - term2) * dx
