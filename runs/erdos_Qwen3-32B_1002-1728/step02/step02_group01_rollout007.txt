import numpy as np
from scipy.optimize import minimize, SLSQP
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:  # Early stop
        n_attempt += 1
        # Generate perturbed initial guess with increased noise
        perturbed = best_h + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Adjust to meet the integral constraint
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        # Run SLSQP optimization with increased iterations
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-10}
        )

        if result.success:
            new_h = result.x.reshape(n_points)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            # Validate constraints
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h

    # Final correction to exactly satisfy the sum constraint
    current_sum = np.sum(best_h)
    delta = target_sum - current_sum
    adjustment_idx = np.argmin(best_h)  # Adjust the smallest value
    best_h[adjustment_idx] = max(0.0, min(1.0, best_h[adjustment_idx] + delta))
    best_c5 = objective(best_h)

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
