            c5_new = objective_func(h_vec)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_vec.copy()
                h = h_vec.copy()
        if time.time() - start_time > budget_s - 10:
            break

    return best_h, best_c5, n_points
