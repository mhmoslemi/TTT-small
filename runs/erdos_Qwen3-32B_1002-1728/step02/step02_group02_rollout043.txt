import cvxpy as cp
import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h = cp.Parameter(n_points, value=h_initial)
    h_var = cp.Variable(n_points)

    start_time = time.time()
    best_c5 = evaluate(h_initial, dx)
    best_h = h_initial.copy()
    max_iter = 1000
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute current overlaps
        corr_values = np.correlate(h.value, 1 - h.value, mode='full') * dx
        # Find the worst k_samples in the relevant part (shifts of 0 to n-1 steps)
        right_part_start = n_points - 1
        right_corr = corr_values[right_part_start : 2*n_points -1]
        worst_idx = np.argmax(right_corr)
        worst_m = right_part_start + worst_idx
        k_samples = worst_idx  # since worst_idx is the shift in steps (0 to n-1)

        # Now, for k_samples, compute the linear approximation
        # for each i, i + k_samples < n_points
        linear_terms = []
        for i in range(n_points - k_samples):
            # current values
            x0 = h.value[i]
            y0 = h.value[i + k_samples]
            # term is (1 - y0)*h[i] - x0 * h[i + k_samples]
            linear_terms.append( (1 - y0) * h_var[i] - x0 * h_var[i + k_samples] )

        objective = cp.Minimize(sum(linear_terms))
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)

        new_h = h_var.value.copy()
        # Update the current h
        h.value = new_h

        # Update best solution
        current_c5 = evaluate(new_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()

    return best_h, best_c5, n_points

def evaluate(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
