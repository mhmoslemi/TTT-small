def run(seed=42, budget_s=1100, **kwargs):

    import numpy as np
    import math

    start_time = time.time()

    target_sum = K / 2.0  # since 1 = sum(h) * dx => sum(h) = 1 / dx * 1 => sum(h) = K/2
    np.random.seed(seed)
    # Initialize h as alternating 1 and 0
    h = np.zeros(K)
    h[::2] = 1.0  # even indices are 1, odd are 0
    h = h.copy()  # ensure it's a numpy array

    # Function to compute C5
    def compute_C5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_C5 = compute_C5(h)
    best_h = h.copy()

    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-4

    iteration = 0
    while time.time() - start_time < budget_s:
        for _ in range(100):  # perform 100 steps per iteration to save time
            if time.time() - start_time >= budget_s:
                break
            i = np.random.randint(0, K)
            j = np.random.randint(0, K)
            if h[i] == h[j]:
                continue  # only swap if different
            # Swap i and j
            h[[i,j]] = h[[j,i]]
            current_C5 = compute_C5(h)
            delta = current_C5 - best_C5
            if delta < 0:  # improvement
                best_C5 = current_C5
                best_h = h.copy()
            else:
                # Accept worse solution with a probability
                prob = math.exp(-delta / temperature)
                if np.random.random() > prob:
                    # Revert the swap
                    h[[i,j]] = h[[j,i]]
            iteration += 1
        temperature = max(temperature * cooling_rate, min_temperature)
    # Now validate
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

import time
import numpy as np
import math

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    K = 64  # Number of segments
    dx = 2.0 / K
    target_sum = K / 2.0  # ∑h = K/2 for ∫h dx = 1
    np.random.seed(seed)
    
    # Initialize with alternating 1s and 0s
    h = np.zeros(K)
    h[::2] = 1.0
    h = h.copy()

    def compute_C5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_C5 = compute_C5(h)
    best_h = h.copy()

    temperature = 1.0
    cooling_rate = 0.999
    min_temperature = 1e-4

    while time.time() - start_time < budget_s:
        for _ in range(100):
            if time.time() - start_time >= budget_s:
                break
            i, j = np.random.randint(0, K, 2)
            if h[i] == h[j]:
                continue  # Only swap different values
            h[[i, j]] = h[[j, i]]  # Swap
            current_C5 = compute_C5(h)
            delta = current_C5 - best_C5
            if delta < 0:
                best_C5 = current_C5
                best_h[:] = h[:]
            else:
                if temperature > 1e-20 and np.random.rand() < math.exp(-delta / temperature):
                    pass
                else:
                    h[[i, j]] = h[[j, i]]  # Undo swap
        temperature = max(temperature * cooling_rate, min_temperature)

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_C5, K)
