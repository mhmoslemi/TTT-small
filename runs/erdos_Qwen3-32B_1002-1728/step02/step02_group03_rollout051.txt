def evaluate(candidates):
    results = []
    for h in candidates:
        n = len(h)
        dx = 2.0 / n
        max_overlap = 0.0
        for k in range(n):
            upper = n - k
            overlap = 0.0
            for x in range(upper):
                overlap += h[x] * (1 - h[x + k])
            max_overlap = max(max_overlap, overlap * dx)
        results.append(max_overlap)
    return np.array(results)

def scale_to_sum(h, target_sum):
    s = np.sum(h)
    if abs(s) < 1e-10:
        return h
    scale = target_sum / s
    h_scaled = h * scale
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def compute_gradient(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        upper = i + k
        if upper < n:
            grad[i] += (1 - h[upper])
        lower = i - k
        if lower >= 0:
            grad[i] -= h[lower]
    return grad * dx
