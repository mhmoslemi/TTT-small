import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_initial = h_initial.copy()

    if not np.isclose(current_sum, target_sum, atol=1e-8):
        h_initial = np.clip(h_initial, 0, 1)

    constraints = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}

    start_time = time.time()
    maxiter = int(budget_s * 0.8)  # Empirical safety factor

    result = minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': 1e-8}
    )

    h_opt = result.x if result.success else h_initial
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (target_sum / h_opt.sum()) if h_opt.sum() != 0 else h_opt

    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(corr) * dx

    return h_opt, c5_bound, n_points
