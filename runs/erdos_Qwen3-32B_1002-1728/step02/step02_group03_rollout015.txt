    def evaluate(candidates):

            c5_values.append(np.max(corr))
        return np.array(c5_values)

    def scale_to_sum(h):

        if abs(s) < 1e-10:
            return h
        scale = target_sum / s
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, mutation_std, size=np.sum(mask))
        h = np.clip(h, 0, 1)
        h = scale_to_sum(h)
        return h

    def crossover(parent1, parent2):
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        child = scale_to_sum(child)
        return child

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 30
    generations = 1200
    mutation_rate = 0.15
    mutation_std = 0.08
    tournament_size = 4
    elitism_ratio = 0.15
    grad_steps = 15
    grad_lr = 0.03

    def evaluate(candidates):
        return np.array([np.max(np.correlate(h, 1 - h, 'full') * dx) for h in candidates])

    def scale_to_sum(h):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return h.copy()
        return np.clip(h * (target_sum / s), 0, 1)

    def mutate(h):
        h = h.copy()
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, mutation_std, size=np.sum(mask))
        return scale_to_sum(h)

    def crossover(p1, p2):
        alpha = np.random.rand()
        child = alpha * p1 + (1 - alpha) * p2
        return scale_to_sum(child)

    def tournament_select(pop, scores, size):
        indices = np.random.randint(len(pop), size=size)
        winner_idx = indices[np.argmin(scores[indices])]
        return pop[winner_idx]

    # Initialize population
    population = [h_initial]
    for _ in range(pop_size - 1):
        h = h_initial + np.random.normal(0, 0.05, n_points)
        h = scale_to_sum(np.clip(h, 0, 1))
        population.append(h)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = np.inf
    start_time = time.time()

    # Main GA loop
    while time.time() - start_time < budget_s - 60:
        scores = evaluate(population)
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h = population[best_idx].copy()

        elitism_size = int(elitism_ratio * pop_size)
        elites = population[np.argsort(scores)[:elitism_size]].copy()

        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, scores, tournament_size)
            p2 = tournament_select(population, scores, tournament_size)
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop = np.vstack([new_pop, child])
        population = new_pop[:pop_size]

    # Gradient refinement
    def compute_overlap(h):
        n = len(h)
        max_corr, k_max = 0, 0
        for k in range(n):
            indices = np.arange(n - k)
            corr = np.sum(h[indices] * (1 - h[indices + k])) * dx
            if corr > max_corr:
                max_corr, k_max = corr, k
        return k_max

    candidates = population[evaluate(population).argsort()][:5]
    for h in candidates:
        h_ref = h.copy()
        for _ in range(grad_steps):
            k_max = compute_overlap(h_ref)
            grad = np.zeros(n_points)
            for i in range(n_points - k_max):
                j = i + k_max
                grad[i] += (1 - h_ref[j]) * dx
                grad[j] -= h_ref[i] * dx
            h_ref -= grad * grad_lr
            h_ref = scale_to_sum(h_ref)
        corr_vals = np.correlate(h_ref, 1 - h_ref, 'full') * dx
        c5 = np.max(corr_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_ref.copy()

    # Final check
    corr = np.correlate(best_h, 1 - best_h, 'full') * dx
    return best_h, np.max(corr), n_points
