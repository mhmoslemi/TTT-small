def threshold_h(h, target_sum):
    # Binarize h to meet the target sum
    # Args:
    #     h: array of continuous values in [0,1]
    #     target_sum: target sum for binarized array
    # Returns:
    #     binarized array, same size as h
    current_sum = np.sum(h)
    if current_sum >= target_sum:
        # Binarize from above
        sorted_indices = np.argsort(-h)
        h_binary = np.zeros(len(h))
        # Determine how many 1s to set
        count = 0
        while current_sum > target_sum and count < len(h):
            h_binary[sorted_indices[count]] = 1
            current_sum -= h[sorted_indices[count]]
            count += 1
        # Adjust the last one if necessary
        if count > 0 and current_sum < target_sum:
            excess = target_sum - (len(h_binary) - count)
            if excess > 0:
                h_binary[sorted_indices[count - 1]] = excess
            else:
                h_binary[sorted_indices[count - 1]] = 1
        return h_binary
    else:
        # Binarize from below
        sorted_indices = np.argsort(h)
        h_binary = np.zeros(len(h))
        count = 0
        while current_sum < target_sum and count < len(h):
            h_binary[sorted_indices[count]] = 1
            current_sum += (1 - h[sorted_indices[count]])
            count += 1
        if count > 0 and current_sum > target_sum:
            underflow = current_sum - target_sum
            h_binary[sorted_indices[count - 1]] = 1 - underflow
        return h_binary

import time
import numpy as np
from scipy.optimize import minimize

def threshold_h(h_vec, target_sum):
    h = h_vec.reshape(-1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-6:
        return h_vec.copy()
    sorted_indices = np.argsort(h)
    binarized = np.zeros_like(h)
    count = 0
    while count < len(h) and np.sum(binarized) < target_sum:
        i = sorted_indices[-count - 1]
        binarized[i] = 1.0
        count += 1
    if np.sum(binarized) > target_sum:
        count -= 1
        binarized[sorted_indices[-count - 1]] = 0.0
        last_idx = sorted_indices[-count - 1]
        binarized[last_idx] = target_sum - np.sum(binarized)
    return binarized.flatten()

def evaluate(h_vec):
    corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    lngs = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        return evaluate(h_vec) * dx

    # Initial optimization
    result = minimize(
        fun=objective,
        x0=h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=lngs,
        options={'maxiter': 300, 'ftol': 1e-9}
    )
    best_h = result.x.copy()
    best_c5 = objective(best_h)

    # Alternating threshold and optimize
    iter_count = 0
    while time.time() - start_time < budget_s * 0.75 and iter_count < 3:
        thresholded_h = threshold_h(best_h, target_sum)
        perturbed_h = thresholded_h + np.random.RandomState(seed + iter_count).normal(0, 0.01, n_points)
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        scale_factor = target_sum / np.sum(perturbed_h)
        perturbed_h *= scale_factor

        result_alt = minimize(
            fun=objective,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=lngs,
            options={'maxiter': 200, 'ftol': 1e-9}
        )
        if result_alt.success:
            new_h = np.clip(result_alt.x, 0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
        iter_count += 1

    # Local search refinement
    while time.time() - start_time < budget_s * 0.95:
        perturbed = best_h + np.random.RandomState(seed + int(time.time())).normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if abs(perturbed_sum - target_sum) > 1e-6:
            scale = target_sum / perturbed_sum
            perturbed *= scale

        result_rs = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=lngs,
            options={'maxiter': 150, 'ftol': 1e-9}
        )
        if result_rs.success:
            h_candidate = np.clip(result_rs.x, 0, 1)
            candidate_sum = np.sum(h_candidate)
            if abs(candidate_sum - target_sum) < 1e-6:
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.98:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
