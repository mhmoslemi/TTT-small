def run(seed=42, budget_s=1100, **kwargs):

    h_initial = initial_h_values.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    n_particles = 80
    max_iter = 800
    omega = 0.65
    phi_c = 1.2
    phi_s = 1.8
    v_max = 0.1
    local_search_top = 3

    def rescale_particle(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if np.isclose(s, target_sum):
            return h_clipped
        delta = target_sum - s
        if delta > 0:
            under_mask = h_clipped < 1
            idx = np.where(under_mask)[0]
            if len(idx) == 0:
                return h_clipped
            increment = delta / len(idx)
            h_clipped[under_mask] += increment
            h_clipped = np.clip(h_clipped, 0, 1)
        else:
            over_mask = h_clipped > 0
            idx = np.where(over_mask)[0]
            if len(idx) == 0:
                return h_clipped
            decrement = -delta / len(idx)
            h_clipped[over_mask] -= decrement
            h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    particles = np.tile(h_initial, (n_particles, 1)).copy()
    np.random.seed(seed + 1)
    particles += 0.05 * np.random.randn(n_particles, n_points)
    for i in range(n_particles):
        particles[i] = rescale_particle(particles[i])

    velocities = np.zeros_like(particles)
    personal_best = particles.copy()
    personal_best_scores = np.array([np.max(np.correlate(h, 1-h, "full") * dx) for h in particles])
    best_idx = np.argmin(personal_best_scores)
    global_best = particles[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]

    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i]
                             + phi_c * r1 * (personal_best[i] - particles[i])
                             + phi_s * r2 * (global_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
