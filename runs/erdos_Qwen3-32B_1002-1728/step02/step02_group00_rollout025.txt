import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_init.copy()
    best_c5 = objective(best_h)
    np.random.seed(seed)

    # Generate multiple perturbed starts
    num_workers = 5
    for worker_id in range(num_workers):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Create perturbed initialization
        perturb_scale = 0.05
        rs = np.random.RandomState(seed + worker_id)
        perturbed = best_h + rs.normal(0, perturb_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = (perturbed / perturbed_sum) * target_sum if perturbed_sum != 0 else best_h

        # COBYLA optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'tol': 1e-8}
        )

        if result.success:
            h_candidate = result.x.reshape(n_points)
            if abs(np.sum(h_candidate) - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
