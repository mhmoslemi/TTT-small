import numpy as np
import time

def evaluate_c5(h, dx, target_sum):
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total < 1e-6:
        return float('inf')
    h = h * (target_sum / total)
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is valid
    h = np.clip(h, 0, 1).copy()
    total = h.sum()
    h = h * (target_sum / total)
    h = np.clip(h, 0, 1).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx, target_sum)

    # Annealing parameters
    temp = 0.1
    alpha = 0.99
    max_iter_per_temp = 100
    perturbation_scale = 0.1

    # Random state
    np.random.seed(seed)
    rnd = np.random.RandomState(seed)

    while time.time() - start_time < budget_s - 60:
        for _ in range(max_iter_per_temp):
            # Perturb
            noise = perturbation_scale * temp * rnd.randn(n_points)
            new_h = best_h + noise
            new_h = np.clip(new_h, 0, 1)
            total = new_h.sum()
            if total > 1e-6:
                new_h = new_h * (target_sum / total)
            else:
                new_h = np.full(n_points, target_sum / n_points)

            # Evaluate
            new_c5 = evaluate_c5(new_h, dx, target_sum)
            delta = new_c5 - best_c5

            # Acceptance criteria
            if delta <= 0 or rnd.rand() < np.exp(-delta / temp):
                best_h = new_h.copy()
                best_c5 = new_c5

        # Cool down
        temp *= alpha

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    total = best_h.sum()
    best_h = best_h * (target_sum / total)
    best_h = np.clip(best_h, 0, 1)
    assert abs(best_h.sum() - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
