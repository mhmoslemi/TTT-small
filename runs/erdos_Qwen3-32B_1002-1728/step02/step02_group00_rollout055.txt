import time
import numpy as np

def compute_c5(h, K, dx):
    g = 1.0 - h
    cross_corr = np.correlate(h, g, mode='full')
    sum1_values = cross_corr[K:2*K-1]
    max_total = 0.0
    for k in range(1, K):
        current_sum1 = sum1_values[k-1]
        current_sum2 = np.sum(h[K - k:])
        total = (current_sum1 + current_sum2) * dx
        if total > max_total:

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    K = 64
    dx = 2.0 / K
    np.random.seed(seed)
    h = np.tile([1, 0], K // 2)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, K, dx)
    T = 1.0
    T_final = 1e-4
    cooling_rate = 0.995
    while time.time() - start_time < budget_s * 0.95:
        indices_1 = np.where(h == 1)[0]
        indices_0 = np.where(h == 0)[0]

            if np.random.rand() < prob:
