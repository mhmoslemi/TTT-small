import numpy as np
import time

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def ensure_feasible(candidate, n_points, required_sum):
    candidate = np.clip(candidate, 0, 1)
    current_sum = np.sum(candidate)
    if current_sum > 1e-10:
        scaling = required_sum / current_sum
        candidate *= scaling
        candidate = np.clip(candidate, 0, 1)
    if abs(np.sum(candidate) - required_sum) > 1e-8:
        diff = required_sum - np.sum(candidate)
        candidate[0] += diff
        candidate = np.clip(candidate, 0, 1)
    return candidate

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    np.random.seed(seed)
    h = ensure_feasible(h_initial, n_points, required_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    alpha = 0.005
    decay_rate = 0.95
    decay_steps = 30
    perturb_interval = 150
    perturb_strength = 0.05
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        n = n_points
        lag = max_idx - (n - 1)
        k = lag

        grad = np.zeros(n)
        for m in range(n):
            shifted_plus = (m + k) % n
            shifted_minus = (m - k) % n
            grad[m] = (1.0 - h[shifted_plus]) - h[shifted_minus]
