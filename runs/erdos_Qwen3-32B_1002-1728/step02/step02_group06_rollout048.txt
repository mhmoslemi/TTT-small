            h[0] = 1.0
            curr_sum = 1.0
        h = np.clip(h * (target_sum / curr_sum), 0, 1)
        delta = target_sum - h.sum()
        h[0] += delta
        h = np.clip(h, 0, 1)
        return h

    def generate_periodic(p, n_points):
        pattern = np.zeros(n_points)
        half_p = p // 2
        for i in range(n_points):
            if i % p < half_p:
                pattern[i] = 1.0
        return pattern

    population_size = 30
    population = [h_initial.copy()]
    for p in [2, 4, 6, 8, 10]:
        population.append(generate_periodic(p, n_points))
    while len(population) < population_size:
        population.append(generate_random_candidate(n_points, target_sum))

    best_c5 = np.inf
    best_h = None
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        fitness = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            fitness.append(np.max(corr))
        idx = np.argsort(fitness)
        sorted_pop = [population[i] for i in idx]
        if fitness[idx[0]] < best_c5:
            best_c5 = fitness[idx[0]]
            best_h = sorted_pop[0].copy()
        num_parents = int(0.2 * population_size)
        parents = sorted_pop[:num_parents]
        new_offspring = []
        for _ in range(num_parents):
            p1, p2 = np.random.choice(parents, 2, replace=True)
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * p1 + (1 - alpha) * p2
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            curr_sum = child.sum()
            delta = target_sum - curr_sum
            child[0] += delta
            child = np.clip(child, 0, 1)
            new_offspring.append(child)
        population[-num_parents:] = new_offspring

    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr)
    return best_h, c5_bound, n_points
