import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):

        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()
    
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()

        x0_perturbed = best_h.copy() + noise_level * np.random.randn(n_points)

        x0_perturbed = adjust_sum(x0_perturbed, target_sum)
        
        result = minimize(
