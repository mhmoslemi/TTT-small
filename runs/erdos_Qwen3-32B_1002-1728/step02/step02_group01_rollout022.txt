import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h_mat = h_vec.reshape(n_points)
        corr = np.correlate(h_mat, 1 - h_mat, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    start_time = time.time()
    population = []
    population.append(h.copy())

    def generate_gaussian_perturbation(h_base, target_sum):
        h = h_base.copy()
        h += np.random.normal(0, 0.05, size=h.shape)
        h = np.clip(h, 0, 1)
        total = h.sum()
        h = h * target_sum / total if total else h
        h = np.clip(h, 0, 1)
        return h

    def generate_boundary_jitter(h_base, target_sum):
        h = h_base.copy()
        indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        for i in indices:
            h[i] = np.random.choice([0.0, 1.0])
        total = h.sum()
        h = h * target_sum / total if total else h
        h = np.clip(h, 0, 1)
        return h

    def generate_block_perturbation(h_base, target_sum):
        h = h_base.copy()
        blocks = np.split(h, [35, 70])[:3]
        for i, blk in enumerate(blocks):
            if blk.sum() == 0:
                continue
            new_blk = np.random.rand(len(blk))
            new_blk = new_blk * blk.sum() / new_blk.sum()
            h[len(np.concatenate(blocks[:i])):len(np.concatenate(blocks[:i+1]))] = new_blk
        total = h.sum()
        h = h * target_sum / total if total else h
        h = np.clip(h, 0, 1)
        return h

    # Generate diverse candidates
    for _ in range(10): population.append(generate_gaussian_perturbation(h, target_sum))
    for _ in range(5): population.append(generate_boundary_jitter(h, target_sum))
    for _ in range(5): population.append(generate_block_perturbation(h, target_sum))

    # Optimization loop
    for candidate in population:
        if time.time() - start_time >= budget_s - 60:
            break
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-9}
        )
        if result.success:
            new_h = result.x.copy()
            new_h = np.clip(new_h, 0, 1)
            total = new_h.sum()
            new_h = new_h * target_sum / total if total else new_h
            new_c5 = objective(new_h)
            if new_c5 < best_c5 and np.all(np.isfinite(new_h)):
                best_h[:] = new_h
                best_c5 = new_c5

    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
