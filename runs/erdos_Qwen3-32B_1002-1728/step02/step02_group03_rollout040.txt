        positions.append(h)

    velocities = np.zeros_like(positions)

    # Personal best initialization
    personal_best_positions = positions.copy()
    personal_best_scores = np.full(pop_size, np.inf)
    for i in range(pop_size):
        personal_best_scores[i] = compute_c5(positions[i], n_points)

    # Global best initialization
    global_best_idx = np.argmin(personal_best_scores)
    global_best_score = personal_best_scores[global_best_idx]
    global_best_position = personal_best_positions[global_best_idx].copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Generate random components for velocity update
        r1 = np.random.rand(pop_size, 1)
        r2 = np.random.rand(pop_size, 1)

        # Cognitive and social components
        cognitive = r1 * (personal_best_positions - positions)
        social = r2 * (global_best_position[None, :] - positions)

        # Update velocities
        velocities = omega * velocities + phi_c * cognitive + phi_s * social
        velocities = np.clip(velocities, -velocity_clamp, velocity_clamp)

        # Update positions
        positions += velocities
        positions = np.clip(positions, 0, 1)

        # Rescale each particle to maintain sum constraint
        for i in range(pop_size):
            positions[i] = rescale_to_sum(positions[i], target_sum)

        # Evaluate new positions
        for i in range(pop_size):
            new_score = compute_c5(positions[i], n_points)
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_best_positions[i] = positions[i].copy()

        # Update global best
        current_global_idx = np.argmin(personal_best_scores)
        current_global_score = personal_best_scores[current_global_idx]
        if current_global_score < global_best_score:
            global_best_score = current_global_score
            global_best_position = personal_best_positions[current_global_idx].copy()

    final_c5 = compute_c5(global_best_position, n_points)
    return (global_best_position, final_c5, n_points)
