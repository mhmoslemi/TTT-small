            offsprings.append(mutate(np.array(c2)))

        offsprings = [o for o in offsprings if o is not None]
        off_scores = np.array([evaluate(o) for o in offsprings])
        # Combine and select top
        combined = np.vstack([pop, offsprings])
        combined_scores = np.concatenate([scores, off_scores])
        indices = argsort(combined_scores)[:population_size]
        pop = combined[indices]
        scores = combined_scores[indices]

        new_idx = argmin(scores)
        if scores[new_idx] < best_c5:
            best_c5 = scores[new_idx]
            best_sol = pop[new_idx].copy()

    return best_sol, best_c5, n_points
