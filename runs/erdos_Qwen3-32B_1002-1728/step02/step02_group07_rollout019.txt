import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h = h.ravel()
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    sum_constraint = LinearConstraint(np.ones(n_points), lb=target_sum, ub=target_sum)
    bounds = Bounds(0.0, 1.0)

    # Normalize initial to satisfy sum constraint
    h_initial = np.clip(h_initial, 0, 1)
    h_initial *= target_sum / h_initial.sum()
    h_initial = np.clip(h_initial, 0, 1)
    x0 = h_initial.copy()

    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()

    while time.time() - start_time < budget_s - 5:
        # Perturb best_h to create new starting point
        perturbed = best_h + 0.02 * np.random.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        scale = target_sum / perturbed.sum()
        x0_perturbed = np.clip(perturbed * scale, 0, 1)

        # Optimize from perturbed point
        result = minimize(
            objective,
            x0=x0_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 150, 'ftol': 1e-7}
        )

        # Enforce constraints on final solution
        curr_h = np.clip(result.x, 0, 1)
        curr_h *= target_sum / curr_h.sum()
        curr_h = np.clip(curr_h, 0, 1)

        # Evaluate objective
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)

        # Update best solution
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()

        # Reschedule perturbation for next restart
        if time.time() - start_time > budget_s - 20:
            break

    return best_h, best_c5, n_points
