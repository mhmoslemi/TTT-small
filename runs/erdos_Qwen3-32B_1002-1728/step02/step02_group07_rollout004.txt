import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if can_incr.any():
            total_possible = (1.0 - h_adj[can_incr]).sum()
            h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if can_decr.any():
            total_possible = h_adj[can_decr].sum()
            h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr_vals)

    population_size = 30
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = h_initial.copy() + 0.05 * np.random.randn(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = adjust_sum(h, target_sum)
        population.append(h)

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()

    while time.time() - start_time < budget_s - 20:
        population.sort(key=compute_c5)
        top_fraction = 0.33
        top_idx = max(1, int(population_size * top_fraction))
        selected = population[:top_idx]

        offspring = []
        sigma = 0.15 * (1 - (time.time() - start_time) / budget_s)
        for h in selected:
            for _ in range(3):
                h_new = h + sigma * np.random.randn(n_points)
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = adjust_sum(h_new, target_sum)
                offspring.append(h_new)

        combined = population + offspring
        combined.sort(key=compute_c5)
        population = combined[:population_size]

        current_c5 = compute_c5(population[0])
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = population[0].copy()

    return best_h, best_c5, n_points
