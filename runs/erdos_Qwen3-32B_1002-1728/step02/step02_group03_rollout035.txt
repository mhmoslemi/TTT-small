        current_c5 = np.max(corr)

            best_c5 = current_c5
            best_h = h.copy()

        top_indices = np.argsort(corr)[-max_top_shifts:]
        grad_dir = compute_top_gradients(h, top_indices, dx)
        h = h - alpha * grad_dir
        h = np.clip(h, 0, 1)

        h_sum = np.sum(h)
        scale = target_sum / h_sum if abs(h_sum) > 1e-10 else 1.0
        h = h * scale
        h = np.clip(h, 0, 1)

        alpha *= alpha_decay

    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(final_corr), n_points
