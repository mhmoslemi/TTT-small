import numpy as np
import time

def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    for i in range(n_points):
        if i % p < p // 2:
            pattern[i] = 1.0
    return pattern

def crossover(parent1, parent2):
    alpha = np.random.uniform(0.3, 0.7)
    child = alpha * parent1 + (1 - alpha) * parent2
    n_points = len(child)
    child = np.clip(child, 0, 1)
    current_sum = child.sum()
    if abs(current_sum - n_points / 2.0) > 1e-8:
        child[0] += (n_points / 2.0 - current_sum)
        child = np.clip(child, 0, 1)
    return child

def mutate(h, sigma=0.05):
    mutated = h + np.random.normal(0, sigma, len(h))
    mutated = np.clip(mutated, 0, 1)
    n_points = len(mutated)
    current_sum = mutated.sum()
    if abs(current_sum - n_points / 2.0) > 1e-8:
        mutated[0] += (n_points / 2.0 - current_sum)
        mutated = np.clip(mutated, 0, 1)
    return mutated

def evaluate_fitness(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    population_size = 50
    population = []
    population.append(h_initial.copy())
    
    p_list = [2, 4, 6, 8, 10]
    for p in p_list:
        pattern = generate_periodic_pattern(p, n_points)
        population.append(pattern)
    
    while len(population) < population_size:
        new_h = np.random.rand(n_points)
        new_h = np.clip(new_h, 0, 1)
        new_sum = new_h.sum()
        new_h[0] += max(0, (target_sum - new_sum))
        new_h[0] = min(1.0, new_h[0])
        population.append(new_h)

    fitness_values = [evaluate_fitness(ind, n_points) for ind in population]
    best_idx = np.argmin(fitness_values)
    best_h = population[best_idx].copy()
    best_c5 = fitness_values[best_idx]

    while time.time() - start_time < budget_s - 10:
        parent1 = tournament_selection(population, fitness_values, tournament_size=3)
        parent2 = tournament_selection(population, fitness_values, tournament_size=3)
        child = crossover(parent1, parent2)
        if np.random.rand() < 0.1:
            child = mutate(child)
        child_c5 = evaluate_fitness(child, n_points)
        
        worst_idx = np.argmax(fitness_values)
        if child_c5 < fitness_values[worst_idx]:
            population[worst_idx] = child
            fitness_values[worst_idx] = child_c5
            if child_c5 < best_c5:
                best_c5 = child_c5
                best_h = child.copy()
    
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n_points)
    return best_h, np.max(corr_values), n_points

def tournament_selection(population, fitness_values, tournament_size=3):
    indices = np.random.randint(0, len(population), tournament_size)
    best_idx = indices[0]
    for idx in indices[1:]:
        if fitness_values[idx] < fitness_values[best_idx]:

    return population[best_idx].copy()
