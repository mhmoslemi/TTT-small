import time
from scipy.optimize import minimize
from scipy.signal import correlate

def _adjust_sum(h, target_sum):
    h = h.copy()
    delta = np.sum(h) - target_sum
    if abs(delta) < 1e-10:
        return h
    h += -delta / len(h)
    h = np.clip(h, 0, 1)
    return _project(h, target_sum, len(h))

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    return np.clip(clamped * (target_sum / clamped_sum), 0, 1)
