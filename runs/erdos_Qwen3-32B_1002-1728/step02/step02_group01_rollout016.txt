        total = np.sum(perturbed)
        if total > 1e-6:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            objective_phi,
            perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        if result.success:
            new_h = result.x.reshape(n_points)
            new_h = np.clip(new_h, 0, 1)
            total_new = np.sum(new_h)
            if total_new < 1e-6:
                continue
            new_h = (new_h / total_new) * target_sum
            new_h = np.clip(new_h, 0, 1)
            corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(corr) * dx
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
