def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    h_init = np.array(initial_h_values).copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    np.random.seed(seed)

    pop_size = 25
    population = [h_init.copy()]

    # Generate initial population
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1.0)
        s = h.sum()
        if s < 1e-8:
            h[0] = 1.0
            s = h.sum()
        h = (h / s) * target_sum
        h = np.clip(h, 0, 1.0)
        population.append(h)

    best_c5 = np.inf
    best_h = None

    while time.time() - t0 < budget_s - 60:
        # Evaluate fitness
        fitness = []
        for h in population:
            cor = np.correlate(h, 1 - h, mode="full")
            val = np.max(cor) * dx
            fitness.append(val)

        # Update best solution
        idx = np.argmin(fitness)
        current_c5 = fitness[idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = population[idx].copy()
            assert abs(np.sum(best_h) - target_sum) < 1e-6
            assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

        # Selection
        top_count = int(pop_size * 0.2)
        top_indices = np.argsort(fitness)[:top_count]
        top_candidates = [population[i] for i in top_indices]

        new_population = top_candidates.copy()

        # Generate new individuals via mutation or crossover
        while len(new_population) < pop_size:
            if np.random.rand() < 0.5:  # Mutation
                parent = top_candidates[np.random.randint(len(top_candidates))]
                noise = np.random.normal(0, 0.05, n_points)
                child = parent + noise
                child = np.clip(child, 0, 1.0)
                s = child.sum()
                if s < 1e-8:
                    child = np.zeros_like(child)
                    child[0] = target_sum
                else:
                    child = (child / s) * target_sum
                child = np.clip(child, 0, 1.0)
                new_population.append(child)
            else:  # Crossover
                p1 = top_candidates[np.random.randint(len(top_candidates))]
                p2 = top_candidates[np.random.randint(len(top_candidates))]
                alpha = np.random.rand()
                child = alpha * p1 + (1.0 - alpha) * p2
                child = np.clip(child, 0, 1.0)
                s = child.sum()
                if s < 1e-8:
                    child = np.zeros_like(child)
                    child[0] = target_sum
                else:
                    child = (child / s) * target_sum
                child = np.clip(child, 0, 1.0)
                new_population.append(child)

        population = new_population

    return (best_h.tolist(), best_c5, n_points)
