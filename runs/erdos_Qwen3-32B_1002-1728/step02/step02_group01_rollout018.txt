import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 25
    pop = [h_init.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        h = pop[i].copy()
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total > 1e-8:
            h *= target_sum / total
        else:
            h.fill(target_sum / n_points)
        pop[i] = h

    best_h = h_init.copy()
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        new_pop = []

        # Evaluate and select elites
        scores = []
        for h in pop:
            g = 1 - h
            len_corr = 2 * n_points - 1
            corr = np.fft.ifft(np.fft.fft(h, len_corr) * np.conj(np.fft.fft(g, len_corr))).real
            max_val = np.max(corr) * dx
            scores.append(max_val)
        elite_indices = np.argsort(scores)[:max(1, pop_size // 4)]
        elites = [pop[i].copy() for i in elite_indices]

        # Fill new population with mutations/cross
        while len(new_pop) < pop_size - len(elites):
            a, b = np.random.choice(pop_size, 2)
            p = np.random.rand()
            child = p * pop[a] + (1-p) * pop[b]
            noise = np.random.normal(0, 0.05, n_points)
            child += noise
            child = np.clip(child, 0, 1)
            total = np.sum(child)
            if total > 1e-8:
                child *= target_sum / total
            else:
                child.fill(target_sum / n_points)
            child = np.clip(child, 0, 1)
            new_pop.append(child)

        # Track global best
        scores = np.array(scores)
        if len(elite_indices) > 0:
            current_best = scores[elite_indices[0]]
            if current_best < best_c5:
                best_c5 = current_best
                best_h = pop[elite_indices[0]].copy()

        # Update population
        pop = elites + new_pop

    # Final checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
