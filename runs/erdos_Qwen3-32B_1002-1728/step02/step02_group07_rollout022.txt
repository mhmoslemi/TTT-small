                h_mut = np.clip(h_mut, 0, 1)
                h_mut = adjust_sum(h_mut, target_sum)
                new_offspring.append(h_mut)
                new_offspring_c5.append(evaluate_c5(h_mut, dx))
        
        combined_pop = np.vstack([population, np.array(new_offspring)])
        combined_c5 = np.concatenate([c5_values, new_offspring_c5])
        elite_indices = np.argsort(combined_c5)[:pop_size]
        population = combined_pop[elite_indices]
        c5_values = combined_c5[elite_indices]
        
        if np.min(c5_values) < best_c5:
            best_c5 = np.min(c5_values)
            best_h = population[np.argmin(c5_values)].copy()
            
        if time.time() - start_time > 700 and pop_size < 40:
            pop_size = min(40, pop_size + 1)
            population = np.vstack([population, generate_random_h()])
            c5_values = np.append(c5_values, evaluate_c5(population[-1], dx))
    
    return best_h, best_c5, n_points
