import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    h = h * (target_sum / (current_sum + 1e-12))
    best_h = h.copy()
    
    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    best_c5 = objective(best_h)
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-6
    np.random.seed(seed)
    
    while time.time() - start_time < budget_s - 10:  # Reserve time buffer
        # Perturb solution
        noise_scale = 0.1 * T
        perturbed = best_h + np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        
        # Normalize sum
        total = perturbed.sum()
        if np.isclose(total, 0):
            perturbed = np.zeros_like(perturbed)
        else:
            perturbed = perturbed * (target_sum / (total + 1e-12))
        perturbed = np.clip(perturbed, 0, 1)  # Re-clip after normalization
        
        new_c5 = objective(perturbed)
        delta_c5 = new_c5 - best_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            best_h = perturbed.copy()
            best_c5 = new_c5
        
        # Cool temperature
        T *= alpha
        if T < min_T:
            T = min_T

    # Final validation

    return (best_h.tolist(), best_c5, n_points)
