def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time();

    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Genetic algorithm parameters
    pop_size = 25
    elite_frac = 0.2
    mutation_strength = 0.05

    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        candidate = np.random.rand(n_points)
        candidate = process_individual(candidate, target_sum, n_points)
        population.append(candidate)
    population = np.array(population)

    best_h = h.copy()
    best_c5 = evaluate_C5(best_h, dx)

    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitness = np.array([evaluate_C5(ind, dx) for ind in population])
        ordered = np.argsort(fitness)
        current_best_idx = ordered[0]
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]

        # Select elites
        n_elite = int(elite_frac * pop_size)
        elites = population[ordered[:n_elite]]

        # Generate new population
        new_pop = [elite for elite in elites]
        top_half_idx = ordered[:int(pop_size * 0.5)]

        while len(new_pop) < pop_size:
            # Crossover
            p1_idx, p2_idx = np.random.choice(top_half_idx, 2, replace=False)
            child = 0.5 * population[p1_idx] + 0.5 * population[p2_idx]
            child = process_individual(child, target_sum, n_points)
            new_pop.append(child)

            # Mutation
            parent_idx = np.random.choice(top_half_idx, 1)[0]
            parent = population[parent_idx].copy()
            noise = mutation_strength * np.random.randn(n_points)
            mutated = parent + noise
            mutated = process_individual(mutated, target_sum, n_points)
            new_pop.append(mutated)

        # Truncate and shuffle
        np.random.shuffle(new_pop)
        population = np.array(new_pop[:pop_size])

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
