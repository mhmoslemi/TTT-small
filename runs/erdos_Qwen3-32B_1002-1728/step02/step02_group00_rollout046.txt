import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Step 1: Continuous optimization from initial_h_values
    if time.time() - start_time < budget_s * 0.8:
        res_continuous = minimize(
            fun=objective,
            x0=h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        h_continuous = res_continuous.x if res_continuous.success else h_init
    else:
        h_continuous = h_init.copy()

    # Step 2: Binarize continuous solution
    indices = np.argsort(h_continuous)[::-1]
    h_binary = np.zeros(n_points)
    h_binary[indices[:n_points // 2]] = 1.0

    # Step 3: Perturb and optimize from binarized h
    best_h = h_binary.copy()
    corr_vals = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr_vals) * dx

    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed + int(time.time()))
        perturbed = best_h + rs.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed = (perturbed / perturbed_sum) * target_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if np.allclose(sum_candidate, target_sum) and np.all((0 <= h_candidate) & (h_candidate <= 1)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
        if time.time() - start_time > budget_s * 0.95:
            break

    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
