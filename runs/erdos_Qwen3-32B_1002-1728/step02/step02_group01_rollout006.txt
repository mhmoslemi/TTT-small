import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        # Generate a new RandomState with a seed-dependent perturbation
        rnd = np.random.RandomState(seed + n_attempt)
        # Perturb with increased magnitude
        perturbed = best_h + rnd.normal(0, 0.15, n_points)  # Larger sigma
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        # Stochastic proximal smoothing with dynamic adaptivity
        @minimize
        def adaptive_objective(h_vec):
            h = h_vec.reshape(n_points)
            overlap = np.correlate(h, 1 - h, mode='full')
            # Adaptive weighting based on overlap deviation
            weights = np.exp(np.maximum(0, overlap - np.mean(overlap)))
            weighted = np.max(weights * overlap)
            return (weighted / np.max(weights)) * dx if np.max(weights) > 0 else 0.0

        # Smaller step for intra-optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
        )

        if result and result.success:
            new_h = result.x.reshape(n_points)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
