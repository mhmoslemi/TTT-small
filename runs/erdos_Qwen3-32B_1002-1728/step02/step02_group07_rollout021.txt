def run(seed=42, budget_s=1100, **kwargs):

    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2
    dx = 2.0 / n_points

    # Define adjust_sum as per original code
    def adjust_sum(h, target_sum):
        h_adj = h.copy()
        current_sum = h_adj.sum()
        D = target_sum - current_sum
        if abs(D) < 1e-9:
            return h_adj
        if D > 0:
            can_incr = (h_adj < 0.9999)
            if not can_incr.any():
                return h_adj
            total_possible = (1.0 - h_adj[can_incr]).sum()
            h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
        else:
            D = -D
            can_decr = (h_adj > 1e-4)
            if not can_decr.any():
                return h_adj
            total_possible = h_adj[can_decr].sum()
            h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
        return h_adj

    population_size = 20  # initial population size
    population = []

    # Initialize population
    population.append(h_initial.copy())
    np.random.seed(seed)
    for _ in range(population_size - 1):
        h_rand = np.random.rand(n_points)
        h_rand = adjust_sum(h_rand, target_sum)
        # Since adjust_sum keeps it in [1e-4, 0.9999], clip to [0,1]
        h_rand = np.clip(h_rand, 0.0, 1.0)
        population.append(h_rand)

    # Now evaluate each individual in the population
    def evaluate_individual(h):
        h_eval = np.clip(h, 0.0, 1.0)
        h_eval = adjust_sum(h_eval, target_sum)
        corr = np.correlate(h_eval, 1 - h_eval, mode='full') * dx
        return np.max(corr)

    # Evaluate all individuals
    population = [adjust_sum(np.clip(ind, 0.0, 1.0), target_sum) for ind in population]
    scores = [evaluate_individual(ind) for ind in population]

    # Track best solution
    best_idx = np.argmin(scores)
    best_c5 = scores[best_idx]
    best_h = population[best_idx].copy()

    # Start timer
    start_time = time.time()
    while time.time() - start_time < budget_s - 5:  # leave 5s buffer
        # Select top individuals for mutation
        num_selected = max(1, int(population_size * selection_fraction))
        selected_indices = np.argsort(scores)[:num_selected]

        new_population = []
        for idx in selected_indices:
            parent = population[idx]
            # Mutate to create 3 new individuals
            for _ in range(3):  # each parent creates 3 offspring
                # Mutation step
                sigma = mutation_std * (1.0 - (time.time() - start_time) / budget_s)  # decay mutation strength
                noise = np.random.normal(0, sigma, n_points)
                child = parent + noise
                child = np.clip(child, 0.0, 1.0)
                child = adjust_sum(child, target_sum)
                new_population.append(child)

        # Evaluate new population
        new_scores = [evaluate_individual(ind) for ind in new_population]
        # Combine old and new
        combined = population + new_population
        combined_scores = scores + new_scores

        # Select the top population_size individuals
        combined = [combined[i] for i in np.argsort(combined_scores)[:population_size]]
        combined_scores = np.sort(combined_scores)[:population_size]

        # Update population
        population = combined
        scores = combined_scores

        # Update best solution
        current_best_idx = np.argmin(scores)
        current_best_c5 = scores[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()

        # Increase population if time allows
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 600:  # if over 10 mins left, increase population
            population_size = min(population_size + 5, 50)

    return best_h, best_c5, n_points
