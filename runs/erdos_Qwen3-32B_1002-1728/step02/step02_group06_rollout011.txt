import numpy as np
from scipy.optimize import Bounds
import time

def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def repair(individual, target_sum):
    individual = np.clip(individual, 0.0, 1.0)
    curr_sum = np.sum(individual)
    if curr_sum == 0 or curr_sum == n_points:
        scale = target_sum / curr_sum
        individual = individual * scale
        individual = np.clip(individual, 0.0, 1.0)
    else:
        scale = target_sum / curr_sum
        individual = individual * scale
        individual = np.clip(individual, 0.0, 1.0)
    diff = np.sum(individual) - target_sum
    individual[0] += diff
    individual = np.clip(individual, 0.0, 1.0)
    return individual

def objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, fitness, tournament_size=3):
    n_parents = max(1, len(population) // 5)
    indices = np.random.choice(len(population), size=tournament_size * n_parents)
    indices = indices.reshape((n_parents, tournament_size))
    parents = []
    for t_indices in indices:
        tournament = [population[i] for i in t_indices]
        f_tournament = [fitness[i] for i in t_indices]
        winner_index = np.argmin(f_tournament)
        parents.append(tournament[winner_index])
    return np.array(parents)

def crossover(parents, crossover_rate=0.7, alpha_range=(0.3, 0.7)):
    n_offspring = len(parents)
    offspring = []
    for i in range(n_offspring):
        p1, p2 = parents[np.random.randint(len(parents))], parents[np.random.randint(len(parents))]
        if np.random.rand() < crossover_rate:
            alpha = np.random.uniform(*alpha_range)
            child = alpha * p1 + (1 - alpha) * p2
            child = repair(child, target_sum)
        else:
            child = p1.copy()
        offspring.append(child)
    return offspring

def mutate(individual, mutation_step, mutation_rate=0.1):
    if np.random.rand() < mutation_rate:
        individual = individual + np.random.normal(0, mutation_step, size=individual.shape)
        individual = np.clip(individual, 0.0, 1.0)
    return individual

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values, n_points
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    pop_size = 50
    mutation_step = 0.05
    mutation_rate = 0.1
    crossover_rate = 0.7
    stagnation_counter = 0
    max_stagnation = 10
    last_best = float('inf')

    def evaluate(individual):
        return objective(individual, dx)

    # Initialize population
    population = [h_initial]
    p_list = [2, 4, 6, 8, 10]
    for p in p_list:
        population.append(generate_periodic_pattern(p, n_points))
    while len(population) < pop_size:
        # Generate random candidate with sum constraint
        candidate = np.random.uniform(0, 1, n_points)
        candidate = repair(candidate, target_sum)
        population.append(candidate)

    population = np.array(population)
    best_h = h_initial.copy()
    best_c5 = objective(best_h, dx)
    start_time = time.time()

    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = np.array([evaluate(ind) for ind in population])
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx].copy()
        current_best_c5 = fitness[current_best_idx]

        if current_best_c5 < best_c5:
            best_h = current_best.copy()

        parents = tournament_selection(population, fitness)
