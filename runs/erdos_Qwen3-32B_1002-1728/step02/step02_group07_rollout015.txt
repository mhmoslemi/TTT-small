        # Current h is best_h
        h_t = best_h.copy()
        # Precompute all vk for each k
        # For all k in -n_points +1 to n_points-1
        n = n_points
        v_list = []
        for k in range(-n+1, n):
            vk = np.zeros(n)
            for i in range(n):
                j = i + k
                if 0 <= j < n:
                    vk[i] = 1 - h_t[j]
                else:
                    vk[i] = 1.0

        h_var = cp.Variable(n)
        objective_expr = 0

        h_new = h_var.value

        # Adjust sum
        current_sum = np.sum(h_new)

        V = np.zeros((n_shifts, n_points))
        for k_idx in range(n_shifts):
            shift = (k_idx - (n_points - 1))
            for i in range(n_points):
                j = i + shift
                if 0 <= j < n_points:
                    V[k_idx, i] = 1 - h_t[j]
                else:
                    V[k_idx, i] = 1.0

        h_var = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(V @ h_var))
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var <= 1.0,
            h_var >= 0.0
        ]
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.CVXOPT, verbose=False)

        h_new = np.clip(h_var.value, 0, 1)
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-6:
            h_new = h_new * (target_sum / current_sum)

        corr_values = np.correlate(h_new, 1 - h_new, mode='full') * dx
        new_c5 = np.max(corr_values)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()

    return best_h, best_c5, n_points
