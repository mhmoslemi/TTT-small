import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_energy(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = h_init.copy()
    best_h = current_h.copy()
    current_energy = compute_energy(current_h)
    best_energy = current_energy

    temp = 1.0
    cooling_rate = 0.995
    rs = np.random.RandomState(seed)

    while time.time() - start_time < budget_s * 0.99:
        # Generate a new candidate
        h_new = current_h.copy()
        attempt = 0
        while True:
            i = rs.randint(n_points)
            j = rs.randint(n_points)
            if i != j:
                break
        # Try to move in both directions
        move_succeeded = False
        if rs.rand() < 0.5:
            # Try to increase h[i] and decrease h[j]
            max_add = min(1.0 - h_new[i], h_new[j])
            if max_add > 0:
                delta = rs.uniform(0, max_add)
                h_new[i] += delta
                h_new[j] -= delta
                move_succeeded = True
        else:
            # Try to decrease h[i] and increase h[j]
            max_remove = min(h_new[i], 1.0 - h_new[j])
            if max_remove > 0:
                delta = rs.uniform(0, max_remove)
                h_new[i] -= delta
                h_new[j] += delta
                move_succeeded = True
        if not move_succeeded:
            continue  # Skip this iteration, no valid move
        new_energy = compute_energy(h_new)
        delta_energy = new_energy - best_energy
        # Acceptance criterion
        if delta_energy < 0 or rs.rand() < np.exp(-delta_energy / temp):
            current_h[:] = h_new
            current_energy = new_energy
            if current_energy < best_energy:
                best_energy = current_energy
                best_h[:] = current_h.copy()
        # Cooling
        temp *= cooling_rate

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_energy, n_points)
