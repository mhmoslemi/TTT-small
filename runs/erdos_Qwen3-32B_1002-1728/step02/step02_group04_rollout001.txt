import numpy as np
import time

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def ensure_feasible(candidate, n_points, required_sum):
    candidate = np.clip(candidate, 0, 1)
    current_sum = np.sum(candidate)
    if current_sum > 1e-10:
        scaling = required_sum / current_sum
        candidate *= scaling
        candidate = np.clip(candidate, 0, 1)
    offset = np.sum(candidate) - required_sum
    if np.abs(offset) > 1e-10:
        candidate[0] -= offset
    return candidate

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    h = ensure_feasible(h_initial, n_points, required_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    start_time = time.time()
    alpha = 0.01
    decay_rate = 0.99
    perturb_interval = 200  # every 200 iterations
    iteration = 0

    while time.time() - start_time < budget_s - 10:
        iteration += 1
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        m = max_idx
        k = m - (n_points - 1)

        gradient = np.zeros(n_points)
        for i in range(n_points):
            i_k = i + k
            if 0 <= i_k < n_points:
                gradient[i] = 1.0 - h[i_k]
            else:
                gradient[i] = 1.0

        h -= alpha * gradient
        h = ensure_feasible(h, n_points, required_sum)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

            pert_radius = 0.05
            pert_indices = np.random.choice(n_points, int(0.1 * n_points), replace=False)
            h[pert_indices] += np.random.normal(0, pert_radius, len(pert_indices))
            h = ensure_feasible(h, n_points, required_sum)

        alpha *= decay_rate
        alpha = max(alpha, 1e-6)
