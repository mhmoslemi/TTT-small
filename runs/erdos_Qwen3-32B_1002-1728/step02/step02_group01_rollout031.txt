import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h_flat = h_vec.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 10:  # Reserve time for final checks
        n_attempt += 1
        current_time = time.time() - start_time
        time_frac = min(1.0, current_time / budget_s)
        epsilon = 0.1 * (1 - time_frac) + 0.01 * time_frac

        # Generate perturbation from initial_h_values with time-varying epsilon
        rnd = np.random.RandomState(seed + n_attempt)
        perturbed = np.array(initial_h_values) + rnd.normal(0, epsilon, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        # Optimization with COBYLA to handle non-smooth objective
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='COBYLA',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'tol': 1e-6, 'disp': False}
        )

        if result.success:
            new_h = result.x.reshape(n_points)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            # Post-optimization projection
            new_h = np.clip(new_h, 0, 1)
            total_new = np.sum(new_h)
            if total_new > 1e-8:
                new_h = (new_h / total_new) * target_sum
                new_h = np.clip(new_h, 0, 1)
            new_c5 = objective(new_h)
            # Validate constraints
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
