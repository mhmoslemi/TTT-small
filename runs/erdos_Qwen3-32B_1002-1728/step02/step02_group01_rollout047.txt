import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial_h_values meet constraints
    h = np.clip(h, 0, 1)
    h = h / h.sum() * target_sum
    h = np.clip(h, 0, 1)
    
    bounds = [(0.0, 1.0)] * n_points
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        s = h_vec.sum()
        if s != 0:
            h_vec = (h_vec / s) * target_sum
        else:
            h_vec.fill(0)
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec

    best_h = normalize(h)
    best_c5 = objective(best_h)
    
    np.random.seed(seed)
    temp = 0.1
    alpha = 0.995
    max_iter_per_temp = 100
    perturb_scale = 0.1
    
    while time.time() - start_time < budget_s:
        for _ in range(max_iter_per_temp):
            if time.time() - start_time >= budget_s:
                break
                
            # Generate perturbation
            rnd = np.random.RandomState(int(time.time() % 1e6))
            perturbed = best_h + rnd.normal(0, perturb_scale * temp, n_points)
            perturbed = normalize(perturbed)
            
            new_c5 = objective(perturbed)
            delta = new_c5 - best_c5

                best_h = perturbed.copy()
                best_c5 = new_c5
        
        if temp < 1e-6:
            break
        temp *= alpha

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
