import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Precompute masks for each k in 1..n_points-1
    masks_i = []
    masks_k_plus = []
    for k in range(1, n_points):
        mask_i = np.zeros(n_points)
        mask_i[:n_points - k] = 1
        masks_i.append(mask_i)
        mask_k_plus = np.zeros(n_points)
        mask_k_plus[k:] = 1
        masks_k_plus.append(mask_k_plus)

    # Binary search initialization
    low = 0.0
    high = 0.38085  # Upper bound to beat
    best_c = high
    best_h = h_initial.copy()
    start_time = time.time()
    max_binary_steps = 100
    binary_steps = 0
    epsilon = 1e-6

    while binary_steps < max_binary_steps and (time.time() - start_time) < budget_s - 10:
        mid = (low + high) / 2
        binary_steps += 1

        # Define constraints for LP feasibility problem
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        for k_idx, k in enumerate(range(1, n_points)):
            mask_i = masks_i[k_idx]
            mask_kp = masks_k_plus[k_idx]
            sum_valid_i = mask_i @ h
            sum_valid_kp = mask_kp @ h
            constraint = 0.5 * dx * (sum_valid_i - sum_valid_kp + (n_points - k)) <= mid
            constraints.append(constraint)

        # Solve LP
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:

                    best_c = mid
                    best_h = h.value.copy()
                    high = mid
                else:
                    low = mid
            else:
                low = mid
        except Exception:
            low = mid

        if high - low < epsilon:
            break

    # Post-process best_h
    best_h = np.clip(best_h, 0, 1)
    best_h = np.where(np.isclose(best_h, 0, atol=1e-6), 0.0, best_h)
