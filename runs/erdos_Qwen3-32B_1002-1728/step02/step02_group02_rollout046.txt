import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def _project(p):
        clamped = np.clip(p, 0, 1)
        clamped_sum = clamped.sum()
        if clamped_sum == 0:
            return np.ones_like(clamped) * 0.5
        temp = clamped * (target_sum / clamped_sum)
        return np.clip(temp, 0, 1)

    def compute_score(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    current_h = h.copy()
    current_score = compute_score(current_h)
    best_h = h.copy()
    best_score = current_score

    # Simulated Annealing parameters
    T_initial = 5.0
    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        progress = (time.time() - start_time) / budget_s
        T = T_initial * (1 - progress)
        if T < 1e-6:
            break

        # Perturb
        new_h = current_h.copy()
        indices = np.random.choice(n_points, size=3, replace=False)
        deltas = T * 0.5 * np.random.uniform(-1.0, 1.0, size=3)
        new_h[indices] += deltas
        new_h = _project(new_h)

        new_score = compute_score(new_h)

        # Update best solution
        if new_score < best_score:
            best_h[:] = new_h.copy()
            best_score = new_score

        # Accept or reject
        if new_score < current_score or np.random.rand() < np.exp((current_score - new_score) / T):
            current_h[:] = new_h.copy()
            current_score = new_score

    final_corr = correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(final_corr), n_points
