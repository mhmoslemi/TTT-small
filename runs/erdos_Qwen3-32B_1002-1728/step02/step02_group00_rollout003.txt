import time
import numpy as np
import random

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 64
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial binary step function: alternating 1 and 0
    h = np.zeros(n_points)
    h[::2] = 1.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-4

    np.random.seed(seed)
    random.seed(seed)

    # Main simulated annealing loop
    while time.time() - start_time < budget_s - 5:
        # Propose a move: flip two indices with different values
        while True:
            i = np.random.randint(0, n_points)
            j = np.random.randint(0, n_points)
            if i != j and h[i] != h[j]:
                break

        # Create new candidate
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]

        # Compute new C5
        current_c5 = compute_c5(new_h, dx)

        # Accept or update
        if current_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = current_c5
            h = new_h.copy()
        else:
            delta = current_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = new_h.copy()

        # Cool down
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp

        # Periodic energy checks
        if (time.time() - start_time) % 10 == 0:
            np.random.seed(seed + int(time.time()))
            random.seed(seed + int(time.time()))

    # Final checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
