import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5

    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling rate
    start_time = time.time()
    budget = budget_s

    while time.time() - start_time < budget - 5:  # Leave buffer for termination
        # Perturb
        h_perturbed = current_h + np.random.normal(0, 0.1 * T, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if abs(current_sum - target_sum) > 1e-8:
            h_perturbed[0] += (target_sum - current_sum)
            h_perturbed = np.clip(h_perturbed, 0, 1)
        new_c5 = compute_c5(h_perturbed)
        # Metropolis criterion
        delta = current_c5 - new_c5
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            current_h = h_perturbed.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        T *= alpha
    return best_h, best_c5, n_points
