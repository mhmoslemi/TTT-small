                c5_plus = compute_c5(h_plus, dx)
                h_minus = h.copy()
                h_minus[j] = np.clip(original_val - 0.05, 0, 1)
                h_minus = repair(h_minus, target_sum)
                c5_minus = compute_c5(h_minus, dx)
                if c5_plus < current_c5 or c5_minus < current_c5:
                    if c5_plus < c5_minus:
                        h = h_plus.copy()
                        new_c5 = c5_plus
                    else:
                        h = h_minus.copy()
                        new_c5 = c5_minus
                    current_c5 = new_c5
                    if current_c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = current_c5
                    improved = True
            if not improved:
                no_improvement_passes += 1
            else:
                no_improvement_passes = 0

    return best_h, best_c5, n_points
