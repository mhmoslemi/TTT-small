```python
import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 100000
    alpha0 = 0.01
    alpha_decay = 0.9995
    alpha_min = 1e-6
    alpha = alpha0
    start_time = time.time()
    best_h = h.copy()
    best_c5 = np.inf

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute overlap
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Find maximum overlap shift k
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        # Compute gradient for k
        grad = np.zeros_like(h)
        i_min = max(0, -k)
        i_max = min(n_points - 1, n_points - 1 - k)
        for i in range(i_min, i_max + 1):
            j = i + k
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]

        # Update h with gradient
        h -= alpha * grad
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if abs(sum_h) > 1e-10:
            scale = target_sum / sum_h
            h *= scale
            h = np.clip(h, 0, 1)

        # Adaptive learning rate
        alpha = max(alpha_min, alpha * alpha_decay)

    return best_h, best_c5, n_points
```
