def run(seed=42, budget_s=1100, **kwargs):

        return lse * dx

        if total != 0:
            x = (x / total) * target_sum
            x = np.clip(x, 0, 1)
        return x

    best_h = h.copy()

        phase = int((time.time() - start_time) / budget_s * n_iter)
