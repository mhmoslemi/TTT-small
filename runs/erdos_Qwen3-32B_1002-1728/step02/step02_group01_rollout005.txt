import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    sigma_gauss = 0.05
    sigma_block = 0.1
    perturbation_dr = 0.95  # Annealing decay rate

    while time.time() - start_time < budget_s - 60:
        candidates = []

        # Gaussian perturbation
        p = best_h + np.random.normal(0, sigma_gauss, n_points)
        p = np.clip(p, 0, 1)
        total = np.sum(p)
        p = (p / total * target_sum) if total else p
        p = np.clip(p, 0, 1)
        candidates.append(p)

        # Boundary jitter
        mask = np.random.rand(n_points) < 0.2
        p_jitter = best_h.copy()
        p_jitter[mask] += np.random.uniform(-0.3, 0.3, size=np.sum(mask))
        p_jitter = np.clip(p_jitter, 0, 1)
        total = np.sum(p_jitter)
        p_jitter = (p_jitter / total * target_sum) if total else p_jitter
        p_jitter = np.clip(p_jitter, 0, 1)
        candidates.append(p_jitter)

        # Block perturbation
        start_idx = np.random.randint(0, n_points)
        end_idx = min(start_idx + np.random.randint(5, 16), n_points)
        p_block = best_h.copy()
        block = p_block[start_idx:end_idx] + np.random.normal(0, sigma_block, end_idx - start_idx)
        p_block[start_idx:end_idx] = np.clip(block, 0, 1)
        total = np.sum(p_block)
        p_block = (p_block / total * target_sum) if total else p_block
        p_block = np.clip(p_block, 0, 1)
        candidates.append(p_block)

        for candidate in candidates:
            result = minimize(
                fun=objective,
                x0=candidate,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 40, 'ftol': 1e-6}
            )
            if result.success:
                new_h = result.x.copy()
                new_h = np.clip(new_h, 0, 1)
                total = np.sum(new_h)
                new_h = (new_h / total * target_sum) if total else new_h
                new_h = np.clip(new_h, 0, 1)
                if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                    np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5

        # Annealing perturbation
        sigma_gauss *= perturbation_dr
        sigma_block *= perturbation_dr

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
