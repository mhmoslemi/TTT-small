        result = minimize(
            fun=objective,
            x0=h_bin,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6},
        )
        if result.success:
            h_candidate = result.x
