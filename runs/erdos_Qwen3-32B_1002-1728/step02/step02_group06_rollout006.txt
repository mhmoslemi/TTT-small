def generate_periodic_pattern(p, n_points):

    half_p = p // 2

            pattern[i] = 1.0
        else:

def adjust_h(h, target_sum):

    current_sum = np.sum(h_clipped)

    if abs(delta) < 1e-8:

    h_clipped[0] += delta
