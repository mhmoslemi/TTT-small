    for k in range(-(n_points-1), n_points):  # k from -(n-1) to n-1, i.e., n_points * 2 -1 possible k's

    # Group u by k
    from collections import defaultdict
    k_to_u = defaultdict(list)
    for idx, (i, k, j) in enumerate(valid_ikj):

    constraints = []

    # Constraints on u
    for idx, (i, k, j) in enumerate(valid_ikj):
        u = u_list[idx]
        constraints += [
            u >= h[i] + h[j] - 1,

    # Problem
    problem = cp.Problem(cp.Minimize(0), constraints)  # Feasibility problem
    # Solve with ECOS
    result = problem.solve(solver=cp.ECOS, max_iters=1000, verbose=False)
