                delta = target_sum - np.sum(init)
                init[:int(len(init)/2)] += delta / (len(init)/2)  # spread delta
                init = np.clip(init, 0.0, 1.0)
        # Proceed with optimization
        result = minimize(
            objective, x0=init, method='SLSQP', bounds=bounds, constraints=sum_constraint,
            options={'maxiter': 300, 'ftol': 1e-7})
        curr_h = result.x.copy() if result.success else init.copy()

        # Repair the solution
        curr_h = np.clip(curr_h, 0, 1)
        # Re-adjust sum constraint
        curr_sum = np.sum(curr_h)
        delta = target_sum - curr_sum
        if delta != 0:
            # Distribute delta to the first 10 elements
            idxs = np.argsort(curr_h)[:int(len(curr_h)/2)]
            curr_h[idxs] += delta / len(idxs)
            curr_h = np.clip(curr_h, 0, 1)

        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()

        # Add a perturbation and optimize again
        if time.time() - start_time < budget_s - 20:
            perturbation = np.random.normal(0, 0.03, n_points)
            new_init = np.clip(curr_h + perturbation, 0, 1)
            new_sum = np.sum(new_init)
            new_init *= target_sum / (new_sum + 1e-10)  # scale to target_sum
            new_init = np.clip(new_init, 0, 1)

            result_perturb = minimize(
                objective, x0=new_init, method='SLSQP', bounds=bounds, constraints=sum_constraint,
                options={'maxiter': 200, 'ftol': 1e-7})
            perturbed_h = result_perturb.x.copy() if result_perturb.success else new_init.copy()
            perturbed_h = np.clip(perturbed_h, 0, 1)

            corr_perturbed = np.correlate(perturbed_h, 1 - perturbed_h, mode='full') * dx
            c5_perturbed = np.max(corr_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = perturbed_h.copy()

    # Final evaluation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
