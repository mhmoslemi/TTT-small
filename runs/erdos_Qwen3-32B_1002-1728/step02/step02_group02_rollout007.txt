def run(seed=42, budget_s=1100):
    import numpy as np
    import time
    import cvxpy as cp
    import scipy.signal

    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()

    best_c5 = float('inf')
    best_h = h.copy()

    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]

    while time.time() - start_time < budget_s - 0.1:
        # Compute current correlation
        corrs = scipy.signal.correlate(h, 1 - h, mode='full') 
        corrs *= dx
        max_val = np.max(corrs)
        max_idx = np.argmax(corrs)
        k = max_idx - (n_points - 1)

        # Find valid i's for this k
        start_i = max(0, -k)
        end_i = min(n_points -1, n_points -1 -k)
        valid_i = np.arange(start_i, end_i + 1)
        pairs = [(i, i +k) for i in valid_i]

        # Compute coefficients for the linear objective
        coeffs = np.zeros(n_points)
        for i, j in pairs:
            coeffs[i] += h[j]
            coeffs[j] += h[i]

        # Formulate and solve the LP
        if len(pairs) == 0:
            # No pairs, so current max is fixed? Not possible, so move on
            break
        objective = cp.Maximize(coeffs @ h_var)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.MOSEK, verbose=False)

        if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            # Unable to solve, perhaps skip
            break

        new_h = h_var.value
        if new_h is None:
            break

        # Ensure the new h is in [0,1] and sum is correct
        h = np.clip(new_h, 0, 1)
        # Project to ensure sum is correct
        h = _project(h, target_sum, n_points)

        # Recompute C5 and check if improved
        current_c5 = np.max(scipy.signal.correlate(h, 1 - h, mode='full') * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # Maybe add a convergence check here
        elif abs(best_c5 - current_c5) < 1e-8:
            # No improvement, break
            break

    return best_h, best_c5, n_points

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    return np.clip(clamped * (target_sum / clamped_sum), 0, 1)

import numpy as np
import time
import cvxpy as cp
import scipy.signal

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()

    h_var = cp.Variable(n_points)
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]

    while time.time() - start_time < budget_s - 0.1:
        corrs = scipy.signal.correlate(h, 1 - h, mode='full') * dx
        max_val = np.max(corrs)
        if max_val >= best_c5:
            best_c5 = max_val
        max_idx = np.argmax(corrs)
        k = max_idx - (n_points - 1)

        start_i = max(0, -k)
        end_i = min(n_points - 1, n_points - 1 - k)
        valid_i = np.arange(start_i, end_i + 1)
        pairs = [(i, i + k) for i in valid_i]

        if not pairs:
            break

        coeffs = np.zeros(n_points)
        for i, j in pairs:
            coeffs[i] += h[j]
            coeffs[j] += h[i]

        objective = cp.Maximize(coeffs @ h_var)
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_candidate = np.clip(h_var.value, 0, 1)
                h = _project(h_candidate, target_sum, n_points)
                corrs_new = scipy.signal.correlate(h, 1 - h, mode='full') * dx
                new_c5 = np.max(corrs_new)
                if new_c5 < best_c5 - 1e-6:
                    best_c5 = new_c5
                    best_h = h.copy()
                elif abs(best_c5 - new_c5) < 1e-8:
                    break
        except Exception:
            pass

    return best_h, best_c5, n_points

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    return np.clip(clamped * (target_sum / clamped_sum), 0, 1)
