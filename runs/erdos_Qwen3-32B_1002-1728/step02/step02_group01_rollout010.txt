import numpy as np
import time

def evaluate_C5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h is valid
    h = np.clip(h, 0, 1)
    total = h.sum()
    if total > 1e-10:
        h = (h / total) * target_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_C5(best_h, dx)

    T = 0.1
    alpha = 0.995
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        current_h = best_h.copy()
        perturb = np.random.normal(0, T * 0.05, size=n_points)
        new_h = np.clip(current_h + perturb, 0, 1)
        total = new_h.sum()
        if total > 1e-10:
            new_h = (new_h / total) * target_sum
        else:
            new_h = np.full(n_points, target_sum / n_points)
