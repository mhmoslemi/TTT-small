import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 100000  # High upper limit, time-controlled
    alpha = 0.01  # Initial learning rate
    start_time = time.time()
    best_h = h_current.copy()
    best_c5 = np.inf

    def scale_to_sum(h):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return h
        scale = target_sum / s
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute overlaps
        corr = np.correlate(h_current, 1 - h_current, mode='full') * dx
        idx_max = np.argmax(corr)
        current_c5 = corr[idx_max]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()

        # Compute shift k
        k = idx_max - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k
            m = i - k
            term = 0.0
            if 0 <= j < n_points:
                term += (1 - h_current[j])
            if 0 <= m < n_points:
                term -= h_current[m]
            grad[i] = dx * term

        # Update h with learning rate adjustment
        h_current -= alpha * grad

        # Clip and project
        h_current = np.clip(h_current, 0, 1)
        h_current = scale_to_sum(h_current)

        if iteration > max_iter // 4 and iteration % 100 == 0:

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
