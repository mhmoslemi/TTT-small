import numpy as np
from scipy.optimize import basinhopping
import time

class ProjectedStep:
    def __init__(self, sigma):
        self.sigma = sigma

    def __call__(self, x):
        dx = np.random.normal(0, self.sigma, len(x))
        x_new = x + dx
        x_new = np.clip(x_new, 0, 1)
        total = np.sum(x_new)
        if total != 0:
            x_new = (x_new / total) * (len(x) / 2)
        x_new = np.clip(x_new, 0, 1)
        return x_new

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    step_taker = ProjectedStep(sigma=0.1)

    minimizer_kwargs = {
        'method': 'SLSQP',
        'bounds': bounds,
        'constraints': cons,
        'options': {'maxiter': 120, 'ftol': 1e-9}
    }

    bh = basinhopping(
        func=objective,
        x0=h,
        niter=1,
        T=1.0,
        stepsize=0.1,
        take_step=step_taker,
        minimizer_kwargs=minimizer_kwargs,
        seed=seed
    )

    remaining_time = max(budget_s - 60, 30)
    estimated_iter = max(int(remaining_time // 1.5), 1)
    
    for i in range(estimated_iter):
        if time.time() - start_time > budget_s - 30:
            break
        bh.run(1)

    best_h = np.array(bh.lowest_coords)
    best_c5 = bh.fun

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"

    return (best_h.tolist(), best_c5, n_points)
