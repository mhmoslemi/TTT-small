import numpy as np
import time

def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def repair(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    delta = target_sum - current_sum
    h[0] += delta
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    population_size = 50
    tournament_size = 3
    mutation_rate = 0.1
    sigma = 0.05
    num_elites = int(population_size * 0.2)

    population = [h_initial.copy()]

    p_list = [2, 4, 6, 8, 10]
    for p in p_list:
        population.append(generate_periodic_pattern(p, n_points))
    
    while len(population) < population_size:
        h_rand = np.random.rand(n_points)
        h_rand = repair(h_rand, target_sum)
        population.append(h_rand)
    
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    while time.time() - start_time < budget_s - 1:
        population_c5 = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            population_c5.append((c5, h))
        population_c5.sort(key=lambda x: x[0])
        
        current_best_c5, current_best_h = population_c5[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()

        next_population = [population_c5[i][1] for i in range(num_elites)]
        remaining = population_size - num_elites

        def tournament_select(population_with_c5):
            indices = np.random.choice(len(population_with_c5), tournament_size)
            candidates = [population_with_c5[i] for i in indices]
            return min(candidates, key=lambda x: x[0])[1]
        
        for _ in range(remaining):
            parent1 = tournament_select(population_c5)
            parent2 = tournament_select(population_c5)
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = repair(child, target_sum)
            if np.random.rand() < mutation_rate:
                child += np.random.normal(0, sigma, n_points)
                child = repair(child, target_sum)
            next_population.append(child)
        
        population = next_population

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
