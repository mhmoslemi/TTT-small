def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    n_particles = 50
    omega = 0.7
    phi_c = 1.5
    phi_s = 1.5
    max_vel = 0.1
    local_search_n = 5
    local_steps = 3
    eps = 1e-4

    particles = []
    for _ in range(n_particles):
        h = h_initial + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)

    global_best_score = np.inf
    global_best_position = h_initial.copy()
