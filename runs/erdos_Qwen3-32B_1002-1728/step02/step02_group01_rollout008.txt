from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 10:  # Margin for final checks
        n_attempt += 1

        noise_scale = max(0.1 * (0.95 ** (n_attempt / 100)), 0.01)
        perturbed = best_h + np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)

        # Adjust sum to constraint
        total = np.sum(perturbed)
        if abs(total - target_sum) < 1e-8:

        # Optimization with tighter tolerance and more iterations
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
