        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Generate additional structured initial guesses
    initial_guesses = [h_init]
    if n_points % 2 == 0:
        h_two_block = np.array([1.0] * (n_points // 2) + [0.0] * (n_points // 2))
        initial_guesses.append(h_two_block)

    best_c5 = np.inf
    best_h = h_init.copy()

    for guess in initial_guesses:
        for maxiter in [500, 300]:
            result = minimize(
                fun=objective,
                x0=guess,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-8}
            )
            if result.success:
                h_candidate = result.x.reshape(n_points)
                sum_candidate = np.sum(h_candidate)
                if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                    corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                    c5_candidate = np.max(corr_candidate) * dx
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate

    # Iterative improvement with perturbations
    while time.time() - start_time < budget_s * 0.95:
        rs = np.random.RandomState(seed + int(time.time()))
        perturbed = best_h + rs.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-6:
            perturbed = (perturbed / perturbed_sum) * target_sum
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-8}
        )
        if result.success:
            h_candidate = result.x.reshape(n_points)
            sum_candidate = np.sum(h_candidate)
            if abs(sum_candidate - target_sum) < 1e-6 and np.all((h_candidate >= 0) & (h_candidate <= 1)):
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate

    # Final assert and return
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
