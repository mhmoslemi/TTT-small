from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    iteration = 0
    initial_sigma = 0.05
    decay_rate = 0.95

    while time.time() - start_time < budget_s - 60:  # Stop early to avoid timeout
        iteration += 1
        current_sigma = initial_sigma * (decay_rate ** iteration)
        # Generate 3 perturbations
        for j in range(3):
            sub_seed = seed + iteration * 3 + j
            rnd = np.random.RandomState(sub_seed)
            if j == 0:
                # Gaussian perturbation
                perturbed = best_h + rnd.normal(0, current_sigma, n_points)
            elif j == 1:
                # Boundary jitter
                perturbed = best_h.copy()

                indices = rnd.choice(n_points, size=n_perturb, replace=False)
                perturbed[indices] = np.where(best_h[indices] < 0.5, 0.0, 1.0)
            else:

                perturbed = best_h.copy()

                start_idx = rnd.randint(0, n_points - block_len)
                end_idx = start_idx + block_len
                perturbed[start_idx:end_idx] += rnd.normal(0, 0.1, block_len)
            # Clip to bounds
            perturbed = np.clip(perturbed, 0, 1)
            # Compute total and scale to target_sum
            total = np.sum(perturbed)
            if total != 0:
                perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
            # Optimization with increased maxiter and tighter tolerance
            result = minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
            )
            if result.success:
                new_h = result.x.reshape(n_points)
                new_corr = np.correlate(new_h, 1 - new_h, mode='full')

                # Validate constraints

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    iter_count = 0
    initial_sigma = 0.05
    decay_rate = 0.95

    while time.time() - start_time < budget_s - 60:
        iter_count += 1
        curr_sigma = initial_sigma * (decay_rate ** iter_count)
        for pert_type in range(3):
            sub_seed = seed + iter_count * 3 + pert_type
            rnd = np.random.RandomState(sub_seed)
            if pert_type == 0:
                perturbed = best_h + rnd.normal(0, curr_sigma, n_points)
            elif pert_type == 1:
                perturbed = best_h.copy()
                n_perturb = int(0.2 * n_points)
                indices = rnd.choice(n_points, size=n_perturb, replace=False)
                perturbed[indices] = np.where(best_h[indices] < 0.5, 0.0, 1.0)
            else:
                perturbed = best_h.copy()
                block_len = rnd.randint(5, 15)
                start_idx = rnd.randint(0, n_points - block_len)
                end_idx = start_idx + block_len
                perturbed[start_idx:end_idx] += rnd.normal(0, 0.1, block_len)

            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total != 0:
                perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            result = minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-9, 'disp': False}
            )

            if result.success:
                new_h = result.x.reshape(n_points)
                corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(corr) * dx
                if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                    np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
