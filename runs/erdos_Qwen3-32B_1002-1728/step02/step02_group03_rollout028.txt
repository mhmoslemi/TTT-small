import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    max_iterations = 10000
    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    iteration = 0

    def compute_gradient(h, k_abs, dx):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n - k_abs):
            grad[i] += 1 - h[i + k_abs]
            grad[i + k_abs] += -h[i]
        grad *= dx
        return grad

    def project(h, target_sum):
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        h *= scale
        h = np.clip(h, 0, 1)
        return h

    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:

        k_val = max_index - (n_points - 1)

            if k_abs < n_points:

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
