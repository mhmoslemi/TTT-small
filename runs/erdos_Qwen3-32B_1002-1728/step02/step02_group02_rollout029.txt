import numpy as np
import time
from scipy.signal import correlate

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    temp = clamped * (target_sum / clamped_sum)
    return np.clip(temp, 0, 1)

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    current_h = h_initial.copy()
    current_C5 = compute_C5(current_h)
    best_h = current_h.copy()
    best_C5 = current_C5

    initial_T = 1.0
    cooling_rate = 0.999
    current_T = initial_T

    start_time = time.time()

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break

        # Generate perturbed candidate
        h_candidate = current_h.copy()
        num_perturb = np.random.randint(1, 5)
        indices = np.random.choice(n_points, size=num_perturb, replace=False)

        for idx in indices:
            delta = np.random.uniform(-0.2, 0.2)
            h_candidate[idx] += delta

        h_candidate = _project(h_candidate, target_sum, n_points)
        new_C5 = compute_C5(h_candidate)

        delta_C5 = new_C5 - current_C5

        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / current_T):
            current_h = h_candidate.copy()
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()

        current_T *= cooling_rate

    final_C5 = compute_C5(best_h)
    return best_h, final_C5, n_points
