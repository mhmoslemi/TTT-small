def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    np.random.seed(seed)
    h = ensure_feasible(h_initial.copy(), n_points, required_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    alpha = 0.01
    alpha_decay = 0.995
    perturb_interval = 150
    iteration = 0

    while time.time() - start_time < budget_s - 10:
        iteration += 1

        # Compute correlation and find worst displacement
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_index = np.argmax(corr)
        d = max_index - (n_points - 1)

        # Compute gradient based on displacement d
        if d >= 0:
            k = d
            valid = n_points - k
            grad = np.zeros(n_points)
            grad[:valid] = 1.0 - h[k:]
        else:
            k = -d
            valid = n_points - k
            grad = np.zeros(n_points)
            grad[k:] += 1.0 - h[:valid]
            grad[:valid] -= h[k:]

        # Gradient step
        h -= alpha * grad
        h = ensure_feasible(h, n_points, required_sum)

        # Perturbation
        if iteration % perturb_interval == 0:
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            h = ensure_feasible(h, n_points, required_sum)
            alpha = max(0.001, alpha * 0.9)  # decrease alpha after perturbation

        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Decay learning rate
        alpha *= alpha_decay

        # Check if time is up
        if time.time() - start_time > budget_s - 5:
            break

    return best_h, best_c5, n_points
