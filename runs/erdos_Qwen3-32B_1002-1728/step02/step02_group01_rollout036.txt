import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 40
    mutation_rate = 0.05
    elite_fraction = 0.1
    np.random.seed(seed)

    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        candidate = np.random.rand(n_points)
        total = candidate.sum()
        if total > 1e-8:
            candidate = (candidate / total) * target_sum
        population.append(np.clip(candidate, 0, 1))
    population = np.array(population)

    def evaluate(individual):
        corr = np.correlate(individual, 1 - individual, mode='full')
        return np.max(corr) * dx

    best_idx = 0
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 60:  # Leave buffer for final steps
        fitness = np.array([evaluate(ind) for ind in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_idx = current_best_idx
            best_c5 = fitness[current_best_idx]

        # Selection
        sorted_indices = np.argsort(fitness)
        elites_N = max(1, int(pop_size * elite_fraction))
        elites = population[sorted_indices[:elites_N]]
        parents = population[sorted_indices[:int(pop_size * 0.3)]]

        # Generate offspring
        offspring = []
        for _ in range(pop_size - elites_N):
            # Crossover
            parent1, parent2 = parents[np.random.choice(len(parents), 2, replace=False)]
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            # Normalize and clip
            child = np.clip(child, 0, 1)
            total = child.sum()
            if total > 1e-8:
                child = (child / total) * target_sum
            child = np.clip(child, 0, 1)

            # Mutation
            if np.random.rand() < 0.5:
                noise = np.random.normal(0, mutation_rate, n_points)
                mutated = child + noise
                mutated = np.clip(mutated, 0, 1)
                total_m = mutated.sum()
                if total_m > 1e-8:
                    mutated = (mutated / total_m) * target_sum
                mutated = np.clip(mutated, 0, 1)
                child = mutated

            offspring.append(child)

        # Form new population
        new_population = list(elites)
        new_population += [population[sorted_indices[0]] for _ in range(int(pop_size * 0.3))]
        new_population += offspring
        if len(new_population) < pop_size:
            pad = [np.random.rand(n_points) for _ in range(pop_size - len(new_population))]
            for p in pad:
                total = p.sum()
                if total > 1e-8:
                    p = (p / total) * target_sum
                p = np.clip(p, 0, 1)
                new_population.append(p)
        population = np.array(new_population)

    # Final refinement of best candidate
    best_h = population[best_idx]
    best_h = np.clip(best_h, 0, 1)
    total = best_h.sum()
    if total > 1e-8:
        best_h = (best_h / total) * target_sum
    best_h = np.clip(best_h, 0, 1)

    # Final constraint validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
