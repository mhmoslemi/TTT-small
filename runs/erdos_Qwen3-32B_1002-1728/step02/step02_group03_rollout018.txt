            overlap = np.dot(h[:valid], 1 - h[m:valid + m]) * dx
            overlaps.append(overlap)
        return overlaps

    def compute_gradient_for_m(h, m):
        grad = np.zeros(n_points)
        for i in range(n_points - m):
            grad[i] += (1 - h[i + m]) * dx
            grad[i + m] -= h[i] * dx
        return grad

    def scale_to_sum(h):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return h
        h_scaled = h * (target_sum / s)
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    best_h = h.copy()
    best_c5 = np.inf
    start_time = time.time()
    step_size = 0.01
    patience = 0
    max_step_increase = 1.1
    min_step_size = 1e-5

    while time.time() - start_time < budget_s - 1:
        overlaps = compute_overlaps(h)
        current_c5 = max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        top_m_indices = np.argsort(overlaps)[-5:]  # take top 5 shifts
        top_m = [i + 1 for i in top_m_indices]  # convert to actual m values

        total_grad = np.zeros(n_points)
        for m in top_m:
            grad = compute_gradient_for_m(h, m)
            total_grad += grad

        # Normalize by number of gradients
        avg_grad = total_grad / len(top_m)

        # Take gradient step
        h = h - step_size * avg_grad
        h = np.clip(h, 0, 1)
        h = scale_to_sum(h)

        # Adjust step size based on progress
        if patience == 0:
            step_size = min(step_size * 1.1, 0.1)
        elif patience > 30:
            step_size = max(step_size * 0.9, min_step_size)
            patience = 0

    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    return best_h, final_c5, n_points
