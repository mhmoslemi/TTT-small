import numpy as np
import time
from scipy.signal import correlate

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    temp = clamped * (target_sum / clamped_sum)
    return np.clip(temp, 0, 1)

def evaluate(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, scores, size=5, top=2):
    selected_indices = np.random.choice(len(population), size=size, replace=False)
    candidates = [scores[i] for i in selected_indices]
    indices_to_select = np.argsort(candidates)[:top]
    return [population[selected_indices[i]] for i in indices_to_select]

def crossover(parent1, parent2, target_sum, n_points):
    idx = np.random.randint(1, n_points)
    child1 = np.concatenate((parent1[:idx], parent2[idx:])).copy()
    child2 = np.concatenate((parent2[:idx], parent1[idx:])).copy()
    child1 = _project(child1, target_sum, n_points)
    child2 = _project(child2, target_sum, n_points)
    return child1, child2

def mutate(h, target_sum, n_points):
    h = h.copy()
    mut_type = np.random.rand()
    if mut_type < 0.1:  # jitter
        h += np.random.randn(n_points) * 0.05
    elif mut_type < 0.3:  # invert segment
        start = np.random.randint(0, n_points)
        length = np.random.randint(2, 11)
        end = min(start + length, n_points)
        h[start:end] = 1 - h[start:end]
    else:  # slide

        end = min(start + length, n_points)
        delta = 0.1 * np.random.randn()
        h[start:end] = np.clip(h[start:end] + delta, 0, 1)
    h = _project(h, target_sum, n_points)
    return h

    return np.clip(temp, 0, 1)

def evaluate(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, scores, size=5, top=2):
    indices = np.random.choice(len(population), size=size, replace=False)
    candidates = [scores[i] for i in indices]
    selected_indices = indices[np.argsort(candidates)[:top]]
    return [population[i] for i in selected_indices]

def crossover(parent1, parent2, target_sum, n_points):
    idx = np.random.randint(1, n_points)
    child1 = np.concatenate((parent1[:idx], parent2[idx:])).copy()
    child2 = np.concatenate((parent2[:idx], parent1[idx:])).copy()
    child1 = _project(child1, target_sum, n_points)
    child2 = _project(child2, target_sum, n_points)
    return child1, child2

def mutate(h, target_sum, n_points):
    h = h.copy()
    mut_type = np.random.rand()
    if mut_type < 0.1:  # Jitter
        h += np.random.randn(n_points) * 0.05
    elif mut_type < 0.3:  # Invert segment
        start = np.random.randint(0, n_points)
        length = np.random.randint(2, 11)
        end = min(start + length, n_points)
        h[start:end] = 1 - h[start:end]
    else:  # Slide segment
        start = np.random.randint(0, n_points)
        length = np.random.randint(2, 11)
        end = min(start + length, n_points)
        delta = 0.1 * np.random.randn()
        h[start:end] = np.clip(h[start:end] + delta, 0, 1)
    h = _project(h, target_sum, n_points)
    return h

def replace_worst(population, scores, new_candidates):
    new_h_list = [child for score, child in new_candidates]
    new_scores_list = [score for score, child in new_candidates]
    combined_pop = population + new_h_list
    combined_scores = scores + new_scores_list
    idxs = np.argsort(combined_scores)
    selected_indices = idxs[:len(population)]
    population[:] = [combined_pop[i] for i in selected_indices]
    scores[:] = [combined_scores[i] for i in selected_indices]

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 50
    population = []
    for _ in range(pop_size):
        if np.random.rand() < 0.3:
            h = h_initial.copy() + np.random.randn(n_points) * 0.1
        else:
            h = np.random.rand(n_points)
        h = _project(h, target_sum, n_points)
        population.append(h)

    scores = [evaluate(h, dx) for h in population]
    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]
    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        parents = tournament_selection(population, scores)
        child1, child2 = crossover(parents[0], parents[1], target_sum, n_points)
        child1 = mutate(child1, target_sum, n_points)
        child2 = mutate(child2, target_sum, n_points)
        c1_score = evaluate(child1, dx)
        c2_score = evaluate(child2, dx)
        new_candidates = [(c1_score, child1), (c2_score, child2)]
        replace_worst(population, scores, new_candidates)
        current_best_score = np.min(scores)
        if current_best_score < best_score:
            best_idx = np.argmin(scores)
            best_h = population[best_idx].copy()
            best_score = current_best_score

    corr_values = correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr_values), n_points
