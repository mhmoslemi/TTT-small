        if time.time() - start_time > budget_s - 30:
            break
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[sum_constraint],
            options={'maxiter': 100, 'ftol': 1e-5}
        )
        if result.success:
            curr_h = np.clip(result.x, 0, 1)
            offset = np.sum(curr_h) - target_sum
            curr_h[0] -= offset
            curr_h[0] = np.clip(curr_h[0], 0, 1)
            corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            curr_c5 = np.max(corr_val)
            if best_c5 is None or curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h[:] = curr_h[:]

    # Adaptive refinement with time-decreasing perturbation
    while time.time() - start_time < budget_s - 10:
        progress = (time.time() - start_time) / (budget_s - 10)
        noise_std = 0.2 * (1 - progress)
        x0 = best_h.copy()
        x0 += np.random.normal(0, noise_std, n_points)
        x0 = np.clip(x0, 0, 1)
        offset = np.sum(x0) - target_sum
        x0[0] -= offset
        x0[0] = np.clip(x0[0], 0, 1)
            
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[sum_constraint],
            options={'maxiter': 150, 'ftol': 1e-6}
        )
        if result.success:
            curr_h = np.clip(result.x, 0, 1)
            offset = np.sum(curr_h) - target_sum
            curr_h[0] -= offset
            curr_h[0] = np.clip(curr_h[0], 0, 1)
            corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
            curr_c5 = np.max(corr_val)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h[:] = curr_h[:]
                
    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
