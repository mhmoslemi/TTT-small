import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    current_h = h_initial.copy()
    current_c5 = objective(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    np.random.seed(seed)
    T_initial = max(1.0, current_c5 * 10)
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 5:
        iteration += 1
        T = T_initial / np.log(1 + iteration)
        noise = np.random.normal(0, 1, n_points)
        candidate_h = np.clip(current_h + noise * 0.1 * T, 0, 1)
        candidate_h = adjust_sum(candidate_h, target_sum)
        new_c5 = objective(candidate_h)
        delta = new_c5 - current_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h, current_c5 = candidate_h, new_c5

        if current_c5 < best_c5:
            best_h, best_c5 = current_h.copy(), current_c5

    return best_h, best_c5, n_points
