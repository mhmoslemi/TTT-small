import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    best_c5 = None
    best_h = h_initial.copy()
    start_time = time.time()

    # Generate structured initial guesses with triangle and sawtooth patterns
    initial_guesses = [h_initial.copy()]
    periods = [3,4,5,6,7]

    for period in periods:
        for phase in range(period):
            # Square wave
            h_square = np.zeros(n_points)
            h_square[(np.arange(n_points) - phase) % period == 0] = 1.0
            # Normalize to meet sum constraint
            target_sum = n_points / 2.0
            h_square = h_square * target_sum / np.sum(h_square) if np.sum(h_square) > 1e-8 else h_square
            h_square = np.clip(h_square, 0, 1)
            offset = target_sum - np.sum(h_square)
            h_square[0] += offset
            h_square[0] = max(0.0, min(1.0, h_square[0]))
            initial_guesses.append(h_square)
        for phase in range(period):
            # Triangle wave
            positions = (np.arange(n_points) - phase) % period
            mid = period // 2
            up = positions < mid
            h_tri = np.zeros(n_points)
            h_tri[up] = positions[up] / mid
            h_tri[~up] = 2 - 2 * (positions[~up] / mid)

            h_tri = np.clip(h_tri, 0, 1)
            offset = target_sum - np.sum(h_tri)
            h_tri[0] += offset
            h_tri[0] = max(0.0, min(1.0, h_tri[0]))
            initial_guesses.append(h_tri)
        for phase in range(period):
            # Sawtooth wave
            positions = (np.arange(n_points) - phase) % period
            h_saw = positions / (period - 1) if period > 1 else np.full(n_points, 0.5)
            target_sum = n_points / 2.0
            h_saw = h_saw * target_sum / np.sum(h_saw) if np.sum(h_saw) > 1e-8 else h_saw
            h_saw = np.clip(h_saw, 0, 1)
            offset = target_sum - np.sum(h_saw)
            h_saw[0] += offset
            h_saw[0] = max(0.0, min(1.0, h_saw[0]))
            initial_guesses.append(h_saw)

            break
        result = minimize(

            x0=x0,

            bounds=bounds,
            constraints=[sum_constraint],
