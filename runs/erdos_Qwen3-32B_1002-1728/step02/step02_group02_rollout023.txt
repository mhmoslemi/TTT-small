import numpy as np
from scipy.optimize import minimize
import time

def objective(h, n_points, dx):
    h = h.reshape(n_points)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def constraint_sum(h, n_points, target_sum):
    return np.sum(h) - target_sum

def _project(p, target_sum, n_points):
    clamped = np.clip(p, 0, 1)
    clamped_sum = clamped.sum()
    if clamped_sum == 0:
        return np.ones_like(clamped) * 0.5
    temp = clamped * (target_sum / clamped_sum)
    return np.clip(temp, 0, 1)

def run(seed=42, budget_s=1100):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': constraint_sum, 'args': (n_points, target_sum)})
    options = {'maxiter': 10000, 'ftol': 1e-8, 'disp': False, 'eps': 1e-4}

    start_time = time.time()
    result = minimize(
        objective, h_initial, args=(n_points, dx),
        method='SLSQP', bounds=bounds, constraints=constraints, options=options
    )
    elapsed = time.time() - start_time

    h_opt = result.x if result.success else h_initial
    h_opt = _project(h_opt, target_sum, n_points)
    corr_values = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    best_c5 = np.max(corr_values)
    return h_opt, best_c5, n_points
