import numpy as np
import time

def adjust_sum(h, target_sum):
    h_adj = h.copy()
    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)
        if not can_incr.any():
            return h_adj
        total_possible = (1.0 - h_adj[can_incr]).sum()
        h_adj[can_incr] += D * (1.0 - h_adj[can_incr]) / total_possible
    else:
        D = -D
        can_decr = (h_adj > 1e-4)
        if not can_decr.any():
            return h_adj
        total_possible = h_adj[can_decr].sum()
        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    h = adjust_sum(h_initial.copy(), target_sum)
    current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c5 = current_c5
    best_h = h.copy()

    np.random.seed(seed)
    start_time = time.time()
    T_initial = 1.5 * current_c5
    iter_count = 0

    while time.time() - start_time < budget_s - 10:
        # Perturbation with temperature
        perturbation = np.random.normal(0, 0.1 * (T_initial / (1 + np.log(iter_count + 1))), n_points)
        candidate = h + perturbation
        candidate = np.clip(candidate, 0, 1)
        candidate = adjust_sum(candidate, target_sum)

        candidate_c5 = np.max(np.correlate(candidate, 1 - candidate, mode='full')) * dx
        delta = candidate_c5 - current_c5

        T = T_initial / (1 + np.log(iter_count + 1))
        if (delta < 0) or (delta > 0 and np.random.rand() < np.exp(-delta / T)):
            h = candidate.copy()
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        iter_count += 1

    return best_h, best_c5, n_points
