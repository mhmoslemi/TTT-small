import time
from scipy.signal import fftconvolve

def run(seed=42, budget_s=1100, **kwargs):

        ones_minus_h = 1.0 - h
        corr = fftconvolve(h, ones_minus_h[::-1], mode='full')
        return np.max(corr) * dx

    def constrain(free_h):
        h = free_h.copy()
        h = np.clip(h, 0, 1.0)
        total = np.sum(h)
        if abs(total - n_points/2.0) > 1e-8:
            if total > 1e-10:
                scale = (n_points/2.0) / total
                h *= scale
            h[0] += (n_points/2.0 - np.sum(h))
            h[0] = max(0.0, min(1.0, h[0]))
        return h

    def fitness(h):
        h = constrain(h)
        return compute_overlap(h)

    particles = []
    velocities = []
    n_particles = 100

    # Initialize half from initial_h_values, half random
    for i in range(n_particles):

            x = initial_h_values.copy()

        velocities.append(np.random.normal(0.0, 0.01, n_points))

    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [fitness(p) for p in particles]
    global_best = particles[np.argmin(personal_best_fitness)].copy()

    w = 0.9
