from scipy.signal import correlate

def run(seed=42, budget_s=1100):

        'ftol': 1e-8,
        'disp': False,

    }

    start_time = time.time()
    result = opt.minimize(
        fun=objective,
        x0=h_initial.copy().flatten(),

        constraints=constraints,
        method='SLSQP',
        options=options
    )
    elapsed = time.time() - start_time
