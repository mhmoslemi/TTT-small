import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = objective(best_h)
    np.random.seed(seed)
    n_attempt = 0
    
    while time.time() - start_time < budget_s - 60:  # Reserve time for final return
        n_attempt += 1
        current_time = time.time() - start_time
        time_frac = current_time / (budget_s - 60)
        sigma_scale = 0.2 * (1.0 - time_frac)
        rnd = np.random.RandomState(seed + n_attempt)
        perturbed = best_h.copy()
        
        method = rnd.choice(['gaussian', 'boundary', 'block'])
        
        if method == 'gaussian':
            noise = rnd.normal(0, sigma_scale, n_points)
            perturbed += noise
        elif method == 'boundary':
            delta = 0.1 * (1.0 - time_frac)
            for i in range(n_points):
                if rnd.rand() < 0.5:
                    perturbed[i] = max(0, perturbed[i] - delta)
                else:
                    perturbed[i] = min(1, perturbed[i] + delta)
        elif method == 'block':
            block_start = rnd.randint(0, n_points)
            block_len = min(rnd.randint(1, 6), n_points - block_start)
            delta_block = 0.2 * (1.0 - time_frac) * (2 * rnd.rand() - 1)
            perturbed[block_start:block_start+block_len] += delta_block
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total > 1e-6:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        
        # Optimization with adaptive settings
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
        )
        
        if result.success:
            new_h = result.x.reshape(n_points)
            new_c5 = objective(new_h)
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        
        # Early stopping for unproductive searches
        if n_attempt % 50 == 0:
            if time.time() - start_time > budget_s - 120:
                break
    
    # Final constraint validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
