```python
import time
import numpy as np
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = 32
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize binary h with alternating 1s and 0s
    h = np.zeros(n_points, dtype=int)
    h[::2] = 1  # 16 ones and 16 zeros
    assert np.sum(h) == target_sum, "Initial sum incorrect"

    # Track indices of 1s and 0s using sets for O(1) random access
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])

    # Compute initial max overlap
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Simulated annealing parameters
    temp = 10.0
    cooling_rate = 0.9999
    min_temp = 1e-6
    iteration = 0

    # Main loop
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % 10000 == 0:
            print(f"Iteration {iteration}, temp {temp:.4f}, c5 {best_c5:.6f}")

        if not ones or not zeros:
            break  # impossible due to fixed sum constraint

        # Select random indices to swap
        idx_one = random.choice(list(ones))
        idx_zero = random.choice(list(zeros))

        # Perform swap
        h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]

        # Update sets
        ones.remove(idx_one)
        zeros.remove(idx_zero)
        ones.add(idx_zero)
        zeros.add(idx_one)

        # Compute new C5
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5

        if delta > 0:
            # Metropolis acceptance
            if temp > 1e-9 and random.random() < np.exp(-delta / temp):
                # Accept worse solution
                if new_c5 > best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                # Reject, revert swap
                h[idx_one], h[idx_zero] = h[idx_zero], h[idx_one]
                ones.remove(idx_zero)
                zeros.remove(idx_one)
                ones.add(idx_one)
                zeros.add(idx_zero)
        else:
            # Improvement
            best_c5 = new_c5
            best_h = h.copy()

        # Cool down
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp

        # Check time periodically
        if iteration % 1000 == 0:
            if time.time() - start_time > budget_s:
                break

    # Validate and return
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.astype(float).tolist(), best_c5, n_points)
```
