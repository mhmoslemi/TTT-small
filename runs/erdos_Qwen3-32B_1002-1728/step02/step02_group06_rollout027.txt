            fitness.append(np.max(corr) * dx)
        # Update best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()

        # Selection and replacement
        elite_size = int(0.2 * population_size)
        elite_indices = np.argsort(fitness)[:elite_size]
        non_elite_indices = np.argsort(fitness)[elite_size:]
        remaining_size = len(non_elite_indices)

        # Generate offspring
        offspring = []
        for _ in range(remaining_size):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            alpha = np.random.uniform(0.3, 0.7)
            child = alpha * parent1 + (1 - alpha) * parent2
            child = repair_population(child, target_sum)
            # Mutation
            if np.random.rand() < 0.1:
                child += np.random.normal(0, 0.05, n_points)
                child = repair_population(child, target_sum)
            offspring.append(child)

        # Replace non-elites
        for i, idx in enumerate(non_elite_indices):
            population[idx] = offspring[i]

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    return best_h, c5_bound, n_points
