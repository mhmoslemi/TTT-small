        'ftol': 1e-8,
        'disp': False,
        'iprint': 0,
        'callback': callback
    }

    result = minimize(
        fun=objective,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds_obj,
        constraints=constraints,
        options=options
    )

    h_opt = best_h.copy()
    h_opt = np.clip(h_opt, 0.0, 1.0)
    current_sum = np.sum(h_opt)
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        scale = target_sum / current_sum
        h_opt = h_opt * scale
        h_opt = np.clip(h_opt, 0.0, 1.0)

    corr_values = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
    return h_opt, np.max(corr_values), n_points
