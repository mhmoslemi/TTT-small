    n_points = len(h_current)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project_h(h, target_sum):
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        if abs(current_sum - target_sum) < 1e-8:
            return h
        delta = target_sum - current_sum
        # Adjust bottom or top elements to fix sum
        if delta > 0:
            idxs = np.where(h < 1.0)[0]
            if len(idxs) == 0:
                return h
            add = np.clip(delta / len(idxs), 0, 1 - h.min())
            h[idxs] += add
        elif delta < 0:
            idxs = np.where(h > 0.0)[0]
            if len(idxs) == 0:
                return h
            sub = np.clip(-delta / len(idxs), 0, h.min())
            h[idxs] -= sub
        h = np.clip(h, 0.0, 1.0)
        return h

    def compute_gradient(h, lag):
        n = len(h)
        grad = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1.0 - h[i + lag])
                grad[i + lag] -= h[i]
        else:
            m = -lag
            for i in range(m, n):
                grad[i] += (1.0 - h[i - m])
                grad[i - m] -= h[i]
        return grad * dx

    initializations = [h_current.copy()]
    # Binary random samples
    for _ in range(2):
        active = np.random.choice(n_points, size=int(n_points/2), replace=False)
        h_rand = np.zeros(n_points)
        h_rand[active] = 1.0
        initializations.append(h_rand)
    # Periodic patterns
    for p in [3,4,6,8,10]:
        pattern = np.zeros(n_points)
        half_p = p // 2
        for i in range(n_points):
            if i % p < half_p:
                pattern[i] = 1.0
        initializations.append(pattern)

    best_h = h_current.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    start_time = time.time()

    for init in initializations:
        if time.time() - start_time > budget_s - 5:
            break
        h = project_h(init.copy(), target_sum)
        velocity = np.zeros(n_points)
        lr = 0.02
        max_iterations = int(budget_s * 0.5)
        for iter in range(max_iterations):
            if time.time() - start_time > budget_s - 1:
                break
            corr_full = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr_full)
            lag = max_idx - (n_points - 1)
            grad = compute_gradient(h, lag)
            # Add L2 regularization (gradient descent + momentum)
            momentum = 0.9
            velocity = momentum * velocity + grad
            h -= lr * velocity
            h = project_h(h, target_sum)
            c5 = np.max(corr_full) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.9999

    return best_h, best_c5, n_points
