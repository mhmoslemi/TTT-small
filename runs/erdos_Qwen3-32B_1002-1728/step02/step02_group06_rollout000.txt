import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_current.copy()
    best_energy = objective(best_h)

    T = 0.1
    alpha = 0.999
    start_time = time.time()

    while time.time() - start_time <= budget_s:
        if T < 1e-6:
            break

        current_energy = objective(h_current)
        delta = np.random.normal(0, 0.1 * T, n_points)
        h_perturbed = h_current + delta
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        if abs(current_sum - target_sum) > 1e-8:
            scale = target_sum / current_sum
            h_perturbed *= scale
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        new_energy = objective(h_perturbed)

        if new_energy < best_energy:
            best_h[:] = h_perturbed[:]
            best_energy = new_energy

        delta_energy = new_energy - current_energy
        if delta_energy < 0:
            h_current[:] = h_perturbed[:]
        else:
            if np.random.rand() < np.exp(-delta_energy / T):
                h_current[:] = h_perturbed[:]

        T *= alpha

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h, np.max(corr_values), n_points
