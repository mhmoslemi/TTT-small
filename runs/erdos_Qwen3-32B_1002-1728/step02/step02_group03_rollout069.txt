import numpy as np
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    def scale_to_sum(h):
        s = np.sum(h)
        if abs(s) < 1e-10:
            return h
        scale = target_sum / s
        h_scaled = h * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    # Evolutionary search parameters
    pop_size = 40
    generations = 1500
    mutation_rate = 0.04
    mutation_std = 0.1
    tournament_size = 5
    elitism_ratio = 0.2

    def evaluate(candidates):
        c5_values = []
        for h in candidates:
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5_values.append(np.max(corr))
        return np.array(c5_values)

    def mutate(h):
        h = np.copy(h)
        mask = np.random.rand(n_points) < mutation_rate
        h[mask] += np.random.normal(0, mutation_std, size=np.sum(mask))
        h = np.clip(h, 0, 1)
        h = scale_to_sum(h)
        return h

    def crossover(parent1, parent2):
        alpha = np.random.rand()
        child = alpha * parent1 + (1 - alpha) * parent2
        child = scale_to_sum(child)
        return child

    def tournament_select(candidates, scores, size):
        indices = np.random.choice(len(candidates), size)
        winner_idx = indices[np.argmin(scores[indices])]
        return candidates[winner_idx]

    population = [h_initial]
    for _ in range(pop_size - 1):
        mut = h_initial + np.random.normal(0, 0.05, n_points)
        mut = scale_to_sum(np.clip(mut, 0, 1))
        population.append(mut)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = np.inf
    start_time = time.time()

    while time.time() - start_time < budget_s - 120:  # Save time for gradient step
        scores = evaluate(population)
        curr_best_idx = np.argmin(scores)
        if scores[curr_best_idx] < best_c5:
            best_c5 = scores[curr_best_idx]
            best_h = population[curr_best_idx].copy()

        elitism_size = max(1, int(elitism_ratio * pop_size))
        elitists = population[np.argsort(scores)[:elitism_size]].copy()

        new_population = elitists.copy()
        while len(new_population) < pop_size:
            parent1 = tournament_select(population, scores, tournament_size)
            parent2 = tournament_select(population, scores, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population = np.vstack([new_population, child])

        population = new_population[:pop_size]

    # Gradient refinement on best_h
    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Ensure initial point is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = scale_to_sum(best_h)

    # Constraints for optimization
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = Bounds(0, 1)

    try:
        result = minimize(
            objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-6},
            tol=1e-6
        )
        if result.success:
            optimized_h = result.x.reshape((n_points,))
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_h = scale_to_sum(optimized_h)
            corr_values = np.correlate(optimized_h, 1 - optimized_h, mode='full') * dx
            final_c5 = np.max(corr_values)
            best_h = optimized_h
            best_c5 = final_c5
    except Exception:
        pass  # Fallback to evolutionary best

    return best_h, best_c5, n_points
