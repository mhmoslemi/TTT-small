import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        # Penalize large deviations from the desired sum
        penalty = max(0, abs(np.sum(h) - target_sum) - 1e-5) * 1e6
        return c5 + penalty

    def constraint_func(x):
        return np.sum(x) - target_sum

    # Setup inequality constraint for the sum
    cons = (
        {'type': 'eq', 'fun': constraint_func}
    )

    # Differential Evolution with bounds and constraints
    result_de = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        maxiter=300,
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=False,
        init='sobol',
        seed=seed,
        callback=lambda x, c: time.time() - start_time > budget_s - 60
    )

    best_h_de = result_de.x.copy().reshape(n_points)
    
    # Post-process to project back to feasible region
    h_proj = np.clip(best_h_de / np.sum(best_h_de) * target_sum, 0, 1)
    # Local optimization refinement
    result = minimize(
        fun=objective,
        x0=h_proj,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    best_h = result.x.reshape(n_points)
    best_h = np.clip(best_h / np.sum(best_h) * target_sum, 0, 1)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(final_corr) * dx

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-5
    assert np.all((best_h >= -1e-7) & (best_h <= 1.0 + 1e-7))
    
    return (best_h.tolist(), best_c5, n_points)
