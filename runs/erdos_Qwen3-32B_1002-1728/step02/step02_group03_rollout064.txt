def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # parameters
    alpha = 0.01
    max_iter = 10000
    start_time = time.time()

    best_h = h.copy()
    best_c5 = np.inf

    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # compute overlaps for all possible k
        corr = np.correlate(h, 1 - h, mode='full') * dx
        positive_k_start = n_points - 1
        if len(corr) > positive_k_start:
            positive_corr = corr[positive_k_start:]
        else:
            positive_corr = corr
        if len(positive_corr) ==0:
            continue
        k_max_index = np.argmax(positive_corr)
        k_max = k_max_index  # since index in positive_corr is k
        current_c5 = positive_corr[k_max_index]

        # compute gradient for k_max
        gradient = np.zeros(n_points)
        # for each i where i +k_max <n_points
        for i in range(n_points - k_max):
            j = i + k_max
            gradient[i] += (1 - h[j]) * dx
            gradient[j] -= h[i] * dx

        # gradient descent step
        h_new = h - alpha * gradient
        # clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # project to feasible sum
        sum_h = np.sum(h_new)
        if abs(sum_h) < 1e-10:
            continue
        scale_factor = target_sum / sum_h
        h_new *= scale_factor
        h_new = np.clip(h_new, 0, 1)

        h = h_new

        # update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    return (best_h, final_c5, n_points)
