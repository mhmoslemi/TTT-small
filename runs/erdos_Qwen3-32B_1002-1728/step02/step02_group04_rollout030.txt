import numpy as np
import time

def compute_c5(h, dx):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def ensure_feasible(candidate, n_points, required_sum):
    candidate = np.clip(candidate, 0, 1)
    current_sum = np.sum(candidate)
    if current_sum > 1e-10:
        scaling = required_sum / current_sum
        candidate *= scaling
        candidate = np.clip(candidate, 0, 1)
    offset = np.sum(candidate) - required_sum
    if offset != 0:
        candidate[0] -= offset
    return candidate

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    np.random.seed(seed)
    h = ensure_feasible(h_initial, n_points, required_sum)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    start_time = time.time()
    alpha = 0.02
    decay_rate = 0.93
    decay_steps = 50
    perturb_interval = 150
    perturb_std = 0.07
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        # Compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        k_val = idx_max - (n_points - 1)

        # Gradient for worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_val) % n_points
            m = (i - k_val) % n_points
            grad[i] = (1 - h[j]) - h[m]
        h = h - alpha * grad
        h = ensure_feasible(h, n_points, required_sum)

        # Perturbation and learning rate
        if iteration % perturb_interval == 0:
            mi = np.random.choice(n_points, size=int(0.15 * n_points), replace=False)
