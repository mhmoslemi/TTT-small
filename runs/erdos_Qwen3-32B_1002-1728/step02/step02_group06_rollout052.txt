import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    current_h = best_h.copy()
    current_c5 = best_c5

    T = 1.0
    alpha = 0.9995  # Adjust cooling rate based on problem difficulty
    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        # Perturb current_h
        perturbation = np.random.normal(0, T * 0.1, size=n_points)
        h_perturbed = current_h + perturbation

        # Repair constraints
        h_perturbed = np.clip(h_perturbed, 0, 1)
        s = h_perturbed.sum()
        if s > 1e-10:
            scale = target_sum / s
            h_perturbed *= scale
            h_perturbed = np.clip(h_perturbed, 0, 1)

        # Compute new_c5
        new_c5 = compute_c5(h_perturbed)

        # Acceptance criterion
        delta_E = new_c5 - current_c5
        if delta_E < 0:
            current_h = h_perturbed.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            if np.random.rand() < np.exp(-delta_E / T):
                current_h = h_perturbed.copy()
                current_c5 = new_c5

        # Cool down
        T *= alpha
        if T < 1e-5:
            break

    return best_h, best_c5, n_points
