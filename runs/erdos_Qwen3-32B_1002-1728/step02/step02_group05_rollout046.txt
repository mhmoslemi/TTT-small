import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    low = 0.0
    high = 0.38085  # Current best bound
    best_c5 = high
    best_h = h_initial.copy()
    start_time = time.time()
    max_iter = 20

    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 20:
            break
        C_mid = (low + high) / 2

        h = cp.Variable(n_points, nonneg=True)
        u = cp.Variable((n_points, n_points), nonneg=True)

        constraints = [
            cp.sum(h) == n_points / 2,
            h <= 1
        ]

        for i in range(n_points):
            for j in range(n_points):
                if i != j:
                    constraints += [
                        u[i, j] >= h[i] + h[j] - 1,
                        u[i, j] <= h[i],
                        u[i, j] <= h[j]
                    ]

        for k in range(n_points):
            if k == 0:
                indices = [(i, i) for i in range(n_points)]
            else:
                indices = [(i, i + k) for i in range(n_points - k)]
            sum_u = sum(u[i, j] for i, j in indices)
            constraints.append(sum_u <= C_mid / dx)

        problem = cp.Problem(cp.Minimize(0), constraints)
        problem.solve(solver=cp.ECOS, verbose=False)

        if problem.status == cp.OPTIMAL:
            feasible_h = h.value
            best_h = feasible_h.copy()
            best_c5 = C_mid
            high = C_mid
        else:
            low = C_mid

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
