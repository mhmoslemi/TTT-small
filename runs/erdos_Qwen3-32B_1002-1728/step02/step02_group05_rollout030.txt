    dx = 2.0 / n_points

    def compute_C(h_array, dx_val):
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx_val

    best_h = h_initial.copy()
    best_c = compute_C(best_h, dx)
    start_time = time.time()

    low = 0.0
    high = best_c
    tolerance = 1e-6
    num_steps = 0
    max_steps = 100

    while num_steps < max_steps and (time.time() - start_time) < budget_s - 10:
        num_steps += 1
        mid = (low + high) / 2
        # Build and solve the LP problem
        h = cp.Variable(n_points)
        u = cp.Variable((n_points, n_points - 1))
        constraints = [
            cp.sum(h) == n_points / 2.0,
        ]
        for i in range(n_points):
            constraints += [0 <= h[i], h[i] <= 1]

        feasible = False
        for k_idx in range(n_points - 1):
            k = k_idx + 1
            for i in range(n_points):

                constraints += [

                    u[i, k_idx] <= h[i],
                    u[i, k_idx] <= 1 - h[j],
                ]
            constraints += [cp.sum(u[:, k_idx]) * dx <= mid]

        prob = cp.Problem(cp.Minimize(0.), constraints)

                feasible = True
        except:
            feasible = False

        if feasible:
            high = mid
        else:
            low = mid

import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def compute_C(h_array):
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    best_c = compute_C(best_h)
    start_time = time.time()

    low, high = 0.0, best_c
    step = 0

    while (time.time() - start_time) < budget_s - 5 and step < 30:
        mid = (low + high) / 2
        h = cp.Variable(n_points)
        u = cp.Variable((n_points, n_points - 1))

        constraints = [
            cp.sum(h) == n_points / 2.0,
            [0 <= h[i] for i in range(n_points)] + [h[i] <= 1 for i in range(n_points)]
        ]
        for k in range(n_points - 1):
            for i in range(n_points):
                j = (i + k + 1) % n_points
                constraints.append(u[i, k] >= h[i] + (1 - h[j]) - 1)
                constraints.append(u[i, k] <= h[i])
                constraints.append(u[i, k] <= 1 - h[j])
            constraints.append(cp.sum(u[:, k]) * dx <= mid)

        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, max_iter=100, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                sol = h.value.clip(0, 1)
                val = compute_C(sol)
                if val < best_c:
                    best_c = val
                    best_h[:] = sol[:]
                high = mid
            else:
                low = mid
        except:
            low = mid
        step += 1

    best_h = np.clip(best_h, 0, 1)
    offset = np.sum(best_h) - n_points / 2
    best_h[0] -= offset
    best_h[0] = np.clip(best_h[0], 0, 1)
