import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    best_c5 = None
    best_h = h_initial.copy()
    start_time = time.time()

    # Generate structured initial guesses
    initial_guesses = []

    # Square waves
    periods_square = [3,4,5,6,7]
    for period in periods_square:
        for phase in range(period):
            h = np.zeros(n_points)
            indices = ((np.arange(n_points) - phase) % period) == 0
            h[indices] = 1.0
            target_sum = n_points / 2.0
            current_sum = np.sum(h)
            if current_sum > 1e-8:
                scale = target_sum / current_sum
                h_scaled = h * scale
                h_scaled = np.clip(h_scaled, 0, 1)
                offset = target_sum - np.sum(h_scaled)
                h_scaled[0] += offset
                h_scaled[0] = max(0.0, min(1.0, h_scaled[0]))
                initial_guesses.append(h_scaled.copy())

    # Sinusoidal waves
    periods_sine = [3,4,5,6]
    for period in periods_sine:
        x = 2 * np.pi * np.arange(n_points) / period
        wave = np.sin(x)
        scaled_wave = (wave - wave.min()) / (wave.max() - wave.min())  # normalize to 0-1
        current_sum = np.sum(scaled_wave)
        if current_sum > 1e-8:
            scale = n_points / 2.0 / current_sum
            h = scaled_wave * scale
            h = np.clip(h, 0, 1)
            offset = n_points / 2.0 - np.sum(h)
            h[0] += offset
            h[0] = max(0.0, min(1.0, h[0]))
            initial_guesses.append(h.copy())

    # Triangular waves
    periods_tri = [3,4,5,6,7]
    for period in periods_tri:
        for phase in range(period):
            x = (np.arange(n_points) - phase) % period
            wave = np.zeros(n_points)
            for i in range(n_points):
                if x[i] <= period/2:
                    wave[i] = x[i]
                else:
                    wave[i] = period - x[i]
            wave = wave / (period/2)  # normalize to 0-1
            current_sum = np.sum(wave)
            if current_sum > 1e-8:
                scale = n_points / 2.0 / current_sum
                h = wave * scale
                h = np.clip(h, 0, 1)
                offset = n_points / 2.0 - np.sum(h)
                h[0] += offset
                h[0] = max(0.0, min(1.0, h[0]))
                initial_guesses.append(h.copy())

    # Add original initial guess if not already included
    if not any(np.allclose(h_initial, guess) for guess in initial_guesses):
        initial_guesses.append(h_initial.copy())

    # Evaluate all generated initial guesses
    for x0 in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[sum_constraint],
            options={'maxiter': 150, 'ftol': 1e-7}
        )
        if not result.success:
            continue
        curr_h = np.clip(result.x.copy(), 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset
        curr_h[0] = max(0.0, min(1.0, curr_h[0]))
        corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_val)
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h[:] = curr_h[:]

    # Refine with perturbation and decaying noise
    perturbation_start = time.time()
    while time.time() - start_time < budget_s - 5:
        time_elapsed = time.time() - perturbation_start
        # Exponential decay of noise
        noise_sd = max(0.02, 0.15 * np.exp(- time_elapsed / 200))
        x0 = best_h.copy()
        x0 += np.random.normal(0, noise_sd, n_points)
        x0 = np.clip(x0, 0, 1)
        offset = np.sum(x0) - n_points/2
        x0[0] -= offset
        x0[0] = max(0.0, min(1.0, x0[0]))
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=[sum_constraint],
            options={'maxiter': 100, 'ftol': 1e-7}
        )
        if not result.success:
            continue
        curr_h = np.clip(result.x.copy(), 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset
        curr_h[0] = max(0.0, min(1.0, curr_h[0]))
        corr_val = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_val)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h[:] = curr_h[:]
            # After improvement, reset the perturbation time to allow longer decay
            perturbation_start = time.time()

    # Final validation and return
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
