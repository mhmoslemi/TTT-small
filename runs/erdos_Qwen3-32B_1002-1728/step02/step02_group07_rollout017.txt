    best_c5 = float('inf')
    best_h = h.copy()

    np.random.seed(seed)
    start_time = time.time()
    lambda_ = 1e-4  # Regularization to ensure convexity

    while time.time() - start_time < budget_s - 10:
        # Compute current C5 and find the max overlap index
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        k_max = np.argmax(corr_vals)
        overlap_k = corr_vals[k_max]

        # Compute gradient of overlap_k w.r.t h using finite differences
        eps = 1e-5
        n = n_points
        gradient = np.zeros(n)
        for i in range(n):
            h_pert = h.copy()
            h_pert[i] += eps
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full') * dx
            overlap_pert = corr_pert[k_max]
            gradient[i] = (overlap_pert - overlap_k) / eps

        # Form and solve convex QP
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(
            gradient @ h_var + 0.5 * cp.sum_squares(h_var - h) * lambda_
        )
        constraints = [
            cp.sum(h_var) == target_sum,
            0 <= h_var,
            h_var <= 1
        ]
        problem = cp.Problem(obj, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, max_iter=1000)
        except:
            h = best_h.copy()
            continue

        if h_var.value is not None:
            h = h_var.value.copy()
            # Enforce constraints more strictly
            h = np.clip(h, 0, 1)
            current_sum = h.sum()
            if np.abs(current_sum - target_sum) > 1e-6:
                h = h * target_sum / current_sum

    return best_h, best_c5, n_points
