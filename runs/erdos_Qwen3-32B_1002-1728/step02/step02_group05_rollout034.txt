    # Initialize with initial_h_values

    # PSO parameters

    c1 = 2.0  # cognitive weight
    c2 = 2.0  # social weight

    # Initialize swarm
    positions = []
    velocities = []
    pbest_positions = []
    pbest_fitnesses = []

    # Create initial particles
    # Half from initial_h_values, half random
    for i in range(n_particles):
        if i < n_particles // 2:
            h = h_initial.copy()
            h += np.random.uniform(-0.1, 0.1, h.shape)  # small random perturbation
        else:
            h = np.random.uniform(0.0, 1.0, n_points)
        # Enforce sum constraint
        h = enforce_sum(h, target_sum)
        positions.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))  # Initial velocities
        pbest_positions.append(h.copy())
        pbest_fitnesses.append(fitness_function(h, dx))

    # Find initial global best
    gbest_index = np.argmin(pbest_fitnesses)
    gbest_position = pbest_positions[gbest_index]
    gbest_fitness = pbest_fitnesses[gbest_index]

    # Start time
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        # Calculate time elapsed
        elapsed_time = time.time() - start_time
        # Update inertia weight with cooling schedule
        w = max(0.4, w - cooling_rate * elapsed_time)

        for idx in range(n_particles):
            # Skip if time is up
            if time.time() - start_time > budget_s - 5:

            # Current position and velocity
            current_pos = positions[idx]

import numpy as np
import time

def enforce_sum(h, target_sum):
    h_clamped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clamped)
    delta = target_sum - current_sum
    h_clamped[0] += delta
    h_clamped[0] = np.clip(h_clamped[0], 0.0, 1.0)
    return h_clamped

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()

    target_sum = n_points / 2.0

    np.random.seed(seed)

    n_particles = 100
    c1, c2 = 2.0, 2.0
    w, w_min = 0.9, 0.4
    cooling_rate = (w - w_min) / (budget_s / 100.0)

    positions = []
    velocities = []
    pbest_positions = []
    pbest_fitness = []

    for i in range(n_particles):
        if i < n_particles // 2:
            h = h_initial.copy()
            h += np.random.uniform(-0.1, 0.1, n_points)
        else:
            h = np.random.uniform(0.0, 1.0, n_points)
        h = enforce_sum(h, target_sum)
        positions.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest_positions.append(h.copy())
        pbest_fitness.append(np.max(np.correlate(h, 1 - h, 'full')) * dx)

    gidx = np.argmin(pbest_fitness)
    gbest = pbest_positions[gidx].copy()
    gbest_fitness = pbest_fitness[gidx]

    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        elapsed = time.time() - start_time
        w = max(w_min, w - cooling_rate * elapsed)
        for i in range(n_particles):
            if time.time() - start_time > budget_s - 5:
                break
            pos = positions[i]
            vel = velocities[i]
            pb = pbest_positions[i]
            pb_f = pbest_fitness[i]

            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = c1 * r1 * (pb - pos)
            social = c2 * r2 * (gbest - pos)
            new_vel = w * vel + cognitive + social

            new_pos = pos + new_vel
            new_pos = enforce_sum(new_pos, target_sum)

            corr = np.correlate(new_pos, 1 - new_pos, 'full')
            new_f = np.max(corr) * dx

            if new_f < pb_f:
                pbest_positions[i] = new_pos.copy()
                pbest_fitness[i] = new_f

            if new_f < gbest_fitness:
                gbest[...] = new_pos[...]
                gbest_fitness = new_f

            positions[i] = new_pos
            velocities[i] = new_vel
