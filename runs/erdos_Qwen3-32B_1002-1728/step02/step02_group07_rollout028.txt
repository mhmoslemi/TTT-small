        h_adj[can_decr] -= D * h_adj[can_decr] / total_possible
    return h_adj

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)
    start_time = time.time()

    m_list = [3, 5, 9, 15, 45]
    for m in m_list:
        if n_points % m != 0:
            continue
        points_per_interval = n_points // m

        def objective_m(h_m):
            h_full = np.repeat(h_m, points_per_interval)
            corr_values = np.correlate(h_full, 1 - h_full, mode='full') * dx
            return np.max(corr_values)

        h_m_prev = best_h.reshape(m, points_per_interval).mean(axis=1)
        h_m_prev = adjust_sum(h_m_prev, m / 2)
        h_m_prev = np.clip(h_m_prev, 0.0, 1.0)

        sum_con = LinearConstraint(np.ones(m), lb=m / 2, ub=m / 2)
        bounds = Bounds(0.0, 1.0)

        result = minimize(
            objective_m,
            x0=h_m_prev,
            method='COBYLA',
            bounds=bounds,
            constraints=sum_con,
            options={'maxiter': 100}
        )

        if result.success or result.message == 'Maximum number of function evaluations has been exceeded.':
            h_m_opt = result.x
            h_m_opt = adjust_sum(h_m_opt, m / 2)
            h_m_opt = np.clip(h_m_opt, 0.0, 1.0)
            h_full_opt = np.repeat(h_m_opt, points_per_interval)
            h_full_opt = adjust_sum(h_full_opt, target_sum)
            corr_values = np.correlate(h_full_opt, 1 - h_full_opt, mode='full') * dx
            curr_c5 = np.max(corr_values)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h_full_opt.copy()

        if time.time() - start_time >= budget_s - 20:
            break

    return best_h, best_c5, n_points
