import time

import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    K = 64

    np.random.seed(seed)

    h = np.tile([1.0, 0.0], K//2)

    dx = 2.0 / K

    target_sum = K / 2.0

    def compute_c5(h_vec):

        corr = np.correlate(h_vec, 1 - h_vec, mode='full')

        return np.max(corr) * dx

    best_h = h.copy()

    best_c5 = compute_c5(best_h)

    current_h = h.copy()

    current_c5 = best_c5

    # Simulated Annealing parameters

    initial_temp = 1.0

    cooling_rate = 0.99

    temp = initial_temp

    while time.time() - start_time < budget_s * 0.95:

        # Pick two random indices

        i, j = np.random.choice(K, size=2, replace=False)

        if current_h[i] == current_h[j]:

            # Not useful, need to swap a 0 and 1

            continue

        # Swap values

        new_h = current_h.copy()

        new_h[i], new_h[j] = new_h[j], new_h[i]  # swap values

        # Compute new C5

        new_c5 = compute_c5(new_h)

        delta = new_c5 - current_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / temp):

            # Accept the move

            current_h = new_h

            current_c5 = new_c5

            if current_c5 < best_c5:

                best_h = current_h.copy()

                best_c5 = current_c5

        # Cool down

        temp *= cooling_rate

    # After loop, return the best solution

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
