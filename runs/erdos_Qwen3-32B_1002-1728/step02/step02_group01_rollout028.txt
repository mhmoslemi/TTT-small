import numpy as np
import time

def objective(h_vec, dx):
    corr = np.correlate(h_vec, 1 - h_vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial_h_values to meet constraints
    h = np.clip(h, 0.0, 1.0)
    sum_h = np.sum(h)
    if sum_h != 0.0:
        h = (h / sum_h) * target_sum
    h = np.clip(h, 0.0, 1.0)
    current_h = h.copy()
    best_h = h.copy()
    current_c5 = objective(current_h, dx)
    best_c5 = current_c5

    # Simulated Annealing Parameters
    T = 0.1  # Initial temperature
    alpha = 0.99  # Cooling factor
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 10:
        # Generate perturbation
        perturbed = current_h + np.random.normal(0, T * 0.05, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_pert = np.sum(perturbed)
        if sum_pert != 0.0:
            perturbed = (perturbed / sum_pert) * target_sum
        perturbed = np.clip(perturbed, 0.0, 1.0)

        new_c5 = objective(perturbed, dx)
        delta = new_c5 - current_c5

        if delta <= 0:
            current_h[:] = perturbed[:]
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h[:] = perturbed[:]
                best_c5 = new_c5
        else:
            prob = np.exp(-delta / T)
            if np.random.rand() < prob:
                current_h[:] = perturbed[:]
                current_c5 = new_c5

        T *= alpha  # Cool the temperature

    # Final constraint validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
