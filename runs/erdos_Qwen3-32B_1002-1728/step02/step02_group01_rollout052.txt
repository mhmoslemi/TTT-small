import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    budget_used = 0

    # Vary perturbation scales and optimize with different settings
    pert_scales = [0.1, 0.05, 0.02]
    n_attempts = 300  # Total attempts budgeted
    attempts = 0

    while time.time() - start_time < budget_s and attempts < n_attempts:
        attempts += 1
        sigma = pert_scales[attempts % len(pert_scales)]
        # Generate noisy starting point
        perturbed = best_h + np.random.normal(0, sigma, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Normalize to meet integral constraint
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        # Optimize with weighted parameters
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            new_h = result.x.reshape(n_points)
            new_h = new_h.copy()
            new_c5 = objective(new_h)
            # Validate constraints
            if (abs(np.sum(new_h) - target_sum) < 1e-6 and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
