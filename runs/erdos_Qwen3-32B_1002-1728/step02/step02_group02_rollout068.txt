import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    pop_size = 50

    def _evaluate(individual):
        corr = correlate(individual, 1 - individual, mode='full')
        return np.max(corr) * dx

    def _project(p, target_sum, n_points):
        clamped = np.clip(p, 0, 1)
        clamped_sum = clamped.sum()
        if clamped_sum == 0:
            return np.ones_like(clamped) * 0.5
        temp = clamped * (target_sum / clamped_sum)
        return np.clip(temp, 0, 1)

    # Initialize population
    population = []
    for _ in range(pop_size):
        if _ < 5:  # 10% perturbations
            mut = h_initial + np.random.normal(0, 0.05, size=n_points)
        elif _ < 15:  # 20% random
            mut = np.random.rand(n_points)
        else:  # 70% mutants
            mut = h_initial.copy()
            a, b = np.sort(np.random.randint(0, n_points, 2))
            mut[a:b+1] = 1 - mut[a:b+1]
        population.append(_project(mut, target_sum, n_points))
    population = np.stack(population)
    scores = np.array([_evaluate(ind) for ind in population])
    best_idx = np.argmin(scores)
    best_individual = population[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()
    generation = 0
    while time.time() - start_time < budget_s - 0.1:
        # Tournament selection
        def tournament():
            k = int(pop_size * 0.05)
            k = max(2, k)
            idx = np.random.choice(pop_size, k, replace=False)
            return idx[np.argmin(scores[idx])]

        p1_idx = tournament()
        p2_idx = tournament()
        p1 = population[p1_idx]
        p2 = population[p2_idx]

        # Crossover (5-point)
        points = np.sort(np.random.choice(n_points, 5, replace=False))
        child = np.empty_like(p1)
        parents = [p1, p2]
        current = 0
        prev = 0
        for pt in points:
            child[prev:pt] = parents[current][prev:pt]
            prev = pt
            current = 1 - current
        child[prev:] = parents[current][prev:]
        child = _project(child, target_sum, n_points)

        # Mutation
        m = np.random.rand()
        if m < 0.1:  # Jitter
            child += np.random.normal(0, 0.05, size=n_points)
        elif m < 0.3:  # Invert segment
            a, b = np.sort(np.random.randint(0, n_points, 2))
            child[a:b+1] = 1 - child[a:b+1]
        else:  # Slide
            a, b = np.sort(np.random.randint(0, n_points, 2))
            delta = np.random.uniform(-0.1, 0.1)
            child[a:b+1] += delta
        child = _project(child, target_sum, n_points)

        # Evaluate and replace worst
        score = _evaluate(child)
        worst_idx = np.argmax(scores)
        if score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = score
            if score < best_score:
                best_individual = child.copy()
                best_score = score

        generation += 1

    return best_individual, best_score, n_points
