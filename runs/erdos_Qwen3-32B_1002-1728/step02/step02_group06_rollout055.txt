import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_current = initial_h_values.copy()
    n_points = len(h_current)
    dx = 2.0 / n_points

    def compute_c5(h):
        h1 = 1 - h
        corr = np.correlate(h, h1, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()

    temp = 0.1
    alpha = 0.9999
    pert_scale = 0.2
    start_time = time.time()

    while time.time() - start_time < budget_s - 5:
        # Generate perturbation
        perturb = np.random.normal(0, temp * pert_scale, n_points)
        h_new = h_current + perturb
        h_new = np.clip(h_new, 0.0, 1.0)
        s = np.sum(h_new)
        if abs(s) < 1e-10:
            h_new = h_current.copy()
        else:
            scale = (n_points / 2.0) / s
            h_new *= scale
            h_new = np.clip(h_new, 0.0, 1.0)
        # Compute new cost
        new_c5 = compute_c5(h_new)
        # Acceptance criteria
        if new_c5 < best_c5:
            h_current = h_new.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            delta = best_c5 - new_c5
            if np.random.rand() < np.exp(delta / temp):
                h_current = h_new.copy()
        temp *= alpha

    # Final constraint enforcer
    if abs(np.sum(best_h) - n_points / 2.0) > 1e-8:
        scale = (n_points / 2.0) / np.sum(best_h)
        best_h *= scale
        best_h = np.clip(best_h, 0.0, 1.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
