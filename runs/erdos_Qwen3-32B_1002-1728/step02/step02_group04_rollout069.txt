        h = np.clip(h, 0, 1)

    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        overlaps = []
        for k in range(1, n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += best_h[i] * best_h[j]
            overlaps.append(-overlap)
        k_worst = np.argmin(overlaps) + 1

        P = build_qp_matrix(n_points, k_worst)
        Pcvx = matrix(P)
        qcvx = matrix(0.0, (n_points, 1))
        G_upper = matrix(-np.eye(n_points).astype(float))
        G_lower = matrix(np.eye(n_points).astype(float))
        G = matrix(np.vstack([G_upper, G_lower]))
        h_c = matrix(np.vstack([np.zeros((n_points,1)), np.ones((n_points,1))]))
        A = matrix(np.ones(n_points).astype(float), (1,n_points))
        b = matrix([required_sum])
        solvers.options["show_progress"] = False
        sol = solvers.qp(Pcvx, qcvx, G, h_c, A, b)
        new_h = np.array(sol["x"]).flatten()

        new_h = project_to_feasible(new_h, n_points, required_sum)
        perturbation = np.random_normal(0, 0.05, n_points)
        new_h += perturbation
        new_h = project_to_feasible(new_h, n_points, required_sum)

        current_c5 = compute_c5(new_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = new_h.copy()

        if time.time() - start_time + 10 > budget_s:
            break

    return project_to_feasible(best_h, n_points, required_sum), best_c5, n_points
