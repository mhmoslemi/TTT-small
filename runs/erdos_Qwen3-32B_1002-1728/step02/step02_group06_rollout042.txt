    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    delta = 0.05

    initial_guesses = [h_initial.copy()]
    p_list = [5, 7, 12]  # different periods for structured patterns
    for p in p_list:
        pattern = generate_periodic_pattern(p, n_points)
        adjusted_pattern = pattern.copy()
        adjusted_pattern[0] += (target_sum - adjusted_pattern.sum())
        adjusted_pattern = np.clip(adjusted_pattern, 0, 1)
        initial_guesses.append(adjusted_pattern)
    initial_guesses.append(generate_random_binary_like(n_points, target_sum))
    initial_guesses.append(generate_random_array(n_points, target_sum))

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    for init in initial_guesses:
        if time.time() - start_time > budget_s - 10:
            break
        h = init.copy()
        # Ensure sum constraint
        current_sum = h.sum()
        h[0] += (target_sum - current_sum)
        h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        no_progress_threshold = 1000
        no_progress = 0
        while time.time() - start_time < budget_s - 10:
            h_before = h.copy()
            coords = np.random.permutation(n_points)
            for j in coords:
                if time.time() - start_time > budget_s - 0.1:
                    break
                # Try +delta
                temp_h_plus = h.copy()
                temp_h_plus[j] = np.clip(temp_h_plus[j] + delta, 0, 1)
                current_sum = temp_h_plus.sum()
                temp_h_plus[0] += (target_sum - current_sum)
                temp_h_plus = np.clip(temp_h_plus, 0, 1)
                c5_plus = compute_c5(temp_h_plus, dx)

                # Try -delta
                temp_h_minus = h.copy()
                temp_h_minus[j] = np.clip(temp_h_minus[j] - delta, 0, 1)
                current_sum = temp_h_minus.sum()
                temp_h_minus[0] += (target_sum - current_sum)
                temp_h_minus = np.clip(temp_h_minus, 0, 1)
                c5_minus = compute_c5(temp_h_minus, dx)

                # Choose the best option
                best_val = current_c5
                best_h_choice = h.copy()
                if c5_plus < best_val:
                    best_val = c5_plus
                    best_h_choice = temp_h_plus
                if c5_minus < best_val:
                    best_val = c5_minus
                    best_h_choice = temp_h_minus
                h = best_h_choice.copy()
                current_c5 = best_val

            if np.allclose(h, h_before):
                no_progress += 1
                if no_progress > no_progress_threshold:
                    break
            else:
                no_progress = 0

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    return best_h, best_c5, n_points
