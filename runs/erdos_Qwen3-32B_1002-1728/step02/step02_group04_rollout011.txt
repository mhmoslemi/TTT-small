    decay_steps = 100

    while time.time() - start_time < budget_s:
        cur_iter += 1
        # Find worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)
        # Update h
        grad = compute_gradient(h, k, n_points)
        h = h - alpha * grad
        h = project(h, required_sum)
        # Update best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        # Perturb
        if cur_iter % perturb_interval == 0:
            h += np.random.normal(0, 0.05, n_points)
            h = project(h, required_sum)
        # Decay alpha
        if cur_iter % decay_steps == 0:
            alpha *= decay_rate
        # Budget check
        if time.time() - start_time >= budget_s:
            break
    return best_h, best_c5, n_points
