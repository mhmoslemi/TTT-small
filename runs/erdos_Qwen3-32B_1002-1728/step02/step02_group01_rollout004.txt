import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = np.array([(0.0, 1.0)] * n_points)
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def compute_c5(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, 'full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    np.random.seed(seed)
    n_pop = 20
    population = [h.copy()]

    # Generate diverse starting candidates
    for _ in range(n_pop - 1):
        base = population[0].copy()
        pert = np.random.choice(['gaussian', 'boundary', 'block'])
        if pert == 'gaussian':
            sigma = 0.1 + 0.05 * np.random.rand()
            candidate = base + np.random.normal(0, sigma, n_points)
        elif pert == 'boundary':
            n_jitter = int(min(20, 0.1 * n_points))
            idxs = np.random.choice(n_points, n_jitter, replace=False)
            candidate = base.copy()
            candidate[idxs] = np.random.choice([0, 1], n_jitter)
        else:  # block perturbation
            start_idx = np.random.randint(0, n_points - 10)
            length = np.random.randint(5, 15)
            end_idx = min(start_idx + length, n_points)
            candidate = base.copy()
            candidate[start_idx:end_idx] += np.random.normal(0, 0.3, end_idx - start_idx)
            
        # Normalize and clip
        candidate = np.clip(candidate, 0, 1)
        norm = candidate.sum()
        if norm > 1e-8:
            candidate = candidate * (target_sum / norm)
        candidate = np.clip(candidate, 0, 1)
        population.append(candidate)
    
    while time.time() - start_time < budget_s - 60:
        if not population:
            break
            
        current_h = population.pop(0).copy()
        
        # Adaptive optimization parameters
        remaining = budget_s - (time.time() - start_time)
        max_iter = int(100 if remaining > 600 else 50)
        
        result = minimize(
            compute_c5, 
            current_h, 
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter, 'ftol': 1e-8, 'disp': False}
        )
        
        if result.success:
            new_h = result.x.reshape(n_points)
            # Projection step
            new_h = np.clip(new_h, 0, 1)
            new_sum = new_h.sum()
            if new_sum > 1e-8:
                new_h = new_h * (target_sum / new_sum)
            new_h = np.clip(new_h, 0, 1)
            
            new_c5 = compute_c5(new_h)
            
            # Validate bounds
            valid = (np.allclose(new_h.sum(), target_sum, atol=1e-6) and 
                    np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6)))
            
            if valid and new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5
                
                # Add local perturbations around improved solution
                for _ in range(2):
                    if time.time() - start_time < budget_s - 30:
                        idxs = np.random.choice(n_points, int(n_points * 0.3), replace=False)
                        sigma = 0.02 * (0.95 ** (time.time() - start_time))
                        pert = best_h.copy()
                        pert[idxs] += np.random.normal(0, sigma, len(idxs))
                        pert = np.clip(pert, 0, 1)
                        pert = pert * (target_sum / pert.sum())
                        population.append(pert)
        
        # Keep population refreshed
        if len(population) < 5 and time.time() - start_time < budget_s - 100:
            base = best_h.copy()
            pert = np.random.choice(['gaussian', 'boundary'])
            if pert == 'gaussian':
                sigma = 0.05 * (0.9 ** (time.time() - start_time))
                candidate = base + np.random.normal(0, sigma, n_points)
            else:
                n_switch = int(n_points * 0.15)
                flip_idx = np.random.choice(n_points, n_switch, replace=False)
                candidate = base.copy()
                candidate[flip_idx] = 1 - candidate[flip_idx]
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (target_sum / candidate.sum())
            population.append(candidate)
    
    # Final validation
    assert np.allclose(best_h.sum(), target_sum, atol=1e-6)
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
