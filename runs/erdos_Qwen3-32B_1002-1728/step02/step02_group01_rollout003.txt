import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    time_budget = budget_s - 60  # Reserve some time for termination
    n_attempt = 0

    while time.time() - start_time < time_budget:
        n_attempt += 1
        rnd = np.random.RandomState(seed + n_attempt)
        perturbation_method = 'gaussian' if n_attempt % 3 == 0 else 'block' if n_attempt % 3 == 1 else 'boundary'
        perturbed = h.copy()

        # Generate perturbation
        if perturbation_method == 'gaussian':
            noise = rnd.normal(0, 0.15, size=n_points)
            perturbed += noise
        elif perturbation_method == 'block':
            idxs = np.argsort(rnd.rand(n_points))[:n_points // 5]
            perturbed[idxs] = 1 - perturbed[idxs]
        else:  # 'boundary'
            mask = (rnd.rand(n_points) < 0.5).astype(int)
            perturbed = np.where(mask, 0.95, 0.05)

        # Enforce integral constraint
        total = np.sum(perturbed)
        if abs(total) < 1e-10:
            perturbed[:] = 0.5 / dx * h.shape[0]
        else:
            perturbed = perturbed * (target_sum / total)
        perturbed = np.clip(perturbed, 0, 1)

        # Optimization
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )

        if result.success:
            new_h = result.x
            new_c5 = objective(new_h)
            if (np.isclose(np.sum(new_h), target_sum) and
                np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
