import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points

    def objective_de(x):
        x = x.reshape(n_points)
        s = x.sum()
        if s == 0:
            scaled = x.copy()
        else:
            scaled = x * (target_sum / s)
        clipped = np.clip(scaled, 0, 1)
        s_clipped = clipped.sum()
        # Use penalty to encourage the constraint
        penalty = 1e4 * abs(s_clipped - target_sum)
        corr = np.correlate(clipped, 1 - clipped, mode='full')
        return np.max(corr) * dx + penalty

    # Run differential evolution with constraint-encouraging objective
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        popsize=15,
        maxiter=30,
        polish=False,
        x0=h.flatten() if h.size > 0 else None,
        tol=1e-4,
        updating='deferred'
    )

    # Refine with SLSQP using strict constraint
    def objective_slsqp(x):
        x = x.reshape(n_points)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    x_de = result_de.x
    x_de_sum = x_de.sum()
    if x_de_sum > 1e-6:  # Safe scaling
        x_de = (x_de / x_de_sum) * target_sum
    x_de = np.clip(x_de, 0, 1)

    result_slsqp = minimize(
        objective_slsqp,
        x0=x_de,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )

    best_h = result_slsqp.x
    best_c5 = result_slsqp.fun

    # Ensure we stay within time budget and validate constraints
    if time.time() - start_time > budget_s - 10:
        # Last-minute validation fallback
        def fallback_objective(x):
            return np.max(np.correlate(x, 1 - x, mode='full')) * dx
        result_fallback = minimize(
            fallback_objective,
            x0=initial_h_values,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-6, 'disp': False}
        )
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
        if result_fallback.success and result_fallback.fun < best_c5:
            best_h = result_fallback.x
            best_c5 = result_fallback.fun

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
