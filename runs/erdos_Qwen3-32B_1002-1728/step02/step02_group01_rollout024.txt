        corr = np.correlate(h, 1 - h, mode='full')

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    def generate_candidate(h, perturbation_type, sigma, seed):
        if perturbation_type == 'gaussian':
            h_candidate = h.copy()
            rnd = np.random.RandomState(seed)
            noise = rnd.normal(0, sigma, n_points)
            h_candidate += noise
            h_candidate = np.clip(h_candidate, 0, 1)
            total = h_candidate.sum()
            if total != 0:
                h_candidate = (h_candidate / total) * target_sum
            else:
                h_candidate = h.copy()
        elif perturbation_type == 'boundary':
            h_candidate = h.copy()
            rnd = np.random.RandomState(seed)
            idxs = rnd.choice(n_points, size=min(10, n_points), replace=False)
            h_candidate[idxs] = rnd.randint(0, 2, size=idxs.shape)
            h_candidate = np.clip(h_candidate, 0, 1)
            total = h_candidate.sum()
            if total != 0:
                h_candidate = (h_candidate / total) * target_sum
            else:
                h_candidate = h.copy()
        elif perturbation_type == 'block':
            h_candidate = h.copy()
            rnd = np.random.RandomState(seed)
            block_len = rnd.randint(1, n_points // 2)
            start_idx = rnd.randint(0, n_points - block_len)
            end_idx = start_idx + block_len
            shift = rnd.normal(0, sigma, 1)
            h_candidate[start_idx:end_idx] += shift
            h_candidate = np.clip(h_candidate, 0, 1)
            total = h_candidate.sum()
            if total != 0:
                h_candidate = (h_candidate / total) * target_sum
            else:
                h_candidate = h.copy()
        else:
            raise ValueError("Unknown perturbation type")
        return h_candidate

    population = []
    base_sigma = 0.2
    np.random.seed(seed)
    for i in range(20):
        pt_type = np.random.choice(['gaussian', 'boundary', 'block'])
        sigma = base_sigma * (0.9 ** i)
        seed_val = seed + i
        candidate = generate_candidate(h_initial if i == 0 else population[-1], pt_type, sigma, seed_val)
        population.append(candidate)

    max_iter = 100
    while time.time() - start_time < budget_s - 60:
        if not population:
            break
        candidate = population.pop(0)
        result = minimize(
            fun=objective,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-9}
        )
        if result.success:
            new_h = result.x.reshape(n_points)
            new_h = np.clip(new_h, 0, 1)
            total = new_h.sum()
            if total != 0:
                new_h = (new_h / total) * target_sum
                new_h = np.clip(new_h, 0, 1)
            new_corr = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            constraint_violation = abs(new_h.sum() - target_sum)
            if constraint_violation < 1e-6 and np.all((new_h >= -1e-6) & (new_h <= 1.0 + 1e-6)):
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
        for _ in range(2):
            if time.time() - start_time > budget_s - 60:
                break
            pt_type = np.random.choice(['gaussian', 'boundary', 'block'])
            sigma = base_sigma * 0.5
            seed_val = seed + len(population) + 100
            new_candidate = generate_candidate(best_h, pt_type, sigma, seed_val)
            population.append(new_candidate)

    assert abs(best_h.sum() - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
