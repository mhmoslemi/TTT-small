import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0
    h = h * target_sum / np.sum(h)
    h = np.clip(h, 0.0, 1.0)
    h = h * target_sum / np.sum(h)
    dx = 2.0 / n_points

    alpha = 0.01  # Step size for gradient descent
    best_c5 = np.inf
    best_h = h.copy()
    start_time = time.time()

    while time.time() - start_time < budget_s - 1:
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_max = np.max(corr)
        if current_max < best_c5:
            best_c5 = current_max
            best_h = h.copy()

        # Find the index of the maximum overlap
        index_max = np.argmax(corr)
        k_full = index_max - (n_points - 1)

        # Compute gradient for this k_full
        grad = np.zeros(n_points)
        for i in range(n_points):
            # term1: contribution from h[i + k_full]
            i_plus_k = i + k_full
            if 0 <= i_plus_k < n_points:
                term1 = 1.0 - h[i_plus_k]
            else:
                term1 = 0.0

            # term2: contribution from h[i - k_full]
            i_minus_k = i - k_full
            if 0 <= i_minus_k < n_points:
                term2 = -h[i_minus_k]
            else:
                term2 = 0.0

            grad[i] = term1 + term2

        # Gradient update
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if np.abs(current_sum) > 1e-10:
            h = h * (target_sum / current_sum)

    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr)
    return best_h, c5_bound, n_points
