def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)

    def objective(h_vec):

        max_overlap = 0.0
        for k in range(1, n_points):
            valid_indices = n_points - k
            overlap = np.sum(h[:valid_indices] * (1.0 - h[k:k+valid_indices]))
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    best_c5 = np.inf

    target_sum = n_points / 2.0

        initial_guesses.append(h_candidate)

            break
