    for _ in range(5):
        while True:
            r = np.random.rand(n_points)
            r = (r / r.sum()) * target_sum  # Feasible sum
            r = np.clip(r, 0, 1)
            if np.all((r >= 0) & (r <= 1)):
                init_pop.append(r.tolist())
                break

    # Configure DE to use available time efficiently
    remaining_time = budget_s - (time.time() - start_time)
    de_config = {
        'bounds': bounds,
        'init': init_pop,
        'seed': seed,
        'mutation': (0.5, 1.0),
        'recombination': 0.9,
        'popsize': 5,
        'maxiter': int(remaining_time * 0.6 / (n_points * 5)),  # Allocate ~60% of time to DE
        'strategy': 'best1bin',
        'tol': 1e-6,
        'updating': 'replace',
        'workers': 1
    }

    de_result = differential_evolution(objective_de, **de_config)

    best_h = np.array(de_result.x)
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    # Post-process to ensure constraint compliance
    current_sum = best_h.sum()
    if abs(current_sum - target_sum) > 1e-6:
        scale = target_sum / current_sum
        best_h = np.clip(best_h * scale, 0, 1)

    # Local refinement if time remains
    if time.time() - start_time < budget_s - 60:
        def objective_local(x):
            h_vec = np.array(x).reshape(n_points)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx

        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(
            objective_local,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        if result.success:
            improved_h = result.x
            if np.all((improved_h >= 0) & (improved_h <= 1)) and abs(np.sum(improved_h) - target_sum) < 1e-6:
                improved_c5 = np.max(np.correlate(improved_h, 1 - improved_h, mode='full')) * dx
                if improved_c5 < best_c5:
                    best_h = improved_h
                    best_c5 = improved_c5

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds exceeded"

    return (best_h.tolist(), best_c5, n_points)
