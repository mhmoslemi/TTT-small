import numpy as np
import time

def generate_periodic_pattern(p, n_points):
    pattern = np.zeros(n_points)
    half_p = p // 2
    for i in range(n_points):
        if i % p < half_p:
            pattern[i] = 1.0
        else:
            pattern[i] = 0.0
    return pattern

def generate_random_h(n_points, target_sum):
    h = np.random.rand(n_points)
    scaling = target_sum / h.sum()
    h_scaled = h * scaling
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def generate_gaussian_mixture(n_points, target_sum):
    x = np.linspace(0, 2, n_points)
    mu = 1.0
    sigma = 0.4
    h = np.exp(- (x - mu)**2 / (2 * sigma**2))
    h = h / h.sum() * target_sum
    h = np.clip(h, 0, 1)
    return h
