import numpy as np

def compute_c5(h, dx, n_points):
    h = np.ascontiguousarray(h)
    corr = np.correlate(h, 1 - h, mode='full')
    shift_values = corr[n_points - 1:]
    max_overlap = np.max(shift_values)
    return max_overlap * dx

def compute_gradient(h, k, n_points):
    grad = np.zeros_like(h)
    for j in range(n_points):
        term1 = 0.0
        if j + k < n_points:
            term1 = 1.0 - h[j + k]
        term2 = 0.0
        if j - k >= 0:
            term2 = -h[j - k]
        grad[j] = term1 + term2
    return grad

def project_feasible(candidate, required_sum):
    candidate = np.clip(candidate, 0, 1)
    current_sum = np.sum(candidate)
    if abs(current_sum) < 1e-10:
        candidate[0] = required_sum
        return candidate
    scaling = required_sum / current_sum

    candidate = np.clip(candidate, 0, 1)
    current_sum = np.sum(candidate)
    delta = required_sum - current_sum
    if abs(delta) > 1e-8:
        candidate[0] += delta
        candidate[0] = np.clip(candidate[0], 0, 1)
    return candidate
