                velocities[j] = np.zeros(n_points)
                score = objective(h, dx, n_points)
                if score < personal_best_scores[j]:
                    personal_best_scores[j] = score
                    personal_best[j] = h.copy()
                    if score < global_best_score:
                        global_best_score = score
                        global_best = h.copy()
        
        # Time check
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final validation
    corr_full = correlate(global_best, 1 - global_best, mode='full', method='fft') * dx
    final_c5 = np.max(corr_full)
    return global_best, final_c5, n_points

def objective(h, dx, n_points):
    one_minus_h = 1.0 - h
    corr = correlate(h, one_minus_h, mode='full', method='fft')
    return np.max(corr) * dx
