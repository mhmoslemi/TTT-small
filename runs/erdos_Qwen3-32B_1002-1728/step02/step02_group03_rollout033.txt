        child = scale_to_sum(alpha * p1 + (1 - alpha) * p2)
        return child

    population = [h_initial]
    for _ in range(pop_size - 1):
        mut = mutate(h_initial)
        population.append(mut)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = np.inf
    start_time = time.time()

    def compute_gradient(h, k):
        grad = np.zeros_like(h)
        n = len(h)
        valid_left = np.arange(n) + k < n
        i_left = np.arange(n)[valid_left]
        grad[i_left] += (1 - h[i_left + k]) * dx
        i_right = np.arange(k, n)
        grad[i_right] -= h[i_right - k] * dx
        return grad

    while time.time() - start_time < budget_s - 1:
        scores = evaluate(population)
        idx = np.argmin(scores)
        if scores[idx] < best_c5:
            best_c5 = scores[idx]
            best_h = population[idx].copy()

        elitism_size = max(2, int(elitism_ratio * pop_size))
        elite = population[np.argsort(scores)[:elitism_size]]

        new_pop = [h for h in elite]
        while len(new_pop) < pop_size:
            p1 = population[np.random.choice(np.where(scores == np.min(scores))[0])]
            p2 = population[np.random.choice(np.where(scores == np.min(scores))[0])]
            child = crossover(p1, p2)
            child = mutate(child)
            new_pop.append(child)
        population = np.array(new_pop)

    # Gradient refinement
    patience_counter = 0
    for _ in range(grad_steps):
        if time.time() - start_time >= budget_s - 1:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        k_idx = np.argmax(corr)
        k_active = k_idx - (n_points - 1)
        grad = compute_gradient(best_h, k_active)
        new_h = np.clip(best_h - grad_alpha * grad, 0, 1)
        new_h = scale_to_sum(new_h)
        
        new_corr = np.correlate(new_h, 1 - new_h, mode='full') * dx
        new_c5 = np.max(new_corr)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= grad_patience:
                break

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    # Ensure constraints
    np.testing.assert_allclose(best_h.sum(), target_sum, rtol=1e-3)
    return best_h, final_c5, n_points
