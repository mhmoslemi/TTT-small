                for i in range(m_abs, n_points):
                    grad[i] += -h[i - m_abs] * dx

        # Update step
        h_new = h - alpha * grad
        h_new_clipped = np.clip(h_new, 0, 1)
        sum_clipped = h_new_clipped.sum()
        if abs(sum_clipped) > 1e-10:
            h_new_scaled = h_new_clipped * (target_sum / sum_clipped)
            h_new_scaled = np.clip(h_new_scaled, 0, 1)
        else:
            h_new_scaled = h_new_clipped

        # Update if improvement
        corr_new = np.correlate(h_new_scaled, 1 - h_new_scaled, mode='full') * dx
        new_c5 = np.max(corr_new)
        if new_c5 < best_c5:
            h = h_new_scaled.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            alpha *= 0.5  # reduce step size

        # Decay alpha gradually
        alpha *= 0.999

    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_values)
    return best_h, final_c5, n_points
