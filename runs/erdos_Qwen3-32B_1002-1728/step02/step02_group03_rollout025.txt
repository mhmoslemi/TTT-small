def run(seed=42, budget_s=1100, **kwargs):

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute full correlate for all lags
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        m = np.argmax(corr)
        k_lag = m - (n_points - 1)
        abs_k_lag = abs(k_lag)
