    current_sum = h.sum()

def adjust_sum(h, target_sum):

    current_sum = h_adj.sum()
    D = target_sum - current_sum
    if abs(D) < 1e-9:
        return h_adj
    if D > 0:
        can_incr = (h_adj < 0.9999)

            return h_adj

        if not can_decr.any():
            return h_adj
        h_adj[can_decr] -= D * h_adj[can_decr] / h_adj[can_decr].sum()
