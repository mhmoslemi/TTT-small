import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair(h, target_sum=target_sum):
        h = np.clip(h, 0.0, 1.0)
        current_sum = np.sum(h)
        if not np.isclose(current_sum, target_sum, atol=1e-8):
            scale = target_sum / current_sum
            h *= scale
            h = np.clip(h, 0.0, 1.0)
        return h

    best_h = repair(h_initial).copy()
    best_c5 = compute_c5(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    T = 1.0  # Initial temperature
    alpha = 0.9999  # Cooling rate
    noise_scale = 0.1  # Noise factor for perturbation
    start_time = time.time()

    while time.time() - start_time < budget_s - 1:
        # Perturbation
        perturbation = np.random.normal(0, noise_scale * T, n_points)
        h_perturbed = current_h.copy() + perturbation
        h_perturbed = repair(h_perturbed)

        # Compute new C5
        new_c5 = compute_c5(h_perturbed)

        # Acceptance criteria
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            current_h = h_perturbed.copy()
            current_c5 = new_c5

            # Update best solution
            if new_c5 < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = new_c5

        # Cool temperature
        T *= alpha

    return best_h, best_c5, n_points
