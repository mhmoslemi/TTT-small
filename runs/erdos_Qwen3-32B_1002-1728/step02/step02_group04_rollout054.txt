    end_idx = start_idx + n_points
    valid_corr = full_corr[start_idx:end_idx]
    max_corr = np.max(valid_corr)
    return max_corr * dx

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0
    np.random.seed(seed)
    start_time = time.time()
    alpha = 0.01
    best_h = h.copy()
    best_c5 = compute_c5(h, dx, n_points)
    last_improvement = 0
    perturb_counter = 0
    iteration = 0

    while time.time() - start_time < budget_s - 5:
        # Compute overlaps and find max k
        full_corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = full_corr[n_points -1 : 2*n_points -1]
        k_max = np.argmax(valid_corr)
        
        # Compute gradient
        grad = np.ones(n_points)
        if k_max > 0:
            grad[:n_points - k_max] = 1 - h[k_max : n_points]
        
        # Update h
        h = h - alpha * grad
        
        # Project to feasible set
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if abs(current_sum) > 1e-10:
            scaling = required_sum / current_sum
            h_clipped *= scaling
            h_clipped = np.clip(h_clipped, 0, 1)
        h = h_clipped
        
        # Check improvement
        c5_current = compute_c5(h, dx, n_points)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            last_improvement = time.time() - start_time
            alpha = max(0.005, alpha * 1.01)  # Slight increase if improving
        else:
            alpha = max(0.005, alpha * 0.995)  # Decay if not improving
            perturb_counter += 1
        
        # Periodic perturbation to escape local optima
        if perturb_counter >= 30:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if abs(current_sum) > 1e-10:
                scaling = required_sum / current_sum
                h *= scaling
                h = np.clip(h, 0, 1)
            perturb_counter = 0
        
        iteration += 1
    
    return best_h, best_c5, n_points
