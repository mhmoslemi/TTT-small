import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h):
        h_flat = h.reshape(-1)
        corr = correlate(h_flat, 1 - h_flat, mode='full')
        return np.max(corr) * dx

    def repair(h, target_sum):
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        if abs(current_sum - target_sum) > 1e-8:
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0.0, 1.0)
        return h

    best_h = h.copy()
    best_c5 = objective(best_h)
    current_h = h.copy()
    current_c5 = best_c5

    T = 0.1
    alpha = 0.999
    start_time = time.time()
    max_time = budget_s

    while time.time() - start_time < max_time - 0.1:
        T *= alpha
        if T < 1e-4:
            break

        # Perturb
        perturbation = np.random.normal(0, T * 0.1, n_points)
        new_h = current_h + perturbation
        new_h = repair(new_h, target_sum)
        new_c5 = objective(new_h)

        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

    return best_h, best_c5, n_points
