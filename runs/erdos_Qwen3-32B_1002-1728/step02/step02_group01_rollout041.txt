import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def objective(h_vec):
        return np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx

    def perturb(h_vec, method=None, sigma=0.1):
        h = np.clip(h_vec.copy(), 0, 1)
        if method == 'boundary':
            shift = np.random.choice([-0.2, 0.2], size=n_points)
            h += shift
        elif method == 'block':
            start = np.random.randint(0, n_points)
            length = np.random.randint(1, 5)
            h[start:start+length] = 1.0 - h[start:start+length]
        else:  # Gaussian
            h += np.random.randn(n_points) * sigma
        h = np.clip(h, 0, 1)
        total = h.sum()
        h *= target_sum / total if total != 0 else target_sum
        return np.clip(h, 0, 1)

    np.random.seed(seed)
    population = [h]
    best_h = h.copy()
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s - 60:
        if not population:
            break
        current = population.pop()
        # Choose perturbation method based on time
        method = 'block' if np.random.rand() < 0.3 else ('boundary' if np.random.rand() < 0.5 else None)
        sigma = 0.05 * max(1.0 - (time.time() - start_time) / budget_s, 0.2)  # Adaptive sigma
        perturbed = perturb(current, method=method, sigma=sigma)
        result = minimize(
            objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h = np.clip(result.x, 0, 1)
            total = new_h.sum()
            if total > 1e-6:
                new_h = (new_h / total) * target_sum
                new_h = np.clip(new_h, 0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                # Accept if valid
                if abs(new_h.sum() - target_sum) < 1e-6:
                    best_h = np.clip(new_h, 0, 1)
                    best_c5 = new_c5
                    # Generate successors
                    population.extend([
                        perturb(best_h, method='block'),
                        perturb(best_h, method='boundary'),
                        perturb(best_h, sigma=sigma * 0.5)
                    ])

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
