import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Ensure initial h has correct sum
    h_initial = np.copy(h_full_initial)
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-8:
        scale = target_sum / current_sum
        h_initial = np.clip(h_initial * scale, 0.0, 1.0)
    
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, 'full')) * dx

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        # Generate perturbed starting point
        perturbation = (np.random.rand(n_points) * 2 - 1) * 0.3
        perturbation = np.clip(h_full_initial * perturbation, -0.25, 0.25)
        x0 = np.clip(best_h + perturbation, 0.0, 1.0)
        
        # Normalize to maintain area constraint
        x0_sum = x0.sum()
        if abs(x0_sum) > 1e-8:
            x0 = (x0 / x0_sum) * target_sum
            x0 = np.clip(x0, 0.0, 1.0)
        else:
            continue

        # Optimization with SLSQP
        bounds = [(0.0, 1.0)] * n_points
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        result = minimize(
            fun=lambda x: np.max(np.correlate(x, 1 - x, 'full')) * dx,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 80, 'ftol': 1e-10, 'disp': False}
        )

        if result.success:
            x = np.clip(result.x, 0.0, 1.0)
            x_sum = x.sum()
            if abs(x_sum - target_sum) > 1e-6:
                x = (x / x_sum) * target_sum
                x = np.clip(x, 0.0, 1.0)
            
            c5 = np.max(np.correlate(x, 1 - x, 'full')) * dx
            if c5 < best_c5:
                best_h = x.copy()
                best_c5 = c5

    return (best_h.tolist(), best_c5, n_points)
