import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) = 1

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    # Ensure initial guess is feasible
    h_init = np.clip(h_init, 0.0, 1.0)
    current_sum = np.sum(h_init)
    if not np.isclose(current_sum, target_sum):
        h_init = (h_init / current_sum) * target_sum
        h_init = np.clip(h_init, 0.0, 1.0)
    best_h = h_init.copy()
    best_c5 = objective(best_h)

    seed_state = np.random.SeedSequence(seed)
    np.random.seed(seed_state.generate_state(1)[0])

    while time.time() - start_time < budget_s - 30:
        time_frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.25 * (1 - time_frac**2)
        perturbed = best_h + np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if np.isclose(perturbed_sum, 0):
            continue
        perturbed = (perturbed / perturbed_sum) * target_sum
        perturbed = np.clip(perturbed, 0.0, 1.0)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 30, 'ftol': 1e-14, 'disp': False}
        )

        if result.success:
            new_h = np.clip(result.x, 0.0, 1.0)
            new_sum = np.sum(new_h)
            if abs(new_sum - target_sum) < 1e-6:
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5

        # Update seed to avoid redundant reseeding
        seed_state = np.random.SeedSequence(seed_state.entropy)

    # Final validation
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)

    return (best_h.tolist(), best_c5, n_points)
