def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)

    def normalize_h(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        h_unscaled = h * (target_sum / current_sum)

        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return np.zeros_like(h)
        h_normalized = h_clipped * (target_sum / new_sum)
        h_final = np.clip(h_normalized, 0, 1)
        return h_final

    h = normalize_h(h_full_initial.copy(), target_sum)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    current_h = h.copy()
    learning_rate = 0.01
    lr_decay = 0.995

    while time.time() - start_time < budget_s - 30:

        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)

        gradient = np.zeros_like(current_h)
        k = max_idx
        for j in range(n_points):
            term = 0.0

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize_h(h, target_sum):
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.zeros_like(h)
        h_unscaled = h * (target_sum / current_sum)
        h_clipped = np.clip(h_unscaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return np.zeros_like(h)
        h_normalized = h_clipped * (target_sum / new_sum)
        h_final = np.clip(h_normalized, 0, 1)
        return h_final

    # Initialize h
    h = normalize_h(h_full_initial.copy(), target_sum)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    current_h = h.copy()
    learning_rate = 0.01
    lr_decay = 0.995

    while time.time() - start_time < budget_s - 30:
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_idx = np.argmax(corr)
        k_shift = max_idx - (n_points - 1)

        gradient = np.zeros_like(current_h)
        for j in range(n_points):
            term = 0.0
            idx_plus = j + k_shift
            if 0 <= idx_plus < n_points:
                term += (1 - current_h[idx_plus])
            idx_min = j - k_shift
            if 0 <= idx_min < n_points:
                term -= current_h[idx_min]
            gradient[j] = term

        current_h -= learning_rate * gradient
        current_h = normalize_h(current_h, target_sum)

        corr_new = np.correlate(current_h, 1 - current_h, mode='full') * dx
        current_c5_new = np.max(corr_new)

        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h[:] = current_h

        learning_rate *= lr_decay

        if time.time() - start_time > budget_s - 5:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all(best_h <= 1.0 + 1e-6) and np.all(best_h >= -1e-6)
    return (best_h.tolist(), best_c5, n_points)
