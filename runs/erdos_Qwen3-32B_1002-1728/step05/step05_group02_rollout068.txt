            winner_idx = np.argmin(contestants_fitness)
            winner = pop_fitness[contestants[winner_idx]][1].copy()

            # Crossover
            if np.random.rand() < 0.7:
                other = pop_fitness[np.random.randint(pop_size)][1].copy()
                alpha = np.random.rand()
                winner = alpha * winner + (1 - alpha) * other
            winner = np.clip(winner, 0, 1)
            total = winner.sum()
            if total != 0:
                winner = (winner / total) * target_sum_half
            winner = np.clip(winner, 0, 1)

            # Mutation
            if np.random.rand() < 0.1:
                noise = np.random.randn(half_n) * 0.2 * (budget_s - (time.time() - start_time)) / budget_s
                winner += noise
                winner = np.clip(winner, 0.0, 1.0)
            total = winner.sum()
            if total != 0:
                winner = (winner / total) * target_sum_half
            winner = np.clip(winner, 0, 1)
            new_population.append(winner)

        population = new_population

        # Local optimization
        for c5, h_half in top_candidates[:3]:
            def opt_obj(x, target_sum=target_sum_half):
                x = np.clip(x, 0, 1)
                s = x.sum()
                if s < 1e-8:
                    return float('inf')
                x = (x / s) * target_sum
                x = np.clip(x, 0, 1)
                return objective(x)
            constraints_local = {"type": "eq", "fun": lambda x: np.sum(x) * dx * 0.5 - (target_sum_half * dx)}
            result = minimize(opt_obj, x0=h_half, method='L-BFGS-B', bounds=[(0, 1)] * half_n)
            if result.success:
                optimized = np.clip(result.x, 0, 1)
                total = optimized.sum()
                optimized = (optimized / total) * target_sum_half if total != 0 else optimized
                optimized = np.clip(optimized, 0, 1)
                c5_new = objective(optimized)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h_half[:] = optimized.copy()

    # Final validation
    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), f"Invalid Value: {np.min(h_full)}, {np.max(h_full)}"

    return (h_full.tolist(), best_c5, n_points)
