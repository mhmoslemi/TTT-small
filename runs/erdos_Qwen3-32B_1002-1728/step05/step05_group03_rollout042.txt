            group_dict[k].append((i, j))

    # Define variables
    h = cp.Variable(n_points, nonneg=True, ub=1)
    t = cp.Variable()

    # Objective: minimize t
    objective = cp.Minimize(t)

    # Constraints
    constraints = [cp.sum(h) == n_points / 2.0]  # Sum constraint

    for k in group_dict:
        pairs = group_dict[k]
        terms = [h[i] * (1 - h[j]) for i, j in pairs]
        overlap = cp.sum(terms) * dx
        constraints.append(overlap <= t)

    # Form problem
    problem = cp.Problem(objective, constraints)

    # Set initial guess (warm start)
    h.value = h_initial

    # Solve with available solvers
    try:
        problem.solve(solver=cp.MOSEK, verbose=False, warm_start=True)
    except:
        try:
            problem.solve(solver=cp.CVXOPT, verbose=False)
        except:
            problem.solve(solver=cp.ECOS, verbose=False)

    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        # Fallback to initial solution if time budget is exhausted
        h_values = h_initial.tolist()
        corr = np.correlate(h_initial, 1 - h_initial, mode='full') * dx
        return (h_values, np.max(corr), n_points)

    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_values = h_initial.tolist()
        corr = np.correlate(h_initial, 1 - h_initial, mode='full') * dx
        return (h_values, np.max(corr), n_points)

    h_solution = h.value
    h_solution = np.clip(h_solution, 0, 1)
    corr = np.correlate(h_solution, 1 - h_solution, mode='full') * dx
    c5_bound = np.max(corr)

    return (h_solution.tolist(), c5_bound, n_points)
