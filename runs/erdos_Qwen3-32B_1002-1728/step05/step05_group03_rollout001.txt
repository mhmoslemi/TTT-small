        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    population_size = 50
    np.random.seed(seed)
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_normalized = normalize_h_half(h_half_initial, target_sum_half)

    positions = []
    velocities = []
    personal_best_positions = []
    personal_best_values = []
    for _ in range(population_size):
        perturbed = h_half_normalized + np.random.normal(0, 0.05, half_n)
        perturbed_clipped = np.clip(perturbed, 0, 1)
        perturbed_normalized = normalize_h_half(perturbed_clipped, target_sum_half)
        positions.append(perturbed_normalized)
        velocities.append(np.random.normal(0, 0.01, half_n))
        h_full_pb = create_full_h(perturbed_normalized)
        corr_pb = np.correlate(h_full_pb, 1 - h_full_pb, mode='full')
        c5_pb = np.max(corr_pb) * dx
        personal_best_positions.append(perturbed_normalized.copy())
        personal_best_values.append(c5_pb)

    best_c5 = min(personal_best_values)
    best_h_half = personal_best_positions[personal_best_values.index(best_c5)]

    while time.time() - start_time < budget_s - 30:
        current_time = time.time() - start_time
        time_frac = min(current_time / budget_s, 1.0)
        inertia_weight = 0.7 - 0.6 * time_frac

        for i in range(population_size):
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive = 2.0 * r1 * (personal_best_positions[i] - positions[i])
            social = 2.0 * r2 * (best_h_half - positions[i])
            velocities[i] = inertia_weight * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            new_position = positions[i] + velocities[i]
            new_position = np.clip(new_position, 0, 1)
            new_position = normalize_h_half(new_position, target_sum_half)
            h_full_new = create_full_h(new_position)
            corr_new = np.correlate(h_full_new, 1 - h_full_new, mode='full')
            c5_new = np.max(corr_new) * dx
            if c5_new < personal_best_values[i]:
                personal_best_positions[i] = new_position.copy()
                personal_best_values[i] = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h_half = new_position.copy()
            positions[i] = new_position

        min_pb = min(personal_best_values)
        if min_pb < best_c5:
            best_c5 = min_pb
            best_h_half = personal_best_positions[personal_best_values.index(min_pb)]

    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
