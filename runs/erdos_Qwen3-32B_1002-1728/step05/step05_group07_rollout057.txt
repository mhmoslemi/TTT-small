    if h_half_initial is not None:
        h_half.value = h_half_initial

    # Iteratively add worst-case constraints
    max_outer = 5
    for outer_iter in range(max_outer):
        # Solve with current constraints
        problem.solve(solver=cp.ECOS, verbose=False, max_iter=1000)
        if problem.status != cp.OPTIMAL:
            break
        current_t = t.value

        # Extract current h_full solution
        current_h_full = np.zeros(n_points)
        for i in range(half_n):
            current_h_full[i] = h_half.value[i]
            current_h_full[n_points - 1 - i] = h_half.value[i]
        current_h_full = np.clip(current_h_full, 0, 1.0)

        # Evaluate cross-correlation for all possible k in 0..n_points-1
        max_k = -1
        max_val = -np.inf
        for k_candidate in range(n_points):
            sum_val = 0.0
            for i in range(n_points - k_candidate):
                sum_val += current_h_full[i] * (1 - current_h_full[i + k_candidate])
            val = sum_val * dx
            if val > max_val:
                max_val = val
                max_k = k_candidate

        # Add constraint for this max_k
        sum_expr = 0.0
        for i in range(n_points - max_k):
            sum_expr += h_full[i] * (1 - h_full[i + max_k])
        new_constraint = t >= sum_expr * dx
        constraints.append(new_constraint)
        problem = cp.Problem(cp.Minimize(t), constraints)

        # Check time budget
        time_left = budget_s - (time.time() - start_time)
        if time_left < 5:
            break

    # Final solve
    problem.solve(solver=cp.ECOS, verbose=False, max_iter=1000)

    # Construct the solution
    best_h_half = h_half.value
    best_t = t.value

    h_full_solution = np.zeros(n_points)
    for i in range(half_n):
        h_full_solution[i] = best_h_half[i]
        h_full_solution[n_points - 1 - i] = best_h_half[i]
    h_full_solution = np.clip(h_full_solution, 0, 1.0)

    assert abs(np.sum(h_full_solution) - n_points / 2.0) < 1e-6

    return (h_full_solution.tolist(), best_t, n_points)
