def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to meet constraint
    current_sum = np.sum(h)
    if current_sum != 0:
        h *= target_sum / current_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx

    # Optimization parameters
    momentum = 0.9
    learning_rate_start = 0.1
    learning_rate_end = 0.0001
    velocity = np.zeros_like(h)

    max_iterations = int(1e5)  # Upper bound for loop control

    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        if iteration > max_iterations:
            break

        # Compute worst shift k
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr_index = np.argmax(corr)
        lag = max_corr_index - (n_points - 1)

        # Compute gradient for current k
        grad = np.zeros_like(h)
        for m in range(n_points):
            m_plus = (m + lag) % n_points
            m_minus = (m - lag) % n_points
            term1 = 1.0 - h[m_plus]
            term2 = h[m_minus]
            grad[m] = term1 - term2

        # Learning rate schedule
        progress = min(1.0, iteration / 10000.0)
        learning_rate = learning_rate_end + (learning_rate_start - learning_rate_end) * (1 - progress)

        # Update with momentum

        h += velocity
