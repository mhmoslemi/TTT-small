            best_h_half = current_best_h_half.copy()
        new_population = [best_h_half.copy()]
        for _ in range(population_size - 1):
            parent1 = tournament_selection(population, c5_list, size=2)
            parent2 = tournament_selection(population, c5_list, size=2)
            beta = np.random.uniform(0.3, 0.7)
            child = beta * parent1 + (1 - beta) * parent2
            mutation_noise = np.random.normal(0, 0.01, half_n)
            child += mutation_noise
            child = np.clip(child, 0, 1)
            current_sum = child.sum()
            if np.isclose(current_sum, 0, atol=1e-6):
                child = best_h_half.copy()
            else:
                scale = target_sum_half / current_sum
                child *= scale
                child = np.clip(child, 0, 1)
            new_population.append(child.copy())
        population = new_population
    best_h_full = create_full_h(best_h_half)
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    assert np.abs(np.sum(best_h_full) - n_points / 2.0) < 1e-6
    return (best_h_full.tolist(), best_c5, n_points)
