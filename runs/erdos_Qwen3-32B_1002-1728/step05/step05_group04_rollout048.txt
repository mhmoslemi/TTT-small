        result = time_limited_de()
        if result.success:
            best_h = normalize_h(result.x)
        else:
            best_h = normalize_h(h_full_initial)
    except:
        best_h = normalize_h(h_full_initial)

    best_c5 = objective_func(best_h)
    return (best_h.tolist(), best_c5, n_points)
