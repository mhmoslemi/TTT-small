        for i in range(half_n):
            j = n_points - 1 - i
            h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym = np.clip(h_full_sym * (n_points / 2 / np.sum(h_full_sym)), 0, 1)

        x = np.linspace(0, 2.0, n_points)
        hats = []
        i = 0
        while i < n_points:
            if h_full_sym[i] <= 1e-6:
                i += 1
                continue
            start = i
            while i < n_points and h_full_sym[i] > 1e-6:
                i += 1
            end = i - 1
            if end > start:
                start_x = x[start]
                end_x = x[end]
                w = end_x - start_x
                integral = np.sum(h_full_sym[start:end+1]) * dx
                a_i = 2 * integral / w
                c_i = (start_x + end_x) / 2.0
