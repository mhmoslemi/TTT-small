import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_full = h_full_initial.copy()
    current_sum = np.sum(h_full)
    if abs(current_sum) > 1e-10:
        h_full = (h_full / current_sum) * target_sum
    h_full = np.clip(h_full, 0, 1)
    best_h = h_full.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 10:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        noise_scale = 0.3 * (1 - (frac ** 0.75))
        
        for _ in range(3):
            perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if abs(total) < 1e-10:
                continue
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            for method in ['COBYLA', 'L-BFGS-B']:
                result = minimize(
                    fun=lambda x: objective(x),
                    x0=perturbed,
                    method=method,
                    bounds=[(0.0, 1.0)] * n_points,
                    constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                    options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
                )
                if result.success:
                    new_h = np.clip(result.x, 0, 1)
                    new_sum = np.sum(new_h)
                    if abs(new_sum) > 1e-10:
                        new_h = (new_h / new_sum) * target_sum
                        new_h = np.clip(new_h, 0, 1)
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5

    h_full = best_h.copy()
    assert abs(np.sum(h_full) - target_sum) < 1e-6
