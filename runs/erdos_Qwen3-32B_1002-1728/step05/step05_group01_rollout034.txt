                scaling = (n_points / 2.0) / current_sum
                h_scaled = phi_values * scaling
                h_clipped = np.clip(h_scaled, 0, 1)
                corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
                c5_val = np.max(corr) * dx
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_c = c
                    best_h = h_clipped

    if best_h is None:
        # Fallback to initial solution
        phi_values = phi_matrix @ best_c
        current_sum = phi_values.sum()
        scaling = (n_points / 2.0) / current_sum
        best_h = np.clip(phi_values * scaling, 0, 1)

    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
