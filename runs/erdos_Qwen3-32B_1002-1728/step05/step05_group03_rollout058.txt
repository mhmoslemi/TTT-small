        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    def compute_c5(h_half):
        full_h = create_full_h(h_half)
        corr = np.correlate(full_h, 1 - full_h, mode='full')
        return np.max(corr) * dx

    initial_h_half = h_full_initial[:half_n].copy()
    h_current = normalize_h_half(initial_h_half, target_sum_half)
    c5_current = compute_c5(h_current)
    best_h_half = h_current.copy()
    best_c5 = c5_current

    np.random.seed(seed)
    learning_rate = 0.05
    epsilon = 1e-5
    perturb_interval = 75
    max_grad_steps = 20
    iterations = 0

    while time.time() - start_time < budget_s - 10:
        for _ in range(max_grad_steps):
            if time.time() - start_time >= budget_s - 10:
                break
            grad = np.zeros_like(h_current)
            for i in range(half_n):
                h_plus = h_current.copy()
                h_plus[i] += epsilon
                h_plus = normalize_h_half(h_plus, target_sum_half)
                c_plus = compute_c5(h_plus)
                h_minus = h_current.copy()
                h_minus[i] -= epsilon
                h_minus = normalize_h_half(h_minus, target_sum_half)
                c_minus = compute_c5(h_minus)
                grad[i] = (c_plus - c_minus) / (2 * epsilon)
            h_candidate = h_current - learning_rate * grad
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h_half(h_candidate, target_sum_half)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < c5_current:
                h_current = h_candidate
                c5_current = c5_candidate
            if c5_current < best_c5:
                best_h_half = h_current.copy()
                best_c5 = c5_current
            iterations += 1

        if iterations % perturb_interval == 0:
            perturbation = np.random.normal(0, 0.2, h_current.shape)
            h_current = h_current + perturbation
            h_current = np.clip(h_current, 0, 1)
            h_current = normalize_h_half(h_current, target_sum_half)
            c5_candidate = compute_c5(h_current)
            if c5_candidate < c5_current:
                c5_current = c5_candidate
            if c5_current < best_c5:
                best_h_half = h_current.copy()
                best_c5 = c5_current

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), best_c5, n_points)
