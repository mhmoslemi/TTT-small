import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    np.random.seed(seed)
    num_particles = 20
    max_iter_pso = 20
    inertia = 0.8
    cog, soc = 1.2, 1.2

    # Initialize particles and velocities
    particles = []
    velocities = []
    for _ in range(num_particles):
        h = h_initial.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (target_sum / h_sum) if h_sum != 0 else h_initial
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)

    fitness = np.array([compute_c5(p) for p in particles])
    pbest_pos = particles.copy()
    pbest_fit = fitness.copy()
    gbest_index = np.argmin(pbest_fit)
    gbest_pos, gbest_val = particles[gbest_index].copy(), pbest_fit[gbest_index]

    # Particle Swarm Optimization loop
    for _ in range(max_iter_pso):
        if time.time() - start_time > budget_s - 30:
            break
        r1 = np.random.rand(num_particles, 1)
        r2 = np.random.rand(num_particles, 1)
        velocities = inertia * velocities + cog * r1 * (pbest_pos - particles) + soc * r2 * (gbest_pos - particles)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(num_particles):
            total = particles[i].sum()
            if total != 0:
                particles[i] *= target_sum / total
        current_fitness = np.array([compute_c5(p) for p in particles])
        improved = current_fitness < pbest_fit
        pbest_pos[improved], pbest_fit[improved] = particles[improved], current_fitness[improved]
        current_gbest_index = np.argmin(pbest_fit)
        if pbest_fit[current_gbest_index] < gbest_val:
            gbest_pos, gbest_val = particles[current_gbest_index].copy(), pbest_fit[current_gbest_index]

    # Local Optimization with SLSQP
    bounds = [(0.0, 1.0)] * n_points
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = minimize(compute_c5, gbest_pos, method='SLSQP', bounds=bounds, constraints=constraints,
                   options={'maxiter': 150, 'ftol': 1e-8})
    if res.success:
        optimized = np.clip(res.x, 0, 1)
        optimized_sum = optimized.sum()
        optimized = optimized * (target_sum / optimized_sum) if optimized_sum != 0 else gbest_pos
        final_c5 = compute_c5(optimized)
        if final_c5 < gbest_val:
            gbest_pos, gbest_val = optimized, final_c5

    final_h = gbest_pos
    final_h = np.clip(final_h, 0, 1)
    finalize_sum = final_h.sum()
    if abs(finalize_sum - target_sum) > 1e-6:
        final_h *= target_sum / finalize_sum
    return (final_h.tolist(), gbest_val, n_points)
