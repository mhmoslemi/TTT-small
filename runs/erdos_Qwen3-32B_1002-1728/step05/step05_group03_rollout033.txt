import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = np.sum(h_copy)
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    np.random.seed(seed)
    initial_h_half = h_full_initial[:half_n].copy()
    h_half = normalize_h_half(initial_h_half, target_sum_half)
    best_h_half = h_half.copy()
    best_c5 = float('inf')
    learning_rate = 0.1
    max_no_improvement = 50
    no_improvement_steps = 0

    while time.time() - start_time < budget_s - 30:
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr_idx = np.argmax(corr)
        k_lag = max_corr_idx - (n_points - 1)
        
        # Compute gradient for full h
        grad_full = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_lag
            if 0 <= j < n_points:
                grad_full[i] = 1 - h_full[j]
            else:
                grad_full[i] = 0.0
        
        # Compute gradient for half
        grad_half = np.zeros(half_n)
        for i in range(half_n):
            mirror_i = n_points - 1 - i
            grad_half[i] = grad_full[i] + grad_full[mirror_i]
        
        # Update h_half
        h_half_candidate = h_half - learning_rate * grad_half
        h_half_clipped = np.clip(h_half_candidate, 0, 1)
        h_half_normalized = normalize_h_half(h_half_clipped, target_sum_half)
        
        # Evaluate improvement
        h_full_new = np.zeros(n_points)
        for i in range(half_n):
            h_full_new[i] = h_full_new[n_points - 1 - i] = h_half_normalized[i]
        corr_new = np.correlate(h_full_new, 1 - h_full_new, mode='full') * dx
        current_c5 = np.max(corr_new)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half_normalized.copy()
            no_improvement_steps = 0
        else:
            no_improvement_steps += 1
            if no_improvement_steps >= max_no_improvement:
                perturbation = np.random.normal(0, 0.1, half_n)
                h_half_candidate = best_h_half + perturbation
                h_half_candidate = np.clip(h_half_candidate, 0, 1)
                h_half_normalized = normalize_h_half(h_half_candidate, target_sum_half)
                no_improvement_steps = 0
                learning_rate *= 0.9
        
        h_half = h_half_normalized.copy()
    
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = best_h_half[i]
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
