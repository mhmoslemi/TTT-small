from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def objective(h_half):
        h_full = create_full_h(np.clip(h_half, 0, 1))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    h_half_initial = h_initial[:half_n].copy()
    bounds_half = [(0, 1) for _ in range(half_n)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_half}

    best_c5 = float('inf')
    best_h_half = h_half_initial.copy()
    best_h_full = create_full_h(best_h_half)

    # Run optimization within budget
    while time.time() - start_time < budget_s - 30:
