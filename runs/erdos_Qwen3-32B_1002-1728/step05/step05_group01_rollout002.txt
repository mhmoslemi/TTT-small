import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # e.g., 35 for n=70
    
    def objective(h_vars):
        corr = np.correlate(h_vars, 1 - h_vars, mode='full')
        return np.max(corr) * dx
    
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Normalize initial h to satisfy constraints
    h_best = np.clip(h_initial.copy(), 0, 1)
    current_sum = h_best.sum()
    if abs(current_sum - target_sum) > 1e-10:
        scale = target_sum / current_sum
        h_best = np.clip(h_best * scale, 0, 1)
    best_c5 = objective(h_best)
    
    np.random.seed(seed)
    
    while time.time() - start_time < budget_s - 30:
        # Perturb current h with decreasing noise over time
        noise_scale = 0.3 * (1 - (time.time() - start_time) / budget_s)
        h_perturbed = h_best + np.random.uniform(-noise_scale, noise_scale, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum_perturbed = h_perturbed.sum()
        
        if current_sum_perturbed < 1e-10:
            continue  # skip invalid samples
        
        # Adjust to match target sum
        h_perturbed = h_perturbed * (target_sum / current_sum_perturbed)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Try optimizing with different methods
        for method in ['COBYLA', 'L-BFGS-B']:
            result = minimize(
                fun=objective,
                x0=h_perturbed,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                h_candidate = np.clip(result.x, 0, 1)
                # Enforce sum constraint as safety
                current_sum_candidate = h_candidate.sum()
                if abs(current_sum_candidate - target_sum) > 1e-6:
                    h_candidate = h_candidate * (target_sum / current_sum_candidate)
                    h_candidate = np.clip(h_candidate, 0, 1)
                # Test if better
                candidate_c5 = objective(h_candidate)
                if candidate_c5 < best_c5:
                    h_best = h_candidate
                    best_c5 = candidate_c5
                    break  # Break inner loop if improved early
    
    return (h_best.tolist(), best_c5, n_points)
