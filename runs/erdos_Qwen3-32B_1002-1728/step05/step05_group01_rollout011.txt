    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        corr = np.correlate(h_vars, 1 - h_vars, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    h_initial = np.clip(h_initial, 0, 1)
    current_sum = np.sum(h_initial)
    if abs(current_sum - target_sum) > 1e-6:
        h_initial = h_initial * (target_sum / current_sum)
    h_initial = np.clip(h_initial, 0, 1)
    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        noise_scale = 0.3 * (1 - (frac ** 0.75))
        
        for _ in range(3):  # More perturbations per iteration
            perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if abs(total) < 1e-10:
                continue  # Skip if sum is zero
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            # Try multiple methods
            for method in ['COBYLA', 'L-BFGS-B', 'SLSQP']:
                result = minimize(
                    fun=objective,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons,
                    options={'maxiter': 150, 'ftol': 1e-10, 'disp': False}
                )
                if result.success and result.fun < best_c5:
                    new_h = np.clip(result.x, 0, 1)
                    corr = np.correlate(new_h, 1 - new_h, mode='full')
                    new_c5 = np.max(corr) * dx
                    if abs(np.sum(new_h) - target_sum) < 1e-6 and new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
                        break  # Break inner loop if improved

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
