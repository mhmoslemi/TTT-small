    # Adjusted Simulated Annealing with enhanced exploration
    np.random.seed(seed)
    T = 10.0  # Higher initial temperature for better exploration
    alpha = 0.9995  # Slightly slower cooling to accommodate higher T
    T_min = 1e-6

    while time.time() - start_time < budget_s - 20:
        if T < T_min:
            break

        # Perturb and scale
        perturbed = current_h_half + np.random.normal(0, T * 2, half_n)  # Larger noise with T
        perturbed = np.clip(perturbed, 0, 1)
        current_sum_pert = np.sum(perturbed)
        if not np.isclose(current_sum_pert, 0):
            scale = target_sum / (2 * np.sum(perturbed))  # Because create_full_h doubles the sum
            perturbed *= scale
        perturbed = np.clip(perturbed, 0, 1)

        new_h_full = create_full_h(perturbed)
        new_c5 = evaluate_c5(perturbed)
        delta = new_c5 - current_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h_half = perturbed.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_half[:] = current_h_half[:]
        T *= alpha

    # Final validity checks
    best_h_full = create_full_h(best_h_half)
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    assert np.abs(np.sum(best_h_full) - target_sum) < 1e-6

    return (best_h_full.tolist(), best_c5, n_points)
