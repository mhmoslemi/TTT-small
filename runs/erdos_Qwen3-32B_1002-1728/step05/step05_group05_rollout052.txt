def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_scale(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        current_sum = np.sum(h_full)
        if not np.isclose(current_sum, 0.0):
            scale = target_sum / current_sum
            h_full *= scale
        h_full_clipped = np.clip(h_full, 0.0, 1.0)
        new_h_half = h_full_clipped[:half_n].copy()
        return new_h_half

    # Initialize population
    population = []
    population_size = 50
    initial_h_half = initial_h_values[:half_n].copy()
    for _ in range(population_size):
        perturbed = initial_h_half + np.random.normal(0, 0.05, half_n)
        perturbed = np.clip(perturbed, 0.0, 1.0)
        h_half_final = clip_and_scale(perturbed)
        population.append(h_half_final)

    best_c5 = float('inf')
    best_h_half = initial_h_half.copy()

    while time.time() - start_time < budget_s - 20:
        # Evaluate candidates
        candidates = []
        for h_half in population:
            h_full = np.zeros(n_points)
            for i in range(half_n):
                h_full[i] = h_full[n_points - 1 - i] = h_half[i]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            candidates.append((h_half, c5))
        candidates.sort(key=lambda x: x[1])
        if candidates[0][1] < best_c5:
            best_c5 = candidates[0][1]
            best_h_half = candidates[0][0].copy()

        # Selection
        parents = []
        for _ in range(population_size // 2):
            a, b = np.random.choice(population_size, 2)
            if candidates[a][1] < candidates[b][1]:
                parents.append(candidates[a][0].copy())
            else:
                parents.append(candidates[b][0].copy())

        # Crossover and mutation
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 < len(parents):
                p1 = parents[i]
                p2 = parents[i + 1]
                beta = 0.3 + np.random.rand()
                child1 = beta * p1 + (1 - beta) * p2
                child2 = beta * p2 + (1 - beta) * p1
            else:
                child1 = parents[i].copy()
                child2 = parents[i].copy()
            for child in [child1, child2]:
                # Mutation
                noise = np.random.normal(0, 0.01, half_n)
                child += noise
                child = np.clip(child, 0.0, 1.0)
                # Clip and scale
                h_half_final = clip_and_scale(child)
                offspring.append(h_half_final)

        # Replace population
        new_population = [c[0] for c in candidates[:10]] + offspring
        if len(new_population) < population_size:
            while len(new_population) < population_size:
                idx = np.random.randint(len(candidates))
                new_population.append(candidates[idx][0].copy())
        population = new_population[:population_size]

    # Final best h
    best_h_full = np.zeros(n_points)
    for i in range(half_n):
        best_h_full[i] = best_h_full[n_points - 1 - i] = best_h_half[i]
    current_sum = np.sum(best_h_full)
    if not np.isclose(current_sum, target_sum):
        scale = target_sum / current_sum
        best_h_full *= scale
    best_h_full = np.clip(best_h_full, 0.0, 1.0)
    return (best_h_full.tolist(), best_c5, n_points)
