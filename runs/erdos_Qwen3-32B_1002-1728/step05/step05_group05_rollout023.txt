    def evaluate_c5(h):

        corr = np.correlate(h1, h2, mode='full')
        return np.max(corr) * dx

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    def evaluate_c5(h):
        h1 = h
        h2 = 1.0 - h1
        corr = np.correlate(h1, h2, mode='full')
        return np.max(corr) * dx
    
    n_particles = 50
    c1, c2 = 2.0, 2.0
    
    positions = []
    velocities = []
    pbest = []
    pbest_fitness = []
    
    for _ in range(n_particles):
        h = h_initial.copy()
        noise = np.random.uniform(-0.05, 0.05, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if not np.isclose(current_sum, 0):
            scale = target_sum / current_sum
            h *= scale
            h = np.clip(h, 0, 1)
        positions.append(h.copy())
        velocities.append(np.zeros_like(h))
        fit = evaluate_c5(h)
        pbest.append(h.copy())
        pbest_fitness.append(fit)
    
    gbest_index = np.argmin(pbest_fitness)
    gbest = pbest[gbest_index].copy()
    best_c5 = pbest_fitness[gbest_index]
    
    while time.time() - start_time < budget_s - 10:
        elapsed = time.time() - start_time
        fraction = elapsed / budget_s
        w = max(0.4, 0.9 - 0.5 * fraction)
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            current_pos = positions[i]
            current_vel = velocities[i]
            current_pbest = pbest[i]
            current_pbest_fit = pbest_fitness[i]
            
            cognitive = c1 * r1 * (current_pbest - current_pos)
            social = c2 * r2 * (gbest - current_pos)
            new_vel = w * current_vel + cognitive + social
            new_vel = np.clip(new_vel, -0.1, 0.1)
            
            new_pos = current_pos + new_vel
            new_pos = np.clip(new_pos, 0, 1)
            current_sum = new_pos.sum()
            if not np.isclose(current_sum, 0):
                scale = target_sum / current_sum
                new_pos *= scale
                new_pos = np.clip(new_pos, 0, 1)
            
            new_fit = evaluate_c5(new_pos)
            if new_fit < current_pbest_fit:
                pbest[i] = new_pos.copy()
                pbest_fitness[i] = new_fit
            
            if new_fit < best_c5:
                best_c5 = new_fit
                gbest = new_pos.copy()
            
            positions[i] = new_pos
            velocities[i] = new_vel
        
        gbest_index = np.argmin(pbest_fitness)
        if pbest_fitness[gbest_index] < best_c5:
            best_c5 = pbest_fitness[gbest_index]
            gbest = pbest[gbest_index].copy()
    
    final_h = gbest
    assert np.all(final_h >= 0) and np.all(final_h <= 1)
    assert np.isclose(final_h.sum(), target_sum, atol=1e-6)
    return (final_h.tolist(), best_c5, n_points)
