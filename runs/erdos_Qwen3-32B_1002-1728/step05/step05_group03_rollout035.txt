import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def normalize(h_half):
        s = np.sum(h_half)
        if s == 0:
            return h_half.copy()
        scale = target_sum_half / s
        scaled = np.clip(h_half * scale, 0, 1)
        s2 = np.sum(scaled)
        if s2 == 0:
            return scaled
        return np.clip(scaled * (target_sum_half / s2), 0, 1)

    np.random.seed(seed)
    particles = []
    velocities = []
    for _ in range(50):
        p = h_initial[:half_n].copy()
        p += np.random.normal(0, 0.1, half_n)
        particles.append(normalize(p))
        velocities.append(np.random.uniform(-0.1, 0.1, half_n))

    personal_best_positions = particles.copy()
    personal_best_vals = [np.inf] * len(particles)
    global_best_position = None
    global_best_val = np.inf

    w_max, w_min = 0.7, 0.1
    c1, c2 = 2.0, 2.0

    while time.time() - start_time < budget_s - 10:
        for i in range(len(particles)):
            p = particles[i]
            h_full = create_full_h(p)
            corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
            val = np.max(corr)
            if val < personal_best_vals[i]:
                personal_best_vals[i] = val
                personal_best_positions[i] = p.copy()
                if val < global_best_val:
                    global_best_val = val
                    global_best_position = p.copy()

        # Velocity update with inertia decay
        t = min(1.0, (time.time() - start_time) / (budget_s - 10))
        w = w_max - t * (w_max - w_min)
        for i in range(len(particles)):
            r1, r2 = np.random.rand(2)
            vel = (w * velocities[i] +
                   c1 * r1 * (personal_best_positions[i] - particles[i]) +
                   c2 * r2 * (global_best_position - particles[i]))
            velocities[i] = np.clip(vel, -0.3, 0.3)
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = normalize(particles[i])

    h_full = create_full_h(global_best_position)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
