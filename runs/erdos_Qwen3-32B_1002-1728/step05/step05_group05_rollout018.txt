import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    POP_SIZE = 50
    MUTATION_STD = 0.12
    crossover_rate = 0.6

    np.random.seed(seed)

    def initialize_population():
        pop = []
        for _ in range(POP_SIZE):
            h = h_initial.copy()
            noise = np.random.normal(0, MUTATION_STD, n_points)
            h += noise
            h_clipped = np.clip(h, 0, 1)
            current_sum = np.sum(h_clipped)
            if not np.isclose(current_sum, 0):
                scale = target_sum / current_sum
                h_scaled = h_clipped * scale
                h_final = np.clip(h_scaled, 0, 1)
            else:
                h_final = h_initial.copy()
            pop.append(h_final)
        return np.array(pop)

    population = initialize_population()

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)

    while time.time() - start_time < budget_s - 5:
        current_c5s = np.array([evaluate_c5(ind) for ind in population])
        current_best_idx = np.argmin(current_c5s)
        if current_c5s[current_best_idx] < best_c5:
            best_c5 = current_c5s[current_best_idx]
            best_h[:] = population[current_best_idx].copy()
        elite = population[current_best_idx].copy()

        tournament_size = 2
        parents_idx = np.random.randint(0, POP_SIZE, (POP_SIZE, tournament_size))
        selected_indices = []
        for idxs in parents_idx:
            vals = current_c5s[idxs]
            selected = idxs[np.argmin(vals)]
            selected_indices.append(selected)
        parents = population[selected_indices]

        children = []
        num_children = POP_SIZE - 1
        for _ in range(num_children):
            if np.random.rand() < crossover_rate:
                idxs = np.random.choice(len(parents), 2, replace=False)
                p1 = parents[idxs[0]]
                p2 = parents[idxs[1]]
                beta = np.random.rand()
                child = beta * p1 + (1 - beta) * p2
            else:
                child = parents[np.random.choice(len(parents))].copy()

            noise = np.random.normal(0, MUTATION_STD, n_points)
            child += noise

            if not np.isclose(current_sum, 0):
                scale = target_sum / current_sum
                child_clipped *= scale
                child_clipped = np.clip(child_clipped, 0, 1)
