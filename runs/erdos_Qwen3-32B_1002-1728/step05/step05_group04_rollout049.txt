import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Genetic Algorithm parameters
    POPULATION_SIZE = 40
    GENERATIONS = 150
    ELITE_FRACTION = 0.2
    MUTATION_SCALE = 0.15
    np.random.seed(seed)

    # Initialize population with clipped and scaled perturbations
    population = []
    for _ in range(POPULATION_SIZE):
        perturbed = h_initial + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = perturbed * (target_sum / total)
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s * 0.85:
        fitness_scores = np.array([objective(ind) for ind in population])
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_c5 = fitness_scores[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Select elites
        n_elite = int(ELITE_FRACTION * POPULATION_SIZE)
        elite_indices = np.argsort(fitness_scores)[:n_elite]
        elites = population[elite_indices]

        # Generate next generation
        new_population = [x for x in elites]
        while len(new_population) < POPULATION_SIZE:
            # Crossover: single-point blend
            parent1, parent2 = np.random.choice(population, 2)
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            
            # Mutation
            if np.random.rand() < 0.5:
                noise = np.random.normal(0, MUTATION_SCALE, n_points)
                child = np.clip(child + noise, 0, 1)
            # Repair constraint
            total = np.sum(child)
            if total != 0:
                child = child * (target_sum / total)
            child = np.clip(child, 0, 1)
            new_population.append(child)

        population = np.array(new_population)

    # Final local optimization refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )

    if result.success and result.fun < best_c5:
        refined_h = np.clip(result.x, 0, 1)
        if abs(np.sum(refined_h) - target_sum) < 1e-6:
            best_h = refined_h
            best_c5 = objective(best_h)

    return (best_h.tolist(), best_c5, n_points)
