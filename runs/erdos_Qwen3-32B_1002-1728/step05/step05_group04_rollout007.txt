    if initial_sum != 0:
        h_initial_normalized = h_full_initial * (target_sum / initial_sum)
    else:
        h_initial_normalized = h_full_initial.copy()
    h_initial_normalized = np.clip(h_initial_normalized, 0.0, 1.0)
    initial_c5 = evaluate_erdos_solution(h_initial_normalized)

    def objective_func(h_candidate):
        sum_h = np.sum(h_candidate)
        if sum_h == 0:
            return np.inf
        h = np.clip(h_candidate * (target_sum / sum_h), 0.0, 1.0)
        c5 = evaluate_erdos_solution(h)
        return c5

    bounds = [(0.0, 1.0)] * n_points
    popsize = 25
    maxiter = int(0.8 * budget_s // 10)

    np.random.seed(seed)
    init_population = [h_initial_normalized]
    for _ in range(popsize - 1):
        vec = np.random.rand(n_points)
        init_population.append(vec)
    init_population = np.array(init_population)

    start_time = time.time()
    result = differential_evolution(
        func=objective_func,
        bounds=bounds,
        popsize=popsize,
        init=init_population,
        seed=seed,
        strategy='best1bin',
        maxiter=maxiter,
        mutation=(0.5, 1.0),
        recombination=0.8,
        polish=False,
        workers=1,
    )
    elapsed = time.time() - start_time

    best_h = h_initial_normalized.copy()
    best_c5 = initial_c5

    if result.success and elapsed < budget_s - 60:
        h_opt = result.x
        sum_h = np.sum(h_opt)
        if sum_h != 0:
            h_opt = np.clip(h_opt * (target_sum / sum_h), 0.0, 1.0)
        c5_opt = evaluate_erdos_solution(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    return (best_h.tolist(), best_c5, n_points)
