def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4.0 => sum(h_full) = n_points/2.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5_and_worst_k(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr)
        worst_k = np.argmax(corr) - (n_points - 1)
        return max_corr * dx, worst_k

    def initial_pop():
        h_full_sym = h_full_initial.copy()
        for i in range(half_n):
            j = n_points - 1 - i
            avg = (h_full_initial[i] + h_full_initial[j]) / 2
            h_full_sym[i] = h_full_sym[j] = avg
        current_sum = np.sum(h_full_sym)
        if current_sum != 0:
            scale = target_sum_half * 2.0 / current_sum
            h_full_sym *= scale
        h_full_sym = np.clip(h_full_sym, 0, 1)
        initial_h_half = h_full_sym[:half_n]
        return initial_h_half

    np.random.seed(seed)
    population_size = 40

    initial_h_half = initial_pop()

        perturbed = np.clip(perturbed, 0, 1)

        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

    best_h_half = initial_h_half.copy()

    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1

        candidates = [(compute_c5_and_worst_k(h)[0], h.copy()) for h in population]
        candidates.sort(key=lambda x: x[0])

        new_population = parents.copy()
