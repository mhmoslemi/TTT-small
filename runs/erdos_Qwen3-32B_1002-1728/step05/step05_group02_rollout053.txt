import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to satisfy the area constraint
    current_sum = h_initial.sum()
    if current_sum != 0:
        h_initial = target_sum / current_sum * h_initial
    h_initial = np.clip(h_initial, 0, 1)

    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]

    best_h = h_initial.copy()
    best_c5 = obj_func(best_h)
    seed_seq = np.random.SeedSequence(seed).generate_state(1)[0]
    rng = np.random.Generator(np.random.PCG64(seed_seq))

    while time.time() - start_time < budget_s - 30:
        remaining = budget_s - (time.time() - start_time)
        frac = remaining / budget_s
        noise_scale = 0.15 * (1.0 - frac**2)

        # Create perturbation
        perturbed = best_h.copy()
        noise = rng.normal(0, noise_scale, n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)

        if not np.isclose(total, 0):
            perturbed = (target_sum / total) * perturbed
            perturbed = np.clip(perturbed, 0, 1)

        # Run optimization
        method = ['L-BFGS-B', 'SLSQP', 'Nelder-Mead'][int(rng.integers(0, 3))]
        options = {'maxiter': 40, 'ftol': 1e-10, 'disp': False}
        if method == 'Nelder-Mead':
            options['xatol'] = 1e-7
            options['fatol'] = 1e-7

        result = minimize(
            fun=obj_func,
            x0=perturbed,
            method=method,
            bounds=bounds,
            constraints=cons,
            options=options
        )

        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_c5 = obj_func(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

        # Exit early for final evaluations
        if remaining < 5:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
