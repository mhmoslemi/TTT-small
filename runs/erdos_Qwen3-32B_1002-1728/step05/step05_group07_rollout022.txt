import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    def project_to_constraints(h_half, target_sum=target_sum_half):
        h_half = np.clip(h_half, 0, 1)
        current_sum = np.sum(h_half)
        if current_sum != 0:
            h_half = (h_half / current_sum) * target_sum
        h_half = np.clip(h_half, 0, 1)
        return h_half

    np.random.seed(seed)
    best_h_half = np.zeros(half_n)
    for i in range(half_n):
        avg = (h_full_initial[i] + h_full_initial[n_points - 1 - i]) / 2
        best_h_half[i] = avg
    best_h_half = project_to_constraints(best_h_half)
    best_c5 = compute_c5(best_h_half)

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    perturbation_radius = 0.3
    max_tries = 20
    try_n = 0
    while time.time() - start_time < budget_s and try_n < max_tries:
        try_n += 1
        current_h = best_h_half + np.random.normal(0.0, 0.1, half_n)
        current_h = project_to_constraints(current_h)

        result = opt.minimize(
            fun=compute_c5,
            x0=current_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 500, 'ftol': 1e-6, 'disp': False}
        )

        if result.success:
            candidate = project_to_constraints(result.x)
            candidate_c5 = compute_c5(candidate)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h_half = candidate
