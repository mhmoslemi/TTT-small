            v = w * v + c1 * r1 * (pbest - current_h) + c2 * r2 * (gbest - current_h)
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            # Update position
            new_h = current_h + v
            new_h = np.clip(new_h, 0, 1)
            current_sum = new_h.sum()
            if current_sum > 1e-8:
                scale = target_sum / current_sum
                new_h *= scale
                new_h = np.clip(new_h, 0, 1)
            particles[i] = new_h
            # Evaluate and update
            new_c5 = evaluate_c5(new_h)
            if new_c5 < personal_best_c5[i]:
                personal_best[i] = new_h.copy()
                personal_best_c5[i] = new_c5
                if new_c5 < best_c5:
                    gbest[:] = new_h
                    best_c5 = new_c5

    # Validation
    assert np.allclose(gbest.sum(), target_sum, atol=1e-6)
    assert np.all((gbest >= 0) & (gbest <= 1))
    return (gbest.tolist(), best_c5, n_points)
