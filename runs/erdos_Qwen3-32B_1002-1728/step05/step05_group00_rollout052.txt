                fun=objective_local, x0=perturbed, method=method,
                bounds=bounds, constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
            )
        else:
            res = minimize(
                fun=objective_local, x0=perturbed, method=method,
                bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
            )
        if res.success:
            new_h = res.x
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if new_sum != 0:
                new_h = (new_h / new_sum) * target_sum
            new_h = np.clip(new_h, 0, 1)
            new_c5 = objective_full(new_h)
            if new_c5 < best_c5:
                best_h_full = new_h
                best_c5 = new_c5

    # Final constraint enforcement and validation
    assert abs(np.sum(best_h_full) - target_sum) < 1e-6
    return (best_h_full.tolist(), best_c5, n_points)
