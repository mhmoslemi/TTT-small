            position_scaled = np.zeros(n_points)
        particles.append(position_scaled)
        velocities.append(0.05 * np.random.randn(n_points))

    particles = np.array(particles)
    velocities = np.array(velocities)
    personal_best_positions = particles.copy()

    personal_best_values = []
    for i in range(num_particles):
        h = particles[i]
        h_clipped = np.clip(h, 0.0, 1.0)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        obj = np.max(corr_vals) * dx
        personal_best_values.append(obj)
    personal_best_values = np.array(personal_best_values)
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = particles[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]

    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(num_particles):
            current_position = particles[i]
            current_velocity = velocities[i]
            pbest = personal_best_positions[i]
            gbest = global_best_position

            r1, r2 = np.random.rand(), np.random.rand()
            new_velocity = (inertia * current_velocity +
                            cognitive * r1 * (pbest - current_position) +
                            social * r2 * (gbest - current_position))
            new_position = current_position + new_velocity

            new_position_clipped = np.clip(new_position, 0.0, 1.0)
            sum_new = np.sum(new_position_clipped)
            if sum_new != 0:
                new_position_scaled = (new_position_clipped / sum_new) * target_sum
                new_position_scaled = np.clip(new_position_scaled, 0.0, 1.0)
            else:
                new_position_scaled = np.zeros(n_points)

            h_clipped = np.clip(new_position_scaled, 0.0, 1.0)
            corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
            obj = np.max(corr_vals) * dx

            velocities[i] = new_velocity
            particles[i] = new_position_scaled

            if obj < personal_best_values[i]:
                personal_best_positions[i] = new_position_scaled
                personal_best_values[i] = obj
                if obj < global_best_value:
                    global_best_position = new_position_scaled.copy()
                    global_best_value = obj

    h_initial = global_best_position.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    linear_constraint = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    def objective_for_min(x):
        x = np.array(x).copy()
        x_clipped = np.clip(x, 0.0, 1.0)
        corr_vals = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr_vals) * dx

    result = minimize(
        fun=objective_for_min,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=linear_constraint,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        h_opt = np.array(result.x)
        sum_opt = np.sum(h_opt)
        if abs(sum_opt - target_sum) > 1e-6:
            h_opt = (h_opt / sum_opt) * target_sum
            h_opt = np.clip(h_opt, 0.0, 1.0)
        obj_opt = objective_for_min(h_opt)
        if obj_opt < global_best_value:
            global_best_value = obj_opt
            global_best_position = h_opt

    assert abs(np.sum(global_best_position) - target_sum) < 1e-6
    return (global_best_position.tolist(), global_best_value, n_points)
