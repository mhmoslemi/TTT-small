        for i in range(n_particles):
            particles[i] = project_position(particles[i], target_sum)

        current_scores = np.array([compute_objective(p, dx) for p in particles])
        improved = current_scores < personal_best_scores
        personal_best_positions[improved] = particles[improved]
        personal_best_scores[improved] = current_scores[improved]

        current_global_idx = np.argmin(personal_best_scores)
        if personal_best_scores[current_global_idx] < global_best_score:
            global_best_score = personal_best_scores[current_global_idx]
            global_best_position = personal_best_positions[current_global_idx].copy()

    top_ratio = 0.2
    n_top = max(1, int(n_particles * top_ratio))
    top_indices = np.argsort(personal_best_scores)[:n_top]
    top_candidates = personal_best_positions[top_indices]

    best_overall = global_best_position
    best_obj = global_best_score

    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 30:
            break

        result = minimize(
            fun=lambda x: compute_objective(x, dx),
            x0=candidate,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
            options={'maxiter': 50, 'ftol': 1e-12}
        )

        if result.success:
            refined = project_position(result.x, target_sum)
            refined_obj = compute_objective(refined, dx)
            if refined_obj < best_obj:
                best_overall = refined
                best_obj = refined_obj

    h_final = best_overall
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6))
    assert abs(np.sum(h_final) - target_sum) < 1e-6

    return (h_final.tolist(), best_obj, n_points)
