import numpy as np
import cvxpy as cp
import time
from collections import defaultdict

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def evaluate_c5(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    initial_c5 = evaluate_c5(h_initial)
    best_h = h_initial.copy()
    best_c5 = initial_c5
    low = 0.0
    high = initial_c5
    max_iter = 30

    half_n = n_points // 2
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 20:
            break
        mid = (low + high) / 2
        # Generate (i,j) pairs where j = i + k for 1 <= k < n_points
        ijs = []
        for k in range(1, n_points):
            for i in range(n_points - k):
                ijs.append((i, i + k))
        # Setup variables and constraints
        h = cp.Variable(n_points)
        w_vars = {}
        constraints = []
        # Symmetry constraints for h
        for i in range(half_n):
            constraints.append(h[i] == h[n_points - 1 - i])
        # Domain constraints for h
        constraints += [
            cp.sum(h) == n_points / 2.0,
            0 <= h,
            h <= 1
        ]
        # Bilinear constraints via McCormick envelopes
        for (i, j) in ijs:
            w = cp.Variable(nonneg=True)
            w_vars[(i, j)] = w
            constraints += [
                w >= h[i] + h[j] - 1,
                w <= h[i],
                w <= h[j],
                w >= 0
            ]
        # Overlap integral constraints
        k_groups = defaultdict(list)
        for (i, j) in ijs:
            k_groups[j - i].append((i, j))
        for k in k_groups:
            indices_list = k_groups[k]
            sum_w = sum(w_vars[(i, j)] for (i, j) in indices_list)
            constraint = sum_w * dx >= 1 - mid
            constraints.append(constraint)
        # Solve convex problem
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=100, verbose=False)
        except:
            prob.status = cp.INFEASIBLE
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h.value
