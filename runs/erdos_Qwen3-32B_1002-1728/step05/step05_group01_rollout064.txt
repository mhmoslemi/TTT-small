        learning_rate *= decay_rate
    
    # Final projection to feasible region
    h_best = np.fft.irfft(best_H, n_points)
    h_sum = np.sum(h_best)
    if abs(h_sum) > 1e-10:
        h_best = h_best * (target_sum / h_sum)
    h_best = np.clip(h_best, 0, 1)
    
    return (h_best.tolist(), best_c5, n_points)
