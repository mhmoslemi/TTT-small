import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        h_clipped = np.clip(h * (target_sum / np.sum(h)), 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 50
    population = [h_initial.copy() for _ in range(5)]
    for _ in range(pop_size - 5):
        h = np.random.rand(n_points)
        h *= target_sum / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)

    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)

    while time.time() - start_time < budget_s - 30:
        candidates = []
        for h in population:
            h_copy = h * (target_sum / np.sum(h))
            h_copy = np.clip(h_copy, 0, 1)
            c5 = evaluate(h_copy)
            candidates.append((c5, h_copy))
        candidates.sort(key=lambda x: x[0])
        if candidates[0][0] < best_c5:
            best_h = candidates[0][1].copy()
            best_c5 = candidates[0][0]

        selected = [c[1] for c in candidates[:5]]
        new_population = selected.copy()
        while len(new_population) < pop_size:
            a, b = np.random.choice(len(selected), 2)
            p1, p2 = selected[a], selected[b]
            split = np.random.randint(1, n_points - 1)
            c1 = np.concatenate((p1[:split], p2[split:]))
            c2 = np.concatenate((p2[:split], p1[split:]))
            for c in [c1, c2]:
                c_scaled = c * (target_sum / np.sum(c))
                c_scaled = np.clip(c_scaled, 0, 1)
                if np.random.rand() < 0.1:
                    noise = np.random.normal(0, 0.05, n_points)
                    c_scaled += noise
                    c_scaled = np.clip(c_scaled, 0, 1)
                    c_scaled *= target_sum / np.sum(c_scaled)
                new_population.append(c_scaled)
        population = new_population[:pop_size]
    return (best_h.tolist(), best_c5, n_points)
