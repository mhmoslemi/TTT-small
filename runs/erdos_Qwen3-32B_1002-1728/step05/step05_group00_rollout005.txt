            x_opt = np.clip(x_opt, 0, 1)
            c5 = evaluate_objective(x_opt, dx)
            if c5 < best_c5:
                return x_opt
        return best_h

    candidates = [best_h.copy(), de_result.x if de_result.success else best_h.copy()]
    for candidate in candidates:
        best_h = local_optimize(candidate)

    # Perturbation loop
    while time.time() - start_time < budget_s - 30:
        perturbation = np.random.normal(0, 0.1, n_points)
        new_candidate = best_h + perturbation
        new_candidate = np.clip(new_candidate, 0, 1)
        sum_new = np.sum(new_candidate)
        if sum_new != 0:
            new_candidate = new_candidate * (target_sum / sum_new)
        new_candidate = np.clip(new_candidate, 0, 1)
        c5_quick = evaluate_objective(new_candidate, dx)
        if c5_quick < best_c5:
            best_h = local_optimize(new_candidate)

    # Final validation
    best_h[np.isclose(best_h, 0)] = 0.0
    best_h[np.isclose(best_h, 1)] = 1.0
    sum_best = np.sum(best_h)
    if not np.isclose(sum_best, target_sum, atol=1e-6):
        best_h = best_h * (target_sum / sum_best)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_objective(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)
