import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    dx = 2.0 / n_points

    np.random.seed(seed)

    elite_fraction = 0.1
    mutation_strength = 0.3
    population = [np.random.rand(n_points) for _ in range(population_size)]
    
    for h in population:
        h[:] = h.astype(np.float64)
        h[:] = h / h.sum() * target_sum
        h[:] = np.clip(h, 0, 1)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def mutate(h):
        noise = np.random.normal(0, mutation_strength, n_points)
        new_h = h + noise
        new_h[:] = new_h.astype(np.float64)
        new_h[:] = new_h / np.sum(new_h) * target_sum
        new_h[:] = np.clip(new_h, 0, 1)
        return new_h
    
    def crossover(p1, p2):
        alpha = np.random.uniform(0.3, 0.7)

        child[:] = child.astype(np.float64)
        child[:] = child / np.sum(child) * target_sum
        child[:] = np.clip(child, 0, 1)
        return child

    best_h = h_full_initial.copy()
