    sequence = []
    for i in range(n):
        bits = []
        num = i
        while num > 0:
            bits.append(num % base)
            num = num // base
        val = 0.0
        for k in range(len(bits)):
            val += bits[k] / (base ** (k + 1))
        sequence.append(val)
    return np.array(sequence)
