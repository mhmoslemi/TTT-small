def run(seed=42, budget_s=1100, **kwargs):

        h_half_initial *= scale

    population = []
    for _ in range(population_size):
        h_half = np.random.uniform(-0.05, 0.05, half_n) + h_half_initial
        h_half = np.clip(h_half, 0, 1)
        current_sum = np.sum(h_half)
        if current_sum > 1e-8:
            h_half *= target_sum_half / current_sum
        h_half = np.clip(h_half, 0, 1)
        population.append(h_half)
    population = np.array(population)

    def evaluate_c5(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]

        return np.max(corr) * dx

    def mutate(h_half, strength=0.01):
        h_mut = h_half + np.random.normal(0, strength, half_n)

        if current_sum > 1e-8:

        h_mut = np.clip(h_mut, 0, 1)
        return h_mut

    def crossover(p1, p2):

    while time.time() - start_time < budget_s - 5:
        new_pop = []
        for _ in range(population_size):
