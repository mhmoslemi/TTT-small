import numpy as np

def run(seed=42, budget_s=1100, **kwargs):

        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 40
    population = []

    # Initialize population with perturbed variants of initial solution
    for _ in range(population_size):
        h_half = h_full_initial[:half_n].copy()
        h_half += np.random.normal(0, 0.1, half_n)

    # Evolutionary optimization loop
    while time.time() - start_time < budget_s - 5:
