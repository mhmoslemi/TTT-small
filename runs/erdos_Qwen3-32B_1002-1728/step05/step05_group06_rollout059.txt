                g_full[m] -= current_h[j2] * dx

        # Project gradient to symmetric half
        g_half = (g_full[:half_n] + g_full[-half_n:][::-1]) / 2

        # Gradient descent update
        h_half = h_half - learning_rate * g_half
        h_half = np.clip(h_half, 0.0, 1.0)

        # Reconstruct and normalize h_full
        h_full = np.concatenate([h_half, h_half[::-1][:- (n_points % 2)]])
        current_sum = h_full.sum()
        if current_sum != 0:
            h_full = (h_full / current_sum) * target_sum
        h_full = np.clip(h_full, 0.0, 1.0)
        current_sum = h_full.sum()
        if current_sum != 0:
            h_full = (h_full / current_sum) * target_sum
        h_full = np.clip(h_full, 0.0, 1.0)

        current_c5 = compute_c5(h_full)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full.copy()
        learning_rate *= lr_decay

    return (best_h.tolist(), best_c5, n_points)
