    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    # Function to evaluate C5
    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Function to normalize h_half to sum to target_sum_half
    def normalize(h_half):
        current_sum = np.sum(h_half)
        if np.isclose(current_sum, 0):
            return h_half  # Shouldn't happen
        scale = target_sum_half / current_sum
        return np.clip(h_half * scale, 0, 1)

    # Initialize particles
    np.random.seed(seed)
    n_particles = 50
    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_scores = []
    positions = []

    initial_h_half = h_full_initial[:half_n].copy()

    for _ in range(n_particles):

        h_perturbed = np.clip(initial_h_half + perturbation, 0, 1)
        h_perturbed = normalize(h_perturbed)
        positions.append(h_perturbed)

        personal_best_positions.append(h_perturbed.copy())
        personal_best_scores.append(evaluate_c5(h_perturbed))
    positions = np.array(positions)
    velocities = np.array(velocities)
    personal_best_positions = np.array(personal_best_positions)
    personal_best_scores = np.array(personal_best_scores)

    global_best_idx = np.argmin(personal_best_scores)

            positions[i] = normalize(positions[i])

            if current_score < personal_best_scores[i]:

                personal_best_positions[i] = positions[i].copy()
                # Update global best

                    global_best_score = current_score
                    global_best_position = positions[i].copy()
                    no_improve_count = 0
            else:
                # Personal best not updated
                pass
        # Increment no_improve count and check stopping criterion
        no_improve_count += 1
        if no_improve_count >= patience:
            print("Early stopping due to no improvement")
            break

    assert np.allclose(np.sum(best_h_full), n_points / 2.0, atol=1e-6)

            vel_cognitive = c1 * r1 * (personal_best_positions[i] - positions[i])
            vel_social = c2 * r2 * (global_best_position - positions[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            positions[i] = normalize(positions[i])

        for i in range(n_particles):
            current_score = evaluate_c5(positions[i])
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_positions[i] = positions[i].copy()
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best_position = positions[i].copy()
                    no_improve_count = 0
            else:
                pass
        if no_improve_count >= patience:
            break

    best_h_half = global_best_position
    best_h_full = create_full_h(best_h_half)
    best_h_full = np.clip(best_h_full, 0, 1)
    assert np.allclose(np.sum(best_h_full), n_points / 2.0, atol=1e-6)
    return (best_h_full.tolist(), global_best_score, n_points)
