    delta_x_control = x_control[1] - x_control[0]
    alphas = (x_samples - x_control[j_indices]) / delta_x_control

    def create_h_from_p(p):
        p = np.array(p, dtype=np.float64)
        border_hj0 = p[j_indices]
        border_hj1 = p[j_indices + 1]
        h = border_hj0 + alphas * (border_hj1 - border_hj0)
        h_sum = np.sum(h) * dx

            return np.zeros_like(h)
        scale = target_sum / h_sum
        h_scaled = h * scale
