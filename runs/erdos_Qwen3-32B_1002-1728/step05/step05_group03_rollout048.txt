import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_gradient(h_full, k):
        n = len(h_full)
        grad = np.zeros(n)
        if k == 0:
            grad = 1.0 - 2.0 * h_full
        else:
            k_abs = abs(k)
            valid_slices = slice(0, n - k_abs)
            # Term1: 1 - h[i +k_abs]
            grad[valid_slices] += 1.0 - h_full[k_abs:]
            # Term2: -h[i -k_abs]
            if k_abs > 0:
                grad[k_abs:] -= h_full[0 : n - k_abs]
        return grad

    def normalize_h_half(h_half, target_sum):
        h = h_half.copy()
        current_sum = np.sum(h)
        if current_sum == 0:
            return h.copy()
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        new_sum = np.sum(h)
        if new_sum == 0:
            return h.copy()
        h = h * (target_sum / new_sum)
        return np.clip(h, 0, 1)

    np.random.seed(seed)
    initial_h_half = h_full_initial[:half_n].copy()
    h_half = normalize_h_half(initial_h_half, target_sum_half)
    best_h_half = h_half.copy()
    best_c5 = float('inf')

    learning_rate = 0.1
    decay_rate = 0.95
    perturb_interval = 50
    no_improve_limit = 100

    iteration = 0
    no_improve_count = 0
