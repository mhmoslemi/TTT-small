import numpy as np
from numpy.random import Generator, PCG64
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return h_clipped.copy()
        scale = target_sum / total
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)
    
    h_initial = project(h_full_initial)
    rng = Generator(PCG64(seed))
    population = [h_initial.copy()]
    noise = rng.normal(0, 0.1, (19, n_points))
    new_individuals = h_initial.copy()[None, :] + noise
    for h in new_individuals:
        population.append(project(h))
    
    def evaluate(individual):
        h = project(individual)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx
    
    scores = [evaluate(p) for p in population]
    best_idx = np.argmin(scores)
    best_individual = population[best_idx].copy()
    best_score = scores[best_idx]

    max_generations = 30
    population_size = len(population)
    elite_rate = 0.05
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 30:
            break

        elite_count = max(1, int(elite_rate * population_size))
        scores_np = np.array(scores)
        elite_indices = np.argsort(scores_np)[:elite_count]
        elites = [population[i] for i in elite_indices]
        new_population = elites.copy()

        num_required = population_size - elite_count
        for _ in range(num_required):
            parent1 = population[rng.choice(len(population), p=scores_np/np.sum(scores_np))]
            parent2 = population[rng.choice(len(population), p=scores_np/np.sum(scores_np))]
            if rng.random() < 0.8:
                child = np.where(rng.random(n_points) < 0.5, parent1, parent2)
            else:
                child = parent1.copy()
            mut_rate = max(0.05, 0.15 * (1 - generation / max_generations))
            if rng.random() < mut_rate:
                mask = rng.random(n_points) < mut_rate
                child[mask] += rng.normal(0, 0.15, size=np.sum(mask))
            child = project(child)
            new_population.append(child)
        population = new_population
        scores = [evaluate(p) for p in population]
        new_best_idx = np.argmin(scores)
        if scores[new_best_idx] < best_score:
            best_individual = population[new_best_idx].copy()
            best_score = scores[new_best_idx]

    top_indices = np.argsort(scores)[:5]
    for idx in top_indices:
        if time.time() - start_time > budget_s - 5:
            break
        x0 = population[idx].copy()
        def obj(x):
            h = project(x)
            corr_vals = np.correlate(h, 1 - h, mode='full')
            return np.max(corr_vals) * dx
        bounds = [(0, 1)] * n_points
        result = minimize(obj, x0=x0, method='COBYLA', bounds=bounds, options={'maxiter': 200, 'rhobeg': 0.1, 'disp': False})
