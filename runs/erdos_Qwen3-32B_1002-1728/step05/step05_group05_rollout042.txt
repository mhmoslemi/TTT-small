    for i in range(n_points):
        pairs.append((i, i))  # k=0 case
    for i in range(n_points):
        for k in range(1, n_points):
            j = i - k
            if j >= 0:
                pairs.append((i, j))
    for i in range(n_points):
        for k in range(1, n_points):
            j = i + k
            if j < n_points:
                pairs.append((i, j))

    # Map each (i,j) to its k = i-j
    k_to_pairs = collections.defaultdict(list)
    for i, j in pairs:
        k = i - j
        k_to_pairs[k].append((i, j))

    # Initialize best solution
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    start_time = time.time()
    low = 0.0
    high = 0.3809  # Target bound

    while time.time() - start_time < budget_s - 20:
        mid = (low + high) / 2

        # Define optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2.0, h <= 1]

        # McCormick variables for each (i,j) pair
        z_dict = {}
        for i, j in pairs:
            z = cp.Variable()
            z_dict[(i, j)] = z
            constraints += [
                z >= h[i] - h[j],
                z <= h[i],
                z <= 1 - h[j],
                z >= 0
            ]

        # Add constraints for each k
        for k, k_pairs in k_to_pairs.items():
            sum_z = cp.sum([z_dict[(i, j)] for i, j in k_pairs])
            constraints.append(sum_z <= mid / dx)

        problem = cp.Problem(cp.Minimize(0), constraints)

        # Solve using best available solver
        solver = cp.GUROBI if 'GUROBI' in cp.installed_solvers() else cp.CVXOPT
        problem.solve(solver=solver, verbose=False)

        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_vals = np.array([v.value for v in h])
            # Post-process: clip and scale
            h_vals = np.clip(h_vals, 0, 1)
            sum_h = h_vals.sum()
            if abs(sum_h - n_points / 2.0) > 1e-6:
                scale = (n_points / 2.0) / sum_h
                h_vals *= scale
                h_vals = np.clip(h_vals, 0, 1)
            # Evaluate actual C5
            corr = np.correlate(h_vals, 1 - h_vals, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals.copy()
            high = mid
        else:
            low = mid

        # Break early to respect time budget
        if time.time() - start_time > budget_s - 10:
            break

    # Final validation and return
    assert np.allclose(best_h.sum(), n_points / 2.0, atol=1e-6)
    return (best_h.tolist(), best_c5, n_points)
