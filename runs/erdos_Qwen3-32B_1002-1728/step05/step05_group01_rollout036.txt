        perturbed = (np.random.rand(len(best_h)) - 0.5) * 0.2
        perturbed = np.clip(best_h + perturbed, -max_diffs, max_diffs)
        for meth in ['COBYLA', 'SLSQP']:
            result = minimize(objective, perturbed, bounds=bounds, method=meth, options={'maxiter': 100})
            if result.success and result.fun < best_c5:
                best_c5 = result.fun
                best_h = result.x
        attempts += 1
    
    h_final = np.zeros(n_points)
    for i in range(len(avg)):
        h_final[2*i] = avg[i] + best_h[i]
        h_final[2*i + 1] = avg[i] - best_h[i]
    h_clipped = np.clip(h_final, 0, 1)
    sum_h = h_clipped.sum()
    scaling = (n_points / 2.0) / sum_h if sum_h > 1e-10 else 1.0
    h_normalized = np.clip(h_clipped * scaling, 0, 1)
    
    corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
    final_c5 = np.max(corr) * dx
    return (h_normalized.tolist(), final_c5, n_points)
