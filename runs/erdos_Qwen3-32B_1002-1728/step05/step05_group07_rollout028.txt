import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full = np.array(initial_h_values)
    n_points = len(h_full)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize and normalize initial h
    h_full = (h_full + h_full[::-1]) / 2.0
    total = np.sum(h_full)
    if total != 0:
        h_full = h_full * (target_sum / total)
    h_full = np.clip(h_full, 0, 1)

    best_h = h_full.copy()
    best_c5 = np.inf

    # Optimization parameters
    learning_rate = 0.1
    lr_decay = 0.995
    iteration = 0

    while time.time() - start_time < budget_s - 5:
        iteration += 1
        # Compute overlaps and find worst shift k
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlaps = corr * dx
        current_c5 = np.max(overlaps)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full.copy()

        worst_k = np.argmax(overlaps)

        # Coordinate-wise gradient updates
        for i in range(half_n):
            j = n_points - 1 - i
            # Compute gradients for i and j
            i_plus_k_i = (i + worst_k) % n_points
            i_minus_k_i = (i - worst_k + n_points) % n_points
            grad_i = (1 - h_full[i_plus_k_i] - h_full[i_minus_k_i]) * dx
            h_full[i] -= learning_rate * grad_i
