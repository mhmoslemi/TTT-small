import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half.copy()
        h_full[half_n:] = h_half[::-1].copy()
        return h_full

    def objective(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Enforce symmetry and scale to target_sum_half
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    initial_sum = np.sum(h_full_sym)
    scale = target_sum_half * 2.0 / initial_sum if initial_sum != 0 else 1.0
    h_full_sym = np.clip(h_full_sym * scale, 0, 1)
    initial_h_half = h_full_sym[:half_n]

    bounds = Bounds(0, 1)
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum_half}]

    np.random.seed(seed)
    best_c5 = float('inf')
    best_h_full = create_full_h(initial_h_half)

    restarts = [initial_h_half.copy()]

    # Generate perturbed initial guesses
    for _ in range(20):
        perturbed = initial_h_half + np.random.normal(0, 0.05, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        restarts.append(perturbed)

    # Try each restart
    for x0 in restarts:
        if time.time() - start_time >= budget_s - 10:
            break
        result = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if not result.success:
            continue
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h_half = result.x
            best_h_full = create_full_h(best_h_half)
            best_c5 = current_c5

    assert abs(np.sum(best_h_full) - n_points / 2.0) < 1e-6
    return (best_h_full.tolist(), best_c5, n_points)
