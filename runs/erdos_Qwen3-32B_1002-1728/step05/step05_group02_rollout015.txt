        h_half_initial = h_half_initial * (target_sum_half / current_sum)
    h_half_initial = np.clip(h_half_initial, 0, 1)

    # Initialize population
    pop_size = 30
    np.random.seed(seed)
    population = []
    for _ in range(pop_size):
        h_half = h_half_initial + np.random.normal(0, 0.1, half_n)
        h_half = np.clip(h_half, 0, 1)
        total = h_half.sum()
        if total != 0:
            h_half = (h_half / total) * target_sum_half
        h_half = np.clip(h_half, 0, 1)
        population.append(h_half)

    best_c5 = np.inf
    best_h_half = h_half_initial.copy()

    while time.time() - start_time < budget_s - 30:
        evaluated = [(evaluate_c5(h), h) for h in population]
        evaluated.sort()
        current_c5, current_h = evaluated[0]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = current_h.copy()

        top_n = pop_size // 2
        top_h = [h for c, h in evaluated[:top_n]]

        # Crossover: average top candidates
        children = []
        for _ in range(pop_size // 2):
            p1 = top_h[np.random.randint(len(top_h))]
            p2 = top_h[np.random.randint(len(top_h))]
            child = (p1 + p2) / 2.0
            child = np.clip(child, 0, 1)
            total = child.sum()
            if total != 0:
                child = (child / total) * target_sum_half
            child = np.clip(child, 0, 1)
            children.append(child)

        # Mutation
        for i in range(len(children)):
            if np.random.rand() < 0.3:
                noise = np.random.normal(0, 0.2 * (1 - (time.time()-start_time)/budget_s), half_n)
                children[i] += noise
                children[i] = np.clip(children[i], 0, 1)
                total = children[i].sum()
                if total != 0:
                    children[i] = (children[i] / total) * target_sum_half
                children[i] = np.clip(children[i], 0, 1)

        new_population = top_h + children
        population = new_population[:pop_size]

        # Local search
        if time.time() - start_time < budget_s - 60:
            top_optimize = min(5, len(top_h))
            for i in range(top_optimize):
                c, h0 = evaluated[i]
                cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
                result = minimize(
                    evaluate_c5, h0, method='L-BFGS-B',
                    bounds=[(0, 1)]*half_n,
                    constraints=cons,
                    options={'maxiter': 30, 'ftol': 1e-12}
                )
                if result.success:
                    new_h = np.clip(result.x, 0, 1)
                    total = new_h.sum()
                    if total != 0:
                        new_h = (new_h / total) * target_sum_half
                    new_h = np.clip(new_h, 0, 1)
                    new_c = evaluate_c5(new_h)
                    if new_c < best_c5:
                        best_c5 = new_c
                        best_h_half = new_h.copy()
                    population.append(new_h)

    h_full = create_full_h(best_h_half)
    assert abs(h_full.sum() - target_sum_full) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), best_c5, n_points)
