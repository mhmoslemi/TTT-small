    n_control = 14  # 14 control points for 70 samples (5 samples per segment)
    chunk_size = n_points // n_control

    np.random.seed(seed)

    def control_points_to_h(control_points, chunk_size):
        h = np.zeros(n_points)
        for i in range(n_control):
            for j in range(chunk_size):
                t = j / chunk_size
                h[i * chunk_size + j] = (1 - t) * control_points[i] + t * control_points[(i + 1) % n_control]
        return h

    def generate_h(control_points):
        h = control_points_to_h(control_points, chunk_size)
        total = np.sum(h)
        if total < 1e-12:
            return np.zeros(n_points)
        scale = (n_points / 2.0) / total
        return np.clip(h * scale, 0, 1.0)

    def evaluate_c5(ctrl):
        h = generate_h(ctrl)
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr) * dx

    def initialize_population(pop_size):
        pop = []
        for _ in range(pop_size):
            pop.append(np.random.uniform(0.4, 0.6, n_control))  # Avoid 0/1 for better scaling
        # Add initial control points from h_full_initial
        if np.sum(h_full_initial) > 1e-12:
            initial_ctrl = h_full_initial[::chunk_size]
            pop.append(initial_ctrl)
        return pop

    pop_size = 50
    population = initialize_population(pop_size)
    best_ctrl = population[0]
    best_c5 = evaluate_c5(best_ctrl)
    best_h = generate_h(best_ctrl)

    while time.time() - start_time < budget_s - 30:
        # Evaluate population
        evals = [evaluate_c5(ctrl) for ctrl in population]
        idx_sorted = np.argsort(evals)
        # Update best
        if evals[idx_sorted[0]] < best_c5:
            best_c5 = evals[idx_sorted[0]]
            best_ctrl = population[idx_sorted[0]]
            best_h = generate_h(best_ctrl)
        # Select parents (top 50%)
        top_pct = 0.5
        parents = [population[i] for i in idx_sorted[:int(pop_size * top_pct)]]
        # Create new population via crossover and mutation
        new_pop = []
        for _ in range(len(population) - len(parents)):
            if not parents:
                break
            p1 = np.random.choice(parents)
            p2 = np.random.choice(parents)
            # Crossover
            alpha = np.random.beta(0.6, 0.6)
            child = alpha * p1 + (1 - alpha) * p2
            # Mutation
            if np.random.rand() < 0.4:
                child += np.random.normal(0, 0.15, n_control)
                child = np.clip(child, 0.2, 0.8)  # Constrain mutation to safe range
            new_pop.append(child)
        population = parents + new_pop
        # Local refine best candidates
        for _ in range(5):
            if time.time() - start_time > budget_s - 30:
                break
            idx = np.random.randint(len(population))
            res = minimize(evaluate_c5, population[idx], method='L-BFGS-B', bounds=[(0.2, 0.8)] * n_control, options={'maxiter': 30})
            if res.success and res.fun < best_c5:
                best_c5 = res.fun
                best_ctrl = res.x
                best_h = generate_h(best_ctrl)

    # Final population evaluation
    for ctrl in population:
        c5 = evaluate_c5(ctrl)
        if c5 < best_c5:
            best_c5 = c5
            best_ctrl = ctrl
            best_h = generate_h(ctrl)

    # Ensure solution meets constraints
    h_final = best_h
    assert abs(np.sum(h_final) - n_points / 2.0) < 1e-6
    assert all(0 <= x <= 1 + 1e-6 for x in h_final)

    return (h_final.tolist(), best_c5, n_points)
