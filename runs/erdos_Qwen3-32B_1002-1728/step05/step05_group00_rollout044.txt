from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project_feasible(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        scaling_factor = target_sum / current_sum
        h_scaled = h_clipped * scaling_factor
        return np.clip(h_scaled, 0, 1)

    def evaluate_candidate(h):
        h_proj = project_feasible(h)
        corr_vals = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr_vals) * dx

    # PSO parameters
    n_particles = 20
    max_iter = 20
    w = 0.8
    c1 = 1.2
    c2 = 1.2

    # Initialize particles
    np.random.seed(seed)
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    pbest_positions = np.zeros((n_particles, n_points))
    pbest_scores = np.full(n_particles, np.inf)
    gbest_position = np.zeros(n_points)
    gbest_score = np.inf

    for i in range(n_particles):
        particle = h_initial.copy().astype(float)
        pert = np.random.normal(0, 0.1, n_points)
        particle += pert
        particle = project_feasible(particle)
        particles[i] = particle
        pbest_positions[i] = particle.copy()
        score = evaluate_candidate(particle)
        pbest_scores[i] = score
        if score < gbest_score:
            gbest_score = score
            gbest_position = particle.copy()

    # PSO iterations
    while time.time() - start_time < budget_s * 0.8:
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.8:
                break
