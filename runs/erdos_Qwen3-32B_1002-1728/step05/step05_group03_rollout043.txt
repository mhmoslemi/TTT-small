        # Clip to bounds
        current_half = np.clip(current_half, 0, 1)

        # Enforce sum constraint
        full_h = create_full_h(current_half)
        current_sum = np.sum(full_h)
        if abs(current_sum - target_sum) > 1e-6:
            scale_factor = target_sum / (np.sum(current_half) * 2 + 1e-10)
            current_half = current_half * scale_factor
            current_half = np.clip(current_half, 0, 1)

        # Update best solution if improved
        current_c5, _ = evaluate_c5(current_half)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_half.copy()
            best_full = create_full_h(best_h)

    # Final evaluation
    h_values = create_full_h(best_h)
    return (h_values.tolist(), best_c5, n_points)
