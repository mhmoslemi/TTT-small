import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_full = h_vars
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_h_full = h_full_initial.copy()
    best_c5 = objective(best_h_full)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        noise_scale = 0.35 * (1 - (frac ** 0.75))  # Increased exploration early on

        for _ in range(4):  # More perturbations per iteration
            perturbed = best_h_full + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if abs(total) < 1e-10:
                continue  # Skip if sum is zero
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            # Try multiple methods (remove COBYLA for speed)
            for method in ['L-BFGS-B', 'SLSQP']:
                result = minimize(
                    fun=objective,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
                )
                if result.success and result.fun < best_c5:
                    new_h_full = np.clip(result.x, 0, 1)
                    corr = np.correlate(new_h_full, 1 - new_h_full, mode='full')
                    new_c5 = np.max(corr) * dx
                    if abs(np.sum(new_h_full) - target_sum) < 1e-6 and new_c5 < best_c5:
                        best_h_full = new_h_full
                        best_c5 = new_c5
                        break  # Break inner loop if improved

    h_full = best_h_full
    assert abs(np.sum(h_full) - target_sum) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), best_c5, n_points)
