    m = len(control_points)
    dx_bezier = 2.0 / (m - 1) if m > 1 else 2.0
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * (2.0 / n_points)
        # Find which control point interval x is in
        if m == 1:
            h[i] = control_points[0]
        else:
            idx = np.clip(int(x / dx_bezier), 0, m-2)
            x0 = idx * dx_bezier

            h[i] = (1 - t) * control_points[idx] + t * control_points[idx + 1]
    # Compute sum to scale
    current_sum = np.sum(h)
    if current_sum == 0:
        return h  # all zeros, but this is invalid
    scale = (n_points / 2.0) / current_sum

    # Clip to [0, 1]

    def get_h_samples(control_points):
        m = len(control_points)
        h = np.zeros(n_points)
        if m == 1:
            h[:] = control_points[0]
        else:
            dx_bezier = 2.0 / (m - 1)
            for i in range(n_points):
                x = i * dx
                idx = int(x / dx_bezier)
                idx = np.clip(idx, 0, m-2)
                x0 = idx * dx_bezier
                x1 = (idx + 1) * dx_bezier
                t = (x - x0) / (x1 - x0)
                h[i] = (1 - t) * control_points[idx] + t * control_points[idx + 1]
        # Normalize to integral constraint
        total = np.sum(h) * dx
        if np.abs(total) < 1e-10:
