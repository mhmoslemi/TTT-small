        h = h_full_initial + np.random.normal(0, 0.1, n_points)
        h = project_feasible(h)
        population.append(h)

    best_c5 = np.inf
    best_h = h_full_initial.copy()

    def objective(h_candidate):
        h = project_feasible(h_candidate)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # GA loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 100:
            break

        fitness = [objective(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Selection
        new_population = []
        top_k = int(0.05 * pop_size)
        elite_indices = np.argsort(fitness)[:top_k]
        for idx in elite_indices:
            new_population.append(population[idx])

        # Fill rest with offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 3
            idxs = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in idxs]
            winner_idx = idxs[np.argmin(tournament_fitness)]
            parent1 = population[winner_idx]

            idxs = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in idxs]
            winner_idx = idxs[np.argmin(tournament_fitness)]
            parent2 = population[winner_idx]

            # Crossover
            cross_prob = 0.8
            if np.random.rand() < cross_prob:
                mask = np.random.rand(n_points) < 0.5
                child = parent1 * mask + parent2 * (1 - mask)
            else:
                child = parent1.copy()

            # Mutation
            mutation_rate = mutation_start_rate * (1 - generation / generations)
            if np.random.rand() < 0.3:  # Mutation chance per step
                mutate_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                noise = np.random.normal(0, 0.1, len(mutate_indices))
                child[mutate_indices] += noise
                child[mutate_indices] = np.clip(child[mutate_indices], 0, 1)

            child = project_feasible(child)
            new_population.append(child)

        population = new_population

    # Local optimization
    candidates = [project_feasible(ind) for ind in population]
    candidates.append(best_h)
    for candidate in candidates[:5]:
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(
            fun=objective,
            x0=candidate,
            method='COBYLA',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 200, 'disp': False}
        )
        if result.success:
            optimized_h = project_feasible(result.x)
            c5 = objective(optimized_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = optimized_h

    best_h = project_feasible(best_h)
    best_c5 = objective(best_h)
    return (best_h.tolist(), best_c5, n_points)
