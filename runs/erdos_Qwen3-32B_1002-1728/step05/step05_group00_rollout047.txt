    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project(h):
        h_clipped = np.clip(h, 0, 1.0)
        total = h_clipped.sum()
        if total == 0:
            return h_clipped.copy()

        h_clipped *= scale

    generations = 30
    tournament_size = 3
    mutation_rate = 0.15
    elite_ratio = 0.05

        noise = np.random.randn(n_points) * 0.1
        new_h = h_initial + noise
        new_h = project(new_h)
        population.append(new_h.copy())

    def evaluate_C5(h, dx):

        return np.max(corr_vals) * dx

    best_h = h_initial.copy()
