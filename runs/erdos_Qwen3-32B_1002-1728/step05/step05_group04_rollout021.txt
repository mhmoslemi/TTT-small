        for i in range(population_size):
            c5 = evaluate(population[i])
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
                best_idx = i
            eval_count += 1

        # Create new population with mutations and elitism
        new_population = []
        # Preserve best solution
        new_population.append(best_h.copy())

        # Add diverse mutations
        for idx in np.random.permutation(population_size)[:int(0.7 * population_size)]:
            parent = population[idx].copy()
            # Mutation with adaptive range
            mu = 0.05 * max(0.2, 1 - GENERATIONS * 0.005)
            count = 3 + np.random.randint(2)
            positions = np.random.choice(n_points, count, replace=False)
            for pos in positions:
                perturb = np.random.uniform(-1, 1)
                parent[pos] += perturb * mu
            parent = np.clip(parent, 0.0, 1.0)
            # Fix sum constraint
            parent_sum = np.sum(parent)
            if abs(parent_sum - target_sum) > 1e-6:
                parent *= target_sum / parent_sum
            new_population.append(parent)

        # Fill remaining spots with random individuals
        while len(new_population) < population_size:
            new_population.append(create_individual() * target_sum / np.sum(create_individual()))
        
        population = new_population

    return (best_h.tolist(), best_c5, n_points)
