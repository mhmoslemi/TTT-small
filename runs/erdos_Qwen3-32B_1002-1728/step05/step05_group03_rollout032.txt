import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = h_copy.sum()
        if current_sum == 0:
            return h_copy
        scale = target_sum / current_sum
        scaled = h_copy * scale
        clipped = np.clip(scaled, 0, 1)
        new_sum = clipped.sum()
        if new_sum == 0:
            return clipped
        scale2 = target_sum / new_sum
        scaled2 = clipped * scale2
        return scaled2

    np.random.seed(seed)
    h_half = initial_h_values[:half_n].copy()
    h_half = normalize_h_half(h_half, target_sum_half)

    best_h_half = h_half.copy()
    best_c5 = float('inf')

    learning_rate = 0.1
    learning_rate_decay = 0.99
    perturb_interval = 100
    no_improvement_counter = 0
    max_no_improvement = 10000

    while time.time() - start_time < budget_s - 10:
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()
            no_improvement_counter = 0
        else:
            no_improvement_counter += 1

        if no_improvement_counter >= perturb_interval:
            perturbation = np.random.normal(0, 0.05, h_half.shape)
            h_half = np.clip(h_half + perturbation, 0, 1)
            h_half = normalize_h_half(h_half, target_sum_half)
            no_improvement_counter = 0

        idx_max = np.argmax(corr)
        k_corr = idx_max - (n_points - 1)

        grad_full = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k_corr
            if i_plus_k < n_points:
                grad_full[i] += (1 - h_full[i_plus_k]) * dx
            if (i - k_corr) >= 0:
                grad_full[i] -= h_full[i - k_corr] * dx

        grad_half = np.zeros(half_n)
        for i in range(half_n):
            grad_half[i] = grad_full[i] + grad_full[n_points - 1 - i]

        h_half -= learning_rate * grad_half
        h_half = np.clip(h_half, 0, 1)
        h_half = normalize_h_half(h_half, target_sum_half)

        learning_rate *= learning_rate_decay

    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx
    return (h_full.tolist(), final_c5, n_points)
