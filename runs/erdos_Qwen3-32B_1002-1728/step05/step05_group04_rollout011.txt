            objective, bounds, args=(), strategy='best1bin',
            maxiter=int(max(1, time_de // 0.1)), popsize=10,
            recombination=0.9, mutation=0.8, tol=1e-3,
            mutation_strategy='best1bin', seed=seed,
            callback=None, polish=True, init='sobol',
            atol=1e-3
        )
        if result_de.success and objective(result_de.x) < best_c5:
            new_h = np.clip(result_de.x, 0, 1)
            best_h = new_h
            best_c5 = objective(best_h)
    except:
        pass

    # Phase 2: Perturbation with local optimization
    np.random.seed(seed)
    n_attempt = 0
    max_attempts = (budget_s - (time.time() - start_time)) // 30

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        seed += 1
        frac = (time.time() - start_time) / budget_s
        noise_scale = 0.2 * (1 - frac ** 1.5)

        for _ in range(2):
            seed += 1
            rnd = np.random.RandomState(seed)
            perturbed = best_h + rnd.normal(0, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            method = 'COBYLA' if n_attempt % 2 == 0 else 'SLSQP'
            result = minimize(
                fun=objective,
                x0=perturbed,
                bounds=bounds,
                method=method,
                options={
                    'maxiter': 200,
                    'ftol': 1e-8,
                    'disp': False
                }
            )

            if result.success and result.fun < best_c5:
                new_h = np.clip(result.x, 0, 1)
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
