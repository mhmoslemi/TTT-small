from scipy.optimize import minimize, COBYLA
import time

    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_half = np.clip(h_half, 0, 1)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        total = np.sum(h_full)
        if total != 0:
            h_full = (h_full / total) * (n_points / 2.0)
        return np.clip(h_full, 0, 1)

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Prepare initial half-vector and ensure feasibility
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = np.clip(h_half_initial, 0, 1)

        h_half_initial = (h_half_initial / total_val) * target_sum_half
    h_half_initial = np.clip(h_half_initial, 0, 1)

    # Initialize GA population

    population_size = 20
    pop = [h_half_initial.copy() + np.random.normal(0, 0.1, half_n) for _ in range(population_size)]

        current_min_idx = np.argmin(fitness)
        if fitness[current_min_idx] < best_c5:
            best_h_half = current_pop[current_min_idx].copy()

        # Selection via tournament
        new_pop = []
        n_offspring = population_size - n_elite
        for _ in range(n_offspring):
            # Select parents
            participants = np.random.choice(len(current_pop), size=n_tournament, replace=False)
