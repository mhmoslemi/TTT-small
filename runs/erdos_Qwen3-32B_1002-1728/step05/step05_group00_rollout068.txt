import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0
    target_sum_full = n_points / 2.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        h_clipped = np.clip(h_full, 0, 1)
        sum_clipped = h_clipped.sum()
        if sum_clipped == 0:
            h_scaled = h_clipped.copy()
        else:
            h_scaled = h_clipped * (target_sum_full / sum_clipped)
        h_final = np.clip(h_scaled, 0, 1)
        corr_vals = np.correlate(h_final, 1 - h_final, mode='full')
        return np.max(corr_vals) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - target_sum_full) > 1e-6 and current_sum != 0:
        scale = target_sum_full / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    # Differential Evolution Global Optimization
    def objective_de(x_vars):
        target_sum = target_sum_half
        x = np.array(x_vars)
        total = np.sum(x)
        for _ in range(2):  # Two passes of clip-rescale
            if total == 0:
                x_scaled = x.copy()
            else:
                x_scaled = (x / total) * target_sum
            x_clipped = np.clip(x_scaled, 0, 1)
            sum_clipped = x_clipped.sum()
            if sum_clipped == 0:
                x = x_clipped.copy()
                total = 0
            else:
                x = x_clipped * (target_sum / sum_clipped)
                total = target_sum
        h_full = create_full_h(x)
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    bounds_de = [(0.0, 1.0)] * half_n
    np.random.seed(seed)
    de_start = time.time()
    try:
        de_result = differential_evolution(
            objective_de, bounds_de, strategy='best1bin',

            recombination=0.9, seed=seed, workers=1, disp=False

        if de_result.success:

            total = np.sum(de_h_half)

                de_h_half = (de_h_half / total) * target_sum_half
            de_h_half = np.clip(de_h_half, 0, 1)
