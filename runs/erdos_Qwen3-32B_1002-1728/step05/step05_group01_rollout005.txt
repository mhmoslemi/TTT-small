import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_vars):
        h_clipped = np.clip(h_vars, 0.0, 1.0)
        current_sum = np.sum(h_clipped)
        if current_sum < 1e-10:
            return float('inf')
        h_normalized = h_clipped * (target_sum / current_sum)
        corr = np.correlate(h_normalized, 1 - h_normalized, mode='full')
        return np.max(corr) * dx

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum * (1 - 1e-8)})  # Avoid strict equality issues

    h_current = h_initial.copy()
    h_current = np.clip(h_current, 0.0, 1.0)
    current_sum = np.sum(h_current)
    h_current = h_current * (target_sum / current_sum)

    best_h = h_current.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 20:
        remaining = budget_s - (time.time() - start_time)
        trials = max(5, int(remaining / 50))
        for _ in range(trials):
            noise_scale = max(0.1, 0.3 * (1 - (time.time() - start_time) / (budget_s - 60)) ** 1.5)
            perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed)
            if current_sum < 1e-10:
                continue
            perturbed = perturbed * (target_sum / current_sum)

            for method in ['L-BFGS-B', 'COBYLA', 'SLSQP']:
                result = minimize(
                    fun=objective,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_full,
                    constraints=cons,
                    options={'maxiter': 50, 'ftol': 1e-10}
                )
                if result.success:
                    new_h = result.x.copy()
                    # Enforce after-optimization feasibility
                    new_h_clipped = np.clip(new_h, 0, 1)
                    new_sum = np.sum(new_h_clipped)
                    if new_sum < 1e-10:
                        continue
                    new_h = new_h_clipped * (target_sum / new_sum)
                    test_c5 = objective(new_h)
                    if test_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = test_c5
                        break  # Skip other methods if improved

    assert abs(np.sum(best_h) - target_sum) < 1e-8
    assert all(0 <= x <= 1 for x in best_h)
    return (best_h.tolist(), best_c5, n_points)
