import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        corr = np.correlate(h_vars, 1 - h_vars, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Scale initial h_full_initial to have correct sum and bounds
    h_full_initial = h_full_initial.copy()
    total_initial = np.sum(h_full_initial)
    if np.isclose(total_initial, 0.0):
        h_full_initial = np.zeros(n_points)
    else:
        h_full_initial = h_full_initial * (target_sum / total_initial)
    h_full_initial = np.clip(h_full_initial, 0, 1)
    best_h_full = h_full_initial.copy()
    best_c5 = objective(best_h_full)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        frac = remaining / budget_s
        # More aggressive noise decay as time runs out
        noise_scale = 0.45 * (frac ** 3)

        perturb_count = 2 if elapsed < 300 else 1
        for _ in range(perturb_count):
            seed += 1
            rnd = np.random.RandomState(seed)
            perturbed = best_h_full + rnd.normal(0, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if not np.isclose(total, 0.0):
                perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            # Alternate optimizers to avoid local optima
            method = ['SLSQP', 'COBYLA', 'L-BFGS-B'][(n_attempt + elapsed)//10 % 3]
            result = minimize(
                fun=objective,
                x0=perturbed,
                method=method,
                bounds=bounds_per_var,
                constraints=cons,
                options={'maxiter': 40, 'ftol': 1e-12, 'disp': False}
            )

            if result.success:
                new_h_full = np.clip(result.x, 0, 1)
                new_total = np.sum(new_h_full)
                # Rescale if sum constraint drifts due to clipping
                if not np.isclose(new_total, target_sum):
                    new_h_full = (new_h_full / new_total) * target_sum
                    new_h_full = np.clip(new_h_full, 0, 1)
                new_c5 = objective(new_h_full)
                if new_c5 < best_c5 - 1e-8:
                    best_h_full = new_h_full
                    best_c5 = new_c5

    # Final repair step
    best_h_full = np.clip(best_h_full, 0, 1)
    total = np.sum(best_h_full)
    if not np.isclose(total, target_sum):
        best_h_full = (best_h_full / total) * target_sum
        best_h_full = np.clip(best_h_full, 0, 1)

    # Recompute final C5 to ensure accuracy
    final_c5 = objective(best_h_full)
    
    return (best_h_full.tolist(), final_c5, n_points)
