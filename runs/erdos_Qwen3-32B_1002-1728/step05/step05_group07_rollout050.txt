import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to meet constraint
    current_sum = np.sum(h_initial)
    if current_sum != 0:
        scale = target_sum / current_sum
        h_initial = h_initial * scale
    h_initial = np.clip(h_initial, 0.0, 1.0)

    def compute_c5(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 70
    max_generations = 800

    # Initialize population with diverse but valid candidates
    population = []
    for _ in range(population_size):
        if _ == 0:
            candidate = h_initial.copy()
        else:
            if np.random.rand() < 0.3:
                # Perturb initial with larger step
                candidate = h_initial + np.random.normal(0, 0.2, n_points)
            else:
                # Random uniform sample
                candidate = np.random.rand(n_points)
                candidate /= np.sum(candidate) * 2.0
                candidate *= n_points / 2.0
                candidate = np.clip(candidate, 0.0, 1.0)
            # Normalize if needed
            current_sum = np.sum(candidate)
            if current_sum != 0:
                candidate = candidate * (target_sum / current_sum)
            candidate = np.clip(candidate, 0.0, 1.0)
        population.append(candidate)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    generation = 0
    while time.time() - start_time < budget_s - 15 and generation < max_generations:

        # Evaluate and sort candidates
        candidates = [(compute_c5(h), h.copy()) for h in population]

            # Differential evolution-style crossover
