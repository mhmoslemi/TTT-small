        n_perturb = max(1, int(perturb_fraction * half_n))
        perturb_indices = np.random.choice(half_n, size=n_perturb, replace=False)
        sigma = 0.05 * (1.0 + (iter_count % 500) / 500.0)  # Increasing sigma over 500 iters
        perturbation = np.random.normal(0, sigma, n_perturb)
        new_h_half = current_h_half.copy()
        new_h_half[perturb_indices] += perturbation
        new_h_half = np.clip(new_h_half, 0, 1)
        new_h_half = normalize_h_half(new_h_half, target_sum_half)

        # Evaluate new candidate
        h_full = create_full_h(new_h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half[:] = new_h_half
            current_h_half[:] = new_h_half
        # Periodically reset from best with larger perturbation
        if iter_count % 100 == 0:
            current_h_half[:] = best_h_half.copy()
            large_sigma = 0.3
            large_perturb_fraction = 0.5
            n_large = int(large_perturb_fraction * half_n)
            large_indices = np.random.choice(half_n, size=n_large, replace=False)
            large_perturbation = np.random.normal(0, large_sigma, n_large)
            current_h_half[large_indices] += large_perturbation
            current_h_half = np.clip(current_h_half, 0, 1)
            current_h_half = normalize_h_half(current_h_half, target_sum_half)
        iter_count += 1

    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
