    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to meet integral constraint
    current_sum = np.sum(h_full_initial)
    h_full = h_full_initial * (target_sum / current_sum)
    h_full = np.clip(h_full, 0, 1.0)

    best_h = h_full.copy()
    best_c5 = np.inf

    # Configuration for gradient descent
    max_outer_steps = 1000
    lr = 0.005
    lr_decay = 0.99
    project_tol = 1e-8

    for step in range(max_outer_steps):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute correlations for positive shifts
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        relevant_corrs = corr[-n_points:]  # Shifts where 1-h is shifted right
        current_c5 = np.max(relevant_corrs)
        k_max = np.argmax(relevant_corrs)

        # Compute gradient for highest contributing shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k_max
            term1 = 1.0 - h_full[i_plus_k] if i_plus_k < n_points else 1.0
            i_minus_k = i - k_max
            term2 = h_full[i_minus_k] if i_minus_k >= 0 else 0.0
            grad[i] = (term1 - term2) * dx

        # Gradient update with learning rate
        h_full = h_full - lr * grad

        # Projection: clip to [0,1]
        h_full = np.clip(h_full, 0.0, 1.0)

        # Renormalize to maintain integral constraint
        current_sum = np.sum(h_full)
        if abs(current_sum) > 1e-10:
            h_full = h_full * (target_sum / current_sum)
        h_full = np.clip(h_full, 0.0, 1.0)

        # Track progress
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full.copy()

        # Adaptive learning rate decay
        lr *= lr_decay

    # Final result validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-8) & (best_h <= 1.0 + 1e-8))
    return (best_h.tolist(), final_c5, n_points)
