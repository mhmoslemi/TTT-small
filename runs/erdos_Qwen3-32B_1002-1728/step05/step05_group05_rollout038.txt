import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    np.random.seed(seed)

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 50
    # Initialize population with perturbations of initial
    population = []
    for _ in range(population_size):
        perturb = np.random.normal(0, 0.1, n_points)
        h = np.clip(h_initial + perturb, 0, 1)
        s = np.sum(h)
        if np.isclose(s, 0):
            h = np.ones(n_points) * 0.5
        else:
            h = h * (target_sum / s)
            h = np.clip(h, 0, 1)
        population.append(h)
    population = np.array(population)

    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)
    num_eval = 0

    while time.time() - start_time < budget_s - 5:
        # Generate next population via evolutionary steps
        new_pop = []
        while len(new_pop) < population_size:
            # Tournament selection
            idxs = np.random.randint(0, population_size, 2)
            p1, p2 = population[idxs[0]], population[idxs[1]]
            parent = p1 if evaluate_c5(p1) < evaluate_c5(p2) else p2

            child = parent.copy()
            # Apply mutation
            mask = np.random.rand(n_points) < 0.1
            noise = np.random.normal(0, 0.2, n_points)
            child[mask] += noise[mask]
            child = np.clip(child, 0, 1)
            s = np.sum(child)
            if not np.isclose(s, 0):
                child = child * (target_sum / s)
                child = np.clip(child, 0, 1)
            new_pop.append(child)
        population = np.array(new_pop[:population_size])

        # Track best solution
        for h in population:
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_h[:] = h
                best_c5 = current_c5

    # Final validation
    assert np.all(best_h >= -1e-6) and np.all(best_h <= 1 + 1e-6)
    assert np.abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
