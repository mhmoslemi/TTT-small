    target_sum = n_points / 2.0

    best_c5 = objective(best_h)

    # Differential Evolution
    bounds_de = [(0.0, 1.0)] * n_points
    np.random.seed(seed)
    try:
        de_result = differential_evolution(
            func=lambda x: objective(x),
            bounds=bounds_de,
            strategy='best1bin',
            maxiter=5,  # Reduced from 10 to 5 due to more variables
            popsize=15,  # Same as before
            tol=1e-4,
            mutation=(0.5, 1),

            workers=1

        if de_result.success:

            current_sum = np.sum(de_h)

                de_h = np.clip(de_h, 0, 1)
            c5_de = objective(de_h)
            if c5_de < best_c5:
