import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def evaluate_objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    rng = np.random.RandomState(seed)

    while time.time() - start_time < budget_s - 30:
        remaining_time = budget_s - (time.time() - start_time)
        frac = remaining_time / budget_s
        noise_scale = 0.25 * (1 - frac**2)

        perturbed = best_h + noise_scale * rng.randn(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)

        if current_sum != 0 and not np.isclose(current_sum, target_sum):
            perturbed = (perturbed / current_sum) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

        methods = ['SLSQP', 'COBYLA']
        for method in methods:
            result = minimize(
                fun=evaluate_objective,
                x0=perturbed,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 30, 'ftol': 1e-12, 'disp': False}
            )
            if result.success:
                optimized_h = result.x
                optimized_c5 = evaluate_objective(optimized_h)
                if optimized_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = optimized_c5
                    break  # Try next perturbation with new best
            elif np.sum(result.x) > 1e-6:  # Non-zero sum
                adjusted = result.x / np.sum(result.x) * target_sum
                adjusted = np.clip(adjusted, 0, 1)
                c5_candidate = evaluate_objective(adjusted)
                if c5_candidate < best_c5:
                    best_h = adjusted
                    best_c5 = c5_candidate
                    break

    # Ensure final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h)) if np.sum(best_h) > 1e-6 else best_h
    final_c5 = evaluate_objective(best_h)

    return (best_h.tolist(), final_c5, n_points)
