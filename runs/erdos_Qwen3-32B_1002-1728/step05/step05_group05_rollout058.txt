import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0  # sum(h_half) must satisfy sum(h_full) = n_points / 2
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize symmetric half from provided construction
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = np.clip(h_half_initial, 0, 1)
    sum_h_half = h_half_initial.sum()
    h_half_initial *= target_sum / sum_h_half

    # PSO parameters
    n_particles = 50
    max_iter = 1000
    c1, c2 = 2.0, 2.0
    w_max, w_min = 0.9, 0.4
    vel_clip = 0.1

    # Initialize particles
    np.random.seed(seed)
    particles = np.zeros((n_particles, half_n))
    velocities = np.zeros((n_particles, half_n))
    pbest_positions = np.zeros((n_particles, half_n))
    pbest_scores = np.full(n_particles, np.inf)
    particles[0] = h_half_initial.copy()  # Use initial as first particle
    for i in range(1, n_particles):
        perturbation = np.random.uniform(-0.05, 0.05, half_n)
        particles[i] = np.clip(h_half_initial + perturbation, 0, 1)
        particles[i] *= target_sum / particles[i].sum()

    # Evaluate initial scores
    pbest_positions[:] = particles
    for i in range(n_particles):
        pbest_scores[i] = evaluate_c5(particles[i])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    # Main PSO loop
    while time.time() - start_time < budget_s - 10:
        # Adaptive inertia weight
        elapsed = time.time() - start_time
        progress_ratio = min(elapsed / budget_s, 1.0)
        w = w_max - progress_ratio * (w_max - w_min)

        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            velocities[i] = np.clip(velocities[i], -vel_clip, vel_clip)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = particles[i].sum()
            if not np.isclose(current_sum, 0):
                particles[i] *= target_sum / current_sum

        # Update best positions
        for i in range(n_particles):
            current_score = evaluate_c5(particles[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = particles[i].copy()
                if current_score < gbest_score:
                    gbest_score = current_score
                    gbest_position = particles[i].copy()

    # Final validation
    best_h_full = create_full_h(gbest_position)
    best_h_full = np.clip(best_h_full, 0, 1)
    best_h_full_sum = best_h_full.sum()
    if not np.allclose(best_h_full_sum, n_points / 2.0, atol=1e-6):
        best_h_full *= (n_points / 2.0) / best_h_full_sum
    final_c5 = evaluate_c5(gbest_position)

    return (best_h_full.tolist(), final_c5, n_points)
