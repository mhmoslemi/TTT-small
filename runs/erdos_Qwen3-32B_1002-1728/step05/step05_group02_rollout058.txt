    # Run differential evolution
    de_result = differential_evolution(

        popsize=10,
        mutation=(0.5, 1),

        polish=True,
        maxiter=20,
        atol=1e-6,
        workers=1
    )

    # Refine with L-BFGS-B
    lbfgs_result = minimize(

        bounds=bounds,

    )

    # Optionally, run some more iterations
    current_time = time.time()
    max_duration = min(budget_s - 30, current_time + 30)  # Allot 30 seconds for more steps
    best_solution = de_result.x
    best_value = objective(de_result.x)
    if lbfgs_result.success:
        new_best = objective(lbfgs_result.x)
        if new_best < best_value:
            best_value = new_best
            best_solution = lbfgs_result.x

    # Try further refinements if time allows
    while time.time() - start_time < budget_s - 60:
        # Perturb the solution slightly and run another optimizer
        perturbed = best_solution + np.random.RandomState(seed + int(time.time()) % 1000).normal(0, 0.1, n_points)

        sum_p = np.sum(perturbed)

        perturbed = perturbed * (target_sum / sum_p)

        result = minimize(objective, perturbed, method=method, bounds=bounds, options={'maxiter': 50})

from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bound_low, bound_high = 0.0, 1.0

    def objective(h_candidate):
        h = np.clip(h_candidate, bound_low, bound_high)
        h_sum = np.sum(h)
        if h_sum < 1e-10:
            return float('inf')
        h = h * (target_sum / h_sum)
        h = np.clip(h, bound_low, bound_high)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(bound_low, bound_high)] * n_points

    # Global search with differential evolution
    de_result = differential_evolution(
        objective, 
        bounds,
        popsize=10,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        maxiter=20,
        atol=1e-6,
        polish=True,
        workers=1
    )

    best_h = de_result.x
    best_c5 = de_result.fun

    remaining_time = budget_s - (time.time() - start_time)
    max_attempts = int(remaining_time // 20) if remaining_time > 30 else 3

    # Local refinement with adaptive methods
    for attempt in range(max_attempts):
        perturbation_scale = 0.1 * (1 - attempt / max_attempts)
        np.random.seed(int(seed) + attempt)
        perturbed = best_h + np.random.normal(0, perturbation_scale, n_points)
        perturbed = np.clip(perturbed, bound_low, bound_high)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum < 1e-10:
            continue
        perturbed = perturbed * (target_sum / perturbed_sum)
        perturbed = np.clip(perturbed, bound_low, bound_high)

        method = ['SLSQP', 'COBYLA', 'L-BFGS-B'][attempt % 3]
        result = minimize(
            objective,
            x0=perturbed,
            method=method,
            bounds=bounds,

        )
        if not result.success:
