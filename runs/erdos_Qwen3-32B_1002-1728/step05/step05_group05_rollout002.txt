def create_full_h(half, half_n, n_points):
    full = np.zeros(n_points)
    for i in range(half_n):
        full[i] = half[i]
        full[n_points - 1 - i] = half[i]

def evaluate_c5(full, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(full, 1 - full, mode='full')
    return np.max(corr) * dx

def perturb(individual, target_sum_half, noise_std=0.1):
    h_perturbed = individual + np.random.normal(0, noise_std, size=individual.shape)
    h_perturbed = np.clip(h_perturbed, 0, 1)
    sum_perturbed = np.sum(h_perturbed)
    if sum_perturbed == 0:
        return individual.copy()
    scale_factor = target_sum_half / sum_perturbed
    h_scaled = h_perturbed * scale_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
