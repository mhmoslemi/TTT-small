import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    def project(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return np.zeros_like(h)
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    np.random.seed(seed)
    population_size = 20
    n_iter_pso = 20
    positions = []
    for _ in range(population_size):
        pert = np.random.randn(n_points) * 0.1
        pos = project(h_initial + pert)
        positions.append(pos)
    velocities = [np.random.randn(n_points) * 0.05 for _ in range(population_size)]
    pbest_positions = [p.copy() for p in positions]
    pbest_values = [compute_c5(p) for p in positions]
    gbest_value, gbest_position = min((v, p) for v, p in zip(pbest_values, pbest_positions))

    w_inertia = 0.8
    c_cognitive = 1.2
    c_social = 1.2

    for iter in range(n_iter_pso):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(population_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w_inertia * velocities[i] +
                            c_cognitive * r1 * (pbest_positions[i] - positions[i]) +
                            c_social * r2 * (gbest_position - positions[i]))
            positions[i] = positions[i] + velocities[i]
            positions[i] = project(positions[i])
            current_value = compute_c5(positions[i])
            if current_value < pbest_values[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_values[i] = current_value
            if current_value < gbest_value:
                gbest_position = positions[i].copy()
                gbest_value = current_value

    values_positions = list(zip(pbest_values, pbest_positions))
    values_positions.sort(key=lambda x: x[0])
    top_candidates = [p for v, p in values_positions[:5]]

    best_h = gbest_position
    best_c5 = gbest_value

    for candidate in top_candidates:
        if time.time() - start_time > budget_s - 10:
            break
        result = minimize(
            fun=compute_c5,
            x0=candidate,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': 150, 'ftol': 1e-8}
        )
        if result.success:
            refined_h = result.x
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5

    final_h = project(best_h)
    return (final_h.tolist(), best_c5, n_points)
