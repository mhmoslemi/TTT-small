        for h in population:
            fits.append(evaluate_c5(h, n_points, dx))
        idx = np.argmin(fits)
        current_best = fits[idx]
        if current_best < best_c5:
            best_c5 = current_best
            best_h_half = population[idx].copy()
        
        offspring = []
        for _ in range(population_size // 2):
            p1 = tournament_select(population, fits)
            p2 = tournament_select(population, fits)
            beta = np.random.uniform(0.3, 0.7)
            child = beta * p1 + (1 - beta) * p2
            child += np.random.normal(0, 0.01, half_n)
            child = np.clip(child, 0, 1)
            s = child.sum()
            if s > 1e-6:
                child *= target_sum_half / s
            child = np.clip(child, 0, 1)
            offspring.append(child)
        
        combined = population + offspring
        combined_fits = [evaluate_c5(c, n_points, dx) for c in combined]
        argsort = np.argsort(combined_fits)
        population = [combined[i] for i in argsort[:population_size]]
    
    best_h_full = create_full_h(best_h_half, n_points)
    best_h_full = np.clip(best_h_full, 0, 1)
    total = best_h_full.sum()
    if not np.isclose(total, n_points/2.0):
        best_h_full *= (n_points/2.0) / total
        best_h_full = np.clip(best_h_full, 0, 1)
    final_c5 = evaluate_c5(best_h_half, n_points, dx)
    return (best_h_full.tolist(), final_c5, n_points)
