    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def project_to_feasible(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.zeros(n_points)
        scaling_factor = target_sum / sum_clipped
        h_scaled = h_clipped * scaling_factor
        h_projected = np.clip(h_scaled, 0, 1)
        sum_projected = np.sum(h_projected)
        if abs(sum_projected - target_sum) > 1e-6:
            # Need to adjust again
            delta = target_sum - sum_projected
            # Distribute delta equally
            # Since h_projected is already in bounds, add a small value to each element
            adjustment = delta / n_points
            h_projected += adjustment
            h_projected = np.clip(h_projected, 0, 1)
        return h_projected

    # Ensure the initial h is feasible
    h_initial_feasible = project_to_feasible(h_full_initial)

    # Initialize PSO parameters
    num_particles = 20
    w = 0.8
    c1 = 1.2
    c2 = 1.2
    max_pso_iterations = 20

    # Initialize particles
    np.random.seed(seed)
    particles = []
    velocities = []

        perturbation = np.random.normal(0, 0.1, n_points)

    # Initialize pbest and gbest

    pbest_scores = [compute_c5(p) for p in particles]
    gbest_index = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_index].copy()
    gbest_score = pbest_scores[gbest_index]
    for t in range(max_pso_iterations):
        if time.time() - start_time > budget_s - 10:
            break

    x0 = gbest_position.copy()

    return (gbest_position.tolist(), gbest_score, n_points)
