def project(candidate, target_sum):
    h_clipped = np.clip(candidate, 0, 1)
    total = np.sum(h_clipped)
    if total > 1e-10:
        h_scaled = h_clipped * (target_sum / total)
        return np.clip(h_scaled, 0, 1)
    else:
        return h_clipped

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Particle Swarm Optimization parameters
    n_particles = 20
    max_iter_pso = 200
    c1, c2, w = 0.5, 0.5, 0.9

    # Initialize positions and velocities
    np.random.seed(seed)
    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    personal_best_scores = np.full(n_particles, np.inf)
    global_best_score = np.inf
    global_best_position = None

    # Initialize particles around initial_h_values and random samples
    for i in range(n_particles):
        if np.random.rand() < 0.7:  # 70% start near initial_h_values
            pos = h_initial.copy()
            pos += np.random.normal(0, 0.05, n_points)
        else:
            pos = np.random.rand(n_points)
        pos = project(pos, target_sum)
        positions[i] = pos
        score = compute_c5(pos, n_points)
        if score < personal_best_scores[i]:
            personal_best_scores[i] = score
        if score < global_best_score:
            global_best_score = score
            global_best_position = pos.copy()

    # PSO Main Loop
    for _ in range(max_iter_pso):
        if time.time() - start_time >= budget_s - 30:
            break

        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (positions[i] - positions[i]) +  # Personal best to current (used for velocity dynamics)
                c2 * r2 * (global_best_position - positions[i])
            )
            new_pos = positions[i] + velocities[i]
            new_pos = project(new_pos, target_sum)
            positions[i] = new_pos
            score = compute_c5(new_pos, n_points)

            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best_score = score
                    global_best_position = new_pos.copy()

        w *= 0.995  # Inertia weight decay

    # Local refinement with SQP on top candidates
    top_k = max(1, int(0.2 * n_particles))
    top_indices = np.argsort(personal_best_scores)[:top_k]
    for idx in top_indices:
        pos = positions[idx]
        bounds = [(0.0, 1.0)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(
            lambda x: compute_c5(x, n_points),
            x0=pos,
            bounds=bounds,
            constraints=cons,
            method='SLSQP',
            options={'maxiter': 50, 'ftol': 1e-12}
        )
        if result.success and result.fun < global_best_score:
            global_best_score = result.fun
            global_best_position = result.x

    # Final projection and validation
    h_full = project(global_best_position, target_sum)
    h_full = h_full.tolist()

    assert abs(np.sum(h_full) - target_sum) < 1e-6
    assert all(0 <= v <= 1 for v in h_full)

    return (h_full, global_best_score, n_points)
