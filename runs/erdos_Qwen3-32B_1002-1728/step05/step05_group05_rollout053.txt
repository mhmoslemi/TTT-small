import random

def run(seed=42, budget_s=1100, **kwargs):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def generate_perturb(h, mutation_strength=0.1):
        h_pert = h.copy()
        noise = np.random.normal(0, mutation_strength, n_points)
        h_pert += noise
        h_pert = np.clip(h_pert, 0, 1)
        s = np.sum(h_pert)
        if abs(s) < 1e-10:
            return h_pert
        scale = target_sum / s
        h_pert *= scale
        h_pert = np.clip(h_pert, 0, 1)
        return h_pert

    def crossover(a, b):
        alpha = np.random.rand()
        return alpha * a + (1 - alpha) * b

    population_size = 50
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        population.append(generate_perturb(population[0]))

    best_h = h_initial.copy()

    random.seed(seed)

    while time.time() - start_time < budget_s - 5:

                scale = target_sum / p_sum

                best_h = p_clipped.copy()
