import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    assert n_points == 70, "Assumed even n_points for symmetry handling"
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = np.sum(h_copy)
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        return np.clip(h_scaled2, 0, 1)

    def compute_gradient(h_full, k):
        n = h_full.size
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= (i + k) < n:
                grad[i] += 1.0 - h_full[i + k]
            if 0 <= (i - k) < n:
                grad[i] -= h_full[i - k]
        return grad

    np.random.seed(seed)
    h_half = normalize_h_half(h_full_initial[:half_n].copy(), target_sum_half)
    best_c5 = float('inf')
    best_h_half = h_half.copy()
    no_improve_steps = 0
    learning_rate = 0.1
    lrate_decay = 0.95
    max_no_improve = 50
    perturb_sigma = 0.05

    while time.time() - start_time < budget_s - 5:
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        lags = np.arange(len(corr)) - (n_points - 1)
        m = np.argmax(corr)
        k_max = lags[m]
        grad_full = compute_gradient(h_full, k_max)
        grad_half = np.array([grad_full[i] + grad_full[n_points - 1 - i] for i in range(half_n)])
        h_half -= learning_rate * grad_half
        h_half = normalize_h_half(h_half, target_sum_half)

        h_full = create_full_h(h_half)
        current_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full') * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()
            no_improve_steps = 0
        else:
            no_improve_steps += 1

        if no_improve_steps % 50 == 0:
            learning_rate *= lrate_decay
        if no_improve_steps >= max_no_improve:
            h_half = normalize_h_half(h_half + np.random.normal(0, perturb_sigma, half_n), target_sum_half)
            no_improve_steps = 0

    h_full_best = create_full_h(best_h_half)
    corr_best = np.correlate(h_full_best, 1 - h_full_best, mode='full') * dx
    final_c5 = np.max(corr_best)
    assert abs(np.sum(h_full_best) - n_points / 2.0) < 1e-6
    return (h_full_best.tolist(), final_c5, n_points)
