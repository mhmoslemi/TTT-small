import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    half_n = n_points // 2

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Enforce symmetry and normalization
    h_full = h_initial.copy()
    for i in range(half_n):
        avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
        h_full[i] = h_full[n_points - 1 - i] = avg
    current_sum = np.sum(h_full)
    if current_sum != 0:
        h_full = (h_full / current_sum) * target_sum
    h_full = np.clip(h_full, 0.0, 1.0)
    h_half_current = h_full[:half_n]
    current_c5 = compute_c5(h_full)
    best_c5 = current_c5
    best_h_half = h_half_current.copy()

    T = 1.0
    T_min = 1e-6
    alpha = 0.9995  # Faster cooling
    target_sum_half = target_sum / 2.0

    np.random.seed(seed)

    while time.time() - start_time < budget_s and T > T_min:
        # Perturbation with adaptive magnitude
        perturbation = np.random.normal(0, T * 0.1, size=half_n)
        new_h_half = h_half_current + perturbation
        new_h_half = np.clip(new_h_half, 0.0, 1.0)
        total = np.sum(new_h_half)

            scale = target_sum_half / total
