import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0

    def normalize_h(h_vec):
        h_clipped = np.clip(h_vec, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            return h_clipped
        scale = T / total
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    population_size = 30
    tournament_size = 5
    population = [normalize_h(h_initial + np.random.normal(0, 0.05, n_points)) for _ in range(population_size)]

    best_h = normalize_h(h_initial)
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s:

        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        idx_sorted = np.argsort(fitness)
        for idx in idx_sorted:
            h_candidate = population[idx]
            c5_candidate = fitness[idx]
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate

        # Tournament selection
        parents = []
        for _ in range(population_size):
            candidates = np.random.choice(population, tournament_size, replace=False)
            fit = [compute_c5(c) for c in candidates]
            winner_idx = np.argmin(fit)
            parents.append(candidates[winner_idx])

        # Crossover
