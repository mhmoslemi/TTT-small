        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        worst_index = np.argmax(corr)

        # Objective focused on reducing the worst lag
        def objective_worst(h):
            return (np.correlate(h, 1 - h, mode='full') * dx)[worst_index]

        # Local optimization with SLSQP
        result = minimize(
            fun=objective_worst,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-4}
        )

        if result.success:
            new_h = np.clip(result.x, 0, 1)
            if abs(np.sum(new_h) - target_sum) < 1e-6:
                new_c5 = np.max(np.correlate(new_h, 1 - new_h, mode='full') * dx)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

        # Perturb and escape local optima
        if time.time() - start_time < budget_s - 90:
            perturbed = best_h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total_perturbed = np.sum(perturbed)
            if total_perturbed != 0:
                perturbed = perturbed * (target_sum / total_perturbed)
            perturbed = np.clip(perturbed, 0, 1)
            result_perturb = minimize(
                fun=objective_worst,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result_perturb.success:
                alt_h = np.clip(result_perturb.x, 0, 1)
                alt_c5 = np.max(np.correlate(alt_h, 1 - alt_h, mode='full') * dx)
                if alt_c5 < best_c5:
                    best_h = alt_h
                    best_c5 = alt_c5

    return (best_h.tolist(), best_c5, n_points)
