            current_total = np.sum(new_h)
            if current_total != 0:
                new_h = (new_h / current_total) * (target_sum / 2)
                new_h = np.clip(new_h, 0.0, 1.0)
        new_full = normalize_and_clip(create_full_h(new_h))
        new_c5 = compute_c5(new_full)
        delta = new_c5 - c5_best
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            if new_c5 < c5_best:
                c5_best = new_c5
                h_half_best = new_h.copy()
        T *= alpha

    h_full = normalize_and_clip(create_full_h(h_half_best))
    assert np.allclose(np.sum(h_full), target_sum, atol=1e-6)
    assert np.all(h_full >= -1e-6) and np.all(h_full <= 1.0 + 1e-6)

    return (h_full.tolist(), c5_best, n_points)
