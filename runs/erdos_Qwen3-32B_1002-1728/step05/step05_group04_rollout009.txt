import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_vector):
        h = np.array(h_vector)
        total = np.sum(h)
        if abs(total) < 1e-10:
            return 1e6  # Invalid configuration
        h_scaled = h * (target_sum / total)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    np.random.seed(seed)
    
    maxiter = 50
    popsize = 10  # Reduced population size to save time
    result = differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=popsize,
        mutation=(0.4, 1.0),
        recombination=0.8,
        seed=seed,
        polish=False,
        init='sobol',
        atol=1e-6,
        disp=False,
        callback=None
    )

    elapsed = time.time() - start_time

            best_h = best_h * (target_sum / total)
            best_h = np.clip(best_h, 0, 1)
            best_c5 = objective(best_h)
        else:
            best_h, best_c5 = h_initial, objective(h_initial)
