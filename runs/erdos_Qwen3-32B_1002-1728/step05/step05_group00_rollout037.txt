import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    T = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    # PSO parameters
    n_particles = 20
    n_iterations = 20
    w = 0.8
    c1, c2 = 1.2, 1.2
    np.random.seed(seed)

    # Initialize particles
    particles = []
    pbests = []
    for _ in range(n_particles):
        h = h_initial.copy()
        noise = (np.random.rand(n_points) - 0.5) * 0.5  # Symmetric noise around 0
        h += noise
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = (h / current_sum) * T
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        pbests.append(h.copy())

    velocities = [np.random.rand(n_points) * 0.1 for _ in range(n_particles)]
    pbest_c5 = [compute_c5(p) for p in pbests]
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbests[gbest_idx].copy()
    best_c5 = min(pbest_c5)

    # PSO loop
    for iter in range(n_iterations):
        if time.time() - start_time >= budget_s - 30:
            break
        for i in range(n_particles):
            # Update velocity and position
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] 
                            + c1 * r1 * (pbests[i] - particles[i])
                            + c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            if current_sum != 0:
                particles[i] = (particles[i] / current_sum) * T
            particles[i] = np.clip(particles[i], 0, 1)
            # Update pbest and gbest
            current_c5 = compute_c5(particles[i])
            if current_c5 < pbest_c5[i]:
                pbests[i] = particles[i].copy()
                pbest_c5[i] = current_c5
        gbest_idx = np.argmin(pbest_c5)
        gbest = pbests[gbest_idx].copy()
        if pbest_c5[gbest_idx] < best_c5:
            best_c5 = pbest_c5[gbest_idx]

    # Local refinement
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(
        fun=lambda x: compute_c5(x),
        x0=gbest,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - T},
        options={'maxiter': 100, 'ftol': 1e-8}

    if result.success:

        final_c5 = compute_c5(candidate)
        if final_c5 < best_c5:
            best_c5 = final_c5
            gbest = candidate.copy()

    return (np.clip(gbest, 0, 1).tolist(), best_c5, n_points)
