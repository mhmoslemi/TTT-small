        x_i = i * (2.0 / n_points)

        idx = np.searchsorted(x_control, x_i) - 1

np.random.seed(42)

def evaluate_bezier(control_points, n_points):
    m = len(control_points)
    x_control = np.linspace(0.0, 2.0, m)
    h_samples = np.zeros(n_points)
    for i in range(n_points):
        x = 2.0 * i / n_points
        idx = np.searchsorted(x_control, x) - 1
        if idx < 0:
            h_samples[i] = control_points[0]
        elif idx >= m - 1:
            h_samples[i] = control_points[-1]
        else:
            x0, x1 = x_control[idx], x_control[idx+1]
            t = (x - x0) / (x1 - x0)
            h_samples[i] = (1.0 - t) * control_points[idx] + t * control_points[idx+1]
    return h_samples

def normalize_h(h_samples):
    target_sum = len(h_samples) / 2.0
    current_sum = h_samples.sum()
    if current_sum == 0:
        return h_samples
    scale = target_sum / current_sum
    h_norm = np.clip(h_samples * scale, 0.0, 1.0)
    return h_norm

def compute_c5(h_samples):
    dx = 2.0 / len(h_samples)
    corr = np.correlate(h_samples, 1.0 - h_samples, mode='full')
    return np.max(corr) * dx

def objective(ctl, n_points):
    h_samples = evaluate_bezier(ctl, n_points)
    h_samples = normalize_h(h_samples)
