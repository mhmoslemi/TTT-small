def mutate(indiv, mutation_rate):
    copy = indiv.copy()
    n = len(copy)
    num_swaps = int(n * mutation_rate)
    for _ in range(num_swaps):
        ones = np.where(copy == 1.0)[0]
        zeros = np.where(copy == 0.0)[0]

        copy[idx_1], copy[idx_0] = copy[idx_0], copy[idx_1]
    return copy

def crossover(parent1, parent2, target_sum):
    # Single-point crossover with adjustment
    n = len(parent1)
    split = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:split], parent2[split:]])
    # Adjust the sum of 1s
    current_sum = np.sum(child1)
    if current_sum != target_sum:
        delta = target_sum - current_sum
        if delta > 0:
            # Need to add delta 1s
            zero_positions = np.where(child1 == 0.0)[0]
            np.random.shuffle(zero_positions)
            add_count = 0

                add_count += 1

            remove_count = 0
            for i in one_positions[:abs(delta)]:

    return child1

import time
import random

def initial_binary_from_h(h_values, n_points, target_sum):
    indices = np.argsort(h_values)[::-1]
    binary = np.zeros(n_points, dtype=float)
    binary[indices[:target_sum]] = 1.0
    return binary

def mutate(indiv, mutation_rate):
    copy = indiv.copy()
    n = len(copy)
    num_swaps = int(n * mutation_rate)
    for _ in range(num_swaps):
        ones = np.where(copy == 1.0)[0]
        zeros = np.where(copy == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        idx_1 = np.random.choice(ones)
        idx_0 = np.random.choice(zeros)
        copy[idx_1], copy[idx_0] = copy[idx_0], copy[idx_1]
    return copy

def crossover(parent1, parent2, target_sum):
    n = len(parent1)
    split = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:split], parent2[split:]])
    current_sum = np.sum(child1)
    if current_sum != target_sum:
        delta = target_sum - current_sum
        if delta > 0:
            zero_positions = np.where(child1 == 0.0)[0]
            np.random.shuffle(zero_positions)
            add_count = 0
            for i in zero_positions[:delta]:
                child1[i] = 1.0
                add_count += 1
                if add_count == delta:
                    break
        else:
            one_positions = np.where(child1 == 1.0)[0]
            np.random.shuffle(one_positions)
            remove_count = 0
            for i in one_positions[:abs(delta)]:
                child1[i] = 0.0
                remove_count += 1
                if remove_count == abs(delta):
                    break
