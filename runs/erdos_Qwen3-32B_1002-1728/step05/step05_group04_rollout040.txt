        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Selection (tournament with top half)
        sorted_indices = np.argsort(fitness)
        top_half = [population[i] for i in sorted_indices[:pop_size // 2]]

        # Generate new population
        new_population = [best_h.copy()]  # Elitism
        while len(new_population) < pop_size:
            # Select two parents
            p1 = top_half[np.random.randint(len(top_half))]
            p2 = top_half[np.random.randint(len(top_half))]

            # Crossover: swap differing positions
            child = p1.copy()
            diff = np.where(p1 != p2)[0]
            if len(diff) > 0:
                n_swap = min(2, len(diff))
                swap_idx = np.random.choice(diff, n_swap, replace=False)
                for i in swap_idx:
                    child[i] = p2[i]

            # Mutation
            if np.random.rand() < mutation_rate:
                ones = np.where(child == 1)[0]
                zeros = np.where(child == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    one_idx = np.random.choice(ones)
                    zero_idx = np.random.choice(zeros)
                    child[one_idx], child[zero_idx] = 0, 1

            new_population.append(child.copy())

        population = new_population

    return (best_h.tolist(), best_c5, n_points)
