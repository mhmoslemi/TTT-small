import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        # Slightly slower decay in noise scale to encourage exploration
        noise_scale = 0.3 * (1 - (frac ** 0.5))
        
        for _ in range(5):  # More perturbations per iteration
            perturbed = best_h_half + np.random.uniform(-noise_scale, noise_scale, half_n)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if abs(total) < 1e-10:
                continue  # Skip if sum is zero
            perturbed = (perturbed / total) * target_sum_half
            perturbed = np.clip(perturbed, 0, 1)

            # Try multiple methods, but prioritize faster ones
            for method in ['COBYLA', 'SLSQP']:
                result = minimize(
                    fun=objective,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons,
                    options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
                )
                if result.success and result.fun < best_c5:
                    new_h_half = np.clip(result.x, 0, 1)
                    h_full = create_full_h(new_h_half)
                    corr = np.correlate(h_full, 1 - h_full, mode='full')
                    new_c5 = np.max(corr) * dx
                    # Safety: enforce constraints
                    if abs(np.sum(h_full) - n_points / 2.0) < 1e-6 and new_c5 < best_c5:
                        best_h_half = new_h_half
                        best_c5 = new_c5
                        break  # Break if improved

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), best_c5, n_points)
