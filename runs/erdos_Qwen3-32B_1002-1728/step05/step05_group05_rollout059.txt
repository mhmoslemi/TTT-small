def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_initial[i] = h_full_initial[j] = avg
    current_sum = np.sum(h_full_initial)
    if not np.isclose(current_sum, n_points / 2.0):
        scale = (n_points / 2.0) / current_sum
        h_full_initial *= scale
    h_full_initial = np.clip(h_full_initial, 0, 1)
    initial_h_half = h_full_initial[:half_n]

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    num_particles = 50
    positions = np.zeros((num_particles, half_n))
