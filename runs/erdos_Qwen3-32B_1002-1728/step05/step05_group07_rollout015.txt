import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    x = np.linspace(0, 2.0 - dx, n_points)
    K = 15  # Number of hat functions

    def generate_h(params):
        c_list = params[::3]
        w_list = params[1::3]
        a_list = params[2::3]
        area = np.sum(a_list * w_list)
        if area == 0:
            return np.zeros(n_points)
        a_list = a_list / area
        h = np.zeros(n_points)
        for j in range(n_points):
            xj = x[j]
            total = 0.0
            for i in range(K):
                c = c_list[i]
                w = w_list[i]
                a = a_list[i]
                if abs(xj - c) <= w / 2:
                    t = abs((xj - c) / w)
                    total += a * (1 - 2 * t)
            h[j] = total
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return h
        scale = target_sum / current_sum
        h *= scale
        return np.clip(h, 0, 1)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 50
    population = []

    def create_initial_individual():
        K_initial = min(20, n_points)
        indices = np.argsort(h_full_initial)[::-1][:K_initial]
        selected_x = x[indices[:K_initial]]

        w_list = np.full(K, max(0.1, 1.5 * dx))

        a_list = a_list[:K] if len(a_list) >= K else np.pad(a_list, (0, K - len(a_list)))
        area = np.sum(a_list * w_list)

            a_list /= area

    population.append(create_initial_individual())
    for _ in range(population_size - 1):

    generation = 0
