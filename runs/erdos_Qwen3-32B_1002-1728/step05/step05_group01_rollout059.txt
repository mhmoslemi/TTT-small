import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    dec_result = differential_evolution(
        objective_func, bounds, strategy='best1bin',
