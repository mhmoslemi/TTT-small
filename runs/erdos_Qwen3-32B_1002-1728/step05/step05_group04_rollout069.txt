import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate_individual(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize_h(h):
        total = np.sum(h)
        if total == 0:
            h = np.zeros_like(h)
            h[:int(n_points/2)] = 1.0
            return h
        h = h * (target_sum / total)
        h = np.clip(h, 0, 1)
        return h

    def crossover(p1, p2):
        alpha = np.random.rand()
        child = alpha * p1 + (1 - alpha) * p2
        return child

    def mutate(individual, noise_scale):
        mutations = np.random.normal(0, noise_scale, individual.shape)
        mutated = individual + mutations
        return normalize_h(mutated)

    population_size = 100
    population = []
    for _ in range(population_size):
        h = h_initial.copy()
        h += np.random.normal(0, 0.05, h.shape)
        population.append(normalize_h(h))

    best_c5 = np.inf
    best_h = None

    while time.time() - start_time < budget_s - 60:
        fitness = [evaluate_individual(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        top_indices = np.argsort(fitness)[:int(population_size * 0.2)]
        new_population = []
        while len(new_population) < population_size:
            p1_idx, p2_idx = np.random.choice(top_indices, 2)
            p1, p2 = population[p1_idx], population[p2_idx]
            child = crossover(p1, p2)
            child = mutate(child, noise_scale=0.05)
            new_population.append(child)
        population = new_population

    def objective(x):
        x = normalize_h(x)
        return evaluate_individual(x)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    if best_h is None:
        best_h = h_initial.copy()
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success:
        h_opt = np.clip(result.x, 0, 1)
        h_opt = normalize_h(h_opt)
        final_c5 = evaluate_individual(h_opt)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_opt
    return (best_h.tolist(), best_c5, n_points)
