            m_minus_k = m - k_max
            if 0 <= m_minus_k < n_points:
                term2 = -h_full[m_minus_k]
            grad[m] = term1 + term2

        # Gradient step
        h_full -= learning_rate * grad

        # Enforce symmetry
        for i in range(half_n):
            avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
            h_full[i] = avg
            h_full[n_points - 1 - i] = avg

        # Normalize
        current_sum = h_full.sum()
        if current_sum != 0:
            scaling = target_sum / current_sum
            h_full *= scaling
            # Clip to [0, 1]
            h_full = np.clip(h_full, 0.0, 1.0)
        else:
            # This should not happen if h was properly initialized.
            h_full = best_h.copy()

        # Compute current C5
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        current_c5 = np.max(corr) * dx

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full.copy()

        # Decay learning rate
        learning_rate *= lr_decay

    return (best_h.tolist(), best_c5, n_points)
