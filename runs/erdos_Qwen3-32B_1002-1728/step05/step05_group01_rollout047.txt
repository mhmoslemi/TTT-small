import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # ∫h(x)dx = 1 ⇒ sum(h) = n_points/2

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    np.random.seed(seed)

    H = np.fft.fft(h)
    H = H.real + 1j * H.imag  # Ensure it's complex

    noise_scale = 0.1
    anneal_rate = 0.999
    loop_count = 0

    while time.time() - start_time < budget_s:
        loop_count += 1
        if loop_count % 100 == 0:
            noise_scale *= anneal_rate  # Reduce noise over time

        # Generate perturbation in frequency domain
        rand_real = 2 * np.random.rand(n_points) - 1
        rand_imag = 2 * np.random.rand(n_points) - 1
        perturbation = noise_scale * (rand_real + 1j * rand_imag)

        H_perturbed = H + perturbation

        # Inverse FFT to obtain real function
        h_candidate = np.fft.ifft(H_perturbed).real
        h_clipped = np.clip(h_candidate, 0, 1)
        total = np.sum(h_clipped)

        if abs(total) > 1e-8:
            h_candidate = h_clipped * (target_sum / total)
        else:
            h_candidate = h_candidate.copy()

        if not all(0 <= x <= 1 for x in h_candidate):
            continue  # Skip invalid due to numerical errors

        if abs(np.sum(h_candidate) - target_sum) > 1e-6:
            continue

        c5 = compute_c5(h_candidate, dx)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            H = np.fft.fft(best_h)  # Update Fourier components for best solution

        # Early stop check
        if time.time() - start_time > budget_s - 10:
            break

    assert abs(np.sum(best_h)) < 1e-6 + abs(target_sum - 1e-6)
    return (best_h.tolist(), best_c5, n_points)
