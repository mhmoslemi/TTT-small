        return max_overlap, worst_k

    # Initialize symmetric h_full from initial values
    h_full = h_initial.copy()
    for i in range(half_n):
        avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
        h_full[i] = h_full[n_points - 1 - i] = avg
    # Normalize to meet sum constraint
    current_sum = np.sum(h_full)
    if current_sum != 0:
        h_full = (h_full / current_sum) * target_sum
    h_full = np.clip(h_full, 0.0, 1.0)
    h_half_current = h_full[:half_n]
    current_c5, current_worst_k = find_worst_k_and_overlap(h_full)
    best_c5 = current_c5
    best_h_half = h_half_current.copy()

    T = 1.0
    T_min = 1e-5
    alpha = 0.999  # Faster cooling
    target_sum_half = target_sum / 2.0

    np.random.seed(seed)

    while time.time() - start_time < budget_s and T > T_min:
        # Evaluate current worst shift
        h_full_current = create_full_h(h_half_current)
        current_c5, current_worst_k = find_worst_k_and_overlap(h_full_current)
        worst_k = current_worst_k

        # Generate focused perturbation based on worst_k
        mask = np.ones(half_n)
        for i in range(half_n):
            if h_half_current[i] < 0.01 or h_half_current[i] > 0.99:
                mask[i] = 1.5  # Higher perturbation near edges of domain
            if i < worst_k and i < half_n:
                mask[i] = 2.0  # Boost perturbations for relevant indices
            if (n_points - 1 - i) < worst_k:
                mask[i] = 2.0  # Same for symmetric counterparts

        perturbation = np.random.normal(0, T * mask, size=half_n)
        new_h_half = h_half_current + perturbation
        new_h_half = np.clip(new_h_half, 0.0, 1.0)
        total = np.sum(new_h_half)
        if total != 0:
            scale = target_sum_half / total
            new_h_half *= scale
            new_h_half = np.clip(new_h_half, 0.0, 1.0)
        else:
            new_h_half = h_half_current.copy()

        # Compute new h_full and C5
        new_h_full = create_full_h(new_h_half)
        new_c5, _ = find_worst_k_and_overlap(new_h_full)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_half_current = new_h_half.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h_half = h_half_current.copy()
                best_c5 = current_c5
        T *= alpha

    h_full = create_full_h(best_h_half)
    return (h_full.tolist(), best_c5, n_points)
