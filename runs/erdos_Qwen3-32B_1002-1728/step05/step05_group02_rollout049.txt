import numpy as np
import time
from scipy.optimize import minimize

def run():
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def evaluate_individual(individual):
        h = np.array(individual)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros_like(h)
        h = h * (target_sum / h_sum)
        h = np.clip(h, 0, 1)
        return h

    pop_size = 30
    tournament_size = 5
    mutation_rate = 0.1
    crossover_ratio = 0.7

    # Initialize population
    population = [np.zeros(n_points) for _ in range(pop_size)]
    population[0] = h_initial.copy()
    for i in range(1, pop_size):
        population[i] = np.random.uniform(0, 1, n_points)

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    while time.time() - start_time < 1050:
        valid_population = [evaluate_individual(ind) for ind in population]
        fit = [compute_c5(h) for h in valid_population]

        # Update best found
        current_idx = np.argmin(fit)
        if fit[current_idx] < best_c5:
            best_c5 = fit[current_idx]
            best_h = valid_population[current_idx].copy()

        # Selection via tournament
        selected = []
        for _ in range(pop_size):
            candidates = np.random.choice(pop_size, tournament_size)
            candidate_fitness = [fit[i] for i in candidates]
            winner_idx = np.argmin(candidate_fitness)
            winner = valid_population[candidates[winner_idx]]
            selected.append(winner.copy())

        # Crossover and mutation
        new_population = []
        for _ in range(pop_size):
            if np.random.rand() < crossover_ratio:
                p1 = selected[np.random.randint(len(selected))]
                p2 = selected[np.random.randint(len(selected))]
                child = (p1 + p2) / 2.0
            else:
                p = selected[np.random.randint(len(selected))]
                child = p.copy()
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
                h_sum = np.sum(child)
                if h_sum != 0:
                    child = child * (target_sum / h_sum)
                child = np.clip(child, 0, 1)
                h_sum_clipped = np.sum(child)
                if h_sum_clipped != 0:
                    child = child * (target_sum / h_sum_clipped)
                child = np.clip(child, 0, 1)
            new_population.append(child)
        population = new_population

    # Local optimization
    def objective(x):
        return compute_c5(x)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    res = minimize(
        objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-12}
    )
    if res.success:
        optimized_h = res.x
        optimized_c5 = compute_c5(optimized_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    best_h = evaluate_individual(best_h)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
