import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros_like(h_full_initial)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def normalize_h(h_half, target_sum):
        h = h_half.copy()
        current_sum = np.sum(h)
        if current_sum != 0:
            h *= target_sum / current_sum
        h = np.clip(h, 0, 1)
        adjusted_sum = np.sum(h)
        if adjusted_sum != 0 and not np.isclose(adjusted_sum, target_sum):
            h *= target_sum / adjusted_sum
            h = np.clip(h, 0, 1)
        return h

    np.random.seed(seed)
    initial_h_half = h_full_initial[:half_n].copy()
    particles = [initial_h_half + np.random.normal(0, 0.05, half_n) for _ in range(50)]
    velocities = [np.zeros(half_n) for _ in range(50)]
    for i in range(50):
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = normalize_h(particles[i], target_sum_half)

    personal_best_positions = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * 50
    global_best_position = particles[0].copy()
    global_best_score = float('inf')

    for i in range(50):
        h = create_full_h(particles[i])
        corr = np.correlate(h, 1 - h, mode='full')
        score = np.max(corr) * dx
        personal_best_scores[i] = score
