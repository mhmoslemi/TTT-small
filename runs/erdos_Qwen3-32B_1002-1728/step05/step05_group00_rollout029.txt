    initial_h_half = h_full_initial[:half_n]
    population_size = 20
    population = [project_to_feasible(initial_h_half + np.random.normal(0, 0.1, half_n)) for _ in range(population_size)]
    population[0] = project_to_feasible(initial_h_half)

    max_generations = 30
    mutation_rate = 0.15
    mutation_decay = 0.95
    early_stop_patience = 10
    generation_without_improvement = 0
    best_c5 = float('inf')
    best_h = initial_h_half.copy()
    elite_percent = 0.05
    num_elite = max(1, int(elite_percent * population_size))
    tournament_size = 3

    def tournament_selection(population, fitness):
        indices = np.random.choice(len(population), tournament_size, replace=False)
        best_idx = indices[np.argmin([fitness[i] for i in indices])]
        return population[best_idx]

    def crossover(parent1, parent2):
        child = np.where(np.random.rand(len(parent1)) < 0.5, parent1, parent2)
        return child

    def mutate(individual, mutation_rate):
        indices = np.random.choice(len(individual), int(mutation_rate * len(individual)), replace=False)
        mutated = individual.copy()
        mutated[indices] += np.random.normal(0, 0.1, len(indices))
        return np.clip(mutated, 0, 1)

    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 30:
            break

        fitness = [evaluate_individual(ind) for ind in population]
        current_best_index = np.argmin(fitness)
        current_best_c5_gen = fitness[current_best_index]

        if current_best_c5_gen < best_c5:
            best_c5 = current_best_c5_gen
            best_h = population[current_best_index].copy()
            generation_without_improvement = 0
        else:
            generation_without_improvement += 1

        if generation_without_improvement >= early_stop_patience:
            break

        parents = [tournament_selection(population, fitness) for _ in range(population_size // 2)]
        offspring = []
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                break
            parent1 = parents[i]
            parent2 = parents[i + 1]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            offspring.append(project_to_feasible(child1))
            offspring.append(project_to_feasible(child2))

        elite_indices = np.argsort(fitness)[:num_elite]
        elites = [population[i] for i in elite_indices]
        population = elites + offspring[:population_size - len(elites)]
        mutation_rate *= mutation_decay

    local_opt_candidates = [best_h] + [tournament_selection(population, [evaluate_individual(ind) for ind in population]) for _ in range(4)]

    for candidate in local_opt_candidates:
        result = minimize(
            fun=lambda x: evaluate_individual(x),
            x0=candidate,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(half_n)],
            constraints=({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}),
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        if result.success:
            optimized_x = project_to_feasible(result.x)
            candidate_c5 = evaluate_individual(optimized_x)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = optimized_x

    h_full = create_full_h(best_h)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
