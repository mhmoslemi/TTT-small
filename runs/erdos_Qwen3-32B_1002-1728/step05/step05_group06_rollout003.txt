    temp = 0.1
    eps = 1e-4
    max_iter = int(1e4)
    iter_step = 0

    np.random.seed(seed)

    while time.time() - start_time < budget_s and iter_step < max_iter:
        # Create full h and overlaps
        h_full = create_full_h(h_half)
        overlaps = compute_overlaps(h_full)
        soft_max = temp * np.log(np.sum(np.exp(overlaps / temp)))  # Smoothed max

        # Gradient via finite diff
        grad = np.zeros(half_n)
        h_half_ = h_half.copy()
        for i in range(half_n):
            h_perturbed = h_half_.copy()
            h_perturbed[i] += eps
            h_perturbed_sum = np.sum(h_perturbed)
            if h_perturbed_sum != 0:
                h_perturbed = h_perturbed * (target_sum_half / h_perturbed_sum)
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            h_full_p = create_full_h(h_perturbed)
            overlaps_p = compute_overlaps(h_full_p)
            soft_max_p = temp * np.log(np.sum(np.exp(overlaps_p / temp)))
            grad[i] = (soft_max_p - soft_max) / eps

        # Gradient update and projection
        h_half -= lr * grad
        h_half = np.clip(h_half, 0.0, 1.0)
        h_half_sum = np.sum(h_half)
        if h_half_sum != 0:
            h_half = h_half * (target_sum_half / h_half_sum)
            h_half = np.clip(h_half, 0.0, 1.0)

        # Update best solution
        h_full_current = create_full_h(h_half)
        overlaps_current = compute_overlaps(h_full_current)
        current_c5 = np.max(overlaps_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()

        iter_step += 1

    h_full_best = create_full_h(best_h_half)
    assert np.allclose(np.sum(h_full_best), n_points / 2.0, atol=1e-6)
    return (h_full_best.tolist(), best_c5, n_points)
