            # Revert the perturbation
            h_half[j] = original

        # Perform gradient ascent
        h_half += alpha * grad

        # Clip to [0,1]
        h_half = np.clip(h_half, 0, 1)

        # Scale to maintain sum
        current_sum = h_half.sum()
        if current_sum != 0:
            h_half = (h_half / current_sum) * target_sum_half

        h_half = np.clip(h_half, 0, 1)

    return h_half

import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5_and_m(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_index = np.argmax(corr)
        m_worst = max_index - (n_points - 1)
        return np.max(corr) * dx, m_worst

    np.random.seed(seed)
    population_size = 50

    generation = 0
