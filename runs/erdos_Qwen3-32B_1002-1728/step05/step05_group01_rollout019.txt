import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_flat):
        h = h_flat.copy().astype(np.float64)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    # Initial projection
    h = h_initial.copy().astype(np.float64)
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-6:
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    best_h = h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        # Noise schedule decaying with time
        noise_scale = max(0.3 * (1 - (time.time() - start_time)/budget_s)**1.5, 0.02)
        perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0.0, 1.0)

        # Project to feasible region
        s = perturbed.sum()
        if s < 1e-10:
            continue
        perturbed *= target_sum / s
        perturbed = np.clip(perturbed, 0.0, 1.0)

        # Optimization
        method = 'SLSQP'
        result = minimize(
            fun=objective,
            x0=perturbed,
            method=method,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 80, 'ftol': 1e-10, 'disp': False}
        )
        if result.success:
            h_opt = np.clip(result.x, 0.0, 1.0)
            s_opt = h_opt.sum()
            if abs(s_opt - target_sum) > 1e-6:
                h_opt = (h_opt / s_opt) * target_sum
                h_opt = np.clip(h_opt, 0.0, 1.0)
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_h[:] = h_opt.copy()
                best_c5 = c5

        if (time.time() - start_time) > budget_s - 2:
            break

    # Final projection
    best_h = np.clip(best_h, 0.0, 1.0)
    s = best_h.sum()
    if abs(s - target_sum) > 1e-6:
        best_h = (best_h / s) * target_sum
    final_c5 = objective(best_h)
    return (best_h.tolist(), final_c5, n_points)
