import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective_func(h_unscaled):
        h = np.clip(h_unscaled, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            h_values = 0.5 * np.ones_like(h)
        else:
            scale = target_sum / current_sum
            h_values = np.clip(h * scale, 0, 1)
        corr = np.correlate(h_values, 1 - h_values, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    maxiter = 500 if n_points <= 70 else 200  # Conservative estimate
    result = differential_evolution(
        func=objective_func,
        bounds=bounds,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        init='sobol',
        disp=False,
        polish=True,
        workers=1,
        updating='deferred'
    )
    
    h_final = np.clip(result.x, 0, 1)
    total = np.sum(h_final)
    if total != 0:
        h_final = h_final * (target_sum / total)
    h_final = np.clip(h_final, 0, 1)
    best_c5 = objective_func(h_final)
    
    return (h_final.tolist(), best_c5, n_points)
