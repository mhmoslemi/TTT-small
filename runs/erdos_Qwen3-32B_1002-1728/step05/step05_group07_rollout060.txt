        if res.success:
            h_half_candidate = (h_half + res.x).copy()
            h_half_candidate = np.clip(h_half_candidate, 0, 1)
            current_sum = np.sum(h_half_candidate)
            if abs(current_sum - target_sum_half) > 1e-8:
                correction = (target_sum_half - current_sum) / half_n
                h_half_candidate += correction
                h_half_candidate = np.clip(h_half_candidate, 0, 1)
            h_half[:] = h_half_candidate
        else:
            pass  # Skip update on failure

    h_full_best = get_h_full(best_h_half)
    assert abs(np.sum(h_full_best) - n_points / 2.0) < 1e-6
    return (h_full_best.tolist(), best_c5, n_points)
