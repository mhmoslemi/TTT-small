from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)

        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        population[i] = perturbed

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return max(0, np.max(corr) * dx)

        return np.all((h_full >= 0) & (h_full <= 1)) and abs(sum(h_full) - target_sum) < 1e-6

    best_h = h_initial.copy()
