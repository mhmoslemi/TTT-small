import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points

    def evaluate_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx

    initial_C5 = evaluate_c5(h_full_initial)
    low = 0.0
    high = initial_C5
    best_C5 = high
    best_h = h_full_initial.copy()

    # Precompute all (i, j) pairs for each possible k
    k_to_terms = []
    for k in range(- (n_points - 1), n_points):
        terms = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                terms.append((i, j))
        k_to_terms.append(terms)

    # CVXPY variables and constraints
    h_var = cp.Variable(n_points)
    z_var = cp.Variable(n_points, n_points)
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) * dx == 1
    ]

    # McCormick constraints for all (i, j)
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                z_var[i, j] >= h_var[i] - h_var[j],
                z_var[i, j] <= h_var[i],
                z_var[i, j] <= 1 - h_var[j],
                z_var[i, j] >= 0
            ]

    # Parameter for C5 bound
    C5_param = cp.Parameter(pos=True)
    for terms in k_to_terms:
        sum_z = 0.0
        for i, j in terms:
            sum_z += z_var[i, j]
        if sum_z > 0:
            constraints += [sum_z * dx <= C5_param]

    problem = cp.Problem(cp.Minimize(0), constraints)

    while time.time() - start_time < budget_s - 20:
        mid = (low + high) / 2.0
        C5_param.value = mid
        problem.solve(solver=cp.GLOP, verbose=False, max_iter=1000)
        if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            current_h = np.clip(h_var.value, 0, 1)
            current_C5 = evaluate_c5(current_h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = current_h.copy()
            high = mid
        else:
            low = mid

    # Ensure sum is exactly 1.0 * dx with clipping
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * dx
    if not np.isclose(total, 1.0, atol=1e-5):
        best_h *= (0.5 / total)
    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_C5, n_points)
