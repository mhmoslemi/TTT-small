import numpy as np
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def objective(h_half_flat):
        h_half = h_half_flat.reshape(-1)
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = np.sum(h_copy)
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    np.random.seed(seed)
    initial_h_half = h_full_initial[:half_n].copy()
    initial_h_half_normalized = normalize_h_half(initial_h_half, target_sum_half)
    population = [initial_h_half_normalized]
    population_size = 50

    for _ in range(population_size - 1):
        perturbed = initial_h_half_normalized + np.random.normal(0, 0.2, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        normalized = normalize_h_half(perturbed, target_sum_half)
        population.append(normalized)

    best_h_half = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 120:
        population_c5 = []
        for h_half in population:
            current_c5 = objective(h_half)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_half = h_half.copy()
            population_c5.append(current_c5)
        pop_with_c5 = list(zip(population_c5, population))
        pop_with_c5.sort(key=lambda x: x[0])
        top_half = [x[1] for x in pop_with_c5[:population_size // 2]]
        new_population = top_half.copy()
        while len(new_population) < population_size:
            parent_indices = np.random.choice(len(top_half), 2, replace=False)
            p1 = top_half[parent_indices[0]]
            p2 = top_half[parent_indices[1]]
            child = (p1 + p2) / 2.0
            if np.random.rand() < 0.2:
                child += np.random.normal(0, 0.1, half_n)
            child = normalize_h_half(child, target_sum_half)
            new_population.append(child)
        population = new_population

    # Local optimization with constraint on sum(h_half)
    if best_h_half is not None and time.time() - start_time < budget_s - 10:
        bounds_opt = Bounds(0.0, 1.0)
        cons_opt = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
        result = minimize(
            fun=objective,
            x0=best_h_half.ravel(),
            method='SLSQP',
            bounds=bounds_opt,
            constraints=cons_opt,
            tol=1e-8
        )
        if result.success:
            best_h_half = result.x.reshape(-1).copy()
            best_c5 = objective(best_h_half)

    h_full = create_full_h(best_h_half)
