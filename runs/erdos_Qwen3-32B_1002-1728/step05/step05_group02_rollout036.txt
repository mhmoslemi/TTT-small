import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h_candidate):
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr) * dx

    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-8:
        h = np.clip(h * (target_sum / current_sum), 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 60:
        remaining = budget_s - (time.time() - start_time)
        frac = remaining / budget_s
        noise_scale = 0.35 * (1 - frac ** 2)

        perturbed = best_h + np.random.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if s != 0:
            perturbed = (perturbed / s) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=lambda x: evaluate_c5(x),
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 40, 'ftol': 1e-9, 'disp': False}
        )

        if result.success:
            rescaled = np.clip(result.x, 0, 1)
            s = np.sum(rescaled)
            rescaled = (rescaled / max(1e-8, s)) * target_sum
            rescaled = np.clip(rescaled, 0, 1)
            c5 = evaluate_c5(rescaled)
            if c5 < best_c5:
                best_h, best_c5 = rescaled, c5

    best_h = np.clip(best_h, 0, 1)
    s = np.sum(best_h)
    if s != 0:
        best_h = (best_h / s) * target_sum
    best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
