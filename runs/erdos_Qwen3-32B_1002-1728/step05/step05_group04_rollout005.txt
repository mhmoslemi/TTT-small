def crossover(parent1, parent2):
    # One-point crossover
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child = np.zeros(n_points, dtype=int)
    child[:crossover_point] = parent1[:crossover_point]
    child[crossover_point:] = parent2[crossover_point:]
    # Now adjust the number of ones
    target_ones = np.count_nonzero(parent1)  # which is n_ones
    current_ones = np.sum(child)
    diff = target_ones - current_ones
    # find indices to flip
    zeros_indices = np.where(child == 0)[0]
    ones_indices = np.where(child == 1)[0]
    if diff > 0:
        # need to turn diff zeros into ones
        flip_indices = np.random.choice(zeros_indices, size=diff, replace=False)
        child[flip_indices] = 1
    elif diff < 0:
        # need to turn -diff ones into zeros
        flip_indices = np.random.choice(ones_indices, size=-diff, replace=False)
        child[flip_indices] = 0
    # child now has correct number of ones
    return child
