import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to satisfy constraints
    h_initial = h_initial.copy()
    h_initial = np.clip(h_initial, 0, 1)
    total = np.sum(h_initial)
    if total != 0:
        scale = target_sum / total
        h_initial = (h_initial * scale).clip(0, 1)
    h_best = h_initial.copy()
    c5_best = np.max(np.correlate(h_best, 1 - h_best, mode='full')) * dx

    def objective(h_vars):
        return np.max(np.correlate(h_vars, 1 - h_vars, mode='full')) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    np.random.seed(seed)

    total_attempts = 0
    while time.time() - start_time < budget_s - 30:
        frac = (time.time() - start_time) / budget_s
        noise_scale = 0.3 * (1 - (frac ** 0.75))
        
        for _ in range(2):  # Number of perturbations per iteration
            perturbation = np.random.uniform(-noise_scale, noise_scale, size=n_points)
            h_perturbed = h_best + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            total_p = np.sum(h_perturbed)
            if total_p < 1e-8:
                continue
            h_perturbed = (h_perturbed * (target_sum / total_p)).clip(0, 1)

            # Try different optimization methods
            for method in ['COBYLA', 'SLSQP']:
                if time.time() - start_time >= budget_s - 30:
                    break
                res = minimize(
                    fun=lambda x: objective(x),
                    x0=h_perturbed,
                    bounds=bounds,
                    constraints=constraint,
                    method=method,
                    options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
                )
                if res.success:
                    h_candidate = np.clip(res.x, 0, 1)
                    total_c = np.sum(h_candidate)
                    if abs(total_c - target_sum) < 1e-6:
                        c5_candidate = objective(h_candidate)
                        if c5_candidate < c5_best:
                            h_best = h_candidate.copy()
                            c5_best = c5_candidate
        total_attempts += 1

    # Final enforcement of constraints
    h_best = np.clip(h_best, 0, 1)
    total = np.sum(h_best)
    if abs(total - target_sum) > 1e-6:
        h_best = (h_best * (target_sum / total)).clip(0, 1)

    return (h_best.tolist(), c5_best, n_points)
