        idxs = np.where(h_full_initial > 0.5)[0]
        if len(idxs) < K:
            add = np.random.choice(n_points, K - len(idxs), replace=False)
            idxs = np.unique(np.concatenate([idxs, add]))
        idxs = idxs[:K]
        centers = x_vals[idxs]
        widths = np.full(K, 0.4)
        amplitudes = h_full_initial[idxs]
        return handle_constraints(centers, widths, amplitudes)

    def get_random_individual():
        cent = np.random.uniform(0, 2, K)
        wid = np.random.uniform(0.1, 0.5, K)
        amp = np.random.uniform(0, 1, K)
        return handle_constraints(cent, wid, amp)

    np.random.seed(seed)
    population_size = 50
    population = [get_initial_from_h() for _ in range(population_size - 1)] + [get_random_individual()]
    generation = 0

    best_c5 = float('inf')
    best_ind = None

    while time.time() - start_time < budget_s - 10:
        candidates = [
            (compute_c5(*ind), ind) for ind in population
        ]
        candidates.sort()
        parents = [ind for c, ind in candidates[:population_size//2]]
        new_pop = list(parents)

        while len(new_pop) < population_size:
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]

            c1 = (p1[0] + p2[0]) / 2
            w1 = (p1[1] + p2[1]) / 2
            a1 = (p1[2] + p2[2]) / 2

            if np.random.rand() < 0.2:
                mut = 0.3 * (1 - (len(new_pop) / float(population_size)))
                c1 += np.random.normal(0, mut, K)
                w1 += np.random.normal(0, mut, K)
                a1 += np.random.normal(0, mut, K)

            c1, w1, a1 = handle_constraints(c1, w1, a1)
            new_pop.append((c1, w1, a1))

        population = new_pop
        current_c5, current_ind = min(
            [(compute_c5(*ind), ind) for ind in new_pop],
            key=lambda x: x[0]
        )
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_ind = current_ind

    h = create_h(*best_ind)
    assert abs(np.sum(h) - n_points / 2.0) < 1e-6
    return (h.tolist(), best_c5, n_points)
