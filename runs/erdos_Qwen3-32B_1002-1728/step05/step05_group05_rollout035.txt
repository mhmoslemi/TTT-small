    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # sum(h_half) must ensure sum(h_full) = n_points / 2
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_fitness(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population_size = 50
    population = []

    # Initialize population with perturbed initial solution
    initial_h_half = h_full_initial[:half_n].copy()
    population.append(initial_h_half.copy())
    for _ in range(population_size - 1):
        perturbed = initial_h_half + np.random.uniform(-0.05, 0.05, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if not np.isclose(current_sum, 0):
            scale = target_sum_half / current_sum
            perturbed *= scale
            perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_fitness = np.inf
    best_individual = None

    while time.time() - start_time < budget_s - 20:
        # Recompute population fitness
        fitnesses = [evaluate_fitness(ind) for ind in population]
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()

        # Generate offspring
        num_offspring = population_size // 2
        offspring = []
        for _ in range(num_offspring):
            # Tournament selection for two parents
            parent1_idx = np.random.choice(len(population), 2, replace=False)
            idx1, idx2 = parent1_idx[0], parent1_idx[1]
            f1, f2 = fitnesses[idx1], fitnesses[idx2]
            winner1 = idx1 if f1 < f2 else idx2

            parent2_idx = np.random.choice(len(population), 2, replace=False)
            idx3, idx4 = parent2_idx[0], parent2_idx[1]
            f3, f4 = fitnesses[idx3], fitnesses[idx4]
            winner2 = idx3 if f3 < f4 else idx4

            # Crossover
            beta = np.random.uniform(0.3, 0.7)
            child = beta * population[winner1] + (1 - beta) * population[winner2]

            # Mutation
            if np.random.rand() < 0.25:
                noise = np.random.normal(0, 0.01, half_n)
                child += noise
                child = np.clip(child, 0, 1)

            # Scale to target sum and clip again
            child_sum = np.sum(child)
            if not np.isclose(child_sum, 0):
                scale = target_sum_half / child_sum
                child *= scale
                child = np.clip(child, 0, 1)
            offspring.append(child)

        # Replace worst in population with offspring
        replace_indices = np.argsort(fitnesses)[-num_offspring:]
        for i, idx in enumerate(replace_indices):
            population[idx] = offspring[i]

    # Final validation and output
    best_h_full = create_full_h(best_individual)
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    assert np.abs(np.sum(best_h_full) - n_points / 2.0) < 1e-6
    return (best_h_full.tolist(), best_fitness, n_points)
