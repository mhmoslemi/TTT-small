import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def constrain_and_clip(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        h_final = np.clip(h_scaled, 0, 1)
        return h_final

    h = constrain_and_clip(h_initial, target_sum)

    def objective(h_vars):
        corr = np.correlate(h_vars, 1 - h_vars, mode='full')
        return np.max(corr) * dx

    best_h = h.copy()
    best_c5 = objective(best_h)
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        remaining_time = budget_s - (time.time() - start_time)
        noise_scale = max(0.1, 0.3 * (remaining_time / budget_s) ** 2)

        seed += 1
        rnd = np.random.RandomState(seed)
        perturbation = rnd.normal(0, noise_scale, n_points)
        h_perturbed = h.copy()
        h_perturbed += perturbation
        h_perturbed = constrain_and_clip(h_perturbed, target_sum)

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-12, 'disp': False}
        )

        if result.success:
            new_h = np.clip(result.x, 0, 1)
            if abs(np.sum(new_h) - target_sum) < 1e-6:
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    h = new_h.copy()
                    best_c5 = new_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6

    return (best_h.tolist(), best_c5, n_points)
