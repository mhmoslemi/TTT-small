def halton_sequence(n, base=2):
    sequence = np.zeros(n)
    for i in range(n):
        x = 0.0
        b = 1.0 / base
        v = i + 1  # since van der Corput uses i+1
        while v > 0:
            x += (v % base) * b
            v = v // base
            b = b / base
        sequence[i] = x
    return sequence

import numpy as np
import time

def generate_halton(n, base=2):
    sequence = np.zeros(n)
    for i in range(n):
        x = 0.0
        b = 1.0 / base
        v = i + 1
        while v > 0:
            x += (v % base) * b
            v //= base
            b /= base
        sequence[i] = x
    return sequence
