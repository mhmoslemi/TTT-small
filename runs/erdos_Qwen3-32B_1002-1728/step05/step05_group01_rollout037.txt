import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_clipped = np.clip(h_vars, 0.0, 1.0)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons_sum = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    h_initial = np.array(initial_h_values)
    h_initial_clipped = np.clip(h_initial, 0.0, 1.0)
    scaling_factor = target_sum / np.sum(h_initial_clipped)
    h_scaled = h_initial_clipped * scaling_factor
    h_scaled = np.clip(h_scaled, 0.0, 1.0)

    best_h = h_scaled.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        noise_scale = 0.5 * (1 - (frac ** 1.0))
        if noise_scale < 0.01:
            noise_scale = 0.01

        for _ in range(2):
            perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum <= 1e-8:
                continue
            perturbed = (perturbed / perturbed_sum) * target_sum
            perturbed = np.clip(perturbed, 0.0, 1.0)

            method = 'SLSQP' if np.random.random() < 0.3 else 'COBYLA'
            res = minimize(objective, x0=perturbed, method=method, bounds=bounds_per_var, constraints=cons_sum,
                           options={'maxiter': 150 if method == 'SLSQP' else 100, 'ftol': 1e-10})
            if res.success and res.fun < best_c5 - 1e-7:
                new_h = np.clip(res.x, 0.0, 1.0)
                new_h_sum = np.sum(new_h)
                if np.isclose(new_h_sum, target_sum, atol=1e-4):
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                break

        if (n_attempt % 50 == 0) and (np.abs(best_c5 - objective(best_h)) / best_c5 < 1e-5):
            break

    best_h = np.clip(best_h, 0.0, 1.0)
    final_sum = np.sum(best_h)
    if np.abs(final_sum - target_sum) > 1e-4:
        scaling_factor = target_sum / final_sum
        best_h = best_h * scaling_factor
        best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
