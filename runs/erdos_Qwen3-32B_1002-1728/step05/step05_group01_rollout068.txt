import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # RBF parameters
    n_rbf = 50
    x_centers = np.linspace(0, 2, n_rbf)
    gamma = 10.0  # Width parameter

    # Precompute basis functions
    x_samples = np.linspace(0, 2, n_points)
    Phi_matrix = np.zeros((n_points, n_rbf))
    for i in range(n_rbf):
        x_c = x_centers[i]
        Phi_matrix[:, i] = np.exp(-gamma * (x_samples - x_c)**2)

    # Initial guess: uniform coefficients, normalized later
    c_initial = np.ones(n_rbf) / n_rbf

    def objective_c(c):
        h = np.dot(Phi_matrix, c)
        h_clipped = np.clip(h, 0, 1)
        sum_h = h_clipped.sum() * dx
        if abs(sum_h) > 1e-10:
            h_final = h_clipped / sum_h
        else:
            h_final = np.zeros(n_points)
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds_c = [(0.0, None) for _ in range(n_rbf)]
    best_c = c_initial.copy()
    best_c5 = objective_c(best_c)
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        # Perturb current best solution
        perturbation = (np.random.rand(n_rbf) * 2 - 1) * 0.05
        c_proposal = best_c + perturbation
        c_proposal = np.clip(c_proposal, 0, None)
        
        for method in ['L-BFGS-B', 'COBYLA']:
            result = minimize(
                fun=objective_c,
                x0=c_proposal,
                bounds=bounds_c,
                method=method,
                options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                c5_new = result.fun
                if c5_new < best_c5:
                    best_c = result.x.copy()
                    best_c5 = c5_new
                    # Ensure no negative coefficients
                    best_c = np.clip(best_c, 0, None)

    # Final evaluation with best_c
    h_best = np.dot(Phi_matrix, best_c)
    h_best = np.clip(h_best, 0, 1)
    sum_h = h_best.sum() * dx
    if sum_h > 1e-10:
        h_best = h_best / sum_h
    else:
        h_best = h_initial  # fallback to initial if invalid

    # Final check of constraints
    assert abs(h_best.sum() * dx - 1) < 1e-5, f"Integral check failed: {h_best.sum() * dx}"
    assert np.all((h_best >= -1e-7) & (h_best <= 1.0 + 1e-7)), f"Bounds check failed"

    return (h_best.tolist(), best_c5, n_points)
