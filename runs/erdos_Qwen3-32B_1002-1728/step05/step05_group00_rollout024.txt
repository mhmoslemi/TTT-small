import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Repair and normalize initial guess
    h = h_init.copy()
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    if total > 1e-10:
        h = h * target_sum / total
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # Differential evolution over full domain
    def objective(x_vars):
        x = np.clip(np.array(x_vars), 0, 1).copy()
        total = np.sum(x)
        if total > 1e-10:
            x *= target_sum / total
        x = np.clip(x, 0, 1)
        return _compute_c5(x, dx)

    bounds = [(0, 1)] * n_points
    try:
        de_result = differential_evolution(
            objective, bounds, seed=seed, strategy='best1bin',
            maxiter=12, popsize=10, mutation=(0.5, 1), recombination=0.85,
            tol=1e-4, workers=1, disp=False
        )
        if de_result.success:
            de_x = np.clip(de_result.x.copy(), 0, 1)
            total = np.sum(de_x)
            if total > 1e-10:
                de_x *= target_sum / total
            de_x = np.clip(de_x, 0, 1)
            de_c5 = _compute_c5(de_x, dx)
            if de_c5 < best_c5:
                best_h[:] = de_x
                best_c5 = de_c5
    except Exception:
        pass

    # Local optimization with adaptive steps
    def repair(x):
        x = np.clip(x.copy(), 0, 1)
        total = np.sum(x)
        if total > 1e-10:
            x *= target_sum / total
        x = np.clip(x, 0, 1)
        return x

    def opt_obj(x, repair=True):
        if repair:
            x = repair(x)
        return _compute_c5(x, dx)

    attempts = 0
    while time.time() - start_time < budget_s - 30:
        # Perturb current best
        perturbed = best_h + np.random.RandomState(seed).normal(0, 0.1, n_points)
        perturbed = repair(perturbed)
        
        # Try different solvers
        for method in ['SLSQP', 'L-BFGS-B']:
            if time.time() - start_time > budget_s - 30:
                break
            result = minimize(
                opt_obj, x0=perturbed.copy(), method=method,
                bounds=bounds, args=(attempts % 2 == 0),
                options={'maxiter': 120, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                candidate = repair(result.x.copy())
                candidate_c5 = _compute_c5(candidate, dx)
                if candidate_c5 < best_c5:
                    best_h[:] = candidate
                    best_c5 = candidate_c5
        attempts += 1
        seed += 1

    # Final constraint check
    final_total = np.sum(best_h)
    assert abs(final_total - target_sum) < 1e-6
    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)

def _compute_c5(h, dx):
    h = np.array(h)
    h = np.clip(h, 0, 1)
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
