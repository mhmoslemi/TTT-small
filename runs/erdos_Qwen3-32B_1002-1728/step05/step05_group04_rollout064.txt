    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_full = h_full_initial.copy()
    total = np.sum(h_full)
    if total != 0:
        scale = target_sum / total
        h_full = h_full * scale
    h_full = np.clip(h_full, 0, 1)

    best_h = h_full.copy()
    best_c5 = objective(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    np.random.seed(seed)
    n_attempt = 0

    # Differential evolution setup
    def de_objective(h):
        h = h.copy()
        h_total = np.sum(h)
        if h_total == 0:
            return np.inf
        h = h * (target_sum / h_total)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    de_bounds = [(0.0, 1.0)] * n_points
    de_popsize = 15
    de_maxiter = max(1, int((budget_s / 2) // n_points))
    
    # Run differential evolution for a fixed time
    total_elapsed = 0
    while total_elapsed < budget_s - 150:  # 150s remaining for final polish
        de_elapsed = time.time() - start_time
        if de_elapsed > budget_s - 150 - 1:
            break
        # Run DE for 10 iterations at a time, adjusting based on remaining time
        try:
            de_result = differential_evolution(
                func=de_objective,
                bounds=de_bounds,
                maxiter=10,
                popsize=de_popsize,
                strategy='best1bin',
                tol=1e-6,
                seed=seed + n_attempt,
                mutation=(0.5, 1),
                recombination=0.8,
                polish=False,
                disp=False
            )
            h_candidate = de_result.x.copy()
            h_candidate_total = np.sum(h_candidate)
            if h_candidate_total != 0:
                h_candidate = h_candidate * (target_sum / h_candidate_total)
                h_candidate = np.clip(h_candidate, 0, 1)
                if objective(h_candidate) < best_c5:
                    best_h, best_c5 = h_candidate.copy(), objective(h_candidate)
            n_attempt += 1
            total_elapsed = time.time() - start_time
        except:
            break

    bfgs_result = minimize(
        fun=objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 100}
    )
    if bfgs_result.success and objective(bfgs_result.x) < best_c5:
        final_h = np.clip(bfgs_result.x, 0, 1)
        best_c5 = objective(final_h)
        best_h = final_h

    cobyla_result = minimize(
        fun=objective,
        x0=best_h,
        method='COBYLA',
        bounds=bounds,
        options={'tol': 1e-10, 'maxiter': 100}
    )
    if cobyla_result.success and objective(cobyla_result.x) < best_c5:
        final_h = np.clip(cobyla_result.x, 0, 1)
        best_c5 = objective(final_h)
        best_h = final_h

    return (best_h.tolist(), best_c5, n_points)
