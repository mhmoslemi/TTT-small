        return np.clip(h_scaled, 0, 1)

    def compute_overlaps(h_full):
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return corr * (2.0 / n_points)

    def compute_smooth_c5(overlaps, tau):
        max_c = np.max(overlaps)
        exp_vals = np.exp((overlaps - max_c) / tau)
        return max_c + tau * np.log(np.sum(exp_vals) + 1e-10)

    h_half = h_full_initial[:half_n].copy()
    h_half = project(h_half)
    h_full = create_full_h(h_half)
    overlaps = compute_overlaps(h_full)
    best_c5 = np.max(overlaps)

    np.random.seed(seed)
    tau = 0.1
    step_size = 0.01
    best_h_half = h_half.copy()

    while time.time() - start_time < budget_s - 10:
        overlaps = compute_overlaps(create_full_h(h_half))
        smooth_c5 = compute_smooth_c5(overlaps, tau)
        grad = np.zeros(half_n)
        epsilon = 1e-5

        for i in range(half_n):
            h_perturbed = h_half.copy()
            h_perturbed[i] += epsilon
            h_proj = project(h_perturbed)
            h_full_p = create_full_h(h_proj)
            overlaps_p = compute_overlaps(h_full_p)
            smooth_p = compute_smooth_c5(overlaps_p, tau)
            grad[i] = (smooth_p - smooth_c5) / epsilon

        h_candidate = np.clip(h_half - step_size * grad, 0, 1)
        h_candidate = project(h_candidate)
        h_full_candidate = create_full_h(h_candidate)
        overlaps_candidate = compute_overlaps(h_full_candidate)
        new_c5 = np.max(overlaps_candidate)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h_half = h_candidate.copy()
            step_size *= 1.05
        else:
            step_size *= 0.95

        h_half = h_candidate.copy()
    final_h_full = create_full_h(best_h_half)
    return (final_h_full.tolist(), best_c5, n_points)
