import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)
    start_time = time.time()

    # PSO Parameters
    n_particles = 20
    max_pso_iter = 60
    w = 0.8  # Inertia
    c1, c2 = 1.2, 1.2  # Cognitive, Social

    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    pb_vals = []
    for _ in range(n_particles):
        noise = np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_initial + noise, 0, 1)
        h_scaled = (h_perturbed * target_sum) / h_perturbed.sum()
        h_scaled = np.clip(h_scaled, 0, 1)
        particles.append(h_scaled)
        velocities.append(0.01 * np.random.randn(n_points))
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        pb_vals.append(np.max(corr) * dx)
        personal_best.append(h_scaled.copy())

    # Find global best
    global_best_idx = np.argmin(pb_vals)
    global_best = personal_best[global_best_idx].copy()
    best_c5 = pb_vals[global_best_idx]

    # Adaptive max iterations
    max_pso_iter = min(max_pso_iter, int((budget_s - (time.time()-start_time)) / 60))

    for iter in range(max_pso_iter):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (global_best - particles[i]))
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            
            # Constraint projection
            current_sum = particles[i].sum()
            if abs(current_sum) < 1e-10:
                particles[i] = 0.5 * np.ones(n_points)
            else:
                particles[i] = (particles[i] / current_sum) * target_sum
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Update objectives
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < pb_vals[i]:
                personal_best[i] = particles[i].copy()
                pb_vals[i] = current_c5
                if current_c5 < best_c5:
                    global_best = personal_best[i].copy()
                    best_c5 = pb_vals[i]
        w *= 0.95  # Decay inertia

    top_candidates = min(5, n_particles)
    indices = np.argsort(pb_vals)[:top_candidates]
    new_h = None

    for idx in indices:

        bounds = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
        
        result = minimize(
