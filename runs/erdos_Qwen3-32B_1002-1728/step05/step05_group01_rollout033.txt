import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    x = np.linspace(0, 2, n_points)
    dx = 2.0 / n_points
    gamma = 4.0  # controls RBF width

    # Construct radial basis function matrix
    xx = x - x[:, np.newaxis]
    Phi = np.exp(-gamma * (xx ** 2))

    # Integral constraint coefficient
    integral_Aeq = Phi.sum(axis=0) * dx

    # Compute initial coefficient vector via least squares
    h_init = h_full_initial
    try:
        c_init, _, _, _ = np.linalg.lstsq(Phi, h_init, rcond=None)
    except:
        c_init = np.zeros(n_points)
    
    # Normalize to satisfy integral constraint
    current_integral = integral_Aeq @ c_init
    if not np.isclose(current_integral, 0):
        c_init = c_init / (current_integral + 1e-10)

    # Objective function using RBFs
    def objective(c):
        c = np.atleast_1d(c)
        h = Phi @ c
        h_clipped = np.clip(h, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Integral constraint function
    def integral_constraint(c):
        return integral_Aeq @ c - 1.0

    cons = ({'type': 'eq', 'fun': integral_constraint})

    best_c = c_init.copy()
    best_c5 = objective(best_c)
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        # Perturb current best solution
        perturbation = np.random.normal(0, 0.1, n_points)
        new_c = best_c + perturbation
        new_c = new_c / (integral_Aeq @ new_c + 1e-10)  # Rescale to meet constraint

        # Run optimization with different methods
        try:
            res = minimize(
                objective,
                x0=new_c,
                constraints=cons,
                method='COBYLA',
                options={'maxiter': 150, 'tol': 1e-7}
            )
            if res.success and res.fun < best_c5:
                h_test = Phi @ res.x
                if np.allclose((h_test.clip(0,1).sum() * dx), 1, atol=1e-5):
                    best_c5 = res.fun
                    best_c = res.x
        except:
            pass

    h_final = Phi @ best_c
    h_final = h_final.clip(0, 1)
    assert abs(h_final.sum() * dx - 1) < 1e-5, f"Integral mismatch: {(h_final.sum() * dx):.8f} != 1"
    return (h_final.tolist(), best_c5, n_points)
