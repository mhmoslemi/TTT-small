import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def project_h_half(h_vars, target_sum_half):
        h_clipped = np.clip(h_vars, 0, 1)
        total = np.sum(h_clipped)
        if total == 0:
            h_clipped[:] = target_sum_half / half_n
        else:
            h_clipped = h_clipped * (target_sum_half / total)
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    def objective_half(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Symmetrize initial h_full_initial
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        h_center = (h_full_initial[i] + h_full_initial[n_points - 1 - i]) / 2
        h_full_sym[i] = h_full_sym[n_points - 1 - i] = h_center
    h_half_initial = project_h_half(h_full_sym[0:half_n], target_sum_half)

    # PSO Parameters
    n_particles = 20
    max_pso_iter = 100
    w = 0.5
    c1 = 1.5
    c2 = 1.5
    np.random.seed(seed)
    particles = []
    velocities = []
    personal_bests = []
    personal_best_scores = np.full(n_particles, np.inf)

    for _ in range(n_particles):
        noise = np.random.normal(0, 0.1, half_n)
        h_candidate = h_half_initial + noise
        h_candidate = project_h_half(h_candidate, target_sum_half)
        particles.append(h_candidate)
        velocities.append(np.zeros(half_n))
        score = objective_half(h_candidate)
        personal_bests.append(h_candidate.copy())
        personal_best_scores[_] = score

    global_best_idx = np.argmin(personal_best_scores)
    global_best_score = personal_best_scores[global_best_idx]

    for pso_iter in range(max_pso_iter):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break

        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (personal_bests[i] - particles[i]) +
                             c2 * r2 * (personal_bests[global_best_idx] - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = project_h_half(new_pos, target_sum_half)

            score = objective_half(new_pos)
            if score < personal_best_scores[i]:
                personal_bests[i] = new_pos.copy()
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best_score = score
                    global_best_idx = i

    # Refine top candidates
    top_indices = np.argsort(personal_best_scores)[:4]
    best_h_half = personal_bests[global_best_idx].copy()
    best_c5 = global_best_score

    bounds = [(0, 1)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    for idx in top_indices:
        result = minimize(
            fun=objective_half,
            x0=personal_bests[idx],
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-12}
        )
        if result.success:
            refined = project_h_half(result.x, target_sum_half)
            refined_c5 = objective_half(refined)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h_half = refined

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1 + 1e-6) and np.all(h_full >= -1e-6)

    return (h_full.tolist(), best_c5, n_points)
