import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(n_points//2):
            h_full[i] = h_full[n_points - 1 -i] = h_half[i]
        return h_full

    def compute_c5(h_func):
        corr = np.correlate(h_func, 1 - h_func, mode='full')
        return np.max(corr) * dx

    # Initialize h_full symmetrically and scaled
    h_half_initial = h_full_initial[:n_points//2].copy()
    h_full = create_full_h(h_half_initial)
    h_full = h_full / h_full.sum() * target_sum
    h_full = np.clip(h_full, 0, 1.0)
    best_h = h_full.copy()
    best_c5 = compute_c5(best_h)
    
    max_iter = 1000
    step_size = 0.02
    decay_factor = 0.999
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Compute current C5 and find worst shift
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        worst_idx = np.argmax(corr)
        # Determine shift m associated with worst_idx
        m = worst_idx - (n_points - 1)
        
        # Compute gradient for current shift m
        g_full = np.zeros_like(h_full)
        for k in range(n_points):
            val = 0.0
            im = k + m
            if im < n_points:
                val += h_full[im]
            jm = k - m
            if jm >= 0:
                val += h_full[jm]
            g_full[k] = val

        # Convert gradient_full to gradient for h_half
        g_half = g_full[:n_points//2]  # Only left half needed due to symmetry
        # Multiply by 2 because h_full is symmetric (each h_half entry affects two elements)
        g_half = g_full[:n_points//2] * 2.0
        # Gradient is scaled by -dx to reduce the integral
        gradient = -dx * g_half

        # Update h_half
        h_half = h_half_initial - step_size * gradient
        h_half = np.clip(h_half, 0, 1.0)

        # Recompute full h and project to constraints
        h_full_new = create_full_h(h_half)
        h_full_new = np.clip(h_full_new, 0, 1.0)
        current_sum = np.sum(h_full_new) + 1e-12  # Avoid division by zero
        scaling = target_sum / current_sum
        h_full_new = h_full_new * scaling
        h_full_new = np.clip(h_full_new, 0, 1.0)

        # Check if improved
        current_c5 = compute_c5(h_full_new)
        if current_c5 < best_c5:
            best_h = h_full_new.copy()
            best_c5 = current_c5
            h_half_initial = best_h[:n_points//2].copy()  # Update for next iteration
        else:
            # Reduce step size if no improvement
            step_size *= decay_factor

    return (best_h.tolist(), best_c5, n_points)
