            offspring.append(child)
        all_candidates = population + offspring
        all_candidates.sort(key=lambda h: compute_c5(h, dx))
        population = all_candidates[:population_size]

        current_best_c5 = compute_c5(population[0], dx)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()

    # Local optimization
    if time.time() - start_time < budget_s * 0.99:  # Allow 1% buffer
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}

        result = minimize(
            fun=objective,
            x0=best_h,
            args=(n_points, dx),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 100, 'ftol': 1e-12, 'disp': False}
        )
        if result.success:
            optimized_h = result.x.reshape(n_points)
            optimized_h = np.clip(optimized_h, 0, 1)
            optimized_h_sum = optimized_h.sum()
            if abs(optimized_h_sum - target_sum) > 1e-6:
                optimized_h = optimized_h * (target_sum / optimized_h_sum)
            final_c5 = compute_c5(optimized_h, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = optimized_h

    # Final checks
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = best_h.sum()
    if abs(best_h_sum - target_sum) > 1e-6:
        best_h = best_h * (target_sum / best_h_sum)

    assert abs(best_h.sum() - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
