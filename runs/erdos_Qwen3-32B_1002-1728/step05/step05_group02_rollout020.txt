import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values).astype(np.float64)
    n_points = len(h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(x):
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    # Normalize initial h
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-6:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    bounds = [(0., 1.) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})

    np.random.seed(seed)
    perturb_count = 2
    noise_initial = 0.3

    while time.time() - start_time < budget_s - 30:
        remaining = budget_s - (time.time() - start_time)
        frac = remaining / budget_s
        noise_scale = noise_initial * (1 - frac ** 2)

        for _ in range(perturb_count):
            # Generate asymmetric perturbation
            perturbation = np.random.normal(0., noise_scale, n_points)
            candidate = np.clip(best_h + perturbation, 0, 1)
            current_sum = candidate.sum()
            if abs(current_sum) > 1e-10:  # Avoid division by near-zero
                candidate = candidate * (target_sum / current_sum)
            candidate = np.clip(candidate, 0, 1)

            method_idx = int(np.random.random() * 3)
            method = ['L-BFGS-B', 'SLSQP', 'COBYLA'][method_idx]

            result = minimize(
                fun=lambda x: compute_c5(x),
                x0=candidate,
                bounds=bounds,
                constraints=constraints,
                method=method,
                options={'maxiter': 40, 'ftol': 1e-10, 'disp': False}
            )

            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h[:] = new_h
                    best_c5 = new_c5

    assert abs(best_h.sum() - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), f'Bounds violation: {best_h.min()} - {best_h.max()}'
    
    return (best_h.tolist(), best_c5, n_points)
