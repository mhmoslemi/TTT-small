import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)

    NOISE_SCALE = 0.1
    MUTATION_RATE = 0.8
    TOURNAMENT_SIZE = 3

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb_individual(h, noise_scale=NOISE_SCALE):
        h_perturbed = h + np.random.normal(0, noise_scale, len(h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        total = np.sum(h_perturbed)
        if total != 0:
            h_perturbed *= target_sum / total
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed

    # Initialize population
    population = [h_initial.copy()]
    for _ in range(POP_SIZE - 1):
        population.append(perturb_individual(population[0], NOISE_SCALE))
    fitness = [evaluate(ind) for ind in population]

    np.random.seed(seed)
    generation = 0
    while time.time() - start_time < budget_s - 30:
        # Tournament selection

        # Crossover
