import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = np.sum(h_copy)
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    initial_h_half = h_full_initial[:half_n].copy()
    initial_h_half_normalized = normalize_h_half(initial_h_half, target_sum_half)

    np.random.seed(seed)

    # Initialize swarm
    swarm_size = 50
    positions = []
    velocities = []
    for _ in range(swarm_size):
        perturbed = initial_h_half_normalized + np.random.normal(0, 0.05, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        normalized = normalize_h_half(perturbed, target_sum_half)
        positions.append(normalized)
        velocities.append(np.zeros(half_n))

    personal_best_positions = [p.copy() for p in positions]
    personal_best_values = [np.inf] * swarm_size

    global_best_value = np.inf
    global_best_position = None

    while time.time() - start_time < budget_s - 30:
        c5_values = []
        for i in range(swarm_size):
            h_half = positions[i]
            h_full = create_full_h(h_half)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            c5_values.append(c5)
            if c5 < personal_best_values[i]:
                personal_best_positions[i] = h_half.copy()
                personal_best_values[i] = c5

        current_global_c5 = min(c5_values)
        if current_global_c5 < global_best_value:
            global_best_value = current_global_c5
            global_best_position = positions[np.argmin(c5_values)].copy()

        current_time = time.time() - start_time
        fraction = current_time / budget_s
        inertia_weight = 0.7 - 0.6 * fraction

        c1 = 2.0
        c2 = 2.0

        for i in range(swarm_size):
            r1 = np.random.rand()
            r2 = np.random.rand()
            personal_diff = personal_best_positions[i] - positions[i]
            global_diff = global_best_position - positions[i]
            velocities[i] = (inertia_weight * velocities[i] +
                            c1 * r1 * personal_diff +
                            c2 * r2 * global_diff)
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)

            new_position = positions[i] + velocities[i]
            new_position_clipped = np.clip(new_position, 0, 1)
            new_position_normalized = normalize_h_half(new_position_clipped, target_sum_half)
            positions[i] = new_position_normalized

    h_full_best = create_full_h(global_best_position)
    corr = np.correlate(h_full_best, 1 - h_full_best, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full_best) - n_points / 2.0) < 1e-6
    assert np.all(h_full_best <= 1.0 + 1e-6) and np.all(h_full_best >= -1e-6)
    return (h_full_best.tolist(), final_c5, n_points)
