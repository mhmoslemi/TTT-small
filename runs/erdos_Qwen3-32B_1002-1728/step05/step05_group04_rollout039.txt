import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_0 = np.array(initial_h_values)
    n_points = len(h_0)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    m = int(target_sum)
    
    def to_binary(h, m):
        indices = np.argsort(h)[::-1]
        bin_h = np.zeros(n_points, dtype=np.float64)
        bin_h[indices[:m]] = 1.0
        return bin_h

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def tournament(pop, k=3):
        candidates = np.random.choice(pop, k, replace=False)
        min_idx = np.argmin([evaluate_c5(c) for c in candidates])
        return candidates[min_idx]

    def crossover(p1, p2):
        idx1 = np.where(p1 == 1)[0]
        idx2 = np.where(p2 == 1)[0]
        split = np.random.randint(1, len(idx1))
        child1 = np.zeros(n_points)
        child1[np.unique(np.sort(np.concatenate([idx1[:split], idx2[split:]])))] = 1
        child1 = np.random.permutation(np.c_[np.where(child1 == 1)]).flatten()[:m]
        child1 = np.zeros(n_points)
        child1[child1] = 1.0
        return child1

    def mutate(c):
        ones = np.random.choice(np.where(c == 1)[0], 1)
        zeros = np.random.choice(np.where(c == 0)[0], 1)
        new = c.copy()
        new[ones] = 0.0
        new[zeros] = 1.0
        return new

    np.random.seed(seed + 1)
    h_init = to_binary(h_0, m)
    pop_size = 50
    population = [h_init.copy()]
    for _ in range(pop_size - 1):
        perm = np.random.permutation(n_points)[:m]
        indiv = np.zeros(n_points)
        indiv[perm] = 1.0
        population.append(indiv.copy())

    best_c5 = evaluate_c5(h_init)
    best_h = h_init.copy()

    while time.time() - start_time < budget_s - 5:
        next_pop = []
        for _ in range(pop_size // 2):
            p1 = tournament(population)
            p2 = tournament(population)
            c1 = crossover(p1, p2)
            c2 = crossover(p2, p1)
            next_pop.extend([mutate(c1), mutate(c2)])
        population = next_pop            
        
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        if time.time() - start_time > budget_s:
            break

    return (best_h.tolist(), best_c5, n_points)
