import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_full(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Initialize population with symmetric and asymmetric variations
    population_size = 20
    population = []
    h = h_initial * (target_sum / np.sum(h_initial))  # Ensure sum constraint
    h = np.clip(h, 0, 1)
    population.append(h)
    np.random.seed(seed)
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, 0.1, n_points)
        h_pert = h_initial + perturbation
        h_pert = np.clip(h_pert, 0, 1)
        current_sum = np.sum(h_pert)
        if current_sum != 0:
            h_pert *= (target_sum / current_sum)
        h_pert = np.clip(h_pert, 0, 1)
        population.append(h_pert)
    population = np.stack(population)

    best_h = population[0].copy()
    best_c5 = objective_full(best_h)

    # Evolutionary strategy loop
    while time.time() - start_time < budget_s - 60:
        # Evaluate all in population
        c5_values = np.array([objective_full(p) for p in population])
        current_best_idx = np.argmin(c5_values)
        if c5_values[current_best_idx] < best_c5:
            best_c5 = c5_values[current_best_idx]
            best_h = population[current_best_idx].copy()

        # Select top half as parents
        top_k = population_size // 2
        top_idx = np.argsort(c5_values)[:top_k]
        parents = population[top_idx]

        # Generate new population by mutating top parents
        new_pop = []
        frac = (budget_s - 60 - (time.time() - start_time)) / (budget_s - 60)
        noise_scale = 0.3 * (1 - frac**2)  # Decreasing scale

            parent = parents[np.random.randint(len(parents))]
            mutation = parent + np.random.normal(0, noise_scale, n_points)

            current_sum = np.sum(mutation)
            if current_sum != 0:
                mutation *= (target_sum / current_sum)
            mutation = np.clip(mutation, 0, 1)
            new_pop.append(mutation)
        population = np.stack(new_pop)

    # Local optimization
    if time.time() - start_time < budget_s - 30:
        def obj(x):
            return objective_full(x)
        bounds = [(0, 1) for _ in range(n_points)]
