import numpy as np
import time
from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    rng = default_rng(seed)
    
    def fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def process_h(h, target_sum):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total != 0:
            h = h * (target_sum / total)
        h = np.clip(h, 0, 1)
        return h
    
    population_size = 100
    elite_size = 5
    mutation_rate = 0.1
    pop = []
    for _ in range(population_size):
        h = h_initial.copy()
        h += 0.1 * rng.normal(size=h.shape)
        h = process_h(h, target_sum)
        pop.append(h)
    
    best_h = h_initial.copy()
    best_c5 = float('inf')
    
    while time.time() - start_time < budget_s:
        # Evaluate and update best
        for h in pop:
            c5 = fitness(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h[:] = h[:]
        
        # Select elites
        scores = [fitness(h) for h in pop]
        elite_indices = np.argpartition(scores, elite_size)[:elite_size]
        elites = [pop[i] for i in elite_indices]
        
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            p1, p2 = [pop[i] for i in np.argsort(rng.permutation(scores))[:2]]
            cross_point = rng.integers(1, n_points)
            child = np.concatenate((p1[:cross_point], p2[cross_point:]))
            child = process_h(child, target_sum)
            
            if rng.random() < mutation_rate:
                mut_idx = rng.choice(n_points, size=(10,), replace=False)
                noise = 0.05 * rng.normal(size=len(mut_idx))
                child[mut_idx] += noise
                child = process_h(child, target_sum)
            
            new_pop.append(child)
        pop = new_pop
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
