import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum != 0:
            h = (h / current_sum) * target_sum
        h = np.clip(h, 0, 1)
        return h

    h_initial = repair_h(h_initial)

    def evaluate(indiv):
        h = repair_h(indiv)
        corr_vals = np.correlate(h, 1 - h, mode='full')
        return np.max(corr_vals) * dx

    pop_size = 20
    generations = 30
    mutation_rate = 0.15
    mutation_decay = 0.95
    pop = [h_initial + np.random.normal(0, 0.1, n_points) for _ in range(pop_size)]
    pop = np.array([repair_h(indiv) for indiv in pop])
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial.copy()

    for gen in range(generations):
        if time.time() - start_time >= budget_s - 30:
            break
        fit = np.array([evaluate(indiv) for indiv in pop])
        best_idx = np.argmin(fit)
        if fit[best_idx] < best_c5:
            best_c5 = fit[best_idx]
            best_h = pop[best_idx].copy()
        fit = 1 / (1 + fit)  # Maximize fitness for selection
        sel_prob = fit / fit.sum()
        parents_idx = np.random.choice(pop_size, size=pop_size, p=sel_prob)
        parents = pop[parents_idx]
        offspring = []

        for i in range(0, pop_size, 2):
            p1, p2 = parents[i], parents[i+1] if i+1 < pop_size else parents[0]
            cx_point = np.random.randint(1, n_points)
            child = np.concatenate((p1[:cx_point], p2[cx_point:]))
            child = repair_h(child)

                mask = np.random.rand(n_points) < mutation_rate
                noise = np.random.normal(0, 0.1, n_points)
                child[mask] += noise[mask]
                child = repair_h(child)

    # Local refinement
