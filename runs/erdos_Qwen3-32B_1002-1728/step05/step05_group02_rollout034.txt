import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0
    np.random.seed(seed)

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def generate_random_individual():
        h_half = np.random.rand(half_n)
        h_half = np.clip(h_half, 0, 1)
        sum_h = np.sum(h_half)
        if sum_h == 0:
            h_half = np.full_like(h_half, target_sum_half / half_n)
        else:
            h_half = h_half * (target_sum_half / sum_h)
        h_half = np.clip(h_half, 0, 1.0)
        return h_half

    population = []
    # First individual from initial_h_values
    h_half_initial = h_full_initial[:half_n].copy()
    sum_initial = np.sum(h_half_initial)
    if sum_initial == 0:
        h_half_initial = np.full_like(h_half_initial, target_sum_half / half_n)
    else:
        h_half_initial = h_half_initial * (target_sum_half / sum_initial)
    h_half_initial = np.clip(h_half_initial, 0, 1.0)
    population.append(h_half_initial)
    # Generate other individuals
    for _ in range(29):  # 30 total
        population.append(generate_random_individual())

    best_c5 = float('inf')
    best_h = None

    tournament_size = 5
    mutation_rate = 0.1

    while time.time() - start_time < budget_s:
        # Evaluate all individuals
        try:
            population_obj = [compute_c5(ind) for ind in population]
        except:
            break
        valid_indices = [i for i, c5 in enumerate(population_obj) if np.isfinite(c5)]
        if not valid_indices:
            continue
        current_best_idx = valid_indices[np.argmin([population_obj[i] for i in valid_indices])]
        current_c5 = population_obj[current_best_idx]

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = create_full_h(population[current_best_idx])

        # Select new population using tournament selection and crossover/mutation
        new_population = []
        for _ in range(len(population)):
            # Tournament selection for parent1
            indices = np.random.choice(len(population), tournament_size)
            candidate_c5 = [population_obj[i] for i in indices]
            best_tournament_idx = np.argmin(candidate_c5)
            parent1 = population[indices[best_tournament_idx]]
            # Tournament selection for parent2
            indices2 = np.random.choice(len(population), tournament_size)
            candidate_c52 = [population_obj[i] for i in indices2]
            best_tournament_idx2 = np.argmin(candidate_c52)
            parent2 = population[indices2[best_tournament_idx2]]
            # Crossover
            child = (parent1 + parent2) / 2.0
            # Mutation
            noise = np.random.randn(half_n) * mutation_rate
            child += noise
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child == 0:
                child = np.full_like(child, target_sum_half / half_n)
            else:
                child = child * (target_sum_half / sum_child)
            child = np.clip(child, 0, 1)
            new_population.append(child)

        population = new_population

        # Local optimization on top candidates if time remains
        elapsed = time.time() - start_time
        if elapsed < budget_s * 0.9:
            top_indices = np.argsort(population_obj)[-max(1, len(population_obj)//10):]
            for idx in top_indices:
                h_candidate = population[idx]
                def objective(x):
                    return compute_c5(x)
                bnds = [(0.0, 1.0) for _ in h_candidate]
                cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
                result = minimize(
                    fun=objective,
                    x0=h_candidate.copy(),
                    method='SLSQP',
                    bounds=bnds,
                    constraints=cons,
                    options={'maxiter': 30, 'ftol': 1e-12}
                )
                if result.success:
                    optimized_h = np.clip(result.x, 0, 1)
                    sum_optim = np.sum(optimized_h)
                    if sum_optim != 0:
                        optimized_h = optimized_h * (target_sum_half / sum_optim)
                        optimized_h = np.clip(optimized_h, 0, 1)
                    population[idx] = optimized_h

        # Adjust mutation rate based on remaining time
        frac_remaining = (budget_s - (time.time() - start_time)) / budget_s
        mutation_rate = 0.1 * (1 - frac_remaining**2)

    # Final check for best solution
    try:
        final_population_obj = [compute_c5(ind) for ind in population]
    except:
        pass
    valid_indices = [i for i, c5 in enumerate(final_population_obj) if np.isfinite(c5)]
    if valid_indices:
        final_best_idx = valid_indices[np.argmin([final_population_obj[i] for i in valid_indices])]
        final_c5 = final_population_obj[final_best_idx]
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = create_full_h(population[final_best_idx])

    return (best_h.tolist(), best_c5, n_points)
