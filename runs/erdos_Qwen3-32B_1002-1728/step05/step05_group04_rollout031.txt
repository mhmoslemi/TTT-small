import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)

    # GA parameters
    pop_size = 50
    max_generations = 100
    mut_rate = 0.01
    mut_strength = 0.2

    # Initialize population
    np.random.seed(seed)
    population = []
    for _ in range(pop_size):
        h = h_initial.copy()
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = h * target_sum / h_sum
        h = np.clip(h, 0, 1)
        population.append(h)
    population = np.array(population)

    # Evaluate initial population
    fitness = np.array([objective(ind) for ind in population])
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # GA loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 120:
            break

        offspring = []
        for _ in range(pop_size):
            # Tournament selection
            parent_idx = [np.random.randint(pop_size) for _ in range(5)]
            parent_scores = fitness[parent_idx]
            winner_idx = parent_idx[np.argmin(parent_scores)]
            parent1 = population[winner_idx]

            parent_idx2 = [np.random.randint(pop_size) for _ in range(5)]
            parent_scores2 = fitness[parent_idx2]
            winner_idx2 = parent_idx2[np.argmin(parent_scores2)]
            parent2 = population[winner_idx2]

            # Blend crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)

            # Mutation
            mask = np.random.rand(n_points) < mut_rate
            noise = np.random.normal(0, mut_strength, n_points)
            child[mask] += noise[mask]
            child = np.clip(child, 0, 1)

            total = child.sum()
            if total != 0:
                child *= target_sum / total
            child = np.clip(child, 0, 1)
            offspring.append(child)

        offspring = np.array(offspring)
        new_fitness = np.array([objective(ind) for ind in offspring])
        fits = np.concatenate([fitness, new_fitness])
        pop = np.vstack([population, offspring])
        order = np.argsort(fits)
        population = pop[order[:pop_size]]
        fitness = fits[order[:pop_size]]

        new_idx = np.argmin(fitness)
        if fitness[new_idx] < best_c5:
            best_h = population[new_idx].copy()
            best_c5 = fitness[new_idx]

    # Local refinement
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    method = 'SLSQP'
    result = minimize(

        x0=best_h,
        method=method,
        bounds=bounds,
        options={'maxiter': 200, 'ftol': 1e-9},
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]

    if result.success:
        new_h = np.clip(result.x, 0, 1)

        new_h = np.clip(new_h, 0, 1)

    if abs(total - target_sum) > 1e-6:
        best_h = best_h * target_sum / total
    return (best_h.tolist(), best_c5, n_points)
