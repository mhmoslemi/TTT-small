        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            particles[i] = particles[i] + velocities[i]
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            sum_p = np.sum(particles[i])
            if sum_p != 0:
                particles[i] = particles[i] * (target_sum / sum_p)
            particles[i] = np.clip(particles[i], 0, 1)
        # Evaluate and update bests
        current_values = np.array([objective_func(p) for p in particles])
        for i in range(n_particles):
            if current_values[i] < personal_best_values[i]:
                personal_best_values[i] = current_values[i]
                personal_best_positions[i] = particles[i].copy()
        new_global_idx = np.argmin(personal_best_values)
        if personal_best_values[new_global_idx] < global_best_value:
            global_best_value = personal_best_values[new_global_idx]
            global_best_position = personal_best_positions[new_global_idx].copy()

    # Local refinement with SLSQP
    best_h = global_best_position.copy()
    cons_refine = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds_refine = [(0.0, 1.0) for _ in range(n_points)]

    try:
        result_refine = minimize(
            fun=lambda x: objective_func(x),
            x0=best_h,
            method='SLSQP',
            bounds=bounds_refine,
            constraints=cons_refine,
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        if result_refine.success:
            best_h = np.clip(result_refine.x, 0, 1)
            best_c5 = objective_func(best_h)
        else:
            best_c5 = global_best_value
    except:
        best_c5 = global_best_value

    # Final constraint check
    h_full = best_h.copy()
    h_full = np.clip(h_full, 0, 1)
    sum_h = np.sum(h_full)
    if abs(sum_h - target_sum) > 1e-6:
        h_full = h_full * (target_sum / sum_h)
        h_full = np.clip(h_full, 0, 1)
    return (h_full.tolist(), best_c5, n_points)
