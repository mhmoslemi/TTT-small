        x_vals = np.array(x)
        x_clipped = np.clip(x_vals, 0, 1)
        current_sum = np.sum(x_clipped)
        if current_sum == 0:
            x_proj = np.zeros_like(x_vals)
        else:
            x_proj = x_clipped * target_sum / current_sum
        x_proj = np.clip(x_proj, 0, 1)
        corr_vals = np.correlate(x_proj, 1 - x_proj, mode='full')
        return np.max(corr_vals) * dx

    def objective_local(x):
        x = np.clip(x, 0, 1)
        corr_vals = np.correlate(x, 1 - x, mode='full')
        return np.max(corr_vals) * dx

    def constraint_eq(x):
        return np.sum(x) - target_sum

    h_best = h_full_initial.copy()
    h_best = np.clip(h_best, 0, 1)
    current_sum = np.sum(h_best)
    if current_sum != 0:
        h_best = (h_best / current_sum) * target_sum
    h_best = np.clip(h_best, 0, 1)
    best_c5 = objective_local(h_best)

    # Differential Evolution phase
    de_budget = 0.35 * budget_s
    np.random.seed(seed)
    try:
        de_result = differential_evolution(
            objective_de,
            [(0.0, 1.0)] * n_points,
            strategy='best1bin',
            maxiter=6,
            popsize=10,
            mutation=(0.5, 1),
            recombination=0.9,
            seed=seed,
            workers=1,
            tol=1e-4,
            disp=False
        )
        if de_result.success:
            h_candidate = np.clip(de_result.x, 0, 1)
            c5_candidate = objective_local(h_candidate)
            if c5_candidate < best_c5:
                h_best = h_candidate
                best_c5 = c5_candidate
    except Exception:
        pass

    # Local optimization with multiple restarts
    def run_opt_from_point(x0):
        result = minimize(
            objective_local,
            x0=np.clip(x0, 0, 1),
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[{'type': 'eq', 'fun': constraint_eq}],
            options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            return h_opt, objective_local(h_opt)
        else:
            return x0, objective_local(x0)

    def perturb(x, noise_scale=0.05):
        x_pert = x + np.random.normal(0, noise_scale, n_points)
        x_pert = np.clip(x_pert, 0, 1)
        total = np.sum(x_pert)
        if total != 0:
            x_pert = (x_pert / total) * target_sum
        return np.clip(x_pert, 0, 1)

    # Run local optimization on DE candidate
    h_best, best_c5 = run_opt_from_point(h_best)

    # Perturbation and optimization loop
    np.random.seed(seed + 1)
    trials = 0
    while time.time() - start_time < budget_s - 10:
        trials += 1
        if trials % 2 == 0:  # alternate between mild and strong noise
            noise_scale = 0.05 if trials % 4 != 0 else 0.1

        x_pert = perturb(h_best, noise_scale=noise_scale)
        x_opt, c_opt = run_opt_from_point(x_pert)

        if c_opt < best_c5:
            h_best = x_opt
            best_c5 = c_opt

    h_best = np.clip(h_best, 0, 1)
    assert abs(np.sum(h_best) - target_sum) < 1e-6
    return (h_best.tolist(), best_c5, n_points)
