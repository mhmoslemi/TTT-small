from scipy.optimize import minimize
import time

    def objective(h_vars):
        h_full = np.clip(h_vars, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint_sum(h_vars):
        return np.sum(h_vars) - target_sum

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Normalize initial guess to meet the sum constraint
    current_sum = np.sum(h_full_initial)
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    best_h = h_full_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time

            perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
