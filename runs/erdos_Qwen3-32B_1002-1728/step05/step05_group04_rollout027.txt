import time

        return np.max(corr) * dx

    if total != 0:
        scale = target_sum / total
        h_full = h_full * scale

    best_h = h_full.copy()
    best_c5 = objective(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        current_time = max(1e-6, time.time() - start_time)

        if total_perturbed != 0:
            perturbed *= (target_sum / total_perturbed)
        perturbed = np.clip(perturbed, 0, 1)

        method = 'COBYLA'
        result = minimize(
