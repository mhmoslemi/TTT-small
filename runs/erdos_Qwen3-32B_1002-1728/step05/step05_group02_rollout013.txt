                h = (h / s) * target_sum
            h = np.clip(h, 0, 1)
            new_population.append(h)
        population = new_population

    # Final optimization
    def objective(x):
        x = x.reshape(-1)
        h = np.clip(x, 0, 1)
        s = h.sum()
        if s != 0:
            h = (h / s) * target_sum
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Constraint: average value constraint (indirect sum constraint)
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    bounds = [(0, 1)] * n_points

    result = minimize(
        fun=objective,
        x0=best_h.flatten(),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-14}
    )

    if result.success:
        optimized_h = np.clip(result.x, 0, 1)
        s = optimized_h.sum()
        if s != 0:
            optimized_h = (optimized_h / s) * target_sum
        optimized_h = np.clip(optimized_h, 0, 1)
        c5_opt = evaluate_candidate(optimized_h)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = optimized_h

    assert abs(best_h.sum() - target_sum) < 1e-6
    assert all(0 <= x <= 1 for x in best_h.tolist())
    return (best_h.tolist(), best_c5, n_points)
