import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Half vector sum for symmetry

    # Initialize symmetric and normalized h_half
    h_half = (h_initial[:half_n] + h_initial[-half_n:][::-1]) / 2.0
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = h_half[i]
    sum_h = np.sum(h_full)
    if sum_h != 0:
        h_full *= (n_points / 2.0) / sum_h
    h_full = np.clip(h_full, 0.0, 1.0)
    h_half = h_full[:half_n].copy()

    np.random.seed(seed)
    learning_rate = 5e-3
    decay_rate = 0.995
    max_iterations = 1000000
    best_c5 = np.inf
    best_h_half = h_half.copy()

    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s:
            break
        iteration += 1

        # Construct symmetric h_full
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        h_full = np.clip(h_full, 0.0, 1.0)  # Ensure no boundaries violated
        sum_h = np.sum(h_full)
        if sum_h != 0:
            h_full *= (n_points / 2.0) / sum_h

        # Compute overlaps
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_c5 = np.max(corr)
        max_index = np.argmax(corr)
        k = max_index - (n_points - 1)

        # Compute gradient for k
        grad_full = np.zeros(n_points)
        for i in range(n_points):
            i_plus_k = i + k
            i_minus_k = i - k
            term1 = (1 - h_full[i_plus_k]) * dx if 0 <= i_plus_k < n_points else 0.0
            term2 = (-h_full[i_minus_k] * dx) if 0 <= i_minus_k < n_points else 0.0
            grad_full[i] = term1 + term2

        # Project gradient to h_half
        grad_half = np.zeros(half_n)
        for j in range(half_n):
            grad_half[j] = grad_full[j] + grad_full[n_points - 1 - j]

        # Gradient step
        h_half -= learning_rate * grad_half

        # Feasibility projection: clip and normalize
        h_half = np.clip(h_half, 0.0, 1.0)
        sum_h_half = np.sum(h_half)
        if sum_h_half != 0:
            h_half *= target_sum_half / sum_h_half
        h_half = np.clip(h_half, 0.0, 1.0)

        # Early stopping for best
        if current_c5 < best_c5:

            best_h_half = h_half.copy()
        learning_rate *= decay_rate

    h_full = np.zeros(n_points)

        h_full[i] = h_full[n_points - 1 - i] = best_h_half[i]
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    return (h_full.tolist(), final_c5, n_points)
