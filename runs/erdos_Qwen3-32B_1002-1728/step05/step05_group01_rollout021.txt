    n_points = len(h_initial)
    dx = 2.0 / n_points
    T = n_points / 2.0  # because integral must be 1

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def objective(h_vars):
        # Make a copy and normalize
        sum_h = np.sum(h_vars)
        if sum_h == 0:
            return float('inf')  # avoid division by zero
        h_scaled = h_vars * (T / sum_h)
        h_clipped = np.clip(h_scaled, 0.0, 1.0)
        new_sum = np.sum(h_clipped)
        if np.abs(new_sum - T) > 1e-6:
            # Penalize large discrepancies from the target sum
            return float('inf')
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    # Run differential evolution with early stopping based on time
    def stop_fn(xk, convergence):
        return time.time() - start_time >= budget_s - 20

        func=objective,

        strategy='best1bin',

        popsize=10,   # 10 * 70 = 700 population
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        callback=stop_fn,
        polish=False,
        workers=1

    h_opt = result.x

        h_scaled = h_vars * (T / sum_h)
        h_clipped = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
