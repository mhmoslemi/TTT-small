    personal_best_scores = np.array([evaluate_objective(p) for p in particles_pos])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_pos = particles_pos[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # PSO iterations
    for iteration in range(n_iterations):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (personal_best_pos[i] - particles_pos[i]) +
                             social * r2 * (global_best_pos - particles_pos[i]))
            particles_pos[i] = np.clip(particles_pos[i] + velocities[i], 0, 1)

        current_scores = np.array([evaluate_objective(p) for p in particles_pos])
        improved = current_scores < personal_best_scores
        personal_best_pos[improved] = particles_pos[improved].copy()
        personal_best_scores[improved] = current_scores[improved]
        current_global_idx = np.argmin(personal_best_scores)
        if personal_best_scores[current_global_idx] < global_best_score:
            global_best_pos = personal_best_pos[current_global_idx].copy()
            global_best_score = personal_best_scores[current_global_idx]

    # Local optimization
    x0 = global_best_pos.copy()
    sum_x0 = np.sum(x0)
    if sum_x0 == 0:
        x0 = np.ones_like(x0) * target_sum / n_points
    else:
        scaling = target_sum / sum_x0
        x0 = np.clip(x0 * scaling, 0, 1)
        sum_x0_clipped = np.sum(x0)
        if sum_x0_clipped != 0:
            scaling_final = target_sum / sum_x0_clipped
            x0 = np.clip(x0 * scaling_final, 0, 1)

    def objective_for_scipy(x):
        x = x.reshape(n_points)
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    lin_const = LinearConstraint(np.ones(n_points), lb=[target_sum], ub=[target_sum])

    if time.time() - start_time < budget_s - 10:
        try:
            result = minimize(
                fun=objective_for_scipy,
                x0=x0,
                method='SLSQP',
                bounds=bounds,
                constraints=lin_const,
                options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                optimized_h = np.clip(result.x, 0, 1)
                scaling_final = target_sum / np.sum(optimized_h)
                optimized_h = np.clip(optimized_h * scaling_final, 0, 1)
                c5 = evaluate_objective(optimized_h)
                if c5 < global_best_score:
                    global_best_pos = optimized_h.copy()
                    global_best_score = c5
        except Exception:
            pass

    # Final check and return
    h_final = global_best_pos
    h_final = np.clip(h_final, 0, 1)
    sum_h_final = np.sum(h_final)
    if abs(sum_h_final - target_sum) > 1e-6:
        scaling = target_sum / sum_h_final
        h_final = np.clip(h_final * scaling, 0, 1)
    assert abs(np.sum(h_final) - target_sum) < 1e-6, "Feasibility check failed"

    return (h_final.tolist(), global_best_score, n_points)
