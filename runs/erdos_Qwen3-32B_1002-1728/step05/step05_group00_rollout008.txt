from scipy.optimize import minimize, SLSQP

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points

    def create_full_h(h_half):
        return np.concatenate((h_half, h_half[::-1]))

    def projection(h_half):
        h_half_clipped = np.clip(h_half, 0, 1)
        total = np.sum(h_half_clipped)
        if total != 0:
            h_half_clipped = (h_half_clipped / total) * target_sum
        return np.clip(h_half_clipped, 0, 1)

    def obj(h_half):
        h_full = create_full_h(projection(h_half))
        corr_vals = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr_vals) * dx

    # Initialize from parent construction with perturbation

    h_half = projection(h_half_init)

    c2 = 1.2  # Social

    pbest_cost = [obj(p) for p in pbest_pos]
