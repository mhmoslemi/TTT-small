import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        corr = np.correlate(h_vars, 1 - h_vars, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    total_iter = 0

    while time.time() - start_time < budget_s - 15:
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        noise_scale = 0.3 * (1 - frac**0.75)

        # Perturb full vector asymmetrically
        perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
        perturbed_clipped = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed_clipped)

        if abs(current_sum) > 1e-10:
            perturbed_scaled = perturbed_clipped * (target_sum / current_sum)
            perturbed_scaled = np.clip(perturbed_scaled, 0, 1)
        else:
            perturbed_scaled = perturbed_clipped

        # Try optimization with L-BFGS-B and COBYLA
        for method in ['L-BFGS-B', 'COBYLA']:
            result = minimize(
                fun=objective,
                x0=perturbed_scaled,
                method=method,
                bounds=bounds_per_var,
                constraints=cons if method != 'COBYLA' else [],
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_sum = np.sum(new_h)
                if abs(new_sum - target_sum) < 1e-6:
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
                        break  # Stop if improved

        # Occasionally try global SLSQP with adaptive noise
        if np.random.rand() < 0.1:
            global_perturbed = best_h * (1 + np.random.normal(0, 0.1, n_points))
            global_perturbed = np.clip(global_perturbed, 0, 1)
            global_perturbed = global_perturbed * (target_sum / np.sum(global_perturbed))
            result = minimize(
                fun=objective,
                x0=global_perturbed,
                method='SLSQP',
                bounds=bounds_per_var,
                constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-8}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_sum = np.sum(new_h)
                if abs(new_sum - target_sum) < 1e-6:
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5

        total_iter += 1

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
