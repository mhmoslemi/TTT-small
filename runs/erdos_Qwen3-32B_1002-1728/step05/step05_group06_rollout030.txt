import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Symmetrize and normalize initial h
    h_full = h_initial.copy()
    for i in range(half_n):
        avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
        h_full[i] = h_full[n_points - 1 - i] = avg
    current_sum = np.sum(h_full)
    if not np.isclose(current_sum, 0.0):
        h_full = h_full / current_sum * target_sum
    h_full = np.clip(h_full, 0.0, 1.0)
    
    h_half = h_full[:half_n].copy()
    best_h = h_full.copy()
    best_c5 = evaluate_erdos_solution().evaluate(h_full)

    def compute_max_overlap(h_vec):
        n = len(h_vec)
        overlaps = []
        for k in range(n):
            valid = n - k
            if valid <= 0:
                continue
            part1 = h_vec[:valid]
            part2 = h_vec[k:k+valid]
            overlap = np.sum(part1 * (1 - part2)) * dx
            overlaps.append(overlap)
        return max(overlaps), overlaps

    def find_max_k(overlaps):
        max_ov = max(overlaps)
        for k, ov in enumerate(overlaps):
            if np.isclose(ov, max_ov):
                return k
        return 0

    def compute_gradient(h_vec, k):
        grad = np.zeros_like(h_vec, dtype=float)
        for m in range(len(h_vec)):
            if m + k < len(h_vec):
                grad[m] += 1.0 - h_vec[m + k]
            if m >= k:
                grad[m] -= h_vec[m - k]
        grad *= dx
        return grad

    np.random.seed(seed)
    learning_rate = 0.01
    lr_decay = 0.998

    while time.time() - start_time < budget_s:
        overlaps = compute_max_overlap(h_full)[1]
        max_k = find_max_k(overlaps)
        grad_full = compute_gradient(h_full, max_k)
        grad_half = 2 * grad_full[:half_n]

        # Update half and normalize
        h_half -= learning_rate * grad_half
        h_half = np.clip(h_half, 0.0, 1.0)
        target_sum_half = target_sum / 2.0
        current_sum_half = np.sum(h_half)
        if not np.isclose(current_sum_half, 0):
            h_half = h_half / current_sum_half * target_sum_half

        # Build new full h
        new_h_full = np.zeros_like(h_full)
        for i in range(half_n):
            new_h_full[i] = new_h_full[n_points - 1 - i] = h_half[i]
        new_h_full = np.clip(new_h_full, 0.0, 1.0)

        # Final normalization
        current_sum_new = np.sum(new_h_full)
        if not np.isclose(current_sum_new, 0):
            new_h_full = new_h_full / current_sum_new * target_sum
        new_h_full = np.clip(new_h_full, 0.0, 1.0)
        h_full = new_h_full

        current_c5 = evaluate_erdos_solution().evaluate(h_full)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_full.copy()
        learning_rate *= lr_decay

    return (best_h.tolist(), best_c5, n_points)
