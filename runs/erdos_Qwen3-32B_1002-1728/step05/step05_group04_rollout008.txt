from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):

    h_full = np.clip(h_full, 0, 1)

    best_h = h_full.copy()

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    # Generate diverse starting points
    np.random.seed(seed)
    for _ in range(3):  # Add 3 random binary-based h inits

        result = minimize(

                if new_c5 < best_c5:
