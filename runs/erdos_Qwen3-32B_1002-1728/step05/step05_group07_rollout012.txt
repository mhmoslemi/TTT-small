    current_sum = np.sum(h_half)
    if current_sum != 0:
        h_half *= target_sum_half / current_sum
    h_half = np.clip(h_half, 0, 1)

    # Best solution tracking
    best_h_half = h_half.copy()

    def get_h_full(h_half_in):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half_in[i]
        return h_full

    def compute_c5(h_half_in):
        h_full = get_h_full(h_half_in)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx

    momentum = 0.9

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # sum(h_full) = n_points/2 => sum(h_half) = n_points/4
    dx = 2.0 / n_points

    # Initialize symmetric h_half
    h_half = np.zeros(half_n)
    for i in range(half_n):
        h_half[i] = (h_initial[i] + h_initial[n_points - 1 - i]) / 2.0
    current_sum = np.sum(h_half)
    if current_sum != 0:
        h_half *= target_sum_half / current_sum
    h_half = np.clip(h_half, 0, 1)

    best_h_half = h_half.copy()

    def get_h_full(h_half_in):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half_in[i]
        return h_full

    def compute_c5(h_half_in):
        h_full = get_h_full(h_half_in)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    best_c5 = compute_c5(best_h_half)
    best_c5_time = start_time

    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros_like(h_half)
    iter = 0

    while time.time() - start_time < budget_s - 1:

        h_full = get_h_full(h_half)

        current_c5 = np.max(corr)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half[:] = h_half.copy()
            best_c5_time = time.time()

        worst_k = np.argmax(corr)

        # Compute gradients for the worst k
        grad_full = np.zeros(n_points)

            j = (i + worst_k) % n_points
