import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)

    population_size = 50
    population = np.zeros((population_size, n_points))
    for i in range(population_size):
        new_h = h_full_initial.copy() + np.random.normal(0, 0.1, n_points)
        total = new_h.sum()
        if total != 0:
            new_h *= (target_sum / total)
        population[i] = np.clip(new_h, 0, 1)

    best_h = population[0].copy()
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s - 60:
        # Evaluate population
        obj_values = np.array([objective(p) for p in population])
        top_idx = np.argsort(obj_values)[:int(population_size * 0.2)]
        parents = population[top_idx]

        # Generate next generation
        children = []
        for _ in range(population_size):
            idxs = np.random.choice(len(parents), 2)
            p1, p2 = parents[idxs[0]], parents[idxs[1]]
            split = np.random.randint(1, n_points)
            child = np.concatenate([p1[:split], p2[split:]])
            # Mutation
            mutate = np.random.rand(n_points) < 0.05
            child[mutate] += np.random.normal(0, 0.05, np.sum(mutate))
            # Normalize and clip
            total = np.sum(child)
            if total != 0:
                child = child * (target_sum / total)
            children.append(np.clip(child, 0, 1))
        population = np.array(children)

        # Update best
        curr_obj = np.array([objective(p) for p in population])
        best_idx = np.argmin(curr_obj)
        if curr_obj[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = curr_obj[best_idx]

    # Local refinement
    bounds = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result.success:
        refined = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined)
        if abs(refined_sum - target_sum) < 1e-6:
            final_obj = objective(refined)
            if final_obj < best_c5:
                best_c5 = final_obj
                best_h = refined

    return (best_h.tolist(), best_c5, n_points)
