            h[i] = (1 - t)**2 * p[idx] + 2 * t * (1 - t) * p[idx+1] + t**2 * p[idx+2]
        target = n_points / 2
        s = h.sum()
        if s > 1e-8:
            h *= target / s
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, 'full')
        return np.max(corr) * (2.0 / n_points), h

    def mutate_p(p, noise=0.1):
        p = p.copy()
        mask = np.random.rand(num_p) < 0.3
        p[mask] += np.random.normal(0, noise, size=np.sum(mask))
        return np.clip(p, 0, 1)

    def crossover(p1, p2):
        alpha = np.random.uniform(0.3, 0.7)
        return np.clip(alpha * p1 + (1-alpha) * p2, 0, 1)

    def init_p_from_h(h):
        p = np.zeros(num_p)
        dx_be = 2.0 / n_segments
        for i in range(num_p):
            x_be = dx_be * i
            t = 0.5
            idx_h = int((x_be + 0.5*dx_be) * n_points / 2)
            p[i] = h[idx_h]
        return np.clip(p, 0, 1)

    population = []
    if h_init is not None:
        p0 = init_p_from_h(h_init)
        c0, h0 = eval_h(p0)
        population.append(p0)

    for _ in range(pop_size - len(population)):
        population.append(np.random.uniform(0, 1, num_p))

    population = np.array(population)
    best_p = None
    best_c5 = float('inf')

    for gen in range(max_gen):
        if (time.time() - start_time) > budget_s - 60:
            break
            
        scores = []
        for p in population:
            c5, _ = eval_h(p)
            scores.append(c5)
        idx_top = np.argsort(scores)[:pop_size//2]
        top_p = population[idx_top]

        for i in range(pop_size - len(top_p)):
            if np.random.rand() < 0.5:
                p_new = mutate_p(np.random.choice(top_p, 1)[0])
            else:
                p1, p2 = top_p[np.random.choice(len(top_p), 2, replace=False)]
                p_new = mutate_p(crossover(p1, p2))
            top_p = np.vstack((top_p, p_new)) if i > 0 else np.vstack((p_new.reshape(1, -1)))
        population = top_p

        for p in population:
            c5, h = eval_h(p)
            if c5 < best_c5:
                best_c5 = c5
                best_p = p

    if best_p is None:
        best_p = np.random.uniform(0, 1, num_p)
        best_c5, h_out = eval_h(best_p)

    h_out = best_p
    c5_final, h_out = eval_h(best_p)

    return (h_out.tolist(), c5_final, n_points)
