import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Prepare initial h_full with correct constraints
    h_full_clipped = np.clip(h_full_initial, 0.0, 1.0)
    current_sum = h_full_clipped.sum()
    if current_sum > 1e-10:
        h_full_clipped *= target_sum / current_sum
    else:
        h_full_clipped[:] = 0.5
    best_h = h_full_clipped.copy()
    best_c5 = calculate_c5(h_full_clipped, dx)
    np.random.seed(seed)

    # PSO parameters
    num_particles = 20
    num_iterations = 20
    inertia = 0.8
    cognitive = 1.2
    social = 1.2

    def clip_and_rescale(vec):
        vec_clipped = np.clip(vec, 0.0, 1.0)
        total = vec_clipped.sum()
        if total > 1e-10:
            return vec_clipped * target_sum / total
        return np.full(n_points, target_sum / n_points)

    # Initialize particles
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    for i in range(num_particles):
        particles[i] = h_full_clipped.copy()
        particles[i] += np.random.normal(0, 0.1, n_points)
        particles[i] = clip_and_rescale(particles[i])
    personal_best_positions = particles.copy()
    personal_best_scores = [calculate_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # PSO loop
    for _ in range(num_iterations):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i]
                + cognitive * r1 * (personal_best_positions[i] - particles[i])
                + social * r2 * (global_best_position - particles[i])
            )
            particles[i] = clip_and_rescale(particles[i] + velocities[i])
            score = calculate_c5(particles[i], dx)
            if score < personal_best_scores[i]:
                personal_best_positions[i] = particles[i].copy()
                personal_best_scores[i] = score
                if score < global_best_score:
                    global_best_position[:] = particles[i]
                    global_best_score = score

    # Local refinement
    def objective(x):
        return calculate_c5(x, dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum})
    result = minimize(
        objective, 
        global_best_position, 
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        current_c5 = calculate_c5(result.x, dx)
        if current_c5 < best_c5:
            best_h[:] = result.x
            best_c5 = current_c5

    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)

def calculate_c5(h, dx):
    corr_vals = np.correlate(h, 1 - h, mode='full')
    return np.max(corr_vals) * dx
