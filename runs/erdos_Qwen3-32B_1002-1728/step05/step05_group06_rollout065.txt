import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    half_n = n_points // 2
    
    # Enforce symmetry and normalize initial h
    h = h_initial.copy()
    for i in range(half_n):
        avg = (h[i] + h[n_points - 1 - i]) / 2.0
        h[i] = h[n_points - 1 - i] = avg
    current_sum = np.sum(h)
    if current_sum != 0:
        h = (h / current_sum) * target_sum
    h = np.clip(h, 0.0, 1.0)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    # Hyperparameters
    learning_rate = 0.1  # initial learning rate
    lr_decay = 0.9995
    max_iterations = 10000
    
    np.random.seed(seed)
    
    while time.time() - start_time < budget_s:
        # Compute all overlaps
        n = len(h)
        overlaps = []
        for k in range(-n + 1, n):
            total = 0.0
            for i in range(n):
                j1 = i + k
                if 0 <= j1 < n:
                    total += h[i] * (1.0 - h[j1])
            overlaps.append(total)
        overlaps = np.array(overlaps)
        max_overlap = np.max(overlaps) * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Find k_max (corresponding to the original k)
        k_flattened = np.arange(-n + 1, n)  # indices correspond to overlap values
        k_max = k_flattened[np.argmax(overlaps)]
        
        # Compute gradient for this k_max
        g = np.zeros(n)
        k = k_max
        for i in range(n):
            j1 = i + k
            j2 = i - k
            if 0 <= j1 < n:
                g[i] += (1.0 - h[j1])
            if 0 <= j2 < n:
                g[i] += -h[j2]
        
        # Update h with gradient descent
        h -= learning_rate * g
        
        # Enforce symmetry again after gradient step (since the gradient may break it)
        for i in range(half_n):
            avg = (h[i] + h[n_points - 1 - i]) / 2.0
            h[i] = h[n_points - 1 - i] = avg
        
        # Clip to [0,1]
        h = np.clip(h, 0.0, 1.0)
        # Normalize
        current_sum = np.sum(h)
        if current_sum != 0:
            scale = target_sum / current_sum
            h = h * scale
            # Clip again after scaling
            h = np.clip(h, 0.0, 1.0)
        
        # Decay learning rate
        learning_rate *= lr_decay
        
        # Early stopping condition (if no improvement for a while)

    # Final compute of best_c5 to ensure it's up to date
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    return (best_h.tolist(), final_c5, n_points)
