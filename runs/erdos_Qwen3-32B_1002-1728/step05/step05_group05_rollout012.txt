    start_time = time.time()

    target_sum_half = n_points / 4.0  # Because full sum is n_points/2, and h_half is first half.

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    # Initialize h_half to be symmetric and normalized
    current_h_half = h_initial[:half_n].copy()
    current_h_full = create_full_h(current_h_half)
    # Normalize to have sum equal to target
    current_sum = np.sum(current_h_full)
    scale = target_sum_half / current_sum if not np.isclose(current_sum, 0) else 1.0
    current_h_full *= scale
    current_h_full = np.clip(current_h_full, 0, 1)
    # Update h_half
    current_h_half = current_h_full[:half_n].copy()

    best_c5 = np.inf
    best_h_full = current_h_full.copy()

    # Parameters for smoothed gradient
    tau = 0.1
    eta = 0.01  # learning rate
    max_iterations = int(1e5)
    warm_up = 100
    tau_decay = 0.999
    np.random.seed(seed)

    for iter in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break

        # Compute h_full from current h_half
        h_full = create_full_h(current_h_half)
        h_full = np.clip(h_full, 0, 1)
        # Normalize sum
        current_sum = np.sum(h_full)
        if not np.isclose(current_sum, n_points / 2.0):
            scale = (n_points / 2.0) / current_sum
            h_full *= scale
            h_full = np.clip(h_full, 0, 1)

        # Compute overlap values
        overlaps = np.correlate(h_full, 1 - h_full, mode='full') * dx
        # Compute smoothed max
        exp_overlaps = np.exp(overlaps / tau)
        sum_exp = np.sum(exp_overlaps)
        p = exp_overlaps / sum_exp
        smoothed_c5 = tau * np.log(sum_exp) if sum_exp > 0 else 0.0

        # Compute gradient_full
        grad_full = np.zeros(n_points)
        for idx in range(len(overlaps)):
            k_shift = idx - n_points + 1
            for m in range(n_points):
                # Compute term1: 1 - h_full[m + k_shift]
                i_plus = m + k_shift
                term1 = 0.0
                if 0 <= i_plus < n_points:
                    term1 = 1.0 - h_full[i_plus]
                # Compute term2: h_full[m - k_shift]
                i_minus = m - k_shift
                term2 = 0.0
                if 0 <= i_minus < n_points:
                    term2 = h_full[i_minus]
                grad_full[m] += p[idx] * dx * (term1 - term2)

        # Project gradient into h_half space
        grad_half = np.zeros(half_n)
        for i in range(half_n):
            mirrored_idx = n_points - 1 - i
            grad_half[i] = 0.5 * (grad_full[i] + grad_full[mirrored_idx])

        # Gradient ascent (to minimize, we use negative gradient)
        learning_rate = eta
        if iter < warm_up:
            learning_rate *= (iter + 1) / warm_up  # warm-up

        new_h_half = current_h_half - learning_rate * grad_half  # gradient descent

        # Clip to [0,1]
        new_h_half = np.clip(new_h_half, 0, 1)

        # Normalize to target sum_half
        current_sum_half = np.sum(new_h_half)
        if not np.isclose(current_sum_half, 0):

            new_h_half *= scale
        new_h_half = np.clip(new_h_half, 0, 1)

        h_full_candidate = create_full_h(current_h_half)
        corr = np.correlate(h_full_candidate, 1 - h_full_candidate, mode='full')
        c5 = np.max(corr) * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h_full = h_full_candidate.copy()

        # Decay tau
        tau *= tau_decay

    # Final clip
    best_h_full = np.clip(best_h_full, 0, 1)
    # Normalize
    best_sum = np.sum(best_h_full)
    if not np.isclose(best_sum, n_points / 2.0):
        scale = (n_points / 2.0) / best_sum
        best_h_full *= scale
    best_h_full = np.clip(best_h_full, 0, 1.0)

    return (best_h_full.tolist(), best_c5, n_points)
