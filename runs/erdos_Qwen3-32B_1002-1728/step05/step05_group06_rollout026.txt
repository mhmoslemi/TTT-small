import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr_idx = np.argmax(corr[n_points - 1:])
        k = max_corr_idx
        return corr[n_points - 1 + k] * dx

    def compute_gradient(h, k, dx):
        n = len(h)
        grad = np.zeros(n)
        for i in range(n - k):
            grad[i] += (1 - h[i + k]) * dx
        for i in range(k, n):
            grad[i] -= h[i - k] * dx
        return grad

    def project_h(h_candidate, target_sum):
        n = len(h_candidate)
        h = cp.Variable(n)
        objective = cp.Minimize(cp.sum_squares(h - h_candidate))
        constraints = [0 <= h, h <= 1, cp.sum(h) == target_sum]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.OSQP, max_iter=100000, eps_abs=1e-6)
        if prob.status in ["infeasible", "unbounded"]:
            return h_candidate.copy()
        return h.value

    # Initialize and project
    h = h_initial.copy()
    h = project_h(h, target_sum)
    h = np.clip(h, 0.0, 1.0)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Optimization
    np.random.seed(seed)
    max_iterations = 1000
    patience = 50
    no_improve = 0
    learning_rate = 0.5
    iteration = 0

    while time.time() - start_time < budget_s - 10 and iteration < max_iterations:
        corr = np.correlate(h, 1 - h, mode='full')
        k = np.argmax(corr[n_points - 1:])
        grad = compute_gradient(h, k, dx)

        h -= learning_rate * grad
        h = project_h(h, target_sum)
        h = np.clip(h, 0.0, 1.0)

        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve = 0
            learning_rate *= 1.1
        else:
            no_improve += 1
            learning_rate *= 0.95
            if no_improve > patience:
                learning_rate = 0.5
                no_improve = 0

        iteration += 1

    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h.tolist(), best_c5, n_points)
