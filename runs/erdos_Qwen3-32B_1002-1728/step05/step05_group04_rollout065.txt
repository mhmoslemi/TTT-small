import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize initial h
    total_initial = np.sum(h_full_initial)
    h_initial_normalized = np.clip(h_full_initial * (target_sum / total_initial), 0, 1) if total_initial != 0 else h_full_initial.copy()

    def objective(x):
        h = np.array(x).reshape(n_points)
        total = np.sum(h)
        if total == 0:
            return np.inf
        scale = target_sum / total
        h_clipped = np.clip(h * scale, 0, 1)
        corr = np.correlate(h_clipped, 1 - h_clipped, mode='full') * dx
        return np.max(corr)

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Differential Evolution
    pop_size = 20
    pop = [np.clip(h_initial_normalized + np.random.normal(0, 0.05, n_points), 0, 1) for _ in range(pop_size)]
    remaining_time = budget_s - 240  # Reserve time for local optimization
    result_de = differential_evolution(objective, bounds, strategy='best1bin', maxiter=20, popsize=pop_size, 
                                       init=pop, polish=False, disp=False, mutation=(0.5, 1), recombination=0.9)

    best_h_de = np.clip(result_de.x.reshape(n_points) * (target_sum / np.sum(result_de.x)), 0, 1)

    # Local refinement
    result_local = minimize(lambda x: objective(x), x0=best_h_de, method='COBYLA', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-10}, tol=1e-8)

    best_h = best_h_de if not result_local.success else result_local.x
    best_c5 = objective(best_h) if objective(best_h) < result_local.fun else result_local.fun

    return (best_h.tolist(), best_c5, n_points)
