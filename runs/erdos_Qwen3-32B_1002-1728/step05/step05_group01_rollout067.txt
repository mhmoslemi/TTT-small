import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    x_samples = np.linspace(0.0, 2.0 - dx, n_points)
    sigma = 0.1
    # Build RBF matrix
    Phi = np.exp(- ((x_samples[:, None] - x_samples)**2) / (2 * sigma**2))
    target_sum = n_points / 2.0

    # Initial c from least squares
    initial_c = np.linalg.lstsq(Phi, h_init, rcond=None)[0]
    initial_c = np.clip(initial_c, 0.0, None)  # Ensure non-negative

    best_c = initial_c.copy()
    best_c5 = float('inf')

    def objective(c):
        h = Phi @ c
        h_clipped = np.clip(h, 0.0, 1.0)
        h_sum = np.sum(h_clipped)
        if h_sum < 1e-10:
            return 1e9
        scaling = target_sum / h_sum
        h_scaled = h_clipped * scaling
        h_scaled = np.clip(h_scaled, 0.0, 1.0)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, None) for _ in range(n_points)]

    start_time = time.time()
    while time.time() - start_time < budget_s - 10:
        # Perturbation to escape local minima
        perturbed_c = best_c + np.random.normal(0, 0.1, n_points)
        perturbed_c = np.clip(perturbed_c, 0.0, 10.0)  # Limiter

        for method in ['L-BFGS-B', 'Nelder-Mead']:
            result = minimize(
                fun=objective,
                x0=perturbed_c,
                method=method,
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-8}
            )
            if result.success:
                val = result.fun
                if val < best_c5:
                    best_c = result.x
                    best_c5 = val

        # Early exit if improvement is minimal
        if time.time() - start_time > 0.8 * budget_s:
            break

    h_opt = Phi @ best_c
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_sum = np.sum(h_opt)
    if h_sum > 1e-10:
        scaling = target_sum / h_sum
        h_opt *= scaling
    h_opt = np.clip(h_opt, 0.0, 1.0)

    return (h_opt.tolist(), best_c5, n_points)
