import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Total sum for integral = 1

    def objective(h_vars):
        h_full = np.clip(h_vars, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    current_h = np.array(h_full_initial)
    current_h = np.clip(current_h, 0, 1)
    current_sum = np.sum(current_h)
    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        current_h = current_h * (target_sum / current_sum)
    current_h = np.clip(current_h, 0, 1)
    best_h = current_h.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        frac = (budget_s - elapsed) / budget_s
        noise_scale = 0.4 * (frac ** 2)  # Higher noise at start, reduce over time

        seed += 1
        rnd = np.random.RandomState(seed)
        perturbed = best_h + rnd.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        p_sum = np.sum(perturbed)
        if p_sum != 0:
            perturbed = (perturbed / p_sum) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        method_cycle = ['SLSQP', 'COBYLA'] if n_attempt % 2 else ['L-BFGS-B', 'Nelder-Mead']
        optimizer_methods = method_cycle[:2 if n_attempt % 2 else 2]
        for method in optimizer_methods:
            result = minimize(
                fun=objective,
                x0=perturbed,
                method=method,
                bounds=bounds_per_var,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-12, 'disp': False}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                n_sum = np.sum(new_h)
                if n_sum != 0:
                    new_h = (new_h / n_sum) * target_sum
                new_h = np.clip(new_h, 0, 1)
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_h[:] = new_h
                    best_c5 = new_c5
                    break

    # Final validation
    best_h = np.clip(best_h, 0, 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
