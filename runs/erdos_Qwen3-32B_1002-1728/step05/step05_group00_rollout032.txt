    max_attempts = int(1e5)  # Arbitrary large limit; time cutoff will stop it
    attempt_count = 0
    while time.time() - start_time < budget_s - 20:  # Reserve buffer
        attempt_count += 1
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.3 * (1 - frac ** 1.5)  # Aggressive early exploration

        for _ in range(1 if attempt_count % 2 == 0 else 2):  # Alternate attempt counts
            seed += 1
            rnd = np.random.RandomState(seed)
            idxs = np.random.choice(half_n, min(half_n, 6), replace=False)
            perturbed = best_h_half.copy()
            perturbed[idxs] += rnd.normal(0, noise_scale, len(idxs))
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total != 0 and total != target_sum_half:
                perturbed = (perturbed / total) * target_sum_half
            perturbed = np.clip(perturbed, 0, 1)

            result = minimize(
                fun=objective, x0=perturbed, method='Nelder-Mead', bounds=bounds_per_var,
                options={'maxiter': 120, 'xatol': 1e-6, 'fatol': 1e-6, 'disp': False}
            )
            if result.success:
                new_h = result.x
                new_h = np.clip(new_h, 0, 1)
                total = np.sum(new_h)
                if total == 0:
                    continue
                if abs(total - target_sum_half) > 1e-6:
                    new_h = (new_h / total) * target_sum_half
                    new_h = np.clip(new_h, 0, 1)
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h_half = new_h.copy()
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
