import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def preprocess(h):
        h_clipped = np.clip(h, 0, 1)
        s = np.sum(h_clipped)
        if s != 0:
            h = h_clipped * (target_sum / s)
        return np.clip(h, 0, 1)

    # Initialize best solution
    h = preprocess(h_initial)
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Differential Evolution
    bounds = [(0, 1)] * n_points
    np.random.seed(seed)

    def objective(x):
        h_candidate = preprocess(x)
        corr_vals = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        return np.max(corr_vals) * dx

    try:
        de_result = differential_evolution(
            objective, bounds, strategy='best1bin', maxiter=10, popsize=10,
            mutation=(0.5, 1), recombination=0.9, seed=seed, polish=False, tol=1e-5
        )
        if de_result.success:
            h_de = preprocess(de_result.x)
            c5_de = objective(h_de)
            if c5_de < best_c5:
                best_h = h_de
                best_c5 = c5_de
    except: pass

    # Local optimization with constraints
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}

    while time.time() - start_time < budget_s - 5:
        seed += 1
        np.random.seed(seed)
        # Perturb best solution
        perturbed = best_h + np.random.normal(0, 0.1, n_points)
        perturbed = preprocess(perturbed)
        # Optimize with SLSQP
        result = minimize(
            objective, perturbed, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_new = preprocess(result.x)
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert all(0 <= x <= 1 for x in best_h)
    return (best_h.tolist(), best_c5, n_points)
