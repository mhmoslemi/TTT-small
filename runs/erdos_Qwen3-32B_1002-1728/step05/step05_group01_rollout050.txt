        frac = elapsed / budget_s
        noise_scale = 0.3 * (1 - (frac ** 0.75))

        for _ in range(3):
            perturbed = h_current + np.random.uniform(-noise_scale, noise_scale, (n_points,))
            perturbed_clipped = np.clip(perturbed, 0, 1)
            current_sum = np.sum(perturbed_clipped)
            if abs(current_sum) < 1e-10:
                continue
            scale_factor = target_sum / current_sum
            perturbed_scaled = perturbed_clipped * scale_factor
            x0 = np.clip(perturbed_scaled, 0, 1)

            for method in ['COBYLA', 'L-BFGS-B', 'SLSQP']:
                result = minimize(
                    fun=objective,
                    x0=x0,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons,
                    options={'maxiter': 150, 'ftol': 1e-10, 'disp': False}
                )
                if result.success:
                    result.x[np.isnan(result.x)] = 0.5
                    result.x = np.clip(result.x, 0, 1)
                    c5_new = objective(result.x)
                    if c5_new < best_c5 and np.allclose(np.sum(result.x), target_sum, atol=1e-6):
                        best_c5 = c5_new
                        h_current = result.x.copy()

    assert np.allclose(np.sum(h_current), target_sum, atol=1e-6)
    return (h_current.tolist(), best_c5, n_points)
