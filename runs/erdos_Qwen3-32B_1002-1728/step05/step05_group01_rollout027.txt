def run(seed=42, budget_s=1100, **kwargs):

    h_initial_full = np.array(initial_h_values)
    n_points = len(h_initial_full)
    half_n = n_points // 2

        return h_full

        h_clip = np.clip(h_vars, 0, 1)
        current_sum = np.sum(h_clip)

            return h_clip
        scale = target_sum_half / current_sum
        return np.clip(h_clip * scale, 0, 1)

    np.random.seed(seed)
    pop_size = 100

        h_half = np.random.rand(half_n)
        population.append(normalize_sum(h_half))

    best_h = normalize_sum(h_half_init)
    best_c5 = objective(best_h)

    while time.time() - start_time < budget_s - 30:
        # Evaluate population
        evals = []
        for ind in population:
            evals.append((objective(ind), ind))
        evals.sort()
        top_ratio = 0.2
