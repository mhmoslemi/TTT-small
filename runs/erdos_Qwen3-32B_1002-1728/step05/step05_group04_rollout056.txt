import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(initial_h)
    n_ones = n_points // 2

        h = np.zeros(n_points, dtype=np.float64)
        h[ones_set] = 1.0
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(initial_ones)
    best_h = initial_ones.copy()

    # GA parameters
    population_size = 100
    tournament_size = 5
    mutation_swaps = 2
    population = []

    # Initialize population with perturbations of initial solution
    for _ in range(population_size):
        ones = initial_ones.copy()
        current_zeros = np.setdiff1d(np.arange(n_points), ones)
        swaps = np.random.randint(1, 3)
        for _ in range(swaps):
            if len(ones) > 0 and len(current_zeros) > 0:
                i_one = ones.pop(np.random.randint(len(ones)))
                i_zero = current_zeros.pop(np.random.randint(len(current_zeros)))
                ones.append(i_zero)
                current_zeros = np.setdiff1d(np.arange(n_points), ones)
        population.append(sorted(ones))

    # Main GA loop
    while time.time() - start_time < budget_s:

        pop_c5 = []

        parents = []

            contestants = np.random.choice(population, tournament_size)

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    n_ones = n_points // 2
    dx = 2.0 / n_points

    # Extract initial ones indices
    initial_ones = [i for i in range(n_points) if initial_h[i] > 0.5]
    assert len(initial_ones) == n_ones

    # Objective function
    def compute_c5(ones_set):
        h = np.zeros(n_points, dtype=np.float64)
        h[ones_set] = 1.0
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(initial_ones)
    best_h = initial_ones.copy()

    # GA parameters
    population_size = 100
    tournament_size = 5
    mutation_swaps = 2
    population = []

    # Initialize population with perturbations of initial solution
    for _ in range(population_size):
        ones = initial_ones.copy()
        current_zeros = np.setdiff1d(np.arange(n_points), ones)
        swaps = np.random.randint(1, 3)
        for _ in range(swaps):
            if len(ones) > 0 and len(current_zeros) > 0:
                i_one = ones.pop(np.random.randint(len(ones)))
                i_zero = current_zeros[np.random.randint(len(current_zeros))]
                ones.append(i_zero)
                current_zeros = np.setdiff1d(np.arange(n_points), ones)
        population.append(sorted(ones))

    # Main GA loop
    while time.time() - start_time < budget_s:
        # Evaluate current population
        pop_c5 = []
        for ind in population:
            c5 = compute_c5(ind)
            pop_c5.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()

        # Tournament selection
        parents = []
        for _ in range(population_size):
            contestant_indices = np.random.choice(len(population), tournament_size)
            contestant_c5s = [pop_c5[i] for i in contestant_indices]
            winner_idx = contestant_indices[np.argmin(contestant_c5s)]
            parents.append(population[winner_idx])

        # Generate next population via mutation
        next_gen = []
        for parent in parents:
            child = parent.copy()
            if np.random.rand() < 0.1:  # Mutation probability
                current_zeros = np.setdiff1d(np.arange(n_points), child)
                if len(current_zeros) == 0 or len(child) == 0:
                    next_gen.append(child)
                    continue
                for _ in range(mutation_swaps):
                    if len(child) > 0 and len(current_zeros) > 0:
                        i_one = np.random.choice(child)
                        i_zero = np.random.choice(current_zeros)
                        child.remove(i_one)
                        child.append(i_zero)
                        current_zeros = np.setdiff1d(np.arange(n_points), child)
            next_gen.append(sorted(child))

    # Construct final h_values

    h_values[best_h] = 1.0
    return (h_values.tolist(), best_c5, n_points)
