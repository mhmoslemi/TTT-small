        corr = np.correlate(x_clipped, 1 - x_clipped, mode='full')
        return np.max(corr) * dx

    # Run DE
    de_params = {
        'strategy': 'best1bin',
        'maxiter': 10,
        'popsize': 10,
        'mutation': (0.5, 1),
        'recombination': 0.8,
        'seed': seed,
        'polish': False,
        'init': 'sobol',
        'disp': False
    }
    try:
        de_result = differential_evolution(objective_de, bounds, **de_params)
        if de_result.success:
            de_h = np.array(de_result.x)
            s = np.sum(de_h)
            de_h = de_h / s * target_sum
            de_h = np.clip(de_h, 0, 1)
            # Now run local optimize
            result = minimize(
                objective_local,
                de_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-12, 'disp': False}
            )
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_c5 = objective_local(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
    except:
        pass  # Handle errors gracefully

    # Phase 2: Local optimization with perturbations
    np.random.seed(seed)
    t0 = time.time()

    while time.time() - t0 < budget_s - 30:
        seed += 1
        # Generate a perturbed version of best_h
        perturb = np.random.normal(0, 0.1, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        # Normalize sum to target
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_sum = 1e-8  # Avoid division by zero
        new_h = new_h / new_sum * target_sum
        new_h = np.clip(new_h, 0, 1)

        # Choose optimizer
        method = ['SLSQP', 'COBYLA', 'L-BFGS-B'][(int(seed) % 3)]
        try:
            result = minimize(
                objective_local,
                new_h,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-12, 'disp': False}
            )
            if result.success:
                optimized_h = np.clip(result.x, 0, 1)
                c5 = objective_local(optimized_h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = optimized_h
        except:
            pass  # Skip on error

    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)

This code first runs differential evolution to find an initial candidate, then performs a series of local optimizations with perturbations.

The differential evolution is set with parameters that aim to balance exploration and computation time.

In the phase 2 loop, each iteration generates a perturbed version of the best h found so far, and runs a local optimizer to refine it. The perturbations are small (0.1 std), and the sum is adjusted.

This code is also designed to exit early if the time budget is exceeded.

I think this approach has a better chance of escaping local minima than the previous code, which relied solely on symmetric perturbations and different optimizers. By breaking the symmetry and allowing the optimizer to explore asymmetric configurations, it's possible to find a better solution with lower C₅.

import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize initial guess
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial / h_initial.sum() * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    def objective_local(x): 
        # Objective function for constrained optimization
        corr = np.correlate(x, 1 - x, mode='full')
        return np.max(corr) * dx

    def objective_de(x): 
        # Objective function with approximate constraint
        return objective_local(np.clip((x / x.sum() * target_sum), 0, 1)) if x.sum() != 0 else float('inf')

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = h_initial.copy()
    best_c5 = float('inf')

    # Phase 1: Differential Evolution
    de_result = differential_evolution(
        objective_de, 
        bounds, 
        strategy='best1bin',
        maxiter=10,
        popsize=10,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=False,
        init='sobol'
    )

    if de_result.success:
        # Use DE solution as warm start
        x = de_result.x
        x = np.clip((x / x.sum() * target_sum), 0, 1)
        result = minimize(
            objective_local,
            x,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-12}
        )
        if result.success and objective_local(result.x) < best_c5:
            best_h = result.x
            best_c5 = objective_local(best_h)

    # Phase 2: Iterative refinement
    np.random.seed(seed)
    while time.time() - start_time < budget_s - 30:
        seed = seed + 1
        # Perturb current best
        perturb = np.random.normal(0, 0.05, n_points) if np.random.rand() < 0.7 else np.random.normal(0, 0.2, n_points)
        new_h = best_h + perturb
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h / new_h.sum() * target_sum
        new_h = np.clip(new_h, 0, 1)

        method = ['SLSQP', 'COBYLA', 'L-BFGS-B'][(seed % 3)]
        result = minimize(
            objective_local,
            new_h,
            method=method,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-12}
        )

        if result.success and np.sum(result.x) == target_sum:
            candidate = np.clip(result.x, 0, 1)
            candidate_c5 = objective_local(candidate)
            if candidate_c5 < best_c5:
                best_h = candidate
                best_c5 = candidate_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
