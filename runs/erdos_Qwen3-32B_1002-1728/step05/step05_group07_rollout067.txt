import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 70  # Increased population size
    max_generations = 1000  # Increased generations

    # Initialize population using full h vectors
    population = [h_full_initial.copy()]
    for _ in range(population_size - 1):
        perturbed = h_full_initial + np.random.normal(0, 0.15, n_points)  # Higher initial mutation strength
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_h_full = h_full_initial.copy()
    best_c5 = compute_c5(best_h_full)

    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1

        # Evaluate population
        candidates = [(compute_c5(h), h.copy()) for h in population]
        candidates.sort(key=lambda x: x[0])
        parents = [h for c, h in candidates[:population_size // 2]]

        # Create new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(len(parents), 2)
            parent1, parent2 = parents[idx1], parents[idx2]
            parent1_score = compute_c5(parent1)
            parent2_score = compute_c5(parent2)
            if parent1_score < parent2_score:
                chosen = parent1
            else:
                chosen = parent2
            if np.random.rand() < 0.4:  # Increased crossover rate
                # Blend crossover
                blend = 0.5 * (chosen + np.random.choice(parents)[0])
            else:
                blend = chosen.copy()
            if np.random.rand() < 0.2:  # Mutation
                # Adaptive mutation strength
                mutation_strength = 0.2 * (1 - (generation / max_generations) ** 1.2)
                blend += np.random.normal(0, mutation_strength, n_points)
                blend = np.clip(blend, 0, 1)
                total = np.sum(blend)
                if total != 0:
                    blend = (blend / total) * target_sum
                blend = np.clip(blend, 0, 1)
            new_population.append(blend)

        population = new_population
        # Update best solution
        candidates = [(compute_c5(h), h.copy()) for h in population]
        current_best_c5 = min(c for c, _ in candidates)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h_full = candidates[np.argmin([c for c, _ in candidates])][1].copy()

    assert abs(np.sum(best_h_full) - target_sum) < 1e-6
    return (best_h_full.tolist(), best_c5, n_points)
