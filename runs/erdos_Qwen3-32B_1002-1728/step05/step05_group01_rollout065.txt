import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_de(h_unscaled):
        h = np.clip(h_unscaled, 0.0, 1.0)
        total = np.sum(h)
        if total < 1e-6:
            return 1.0
        h_norm = h * (target_sum / total)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Differential evolution phase
    remaining_time = budget_s - (time.time() - start_time)
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        maxiter=300 if remaining_time > 200 else 50,
        popsize=10,
        polish=False,
        disp=False,
        workers=1
    )

    best_h_de_unscaled = result_de.x
    best_h_de = np.clip(best_h_de_unscaled, 0.0, 1.0)
    total = np.sum(best_h_de)
    best_h_de = best_h_de * (target_sum / total)
    best_c5 = np.max(np.correlate(best_h_de, 1 - best_h_de, mode='full')) * dx

    # Local refinement phase
    while time.time() - start_time < budget_s - 5:
        perturbation = 0.1 * np.random.default_rng(seed).uniform(-1, 1, n_points)
        x0 = np.clip(best_h_de + perturbation, 0.0, 1.0)
        x0 = x0 * (target_sum / np.sum(x0))

        cons_local = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        for method in ['L-BFGS-B', 'SLSQP', 'COBYLA']:
            if time.time() - start_time >= budget_s - 5:
                break
            result_loc = minimize(
                fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                x0=x0,
                method=method,
                bounds=[(0.0, 1.0)] * n_points,
                constraints=cons_local,
                options={'ftol': 1e-12, 'maxiter': 200}
            )
            if result_loc.success:
                h_candidate = result_loc.x
                corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr_candidate) * dx

                    best_c5 = c5_candidate
                    best_h_de = h_candidate

    assert np.all((best_h_de >= -1e-6) & (best_h_de <= 1 + 1e-6))

    return (best_h_de.tolist(), best_c5, n_points)
