import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # h_half sum to target_sum_half => full sum n_points/2
    dx = 2.0 / n_points

    # Construct symmetric h_full from h_half
    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[half_n:] = h_vars[::-1]
        return h_full

    # Evaluate the maximum C₅ for a given h_half
    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 50

    # Initialize population with symmetric constraints and normalization
    population = []
    h_half_initial = h_full_initial[:half_n].copy()
    sum_initial = np.sum(h_half_initial)
    if not np.isclose(sum_initial, target_sum_half):
        h_half_initial *= target_sum_half / sum_initial
    h_half_initial = np.clip(h_half_initial, 0, 1)
    for _ in range(pop_size):
        pert = h_half_initial + np.random.uniform(-0.05, 0.05, size=half_n)
        pert = np.clip(pert, 0.0, 1.0)
        scale = target_sum_half / (np.sum(pert) + 1e-10)
        pert *= scale
        pert = np.clip(pert, 0, 1)
        population.append(pert.copy())

    best_c5 = float('inf')
    best_h_half = None

    while time.time() - start_time < budget_s - 10:
        # Evaluate all population members
        scores = []
        for h_half in population:
            try:
                c5 = evaluate_c5(h_half)
                scores.append((h_half, c5))
            except:
                scores.append((h_half, float('inf')))
        # Update best solution
        scores = [(h, c) for h, c in scores if np.isfinite(c)]
        if scores:
            best_current = min(scores, key=lambda x: x[1])
            if best_current[1] < best_c5:
                best_c5 = best_current[1]
                best_h_half = best_current[0].copy()

        # Selection (Binary tournament)
        selected = []
        for _ in range(pop_size // 2):
            idx1, idx2 = np.random.choice(len(population), 2, replace=False)
            h1, c1 = scores[idx1]
            h2, c2 = scores[idx2]
            selected.append(h1 if (c1 < c2) else h2)

        # Crossover and mutation
        offspring = []
        for _ in range(len(selected) // 2):
            # Two parents, blend with β in [0.3, 0.7]
            parents = [selected.pop(0), selected.pop(0)]
            beta = np.random.uniform(0.3, 0.7)
            child1 = beta * parents[0] + (1 - beta) * parents[1]
            child2 = (1 - beta) * parents[0] + beta * parents[1]
            # Mutation
            for child in [child1, child2]:
                child += np.random.normal(0, 0.02, size=half_n)
                child = np.clip(child, 0, 1)
                if np.sum(child) < 1e-10:
                    child = np.zeros_like(child) + 0.5
                scale = target_sum_half / np.sum(child)
                child *= scale
                child = np.clip(child, 0, 1)
                offspring.append(child)

        # Maintain best half + infants
        new_population = [h for h, _ in sorted(scores, key=lambda x: x[1])[:pop_size//2]]
        new_population.extend(offspring)
        population = new_population

    # Final checks
