            child_valid = make_valid(child)
            score = evaluate(child_valid)
            new_population.append((score, child_valid))
        
        # Select top 80 individuals
        new_population.sort()
        fitness = new_population[:population_size]
        if fitness[0][0] < best_fitness:
            best_fitness, best_h = fitness[0]

    return (best_h.tolist(), best_fitness, n_points)
