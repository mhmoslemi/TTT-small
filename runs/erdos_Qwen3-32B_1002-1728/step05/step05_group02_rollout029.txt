import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(h, noise_scale=0.2):
        h = h.copy()
        noise = np.random.normal(0, noise_scale, n_points)
        h += noise
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s != 0:
            h = h * (target_sum / s)
        h = np.clip(h, 0, 1)
        return h

    population_size = 20
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(mutate(population[-1], 0.2))

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s - 30:
        fitness = [compute_c5(ind) for ind in population]
        for i in range(len(fitness)):
            if fitness[i] < best_c5:
                best_h = population[i].copy()
                best_c5 = fitness[i]

        indices = np.argsort(fitness)
        top_idx = int(population_size * 0.5)
        top = [population[i] for i in indices[:top_idx]]

        new_pop = []
        for _ in range(population_size):
            if time.time() - start_time >= budget_s - 30:
                break
            parent = top[np.random.randint(len(top))]
            scale = 0.2 * (budget_s - (time.time() - start_time)) / budget_s
            new_pop.append(mutate(parent, scale))
        population = new_pop

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(compute_c5, best_h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 100, 'ftol': 1e-12})
    if result.success and result.fun < best_c5:
        best_h = np.clip(result.x, 0, 1)
        best_c5 = result.fun

    s = np.sum(best_h)
    if abs(s - target_sum) > 1e-6:
        best_h = best_h * (target_sum / s)
        best_h = np.clip(best_h, 0, 1)

    return (best_h.tolist(), best_c5, n_points)
