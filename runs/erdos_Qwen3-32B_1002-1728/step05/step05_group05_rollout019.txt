        for indiv in population:
            h_full = create_full_h(indiv)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            evaluated.append((indiv, c5))

        # Sort and update best
        evaluated.sort(key=lambda x: x[1])
        if evaluated[0][1] < best_c5:
            best_c5 = evaluated[0][1]
            best_h = evaluated[0][0]

        # Elitism
        elite_size = max(1, int(population_size * 0.1))
        elites = [e[0].copy() for e in evaluated[:elite_size]]

        # Generate new population
        new_pop = elites.copy()
        while len(new_pop) < population_size:
            # Tournament selection
            parents = np.random.choice(len(evaluated), size=2)
            parent1 = evaluated[np.argmin([evaluated[i][1] for i in parents])][0]
            parent2 = evaluated[np.random.choice(len(evaluated))][0]

            # Crossover
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            child = np.clip(child, 0, 1)
            scale_c = target_sum_half / np.sum(child)
            child *= scale_c
            child = np.clip(child, 0, 1)

            # Mutation
            if np.random.rand() < 0.2:
                noise = np.random.normal(0, 0.1, child.shape)
                child += noise
                child = np.clip(child, 0, 1)
                scale_c = target_sum_half / np.sum(child)
                child *= scale_c
                child = np.clip(child, 0, 1)

            new_pop.append(child)

        population = new_pop

    # Final validation and return
    h_full = create_full_h(best_h)
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = np.sum(h_full)
    assert np.isclose(h_full_sum, n_points / 2.0), f"Sum {h_full_sum} != {n_points/2.0}"
    return (h_full.tolist(), best_c5, n_points)
