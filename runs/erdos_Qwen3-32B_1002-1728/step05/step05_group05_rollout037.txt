import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 50
    initial_h_half = h_full_initial[:half_n].copy()
    population = [initial_h_half.copy() for _ in range(pop_size)]

    for i in range(pop_size):
        perturbed = population[i] + np.random.normal(0, 0.1, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        s = np.sum(perturbed)
        if not np.isclose(s, 0):
            scale = target_sum_half / s
            perturbed *= scale
        perturbed = np.clip(perturbed, 0, 1)
        population[i] = perturbed

    best_c5 = float('inf')
    best_h_full = None

    while time.time() - start_time < budget_s - 20:
        candidates = []
        for h in population:
            c5 = evaluate_c5(h)
            candidates.append((h.copy(), c5))
        candidates.sort(key=lambda x: x[1])
        new_current = [c[0] for c in candidates]

        if candidates[0][1] < best_c5:
            best_c5 = candidates[0][1]
            best_h_half = candidates[0][0]
            best_h_full = create_full_h(best_h_half)

        new_population = []
        for _ in range(pop_size // 2):
            # Tournament selection
            idx1, idx2 = np.random.randint(0, pop_size, 2)
            p1, p2 = candidates[idx1], candidates[idx2]
            parent1 = p1[0].copy() if p1[1] < p2[1] else p2[0].copy()
            idx3 = np.random.randint(0, pop_size)
            parent2 = candidates[idx3][0].copy()

            # Crossover (blending)
            alpha = np.random.rand()
