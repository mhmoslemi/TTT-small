import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum(h_half) = n_points/4 because h is symmetric

    # Initialize symmetric h_half from initial_h_values
    h_half = h_full_initial[:half_n].copy()
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = h_half[i]
    current_sum = np.sum(h_full)
    if current_sum != 0:
        scale = (n_points / 2.0) / current_sum
        h_full *= scale
        h_half = h_full[:half_n]
    h_half = np.clip(h_half, 0, 1)

    best_h_half = h_half.copy()
    best_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full') * dx)

    # Optimization parameters
    lambda_val = 1e-3
    max_iters = 500
    patience = 20
    no_improvement = 0

    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break

        # Construct full h from h_half
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        # Correct sum if needed
        current_sum = np.sum(h_full)
        if abs(current_sum - n_points / 2.0) > 1e-6:
            scale = (n_points / 2.0) / current_sum
            h_full *= scale
            for i in range(half_n):
                h_full[i] = h_full[n_points - 1 - i] = h_half[i]

        # Find worst shift k
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        k_worst = np.argmax(corr)
        current_c5 = corr[k_worst]

        # Compute gradient of worst shift k in h_full
        g_full = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_worst) % n_points
            j_prev = (i - k_worst) % n_points
            term = (1 - h_full[j]) - h_full[j_prev]
            g_full[i] = term * dx

        # Project gradient to h_half using symmetry
        G = np.zeros(half_n)
        for i in range(half_n):
            G[i] = g_full[i] + g_full[n_points - 1 - i]

        # Solve QP for step direction
        delta_var = cp.Variable(half_n)
        objective = cp.Minimize( G @ delta_var + 0.5 * lambda_val * cp.sum_squares(delta_var) )
        constraints = [
            cp.sum(delta_var) == 0,
            h_half + delta_var <= 1,
            h_half + delta_var >= 0
        ]
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.OSQP, max_iter=10000, verbose=False)
        except:
            no_improvement += 1
            continue

        if delta_var.value is None:
            no_improvement += 1
            continue

        delta_sol = delta_var.value

        # Update h_half and clip
        h_half += delta_sol
        h_half = np.clip(h_half, 0, 1)

        # Rescale to maintain sum(h_half) == target_sum
        new_sum_h_half = np.sum(h_half)
        if abs(new_sum_h_half - target_sum) > 1e-6:
            scale = target_sum / new_sum_h_half
            h_half *= scale
            h_half = np.clip(h_half, 0, 1)

        # Reconstruct h_full and evaluate improvement
        h_full_update = np.zeros(n_points)
        for i in range(half_n):
            h_full_update[i] = h_full_update[n_points - 1 - i] = h_half[i]
        corr_new = np.correlate(h_full_update, 1 - h_full_update, mode='full') * dx
        new_c5 = np.max(corr_new)

        if new_c5 < best_c5 - 1e-6:
            best_c5 = new_c5
            best_h_half = h_half.copy()
            no_improvement = 0
        else:
            no_improvement += 1

        if no_improvement >= patience:
            break

    # Final check and return
    h_full_final = np.zeros(n_points)
    for i in range(half_n):
        h_full_final[i] = h_full_final[n_points - 1 - i] = best_h_half[i]
    current_sum = np.sum(h_full_final)
    if abs(current_sum - n_points / 2.0) > 1e-6:
        h_full_final *= (n_points / 2.0) / current_sum
    return (h_full_final.tolist(), np.max(np.correlate(h_full_final, 1 - h_full_final, mode='full') * dx), n_points)
