import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = np.sum(h_copy)
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    np.random.seed(seed)
    h_armor = np.array(h_full_initial)
    initial_half = h_armor[:half_n].copy()
    initial_half = normalize_h_half(initial_half, target_sum_half)
    
    swarm_size = 50
    positions = [initial_half + np.random.normal(0, 0.1, half_n) for _ in range(swarm_size)]
    velocities = [np.random.normal(0, 0.05, half_n) for _ in range(swarm_size)]
    positions = [normalize_h_half(p, target_sum_half) for p in positions]
    personal_best_positions = positions.copy()
    personal_best_c5 = [float('inf')] * swarm_size
    global_best_position = initial_half.copy()
    global_best_c5 = float('inf')

    max_iter_time = budget_s - 30
    c1, c2 = 2.0, 2.0
    inertia_start, inertia_end = 0.9, 0.4

    while time.time() - start_time < max_iter_time:
        elapsed_ratio = (time.time() - start_time) / max_iter_time
        inertia = inertia_start - elapsed_ratio * (inertia_start - inertia_end)
        for i in range(swarm_size):
            pos = positions[i]
            vel = velocities[i]
            pbest_pos = personal_best_positions[i]
            gbest_pos = global_best_position

            # Compute current c5
            h_full = create_full_h(pos)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx

            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best_positions[i] = pos.copy()
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_position = pos.copy()

            # Update velocity and position

                   c2 * r2 * (global_best_position - pos))
