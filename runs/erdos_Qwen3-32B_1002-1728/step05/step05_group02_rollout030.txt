import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    dx = 2.0 / n_points
    required_sum = n_points / 2.0

    population_size = 20
    tournament_size = 5
    elitism = 5  # Number of top candidates to preserve per generation

    def evaluate_individual(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def perturb(h, noise_scale):
        h_mutated = h.copy()
        noise = np.random.normal(0, noise_scale, n_points)
        h_mutated += noise
        h_mutated = np.clip(h_mutated, 0, 1)
        s = np.sum(h_mutated)
        if s != 0:
            h_mutated = (h_mutated / s) * required_sum
        h_mutated = np.clip(h_mutated, 0, 1)
        return h_mutated

    # Initialize population with asymmetric perturbations of initial_h_values
    population = []
    np.random.seed(seed)
    for _ in range(population_size):
        candidate = perturb(h_init, 0.15)
        population.append(candidate)

    best_c5 = np.inf
    best_h = h_init.copy()

    # Evolutionary search
    while time.time() - start_time < budget_s - 60:  # Reserve 60s for local optimization
        offspring = []
        for _ in range(population_size):

            candidates = np.random.choice(population_size, tournament_size, replace=False)
            parent_idx = min(candidates, key=lambda i: evaluate_individual(population[i]))

            remaining = (budget_s - (time.time() - start_time)) / budget_s
