    population = [binary_h_initial.copy() for _ in range(pop_size // 2)]
    for _ in range(pop_size // 2):
        new_ind = np.zeros(n_points, dtype=int)
        indices = np.random.choice(n_points, target_sum, replace=False)
        new_ind[indices] = 1
        population.append(new_ind)
    
    best_h = binary_h_initial.copy()
    best_c5 = np.inf
    
    def evaluate(ind):
        h = ind.astype(float)
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx
        
    # Main loop
    while time.time() - start_time < budget_s - 60:
        # Evaluate population
        fitness = np.array([evaluate(ind) for ind in population])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness[current_best_idx]
            
        # Selection and reproduction
        new_pop = []
        for _ in range(pop_size):
            # Tournament selection
            candidates = np.random.choice(pop_size, tournament_k, replace=False)
            parent = population[np.argmin(fitness[candidates])].copy()
            
            # Crossover (one-point with fix)
            if np.random.rand() < 0.8 and pop_size > 1:
                candidates2 = np.random.choice(pop_size, tournament_k, replace=False)
                parent2 = population[np.argmin(fitness[candidates2])].copy()
                cross_point = np.random.randint(n_points)
                child = np.hstack((parent[:cross_point], parent2[cross_point:]))
                child = fix_count(child, target_sum)
            else:
                child = parent.copy()
            
            # Mutation
            if np.random.rand() < mut_rate:
                ones = child[child == 1]
                zeros = child[child == 0]
                if len(zeros) > 0 and len(ones) > 0:
                    z_idx = np.random.choice(np.where(child == 0)[0])
                    o_idx = np.random.choice(np.where(child == 1)[0])
                    child[z_idx] = 1
                    child[o_idx] = 0
            
            new_pop.append(child)
        population = new_pop
        
    h_values = best_h.astype(float)
    return (h_values.tolist(), best_c5, n_points)
