            df_j = 0.0
            if j + k < n_points:
                df_j += (1.0 - h_full_clone[j + k]) * dx
            if j - k >= 0:
                df_j -= h_full_clone[j - k] * dx
            grad[h_idx] = df_i + df_j
        
        # Solve QP
        delta = cp.Variable(half_n)
        obj = cp.Minimize(grad @ delta + 0.5 * cp.sum_squares(delta) * 1e-3)
        constraints = [
            cp.sum(delta) == 0,
            h_half + delta >= 0,
            h_half + delta <= 1
        ]
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.OSQP)
        
        if prob.status != cp.OPTIMAL:
            continue
        
        h_half += delta.value
        h_half = np.clip(h_half, 0, 1)
        current_sum = np.sum(h_half)
        if current_sum != 0:
            scaling_factor = target_sum_half / current_sum
            h_half *= scaling_factor
        h_half = np.clip(h_half, 0, 1)
        
        h_full_current = create_full_h(h_half)
        current_c5 = np.max(np.correlate(h_full_current, 1 - h_full_current, mode='full') * dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()
    
    h_full_result = create_full_h(best_h_half)
    return (h_full_result.tolist(), best_c5, n_points)
