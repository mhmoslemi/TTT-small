def normalize_h_half(h_half, target_sum):

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def normalize_h_half(h_half):
        sum_h = np.sum(h_half)
        if sum_h == 0:
            return h_half.copy()
        scaled = h_half * (target_sum_half / sum_h)
        clipped = np.clip(scaled, 0, 1)
        new_sum = np.sum(clipped)
        if new_sum == 0:
            return clipped.copy()
        return clipped * (target_sum_half / new_sum)

    def compute_correlation_at_p(h_full, p):
        n = h_full.shape[0]
        k = p - (n - 1)
        start = max(0, -k)
        end = min(n - 1, n - 1 - k)
        if start > end:
            return 0.0
        total = 0.0
        for i in range(start, end + 1):
            j = i + k
            if j >= 0 and j < n:
                total += h_full[i] * (1 - h_full[j])
        return total * dx

    h_half_best = normalize_h_half(h_full_initial[:half_n].copy())
    best_c5 = float('inf')
    np.random.seed(seed)

    perturb_interval = 50
    learning_rate = 0.01
    max_iterations = 6000

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break

        h_full = create_full_h(h_half_best)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        current_c5 = np.max(corr)
        best_c5 = min(best_c5, current_c5)
        max_p = np.argmax(corr)

        eps = 1e-5
        gradient = np.zeros(half_n)

        for m in range(half_n):
            h_perturbed = h_half_best.copy()
            h_perturbed[m] += eps
            h_clipped = np.clip(h_perturbed, 0, 1)
            h_full_perturbed = create_full_h(h_clipped)
            perturbed_corr = compute_correlation_at_p(h_full_perturbed, max_p)
            gradient[m] = (perturbed_corr - current_c5) / eps

        step_update = h_half_best - learning_rate * gradient
        h_clipped = np.clip(step_update, 0, 1)
        h_half_new = normalize_h_half(h_clipped)
        h_full_new = create_full_h(h_half_new)
        new_corr = np.max(np.correlate(h_full_new, 1 - h_full_new, mode='full') * dx)

        if new_corr < current_c5:
            h_half_best = h_half_new

        if iteration % perturb_interval == 0:
            noise = np.random.normal(0, 0.1, half_n)
            noisy_h = h_half_best + noise
            noisy_h = normalize_h_half(np.clip(noisy_h, 0, 1))
            h_full_noisy = create_full_h(noisy_h)
            noisy_corr = np.max(np.correlate(h_full_noisy, 1 - h_full_noisy, mode='full') * dx)
            if noisy_corr < best_c5:
                h_half_best = noisy_h

    h_full_final = create_full_h(h_half_best)
    final_corr = np.max(np.correlate(h_full_final, 1 - h_full_final, mode='full') * dx)
    assert abs(np.sum(h_full_final) - n_points / 2.0) < 1e-6
    assert np.all(h_full_final <= 1.0 + 1e-6) and np.all(h_full_final >= -1e-6)
    return (h_full_final.tolist(), final_corr, n_points)
