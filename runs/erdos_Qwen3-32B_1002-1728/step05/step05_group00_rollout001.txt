import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate(h):
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr_vals)

    def process_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            return h_clipped
        scaling_factor = target_sum / current_sum
        scaled = h_clipped * scaling_factor
        return np.clip(scaled, 0, 1)

    best_h = process_h(h_initial.copy())
    best_c5 = evaluate(best_h)

    np.random.seed(seed)
    n_particles = 20
    inertia = 0.8
    cognitive = 1.2
    social = 1.2

    particles = []
    velocities = []
    personal_best_positions = []
    personal_best_values = []
    for _ in range(n_particles):
        pos = np.random.rand(n_points)
        pos = process_h(pos)
        vel = np.random.uniform(-0.1, 0.1, n_points)
        particles.append(pos)
        velocities.append(vel)
        pb = pos.copy()
        pb_val = evaluate(pb)
        personal_best_positions.append(pb)
        personal_best_values.append(pb_val)

    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx]
    global_best_c5 = personal_best_values[global_best_idx]

    for iter_pso in range(60):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(n_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            vel_update = inertia * velocities[i] + cognitive * r1 * (
                personal_best_positions[i] - particles[i]) + social * r2 * (global_best_position - particles[i])
            velocities[i] = np.clip(vel_update, -0.2, 0.2)
            new_pos = particles[i] + velocities[i]
            new_pos = process_h(new_pos)
            new_val = evaluate(new_pos)
            if new_val < personal_best_values[i]:
                personal_best_positions[i] = new_pos.copy()
                personal_best_values[i] = new_val
                if new_val < global_best_c5:
                    global_best_position = new_pos.copy()
                    global_best_c5 = new_val

    x0 = global_best_position.copy()
    def objective(x_vec):
        x = np.copy(x_vec)
        x = process_h(x)
        return evaluate(x)

    bounds = [(0, 1)] * n_points
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, 
                      options={'maxiter': 200, 'ftol': 1e-8})
    if result.success:
        final_h = process_h(result.x)
        final_c5 = evaluate(final_h)
        if final_c5 < evaluate(global_best_position):
            best_c5 = final_c5
            best_h = final_h

    return (process_h(best_h).tolist(), best_c5, n_points)
