                pbest_values[i] = new_c5
                pbest_positions[i] = new_pos.copy()
                if new_c5 < gbest_value:
                    gbest_value = new_c5
                    gbest_pos[:] = new_pos.copy()

    best_h_full = create_full_h(gbest_pos)
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    assert np.abs(np.sum(best_h_full) - n_points / 2.0) < 1e-6

    return (best_h_full.tolist(), gbest_value, n_points)
