import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full
    
    def compute_worst_k(full_h):
        corr = np.correlate(full_h, 1 - full_h, mode='full') * dx
        max_idx = np.argmax(corr)
        return np.max(corr), max_idx
    
    # Initialization with symmetry and scaling
    h_full_initial = create_full_h(h_initial)
    curr_sum = np.sum(h_full_initial)
    if curr_sum != 0:
        scale = target_sum * 2.0 / curr_sum
        h_full_initial *= scale
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_half = h_full_initial[:half_n].copy()

    best_h_half = h_half.copy()
    best_c5 = compute_worst_k(h_full_initial)[0]

        h_full = create_full_h(h_half)
        curr_corr, max_k = compute_worst_k(h_full)

        grad = np.zeros(half_n)
        eps = 1e-5

            up = h_half.copy()
            up[i] += eps
            down = h_half.copy()
            down[i] -= eps
            fu = create_full_h(up)
            fd = create_full_h(down)
            corr_up = np.correlate(fu, 1 - fu, mode='full')[max_k] * dx
            corr_down = np.correlate(fd, 1 - fd, mode='full')[max_k] * dx
            grad[i] = (corr_up - corr_down) / (2 * eps)
