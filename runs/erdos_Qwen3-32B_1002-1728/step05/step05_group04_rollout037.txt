def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    n_points = len(initial_h_list)
    target_sum = int(n_points / 2)
    assert sum(initial_h_list) == target_sum

    dx = 2.0 / n_points

    def evaluate_c5(h_list):
        h = np.array(h_list, dtype=np.float64)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def mutate(individual):
        individual = individual.copy()
        zero_indices = [i for i, val in enumerate(individual) if val == 0]
        one_indices = [i for i, val in enumerate(individual) if val == 1]
        if not zero_indices or not one_indices:
            return individual
        i0 = np.random.choice(zero_indices)
        i1 = np.random.choice(one_indices)
        individual[i0] = 1
        individual[i1] = 0
        return individual

    def fix_sum(individual):
        individual = individual.copy()
        current_sum = sum(individual)
        if current_sum == target_sum:
            return individual
        delta = current_sum - target_sum
        if delta < 0:
            flip_count = abs(delta)
            zero_indices = [i for i, val in enumerate(individual) if val == 0]
            idxs = np.random.choice(zero_indices, flip_count, replace=False)
            for i in idxs:
                individual[i] = 1
        else:
            flip_count = delta
            one_indices = [i for i, val in enumerate(individual) if val == 1]
            idxs = np.random.choice(one_indices, flip_count, replace=False)
            for i in idxs:
                individual[i] = 0
        return individual

    def crossover(p1, p2, n_points=n_points):
        split = np.random.randint(1, n_points - 1)
        c1 = fix_sum(p1[:split] + p2[split:])
        c2 = fix_sum(p2[:split] + p1[split:])
        return c1, c2

    # Initialize population
    population_size = 50
    population = [initial_h_list.copy()]
    for _ in range(population_size - 1):
        candidate = mutate(population[0])
        population.append(candidate)

    best_c5 = evaluate_c5(initial_h_list)
    best_h = initial_h_list.copy()

    while time.time() - start_time < budget_s - 10:

            best_h = combined[idxs[0]]

    return (best_h.tolist(), best_c5, n_points)
