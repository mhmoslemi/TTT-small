from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        h_clipped = np.clip(h_full, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return float('inf')
        h_scaled = h_clipped * (target_sum / sum_h)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    h_current = np.array(h_initial).copy()

    np.random.seed(seed)

            perturbed = (perturbed / sum_p) * target_sum
