import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Constraint sum(h) == n_points / 2

    h = h_initial.copy()
    h = (h + h[::-1]) / 2  # Enforce symmetry around center
    current_sum = np.sum(h)
    if current_sum != 0:
        h *= target_sum / current_sum
    h = np.clip(h, 0, 1)

    # Helper to compute the worst k and correlation
    def evaluate_worst_k(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_idx = np.argmax(corr)
        mid = len(h_vec) - 1
        worst_k = max_idx - mid  # Maps full correlate index to actual shift
        return corr, worst_k

    # Gradient computation for the current worst k
    def compute_gradient(h_vec, worst_k):
        grad = np.zeros_like(h_vec)
        N = len(h_vec)
        for i in range(N):
            j = i + worst_k
            if 0 <= j < N:
                grad[i] += (1 - h_vec[j])
        return grad * dx  # Multiply by dx as per original objective
