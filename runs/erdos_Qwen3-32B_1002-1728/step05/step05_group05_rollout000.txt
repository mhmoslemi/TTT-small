def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 50
    population = []
    initial_h_half = h_full_initial[:half_n].copy()
    for _ in range(population_size):
        noise = np.random.normal(0, 0.1, half_n)
        perturbed = initial_h_half + noise
        # Project to constraints
        for _ in range(3):
            perturbed = np.clip(perturbed, 0, 1)
            s = np.sum(perturbed)
            if np.isclose(s, 0):
                perturbed = np.full_like(perturbed, target_sum_half / half_n)
                break
            scale = target_sum_half / s
            perturbed *= scale
            perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed.copy())
    population = np.array(population)

    best_c5 = np.inf
    best_h_half = None

    while time.time() - start_time < budget_s - 10:
        current_c5_list = [evaluate_c5(ind) for ind in population]
        current_min_c5 = np.min(current_c5_list)
        if current_min_c5 < best_c5:
            best_c5 = current_min_c5
            best_h_half = population[np.argmin(current_c5_list)].copy()

        new_population = []
        for _ in range(population_size):
            idx1, idx2 = np.random.choice(population_size, 2, replace=False)
            parent1_c5 = current_c5_list[idx1]
            parent2_c5 = current_c5_list[idx2]
            if parent1_c5 < parent2_c5:
                parent_idx = idx1
            else:
                parent_idx = idx2
            parent = population[parent_idx].copy()

            if np.random.rand() < 0.5:  # Crossover
                idx3, idx4 = np.random.choice(population_size, 2, replace=False)
                parent3_c5 = current_c5_list[idx3]
                parent4_c5 = current_c5_list[idx4]
                if parent3_c5 < parent4_c5:
                    parent3_idx = idx3
                else:
                    parent3_idx = idx4
                parent3 = population[parent3_idx]
                beta = np.random.beta(1, 1, size=half_n)
                child = beta * parent3 + (1 - beta) * parent
            else:
                noise = np.random.normal(0, 0.1, half_n)
                child = parent + noise

            # Project child to feasible region
            for _ in range(3):
                child = np.clip(child, 0, 1)
                s = np.sum(child)
                if np.isclose(s, 0):
                    child = np.full_like(child, target_sum_half / half_n)
                    break
                scale = target_sum_half / s
                child *= scale
                child = np.clip(child, 0, 1)
            new_population.append(child.copy())

        population = np.array(new_population)

        if time.time() - start_time > budget_s - 5:
            break

    best_h_full = create_full_h(best_h_half)
    best_h_full = np.clip(best_h_full, 0, 1)
    best_h_full_sum = np.sum(best_h_full)
    if not np.isclose(best_h_full_sum, n_points / 2.0, atol=1e-6):
        best_h_full *= (n_points / 2.0) / best_h_full_sum
        best_h_full = np.clip(best_h_full, 0, 1)

    return (best_h_full.tolist(), best_c5, n_points)
