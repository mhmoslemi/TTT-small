import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx

    np.random.seed(seed)
    population_size = 50
    max_generations = 600

    # Initialize population
    initial_sum = np.sum(h_initial)
    if initial_sum != 0:
        h_initial = (h_initial / initial_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)

    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        perturbed = h_initial.copy() + np.random.normal(0, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    generation = 0
    while time.time() - start_time < budget_s - 5 and generation < max_generations:
        generation += 1

        # Evaluate population
        candidates = [(compute_c5(h), h.copy()) for h in population]
        candidates.sort(key=lambda x: x[0])
        top_half = candidates[:population_size // 2]
        parents = [h for _, h in top_half]

        # Create new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            child = (p1 + p2) / 2.0
            if np.random.rand() < 0.2:
                decay = 1.0 - generation / max_generations
                child += np.random.normal(0, 0.5 * decay, n_points)
                child = np.clip(child, 0, 1)
            total = np.sum(child)
            if total != 0:
                child = (child / total) * target_sum
            else:
                child = np.full(n_points, target_sum / n_points)
            child = np.clip(child, 0, 1)
            new_population.append(child)

        population = new_population

        # Update best solution
        if candidates[0][0] < best_c5:
            best_c5 = candidates[0][0]
            best_h = candidates[0][1].copy()

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
