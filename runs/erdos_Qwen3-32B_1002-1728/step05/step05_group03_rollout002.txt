import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def normalize_h(h, target_sum):
        h_copy = h.copy()
        current_sum = np.sum(h_copy)
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = np.sum(h_clipped)
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    def evaluate_h(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    initial_h_half = normalize_h(h_full_initial[:half_n], target_sum_half)
    num_particles = 50
    swarm_positions = np.array([initial_h_half + 0.1 * np.random.randn(half_n) for _ in range(num_particles)])
    swarm_positions = np.clip(swarm_positions, 0, 1)
    for i in range(num_particles):
        swarm_positions[i] = normalize_h(swarm_positions[i], target_sum_half)

    velocities = 0.1 * np.random.randn(num_particles, half_n)
    personal_best_positions = swarm_positions.copy()
    personal_best_scores = np.array([evaluate_h(p) for p in swarm_positions])
    best_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[best_idx].copy()
    global_best_score = personal_best_scores[best_idx]

    max_iterations = 10000
    inertia_start = 0.7
    inertia_end = 0.1
    c1, c2 = 2.0, 2.0

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        t = (time.time() - start_time) / budget_s
        inertia = inertia_start - t * (inertia_start - inertia_end)
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - swarm_positions[i]) +
                             c2 * r2 * (global_best_position - swarm_positions[i]))
            velocities[i] = np.clip(velocities[i], -0.2, 0.2)
            swarm_positions[i] += velocities[i]
            swarm_positions[i] = np.clip(swarm_positions[i], 0, 1)
            swarm_positions[i] = normalize_h(swarm_positions[i], target_sum_half)
            current_score = evaluate_h(swarm_positions[i])
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best_positions[i] = swarm_positions[i].copy()
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best_position = swarm_positions[i].copy()
        best_idx = np.argmin(personal_best_scores)
        global_best_position = personal_best_positions[best_idx].copy()
        global_best_score = personal_best_scores[best_idx]

    h_full = create_full_h(global_best_position)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
