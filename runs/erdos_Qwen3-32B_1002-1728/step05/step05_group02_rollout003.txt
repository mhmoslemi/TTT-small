import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize the initial h to meet the sum constraint

    if abs(current_sum - target_sum) > 1e-10:
        scale = target_sum / current_sum
        h_initial = np.clip(h_initial * scale, 0, 1)

    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 5:
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.35 * (1.0 - frac**2.5)
        num_perturb = 2

        for _ in range(num_perturb):
            perturbed = best_h + np.random.normal(0, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum_p = perturbed.sum()
            if current_sum_p != 0:
                perturbed = (perturbed / current_sum_p) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            method = ['SLSQP', 'L-BFGS-B', 'COBYLA'][int(frac * 3) % 3]

            result = minimize(
                fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                x0=perturbed,
                method=method,
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-10, 'disp': False}
            )

            if result.success:
                new_h = np.clip(result.x, 0, 1)
                delta_sum = new_h.sum() - target_sum
                if abs(delta_sum) > 1e-6:
                    scale = target_sum / new_h.sum()
                    new_h = (new_h * scale).clip(0, 1)
                new_corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(new_corr) * dx
                if new_c5 < best_c5:

    # Final check and fix
    best_h = np.clip(best_h, 0, 1)
