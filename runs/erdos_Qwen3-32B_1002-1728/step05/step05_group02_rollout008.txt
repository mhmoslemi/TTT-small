            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best_positions[i] - positions[i]) +
                             c2 * r2 * (global_best_position - positions[i]))
            new_position = positions[i] + velocities[i]
            new_position = project_h_half(new_position, target_sum_half)
            new_score = objective(new_position, half_n, n_points, dx)
            if new_score < personal_best_scores[i]:
                personal_best_positions[i] = new_position.copy()
                personal_best_scores[i] = new_score
            if new_score < global_best_score:
                global_best_position = new_position.copy()
                global_best_score = new_score
            positions[i] = new_position

    # SQP Refinement
    top_ratio = 0.2
    top_count = int(n_particles * top_ratio)
    top_indices = np.argsort(personal_best_scores)[:top_count]
    top_positions = [personal_best_positions[i] for i in top_indices]
    best_solutions = []

    for hp in top_positions:
        if time.time() - start_time >= budget_s:
            break
        def sqp_objective(x):
            h_full = np.zeros(n_points)
            for i in range(len(x)):
                h_full[i] = h_full[n_points-1-i] = x[i]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        bounds = [(0.0, 1.0)] * half_n
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
        x0 = hp.copy()
        result = minimize(
            sqp_objective, x0=x0, method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-12, 'disp': False})
        if result.success:
            new_h_half = np.clip(result.x, 0, 1)
            new_h_half = project_h_half(new_h_half, target_sum_half)
            h_full = np.zeros(n_points)
            for i in range(half_n):
                h_full[i] = h_full[n_points-1-i] = new_h_half[i]
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            new_score = np.max(corr) * dx
            best_solutions.append((new_score, new_h_half))

    if best_solutions:
        best_solutions.sort()
        best_score, best_h_half = best_solutions[0]
    else:
        best_score = global_best_score
        best_h_half = global_best_position

    # Construct and validate final solution
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points-1-i] = best_h_half[i]
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), best_score, n_points)

def objective(h_vars, half_n, n_points, dx):
    h_full = np.zeros(n_points)
    for i in range(len(h_vars)):
        h_full[i] = h_full[n_points-1-i] = h_vars[i]
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
