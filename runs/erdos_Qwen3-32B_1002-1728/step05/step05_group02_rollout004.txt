from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        scale = target_sum / current_sum
        h_full = h_full * scale
    h_full = np.clip(h_full, 0, 1)

    best_h = h_full.copy()

    np.random.seed(seed)

        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.4 * (1 - frac**2)  # Increased initial noise for stronger exploration

        perturb_count = 2  # Adjusted perturb count for resource balance
        for _ in range(perturb_count):

            perturbed = np.clip(perturbed, 0, 1)

            perturbed = np.clip(perturbed, 0, 1)

            method = ['SLSQP', 'COBYLA', 'L-BFGS-B'][n_attempt % 3]
            result = minimize(
