import numpy as np
import time

def compute_gradient(h, k, n_points):
    dx = 2.0 / n_points
    grad = np.zeros_like(h)
    for j in range(n_points):
        part1 = 0.0
        idx_plus = j + k
        if 0 <= idx_plus < n_points:
            part1 = 1.0 - h[idx_plus]
        part2 = 0.0
        idx_minus = j - k
        if 0 <= idx_minus < n_points:
            part2 = -h[idx_minus]
        grad[j] = (part1 + part2) * dx
    return grad

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
