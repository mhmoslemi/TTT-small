                r1, r2 = np.random.rand(2)
                velocities[i] = (w * velocities[i] +
                                 c1 * r1 * (pbest_positions[i] - positions[i]) +
                                 c2 * r2 * (gbest_position - positions[i]))
                positions[i] += velocities[i]
                positions[i] = np.clip(positions[i], 0, 1)
                current_sum = np.sum(positions[i])
                if current_sum != 0:
                    positions[i] = (positions[i] * (target_sum / current_sum))
                    positions[i] = np.clip(positions[i], 0, 1)

            # Update pbest and gbest
            current_values = np.array([compute_c5(p) for p in positions])
            improved = current_values < pbest_values
            pbest_values[improved] = current_values[improved]
            pbest_positions[improved] = positions[improved]
            new_gbest_idx = np.argmin(pbest_values)
            if pbest_values[new_gbest_idx] < gbest_value:
                gbest_value = pbest_values[new_gbest_idx]
                gbest_position = pbest_positions[new_gbest_idx]

        # Local refinement
        top_indices = np.argsort(pbest_values)[:5]
        for idx in top_indices:
            if time.time() - start_time >= budget_s - 60:
                break
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
            res = minimize(
                compute_c5, x0=pbest_positions[idx], method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
            )
            if res.success:
                new_h = np.clip(res.x, 0, 1)
                new_c5 = compute_c5(new_h)
                if new_c5 < gbest_value:
                    gbest_value = new_c5
                    gbest_position = new_h.copy()

    final_h = gbest_position
    final_h = np.clip(final_h, 0, 1)
    assert abs(np.sum(final_h) - target_sum) < 1e-6
    return (final_h.tolist(), gbest_value, n_points)
