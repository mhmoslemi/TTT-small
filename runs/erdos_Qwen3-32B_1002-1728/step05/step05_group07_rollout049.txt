    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1

        # Evaluate candidates
        candidates = [ (compute_c5(*params), params) for params in population ]
        candidates.sort(key=lambda x: x[0])
        top_c5 = candidates[0][0]
        if top_c5 < best_c5:
            best_c5 = top_c5
            best_params = candidates[0][1]

        # Select parents from top half
        keep = int(population_size / 2)
        parents = [ params for c, params in candidates[:keep] ]

        # Create new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            p1, p2 = np.random.choice(parents, 2).tolist()
            # Crossover
            c_new = (p1[0] + p2[0]) / 2
            w_new = (p1[1] + p2[1]) / 2
            a_new = (p1[2] + p2[2]) / 2

            # Mutation
            if np.random.rand() < 0.2:
                mutation_rate = 0.3 * (1 - generation/max_generations)**mutation_decay
                c_new += np.random.normal(0, mutation_rate, K)
                w_new += np.random.normal(0, mutation_rate, K)
                a_new += np.random.normal(0, mutation_rate, K)

            # Enforce constraints
            c_new = np.clip(c_new, 0.0, 2.0)
            w_new = np.clip(w_new, 0.0, 1.0)
            a_new = np.clip(a_new, 0.0, 1.0)
            
            for i in range(K):
                c_new[i] = min(max(c_new[i], w_new[i]), 2.0 - w_new[i])
                w_new[i] = min(w_new[i], min(c_new[i], 2.0 - c_new[i]))

            new_population.append( (c_new, w_new, a_new) )

        population = new_population

    h_final = compute_h(*best_params)
    return (h_final.tolist(), best_c5, n_points)
