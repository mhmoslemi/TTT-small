        best_h_full = h_full.copy()

    assert np.allclose(np.sum(best_h_full), target_sum, atol=1e-6), "Sum constraint violated"
    assert np.all((best_h_full >= 0) & (best_h_full <= 1 + 1e-6)), "Bounds violated"

    return (best_h_full.tolist(), best_c5, n_points)
