import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_full = np.reshape(h_vars, (n_points,))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    # Prepare initial h
    h_full_initial = h_full_initial.copy()
    current_sum = np.sum(h_full_initial)
    if current_sum != 0:
        h_full_initial = (h_full_initial / current_sum) * target_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)

    best_h = h_full_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        time_remaining = budget_s - elapsed
        frac = time_remaining / budget_s
        noise_scale = 0.4 * (1 - frac**2)

        # Perturbation and optimization
        seed += 1
        rnd = np.random.RandomState(seed)
        perturbed = best_h + rnd.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum
        perturbed = np.clip(perturbed, 0, 1)

        methods = ['L-BFGS-B', 'SLSQP', 'COBYLA']
        method_idx = (n_attempt % len(methods))
        result = minimize(
            fun=objective,
            x0=perturbed,
            method=methods[method_idx],
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 40, 'ftol': 1e-12, 'disp': False}
        )

        # Update best solution if improved
        if result.success:
            new_h = np.clip(result.x, 0, 1)
            if abs(np.sum(new_h) - target_sum) < 1e-6:
                new_c5 = objective(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    # Final check
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
