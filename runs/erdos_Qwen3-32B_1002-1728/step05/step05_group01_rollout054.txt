import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_full = h_vars.copy()
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Project initial h to feasible set
    current_sum = np.sum(h_initial)
    if current_sum != 0:
        scale = target_sum / current_sum
        h_initial_proj = np.clip(h_initial * scale, 0, 1)
    else:
        h_initial_proj = np.zeros(n_points)
        h_initial_proj[0] = target_sum  # so sum is target_sum

    # Ensure sum is correct
    h_initial_proj = h_initial_proj / np.sum(h_initial_proj) * target_sum
    h_initial_proj = np.clip(h_initial_proj, 0, 1)

    best_h = h_initial_proj.copy()
    best_c5 = objective(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s:
        n_attempt += 1
        elapsed = time.time() - start_time
        noise_scale = 0.3 * (1 - (elapsed / budget_s)**0.5)

        for _ in range(2):  # Reduced attempts per iteration
            idx = np.random.permutation(n_points)[:int(n_points ** 0.5)]
            perturbed = best_h.copy()
            perturbed[idx] += np.random.uniform(-noise_scale, noise_scale, size=len(idx))
            perturbed = np.clip(perturbed, 0, 1)
            total = perturbed.sum()
            if abs(total) < 1e-10:
                continue
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            # Try multiple optimizers with adjusted tolerances
            methods = ['COBYLA', 'L-BFGS-B', 'Nelder-Mead']
            for method in methods:
                result = minimize(
                    fun=lambda x: objective(x) + 1e-8 * np.random.randn(),  # Stochastic perturbation
                    x0=perturbed,
                    method=method,
                    bounds=bounds,
                    constraints=constraints if method not in ['Nelder-Mead'] else (),
                    options={'maxiter': 100, 'ftol': 1e-12, 'eps': 1e-8}
                )
                if result.success:
                    new_h = np.clip(result.x, 0, 1)
                    new_h = new_h / sum(new_h) * target_sum  # Final projection
                    new_h = np.clip(new_h, 0, 1)
                    h_full = new_h
                    corr = np.correlate(h_full, 1 - h_full, mode='full')

                        best_h[:] = h_full
                        best_c5 = new_c5
                        break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-7) & (best_h <= 1.0 + 1e-7))

    return (best_h.tolist(), best_c5, n_points)
