import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_vars):
        h_full = np.clip(h_vars, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_h = h_full_initial.copy()
    best_h_sum = np.sum(best_h)
    if best_h_sum > 1e-10:
        best_h = np.clip(best_h * target_sum / best_h_sum, 0, 1)
    best_c5 = objective(best_h)

    while time.time() - start_time < 1100 - 30:
        elapsed = time.time() - start_time
        frac = elapsed / 1100
        noise_scale = 0.3 * (1 - (frac ** 0.75)) if frac < 1.0 else 0.0

        for _ in range(2):
            if frac > 0.95:
                perturbed = best_h + np.random.uniform(-noise_scale*0.5, noise_scale*0.5, n_points)
            else:
                perturbed = best_h + np.random.uniform(-noise_scale, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if abs(perturbed_sum) < 1e-10:
                continue
            perturbed = np.clip(perturbed * target_sum / perturbed_sum, 0, 1)

            for method in ['SLSQP', 'L-BFGS-B', 'COBYLA']:
                do_constraints = (method in ['SLSQP', 'COBYLA'])
                cons = [constraints] if do_constraints else []
                options = {'maxiter': 100, 'ftol': 1e-11, 'disp': False}
                result = minimize(
                    fun=objective,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons,
                    options=options
                )
                if result.success:
                    new_h = np.clip(result.x, 0, 1)
                    new_sum = np.sum(new_h)
                    if abs(new_sum - target_sum) > 1e-6:
                        new_h = np.clip(new_h * target_sum / new_sum, 0, 1)
                    new_c5 = objective(new_h)
                    if np.isfinite(new_c5) and new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5

    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-6:
        best_h = np.clip(best_h * target_sum / final_sum, 0, 1)
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
