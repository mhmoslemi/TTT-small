    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    if total <= 1e-12:
        h_scaled = np.zeros_like(h)
    else:
        h_scaled = h_clipped * (target_sum / total)
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    n_particles = 20
    max_iter_pso = 100
    w, c1, c2 = 0.72984, 1.49445, 1.49445  # Van Veen parameters
    
    np.random.seed(seed)
    particles = []
    velocities = []
    for _ in range(n_particles):
        noise = 0.15 * np.random.randn(n_points)
        p = h_initial + noise
        p = enforce_constraints(p, target_sum)
        particles.append(p)
        velocities.append(0.01 * np.random.randn(n_points))
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    pbest_positions = particles.copy()
    pbest_scores = np.array([evaluate_objective(p) for p in particles])
    idx_gb = np.argmin(pbest_scores)
    gbest_position = pbest_positions[idx_gb].copy()
    gbest_score = pbest_scores[idx_gb]
    
    for iter in range(max_iter_pso):
        if time.time() - start_time > budget_s - 60:
            break
        
        r1 = np.random.rand(n_particles, 1)
        r2 = np.random.rand(n_particles, 1)
        velocities = (
            w * velocities +
            c1 * r1 * (pbest_positions - particles) +
            c2 * r2 * (np.tile(gbest_position[None, :], (n_particles, 1)) - particles)
        )
        particles += velocities
        
        for i in range(n_particles):
            particles[i] = enforce_constraints(particles[i], target_sum)
        
        scores = np.array([evaluate_objective(p) for p in particles])
        pbest_mask = scores < pbest_scores
        pbest_scores[pbest_mask] = scores[pbest_mask]
        pbest_positions[pbest_mask] = particles[pbest_mask]
        
        new_pbest_min = np.min(pbest_scores)
        if new_pbest_min < gbest_score:
            gbest_idx = np.argmin(pbest_scores)
            gbest_position = pbest_positions[gbest_idx].copy()
            gbest_score = new_pbest_min
    
    top_count = max(1, int(n_particles * 0.25))
    top_indices = np.argpartition(pbest_scores, top_count)[:top_count]
    candidates = pbest_positions[top_indices]
    best_h = gbest_position.copy()
    best_c5 = gbest_score
    
    for candidate in candidates:
        def objective(x):
            h = x.reshape(n_points)
            h_clipped = np.clip(h, 0, 1)
            total = np.sum(h_clipped)
            h_scaled = h_clipped * (target_sum / total) if total > 1e-12 else np.zeros_like(h)
            h_scaled = np.clip(h_scaled, 0, 1)
            return evaluate_objective(h_scaled)
        
        bounds = [(0, 1)] * n_points
        result = minimize(
            objective, 
            x0=candidate.copy().ravel(), 
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 40, 'ftol': 1e-10, 'eps': 1e-4}
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            final_h = enforce_constraints(h_opt, target_sum)
            new_c5 = evaluate_objective(final_h)
            if new_c5 < best_c5:
                best_h = final_h
                best_c5 = new_c5
    
    final_sum = np.sum(best_h)
    assert abs(final_sum - target_sum) < 1e-6, f"Sum constraint violated: {final_sum} vs {target_sum}"
    best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), best_c5, n_points)
