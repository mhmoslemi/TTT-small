import numpy as np
import time

def compute_c5(h_full, dx):
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    # Initialize symmetric h_full from initial values
    h_full = h_initial.copy()
    h_full[:half_n] = (h_full[:half_n] + h_full[-half_n:][::-1]) / 2
    h_full[-half_n:] = h_full[:half_n][::-1]
    # Normalize to meet sum constraint
    current_sum = np.sum(h_full)
    if current_sum != 0:
        h_full = (h_full / current_sum) * target_sum
    h_full = np.clip(h_full, 0.0, 1.0)

    best_h_half = h_half_current.copy()

    T = 10.0

    np.random.seed(seed)
