    while current_sum != target_sum:
        delta = current_sum - target_sum
        if delta > 0:
            mask = binary_h == 1
            idxs = np.where(mask)[0]
            if len(idxs) == 0:
                break
            i = np.random.choice(idxs)
            binary_h[i] = 0
            current_sum -= 1
        else:
            mask = binary_h == 0
            idxs = np.where(mask)[0]
            if len(idxs) == 0:
                break
            i = np.random.choice(idxs)
            binary_h[i] = 1
            current_sum += 1
    population = [binary_h]

    # Generate initial population with random swaps
    np.random.seed(seed)
    for _ in range(49):
        ind = binary_h.copy()
        for _ in range(10):
            i, j = np.random.choice(n_points, 2, replace=False)
            if ind[i] != ind[j]:
                ind[i], ind[j] = ind[j], ind[i]
        population.append(ind)

    best_ind = binary_h.copy()
    best_c5 = float('inf')

    def evaluate(ind):
        h = ind.astype(float)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    while time.time() - start_time < budget_s - 10:
        # Evaluate current best
        for ind in population:
            c5 = evaluate(ind)
            if c5 < best_c5:
                best_c5 = c5
                best_ind = ind.copy()

        # Select parents (top half)
        population = sorted(population, key=evaluate)
        selected = population[:len(population)//2]

        # Generate new population
        new_pop = selected.copy()
        while len(new_pop) < len(population):
            p1, p2 = np.random.choice(selected, 2, replace=False)
            split = np.random.randint(1, n_points)
            child = np.concatenate([p1[:split], p2[split:]]).copy()
            
            # Repair sum constraint
            delta = int(np.sum(child) - target_sum)
            if delta != 0:
                if delta > 0:
                    idxs = np.where(child == 1)[0]
                    repair = np.random.choice(idxs, min(len(idxs), abs(delta)), replace=False)
                    child[repair] = 0
                else:
                    idxs = np.where(child == 0)[0]
                    repair = np.random.choice(idxs, min(len(idxs), abs(delta)), replace=False)
                    child[repair] = 1
            new_pop.append(child)
        
        # Mutation
        for ind in new_pop:
            if np.random.rand() < 0.2:
                i, j = np.random.choice(n_points, 2, replace=False)
                if ind[i] != ind[j]:
                    ind[i], ind[j] = ind[j], ind[i]
        population = new_pop

    return (best_ind.astype(float).tolist(), best_c5, n_points)
