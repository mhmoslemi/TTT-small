import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    np.random.seed(seed)
    population_size = 50
    pop = []

    # Initialize population
    initial_half = h_full_initial[:half_n].copy()
    for _ in range(population_size):
        perturbed = initial_half + np.random.normal(0, 0.1, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = max(np.sum(perturbed), 1e-9)
        perturbed *= target_sum_half / total
        perturbed = np.clip(perturbed, 0, 1)
        pop.append(perturbed.copy())
    pop = np.array(pop)

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    best_c5 = float('inf')
    best_h_half = initial_half.copy()

    while time.time() - start_time < budget_s - 5:
        # Evaluate population
        scores = np.array([evaluate_c5(p) for p in pop])
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h_half[:] = pop[best_idx].copy()

        # Binary tournament selection
        parents = []
        for _ in range(population_size // 2):
            candidates = np.random.choice(population_size, 2)
            p1, p2 = candidates[0], candidates[1]
            better = p1 if scores[p1] < scores[p2] else p2
            parents.append(pop[better].copy())

        # Crossover + Mutation
        offspring = []
        for p1, p2 in zip(parents[::2], parents[1::2]):
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2

            # Mutation
            mutation = np.random.normal(0, 0.1, half_n)
            child += mutation
            child = np.clip(child, 0, 1)
            total = max(np.sum(child), 1e-9)
            child *= target_sum_half / total
            child = np.clip(child, 0, 1)
            offspring.append(child)

        # Generate additional diversity
        for _ in range(population_size // 2):
            child = initial_half.copy()
            mutation = np.random.normal(0, 0.2, half_n)
            child += mutation
            child = np.clip(child, 0, 1)
            total = max(np.sum(child), 1e-9)
            child *= target_sum_half / total
            child = np.clip(child, 0, 1)
            offspring.append(child)

        pop = np.concatenate([pop, np.array(offspring)], axis=0)
        # Truncate to population_size best
        scores = np.array([evaluate_c5(p) for p in pop])
        best_indices = np.argsort(scores)[:population_size]
        pop = pop[best_indices]

    # Final validation
    best_h_full = create_full_h(best_h_half)
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    assert np.abs(np.sum(best_h_full) - n_points / 2.0) < 1e-6

    return (best_h_full.tolist(), best_c5, n_points)
