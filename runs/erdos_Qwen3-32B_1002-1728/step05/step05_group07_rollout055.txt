    if not np.isclose(current_sum, n_points / 4):
        best_h_half = best_h_half * (n_points / 4) / current_sum
    best_h_half = np.clip(best_h_half, 0, 1)
    
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = best_h_half[i]
    
    final_c5 = compute_c5(best_h_half)
    
    # Early stopping if remaining time < 5s
    if time.time() - start_time < budget_s - 5:
        pass  # Continue optimization if time permits
        
    return (h_full.tolist(), final_c5, n_points)
