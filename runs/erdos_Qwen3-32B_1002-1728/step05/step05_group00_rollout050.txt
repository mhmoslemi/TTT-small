import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def process_h(h_candidate):
        h_clipped = np.clip(h_candidate, 0, 1)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return np.full_like(h_candidate, 0.5)
        scale = target_sum / sum_clipped
        h_scaled = h_clipped * scale
        return np.clip(h_scaled, 0, 1)

    def objective(h_vars):
        h_processed = process_h(h_vars)
        corr_vals = np.correlate(h_processed, 1 - h_processed, mode='full')
        return np.max(corr_vals) * dx

    h_best = process_h(h_initial.copy())
    c5_best = objective(h_best)

    # Differential Evolution setup with full h vector
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    np.random.seed(seed)
    de_result = differential_evolution(
        objective, bounds, strategy='best1bin', maxiter=20, popsize=15,
        mutation=(0.5, 1), recombination=0.9, seed=seed, workers=1, disp=False
    )
    if de_result.success:
        h_de = process_h(de_result.x)
        c5_de = objective(h_de)
        if c5_de < c5_best:
            h_best = h_de
            c5_best = c5_de

    # SLSQP refinement with equality constraint
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds_per_var = [(0, 1) for _ in range(n_points)]
    slsqp_result = minimize(
        objective, h_best.copy(), method='SLSQP', bounds=bounds_per_var,
        constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8}
    )
    if slsqp_result.success:
        h_slsqp = process_h(slsqp_result.x)
        c5_slsqp = objective(h_slsqp)
        if c5_slsqp < c5_best:
            h_best, c5_best = h_slsqp, c5_slsqp

    # Perturbation and local search iteration
    while time.time() - start_time < budget_s - 10:
        seed = seed + 1
        np.random.seed(seed)
        idxs = np.random.choice(n_points, size=int(n_points * 0.25), replace=False)
        perturbed = h_best.copy()
        perturbed[idxs] += np.random.normal(0, 0.35, len(idxs))
        perturbed_clipped = np.clip(perturbed, 0, 1)
        perturbed = process_h(perturbed_clipped)

        slsqp_result_perturbed = minimize(

            constraints=constraints, options={'maxiter': 150, 'ftol': 1e-8}
        )
        if slsqp_result_perturbed.success:
            h_new = process_h(slsqp_result_perturbed.x)
            c5_new = objective(h_new)
            if c5_new < c5_best:
                h_best, c5_best = h_new, c5_new

    return (h_best.tolist(), c5_best, n_points)
