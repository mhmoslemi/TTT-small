import time

    n_points = len(h_full_initial)

    target_sum = n_points / 2.0

    h_full_initial = h_full_initial.copy()
    current_sum = h_full_initial.sum()

        h_full_initial = np.clip(h_full_initial, 0, 1)

    H_imag = H_initial.imag.copy()

        c5 = np.max(corr) * dx
        return c5, h

    best_c5, best_h = compute_C5(best_H_real, best_H_imag)
