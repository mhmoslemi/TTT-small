import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_full = h_full_initial.copy()
    total = np.sum(h_full)
    if total != 0:
        scale = target_sum / total
        h_full = h_full * scale
    h_full = np.clip(h_full, 0, 1)

    best_h = h_full.copy()
    best_c5 = objective(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        current_time = time.time() - start_time
        if current_time == 0:
            frac = 0.0
        else:
            frac = current_time / budget_s
        noise_scale = 0.3 * (1 - frac ** 1.6)

        for _ in range(3):
            seed += 1
            rnd = np.random.RandomState(seed)
            perturbed = best_h + rnd.normal(0, noise_scale, n_points)
            total_perturbed = np.sum(perturbed)
            if total_perturbed != 0:
                perturbed *= target_sum / total_perturbed
            perturbed = np.clip(perturbed, 0, 1)

            # Alternate between SLSQP and COBYLA based on attempt parity
            method = 'COBYLA' if n_attempt % 2 == 0 else 'SLSQP'
            result = minimize(
                fun=objective,
                x0=perturbed,
                method=method,
                bounds=bounds,
                constraints=cons if method != 'Nelder-Mead' else [],
                options={
                    'maxiter': 80,
                    'ftol': 1e-6,
                    'disp': False
                }
            )

            if result.success and result.fun < best_c5:
                new_h = np.clip(result.x, 0, 1)
                if abs(np.sum(new_h) - target_sum) < 1e-6:
                    new_c5 = objective(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5

    h_full = best_h.copy()
    return (h_full.tolist(), best_c5, n_points)
