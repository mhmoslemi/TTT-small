def check_feasibility(target_c5, dx, n_points):
    h = cp.Variable(n_points, nonneg=True)  # 0 <= h <=1
    constraints = [
        cp.sum(h) == n_points / 2.0,
        h >= 0,
        h <= 1
    ]

    for k in range(-n_points+1, n_points):
        # For this shift k, determine the valid i's and j = i +k
        valid_is = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_is.append(i)
        if not valid_is:  # no contribution to the integral for this k
            continue
        # For each i in valid_is, add z_i as variable
        z_vars = []
        for i in valid_is:
            j = i + k
            z = cp.Variable(nonneg=True)  # 0 <= z since McCormick bounds are 0 <= z <= h[i]
            # McCormick constraints
            constraints += [
                z >= h[i] + h[j] - 1,
                z <= h[i],
                z <= h[j],
                z >= 0
            ]

    prob = cp.Problem(objective=cp.Minimize(0), constraints=constraints)

    feasible = prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]
    if feasible:
        # Return h value
        return feasible, h.value
    else:
        return False, None
