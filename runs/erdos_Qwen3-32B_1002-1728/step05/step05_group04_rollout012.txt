import numpy as np

import time

def run(seed=42, budget_s=1100, **kwargs):

    def objective(h_full):
        h_full = np.clip(h_full, 0, 1).copy()
        total = np.sum(h_full)
        if total != 0:
            h_full *= target_sum / total
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * n_points
    np.random.seed(seed)
    
    # Create initial population from perturbed initial guess
    pop_size = max(20, n_points // 5)
    init_pop = [np.clip(h_full_initial + np.random.normal(0, 0.1, n_points), 0, 1) for _ in range(pop_size)]
    init_pop = np.array(init_pop)
    
    # Normalize and clip initial population
    for i in range(init_pop.shape[0]):
        total = np.sum(init_pop[i])
        if total > 1e-9:
            init_pop[i] *= target_sum / total
        init_pop[i] = np.clip(init_pop[i], 0, 1)
