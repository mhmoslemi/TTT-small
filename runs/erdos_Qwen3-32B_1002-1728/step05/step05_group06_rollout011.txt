import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum of h_full must be this

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize symmetric and normalized h_full
    h_full = h_initial.copy()
    for i in range(half_n):
        avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
        h_full[i] = h_full[n_points - 1 - i] = avg
    current_sum = np.sum(h_full)
    if current_sum != 0:
        h_full = (h_full / current_sum) * target_sum
    h_full = np.clip(h_full, 0.0, 1.0)
    h_half_current = h_full[:half_n].copy()
    target_sum_half = target_sum / 2.0  # sum of first half

    best_c5 = compute_c5(h_full)
    best_h_half = h_half_current.copy()

    # Optimization parameters
    learning_rate = 5e-3
    decay_rate = 0.9995
    max_iterations = 50000
    iteration = 0
    np.random.seed(seed)

    while time.time() - start_time < budget_s and iteration < max_iterations:
        h_full = create_full_h(h_half_current)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        k_max = np.argmax(corr)
        k_shift = k_max - (n_points - 1)

        # Compute gradient_full
        gradient_full = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                gradient_full[i] = 1.0 - h_full[j]
            else:
                gradient_full[i] = 1.0  # 1 - h[j] = 1 if j out of bounds

        # Compute gradient for h_half using symmetry
        gradient_half = np.zeros(half_n)
        for i in range(half_n):
            mirror_idx = n_points - 1 - i
            gradient_half[i] = gradient_full[i] + gradient_full[mirror_idx]

        # Gradient descent step
        h_half_current -= learning_rate * gradient_half
        h_half_current = np.clip(h_half_current, 0.0, 1.0)
        sum_half = np.sum(h_half_current)
        if sum_half > 1e-8:

        # Enforce symmetry and normalization
        h_full = create_full_h(h_half_current)
