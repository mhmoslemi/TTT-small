def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def get_initial_params(H):
        params = []
        mid = n_points // 2
        for k in range(mid + 1):
            if k == 0 or k == mid:
                params.append(H[k].real)
            else:
                params.append(H[k].real)
                params.append(H[k].imag)
        return np.array(params)

    def construct_H_from_params(params):
        H = np.zeros(n_points, dtype=np.complex128)
        idx = 0
        # Handle k=0
        H[0] = complex(params[idx], 0)
        idx += 1

        # Handle k=1 to mid-1
        for k in range(1, n_points // 2):
            if idx + 1 >= len(params):
                break
            real = params[idx]
            imag = params[idx + 1]
            H[k] = complex(real, imag)
            H[n_points - k] = complex(real, -imag)
            idx += 2

        # Handle mid if even
        if n_points % 2 == 0 and idx < len(params):
            H[n_points // 2] = complex(params[idx], 0)
            idx += 1
        return H

    H_initial = np.fft.fft(h_initial)
    params_initial = get_initial_params(H_initial)

    best_C5 = float('inf')
    best_h = h_initial.copy()

    def objective(params):
        H = construct_H_from_params(params)
        h_fft = np.fft.ifft(H).real
        h_clip = np.clip(h_fft, 0, 1)
        sum_h = h_clip.sum()
        if sum_h < 1e-10:
            return np.inf
        h_proj = h_clip * (n_points / 2.0 / sum_h)
        corr = np.correlate(h_proj, 1 - h_proj, mode='full')
        return np.max(corr) * dx

    # First optimization with initial guess
    res = minimize(objective, x0=params_initial, method='L-BFGS-B', options={'maxiter': 200})
    if res.success:
        H = construct_H_from_params(res.x)
        h = np.fft.ifft(H).real
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0 / h.sum()) if h.sum() > 1e-10 else h
        corr = np.correlate(h, 1 - h, mode='full')
        val = np.max(corr) * dx
        if val < best_C5:
            best_C5 = val
            best_h = h

    # Multiple random restarts with differential evolution
    np.random.seed(seed)
    bounds = [(-1.0, 1.0)] * len(params_initial)
    tries = 0
    while time.time() - start_time < budget_s - 30:
        dx = np.random.rand(*params_initial.shape) * 2 - 1
        init = params_initial + dx * 0.1 * tries
        res = differential_evolution(objective, bounds, maxiter=50, workers=1, init=init)
        if res.success:
            H = construct_H_from_params(res.x)
            h = np.fft.ifft(H).real
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0 / h.sum()) if h.sum() > 1e-10 else h
            corr = np.correlate(h, 1 - h, mode='full')
            val = np.max(corr) * dx
            if val < best_C5:
                best_C5 = val
                best_h = h
        tries += 1

    return (best_h.tolist(), best_C5, n_points)
