        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_full = [(0.0, 1.0) for _ in range(n_points)]
    cons_full = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = 0.35 * (1 - frac**2)

        seed += 1
        rnd = np.random.RandomState(seed)
        perturbed = best_h + rnd.normal(0, noise_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)

        method_opt = ['SLSQP', 'COBYLA'][n_attempt % 2]
        result = minimize(
            fun=objective,
            x0=perturbed,
            method=method_opt,
            bounds=bounds_full,
            constraints=cons_full,
            options={'maxiter': 30, 'ftol': 1e-8, 'disp': False}
        )

        if result.success:
            new_h = np.clip(result.x, 0, 1)
            current_sum = np.sum(new_h)
            if abs(current_sum - target_sum) < 1e-6:
                corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
