import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_h = h_initial.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        elapsed = time.time() - start_time
        progress = elapsed / budget_s
        noise_scale = 0.4 * (1.0 - progress**3)

        for _ in range(1):  # Reduce perturbation count for time efficiency
            seed += 1
            rnd = np.random.RandomState(seed)
            # Apply non-symmetric noise
            perturbed = best_h + rnd.normal(0, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if abs(total) > 1e-9:  # Avoid division by zero
                perturbed = (perturbed / total) * target_sum
                perturbed = np.clip(perturbed, 0, 1)

            # Alternate between SLSQP and COBYLA
            method = 'SLSQP' if n_attempt % 2 == 0 else 'COBYLA'
            result = minimize(
                fun=objective,
                x0=perturbed,
                method=method,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 30 if method == 'COBYLA' else 50, 'ftol': 1e-12, 'disp': False}
            )

            if result.success:
                new_h = np.clip(result.x, 0, 1)
                sum_check = np.sum(new_h)
                c5_new = objective(new_h)
                # Only replace if sum is valid and lower c5
                if abs(sum_check - target_sum) < 1e-6 and c5_new < best_c5:
                    best_h = new_h
                    best_c5 = c5_new

    # Final validation
    sum_h = np.sum(best_h)
    assert abs(sum_h - target_sum) < 1e-6, f"Sum constraint violated: {sum_h} vs {target_sum}"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
