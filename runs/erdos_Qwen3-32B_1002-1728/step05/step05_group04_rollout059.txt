                new_sum = np.sum(new_h)
                if new_sum != 0:
                    new_h = new_h * (target_sum / new_sum)
                new_h = np.clip(new_h, 0, 1)
                corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < current_c5:
                    best_h = new_h
                    current_c5 = new_c5
                    time_used = time.time() - start_time
                    if time_used > budget_s - 30:
                        break

    return (best_h.tolist(), current_c5, n_points)
