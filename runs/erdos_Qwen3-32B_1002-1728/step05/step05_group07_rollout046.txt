                term1 = 1.0
            
            # Compute term2
            if 0 <= m < n_points:
                term2 = h[m]
            else:
                term2 = 0.0
            
            grad = term1 - term2
            
            # Momentum update
            velocity[i] = momentum * velocity[i] + grad
            h[i] += current_lr * velocity[i]
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
            h = (h / current_sum) * target_sum
        
        # Evaluate and update best
        corr = np.correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h[:] = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
