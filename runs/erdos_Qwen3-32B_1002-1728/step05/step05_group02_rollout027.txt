from scipy.optimize import minimize

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    s = h_clipped.sum()
    if s == 0:
        return np.ones_like(h) * 0.5
    h_proj = h_clipped * (target_sum / s)
    return np.clip(h_proj, 0, 1)

def evaluate(h, dx):
    h_clipped = np.clip(h, 0, 1)
    corr = np.correlate(h_clipped, 1 - h_clipped, mode='full')
    return np.max(corr) * dx
