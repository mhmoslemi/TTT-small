import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess to ensure sum constraint
    current_sum = h_initial.sum()
    h_initial = h_initial * (target_sum / current_sum)
    h_initial = np.clip(h_initial, 0, 1)
    
    best_h = h_initial.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    np.random.seed(seed)
    perturb_count = 3  # Number of perturbation attempts per main loop iteration
    max_iter_per_opt = 25  # Reduced for faster convergence
    ftol = 1e-8  # Tolerance for optimization convergence

    while (elapsed := time.time() - start_time) < budget_s - 30:
        time_left = budget_s - elapsed
        frac = time_left / budget_s
        noise_scale = 0.35 * (1 - frac**2)

        for _ in range(perturb_count):
            # Perturbation of the current best_h
            perturbed = best_h + np.random.normal(0, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            current_sum = perturbed.sum()
            
            # Rescale to target_sum while preserving [0,1] bounds
            if current_sum != 0:
                perturbed = (perturbed / current_sum) * target_sum
            perturbed = np.clip(perturbed, 0, 1)

            # Setup for constrained optimization
            bounds = [(0.0, 1.0)] * n_points
            constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            result = minimize(
                fun=lambda x: np.max(np.correlate(x, 1 - x, mode='full')) * dx,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter_per_opt, 'ftol': ftol}
            )

            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_sum = new_h.sum()
                
                # Correct potential numeric drift
                if not np.isclose(new_sum, target_sum, atol=1e-6):
                    new_h = (new_h / new_sum) * target_sum
                    new_h = np.clip(new_h, 0, 1)
                
                corr = np.correlate(new_h, 1 - new_h, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
    
    # Final validation
    assert np.allclose(best_h.sum(), target_sum, atol=1e-6), "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
