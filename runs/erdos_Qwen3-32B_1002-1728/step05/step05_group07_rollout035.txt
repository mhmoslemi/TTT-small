        idxs = np.arange(len(population))
        for _ in range(len(population)):
            contestants = np.random.choice(idxs, tournament_size, replace=False)
            best_idx = contestants[np.argmin(scores[contestants])]
            new_population.append(population[best_idx].copy())
        return new_population

    np.random.seed(seed)
    population_size = 100
    max_generations = 400

    # Initialize population
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n]

    population = [initial_h_half.copy()]
    for _ in range(population_size - 1):
        perturbed = initial_h_half + np.random.normal(0, 0.1, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_h_half = initial_h_half.copy()
    best_c5 = compute_c5(best_h_half)

    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1
        # Evaluate population
        candidates = [(compute_c5(h), h.copy()) for h in population]
        candidates_scores = np.array([c for c, _ in candidates])
        candidates.sort(key=lambda x: x[0])
        if candidates[0][0] < best_c5:
            best_c5 = candidates[0][0]
            best_h_half = candidates[0][1].copy()

        # Elitism: keep the best individual
        elite = [best_h_half.copy()]

        # Tournament selection
        selected_parents = tournament_selection(population, candidates_scores, tournament_size=5)

        # Create new population
        new_population = selected_parents[:population_size // 2]  # Elite and others
        while len(new_population) < population_size:
            # Parent selection via tournament
            parent1 = selected_parents[np.random.randint(len(selected_parents))]
            parent2 = selected_parents[np.random.randint(len(selected_parents))]
            child = (parent1 + parent2) / 2.0
            # Mutation
            if np.random.rand() < 0.3:
                mutation_strength = 0.5 * (1 - (generation / max_generations) ** 1.5)
                child += np.random.normal(0, mutation_strength, half_n)
                child = np.clip(child, 0, 1)
            total = np.sum(child)
            if total != 0:
                child = (child / total) * target_sum_half
            child = np.clip(child, 0, 1)
            new_population.append(child)

        population = new_population

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
