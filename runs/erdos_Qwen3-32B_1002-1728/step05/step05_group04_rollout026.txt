    dx = 2.0 / n_points

    def objective(h_full):

        return np.max(corr) * dx

    h_full = h_full_initial.copy()
    total = np.sum(h_full)
    if total != 0:
        scale = target_sum / total
        h_full = h_full * scale
    h_full = np.clip(h_full, 0, 1)

    best_h = h_full.copy()
    best_c5 = objective(best_h)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:

        current_time = time.time() - start_time

        noise_scale = 0.3 * np.exp(-frac * 5)  # Exponential decay
