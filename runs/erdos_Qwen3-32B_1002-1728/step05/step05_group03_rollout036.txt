        inertia = np.clip(inertia, 0, inertia_start)

        for i in range(population_size):
            r1 = np.random.rand()
            r2 = np.random.rand()
            vel_cognitive = r1 * (particles_pbest_positions[i] - particles_positions[i])
            vel_social = r2 * (gbest_position - particles_positions[i])
            particles_velocities[i] = (
                inertia * particles_velocities[i] +
                c1 * vel_cognitive +
                c2 * vel_social
            )
            particles_velocities[i] = np.clip(particles_velocities[i], -0.1, 0.1)
            particles_positions[i] += particles_velocities[i]
            particles_positions[i] = np.clip(particles_positions[i], 0, 1)
            particles_positions[i] = normalize_h_half(particles_positions[i], target_sum_half)

        for i in range(population_size):
            h_full = create_full_h(particles_positions[i])
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < particles_pbest_fitness[i]:
                particles_pbest_positions[i] = particles_positions[i].copy()
                particles_pbest_fitness[i] = current_c5

        current_min_index = np.argmin(particles_pbest_fitness)
        if particles_pbest_fitness[current_min_index] < best_c5:
            best_c5 = particles_pbest_fitness[current_min_index]
            gbest_position = particles_pbest_positions[current_min_index].copy()

    h_best_half = gbest_position
    h_full = create_full_h(h_best_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
