import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0

    def objective_pso_full(h_full):
        h_full_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_full_clipped, 1 - h_full_clipped, mode='full')
        return np.max(corr_vals) * dx

    # Initialize particles with perturbations around the parent solution
    n_particles = 20
    np.random.seed(seed)
    positions = [h_full_initial.copy() for _ in range(n_particles)]
    for i in range(n_particles):
        noise = np.random.normal(0, 0.1, size=n_points)
        positions[i] += noise
        # Feasibility projection
        positions[i] = np.clip(positions[i], 0, 1)
        total = np.sum(positions[i])
        if total != 0:
            positions[i] = (positions[i] / total) * target_sum_full

    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest_positions = [p.copy() for p in positions]
    pbest_values = [objective_pso_full(p) for p in positions]
    gbest_index = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_index].copy()
    gbest_value = pbest_values[gbest_index]

    # PSO parameters
    inertia = 0.8
    cognitive = 1.2
    social = 1.2
    max_pso_iterations = 20

    for iteration in range(max_pso_iterations):
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = inertia * velocities[i] + \
                            cognitive * r1 * (pbest_positions[i] - positions[i]) + \
                            social * r2 * (gbest_position - positions[i])
            positions[i] += velocities[i]
            # Clamp to [0,1] and rescale
            positions[i] = np.clip(positions[i], 0, 1)
            total = np.sum(positions[i])
            if total != 0:
                positions[i] = (positions[i] / total) * target_sum_full
            current_value = objective_pso_full(positions[i])
            if current_value < pbest_values[i]:
                pbest_positions[i] = positions[i].copy()
                pbest_values[i] = current_value
                if current_value < gbest_value:
                    gbest_position = positions[i].copy()
                    gbest_value = current_value
        if time.time() - start_time > budget_s - 30:
            break

    # Local refinement on top 5 candidates
    top_indices = np.argsort(pbest_values)[:5]
    best_candidates = [pbest_positions[i] for i in top_indices]

    best_h_full = gbest_position.copy()
    best_c5 = gbest_value

    for candidate in best_candidates:
        result = minimize(
            fun=objective_pso_full,
            x0=candidate,
            method='SLSQP',
            bounds=[(0, 1)] * n_points,
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_full}
            ],
            options={'maxiter': 150, 'gtol': 1e-8}
        )
        if result.success:
            optimized_h = np.clip(result.x, 0, 1)
            total = np.sum(optimized_h)
            if total != 0:
                optimized_h = (optimized_h / total) * target_sum_full
            optimized_h = np.clip(optimized_h, 0, 1)
            current_value = objective_pso_full(optimized_h)

                best_h_full = optimized_h
                best_c5 = current_value

            break

    best_h_full = np.clip(best_h_full, 0, 1)
    return (best_h_full.tolist(), best_c5, n_points)
