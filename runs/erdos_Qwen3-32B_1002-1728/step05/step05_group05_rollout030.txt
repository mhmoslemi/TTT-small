def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def clip_and_scale(h_half):
        h_clipped = np.clip(h_half, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum == 0:
            return h_clipped
        target_sum_half = target_sum / 2.0
        scale_factor = target_sum_half / current_sum
        scaled = h_clipped * scale_factor
        return np.clip(scaled, 0, 1)

    def evaluate_smoothed_objective(h_half, tau):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        overlaps = corr * dx
        max_value = np.max(overlaps)
        exp_vals = np.exp((overlaps - max_value) / tau)
        sum_exp = np.sum(exp_vals)
        return tau * (np.log(sum_exp) + max_value)

    def compute_gradient(h_half, tau, epsilon=1e-5):
        grad = np.zeros_like(h_half)
        for i in range(len(h_half)):
            h_plus = h_half.copy()
            h_plus[i] += epsilon
            val_plus = evaluate_smoothed_objective(clip_and_scale(h_plus), tau)
            
            h_minus = h_half.copy()
            h_minus[i] -= epsilon
            val_minus = evaluate_smoothed_objective(clip_and_scale(h_minus), tau)
            
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        return grad

    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = clip_and_scale(h_half_initial)
    current_h = h_half_initial.copy()
    best_c5 = evaluate_c5(create_full_h(current_h))
    best_h = current_h.copy()

    step_size = 0.1
    min_step = 1e-4
    tau = 0.1
    tau_decay = 0.995
    threshold = 1e-6

    while time.time() - start_time < budget_s - 10:
        try:
            grad = compute_gradient(current_h, tau)
        except:
            grad = np.zeros_like(current_h)
        
        proposed_h = current_h - step_size * grad
        proposed_h = clip_and_scale(proposed_h)
        current_smooth = evaluate_smoothed_objective(current_h, tau)
        proposed_smooth = evaluate_smoothed_objective(proposed_h, tau)

        if proposed_smooth < current_smooth:
            current_h = proposed_h.copy()
            current_c5 = evaluate_c5(create_full_h(current_h))
            if current_c5 < best_c5 - threshold:
                best_c5 = current_c5
                best_h = current_h.copy()
                step_size = min(step_size * 1.1, 0.2)
            else:
                step_size = max(step_size * 0.5, min_step)
        else:
            step_size = max(step_size * 0.5, min_step)
        
        if np.random.rand() < 0.01:
            tau = max(tau * tau_decay, 0.01)

    best_h_full = create_full_h(best_h)
    assert np.isclose(np.sum(best_h_full), target_sum, atol=1e-6)
    return (best_h_full.tolist(), best_c5, n_points)
