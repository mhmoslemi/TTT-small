import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    K = n_points // 2  # Ensures sum(h)*dx=1
    dx = 2.0 / n_points

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Convert initial to binary with exactly K ones
    indices = np.argsort(-h_full_initial).astype(int)
    h_binary = np.zeros(n_points, dtype=bool)
    h_binary[indices[:K]] = True
    best_h = h_binary.astype(np.float64)
    best_c5 = evaluate(h_binary)

    np.random.seed(seed)
    rand_gen = np.random.default_rng(seed)

    # Initialize a small population for diversity
    population = [h_binary.copy() for _ in range(10)]
    for idx in range(1, 10):
        candidate = population[idx].copy()
        # Perturb by swapping more positions
        zeros = np.where(~candidate)[0]
        ones = np.where(candidate)[0]
        swap_count = np.clip(rand_gen.integers(1, 4), 1, len(zeros) * len(ones))
        for _ in range(swap_count):
            if len(zeros) and len(ones):
                z = rand_gen.choice(zeros)
                o = rand_gen.choice(ones)
                candidate[z], candidate[o] = True, False
                zeros = np.setdiff1d(zeros, [z, o])
                ones = np.setdiff1d(ones, [z, o])
        population[idx] = candidate

    while time.time() - start_time < budget_s - 15:
        new_population = []
        for _ in range(50):  # balance population
            # Select two parents by random tournament
            p1 = population[rand_gen.integers(0, len(population))]
            p2 = population[rand_gen.integers(0, len(population))]
            parent = p1 if evaluate(p1) < evaluate(p2) else p2
            # Mutate
            new_ind = parent.copy()
            zeros = np.where(~new_ind)[0]
            ones = np.where(new_ind)[0]
            swap_count = rand_gen.integers(1, 3)
            for _ in range(swap_count):
                if len(zeros) and len(ones):
                    z = rand_gen.choice(zeros)
                    o = rand_gen.choice(ones)
                    new_ind[z], new_ind[o] = True, False
                    zeros = np.setdiff1d(zeros, [z,o])
                    ones = np.setdiff1d(ones, [z,o])
            new_population.append(new_ind)

        # Evaluate and replace population
        valid_population = []

            c5_val = evaluate(candidate)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h[:] = candidate.astype(np.float64)
            valid_population.append(candidate)
        population = valid_population[:20]  # trim population

    return (best_h.tolist(), best_c5, n_points)
