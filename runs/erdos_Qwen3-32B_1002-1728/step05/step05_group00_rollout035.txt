            current_sum = np.sum(x)
            if current_sum == 0:
                return objective(np.full(n_points, 0.5))
            x_clipped = np.clip(x, 0, 1)
            scale = target_sum / current_sum
            x_clipped = np.clip(x_clipped * scale, 0, 1)
            return objective(x_clipped)

        bounds = [(0.0, 1.0)] * n_points
        result = minimize(
            constrained_objective,
            x0=global_best.ravel(),
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x.ravel()) - target_sum},
            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            final_h = np.clip(result.x.reshape(n_points) * (target_sum / np.sum(result.x)), 0, 1)
            final_score = objective(final_h)
            if final_score < global_best_score:
                global_best = final_h
                global_best_score = final_score

    h_full = np.clip(global_best, 0, 1)
    assert abs(np.sum(h_full) - target_sum) < 1e-6, (np.sum(h_full), target_sum)
    return (h_full.tolist(), global_best_score, n_points)
