import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = h_half[i]
            full_h[n_points - 1 - i] = h_half[i]
        return full_h
    
    def objective(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    def equality_constraint(h_half):
        return 2 * np.sum(h_half) - h_half[-1] - target_sum
    
    bounds = [(0, 1) for _ in range(half_n)]
    constraints = {'type': 'eq', 'fun': equality_constraint}

        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
