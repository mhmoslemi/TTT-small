import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 20
    population = [h_initial.copy() for _ in range(4)]
    for h in population:
        h_sum = h.sum()
        if h_sum != 0:
            h[:] = np.clip(h * target_sum / h_sum, 0, 1)

    while len(population) < pop_size:
        idx = np.random.randint(0, 4)
        h = population[idx].copy()
        noise_scale = 0.2 * np.random.rand() + 0.05
        h += np.random.randn(n_points) * noise_scale
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h[:] = np.clip(h * target_sum / h_sum, 0, 1)
        population.append(h)

    best_h = evaluate_c5(population[0])
    best_idx = 0
    best_population = population.copy()

    while time.time() - start_time < budget_s - 30:
        candidates = []
        for h in population:
            c5 = evaluate_c5(h)
            candidates.append((c5, h))
        candidates.sort()
        new_pop = [c[1] for c in candidates[: int(pop_size * 0.5)]]
        while len(new_pop) < pop_size:
            parent = candidates[np.random.randint(0, len(candidates))][1].copy()
            noise_scale = 0.3 * (1 - (time.time() - start_time)/budget_s)
            mut = np.random.randn(n_points) * noise_scale
            child = np.clip(parent + mut, 0, 1)
            h_sum = child.sum()
            if h_sum != 0:
                child = np.clip(child * target_sum / h_sum, 0, 1)
            new_pop.append(child)
        population = new_pop
        if candidates[0][0] < best_h:
            best_h = candidates[0][0]
            best_population = [c[1] for c in candidates]
        if time.time() - start_time > budget_s - 70:
            break

    best_idx = 0
    final_h = best_population[best_idx]
    final_c5 = evaluate_c5(final_h)
    
    # Local refinement
    def objective(x):
        return evaluate_c5(x)
    
    constraints = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}
    bounds = [(0.0, 1.0)] * n_points
    result = minimize(objective, final_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 100, 'ftol': 1e-12})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        h_sum = refined_h.sum()
        if h_sum != 0:
            refined_h = np.clip(refined_h * target_sum / h_sum, 0, 1)
            final_c5 = evaluate_c5(refined_h)
    
    h_refined = refined_h if 'refined_h' in locals() else final_h
    assert abs(h_refined.sum() - target_sum) < 1e-6
    return (h_refined.tolist(), final_c5, n_points)
