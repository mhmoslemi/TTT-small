import numpy as np
import time

def create_full_h(h_half, n_points, half_n):
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = h_half[i]
    return h_full

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    # Ensure h_full_initial is symmetric
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2.0
        h_full_initial[i] = h_full_initial[j] = avg

    # Initialize PSO parameters
    num_particles = 50
    h_particles = np.zeros((num_particles, half_n))
    velocities = np.zeros((num_particles, half_n))
    pbest_positions = np.zeros((num_particles, half_n))
    pbest_scores = np.full(num_particles, np.inf)
    gbest = None
    gbest_score = np.inf

    # Initialize particles near initial solution
    np.random.seed(seed)
    for i in range(num_particles):
        h_half = h_full_initial[:half_n].copy()
        perturbation = np.random.uniform(-0.05, 0.05, half_n)
        h_half = np.clip(h_half + perturbation, 0, 1)
        current_sum = np.sum(h_half)
        if not np.isclose(current_sum, 0):
            h_half *= target_sum_half / current_sum
            h_half = np.clip(h_half, 0, 1)
        h_particles[i] = h_half
        # Evaluate fitness
        full = create_full_h(h_half, n_points, half_n)
        corr = np.correlate(full, 1 - full, mode='full')
        c5 = np.max(corr) * dx
        if c5 < pbest_scores[i]:
            pbest_scores[i] = c5
            pbest_positions[i] = h_half.copy()
            if c5 < gbest_score:
                gbest_score = c5
                gbest = h_half.copy()

    # PSO parameters
    c1, c2 = 1.5, 1.5
    w_max, w_min = 0.9, 0.4

    while time.time() - start_time < budget_s - 10:
        elapsed = time.time() - start_time
        w = w_max - (w_max - w_min) * (elapsed / budget_s)
        w = max(w, w_min)

        for i in range(num_particles):
            r1, r2 = np.random.rand(half_n), np.random.rand(half_n)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (pbest_positions[i] - h_particles[i]) +
                            c2 * r2 * (gbest - h_particles[i]))
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)

            # Update position and normalize
            new_h = h_particles[i] + velocities[i]
            new_h = np.clip(new_h, 0, 1)
            current_sum = np.sum(new_h)
            if not np.isclose(current_sum, 0):
                new_h *= target_sum_half / current_sum
                new_h = np.clip(new_h, 0, 1)
            h_particles[i] = new_h

            # Evaluate
            full = create_full_h(new_h, n_points, half_n)
            corr = np.correlate(full, 1 - full, mode='full')
            new_c5 = np.max(corr) * dx

            # Update pbest and gbest
            if new_c5 < pbest_scores[i]:
                pbest_scores[i] = new_c5
                pbest_positions[i] = new_h.copy()
                if new_c5 < gbest_score:
                    gbest_score = new_c5
                    gbest = new_h.copy()

    # Construct final h_full and ensure constraints
    best_h_full = create_full_h(gbest, n_points, half_n)
    best_h_full = np.clip(best_h_full, 0, 1)
    current_sum = np.sum(best_h_full)
    best_h_full *= (n_points / 2.0) / current_sum
    best_h_full = np.clip(best_h_full, 0, 1)

    return (best_h_full.tolist(), gbest_score, n_points)
