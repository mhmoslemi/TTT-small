import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initialize population with varied asymmetric candidates
    pop_size = 12  # Balanced population size
    np.random.seed(seed)
    population = []
    for _ in range(pop_size):
        h = h_initial.copy()
        h += 0.3 * (np.random.rand(n_points) * 2 - 1)
        h = np.clip(h, 0, 1)
        curr_sum = np.sum(h)
        if curr_sum != 0:
            h = h * (target_sum / curr_sum)
        h = np.clip(h, 0, 1)
        population.append(h)
        if _ < 2:  # Add more symmetric baseline
            population.append(np.sort(h))
    
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Evolutionary phase
    while time.time() - start_time < budget_s - 60:
        # Evaluate all candidates
        scores = [evaluate_c5(h) for h in population]
        
        # Select top half
        elite_indices = np.argsort(scores)[:len(population)//2]
        elites = [population[i] for i in elite_indices]
        
        # Generate new population with adaptive mutation
        new_population = []
        remaining_time = max(budget_s - 60 - (time.time() - start_time), 1e-9)
        noise_scale = 0.25 * (1 - remaining_time/budget_s)  # Scales from 0 to 0.25
        
        for _ in range(len(population)):
            parent = elites[np.random.randint(len(elites))]
            h_mut = parent.copy()
            # Add asymmetric noise with position-dependent weights
            perturbation = noise_scale * (np.random.rand(n_points) * 2 - 1)
            perturbation = perturbation * (1 + 0.5 * np.sin(2 * np.pi * np.arange(n_points)/n_points))  # Non-uniform mutation
            h_mut += perturbation
            h_mut = np.clip(h_mut, 0, 1)
            curr_sum = np.sum(h_mut)
            h_mut = h_mut * (target_sum / max(curr_sum, 1e-9))
            h_mut = np.clip(h_mut, 0, 1)
            new_population.append(h_mut)
        
        population = new_population
        
        # Update best if improved
        for h in population:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_h = h.copy()

    # Local refinement of top candidates
    candidates = [h for h in population]
    candidates_scores = [evaluate_c5(h) for h in candidates]
    candidates = [h for _, h in sorted(zip(candidates_scores, candidates))]
    
    # Use derivative-free optimization for final polish
    def objective(x):
        h_opt = x.reshape(-1)
        return np.max(np.correlate(h_opt, 1-h_opt, mode='full')) * dx
    
    for i in range(min(4, len(candidates))):  # Optimize top 4 candidates
        h0 = candidates[i].copy()
        # Use SLSQP with sum constraint
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons,
