        width_seg = 2.0 / n_segments
        for i in range(n_points):
            x = i * dx
            seg = int(x / (2.0 / n_segments))
            t_val = (x - seg * width_seg) / width_seg
            if 0 <= seg < n_segments:
                h[i] = (1 - t_val) * p[seg] + t_val * p[seg + 1]
            else:
                h[i] = 0.0
        return h

    # Evaluate objective
    def evaluate_obj(p):
        h = eval_control_points(normalize(p))
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    # Population initialization
    np.random.seed(seed)
    population = []
    for _ in range(pop_size - 2):
        p = np.random.uniform(0.3, 0.8, size=n_segments + 1)
        population.append(normalize(p))
    p0 = create_p_from_initial()
    p0 = normalize(p0)
    population.extend([p0] * 3)
    best_p, best_score = p0.copy(), evaluate_obj(p0)

    # Genetic main loop
    while time.time() - start_time < budget_s - 30:
        # Score population
        scores = [evaluate_obj(p) for p in population]
        idx = np.argmin(scores)
        if scores[idx] < best_score:
            best_p, best_score = population[idx], scores[idx]

        # Selection
        fitness = np.argsort(scores)
        top_pop = [population[i].copy() for i in fitness[:int(0.3*pop_size)]]
        pool = [p for i in range(t) for p in top_pop]

        # Genetic ops
        new_population = []
        while len(new_population) < pop_size:
            # Crossover
            p1, p2 = np.random.choice(pool) if pool else np.random.choice(population)
            child = p1 * np.random.beta(0.5, 0.5) + p2 * np.random.beta(0.5, 0.5)
            # Mutation
            if np.random.random() < 0.3:
                scale = 0.5 - (time.time() - start_time) / budget_s * 0.4
                child += np.random.normal(0, scale, size=child.shape)
                child = np.clip(child, 0, 1)
            new_p = normalize(child)
            new_population.append(new_p)

        # Local search
        for tries in range(min(4, int(budget_s - (time.time()-start_time)) // 6)):
            idx = np.random.choice(fitness[:20])
            p_init = population[idx].copy()
            res = minimize(
                evaluate_obj, p_init, 
                method='SLSQP', 
                bounds=[(0,1)]*(n_segments+1),
                options={'ftol':1e-10, 'maxiter':100}
            )
            if res.success and res.fun < best_score:
                p_opt = normalize(res.x)
                score_opt = evaluate_obj(p_opt)
                if score_opt < best_score:
                    best_p, best_score = p_opt, score_opt

        population = new_population

    # Final check and result
    final_h = eval_control_points(normalize(best_p))
    final_int = integral(normalize(best_p))
    final_score = evaluate_obj(normalize(best_p))
    assert abs(np.sum(final_h) * dx - 1.0) < 1e-6
    return (final_h.tolist(), final_score, n_points)
