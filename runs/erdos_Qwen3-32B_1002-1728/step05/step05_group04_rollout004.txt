    n_elites = 2

    def compute_c5(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def adjust_sum_and_clip(h, target):
        h = np.array(h, dtype=float)
        current_sum = np.sum(h)
        if current_sum == 0:
            h[0] = target_sum
            return np.clip(h, 0, 1)
        scaling = target / current_sum
        h *= scaling
        h_clipped = np.clip(h, 0, 1)
        return h_clipped

    def crossover(p1, p2):
        alpha = np.random.rand()
        child = alpha * p1 + (1 - alpha) * p2
        return child

    def mutate(h):
        h = h.copy()
        idx = np.random.choice(n_points, int(0.1 * n_points), replace=False)
        h[idx] += np.random.normal(0, 0.05, idx.shape)
        return h

    np.random.seed(seed)
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        h = h_initial.copy()
        h += np.random.normal(0, 0.1, h.shape)
        h = adjust_sum_and_clip(h, target_sum)
        population.append(h)

    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s - 20:
        fitnesses = [compute_c5(h) for h in population]
        current_indices = np.argsort(fitnesses)
        current_best = population[current_indices[0]]
        current_best_c5 = fitnesses[current_indices[0]]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best.copy()

        parents = []
        for _ in range(population_size):
            candidates = np.random.choice(population_size, 4)
            winner_idx = candidates[np.argmin([fitnesses[i] for i in candidates])]
            parents.append(population[winner_idx])

        new_population = []
        for i in current_indices[:n_elites]:
            new_population.append(population[i].copy())

        while len(new_population) < population_size:
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            child = crossover(p1, p2)
            if np.random.rand() < 0.1:
                child = mutate(child)
            child = adjust_sum_and_clip(child, target_sum)
            new_population.append(child)

        population = new_population

    return (best_h.tolist(), best_c5, n_points)
