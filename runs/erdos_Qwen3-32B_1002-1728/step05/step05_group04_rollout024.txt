import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    np.random.seed(seed)

    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return float(np.max(corr) * dx)

    def mutate(h, max_steps=2):
        h = h.copy()
        for _ in range(max_steps):
            i, j = np.random.choice(n_points, size=2, replace=False)
            while i == j:
                j = np.random.randint(n_points)
            delta = np.random.uniform(0, min(1 - h[i], h[j]))
            h[i] += delta
            h[j] -= delta
        return h

    population_size = 50
    population = [h_initial.copy()]
    for _ in range(population_size - 1):
        pop = mutate(h_initial.copy())
        population.append(pop)

    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)

    while time.time() - start_time < budget_s - 10:
        def tournament(pool_size=3):
            candidates = [random.choice(population) for _ in range(pool_size)]
            return min(candidates, key=evaluate)

        parent1, parent2 = tournament(), tournament()
        child = (parent1 + parent2) / 2.0
        child = mutate(child)
        child_c5 = evaluate(child)

        fitness_list = [evaluate(ind) for ind in population]
        worst_idx = np.argmax(fitness_list)
        if child_c5 < fitness_list[worst_idx]:

        if child_c5 < best_c5:
