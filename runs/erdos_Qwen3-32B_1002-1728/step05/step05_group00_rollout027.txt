def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def clip_and_project(position, target_sum):
        position_clipped = np.clip(position, 0.0, 1.0)
        current_sum = np.sum(position_clipped)
        if current_sum < 1e-10:
            fallback = np.clip(h_initial.copy(), 0.0, 1.0)
            current_sum = np.sum(fallback)
        else:
            fallback = position_clipped
        scale = target_sum / current_sum
        position_clipped = fallback * scale
        return np.clip(position_clipped, 0.0, 1.0)

    np.random.seed(seed)
    h_initial = clip_and_project(h_initial, target_sum)
    population_size = 20
    particles = np.array([clip_and_project(h_initial + np.random.normal(0, 0.1, n_points), target_sum) for _ in range(population_size)])
    velocities = np.zeros_like(particles)

    def compute_c5(h_vec):
        corr_vals = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr_vals) * dx

    personal_best_scores = np.array([compute_c5(p) for p in particles])
    personal_best_positions = particles.copy()
    global_best_index = np.argmin(personal_best_scores)
    global_best_position = particles[global_best_index].copy()
    global_best_score = personal_best_scores[global_best_index]
    best_c5 = global_best_score
    best_solution = particles[global_best_index].copy()

    max_pso_iterations = 20
    w, c1, c2 = 0.8, 1.2, 1.2

    for _ in range(max_pso_iterations):
        if time.time() - start_time >= budget_s * 0.8:
            break
        for i in range(population_size):
            r1, r2 = np.random.rand(), np.random.rand()
            cognitive = c1 * r1 * (personal_best_positions[i] - particles[i])
            social = c2 * r2 * (global_best_position - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            particles[i] = particles[i] + velocities[i]
            particles[i] = clip_and_project(particles[i], target_sum)
            new_score = compute_c5(particles[i])
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_best_positions[i] = particles[i].copy()
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best_position = particles[i].copy()
            if new_score < best_c5:
                best_c5 = new_score
                best_solution = particles[i].copy()

    top_candidates_indices = np.argsort(personal_best_scores)[:5]
    remaining_time = budget_s - (time.time() - start_time)

    for idx in top_candidates_indices:
        if time.time() - start_time >= budget_s:
            break
        candidate = particles[idx].copy()
        bounds = Bounds(0.0, 1.0)
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(
            fun=compute_c5,
            x0=candidate,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            x_opt = np.clip(result.x, 0.0, 1.0)
            if abs(np.sum(x_opt) - target_sum) > 1e-6:
                x_opt = clip_and_project(x_opt, target_sum)
            new_score = compute_c5(x_opt)
            if new_score < best_c5:
                best_c5 = new_score
                best_solution = x_opt

    h_full = best_solution
    assert abs(np.sum(h_full) - target_sum) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
