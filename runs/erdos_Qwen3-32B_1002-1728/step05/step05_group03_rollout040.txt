import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        full_h = np.zeros(n_points)
        for i in range(half_n):
            full_h[i] = full_h[n_points - 1 - i] = h_half[i]
        return full_h

    def normalize_h_half(h_half, target_sum):
        h_copy = h_half.copy()
        current_sum = h_copy.sum()
        if current_sum == 0:
            return h_copy
        h_scaled = h_copy * (target_sum / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        new_sum = h_clipped.sum()
        if new_sum == 0:
            return h_clipped
        h_scaled2 = h_clipped * (target_sum / new_sum)
        h_clipped2 = np.clip(h_scaled2, 0, 1)
        return h_clipped2

    np.random.seed(seed)
    initial_h_half = h_full_initial[:half_n].copy()
    best_h_half = normalize_h_half(initial_h_half, target_sum_half)
    best_c5 = float('inf')
    step_count = 0

    while time.time() - start_time < budget_s - 30:
        step_count += 1
        # Perturbation parameters
        sigma = min(0.2, 0.05 + 0.05 * (step_count // 50))
        perturbation_fraction = min(0.5, 0.2 + 0.1 * (step_count // 50))
        num_perturb = max(1, int(half_n * perturbation_fraction))
        indices = np.random.choice(half_n, size=num_perturb, replace=False)
        h_perturbed = best_h_half.copy()
        perturbation = np.random.normal(0, sigma, size=num_perturb)
        h_perturbed[indices] += perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h_half(h_perturbed, target_sum_half)
        h_full = create_full_h(h_perturbed)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h_half = h_perturbed.copy()

    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    final_c5 = np.max(corr)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all(h_full <= 1.0 + 1e-6) and np.all(h_full >= -1e-6)
    return (h_full.tolist(), final_c5, n_points)
