        remaining = budget_s - elapsed
        if remaining < 1e-6:
            break
        fraction = 1.0 - remaining / (budget_s)
        w = w_start + (w_end - w_start) * fraction

        r1 = np.random.rand(n_pop, half_n)
        r2 = np.random.rand(n_pop, half_n)

        velocities = w * velocities + c1 * r1 * (pbest_positions - particles) + c2 * r2 * (gbest_position[None, :] - particles)
        velocities = np.clip(velocities, -velocity_clamp, velocity_clamp)

        particles += velocities
        particles = np.clip(particles, 0, 1)

        sums = np.sum(particles, axis=1)
        scales = np.where(sums != 0, target_sum / sums, 1.0)
        particles *= scales[:, None]
        particles = np.clip(particles, 0, 1)

        for i in range(n_pop):
            current_fitness = evaluate_c5(particles[i])
            if current_fitness < pbest_fitness[i]:
                pbest_fitness[i] = current_fitness
                pbest_positions[i] = particles[i].copy()

        current_gbest_idx = np.argmin(pbest_fitness)
        if pbest_fitness[current_gbest_idx] < best_c5:
            best_c5 = pbest_fitness[current_gbest_idx]
            gbest_position = pbest_positions[current_gbest_idx].copy()

    # Final solution
    best_h_half = gbest_position
    best_h_full = np.zeros(n_points)
    for i in range(half_n):
        best_h_full[i] = best_h_full[n_points - 1 - i] = best_h_half[i]
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    assert np.abs(np.sum(best_h_full) - n_points / 2.0) < 1e-6

    return (best_h_full.tolist(), best_c5, n_points)
