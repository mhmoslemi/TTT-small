import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def repair(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if abs(current_sum) < 1e-10:
            return h_clipped
        h_scaled = h_clipped * (target_sum / current_sum)
        return np.clip(h_scaled, 0, 1)

    def objective(h_flattened):
        h = h_flattened.reshape(n_points)
        h_repaired = repair(h)
        corr_vals = np.correlate(h_repaired, 1 - h_repaired, mode='full')
        return np.max(corr_vals) * dx

    pop_size = 20
    generations = 30
    elite_ratio = 0.05
    mutation_rate_start = 0.15
    mutation_decay = 0.95
    crossover_prob = 0.8
    tournament_ratio = 0.3
    top_candidates = 5

    # Initialize population
    np.random.seed(seed)
    population = []
    h_repaired_initial = repair(h_initial)
    population.append(h_repaired_initial.flatten())
    for _ in range(pop_size - 1):
        h_perturbed = h_initial + np.random.normal(0, 0.1, size=n_points)
        population.append(repair(h_perturbed).flatten())
    population = np.array(population)

    best_c5 = float('inf')
    best_h = h_repaired_initial.flatten()
    gen_start = 0

    while time.time() - start_time < budget_s - 30 and gen_start < generations:
        gen_start += 1
        # Evaluate population
        fitness = np.array([objective(ind) for ind in population])
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        # Selection (tournament)
        tournament_size = int(pop_size * tournament_ratio)
        selected_indices = np.argsort(fitness)[:tournament_size]
        selected = population[selected_indices]
        # Elitism
        num_elites = int(pop_size * elite_ratio)
        elites = selected[:num_elites]
        new_population = elites.copy()
        remaining = pop_size - num_elites
        # Crossover and mutation
        for _ in range(remaining):
            parent1, parent2 = np.random.choice(selected.shape[0], size=2, replace=False)
            offspring = np.zeros(n_points)
            for i in range(n_points):
                if np.random.rand() < crossover_prob:
                    offspring[i] = parent1 if np.random.rand() < 0.5 else parent2
                else:
                    if np.random.rand() < mutation_rate_start ** gen_start * mutation_decay ** gen_start:
                        offspring[i] = np.clip(parent2[i] + np.random.normal(0, 0.1), 0, 1)
                    else:
                        offspring[i] = parent2[i]
            new_population = np.vstack((new_population, repair(offspring).flatten()))
        population = new_population

    # Local refinement
    np.random.seed(seed)
    top_indices = np.argsort([objective(ind) for ind in population])[:top_candidates]
    top_solutions = population[top_indices]
    for h_init in top_solutions:
        bounds = [(0, 1)] * n_points
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
        result = minimize(
            fun=objective, x0=h_init, method='COBYLA', bounds=bounds,
            constraints=cons, options={'maxiter': 200, 'tol': 1e-8, 'disp': False}
        )
        if result.success:
            h_optimized = repair(result.x).flatten()
            c5 = objective(h_optimized)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_optimized

    final_h = best_h.reshape(n_points)
    final_h = repair(final_h)
    assert abs(np.sum(final_h) - target_sum) < 1e-6
    return (final_h.tolist(), best_c5, n_points)
