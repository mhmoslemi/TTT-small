            new_population.append(child)

    best_h = [population[i] for i in np.argsort([objective(ind) for ind in population])][0]
    # Refine with SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

        x0=best_h,
        method='SLSQP',
