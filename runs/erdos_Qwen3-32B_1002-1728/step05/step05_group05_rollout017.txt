import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    # Symmetrize and normalize initial h_full
    h_full = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full[i] + h_full[j]) / 2
        h_full[i] = h_full[j] = avg
    h_sum = np.sum(h_full)
    if not np.allclose(h_sum, n_points / 2.0, atol=1e-6):
        scale = (n_points / 2.0) / h_sum
        h_full *= scale
    h_full = np.clip(h_full, 0, 1)
    h_half_initial = h_full[:half_n].copy()

    # Initialize population
    np.random.seed(seed)
    pop_size = 50
    population = []
    for _ in range(pop_size):
        if _ == 0:
            h_half = h_half_initial.copy()
        else:
            h_half = h_half_initial + np.random.normal(0, 0.05, half_n)
        h_half = np.clip(h_half, 0, 1)
        current_sum = np.sum(h_half)
        if not np.isclose(current_sum, 0):
            scale = target_sum_half / current_sum
            h_half *= scale
        h_half = np.clip(h_half, 0, 1)
        population.append(h_half)

    def evaluate(h_half_candidate):
        h_full_candidate = np.zeros(n_points)
        for i in range(half_n):
            h_full_candidate[i] = h_full_candidate[n_points - 1 - i] = h_half_candidate[i]
        corr = np.correlate(h_full_candidate, 1 - h_full_candidate, mode='full')
        return np.max(corr) * dx

    fitness = [evaluate(p) for p in population]
    best_idx = np.argmin(fitness)
    best_c5 = min(fitness)
    best_h_half = population[best_idx].copy()

    # GA parameters
    max_gen = 200
    mutation_rate = 0.25
    crossover_rate = 0.5
    num_offspring = 50

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 20:
            break

        offspring = []
        for _ in range(num_offspring):
            # Tournament selection
            def tournament():
                idxs = np.random.choice(pop_size, 2, replace=False)
                f1, f2 = fitness[idxs[0]], fitness[idxs[1]]
                return population[idxs[0] if f1 < f2 else idxs[1]]
            parent1 = tournament()
            parent2 = tournament()

            # Crossover
            if np.random.rand() < crossover_rate:
                beta = np.random.uniform(0.3, 0.7)
                child = beta * parent1 + (1 - beta) * parent2
            else:
                child = parent1.copy()

            # Mutation

                noise = np.random.normal(0, 0.01, half_n)
                child += noise
                child = np.clip(child, 0, 1)

                if not np.isclose(current_sum, 0):
                    scale = target_sum_half / current_sum
                    child *= scale
                child = np.clip(child, 0, 1)

            offspring.append(child)

        # Evaluate offspring
        off_fitness = [evaluate(c) for c in offspring]
        combined = population + offspring
        combined_f = fitness + off_fitness

        # Select top pop_size individuals
        idxs = np.argsort(combined_f)
        population = [combined[i] for i in idxs[:pop_size]]

    best_h_full = np.zeros(n_points)
