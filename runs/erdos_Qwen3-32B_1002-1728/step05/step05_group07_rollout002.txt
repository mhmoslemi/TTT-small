            else:
                mirror_j = n - 1 - j
                hj = h_half[mirror_j]
            expr += hi * hj
        return expr

    # Iterative constraint adding
    for _ in range(5):  # Up to 5 iterations
        if time.time() - start_time > budget_s - 10:
            break
        # Evaluate current solution
        h_half_val = h_half.value if h_half.value is not None else h_half_initial
        h_full_val = create_full_h(h_half_val)
        # Find worst k
        worst_overlap = -np.inf
        worst_k = 0
        for k in range(n_points):
            # Compute correlation at shift k
            j_vals = (np.arange(n_points) + k) % n_points
            corr = np.sum(h_full_val * h_full_val[j_vals])
            overlap = (n_points_half - corr) * dx
            if overlap > worst_overlap:
                worst_overlap = overlap
                worst_k = k
        if worst_overlap <= 1e-6:  # Already good
            break
        # Add constraint for worst_k
        sum_expr = build_sum_ij_expr(worst_k)
        rhs = n_points_half * (1 - t / dx)
        constraints.append(sum_expr >= rhs)
        # Formulate problem
        objective = cp.Minimize(t)
        problem = cp.Problem(objective, constraints)
        problem.solve(solver=cp.ECOS, max_iters=100)
        if problem.status != "optimal":
            break

    # Final solution
    if h_half.value is not None:
        h_half_opt = np.clip(h_half.value, 0, 1)
    else:
        h_half_opt = h_half_initial.copy()
    h_full_opt = create_full_h(h_half_opt)
    # Ensure constraint
    h_sum = np.sum(h_full_opt)
    if abs(h_sum - n_points_half) > 1e-6:
        h_full_opt = h_full_opt * (n_points_half / h_sum)
    h_full_opt = np.clip(h_full_opt, 0, 1)
    best_c5 = compute_c5(h_full_opt)
    return (h_full_opt.tolist(), best_c5, n_points)
