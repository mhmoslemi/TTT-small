import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum})

    h_current = h_initial.copy()
    current_sum = np.sum(h_current)
    if current_sum != 0 and abs(current_sum - target_sum) > 1e-10:
        h_current = (h_current / current_sum) * target_sum
    h_current = np.clip(h_current, 0.0, 1.0)
    best_h = h_current.copy()
    best_c5 = objective(best_h)

    np.random.seed(seed)
    perturb_count = 3

    while time.time() - start_time < budget_s - 30:
        time_left = budget_s - (time.time() - start_time)
        frac = time_left / budget_s
        noise_scale = 0.35 * (1 - frac**2)

        for _ in range(perturb_count):
            perturbed = best_h + np.random.normal(0, noise_scale, n_points)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum != 0:
                perturbed = (perturbed / perturbed_sum) * target_sum
            perturbed = np.clip(perturbed, 0.0, 1.0)

            result = minimize(
                fun=objective,
                x0=perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-12, 'disp': False}

            if result.success:

                        best_h = new_h
                        best_c5 = new_c5

    assert abs(np.sum(best_h) - target_sum) < 1e-6
