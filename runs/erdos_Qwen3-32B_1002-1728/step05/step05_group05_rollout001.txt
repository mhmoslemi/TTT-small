        fitness.append(fit)
    fitness = np.array(fitness)

    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h_half = population[best_idx].copy()

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 5:
        parent_indices = []
        for _ in range(2 * population_size):
            a, b = np.random.choice(population_size, 2, replace=False)
            parent_indices.append(a if fitness[a] < fitness[b] else b)

        offspring = []
        for i in range(0, len(parent_indices), 2):
            if i + 1 >= len(parent_indices):
                break
            a = parent_indices[i]
            b = parent_indices[i+1]
            p1, p2 = population[a], population[b]
            beta = np.random.rand()
            child = beta * p1 + (1 - beta) * p2
            mutation = np.random.normal(0, 0.05, half_n)
            child += mutation
            child = np.clip(child, 0, 1)
            current_sum = np.sum(child)
            if current_sum > 1e-8:
                scale = target_sum_half / current_sum
                child *= scale
                child = np.clip(child, 0, 1)
            offspring.append(child)

        offspring_fitness = []
        for child in offspring:
            offspring_fitness.append(evaluate_c5(child, n_points, dx))
        offspring_fitness = np.array(offspring_fitness)

        combined = np.concatenate((population, offspring), axis=0)
        combined_fitness = np.concatenate((fitness, offspring_fitness))
        sorted_indices = np.argsort(combined_fitness)
        new_population = combined[sorted_indices[:population_size]]
        fitness = combined_fitness[sorted_indices[:population_size]]

        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h_half = new_population[current_best_idx]

        if time.time() - start_time > budget_s:
            break

    best_h_full = create_full_h(best_h_half, n_points)
    best_h_full = np.clip(best_h_full, 0, 1)
    sum_h = np.sum(best_h_full)
    if not np.isclose(sum_h, n_points / 2.0):
        scale = (n_points / 2.0) / sum_h
        best_h_full *= scale
        best_h_full = np.clip(best_h_full, 0, 1)

    return (best_h_full.tolist(), best_c5, n_points)
