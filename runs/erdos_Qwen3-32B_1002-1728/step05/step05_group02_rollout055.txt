import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def scale_and_clip(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if np.isclose(current_sum, 0):
            return np.zeros_like(h)
        scaling = target_sum / current_sum
        h_scaled = h_clipped * scaling
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    h_best = scale_and_clip(h_full_initial)
    best_c5 = objective(h_best)
    bounds = [(0, 1) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    while time.time() - start_time < budget_s - 30:
        attempt = int((time.time() - start_time) * 1000)
        local_seed = seed + attempt
        rnd = np.random.RandomState(local_seed)
        noise_scale = 0.25 * (1.0 - (time.time() - start_time) / budget_s)
        
        perturbation = rnd.normal(0, noise_scale, n_points)
        h_perturbed = h_best + perturbation
        h_perturbed = scale_and_clip(h_perturbed)

        result = minimize(
            fun=objective,
            x0=h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 60, 'ftol': 1e-12, 'disp': False}
        )

        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                h_best = new_h
                best_c5 = new_c5

    h_best = scale_and_clip(h_best)
    assert np.all((h_best >= 0) & (h_best <= 1))
    assert abs(np.sum(h_best) - target_sum) < 1e-6
    return (h_best.tolist(), best_c5, n_points)
