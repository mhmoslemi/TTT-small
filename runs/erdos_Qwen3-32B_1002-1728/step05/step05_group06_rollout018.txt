def create_full_h(half_h, n_points, half_n):
    full_h = np.zeros(n_points)
    for i in range(half_n):
        full_h[i] = full_h[n_points - 1 - i] = half_h[i]

def compute_smooth_c5(full_h, beta, dx):
    corr = np.correlate(full_h, 1 - full_h, mode='full')
    m = np.max(corr)
    exp_corr = np.exp(beta * (corr - m))
    log_sum = np.log(exp_corr.sum())
    smooth_max = m + log_sum / beta
    return smooth_max * dx
