import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Symmetrize and normalize initial h
    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def normalize(h_full):
        current_sum = np.sum(h_full)
        if current_sum > 1e-8:
            h_full = h_full * (target_sum / current_sum)
        return np.clip(h_full, 0.0, 1.0)

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        valid_corr = corr[n_points - 1:]  # shifts 0 to (n_points-1)
        return np.max(valid_corr) * dx

    h_full = h_initial.copy()
    for i in range(half_n):
        avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
        h_full[i] = h_full[n_points - 1 - i] = avg
    h_full = normalize(h_full)
    h_half = h_full[:half_n].copy()

    best_c5 = compute_c5(h_full)
    best_h_half = h_half.copy()

    learning_rate = 0.1
    lr_decay = 0.999
    epsilon = 1e-5

    np.random.seed(seed)

    while time.time() - start_time < budget_s:
        h_full_current = create_full_h(h_half)
        h_full_current = normalize(h_full_current)
        corr = np.correlate(h_full_current, 1 - h_full_current, mode='full')
        valid_corr = corr[n_points - 1:]
        max_idx = np.argmax(valid_corr)
        k_max = max_idx

        # Compute numerical gradient
        grad = np.zeros(half_n)
        for j in range(half_n):
            h_perturb = h_half.copy()
            h_perturb[j] += epsilon
            hf_perturb = create_full_h(h_perturb)
            hf_perturb = normalize(hf_perturb)
            overlap_perturb = np.sum(hf_perturb[:n_points - k_max] * (1 - hf_perturb[k_max:n_points])) * dx
