        current_c5 = overlaps[k_max]

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_half = h_half.copy()

        n = n_points
        k = k_max
        h_full_arr = h_full.copy()
        g_full = np.zeros(n)
        for m in range(n):
            j = (m + k) % n
            mi = (m - k) % n
            val = 1 - h_full_arr[j] - h_full_arr[mi]
            g_full[m] = dx * val

        g_half = np.zeros(half_n)
        for i in range(half_n):
            g_half[i] = g_full[i] + g_full[n - 1 - i]

        A = np.zeros((n, n))
        for i in range(n):
            A[i, (i + k) % n] = 1.0
        H_full = -dx * (A + A.T)

        J = np.zeros((n, half_n))
        for i in range(half_n):
            J[i, i] = 1.0
            J[n - 1 - i, i] = 1.0
        H_half = J.T @ H_full @ J

        P = H_half + mu * np.eye(half_n)

        delta = cp.Variable(half_n)
        h_curr = h_half.copy()
        constraints = [cp.sum(delta) == 0]
        for i in range(half_n):
            constraints.append(delta[i] >= -h_curr[i])
            constraints.append(delta[i] <= 1 - h_curr[i])

        obj = 0.5 * cp.quad_form(delta, P) + cp.sum(g_half * delta)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)

        if prob.status == cp.OPTIMAL and delta.value is not None:
            new_h_half = h_half + delta.value
            new_h_half = np.clip(new_h_half, 0, 1)
            current_sum = np.sum(new_h_half)
            if abs(current_sum) > 1e-10:
                scale = target_sum_half / current_sum
                new_h_half *= scale
                new_h_half = np.clip(new_h_half, 0, 1)
            h_half = new_h_half

        iteration += 1

    h_full_final = create_full_h(h_half)
    assert abs(np.sum(h_full_final) - n_points / 2.0) < 1e-6
    final_overlaps = compute_overlaps_k(h_full_final)
    return (h_full_final.tolist(), final_overlaps[np.argmax(final_overlaps)], n_points)
