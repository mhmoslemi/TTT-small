import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Total sum of h_full

    def create_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half[:]
        h_full[half_n:] = h_half[::-1]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_and_scale(h_half):
        h_full = create_full(h_half)
        h_clip = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_clip)
        scale = target_sum / current_sum
        h_scaled = h_clip * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        h_half_repair = h_scaled[:half_n]
        return h_half_repair

    # Initialize population
    np.random.seed(seed)
    pop_size = 50
    population = []
    h_initial_half = h_initial[:half_n].copy()
    for _ in range(pop_size):
        perturb = h_initial_half + np.random.uniform(-0.05, 0.05, half_n)
        perturb[...] = repair_and_scale(perturb)[...]
        population.append(perturb.copy())

    best_idx = 0

        # Selection and breeding pool
