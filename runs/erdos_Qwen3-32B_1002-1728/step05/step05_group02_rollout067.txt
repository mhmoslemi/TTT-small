        if current_sum != 0.0:
            new_h *= target_sum / current_sum
            new_h = np.clip(new_h, 0.0, 1.0)
        population.append(new_h)

    # Evaluate initial population
    fitness_list = [compute_c5(ind) for ind in population]
    best_idx = np.argmin(fitness_list)
    best_h = population[best_idx].copy()
    best_c5 = fitness_list[best_idx]

    # GA loop
    while time.time() - start_time < budget_s - 30:
        next_population = []
        tournament_size = 5
        mutation_prob = 0.1

        for _ in range(population_size):
            parent1 = tournament_selection(population, fitness_list, tournament_size)
            parent2 = tournament_selection(population, fitness_list, tournament_size)
            child = (parent1 + parent2) / 2.0

            if np.random.rand() < mutation_prob:
                noise_scale = 0.1 * max(0.1, 1.0 - (time.time() - start_time)/budget_s)
                noise = np.random.normal(0, noise_scale, n_points)
                child += noise
                child = np.clip(child, 0.0, 1.0)
                current_sum = child.sum()
                if current_sum != 0.0:
                    child *= target_sum / current_sum
                    child = np.clip(child, 0.0, 1.0)

            next_population.append(child)

        # Replace population
        population = next_population

        # Re-evaluate and update
        fitness_list = [compute_c5(ind) for ind in population]
        current_best = min(fitness_list)
        if current_best < best_c5:
            best_idx = np.argmin(fitness_list)
            best_h = population[best_idx].copy()
            best_c5 = current_best

        # Local optimization on top candidates
        top_n = max(1, int(0.1 * population_size))
        top_indices = np.argsort(fitness_list)[:top_n]
        for idx in top_indices:
            if time.time() - start_time > budget_s - 20:
                break
            current_h = population[idx].copy()
            current_sum = current_h.sum()

            def obj_func(x):
                x = np.clip(x, 0.0, 1.0)
                x_sum = x.sum() if x.sum() != 0 else 1.0
                x = x / x_sum * target_sum
                x = np.clip(x, 0.0, 1.0)
                return compute_c5(x)

            result = minimize(obj_func, current_h, method="SLSQP", bounds=[(0.0,1.0)]*n_points,
                             options={"maxiter": 30, "ftol": 1e-10})
            if result.success:
                optimized = result.x.reshape(-1)
                optimized = np.clip(optimized, 0.0, 1.0)
                population[idx] = optimized.copy()
                fitness_list[idx] = compute_c5(optimized)
                if fitness_list[idx] < best_c5:
                    best_h = optimized.copy()
                    best_c5 = fitness_list[idx]

    # Final validation
    final_sum = best_h.sum()
    if abs(final_sum - target_sum) > 1e-6:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h *= target_sum / best_h.sum()
    best_c5 = compute_c5(best_h)
    assert abs(best_h.sum() - target_sum) < 1e-6
    assert (0.0 <= best_h).all() and (best_h <= 1.0).all()
    return (best_h.tolist(), best_c5, n_points)
