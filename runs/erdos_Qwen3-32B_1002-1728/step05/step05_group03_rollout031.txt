    max_no_improve = 50
    perturbation_sigma = 0.05
    perturbation_rate = 0.2
    iteration = 0

    while time.time() - start_time < budget_s - 30:
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        c5 = corr[max_idx] * dx

        if c5 < best_c5:
            best_c5 = c5
            best_h_half = h_half.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        grad_full = compute_gradient_full(h_full, k_max)
        grad_half = np.zeros(half_n)
        for i in range(half_n):
            mirrored = n_points - 1 - i
            grad_half[i] = grad_full[i] + grad_full[mirrored]

        h_half = h_half - learning_rate * grad_half
        h_half = normalize_h_half(h_half, target_sum)

        iteration += 1
        if iteration % 100 == 0:
            learning_rate *= 0.95

        if no_improve_count > max_no_improve:
            perturb_mask = np.random.rand(half_n) < perturbation_rate
            noise = np.random.normal(0, perturbation_sigma, half_n)
            h_half += perturb_mask * noise
            h_half = np.clip(h_half, 0, 1)
            h_half = normalize_h_half(h_half, target_sum)
            no_improve_count = 0

    h_full_best = create_full_h(best_h_half)
    corr_best = np.correlate(h_full_best, 1 - h_full_best, mode='full') * dx
    final_c5 = np.max(corr_best)
    assert abs(np.sum(h_full_best) - n_points / 2.0) < 1e-6
    assert np.all(h_full_best <= 1.0 + 1e-6) and np.all(h_full_best >= -1e-6)
    return (h_full_best.tolist(), final_c5, n_points)
