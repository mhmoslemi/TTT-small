import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # Sum constraint for half the points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def project(candidate):
        candidate = np.clip(candidate, 0, 1)
        current_sum = np.sum(candidate)
        delta = target_sum - current_sum
        if abs(delta) < 1e-8:
            return candidate
        if delta > 0:
            idxs = np.where(candidate < 1)[0]
            if len(idxs) > 0:
                amount = delta / len(idxs)
                candidate[idxs] += amount
        else:
            idxs = np.where(candidate > 0)[0]
            if len(idxs) > 0:
                amount = -delta / len(idxs)
                candidate[idxs] -= amount
        return np.clip(candidate, 0, 1)

    def objective(h_half):
        h_half_proj = project(h_half)
        h_full = create_full_h(h_half_proj)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = project(h_half_initial)
    bounds = [(0, 1)] * half_n

    # Estimate maxiter based on remaining time
    remaining = budget_s - (time.time() - start_time)
    maxiter = max(1, int(remaining // 0.1))  # Rough heuristic for small steps

    result = scipy.optimize.differential_evolution(
        objective,
        bounds,
        seed=seed,
        strategy='best1bin',
        maxiter=maxiter,
        popsize=50,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=False,
        init='sobol',
        atol=1e-7,
        disp=False
    )

    best_h_half = project(result.x)
    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(corr) * dx

    return (h_full.tolist(), c5_bound, n_points)
