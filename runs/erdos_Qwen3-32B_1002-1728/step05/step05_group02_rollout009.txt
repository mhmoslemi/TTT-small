        return np.max(corr) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if not np.isclose(current_sum, target_sum) and current_sum != 0:
            h = (h / current_sum) * target_sum
        return np.clip(h, 0, 1)
