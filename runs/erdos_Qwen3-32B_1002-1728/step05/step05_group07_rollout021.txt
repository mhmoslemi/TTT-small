import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    def prepare_initial_guess(perturb_seed):
        np.random.seed(perturb_seed)
        h_full = create_full_h(h_initial[:half_n])
        h_full = (h_full - h_full.min()) / (h_full.max() - h_full.min()) if h_full.ptp() != 0 else h_full
        h_full = np.clip(h_full, 0, 1)
        current_sum = h_full.sum()
        if current_sum > 1e-9:
            h_full = h_full * (n_points / 2.0) / current_sum
        h_full = np.clip(h_full, 0, 1)
        h_half = h_full[:half_n]
        h_half += np.random.normal(0, 0.1, half_n)
        return np.clip(h_half, 0, 1)

    def objective_penalized(p):
        c5 = compute_c5(p)
        cv = (np.sum(p) - target_sum_half)**2
        return c5 + 1e6 * cv

    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = prepare_initial_guess(seed)

    max_tries = 6
    tries = 0
    while time.time() - start_time < budget_s - 30 and tries < max_tries:
        tries += 1
        h0 = prepare_initial_guess(42 * tries)
        res = minimize(objective_penalized, h0, method='Nelder-Mead',
                       options={'maxiter': 200, 'xatol': 1e-4, 'fatol': 1e-4, 'disp': False})
        if not res.success:
            continue
        h_candidate = np.clip(res.x, 0, 1)
        # Fix constraint violation
        current_sum = h_candidate.sum()
        if not np.isclose(current_sum, target_sum_half, atol=1e-3):
            h_candidate = h_candidate * (target_sum_half / current_sum)
        h_candidate = np.clip(h_candidate, 0, 1)
        current_c5 = compute_c5(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate

    # Final pass to enforce constraints
    final_h = create_full_h(best_h)
    total = final_h.sum()
    final_h *= (n_points / 2.0) / total
    final_h = np.clip(final_h, 0, 1)
    return (final_h.tolist(), best_c5, n_points)
