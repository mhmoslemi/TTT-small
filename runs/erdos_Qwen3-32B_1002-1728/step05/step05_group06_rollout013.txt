import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def project(h_full):
        h_full = np.clip(h_full, 0.0, 1.0)
        current_sum = np.sum(h_full)
        if current_sum != 0:
            scale = target_sum / current_sum
            h_full *= scale
            h_full = np.clip(h_full, 0.0, 1.0)
        return h_full

    h_full = h_initial.copy()
    for i in range(half_n):
        avg = (h_full[i] + h_full[n_points - 1 - i]) / 2.0
        h_full[i] = h_full[n_points - 1 - i] = avg
    h_full = project(h_full)
