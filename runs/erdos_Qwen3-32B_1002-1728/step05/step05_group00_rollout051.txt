import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_full):
        h_clipped = np.clip(h_full, 0, 1)
        corr_vals = np.correlate(h_clipped, 1 - h_clipped, mode='full')
        return np.max(corr_vals) * dx

    # Initial solution setup
    h_clipped = np.clip(h_full_initial, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum != 0:
        h_scaled = (h_clipped / current_sum) * target_sum
    else:
        h_scaled = h_clipped.copy()
    h_scaled = np.clip(h_scaled, 0, 1)
    best_full = h_scaled.copy()
    best_c5 = objective(best_full)

    # DE with full vectors
    bounds = [(0.0, 1.0)] * n_points
    def de_obj(x_vars):
        total = np.sum(x_vars)
        if total != 0:
            x_scaled = (x_vars / total) * target_sum
        else:
            x_scaled = x_vars.copy()
        x_clipped = np.clip(x_scaled, 0, 1)
        return objective(x_clipped)

    de_params = {
        'strategy': 'currenttobest1bin',
        'maxiter': 8,
        'popsize': 10,
        'mutation': (0.7, 1.2),
        'recombination': 0.9,
        'tol': 1e-5,
        'seed': seed,
        'workers': 1
    }
    try:
        res_de = differential_evolution(de_obj, bounds, **de_params)
        if res_de.success:
            h_de = np.clip(res_de.x, 0, 1)
            tot = np.sum(h_de)
            if tot != 0:
                h_de = (h_de / tot) * target_sum
            h_de = np.clip(h_de, 0, 1)
            new_c5 = objective(h_de)
            if new_c5 < best_c5:

                best_c5 = new_c5
    except:

    # Perturbation and refinement loop
    np.random.seed(seed)

            if tot != 0:

            perturbed = np.clip(perturbed, 0, 1)

            result = minimize(

                        best_full = new_sol
                        best_c5 = val
