            h_full[i] = h_full[n_points - 1 - i] = h_half[i]

        sum_h_full = h_full.sum()

            scale = (n_points / 2.0) / sum_h_full
            h_full *= scale
            h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.log(np.sum(np.exp(corr / tau))) * tau

    def finite_diff_gradient(h_half, tau, eps=1e-4):
