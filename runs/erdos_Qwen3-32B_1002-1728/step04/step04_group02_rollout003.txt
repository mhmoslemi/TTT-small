            h = h * (target_sum / total)
        return h

    population_size = 50
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = initial_h.copy()
        noise = np.random.normal(0, 0.1, h.shape)
        h += noise
        h = normalize_individual(h)
        population.append(h)
    population = np.array(population)

    best_individual = population[0].copy()
    best_c5 = compute_c5(best_individual)

    generation = 0
    stagnation_counter = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1

        fitness = np.array([compute_c5(ind) for ind in population])
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx].copy()
        current_best_c5 = fitness[current_best_idx]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_individual[:] = current_best
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        if stagnation_counter > 100:
            mutation_rate = 0.3
        else:
            mutation_rate = 0.1

        parents = [tournament_selection(population, fitness) for _ in range(population_size)]
        offspring = []
        for i in range(0, population_size, 2):
            p1 = parents[i]
            p2 = parents[i + 1] if i + 1 < population_size else parents[0]
            cp = np.random.randint(1, n_points)
            c1 = np.concatenate((p1[:cp], p2[cp:]))
            c2 = np.concatenate((p2[:cp], p1[cp:]))
            c1 = normalize_individual(c1)
            c2 = normalize_individual(c2)
            offspring.append(c1)
            offspring.append(c2)

        for child in offspring:
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[j] += np.random.normal(0, 0.1)
                    child[j] = np.clip(child[j], 0, 1)
            child[:] = normalize_individual(child)

        elite_count = 5
        sorted_indices = np.argsort(fitness)
        elites = population[sorted_indices[:elite_count]].copy()
        new_offspring = [normalize_individual(child) for child in offspring[:population_size - elite_count]]
        population = np.array(elites + new_offspring)

    assert abs(np.sum(best_individual) - target_sum) < 1e-6
    assert np.all((best_individual >= -1e-6) & (best_individual <= 1.0 + 1e-6))
    return (best_individual.tolist(), best_c5, n_points)
