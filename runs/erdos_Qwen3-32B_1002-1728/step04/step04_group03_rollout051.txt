    base_pattern = np.array([0,3,3,0,1,2,2,1], dtype=float)
    full_pattern = np.tile(base_pattern, (n_points + 7) // 8)[:n_points]
    h_initial = full_pattern / 3.0  # map to [0, 1]
    target_sum = n_points / 2.0
    current_sum = np.sum(h_initial)
    if current_sum == 0:
        return h_initial  # shouldn't happen
    scaling_factor = target_sum / current_sum
    h_scaled = h_initial * scaling_factor
    h_clipped = np.clip(h_scaled, 0.0, 1.0)
    # Now, check the sum after clipping
    sum_after_clip = np.sum(h_clipped)
    if sum_after_clip < target_sum:
        # Need to adjust to meet the sum. Scale the unclipped elements.
        # Find indices where h_scaled was not clipped.
        mask = (h_scaled < 1.0) & (h_scaled > 0.0)
        if np.sum(mask) == 0:
            # All elements are clipped, cannot adjust.
            return h_clipped
        total_unclipped = sum_after_clip - np.sum(h_clipped[~mask])  # sum of the unclipped elements is total_unclipped
