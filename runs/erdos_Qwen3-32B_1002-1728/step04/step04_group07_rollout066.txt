import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_clipped = np.clip(h_vars, 0, 1)
        total = np.sum(h_clipped)
        if total < 1e-10:
            return float("inf")
        h_rescaled = h_clipped * (target_sum_half / total)
        h_full = create_full_h(h_rescaled)
        corr = np.correlate(h_full, 1 - h_full, mode="full")
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]

    # Initialize from symmetric-rescaled initial_h_values
    h_half_initial = h_full_initial[:half_n].copy()
    init_total = np.sum(h_half_initial)
    if init_total > 1e-10:
        h_half_initial = (h_half_initial / init_total) * target_sum_half
    else:
        h_half_initial = np.ones(half_n) * (target_sum_half / half_n)

    # Time allocation
    de_timer = max(int(min(600, 0.7 * budget_s)), 100)  # Allocate 70% of budget for DE
    local_timer = max(int(budget_s - (time.time() - start_time)), 20)

    # Adaptive DE parameters
    maxiter_de = int(min(500, de_timer // (half_n * 10)))  # Heuristic for DE iterations
    if maxiter_de == 0:
        maxiter_de = 1

    try:
        de_result = differential_evolution(
            objective,
            bounds,
            strategy="best1bin",
            maxiter=maxiter_de,
            popsize=10,
            recombination=0.9,
            tol=1e-3,
            mutation=(0.5, 1.0),
            seed=seed,
            polish=False,
            # Callback to check time in next phase
        )
        de_success = de_result.success
        best_de = de_result.x
    except Exception:
        de_success = False
        best_de = h_half_initial  # Fall back to initialization

    # Rescale DE solution
    h_clipped = np.clip(best_de, 0, 1.0)
    total = np.sum(h_clipped)
    if total > 1e-10:
        h_half_rescaled = h_clipped * (target_sum_half / total)
    else:
        h_half_rescaled = h_clipped

    # Local refinement with SLSQP
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    bounds_local = bounds
    h_refined = h_half_rescaled.copy()

    if de_success and (time.time() - start_time) < (budget_s - 20):
        local_result = minimize(
            fun=objective,
            x0=h_refined,
            bounds=bounds_local,
            constraints=cons,
            method="SLSQP",
            options={'maxiter': 100, 'disp': False}
        )
        if local_result.success:
            h_refined = local_result.x

    # Final evaluation and validation
    h_full = create_full_h(h_refined)
    h_full_clipped = np.clip(h_full, 0, 1.0)
    final_sum = np.sum(h_full_clipped)
    if abs(final_sum - n_points / 2.0) > 1e-6:
        # Normalize if constraints violated
        scaling = (n_points / 2.0) / (final_sum or 1.0)
        h_full_clipped = h_full_clipped * scaling
        h_full_clipped = np.clip(h_full_clipped, 0, 1.0)

    corr = np.correlate(h_full_clipped, 1 - h_full_clipped, mode="full")
    c5_bound = np.max(corr) * dx

    assert np.all(np.logical_and(h_full_clipped >= -1e-6, h_full_clipped <= 1.0 + 1e-6))
    assert abs(np.sum(h_full_clipped) - n_points / 2.0) < 1e-6

    return (h_full_clipped.tolist(), c5_bound, n_points)
