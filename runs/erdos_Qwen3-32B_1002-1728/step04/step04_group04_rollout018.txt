            best_c5 = compute_c5(current_h)
            T = max(T * 2.0, initial_T)
            no_improve_count = 0

        if T < 1e-6:
            T = 1e-6

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Out of bounds"
    return (best_h.tolist(), best_c5, n_points)
