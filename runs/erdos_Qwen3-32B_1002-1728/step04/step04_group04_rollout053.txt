import numpy as np
import time
import random

def evaluate_objective(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    np.random.seed(seed)
    random.seed(seed)

    h_current = h_initial.copy()
    h_total = np.sum(h_current)
    h_current = (h_current / h_total) * target_sum if h_total > 1e-10 else np.zeros_like(h_current)
    h_current = np.clip(h_current, 0, 1)
    best_h = h_current.copy()
    best_c5 = evaluate_objective(best_h, dx)

    T = 1.0
    alpha = 0.999
    perturb_frac = 0.07  # 7% of points to perturb
    min_perturb = 3
    max_perturb = 5

    while time.time() - start_time < budget_s:
        # Dynamically adjust perturb count based on n_points
        perturb_count = min(max(int(n_points * perturb_frac), min_perturb), max_perturb)
        indices = np.random.choice(n_points, perturb_count, replace=False)
        
        # Size of perturbation: proportional to T
        delta = np.random.uniform(-0.2 * T, 0.2 * T, size=perturb_count)
        candidate = h_current.copy()
        candidate[indices] += delta
        candidate = np.clip(candidate, 0, 1)

        # Normalize to satisfy constraint
        candidate_total = np.sum(candidate)
        if np.isclose(candidate_total, 0):
            candidate = h_current.copy()
        else:
            candidate = candidate * (target_sum / candidate_total)
            candidate = np.clip(candidate, 0, 1)

        current_c5 = evaluate_objective(candidate, dx)
        delta_c5 = current_c5 - best_c5

        # Metropolis acceptance
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current = candidate.copy()
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5

        # Aggressive cooling in the final phase
        T *= alpha
        if time.time() - start_time > budget_s - 30 and T > 1e-4:
            T = 1e-4  # Prevent zero-temperature stall

        # Faster cooling in final 60s
        if time.time() - start_time > budget_s - 60:
            alpha = 0.9995

    # Final assertions
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
