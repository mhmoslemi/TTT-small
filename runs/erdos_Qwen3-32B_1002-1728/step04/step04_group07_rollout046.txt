        sum_clipped = np.sum(h_clipped)
        if abs(sum_clipped) < 1e-8:
            return 1e6
        h_clipped = h_clipped * target_sum_half / sum_clipped
        h_full = create_full_h(h_clipped)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n

    h_half = h_full_initial[:half_n].copy()
    sum_h = np.sum(h_half)
    if sum_h != 0:
        h_half = h_half * target_sum_half / sum_h
    h_half = np.clip(h_half, 0, 1)
    best_c5 = objective_de(h_half)
    best_h_half = h_half.copy()

    np.random.seed(seed)

    # Global exploration with DE
    max_de_time = 800.0
    de_start = time.time()
    while time.time() - start_time < max_de_time and time.time() - start_time < budget_s - 50:
        result_de = differential_evolution(
            objective_de, bounds, strategy='best1bin', 
            maxiter=50, popsize=30, mutation=(0.5, 0.9), 
            recombination=0.85, seed=None, polish=False
        )
        if result_de.success:
            candidate_h = result_de.x
            candidate_sum = np.sum(candidate_h)
            if abs(candidate_sum) < 1e-8:
                continue
            h_scaled = candidate_h * target_sum_half / candidate_sum
            h_clipped = np.clip(h_scaled, 0, 1)
            sum_clipped = np.sum(h_clipped)
            if abs(sum_clipped) < 1e-8:
                continue
            h_clipped = h_clipped * target_sum_half / sum_clipped
            h_full_test = create_full_h(h_clipped)
            test_corr = np.correlate(h_full_test, 1 - h_full_test, mode='full')
            test_c5 = np.max(test_corr) * dx
            if test_c5 < best_c5:
                best_h_half = h_clipped.copy()
                best_c5 = test_c5

    # Local refinement with SLSQP
    local_budget = budget_s - max_de_time
    if local_budget > 20:
        def local_objective(x):
            h_full = create_full_h(x)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx
        
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
        bounds_local = bounds
        result_local = minimize(
            local_objective, x0=best_h_half, method='SLSQP', 
            bounds=bounds_local, constraints=cons, 
            options={'maxiter': 500, 'ftol': 1e-10}
        )
        if result_local.success:
            improved_h_half = result_local.x
            h_full = create_full_h(improved_h_half)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            im_c5 = np.max(corr) * dx
            if im_c5 < best_c5:
                best_h_half = improved_h_half
                best_c5 = im_c5

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)  # Final safeguard

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    return (h_full.tolist(), best_c5, n_points)
