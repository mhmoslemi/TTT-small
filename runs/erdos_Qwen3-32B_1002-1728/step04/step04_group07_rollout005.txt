        x0=best_h_half,
        no_local_search=True
    )
    if global_result.success and global_result.fun < best_c5:
        best_h_half = global_result.x
        best_c5 = global_result.fun

    # Local refinement phase
    local_time = 500
    elapsed = time.time() - start_time
    while elapsed + 10 < budget_s:
        current_result = minimize(
            objective_function,
            x0=best_h_half,
            method='L-BFGS-B',
            bounds=bounds_per_var,
            options={'maxiter': 200, 'ftol': 1e-10}
        )
        if current_result.success and current_result.fun < best_c5:
            best_h_half = current_result.x
            best_c5 = current_result.fun
        elapsed = time.time() - start_time

        # Additional perturbations
        if elapsed + 50 >= budget_s:
            break
        perturb = np.random.normal(0, 0.05, size=half_n)
        new_h_half = best_h_half + perturb
        new_h_half = np.clip(new_h_half, 0, 1)
        best_h_half = new_h_half

    h_half_best = normalize_h_half(best_h_half, target_sum_half)
    h_full_best = create_full_h(h_half_best)

    assert abs(np.sum(h_full_best) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full_best >= -1e-6) & (h_full_best <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full_best.tolist(), best_c5, n_points)
