    best_c5 = compute_c5(create_full_h(best_h_half))

    np.random.seed(seed)
    pop_size = 50
    t_size = 3
    mut_rate = 0.1
    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_half})

    def tournament_selection(pop, fit, size):
        idx = np.random.choice(len(pop), size, replace=False)
        best_idx = idx[np.argmin(fit[idx])]
        return pop[best_idx]

    population = [h_half_initial.copy()]
    for _ in range(pop_size - 1):
        new_h = h_half_initial.copy() + np.random.normal(0, 0.05, half_n)
        new_h = np.clip(new_h, 0, 1)
        s = new_h.sum()
        if s != 0:
            new_h *= target_sum_half / s
        new_h = np.clip(new_h, 0, 1)
        population.append(new_h)

    population = np.array(population)

    while time.time() - start_time < budget_s - 60:
        fitness = np.array([compute_c5(create_full_h(ind)) for ind in population])
        new_pop = []
        for _ in range(pop_size // 2):
            p1 = tournament_selection(population, fitness, t_size)
            p2 = tournament_selection(population, fitness, t_size)
            if np.random.rand() < 0.7:
                cp = np.random.randint(1, half_n)
                c1, c2 = np.concatenate([p1[:cp], p2[cp:]])[np.newaxis], np.concatenate([p2[:cp], p1[cp:]])[np.newaxis]
            else:
                c1, c2 = p1[np.newaxis], p2[np.newaxis]
            for ch in [c1, c2]:
                if np.random.rand() < 0.5:
                    mut = ch + np.random.normal(0, mut_rate, (1, half_n))
                    mut = np.clip(mut, 0, 1)
                    s = mut.sum()
                    if s != 0:
                        mut *= target_sum_half / s
                    mut = np.clip(mut, 0, 1)
                    ch = mut
                new_pop.append(ch[0])
        refined = []
        for ch in new_pop:
            res = minimize(lambda x: compute_c5(create_full_h(x)), ch, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-8})
            refined.append(res.x if res.success else ch)
        refined_fitness = np.array([compute_c5(create_full_h(ind)) for ind in refined])
        combined = np.vstack([population, refined])
        combined_fit = np.concatenate([fitness, refined_fitness])
        idx = np.argsort(combined_fit)
        population = combined[idx][:pop_size]
        if np.min(fitness) < best_c5:
            best_h_half = population[0]
            best_c5 = np.min(fitness)

    h_full = create_full_h(best_h_half)
    assert abs(h_full.sum() - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
