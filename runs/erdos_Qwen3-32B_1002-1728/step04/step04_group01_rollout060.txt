import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_full_from_left(left_half):
        full = np.zeros(n_points)
        for i in range(half_n):
            full[i] = left_half[i]
            full[n_points - 1 - i] = left_half[i]
        full_clipped = np.clip(full, 0, 1)
        current_sum = np.sum(full_clipped)
        if current_sum != 0:
            full_scaled = full_clipped * (target_sum / current_sum)
            full_scaled = np.clip(full_scaled, 0, 1)
            return full_scaled
        else:
            return np.zeros(n_points)

    def evaluate_c5(left_half):
        full = create_full_from_left(left_half)
        corr = np.correlate(full, 1 - full, mode='full')
        return np.max(corr) * dx

    pop_size = 40
    mutation_rate = 0.1
    perturb_amount = 0.05
    n_mutate = max(1, int(0.05 * half_n))
    elite_ratio = 0.05
    elite_count = max(1, int(pop_size * elite_ratio))

    population = []
    left_initial = np.zeros(half_n)
    for i in range(half_n):
        left_initial[i] = (h_full_initial[i] + h_full_initial[n_points - 1 - i]) / 2
    population.append(left_initial)

    np.random.seed(seed)
    for _ in range(pop_size - 1):
        left = np.random.uniform(0, 1, half_n)
        population.append(left)

    best_c5 = np.inf
    best_left = None

    while time.time() - start_time < budget_s - 10:
        population_fitness = []
        for i in range(pop_size):
            try:
                c5 = evaluate_c5(population[i])
            except:
                c5 = float('inf')
            population_fitness.append(c5)
        current_gen_best_idx = np.argmin(population_fitness)
        current_gen_best_c5 = population_fitness[current_gen_best_idx]
        if current_gen_best_c5 < best_c5:
            best_c5 = current_gen_best_c5
            best_left = population[current_gen_best_idx].copy()

        sorted_indices = np.argsort(population_fitness)
        new_population = [population[sorted_indices[i]] for i in range(elite_count)]
        remaining = pop_size - elite_count

        for _ in range(remaining):
            contestants = np.random.randint(0, pop_size, 3)
            contestant_fitness = [population_fitness[i] for i in contestants]
            winner_idx = contestants[np.argmin(contestant_fitness)]
            parent1 = population[winner_idx]

            winner_idx = contestants[np.argmin(contestant_fitness)]
            parent2 = population[winner_idx]
            split = np.random.randint(1, half_n)
            child = np.concatenate((parent1[:split], parent2[split:])).copy()
            if np.random.rand() < mutation_rate:
