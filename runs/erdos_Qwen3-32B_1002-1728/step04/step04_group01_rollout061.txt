import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def create_full(h_half):
        full = np.zeros(n_points)
        full[:half_n] = h_half
        full[half_n:] = h_half[::-1]
        return full

    def repair_h(h_half):
        h = create_full(h_half).copy()
        h_sum = np.sum(h)
        if h_sum != 0:
            h = np.clip(h, 0, 1)
            h = (h / h_sum) * target_sum
        return h[:half_n]

    def evaluate(h_half):
        h = create_full(h_half)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)

    population_size = 40
    p_mutate = 0.1
    mutation_strength = 0.05

    # Initialize symmetric population
    pop = []
    h_half = h_init[:half_n].copy()
    pop.append(repair_h(h_half))
    
    rng = np.random.RandomState(seed)
    for _ in range(population_size - 1):
        h_half = rng.rand(half_n)
        pop.append(repair_h(h_half))

    best_c5 = min(evaluate(ind) for ind in pop)
    best_gene = min(pop, key=evaluate)

    while time.time() - start_time < budget_s - 10:
        next_gen = []
        # Tournament selection with pressure
        for _ in range(population_size // 2):
            # Select 2 parents via tournament
            candidates = [rng.choice(pop) for _ in range(3)]
            parent1 = min(candidates, key=evaluate)
            candidates = [rng.choice(pop) for _ in range(3)]
            parent2 = min(candidates, key=evaluate)
            
            # Crossover
            if rng.rand() < 0.8:
                cross_point = rng.randint(1, half_n)
                child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            else:
                child = parent1.copy()
            
            # Mutation
            if rng.rand() < p_mutate:
                mut_mask = rng.rand(half_n) < 0.05
                child[mut_mask] += rng.normal(0, mutation_strength, np.sum(mut_mask))
                child = np.clip(child, 0, 1)
            
            child = repair_h(child)
            next_gen.append(child)
        
        # Elitism
        next_gen.append(best_gene)
        pop = sorted(next_gen + pop, key=evaluate)[:population_size]
        
        # Track best solution
        current_best = min(pop, key=evaluate)
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_gene = current_best
        
        # Emergency termination check
        if time.time() - start_time > budget_s - 5:
            break

    final_h = create_full(best_gene)
    assert abs(np.sum(final_h) - target_sum) < 1e-6
    assert np.all((final_h >= -1e-6) & (final_h <= 1.0 + 1e-6))
    
    return (final_h.tolist(), best_c5, n_points)
