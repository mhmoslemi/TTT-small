            new_p = current + velocity
            # Clip and normalize
            new_p = np.clip(new_p, 0, 1)
            s = np.sum(new_p)
            if s > 1e-6:
                new_p *= target_sum / s
            particles[i] = new_p
            # Evaluate
            cost = compute_c5_fft(new_p)
            if cost < pbest_cost[i]:
                pbest_pos[i] = new_p.copy()
                pbest_cost[i] = cost
                if cost < gbest_cost:
                    gbest_pos[:] = new_p
                    gbest_cost = cost

    h_full = gbest_pos
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    return (h_full.tolist(), gbest_cost, n_points)
