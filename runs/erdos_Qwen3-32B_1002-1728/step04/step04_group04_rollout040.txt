    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_current = h_full_initial.copy()
    current_sum = np.sum(h_current)
    if abs(current_sum) < 1e-10:
        h_current = np.zeros_like(h_current)
    else:
        h_current = (h_current / current_sum) * target_sum
    h_current = np.clip(h_current, 0, 1)
    y_samples = [compute_c5(h_current)]
    X_samples = [h_current.copy()]
    best_h = h_current.copy()
    best_c5 = y_samples[0]

    def rbf_kernel(x1, x2, l):
        return np.exp(-np.sum((x1 - x2)**2) / (2 * l**2))

    def gp_predict(x_star, X, y, l, noise=1e-8):
        n = len(X)
        K = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                K[i, j] = rbf_kernel(X[i], X[j], l)
        K += noise * np.eye(n)
        try:
            K_inv = np.linalg.inv(K)
        except np.linalg.LinAlgError:
            K_inv = np.linalg.pinv(K)
        k_star = np.array([rbf_kernel(x_star, x, l) for x in X])
        mu = k_star @ (K_inv @ y)
        var = rbf_kernel(x_star, x_star, l) - k_star @ K_inv @ k_star
        return mu, var

    def expected_improvement(mu, sigma, f_min):
        if sigma == 0.0:
            return max(f_min - mu, 0.0)
        z = (f_min - mu) / sigma
        def norm_cdf(z):
            return 0.5 * (1 + np.erf(z / np.sqrt(2)))
        def norm_pdf(z):
            return np.exp(-0.5 * z**2) / np.sqrt(2 * np.pi)
        ei = sigma * (z * norm_cdf(z) + norm_pdf(z))
        return ei

    def generate_candidate(n_points, target_sum):
        h = np.random.uniform(0.0, 1.0, n_points)
        h_sum = h.sum()
        if h_sum != 0:
            h = h / h_sum * target_sum
        h = np.clip(h, 0.0, 1.0)
        return h

    l = 1.0
    num_candidate_per_iter = 50
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        current_f_min = best_c5
        candidates = []
        eis = []
        for _ in range(num_candidate_per_iter):
            candidate = generate_candidate(n_points, target_sum)
            mu, var = gp_predict(candidate, X_samples, y_samples, l)
            ei = expected_improvement(mu, np.sqrt(var), current_f_min)
            candidates.append(candidate)
            eis.append(ei)
        best_idx = np.argmax(eis)
        next_candidate = candidates[best_idx].copy()
        new_c5 = compute_c5(next_candidate)
        X_samples.append(next_candidate)
        y_samples.append(new_c5)
        if new_c5 < best_c5:
            best_h = next_candidate.copy()
            best_c5 = new_c5

    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Out of bounds"
    return (h_full.tolist(), best_c5, n_points)
