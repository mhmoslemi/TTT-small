import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # PSO parameters
    n_particles = 50
    inertia = 0.7
    cognitive = 1.0
    social = 1.0
    velocity_max = 0.5

    np.random.seed(seed)
    def compute_c5(h_full):
        n = len(h_full)
        h_fft = np.fft.fft(h_full, 2 * n - 1)
        one_minus_h_fft = np.fft.fft(1.0 - h_full, 2 * n - 1)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * dx

    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_fitness = []

    # First particle: initial_h_values scaled to constraint
    pos = h_initial.copy()
    pos = np.clip(pos, 0, 1)
    s = np.sum(pos)
    if s != 0:
        pos *= target_sum / s
    else:
        pos = h_initial.copy()
    particles.append(pos)
    velocities.append(np.random.uniform(-velocity_max, velocity_max, size=n_points))
    pbest_positions.append(pos.copy())
    pbest_fitness.append(compute_c5(pos))

    # Create remaining particles with perturbed initial values
    for _ in range(n_particles - 1):
        while True:
            noise = np.random.normal(0, 0.1, size=n_points)
            pos = h_initial + noise
            pos = np.clip(pos, 0, 1)
            s = np.sum(pos)
            if s != 0:
                pos *= target_sum / s
                break
        particles.append(pos)
        velocities.append(np.random.uniform(-velocity_max, velocity_max, size=n_points))
        pbest_positions.append(pos.copy())
        pbest_fitness.append(compute_c5(pos))

    best_idx = np.argmin(pbest_fitness)
    gbest = pbest_positions[best_idx].copy()
    best_c5 = pbest_fitness[best_idx]

    while time.time() - start_time < budget_s - 5:

            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (

            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            s = np.sum(particles[i])
            if s != 0:
