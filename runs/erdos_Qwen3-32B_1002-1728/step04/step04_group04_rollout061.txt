import numpy as np
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    total = h_clipped.sum()
    if abs(total) < 1e-12:
        h_projected = np.zeros_like(h)
    else:
        h_projected = (h_clipped / total) * target_sum
        h_projected = np.clip(h_projected, 0, 1)
    return h_projected

def compute_gradient(h, k_shift, n_points):
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Contributions from h[i] * (1 - h[i + k_shift])
        if k_shift > 0:
            if i + k_shift < n_points:
                grad[i] += 1 - h[i + k_shift]
            if i - k_shift >= 0:
                grad[i] += -h[i - k_shift]
        elif k_shift < 0:
            shifted_k = -k_shift
            if i - shifted_k >= 0:
                grad[i] += 1 - h[i - shifted_k]
            if i + shifted_k < n_points:
                grad[i] += -h[i + shifted_k]
        else:  # k_shift == 0
            grad[i] += 1 - h[i]
            grad[i] += -h[i]  # -h[i] from the second term
    return grad

    lr = 0.01
    lr_decay = 0.9995
