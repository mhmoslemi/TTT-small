            continue
        scaling_factor = target_sum / sum_perturbed
        new_h_half_scl = new_h_half * scaling_factor
        # Clip again because scaling might push beyond [0,1]
        new_h_half_scl = np.clip(new_h_half_scl, 0, 1)

        # Evaluate new candidate
        try:
            new_c5 = objective_fft(new_h_half_scl)
        except:
            new_c5 = float('inf')  # Skip invalid candidates

        # Acceptance logic
        if new_c5 < best_c5:
            best_h_half = new_h_half_scl.copy()
            best_c5 = new_c5
        else:
            delta = new_c5 - best_c5
            if np.exp(-delta / current_T) > np.random.rand():
                best_h_half = new_h_half_scl.copy()
                best_c5 = new_c5

        # Cool temperature
        current_T *= alpha_cooling

    # Gradient refinement after simulated annealing
    result = minimize(
        fun=objective_fft,
        x0=best_h_half,
        method='SLSQP',
        bounds=bounds_per_var,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success:
        refined_h_half = result.x
        refined_h_full = create_full_h(refined_h_half)
        if np.all(refined_h_full <= 1 + 1e-6) and np.all(refined_h_full >= 0):
            refined_c5 = result.fun
            if refined_c5 < best_c5:
                best_h_half = refined_h_half
                best_c5 = refined_c5

    # Final verification
    h_full = create_full_h(best_h_half)
    assert np.isclose(np.sum(h_full), n_points / 2.0, atol=1e-6)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
