import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def objective_full(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}

    # Validate and adjust initial_h if necessary
    def adjust_h(h):
        h_clipped = np.clip(h, 0, 1)
        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            return h.copy()
        scale_factor = target_sum / sum_h
        scaled = h_clipped * scale_factor
        return np.clip(scaled, 0, 1)

    best_h = adjust_h(h_initial.copy())
    best_c5 = objective_full(best_h)

    np.random.seed(seed)
    iter_count = 0

    while time.time() - start_time < budget_s - 60:
        iter_count += 1
        # Generate perturbation with decaying noise
        noise_scale = 0.1 * (1 - (iter_count / (budget_s / 100)) ** 0.5)
        perturb = np.random.normal(0, noise_scale, n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective_full,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        if result.success:
            new_h = result.x
            new_c5 = objective_full(new_h)
            if new_c5 < best_c5:
                best_h = new_h
                best_c5 = new_c5

    # Final validation
    best_h = adjust_h(best_h)
    best_c5 = objective_full(best_h)

    return (best_h.tolist(), best_c5, n_points)
