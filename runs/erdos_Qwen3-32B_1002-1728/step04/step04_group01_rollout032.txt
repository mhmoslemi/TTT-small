import numpy as np
from scipy.optimize import minimize_scalar
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Total sum constraint
    target_half_sum = target_sum / 2  # First half sum constraint

    # Create symmetric clusters for the first half
    h_half_initial = h_full_initial[:half_n].copy()
    clusters = []
    start = 0
    current_val = h_half_initial[start]
    for i in range(1, half_n):
        if h_half_initial[i] != current_val:
            clusters.append((start, i-1, current_val))
            start = i
            current_val = h_half_initial[i]
    clusters.append((start, half_n-1, current_val))

    def build_h_full(current_clusters):
        h_half = np.zeros(half_n)
        for s, e, v in current_clusters:
            h_half[s:e+1] = v
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    best_h_clusters = clusters.copy()
    best_h_full = build_h_full(best_h_clusters)
    best_c5 = evaluate_c5(best_h_full)

    np.random.seed(seed)
    elapsed = 0

    while time.time() - start_time < budget_s - 10:
        # Select the cluster with max v*(1-v)
        max_impact = -1
        best_cluster_idx = -1
        for idx, (s, e, v) in enumerate(clusters):
            impact = v * (1 - v)
            if impact > max_impact:
                max_impact = impact
                best_cluster_idx = idx
        if best_cluster_idx == -1:
            break

        # Extract and remove the best cluster to split
        s_split, e_split, v_split = clusters.pop(best_cluster_idx)
        total_len = e_split - s_split + 1
        original_sum = v_split * total_len
        mid = (s_split + e_split) // 2
        len1 = mid - s_split + 1
        len2 = e_split - mid

        # Define the valid range for v1
        v1_min = max(0.0, (original_sum - len2 * 1.0) / len1)
        v1_max = min(1.0, original_sum / len1)
        if v1_min > v1_max:
            clusters.insert(best_cluster_idx, (s_split, e_split, v_split))
            continue

        # Objective function for 1D optimization
        def objective(v1):
            v2 = (original_sum - len1 * v1) / len2
            if v1 < v1_min or v1 > v1_max or v2 < 0.0 or v2 > 1.0:
                return float('inf')
            temp_clusters = []
            temp_clusters.extend(clusters)
            temp_clusters.append((s_split, mid, v1))
            temp_clusters.append((mid+1, e_split, v2))
            h_full = build_h_full(temp_clusters)
            if not (abs(np.sum(h_full) - target_sum) < 1e-6):
                return float('inf')
            return evaluate_c5(h_full)

        # Perform scalar optimization
        result = minimize_scalar(objective, bounds=(v1_min, v1_max), method='bounded', options={'xatol': 1e-4})
        if result.success:
            v1_new = result.x
            v2_new = (original_sum - len1 * v1_new) / len2
            # Add new clusters
            clusters.append((s_split, mid, v1_new))
            clusters.append((mid+1, e_split, v2_new))
            # Check if this is better
            temp_clusters = clusters.copy()
            h_full_new = build_h_full(temp_clusters)
            c5_new = evaluate_c5(h_full_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h_clusters = temp_clusters.copy()
                best_h_full = h_full_new.copy()
        else:
            clusters.insert(best_cluster_idx, (s_split, e_split, v_split))

    h_full_final = build_h_full(best_h_clusters)
    assert abs(np.sum(h_full_final) - target_sum) < 1e-6
    return (h_full_final.tolist(), best_c5, n_points)
