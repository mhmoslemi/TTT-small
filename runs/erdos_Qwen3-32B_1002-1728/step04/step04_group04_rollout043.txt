import numpy as np
import time

def compute_gradient(h, k_lag, n, dx):
    term1 = np.zeros(n)
    term2 = np.zeros(n)
    if k_lag >= 0:
        if k_lag < n:
            term1[k_lag:] = 1.0 - h[:-k_lag]
            term2[:n - k_lag] = h[k_lag:]
    else:
        abs_k = -k_lag
        if abs_k < n:
            term1[:n - abs_k] = 1.0 - h[abs_k:]
            term2[abs_k:] = h[:-abs_k]
    grad = (term1 - term2) * dx
    return grad
