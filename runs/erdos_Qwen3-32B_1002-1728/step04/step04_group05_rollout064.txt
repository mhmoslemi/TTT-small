    dx = 2.0 / n_points
    target_sum = n_points / 4.0

    def create_full_h(y):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = y[i]
        return h_full

    def evaluate_candidate(y):
        h_full = create_full_h(y)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def project_y(y, target_sum):
        y_clamped = np.clip(y, 0.0, 1.0)
        current_sum = np.sum(y_clamped)
        if current_sum < 1e-10:
            return y_clamped.copy()
        scaling_factor = target_sum / current_sum
        y_scaled = y_clamped * scaling_factor
        y_projected = np.clip(y_scaled, 0.0, 1.0)
        return y_projected

    # Initialize symmetric y
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - n_points / 2.0) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (n_points / 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n].copy()
    best_y = initial_h_half.copy()
    best_c5 = evaluate_candidate(best_y)
    current_y = best_y.copy()

    # Optimization parameters
    learning_rate = 1e-3
    perturbation_mag = 0.01
    tabu_size = 50
    max_no_improvement = 50
    no_improvement_count = 0
    max_perturbation_incs = 3
    current_perturbation_inc = 0
    tabu_list = []

    np.random.seed(int(seed) % 2**32)

    while time.time() - start_time < budget_s - 10:
        grad = np.zeros(half_n)
        epsilon = 1e-5
        for i in range(half_n):
            y_p = current_y.copy()
            y_p[i] += epsilon
            y_p_p = project_y(y_p, target_sum)
            y_n = current_y.copy()
            y_n[i] -= epsilon
            y_n_n = project_y(y_n, target_sum)
            grad[i] = (evaluate_candidate(y_p_p) - evaluate_candidate(y_n_n)) / (2 * epsilon)

        current_y = current_y - learning_rate * grad
        current_y = project_y(current_y, target_sum)
        current_c5 = evaluate_candidate(current_y)

        if current_c5 < best_c5:
            best_y = current_y.copy()
            best_c5 = current_c5
            no_improvement_count = 0
            similar = False
            for tabu_y in tabu_list:
                if np.linalg.norm(current_y - tabu_y) < 0.1:
                    similar = True
                    break
            if not similar:
                tabu_list.append(current_y.copy())
                if len(tabu_list) > tabu_size:
                    tabu_list.pop(0)
        else:
            no_improvement_count += 1

        if no_improvement_count > max_no_improvement:
            perturbation = np.random.normal(0, perturbation_mag, half_n)
            current_y = project_y(current_y + perturbation, target_sum)
            no_improvement_count = 0
            new_similar = False
            for tabu_y in tabu_list:
                if np.linalg.norm(current_y - tabu_y) < 0.1:
                    new_similar = True
                    break
            if new_similar:
                perturbation_mag *= 2
                current_perturbation_inc += 1
                if current_perturbation_inc >= max_perturbation_incs:
                    perturbation_mag = 0.01

        learning_rate *= 0.999

    h_full = create_full_h(best_y)
    assert np.isclose(np.sum(h_full), n_points / 2.0, atol=1e-4), f"Invalid sum: {np.sum(h_full)}"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0+1e-6))
    return (h_full.tolist(), best_c5, n_points)
