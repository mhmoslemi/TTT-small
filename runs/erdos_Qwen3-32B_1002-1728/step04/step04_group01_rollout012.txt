import numpy as np
import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum == 0:
            return np.zeros_like(h)
        return h * (target_sum / h_sum)
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    
    # Frequency space initialization and repair
    H = fft(h_initial)
    H_candidate = np.zeros_like(H)
    H_candidate[0] = 0.5  # DC component
    if n_points % 2 == 0:
        H_candidate[n_points//2] = 0.5  # Nyquist for even N
    for k in range(1, n_points//2):
        H_candidate[k] = 0.5
        H_candidate[-k] = 0.5
    
    h_freq = np.real(ifft(H_candidate))
    h_freq = repair_h(h_freq)
    c5_freq = compute_c5(h_freq)
    
    if c5_freq < best_c5:
        best_h = h_freq.copy()
        best_c5 = c5_freq
    
    # Frequency optimization with time budget
    np.random.seed(seed)
    h_best_freq = best_h.copy()
    c5_best = best_c5
    
    while time.time() - start_time < budget_s - 5:
        perturb_freqs = min(3, n_points // 2 - 1)
        H_candidate = 0.5 * np.ones_like(H)
        for _ in range(perturb_freqs):
            k = np.random.randint(1, n_points//2)
            pert = np.random.normal(0, 0.1)
            val = 0.5 + pert
            H_candidate[k] = val
            H_candidate[-k] = val
        
        h_candidate = np.real(ifft(H_candidate))
        h_candidate = repair_h(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        
        if c5_candidate < c5_best:
            h_best_freq = h_candidate.copy()
            c5_best = c5_candidate
    
    # Repair final solution
    final_h = repair_h(h_best_freq)
    final_c5 = compute_c5(final_h)
    
    assert abs(np.sum(final_h) - target_sum) < 1e-6, f"Sum constraint violated: {np.sum(final_h)}"
    assert np.all((final_h >= -1e-6) & (final_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (final_h.tolist(), final_c5, n_points)
