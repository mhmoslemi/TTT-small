        indices = rng.integers(0, len(pop), tournament_size)
        winner_idx = indices[np.argmin(fit[indices])]
        return pop[winner_idx].copy()

    while time.time() - start_time < budget_s - 30:
        fitness = np.array([evaluate_candidate(population[i]) for i in range(population_size)])
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h_half = population[current_best_idx].copy()

        new_offspring = []
        for _ in range(population_size // 2):
            p1 = tournament_select(population, fitness)
            p2 = tournament_select(population, fitness)
            perturbation = rng.normal(0.0, 0.07, half_n)
            donor = p1 + 0.7 * perturbation
            donor = normalize_h_half(donor)
            mask = rng.random(half_n) < 0.9
            child = np.where(mask, donor, p2)
            child = normalize_h_half(child)
            new_offspring.append(child)
        new_offspring = np.array(new_offspring)

        temp_population = np.vstack([population, new_offspring])
        temp_fitness = np.array([evaluate_candidate(temp_population[i]) for i in range(temp_population.shape[0])])
        indices = np.argsort(temp_fitness)
        population = temp_population[indices[:population_size]]

        top_count = max(1, int(0.1 * population_size))
        for i in range(min(top_count, population_size)):
            current_h_half = population[i].copy()
            def objective(x):
                h_full = create_symmetric_h_full(x)
                corr = np.correlate(h_full, 1 - h_full, mode='full')
                return np.max(np.abs(corr)) * dx
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
            bounds = Bounds([0.0]*half_n, [1.0]*half_n)
            try:
                result = minimize(objective, current_h_half, constraints=constraints, bounds=bounds, method='SLSQP', options={'maxiter': 50})
                if result.success:
                    optimized = result.x.copy()
                    optimized = normalize_h_half(optimized)
                    population[i] = optimized
            except Exception:
                pass

    h_full = create_symmetric_h_full(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
