from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    target_sum_half = target_sum / 2.0  # sum of h_half_initial must equal target_sum_half

    # Normalize initial h_full_initial to meet integral constraint
    current_sum = np.sum(h_full_initial)
    if current_sum < 1e-6 or abs(current_sum - target_sum) > 1e-6:
        h_full_initial = np.clip(h_full_initial * (target_sum / current_sum), 0.0, 1.0)

    half_n = n_points // 2

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def evaluate_candidate(h_half_candidate):
        h_full = create_symmetric_h_full(h_half_candidate)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    current_sum = np.sum(h_full_sym)

    population_size = 50

    crossover_rate = 0.7
