        n = len(h)
        h_fft = np.fft.fft(h, n=2 * n)
        s_fft = np.fft.fft(1.0 - h, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

            current_h[:] = candidate.copy()
