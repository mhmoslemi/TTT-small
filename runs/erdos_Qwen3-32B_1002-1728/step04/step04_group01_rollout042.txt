import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_correlation(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def process_individual(h_half):
        h_full = create_full_h(h_half)
        total = np.sum(h_full)
        if total != 0:
            h_full = (h_full / total) * target_sum
        h_full = np.clip(h_full, 0, 1)
        return h_full[:half_n]

    def tournament_select(candidates, scores, size=3):
        idxs = np.random.choice(len(candidates), size)
        best_idx = idxs[np.argmax(scores[idxs])]
        return candidates[best_idx]

    np.random.seed(seed)
    pop_size = 40
    population = []

    # Initialize population
    h_initial_sym = create_full_h((h_initial + h_initial[::-1])/2)
    h_initial_sym = (h_initial_sym / h_initial_sym.sum() * target_sum) if h_initial_sym.sum() != 0 else h_initial_sym
    h_initial_sym = np.clip(h_initial_sym, 0, 1)
    population.append(h_initial_sym[:half_n])

    for _ in range(pop_size - 1):
        h_half = np.random.rand(half_n)
        h_full = create_full_h(h_half)
        total = h_full.sum()
        if total != 0:
            h_full = (h_full / total) * target_sum
        h_full = np.clip(h_full, 0, 1)
        population.append(h_full[:half_n])

    best_c5 = float('inf')
    best_h = None
    start_time = time.time()

    while time.time() - start_time < budget_s:
        # Evaluate current population
        c5_list = [evaluate_correlation(p) for p in population]
        valid = np.isfinite(c5_list)
        if np.any(valid):
            idx_best = np.nanargmin(np.array(c5_list))
            if c5_list[idx_best] < best_c5:
                best_c5 = c5_list[idx_best]
                best_h = population[idx_best].copy()

        # Select elites and breeders
        scores = np.array([-c5 for c5 in c5_list])
        idxs = np.argsort(scores)
        top_k = int(0.05 * pop_size)
        elites = [population[i] for i in idxs[:top_k]]
        
        new_pop = elites[:]
        while len(new_pop) < pop_size:
            p1 = tournament_select(population, scores[valid])
            p2 = tournament_select(population, scores[valid])
            
            # Crossover
            cross_point = np.random.randint(1, half_n)
            child = np.concatenate([p1[:cross_point], p2[cross_point:]])
            
            # Mutation
            mut_idx = np.random.choice(half_n, min(half_n, 3), replace=False)
            child = np.clip(child + np.random.normal(0, 0.05, half_n), 0, 1)
            child = process_individual(child)
            
            new_pop.append(child)

        population = new_pop[:pop_size]

    if best_h is None:
        best_h = population[np.nanargmin(c5_list)].copy()
        best_c5 = evaluate_correlation(best_h)

    h_full = create_full_h(best_h)
    assert abs(np.sum(h_full) - target_sum) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    
    return (h_full.tolist(), best_c5, n_points)
