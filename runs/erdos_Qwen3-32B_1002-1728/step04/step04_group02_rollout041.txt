import numpy as np
import time

def evaluate_c5(h_full, dx):
    n = len(h_full)
    padding_length = 2 * n - 1
    h_fft = np.fft.fft(h_full, padding_length)
    one_minus_h_fft = np.fft.fft(1.0 - h_full, padding_length)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    return np.max(corr) * dx

def tournament_selection(population, dx):
    participants = np.random.choice(population, size=3)
    best = participants[0]
    min_c5 = evaluate_c5(best, dx)
    for p in participants[1:]:
        c5 = evaluate_c5(p, dx)
        if c5 < min_c5:
            best = p
            min_c5 = c5
    return best

def crossover(parent1, parent2):
    n_points = len(parent1)
    split_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate((parent1[:split_point], parent2[split_point:]))
    child2 = np.concatenate((parent2[:split_point], parent1[split_point:]))
    return child1, child2

def mutate(individual, mutation_rate):
    n_points = len(individual)
    if np.random.rand() < mutation_rate:
        mask = np.random.rand(n_points) < 0.1
        noise = np.random.normal(0, 0.05, size=n_points)
        individual = np.where(mask, individual + noise, individual)
    return individual

def adjust_child(individual, target_sum):

    sum_current = np.sum(individual)
    if sum_current != 0:
        individual = individual * (target_sum / sum_current)
    individual = np.clip(individual, 0, 1)
    return individual
