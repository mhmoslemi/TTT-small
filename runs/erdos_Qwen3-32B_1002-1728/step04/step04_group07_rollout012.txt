import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points

        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    # Initial guess for DE
    h_initial_half = h_full_initial[:half_n].copy()

    h_initial_half = (h_initial_half / np.sum(h_initial_half)) * target_sum_half
    h_initial_half = np.clip(h_initial_half, 0, 1)

        if sum_h == 0:
            return float('inf')
