    best_c5 = float('inf')
    best_h_half = population[0].copy()

    def evaluate(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        h_full = np.clip(h_full, 0.0, 1.0)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    def refine_h(h_half, steps=5):
        for _ in range(steps):
            if time.time() - start_time > budget_s - 5:
                break
            h_full = np.zeros(n_points)
            h_full[:half_n] = h_half
            h_full[-half_n:] = h_half[::-1]
            h_full = np.clip(h_full, 0.0, 1.0)
            corr = np.correlate(h_full, 1.0 - h_full, mode='full')
            max_idx = np.argmax(np.abs(corr))
            grad = np.zeros(half_n)
            eps = 1e-4
            for i in range(half_n):
                cand = h_half.copy()
                cand[i] += eps
                cand = cand * target_sum_half / np.sum(cand)
                cand = np.clip(cand, 0.0, 1.0)
                h_full_p = np.zeros(n_points)
                h_full_p[:half_n] = cand
                h_full_p[-half_n:] = cand[::-1]
                corr_p = np.correlate(h_full_p, 1.0 - h_full_p, mode='full')
                diff = (np.abs(corr_p[max_idx]) - np.abs(corr[max_idx])) / eps
                grad[i] = diff
            step_size = 0.05
            h_half -= step_size * grad
            h_half = h_half * target_sum_half / np.sum(h_half)
            h_half = np.clip(h_half, 0.0, 1.0)
        return h_half

    while time.time() - start_time < budget_s - 5:
        scores = np.array([evaluate(population[i]) for i in range(population_size)])
        best_idx = np.argmin(scores)
        if scores[best_idx] < best_c5:
            best_c5 = scores[best_idx]
            best_h_half = population[best_idx].copy()

        if iterations % refine_freq == 0:
            top_idx = np.argsort(scores)[:min(10, population_size)]
            for idx in top_idx:
                if time.time() - start_time > budget_s - 5:
                    break
                population[idx] = refine_h(population[idx].copy())

        new_population = []
        for i in range(population_size):
            if time.time() - start_time > budget_s - 5:
                break
            idxs = rng.choice(population_size, 3, replace=False)
            a, b, c = population[idxs]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = mutant * target_sum_half / np.sum(mutant)
            cross_mask = rng.random(half_n) < CR
            trial = np.where(cross_mask, mutant, population[i])
            trial = trial * target_sum_half / np.sum(trial)
            trial = np.clip(trial, 0.0, 1.0)
            new_population.append(trial)
        population = np.vstack((population, np.array(new_population)))[:population_size]

        elite = np.argsort(scores)[:population_size // 2]
        worse = np.argsort(scores)[population_size // 2:]
        population[worse] = population[elite]
        iterations += 1

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0.0, 1.0)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
