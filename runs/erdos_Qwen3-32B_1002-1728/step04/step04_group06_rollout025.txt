                        best_c5 = new_c5
                        h_reshaped = create_symmetric_h(np.zeros(half_n))
                        h_reshaped[:half_n] = optimized_h[:half_n]
                        best_h_half = h_reshaped[:half_n]

        # Time check
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break

    # Final best
    final_full_h = create_symmetric_h(best_h_half)
    final_full_h = normalize_h(final_full_h)
    final_c5 = compute_c5(final_full_h)
    
    # Ensure constraints
    assert abs(np.sum(final_full_h) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((final_full_h >= -1e-6) & (final_full_h <= 1 + 1e-6)), "Bounds violated"
    
    return (final_full_h.tolist(), final_c5, n_points)
