                new_c5 = evaluate_candidate(h_opt)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h_half = h_opt.copy()

        # Generate new population via DE
        for i in range(population_size):
            if time.time() - start_time >= budget_s - 60:
                break
            idxs = rng.choice(population_size, size=3, replace=False)
            a, b, c = population[idxs]
            trial = a + de_mut * (b - c)
            cross_idx = rng.integers(half_n)
            mutant = np.copy(population[i])
            for j in range(half_n):
                if j == cross_idx or rng.random() < de_cross:
                    mutant[j] = trial[j]
            mutant = np.clip(mutant, 0.0, 1.0)
            current_sum = np.sum(mutant)
            if current_sum > 1e-8:
                scale_f = target_sum_half / current_sum
            else:
                scale_f = 0.0
            mutant = np.clip(mutant * scale_f, 0.0, 1.0)
            current_fit = fitness[i]
            mutant_fit = evaluate_candidate(mutant)
            if mutant_fit < current_fit:
                population[i] = mutant.copy()
                if mutant_fit < best_c5:
                    best_c5 = mutant_fit
                    best_h_half = mutant.copy()

    # Final validation
    h_full = create_symmetric_h_full(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
