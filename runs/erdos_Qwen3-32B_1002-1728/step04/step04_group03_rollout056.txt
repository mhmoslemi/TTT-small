import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective_fft(h_full):
        n = len(h_full)
        h_fft = np.fft.fft(h_full, n=2 * n)
        s_fft = np.fft.fft(1.0 - h_full, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

    current_h = h_initial.copy()
    best_h = current_h.copy()
    best_c5 = objective_fft(best_h)
    np.random.seed(seed)
    temp = 10.0
    cooling_rate = 0.995

    while time.time() - start_time < budget_s - 30:
        # Perturb with scaled Gaussian noise
        noise = np.random.normal(0, 1, n_points) * temp / n_points
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        # Enforce sum constraint via uniform scaling
        current_sum = np.sum(new_h)
        if abs(current_sum) > 1e-9:
            new_h = new_h * (target_sum / current_sum)
        new_h = np.clip(new_h, 0, 1)
        # Compute new objective
        new_c5 = objective_fft(new_h)
        # Accept new h with Metropolis rule
        if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / max(temp, 1e-6)):
            current_h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        # Cool down
        temp *= cooling_rate

    # Local refinement using SLSQP
    bounds = [(0.0, 1.0)] * n_points
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result = minimize(
        fun=objective_fft,
        x0=current_h.copy(),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    if result.success:
        refined_h = result.x.clip(0, 1)
        refined_sum = np.sum(refined_h)
