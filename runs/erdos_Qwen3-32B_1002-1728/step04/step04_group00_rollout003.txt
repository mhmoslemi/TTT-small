import numpy as np

import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective_clipped_scaled(h_half):
        h_half_clipped = np.clip(h_half, 0, 1)
        current_sum = np.sum(h_half_clipped)
        if current_sum < 1e-10:
            h_half_clipped = np.zeros_like(h_half_clipped)
            h_half_clipped += (target_sum_half / h_half.size) * 1e-6
            h_half_clipped = np.clip(h_half_clipped, 0, 1)
        else:
            scaling_factor = target_sum_half / current_sum
            h_half_clipped *= scaling_factor
            h_half_clipped = np.clip(h_half_clipped, 0, 1)
        h_full = create_full_h(h_half_clipped)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    # Generate symmetric initial solution
    h_full_initial_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_initial_sym[i] = h_full_initial_sym[j] = avg
    h_half_initial = h_full_initial_sym[:half_n]
    current_sum = np.sum(h_half_initial)
    if abs(current_sum - target_sum_half) > 1e-6:
        h_half_initial = h_half_initial * (target_sum_half / current_sum)
    h_half_initial = np.clip(h_half_initial, 0, 1)

    bounds = [(0.0, 1.0)] * half_n

    # Custom callback to monitor time
    class StopEarly(Exception): pass

    def time_callback(x, f, c):
        if time.time() - start_time > budget_s - 30:
            raise StopEarly()
        return False

    minimizer_kwargs = {"method": "L-BFGS-B"}

    best_h = h_half_initial.copy()
    best_c5 = objective_clipped_scaled(best_h)

        result = basinhopping(
