    import numpy as np
    import time

    # Load initial h
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # CVXPY variables
    h = cp.Variable(n_points, nonneg=True)
    t = cp.Variable()

    # Constraints
    constraints = [
        cp.sum(h) == target_sum,
        h <= 1.0
    ]

    # Generate all possible k shifts and corresponding constraints
    for k in range(-n_points + 1, n_points):
        # Find valid i where i +k is in [0, n_points-1]
        valid_i = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_i.append(i)
        if not valid_i:
            continue  # no overlap for this k
        # Compute sum_h_i = sum_{i in valid_i} h[i]
        sum_h_i = cp.sum(h[valid_i])
        # sum_hihj = sum_{i in valid_i} h[i] * h[i +k]
        indices = [i +k for i in valid_i]
        sum_hihj = cp.sum(h[valid_i] * h[indices])
        # Constraint: dx * (sum_h_i - sum_hihj) <= t
        constraints.append(dx * (sum_h_i - sum_hihj) <= t)

    # Objective: minimize t
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)

    # Set up solver
    prob.solve(solver=cp.MOSEK, verbose=True, mosek_params={"MSK_DPAR_OPT_TOL_FEAS": 1e-8, "MSK_DPAR_OPT_TOL_REL_GAP": 1e-6, "MSK_IPAR_LOG": 0}, warm_start=True)

    # Check solver status
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_values = h.value
        c5_bound = t.value
    else:
        # Fallback to initial solution if optimization failed
        h_values = h_initial
        c5_bound = np.max(np.correlate(h_values, 1 - h_values, mode='full') * dx)

    return (h_values.tolist(), c5_bound, n_points)
