import numpy as np
import time
from scipy.optimize import minimize
from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def normalize_h_half(h_half):
        current_sum = np.sum(h_half)
        if current_sum == 0:
            h_half = np.full_like(h_half, target_sum_half / half_n)
        else:
            scale = target_sum_half / current_sum
            h_half = h_half * scale
        return np.clip(h_half, 0.0, 1.0)

    h_half_initial = np.zeros(half_n)
    for i in range(half_n):
        j = n_points - 1 - i
        h_half_initial[i] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_half_initial)
    if abs(current_sum - target_sum_half) > 1e-6 and current_sum != 0:
        h_half_initial = h_half_initial * target_sum_half / current_sum
    h_half_initial = np.clip(h_half_initial, 0.0, 1.0)

    def evaluate(h_half):
        h_full = create_symmetric_h_full(h_half)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    population_size = 40
    rng = default_rng(seed)

    population = np.zeros((population_size, half_n))
    population[0] = h_half_initial.copy()
    for i in range(1, population_size):
        perturb = rng.normal(0.0, 0.1, half_n)
        population[i] = normalize_h_half(h_half_initial + perturb)

    best_score = evaluate(population[0])
    best_h_half = population[0].copy()

    while time.time() - start_time < budget_s - 30:
        # Differential Evolution phase
        for _ in range(10):
            if time.time() - start_time >= budget_s - 30:
                break
            for i in range(population_size):
                idxs = rng.permutation(population_size)
                a, b, c = population[idxs[:3]]

                mutant = a + 0.7 * (b - c)
                cross_mask = rng.random(half_n) < 0.9
                trial = np.where(cross_mask, mutant, population[i])
                trial = normalize_h_half(trial)
                if evaluate(trial) < evaluate(population[i]):
                    population[i] = trial

                current_score = evaluate(population[i])
                if current_score < best_score:
                    best_score = current_score
                    best_h_half = population[i].copy()

        # Local optimization with SLSQP
        def objective(x):
            h_full = create_symmetric_h_full(x)
            corr = np.correlate(h_full, 1.0 - h_full, mode='full')
            return np.max(np.abs(corr)) * dx

        def constraint_sum(x):
            return np.sum(x) - target_sum_half

        if time.time() - start_time < budget_s - 30:
            x0 = best_h_half.copy()
            bounds = Bounds(np.zeros(half_n), np.ones(half_n))
            con_sum = {'type': 'eq', 'fun': constraint_sum}
            res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=con_sum, tol=1e-8)
