import numpy as np
import time
import random

def compute_c5_fast(h_full):
    n = len(h_full)
    dx = 2.0 / n
    h_fft = np.fft.fft(h_full, 2 * n - 1)
    one_minus_h = 1.0 - h_full
    one_minus_h_fft = np.fft.fft(one_minus_h, 2 * n - 1)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    return np.max(corr) * dx

def tournament_selection(fitness, size=3):
    indices = np.random.choice(len(fitness), size, replace=False)
    best_idx = min(indices, key=lambda i: fitness[i])
    return best_idx

def crossover(parent1, parent2, crossover_rate):
    if np.random.rand() < crossover_rate:
        crossover_point = np.random.randint(1, len(parent1)-1)
        child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        return child
    return parent1.copy() if np.random.rand() < 0.5 else parent2.copy()
