import numpy as np
import time
from functools import partial

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial is feasible
    h_initial = np.clip(h_initial / h_initial.sum() * target_sum, 0, 1)
    
    def objective(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Population initialization
    pop_size = 100
    pop = np.zeros((pop_size, n_points))
    pop[0] = h_initial.copy()
    for i in range(1, pop_size):
        pop[i] = np.random.uniform(0, 1, n_points)
        pop[i] = np.clip(pop[i], 0, 1)
        pop[i] = pop[i] / pop[i].sum() * target_sum
        pop[i] = np.clip(pop[i], 0, 1)

    best_idx = 0
    best_c5 = objective(pop[0])
    start_time = time.time()

    def tournament_select(population, fitness, k=3):
        indices = np.random.choice(len(population), k)
        return indices[np.argmin(fitness[indices])]

    def crossover(p1, p2, cr=0.7):
        if np.random.rand() > cr:
            return p1.copy()
        idx = np.random.randint(1, n_points)
        child = np.hstack([p1[:idx], p2[idx:]])
        return child

    def mutate(ind, mr=0.2):
        mut = np.random.randn(n_points) * 0.1
        ind = np.clip(ind + mut, 0, 1)
        ind = ind / ind.sum() * target_sum
        return np.clip(ind, 0, 1)

    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = np.array([objective(ind) for ind in pop])
        current_idx = int(np.argmin(fitness))
        if fitness[current_idx] < best_c5:
            best_idx = current_idx
            best_c5 = fitness[current_idx]

        # Create new population
        new_pop = pop.copy()
        for i in range(pop_size // 2):
            # Selection
            p1_idx = tournament_select(pop, fitness, 3)
            p2_idx = tournament_select(pop, fitness, 3)
            # Crossover
            child1 = crossover(pop[p1_idx], pop[p2_idx], 0.7)
            child2 = crossover(pop[p2_idx], pop[p1_idx], 0.7)
            # Mutation
            child1 = mutate(child1, 0.2)
            child2 = mutate(child2, 0.2)
            # Replace worst
            replace_idx = np.argsort(fitness)[-2:]
            new_pop[replace_idx[0]] = child1
            new_pop[replace_idx[1]] = child2

        pop = new_pop

        # Early stopping if no improvement
        if time.time() - start_time > budget_s - 30:
            break

    h_best = pop[best_idx]
    assert abs(h_best.sum() - target_sum) < 1e-6
    assert np.all((h_best >= 0) & (h_best <= 1))
    return (h_best.tolist(), best_c5, n_points)
