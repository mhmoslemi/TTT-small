            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0.0, 1.0)
            current_sum = particles[i].sum()
            if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
                particles[i] *= target_sum / current_sum
                particles[i] = np.clip(particles[i], 0.0, 1.0)
            current_cost = compute_c5(particles[i])
            if current_cost < pbest_cost[i]:
                pbest[i] = particles[i].copy()
                pbest_cost[i] = current_cost
                if current_cost < gbest_cost:
                    gbest[:] = particles[i]
                    gbest_cost = current_cost
    assert abs(gbest.sum() - target_sum) < 1e-6
    assert np.all((gbest >= 0) & (gbest <= 1))
    return (gbest.tolist(), gbest_cost, n_points)

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    n_particles = 50

    # Initialize particles
    particles = np.zeros((n_particles, n_points))
    particles[0] = h_initial.copy()
    for i in range(1, n_particles):
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        particles[i] = np.clip(particles[0] + perturb, 0, 1)
        s = particles[i].sum()
        if abs(s) > 1e-8:
            particles[i] *= target_sum / s

    # Initialize velocities
    velocities = np.zeros_like(particles)

    # Evaluate initial particles and set personal bests
    def compute_c5(h_full):
        n = len(h_full)
        h_fft = np.fft.fft(h_full, 2*n -1)
        one_minus_h_fft = np.fft.fft(1.0 - h_full, 2*n -1)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * (2.0 / n)

    pbest = particles.copy()
    pbest_cost = np.array([compute_c5(p) for p in particles])
    gbest_idx = np.argmin(pbest_cost)
    gbest = particles[gbest_idx].copy()
    gbest_cost = pbest_cost[gbest_idx]

    # PSO parameters
    w = 0.7; c1 = 1.0; c2 = 1.0; velocity_clamp = 0.5

    while time.time() - start_time < budget_s - 10:
        for i in range(n_particles):
            if time.time() - start_time >= budget_s - 10:
                break

            r1, r2 = np.random.rand(2)
            velocities[i] = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)

            s = particles[i].sum()
            # Normalize while maintaining bounds
            if abs(s - target_sum) > 1e-8 and s != 0:
                particles[i] *= target_sum / s
                particles[i] = np.clip(particles[i], 0, 1)

            # Evaluate and update pbest/gbest
            current_cost = compute_c5(particles[i])
            if current_cost < pbest_cost[i]:
                pbest[i] = particles[i].copy()
                pbest_cost[i] = current_cost

                    gbest_cost = current_cost

    assert np.all((gbest >= -1e-8) & (gbest <= 1.0 + 1e-8)), "Value bounds violated"
    return (gbest.tolist(), gbest_cost, n_points)
