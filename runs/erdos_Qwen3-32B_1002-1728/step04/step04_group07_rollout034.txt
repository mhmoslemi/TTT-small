        if iteration >= max_pso_iter or (time.time() - start_time > budget_s - 100):
            break

    if gbest_value < best_c5:
        best_h_half = gbest_position.copy()
        best_c5 = gbest_value

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    result = minimize(
        fun=objective,
        x0=best_h_half,
        method='SLSQP',
        bounds=bounds_per_var,
        constraints=cons,
        options={'maxiter': 300, 'disp': False, 'ftol': 1e-10}
    )
    if result.success:
        new_h_half = result.x
        if (np.all((new_h_half >= -1e-6) & (new_h_half <= 1.0 + 1e-6))):
            h_full_new = create_full_h(new_h_half)
            new_c5 = objective(new_h_half)
            if new_c5 < best_c5:
                best_h_half = new_h_half
                best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - target_sum_full) < 1.0e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    return (h_full.tolist(), best_c5, n_points)
