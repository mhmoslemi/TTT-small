import numpy as np
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0
    dx = 2.0 / n_points
    
    # Initialize symmetric h_half from initial_h_values
    h_half = np.array([ (h_full_initial[i] + h_full_initial[n_points-1-i])/2 for i in range(half_n) ], dtype=float)
    total = h_half.sum()
    if total != 0:
        h_half *= target_sum / total
    h_half = np.clip(h_half, 0.0, 1.0)
    best_h_half = h_half.copy()
    best_c5 = float('inf')
    
    def objective(h_val):
        h_full = np.empty(n_points)
        h_full[:half_n] = h_val
        h_full[half_n:] = h_val[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx
    
    best_c5 = objective(best_h_half)
    
    # Simulated Annealing parameters
    temp = 1.0
    alpha = 0.998
    min_temp = 1e-5
    start_time = time.time()
    
    while (time.time() - start_time < budget_s - 30) and (temp > min_temp):
        # Dynamic noise scaling by temperature and remaining time
        time_frac = (budget_s - (time.time() - start_time)) / budget_s
        noise_scale = temp * (0.3 + 0.7 * time_frac)
        
        perturbation = np.random.normal(0, noise_scale, half_n)
        new_h_half = best_h_half + perturbation
        new_h_half = np.clip(new_h_half, 0.0, 1.0)
        
        total_new = new_h_half.sum()
        if total_new != 0:
            new_h_half = np.clip(new_h_half * (target_sum / total_new), 0.0, 1.0)
        
        current_c5 = objective(new_h_half)
        delta = current_c5 - best_c5
        
        if (delta < 0) or (np.random.rand() < math.exp(-delta / temp)):
            best_h_half = new_h_half
            best_c5 = current_c5
        
        temp *= alpha
    
    # Final h_full construction
    h_full = np.empty(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0.0, 1.0)
    
    assert abs(h_full.sum() - n_points/2.0) < 1e-6
