import numpy as np
import time
from scipy.optimize import minimize

def objective(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_full_initial = initial_h_values
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize initial h
    current_h = h_full_initial.astype(float).copy()
    current_sum = np.sum(current_h)
    if not np.isclose(current_sum, target_sum):
        current_h = current_h * target_sum / current_sum
    current_h = np.clip(current_h, 0, 1)

    current_c5 = objective(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Simulated Annealing phase
    sa_start = time.time()
    T = 1.5  # Initial temperature
    alpha = 0.99  # Cooling factor per step
    while time.time() - sa_start < budget_s - 200 and time.time() < budget_s:
        # Perturb h
        noise = np.random.normal(0, 0.1, n_points)
        new_h = current_h + noise
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)

        if new_sum < 1e-12:
            new_h = np.full_like(new_h, target_sum / n_points)
            new_sum = target_sum
        else:
            scale = target_sum / new_sum
            new_h = np.clip(new_h * scale, 0, 1)

        # Compute new_c5
        new_c5 = objective(new_h)

        # Acceptance probability
        if new_c5 < best_c5:
            accepted = True
        else:
            temp = T * (0.99 ** (time.time() - sa_start))
            prob = np.exp((best_c5 - new_c5) / (temp + 1e-8))
            if np.random.rand() < prob:
                accepted = True
            else:
                accepted = False

        if accepted:
            current_h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5

        # Cool down
        T *= alpha

    # Deterministic refinement phase
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}]
    method = 'SLSQP' if n_points < 1000 else 'L-BFGS-B'
    result = minimize(
        fun=lambda h: objective(h),
        x0=best_h,
        method=method,
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_sum = np.sum(refined_h)
        if refined_sum > 1e-12:
            refined_h = refined_h * target_sum / refined_sum
            refined_h = np.clip(refined_h, 0, 1)
        refined_c5 = objective(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5

    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if not np.isclose(final_sum, target_sum):
        final_h = final_h * target_sum / (final_sum + 1e-12) * (1 - 1e-6)
        final_h = np.clip(final_h, 0, 1)
    final_c5 = objective(final_h)

    return (final_h.tolist(), final_c5, n_points)
