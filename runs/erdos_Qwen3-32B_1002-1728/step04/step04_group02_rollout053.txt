import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Fast FFT-based overlap integral computation
    def compute_c5(h):
        n = len(h)
        n_full = 2 * n - 1
        h_fft = np.fft.fft(h, n_full)
        one_minus_h_fft = np.fft.fft(1.0 - h, n_full)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * dx

    # PSO parameters
    np.random.seed(seed)
    num_particles = 50
    particles = []
    velocities = []

    # Initialize particles around initial solution
    for _ in range(num_particles):
        noise = 0.15 * np.random.randn(n_points)
        h = h_initial + noise
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 1e-8:
            scaling = target_sum / sum_h
            h = h * scaling
        h = np.clip(h, 0.0, 1.0)
        particles.append(h)
        velocities.append(0.02 * np.random.randn(n_points))

    particles = np.array(particles)
    velocities = np.array(velocities)

    # Personal bests
    personal_bests = particles.copy()
    personal_best_values = np.array([compute_c5(p) for p in particles])
    gbest_idx = np.argmin(personal_best_values)
    gbest_value = personal_best_values[gbest_idx]
    gbest_pos = particles[gbest_idx].copy()

    # PSO config
    w = 0.7
    c1 = 1.5
    c2 = 0.5
    max_velocity = 0.3
    max_no_improve = 80
    no_improve = 0

    while time.time() - start_time < budget_s - 60:
        for i in range(num_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_bests[i] - particles[i]) +
                            c2 * r2 * (gbest_pos - particles[i]))  
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
