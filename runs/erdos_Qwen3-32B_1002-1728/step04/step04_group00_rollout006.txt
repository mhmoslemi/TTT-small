import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values, copy=True)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def adjust_sum(h_half, target):
        h_clipped = np.clip(h_half, 0, 1)
        s = h_clipped.sum()
        if not np.isclose(s, 0):
            scaling = target / s
            h_clipped *= scaling
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    # Normalize initial h_half
    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = adjust_sum(h_half_initial, target_sum_half)

    np.random.seed(seed)
    population_size = 50
    population = [h_half_initial.copy() for _ in range(population_size)]
    for i in range(population_size):
        perturbed = population[i] + 0.1 * np.random.randn(half_n)
        population[i] = adjust_sum(perturbed, target_sum_half)

    best_h_half = h_half_initial.copy()
    best_c5 = float('inf')

    def evaluate(ind):
        h_full = create_full_h(ind)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    generation = 0
    max_gen = 500

    while (time.time() - start_time) < budget_s and generation < max_gen:
        generation += 1

        # Evaluate and find current best
        fitness = [evaluate(ind) for ind in population]
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_h_half = population[idx].copy()
            best_c5 = fitness[idx]

        # Sort population by fitness
        order = np.argsort(fitness)
        sorted_pop = [population[i] for i in order]
        new_pop = sorted_pop[:population_size // 4]  # Elitism

        # Crossover and mutation
        while len(new_pop) < population_size:
            # Select parents from top 50%
            p1, p2 = np.random.choice(population_size // 2, size=2, replace=False)
            parent1 = sorted_pop[p1]
            parent2 = sorted_pop[p2]

            # Crossover: blend with noise
            alpha = np.random.rand(half_n)
            child = alpha * parent1 + (1 - alpha) * parent2
            # Mutation
            if np.random.rand() < 0.2:
                noise = 0.3 * np.random.randn(half_n) * (1 - generation / max_gen)
                child += noise
            child = adjust_sum(child, target_sum_half)
            new_pop.append(child)

        population = new_pop

    h_full = create_full_h(best_h_half)
    assert abs(h_full.sum() - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), best_c5, n_points)
