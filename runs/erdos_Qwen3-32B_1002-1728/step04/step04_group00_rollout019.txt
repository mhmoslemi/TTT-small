import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    if current_sum != 0:
        scaling = target_sum / current_sum

    if abs(np.sum(h_full) - target_sum) > 1e-6:

        h_full = np.clip(h_full, 0, 1)
    best_h = h_full.copy()
    best_c5 = objective(best_h)

    bounds_per_var = [(0.0, 1.0)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    np.random.seed(seed)
    n_attempts = 0

    def noise_scale(min_time=60, max_scale=0.3):

        frac = time_left / budget_s
        return max_scale * (frac ** 1.5)

    while time.time() - start_time < budget_s - 30:
