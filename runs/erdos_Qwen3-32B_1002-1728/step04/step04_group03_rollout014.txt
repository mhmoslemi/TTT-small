from scipy.optimize import minimize
import time

        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        cross_corr = np.correlate(h_full, 1 - h_full, mode='full').real
        return np.max(cross_corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]

    current_sum = np.sum(h_full_sym)

    while time.time() - start_time < budget_s - 30:

        for method in methods:
            if time.time() - start_time >= budget_s - 30:

            try:
