import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    half_n = len(h_init) // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_sym):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_sym[i]
        return h_full

    def objective(h_sym):
        h_full = create_full_h(h_sym)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_constraints(h_sym):
        h_clipped = np.clip(h_sym, 0.0, 1.0)
        s = np.sum(h_clipped)
        if s == 0:
            h_repaired = np.zeros_like(h_clipped)
            h_repaired[0] = target_sum_half
            h_repaired = np.clip(h_repaired, 0, 1)
        else:
            scale = target_sum_half / s
            h_repaired = np.clip(h_clipped * scale, 0.0, 1.0)
        return h_repaired

    np.random.seed(seed)
    # Initialize from symmetric version of h_init
    h_sym_init = np.zeros(half_n)
    for i in range(half_n):
        j = n_points - 1 - i
        h_sym_init[i] = (h_init[i] + h_init[j]) / 2
    h_sym_init = repair_constraints(h_sym_init)

    n_particles = 30
    particles = [h_sym_init + np.random.uniform(-0.3, 0.3, half_n) for _ in range(n_particles)]
    particles = [repair_constraints(p) for p in particles]
    velocities = [np.zeros(half_n) for _ in range(n_particles)]

    w = 0.7
    c1 = 1.5
    c2 = 1.5

    pbest_vals = [objective(p) for p in particles]
    pbests = particles[:]
    gbest_idx = np.argmin(pbest_vals)
    gbest = particles[gbest_idx]

    check_time = start_time
    while time.time() - start_time < budget_s - 30:
        for _ in range(5):
            for i in range(n_particles):
                r1, r2 = np.random.rand(2)
                vel_update = (
                    w * velocities[i] + 
                    c1 * r1 * (pbests[i] - particles[i]) + 
                    c2 * r2 * (gbest - particles[i])
                )
                velocities[i] = np.clip(vel_update, -0.5, 0.5)
                particles[i] = particles[i] + velocities[i]
                particles[i] = repair_constraints(particles[i])
            pbest_vals = [min(pbest_vals[i], objective(particles[i])) for i in range(n_particles)]
            pbests = [pbests[i] if pbest_vals[i] == pbest_vals[i] else particles[i].copy() for i in range(n_particles)]
            gbest_idx = np.argmin(pbest_vals)
            gbest = pbests[gbest_idx]

    # Local refinement
    bounds = [(0, 1)] * half_n
    res = minimize(
        objective,
        x0=gbest,
        method='COBYLA',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}],
        options={'maxiter': 300, 'tol': 1e-10}
    )
    h_final = res.x
    h_full = create_full_h(h_final)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1 + 1e-6))
    return (h_full.tolist(), res.fun, n_points)
