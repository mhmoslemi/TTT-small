import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Symmetric initialization
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 20:
        n_attempt += 1
        if n_attempt > 150:
            break
        # Perturb with smaller range first
        perturbed = best_h_half + np.random.uniform(-0.1, 0.1, size=half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 300, 'disp': False, 'tol': 1e-8}
        )

        if result.success:
            new_h_half = result.x
            # Project to feasible region
            total = np.sum(new_h_half)
            if total != 0:
                new_h_half = new_h_half * (target_sum_half / total)
            new_h_half = np.clip(new_h_half, 0, 1)
            h_full_new = create_full_h(new_h_half)
            new_corr = np.correlate(h_full_new, 1 - h_full_new, mode='full')
            new_c5 = np.max(new_corr) * dx
            if (np.isclose(np.sum(new_h_half), target_sum_half, atol=1e-6) and
                np.all((new_h_half >= -1e-6) & (new_h_half <= 1.0 + 1e-6))):
                if new_c5 < best_c5:
                    best_h_half = new_h_half
                    best_c5 = new_c5

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
