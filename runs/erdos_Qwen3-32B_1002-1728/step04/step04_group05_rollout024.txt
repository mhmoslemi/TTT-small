import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(y):
        h_full = np.zeros(n_points)
        h_full[:half_n] = y
        h_full[half_n:] = y[::-1]
        if n_points % 2 == 1:
            h_full[half_n] = h_full[half_n - 1]
        return h_full

    def evaluate(y):
        h_full = create_full_h(y)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def normalize_y(y):
        s = np.sum(y)
        if s == 0:
            return y.copy()
        y = y / s * target_sum_half
        y = np.clip(y, 0, 1)
        return y

    # Symmetrize initial
    y_initial = h_initial[:half_n].copy()
    y_initial = normalize_y(y_initial)

    # PSO parameters
    SWARM_SIZE = 40
    MAX_VEL = 0.1
    W_INIT, W_FINAL = 0.9, 0.4
    C1, C2 = 2.0, 2.0
    MAX_ITER = 2000

    np.random.seed(seed)
    # Initialize swarm
    swarm = [y_initial.copy() for _ in range(SWARM_SIZE)]
    velocities = [np.random.rand(half_n) * 0.01 for _ in range(SWARM_SIZE)]
    pbest = [y.copy() for y in swarm]
    pbest_scores = [evaluate(y) for y in swarm]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    best_c5 = pbest_scores[gbest_idx]

    # Main loop
    for iter in range(MAX_ITER):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
        # Update inertia weight
        t_ratio = min(iter / MAX_ITER, 1.0)
        inertia = W_INIT - t_ratio * (W_INIT - W_FINAL)
        
        for i in range(SWARM_SIZE):
            y = swarm[i]
            v = velocities[i]
            yp = pbest[i]
            yg = gbest
            
            # Velocity update
            r1, r2 = np.random.rand(2)
            v = (inertia * v +
                 C1 * r1 * (yp - y) +
                 C2 * r2 * (yg - y))
            v = np.clip(v, -MAX_VEL, MAX_VEL)
            
            # Position update
            y_new = y + v
            y_new = np.clip(y_new, 0, 1)
            y_new = normalize_y(y_new)
            
            # Evaluate new position
            score_new = evaluate(y_new)
            
            # Update personal best
            if score_new < pbest_scores[i]:
                pbest[i] = y_new.copy()
                pbest_scores[i] = score_new
                
                # Update global best
                if score_new < best_c5:
                    gbest = y_new.copy()
                    best_c5 = score_new
            
            # Update swarm and velocities
            swarm[i] = y_new
            velocities[i] = v

    # Construct final solution
    h_full = create_full_h(gbest)
    h_full = np.clip(h_full, 0, 1)
    assert np.all(h_full >= -1e-6) and np.all(h_full <= 1.0 + 1e-6)
    assert abs(np.sum(h_full) - n_points/2.0) < 1e-6
    
    return (h_full.tolist(), best_c5, n_points)
