            return h.copy()
        return perturbed * (target_sum / total)

    while time.time() - start_time < budget_s - 60:
        num_attempts += 1
        seed += 1
        np.random.seed(seed)
        new_h0 = perturb_h(h_current)
        # Try different method to avoid local minima
        for method in method_sequence:
            if time.time() - start_time >= budget_s - 60:
                break
            result = minimize(
                objective,
                new_h0,
                method=method,
                bounds=bounds,
                constraints=cons if method not in ['L-BFGS-B'] else (),
                options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                h_candidate = np.clip(result.x, 0.0, 1.0)
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    h_current = h_candidate
                    best_c5 = c5_candidate
    # Final checks
    assert np.all(h_current >= -1e-6) and np.all(h_current <= 1.0 + 1e-6)
    assert np.isclose(np.sum(h_current), target_sum, atol=1e-6)
    return (h_current.tolist(), best_c5, n_points)
