import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    np.random.seed(seed)

    n_low = 14
    group_size = n_points // n_low
    h_low_init = np.zeros(n_low)
    for i in range(n_low):
        start = i * group_size
        end = start + group_size
        h_low_init[i] = np.mean(h_full_initial[start:end])
    target_sum_low = n_low / 2.0
    current_sum = np.sum(h_low_init)
    if current_sum != 0:
        h_low_init = h_low_init * (target_sum_low / current_sum)
    h_low_init = np.clip(h_low_init, 0, 1)

    best_c5 = np.inf
    best_h = h_full_initial.copy()

    num_trials = 5
    for trial in range(num_trials):
        if time.time() - start_time > budget_s - 60:
            break
        perturbation = np.random.uniform(-0.05, 0.05, size=n_low)
        h_low_start = h_low_init + perturbation
        h_low_start = np.clip(h_low_start, 0, 1)
        current_sum = np.sum(h_low_start)
        if current_sum != 0:
            h_low_start = h_low_start * (target_sum_low / current_sum)
        h_low_start = np.clip(h_low_start, 0, 1)

        def objective_low(x):
            h = x.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            dx = 2.0 / n_low
            return np.max(corr) * dx

        cons_low = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_low}]
        bounds_low = [(0, 1)] * n_low

        result_low = minimize(
            objective_low,
            h_low_start,
            method='trust-constr',
            bounds=bounds_low,
            constraints=cons_low,
            options={'maxiter': 100},
        )

        if not result_low.success:
            continue
        h_low_opt = result_low.x

        h_high = np.zeros(n_points)
        for i in range(n_low):
            h_high[i * group_size : (i+1)*group_size] = h_low_opt[i]
        target_sum_high = n_points / 2.0
        current_sum_high = np.sum(h_high)
        if current_sum_high != 0:
            h_high = h_high * (target_sum_high / current_sum_high)
        h_high = np.clip(h_high, 0, 1)

        def objective_high(x):
            h = x.copy()
            corr = np.correlate(h, 1 - h, mode='full')
            dx = 2.0 / n_points
            return np.max(corr) * dx

        cons_high = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_high}]
        bounds_high = [(0, 1)] * n_points

        result_high = minimize(
            objective_high,
            h_high,
            method='trust-constr',
            bounds=bounds_high,
            constraints=cons_high,
            options={'maxiter': 150},
        )

        if result_high.success:
            h_high_opt = result_high.x
            corr_high = np.correlate(h_high_opt, 1 - h_high_opt, mode='full')
            dx_high = 2.0 / n_points
            c5_high = np.max(corr_high) * dx_high
            if c5_high < best_c5:
                best_c5 = c5_high
                best_h = h_high_opt.copy()

    corr_initial = np.correlate(h_full_initial, 1 - h_full_initial, mode='full')
    c5_initial = np.max(corr_initial) * (2.0 / n_points)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = h_full_initial.copy()

    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
