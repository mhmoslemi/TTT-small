        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    bounds_per_variable = [(0, 1) for _ in range(half_n)]

    # Phase 1: Differential Evolution
    de_result = differential_evolution(
        func=objective_candidate,
        bounds=bounds_per_variable,
        x0=h_half_initial,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        maxiter=100,
        disp=False
    )
    best_h_half_de = de_result.x
    best_c5_de = de_result.fun
    elapsed = time.time() - start_time
    if elapsed > budget_s - 60:
        return (create_symmetric_h_full(normalize_h_half(best_h_half_de)).tolist(), best_c5_de, n_points)

    # Phase 2: Local refinement with L-BFGS-B
    def objective_minimize(h_half):
        h_half_norm = normalize_h_half(h_half)
        h_full = create_symmetric_h_full(h_half_norm)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    qp_result = minimize(
        fun=objective_minimize,
        x0=best_h_half_de,
        bounds=bounds_per_variable,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    best_h_half_qp = qp_result.x
    best_c5_qp = objective_minimize(best_h_half_qp)
    final_h_full = create_symmetric_h_full(normalize_h_half(best_h_half_de)) if best_c5_de <= best_c5_qp else create_symmetric_h_full(normalize_h_half(best_h_half_qp))
    final_c5 = best_c5_de if best_c5_de <= best_c5_qp else best_c5_qp

    assert abs(np.sum(final_h_full) - target_sum) < 1e-6
    assert np.all((final_h_full >= -1e-6) & (final_h_full <= 1.0 + 1e-6))
    
    return (final_h_full.tolist(), final_c5, n_points)
