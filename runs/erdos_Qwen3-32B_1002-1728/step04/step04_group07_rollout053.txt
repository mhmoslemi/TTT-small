import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_position(pos):
        avg = pos.mean()
        pos_clipped = np.clip(pos, 0, 1)
        total = np.sum(pos_clipped)
        if total <= 1e-10:
            pos_clipped = np.random.uniform(0, 1, half_n)
            pos_clipped = (pos_clipped / total) * target_sum_half
            pos_clipped = np.clip(pos_clipped, 0, 1)
        else:
            pos_clipped = (pos_clipped / total) * target_sum_half
            pos_clipped = np.clip(pos_clipped, 0, 1)
        return pos_clipped

    # Symmetric initialization
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]
    
    h_half = repair_position(h_half)

    # Setup for PSO
    np.random.seed(seed)
    n_particles = 30
    w = 0.7

    c2 = 1.5

    # Initialize positions around the symmetric initial
    positions = []
    velocities = []
    for _ in range(n_particles):

        perturbed = h_half.copy() + np.random.uniform(-0.1, 0.1, size=half_n)
        perturbed = repair_position(perturbed)
        positions.append(perturbed)
        velocities.append(np.random.uniform(-0.05, 0.05, size=half_n))
    positions = np.array(positions)
    velocities = np.array(velocities)

    # Personal best and global best
    pbest_positions = positions.copy()

    gbest_index = np.argmin(pbest_scores)

    gbest_score = pbest_scores[gbest_index]
    best_h_half = gbest.copy()
    best_c5 = gbest_score

    while time.time() - start_time < budget_s * 0.7:  # 70% of budget for PSO
        # Update velocities and positions
        r1 = np.random.rand(n_particles, half_n)
        r2 = np.random.rand(n_particles, half_n)
        velocities = (w * velocities +
                      c1 * r1 * (pbest_positions - positions) +
                      c2 * r2 * (gbest - positions))
        positions += velocities

        # Repair positions
        for i in range(n_particles):
            positions[i] = repair_position(positions[i])

        # Update personal best and global best
        for i in range(n_particles):
            current_score = objective(positions[i])
            if current_score < pbest_scores[i]:
                pbest_scores[i] = current_score
                pbest_positions[i] = positions[i].copy()

            gbest = pbest_positions[current_gbest_index].copy()
            best_h_half = gbest.copy()
            best_c5 = gbest_score

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]

        new_corr = np.correlate(h_full_new, 1 - h_full_new, mode='full')
        new_c5 = np.max(new_corr) * dx

import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_position(pos):
        pos_clipped = np.clip(pos, 0, 1)
        total = np.sum(pos_clipped)
        if abs(total) < 1e-10:
            pos_clipped = np.random.uniform(0, 1, half_n) * (target_sum_half / half_n)
        else:
            pos_clipped = (pos_clipped / total) * target_sum_half
        pos_clipped = np.clip(pos_clipped, 0, 1)
        return pos_clipped

    # Symmetric initialization
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - n_points / 2.0) > 1e-6:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]
    h_half = repair_position(h_half)

    np.random.seed(seed)
    n_particles = 20
    w = 0.7
    c1 = 1.5
    c2 = 1.5

    positions = []
    velocities = []
    for _ in range(n_particles):
        perturbed = h_half.copy() + np.random.uniform(-0.1, 0.1, half_n)
        perturbed = repair_position(perturbed)
        positions.append(perturbed)
        velocities.append(np.random.uniform(-0.05, 0.05, half_n))
    positions = np.array(positions)
    velocities = np.array(velocities)

    pbest_positions = positions.copy()
    pbest_scores = np.array([objective(p) for p in pbest_positions])
    gbest = pbest_positions[np.argmin(pbest_scores)].copy()
    best_c5 = objective(gbest)
    best_h_half = gbest.copy()

    while time.time() - start_time < budget_s * 0.8:
        r1 = np.random.rand(n_particles, half_n)
        r2 = np.random.rand(n_particles, half_n)
        velocities = w * velocities + c1 * r1 * (pbest_positions - positions) + c2 * r2 * (gbest - positions)
        positions += velocities
        for i in range(n_particles):
            positions[i] = repair_position(positions[i])
        current_scores = np.array([objective(p) for p in positions])
        improved_mask = current_scores < pbest_scores
        pbest_scores[improved_mask] = current_scores[improved_mask]
        pbest_positions[improved_mask] = positions[improved_mask]
        new_gbest_idx = np.argmin(pbest_scores)
        new_gbest = pbest_positions[new_gbest_idx]
        if objective(new_gbest) < best_c5:
            best_c5 = objective(new_gbest)
            best_h_half = new_gbest.copy()

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    result = minimize(objective, best_h_half, bounds=bounds_per_var, constraints=cons, method='SLSQP', options={'maxiter': 500, 'tol': 1e-8})
    if result.success:
        candidate = result.x
        candidate = repair_position(candidate)
        new_c5 = objective(candidate)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h_half = candidate

    while time.time() - start_time < budget_s - 20:
        perturbed = best_h_half.copy() + np.random.uniform(-0.1, 0.1, half_n)
        perturbed = repair_position(perturbed)
        result = minimize(objective, perturbed, bounds=bounds_per_var, constraints=cons, method='COBYLA', options={'maxiter': 200})
        if result.success:
            candidate = result.x
            candidate = repair_position(candidate)
            new_c5 = objective(candidate)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_half = candidate

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
