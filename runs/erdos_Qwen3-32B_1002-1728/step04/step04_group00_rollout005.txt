import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    population_size = 50
    max_generations = 500

    # Initialize population
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n]

    population = [initial_h_half.copy()]
    for _ in range(population_size - 1):
        perturbed = initial_h_half + np.random.normal(0, 0.1, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_h_half = initial_h_half.copy()
    best_c5 = compute_c5(best_h_half)

    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1

        # Evaluate population
        candidates = [(compute_c5(h), h.copy()) for h in population]
        candidates.sort(key=lambda x: x[0])
        parents = [h for c, h in candidates[:population_size // 2]]

        # Create new population
        new_population = parents.copy()
        while len(new_population) < population_size:
            p1 = parents[np.random.randint(len(parents))]

            child = (p1 + p2) / 2.0
            if np.random.rand() < 0.2:
                mutation_strength = 0.3 * (1 - (generation / max_generations) ** 1.5)
                child += np.random.normal(0, mutation_strength, half_n)
                child = np.clip(child, 0, 1)

            if total != 0:

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
