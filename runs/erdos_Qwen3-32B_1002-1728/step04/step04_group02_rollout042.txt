            p[:] = 0.5
        particles.append(p)
    particles = np.array(particles)
    velocities = np.zeros_like(particles)

    # Evaluate initial C5 and pbest
    pbest_positions = particles.copy()
    current_c5 = [compute_c5(p, dx) for p in particles]
    pbest_c5 = np.array(current_c5)
    gbest_idx = np.argmin(pbest_c5)
    gbest_position = pbest_positions[gbest_idx]
    gbest_c5 = pbest_c5[gbest_idx]
    best_c5 = gbest_c5
    best_particle = gbest_position.copy()

    while time.time() - start_time < budget_s and num_iterations_without_improvement < max_no_improvement:
        for i in range(num_particles):
            current_p = particles[i]
            current_v = velocities[i]
            personal_best = pbest_positions[i]

            r1, r2 = np.random.rand(), np.random.rand()
            cognitive_vec = cognitive * r1 * (personal_best - current_p)
            social_vec = social * r2 * (gbest_position - current_p)
            new_v = inertia * current_v + cognitive_vec + social_vec
            new_v = np.clip(new_v, -velocity_max, velocity_max)
            new_p = current_p + new_v
            new_p = np.clip(new_p, 0.0, 1.0)
            s = np.sum(new_p)
            if s != 0:
                new_p *= target_sum / s
            else:
                new_p[:] = 0.5
            velocities[i] = new_v
            particles[i] = new_p

        # Evaluate new particles
        new_c5 = [compute_c5(p, dx) for p in particles]
        new_c5 = np.array(new_c5)
        for i in range(num_particles):
            if new_c5[i] < pbest_c5[i]:
                pbest_positions[i] = particles[i].copy()
                pbest_c5[i] = new_c5[i]
        current_gbest_idx = np.argmin(pbest_c5)
        current_gbest_c5 = pbest_c5[current_gbest_idx]
        if current_gbest_c5 < gbest_c5:
            gbest_c5 = current_gbest_c5
            gbest_position = pbest_positions[current_gbest_idx].copy()
            num_iterations_without_improvement = 0
        else:
            num_iterations_without_improvement += 1

    h_full = gbest_position.copy()
    c5_bound = gbest_c5

    # Validate constraints
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1 + 1e-6)), "Bounds violated"

    return (h_full.tolist(), c5_bound, n_points)
