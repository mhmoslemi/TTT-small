        h_half = np.clip(h_half_var, 0, 1)
        h_half = normalize_h_half(h_half)
        h_full = create_symmetric_h_full(h_half)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(half_n)]
    res = minimize(objective, best_h_half, method='SLSQP', bounds=bounds, options={'maxiter': 100, 'disp': False})
    if res.success:
        best_h_half = np.clip(res.x, 0, 1)
        best_h_half = normalize_h_half(best_h_half)

    h_full = create_symmetric_h_full(best_h_half)
    corr_values = np.correlate(h_full, 1.0 - h_full, mode='full')
    c5_bound = np.max(corr_values) * dx
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), c5_bound, n_points)
