        combined_pop = np.vstack([population, new_offspring_array])
        indices = np.argsort([evaluate_candidate(cp) for cp in combined_pop])[:population_size]
        population = combined_pop[indices]

        top_count = max(1, int(population_size * 0.1))
        for i in range(top_count):
            def objective_sqp(x_flat):
                x = x_flat.reshape(-1)
                x_constr = np.clip(x, 0.0, 1.0)
                x_constr = normalize_h_half(x_constr)
                h_full = create_symmetric_h_full(x_constr)
                corr = np.correlate(h_full, 1 - h_full, mode='full')
                return np.max(corr) * dx

            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
            x0 = population[i].copy()
            x0 = np.clip(x0, 0.0, 1.0)
            x0_sum = np.sum(x0)
            if abs(x0_sum - target_sum_half) > 1e-6 and x0_sum != 0:
                x0 = x0 * (target_sum_half / x0_sum)
                x0 = np.clip(x0, 0.0, 1.0)
            result = scipy.optimize.minimize(objective_sqp, x0, method='SLSQP', 
                                             bounds=[(0.0, 1.0)] * half_n,
                                             constraints=[cons], 
                                             options={'maxiter': 10, 'ftol': 1e-6})
            if result.success:
                refined = result.x
                if evaluate_candidate(refined) < best_c5:
                    best_c5 = evaluate_candidate(refined)
                    best_h_half = refined.copy()
                population[i] = refined

    h_full = create_symmetric_h_full(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), best_c5, n_points)
