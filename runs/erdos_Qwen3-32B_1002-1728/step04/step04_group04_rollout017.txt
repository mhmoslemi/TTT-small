        perturbed = np.clip(perturbed, 0, 1)
        
        # Tune optimization parameters based on time remaining
        maxiter = min(int(120 * (time_left / budget_s)), 200)
        ftol = 1e-8 if time_left > 200 else 1e-6
        
        result = minimize(
            fun=objective,
            x0=perturbed,
            method='SLSQP',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol, 'disp': False}
        )
        
        # Constraint recovery
        if result.success:
            h_new = result.x.copy()
            h_new = np.clip(h_new, 0, 1)
            total_new = np.sum(h_new)
            
            if abs(total_new - target_sum) > 1e-8:
                h_new = (h_new / total_new) * target_sum
                h_new = np.clip(h_new, 0, 1)
                
            c5_new = objective(h_new)
            
            if np.all((h_new >= -1e-6) & (h_new <= 1.0 + 1e-6)) and abs(np.sum(h_new) - target_sum) < 1e-6:
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    patience = 0  # Reset patience for improvement
                elif patience < max_patience:
                    patience += 1
        elif abs(np.sum(perturbed) - target_sum) < 1e-6:
            c5_new = objective(perturbed)
            if c5_new < best_c5:
                best_h = perturbed
                best_c5 = c5_new
        
        # Adjust perturbation scale based on lack of progress
        if patience >= min_patience_threshold and perturbation_scale < 0.5 * target_sum:
            perturbation_scale *= 1.5  # Kick out of local optima
            patience = 0
    
    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Out of bounds"
    return (h_full.tolist(), best_c5, n_points)
