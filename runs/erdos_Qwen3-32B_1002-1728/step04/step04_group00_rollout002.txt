import numpy as np
from scipy.optimize import minimize
import time

def upscale(lower_half, n_low_half, n_high_half):
    lower_x = np.linspace(0, 1, n_low_half)
    higher_x = np.linspace(0, 1, n_high_half)
    return np.interp(higher_x, lower_x, lower_half)

def create_full(h_vars, n_points):
    half_n = n_points // 2
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
    return h_full

def run(seed=42, budget_s=1100, **kwargs):

    # Step 1: Coarse grid optimization
    n_low = 30
    half_low = n_low // 2
    dx_low = 2.0 / n_low
    target_sum_low = n_low / 4.0
    initial_lower_half = h_full_initial[0:half_low]  # from parent constructor's first half

    def objective_low(h_vars):
        h_full = create_full(h_vars, n_low)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx_low
        return np.max(corr)

    bounds_per_var_low = [(0.0, 1.0) for _ in range(half_low)]
    cons_low = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_low})

    # Optimize lower grid for 60s or until time runs out
    h_low_best_half = initial_lower_half.copy()
    h_low_best_c5 = objective_low(h_low_best_half)
