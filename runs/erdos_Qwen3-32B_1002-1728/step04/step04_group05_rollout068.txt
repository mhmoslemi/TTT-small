import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(y_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = y_half[i]
            h_full[n_points - 1 - i] = y_half[i]
        return h_full

    def evaluate_candidate(y_half):
        h_full = create_full_h(y_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize current and best y_half
    current_y = h_full_initial[:half_n].copy()
    s = current_y.sum()
    if s > 1e-9:
        current_y = current_y * (target_sum_half / s)
    current_y = np.clip(current_y, 0, 1)
    best_y = current_y.copy()
    best_c5 = evaluate_candidate(best_y)

    # SA Parameters
    np.random.seed(seed)
    T = 1.0  # Initial temperature
    alpha = 0.999  # Cooling factor
    perturbation_radius = 0.1  # Initial perturbation magnitude
    cooling_rate = 0.9995  # Perturbation radius decay rate
    max_no_improvement = 0
    iterations = 0

    while time.time() - start_time < budget_s - 10:
        # Generate random perturbation
        y_new = current_y + np.random.normal(0, perturbation_radius, size=half_n)
        y_new = np.clip(y_new, 0, 1)
        s = y_new.sum()
        if s > 1e-9:
            y_new = y_new * (target_sum_half / s)
        else:
            y_new = current_y.copy()

        # Evaluate new candidate
        new_c5 = evaluate_candidate(y_new)
        delta = new_c5 - best_c5

        # Accept or reject the new solution
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_y = y_new
            if new_c5 < best_c5:
                best_y = y_new.copy()
                best_c5 = new_c5
                max_no_improvement = 0
            else:
                max_no_improvement = 0
        else:
            max_no_improvement += 1

        # Adaptive cooling
        if max_no_improvement >= 200:
            T = max(T * alpha**2, 1e-6)
        else:
            T = max(T * alpha, 1e-6)
        perturbation_radius = max(perturbation_radius * cooling_rate, 1e-4)
        iterations += 1

    h_full = create_full_h(best_y)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
