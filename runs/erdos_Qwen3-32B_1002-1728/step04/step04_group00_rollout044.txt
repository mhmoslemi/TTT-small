import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Initialize symmetric h
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = (h_initial[i] + h_initial[n_points - 1 - i]) / 2
    h_full = np.clip(h_full, 0, 1)
    current_sum = np.sum(h_full)
    if current_sum != 0:
        h_full = h_full * (target_sum_half * 2 / current_sum)
    h_full = np.clip(h_full, 0, 1)
    best_h = h_full.copy()
    best_c5 = compute_c5(best_h, dx)

    np.random.seed(seed)
    T = 1.0  # initial temperature
    alpha = 0.999  # cooling rate
    T_min = 1e-5

    while time.time() - start_time < budget_s - 20:
        # Perturb best_h's first half
        h_half = best_h[:half_n].copy()
        noise = np.random.normal(0, T, half_n)
        perturbed = h_half + noise
        perturbed = np.clip(perturbed, 0, 1)
        current_sum = np.sum(perturbed)
        if current_sum == 0:
            perturbed = np.full_like(perturbed, target_sum_half / half_n)
        else:
            perturbed = perturbed * (target_sum_half / current_sum)
        perturbed = np.clip(perturbed, 0, 1)

        # Build full h from perturbed half
        h_full_perturbed = np.zeros(n_points)
        for i in range(half_n):
            h_full_perturbed[i] = h_full_perturbed[n_points - 1 - i] = perturbed[i]
        new_c5 = compute_c5(h_full_perturbed, dx)
        delta = new_c5 - best_c5

        # Acceptance criterion
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            best_c5 = new_c5
            best_h[:] = h_full_perturbed

        T *= alpha
        if T < T_min:
            T = T_min

    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6
