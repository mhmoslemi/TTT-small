import numpy as np
import time
from scipy.optimize import minimize

def evaluate_C5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize initial h to meet sum constraint
    current_sum = np.sum(h_initial)
    if current_sum != 0:
        h_scaled = h_initial * (target_sum / current_sum)
    else:
        h_scaled = 0.5 * np.ones(n_points)
    h_scaled = np.clip(h_scaled, 0, 1)
    current_sum_clipped = np.sum(h_scaled)
    if current_sum_clipped != 0:
        h_scaled = h_scaled * (target_sum / current_sum_clipped)
    h_scaled = np.clip(h_scaled, 0, 1)
    h_current = h_scaled.copy()
    best_h = h_current.copy()
    best_c5 = evaluate_C5(h_current, dx)

    # Simulated Annealing Phase 1
    np.random.seed(seed)
    T = 100.0  # Initial temperature
    T_final = 1e-5
    alpha = 0.99  # Cooling rate
    noise_scale = 0.1

    max_sa_duration = budget_s - 60  # Reserve 60 seconds for refinement

    while time.time() - start_time < max_sa_duration:
        # Perturb h
        perturbation = np.random.normal(0, noise_scale, n_points)
        h_candidate = h_current + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        scalar = target_sum / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate * scalar, 0, 1)

        # Evaluate objectives
        current_c5 = evaluate_C5(h_current, dx)
        candidate_c5 = evaluate_C5(h_candidate, dx)

        delta = candidate_c5 - current_c5
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-8)):
            h_current = h_candidate
            current_c5 = candidate_c5

        if candidate_c5 < best_c5:
            best_h = h_candidate.copy()
            best_c5 = candidate_c5

        # Cool down
        T *= alpha

    # Local Refinement Phase 2
    def objective_flat(x):
        h = x.reshape(n_points)
        return evaluate_C5(h, dx)

    def constraint_sum(x):
        return np.sum(x) - target_sum

    constraints = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1) for _ in range(n_points)]

    refine_time_remain = budget_s - (time.time() - start_time)
    if refine_time_remain > 0:
        result = minimize(
            fun=objective_flat,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': int(refine_time_remain * 20), 'ftol': 1e-8}
        )
        if result.success:
            h_refined = result.x
            refined_c5 = evaluate_C5(h_refined, dx)
            if refined_c5 < best_c5:
                best_h = h_refined
                best_c5 = refined_c5

    # Constraint validation
    final_sum = np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / final_sum) if final_sum != 0 else best_h

    return (best_h.tolist(), best_c5, n_points)
