            full_h[n_points - 1 - i] = left_half[i]

        current_sum = sum(left_half)

        initial_guess = (initial_guess / initial_sum) * target_sum
    initial_guess = np.clip(initial_guess, 0, 1)

    best_left = initial_guess.copy()
    best_c5 = objective_de(best_left)

    # Run Differential Evolution
    max_de_time = 600.0
    de_start = time.time()
    while time.time() - start_time < budget_s - 200:
        try:
            res_de = differential_evolution(
                func=objective_de,
                bounds=bounds,
                popsize=20,
                mutation=(0.5, 1.0),
                recombination=0.9,
                strategy='best1bin',
                polish=False,
                updating='deferred',
                workers=1,
                disp=False,
                maxiter=50
            )
            if res_de.success:
                candidate_left = res_de.x
                candidate_left = np.clip(candidate_left * (target_sum / np.sum(candidate_left)), 0, 1)
                h_full = create_full_h(candidate_left)
                corr = np.correlate(h_full, 1 - h_full, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_left = candidate_left
                    best_c5 = new_c5
        except:
            pass
        if time.time() - de_start > max_de_time:
            break

    # Local refinement with SLSQP
    def objective_local(x):
        h_full = create_full_h(x)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    cons_local = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds_local = bounds

    if best_c5 < np.inf:
        try:
            result_local = minimize(
                fun=objective_local,
                x0=best_left,
                method='SLSQP',
                bounds=bounds_local,
                constraints=cons_local,
                options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
            )
            if result_local.success:
                refined_left = np.clip(result_local.x, 0, 1)
                full_h = create_full_h(refined_left)
                if abs(np.sum(full_h) - n_points / 2) < 1e-6:
                    corr = np.correlate(full_h, 1 - full_h, mode='full')
                    refined_c5 = np.max(corr) * dx
                    if refined_c5 < best_c5:
                        best_left = refined_left
                        best_c5 = refined_c5
        except:
            pass

    h_full = create_full_h(best_left)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2) < 1e-6, "Sum constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    final_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx

    return (h_full.tolist(), final_c5, n_points)
