def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)

    while time.time() - start_time < budget_s - 30:
        # Perturbation parameters
        num_perturb = max(1, int(0.05 * n_points))

        delta = np.random.uniform(-0.2, 0.2, num_perturb)
