        # Generate perturbed candidate
        perturbed = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        if np.sum(perturbed) != 0:
            perturbed = perturbed * (target_sum / np.sum(perturbed))
        # Try optimization
        methods = ['SLSQP', 'L-BFGS-B', 'TNC']
        for method in methods:
            if time.time() - start_time >= budget_s - 30:
                break
            result = minimize(
                fun=objective_fft,
                x0=perturbed,
                method=method,
                bounds=bounds_per_var,
                constraints=cons if method != 'L-BFGS-B' else (),
                options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                new_h = result.x.copy()
                # Final constraint enforcement
                if not np.isclose(np.sum(new_h), 0):
                    new_h = new_h * (target_sum / np.sum(new_h))
                new_h = np.clip(new_h, 0, 1)
                if np.all(new_h >= -1e-6) and np.all(new_h <= 1.0 + 1e-6):
                    new_c5 = objective_fft(new_h)
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                        break

    # Final validation
    assert np.isclose(np.sum(best_h) - target_sum, 0, atol=1e-6), "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), best_c5, n_points)
