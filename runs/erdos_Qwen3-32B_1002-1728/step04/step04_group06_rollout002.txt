import numpy as np
import time
import cvxpy as cp
from numpy.random import default_rng

def create_symmetric_h_full(h_half):
    n_points = len(h_half) * 2
    h_full = np.zeros(n_points)
    h_full[:len(h_half)] = h_half.copy()
    h_full[-len(h_half):] = h_half[::-1].copy()
    return h_full

def normalize_h_half(h_half, target_sum=None):
    n = len(h_half)
    half_n = n
    if target_sum is None:
        target_sum = n * 0.5  # sum of h_full is n * 1.0, so h_half sum is n * 0.5
    current_sum = np.sum(h_half)
    if current_sum == 0:
        return np.ones_like(h_half) * target_sum / half_n
    scale = target_sum / current_sum
    h_half_norm = h_half * scale
    return np.clip(h_half_norm, 0.0, 1.0)

def refine_candidate(h_half_candidate, n_points, dx):
    h_full = create_symmetric_h_full(h_half_candidate)
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    max_corr_idx = np.argmax(np.abs(corr))
    k_idx = max_corr_idx - (n_points) + 1
    k = abs(k_idx) % n_points

    h_full_qp = cp.Variable(n_points)
    sum_constr = cp.Constraint(sum(h_full_qp) == n_points / 2.0)
    bound_constr = [0 <= h_full_qp, h_full_qp <= 1.0]
    symmetry_constr = []
    half_n = n_points // 2
    for i in range(half_n):
        symmetry_constr.append(h_full_qp[i] == h_full_qp[n_points - i - 1])
    all_constr = [sum_constr] + bound_constr + symmetry_constr

    obj_expr = 0.0
    for i in range(n_points):
        j = (i + k) % n_points
        obj_expr += h_full_qp[i] * h_full_qp[j]
    obj = cp.Minimize(obj_expr)

    prob = cp.Problem(obj, all_constr)
    try:
        prob.solve(solver=cp.OSQP, max_iter=10000, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            new_h_full = h_full_qp.value.copy()

            new_h_half = new_h_full[:half_n]
