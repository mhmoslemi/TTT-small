import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def calculate_c5(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    np.random.seed(seed)
    pop_size = 100
    crossover_rate = 0.7
    mutation_rate = 0.2
    population = []

    # Initialize population with initial and perturbed versions
    h = h_initial.copy()
    total = h.sum()
    if total > 1e-10:
        h = (h / total) * target_sum
    h = np.clip(h, 0, 1)
    population.append(h)

    for _ in range(pop_size - 1):
        perturbed = h + np.random.normal(0, 0.2, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total_p = perturbed.sum()
        if abs(total_p) > 1e-10:
            perturbed = (perturbed / total_p) * target_sum
        else:
            perturbed = np.zeros(n_points)
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        c5_values = []
        for idx, candidate in enumerate(population):
            c5 = calculate_c5(candidate)
            c5_values.append((c5, idx))
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate

        # Sort population by C5 (lower is better)
        c5_values.sort()
        population = [population[idx] for c5, idx in c5_values]

        # Generate offspring
        new_population = population.copy()
        for _ in range(pop_size):
            # Tournament selection
            parent1 = population[np.random.choice([0, 1], p=[0.8, 0.2])]
            parent2 = population[np.random.choice([0, 1], p=[0.8, 0.2])]
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                cross_point = np.random.randint(1, n_points)
                child1 = np.concatenate((parent1[:cross_point], parent2[cross_point:]))
                child2 = np.concatenate((parent2[:cross_point], parent1[cross_point:]))
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    # Apply mutation
                    noise = np.random.normal(0, 0.1, n_points)
                    child += noise
                    child = np.clip(child, 0, 1)
                    total_c = child.sum()
                    if abs(total_c) > 1e-10:
                        child = (child / total_c) * target_sum
                    child = np.clip(child, 0, 1)

                # Enforce constraints once more
                total_c = child.sum()

                    child = (child / total_c) * target_sum
                    child = np.clip(child, 0, 1)

        # Truncate to maintain population size
        population = new_population[:pop_size]
