def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h0 = np.array(initial_h_values)
    n_points = len(h0)
    target_sum = n_points / 2.0

            return np.zeros_like(h_vec)

    population = create_population(h0, POP_SIZE, noise_std=0.05)
