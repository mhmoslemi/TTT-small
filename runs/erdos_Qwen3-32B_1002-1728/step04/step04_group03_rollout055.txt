            delta_sum + n1 * (1/3) + n2 * (2/3),
            1/3  # l2_guess - l1_guess
        ])
        try:
            sol = np.linalg.solve(A, b)
            new_l1, new_l2 = sol
        except np.linalg.LinAlgError:
            # Fall back to adjusting one level
            x = delta_sum / (n1 + n2)
            new_l1 = 1 / 3 + x
            new_l2 = 2 / 3 + x
            new_l1 = np.clip(new_l1, 0.0, 1.0)
            new_l2 = np.clip(new_l2, 0.0, 1.0)

    # Ensure l1 <= l2 within [0, 1]
    new_l1 = max(min(new_l1, new_l2), 0.0)
    new_l2 = min(max(new_l2, new_l1), 1.0)

    # Build adjusted h_values
    h_values = np.zeros(n_points)
    for i in range(n_points):
        if digitized[i] == 0:
            h_values[i] = l0
        elif digitized[i] == 1:
            h_values[i] = new_l1
        elif digitized[i] == 2:
            h_values[i] = new_l2
        else:
            h_values[i] = l3

    # Final validation
    assert np.all(h_values >= -1e-8) and np.all(h_values <= 1.0001), "Value bounds violated"
    total = np.sum(h_values)
    assert np.isclose(total, target_sum, atol=1e-6), "Sum constraint violated"

    # Evaluate C5 using correlation
    cross_corr = np.correlate(h_values, 1 - h_values, mode="full")
    c5_bound = np.max(cross_corr) * dx

    return (h_values.tolist(), c5_bound, n_points)
