import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    def project(x, target_sum):
        total = np.sum(x)
        if total == 0:
            x_proj = np.zeros_like(x)
        else:
            x_proj = (x / total) * target_sum
        x_proj = np.clip(x_proj, 0, 1)
        return x_proj

    def objective_de(x):
        return objective(project(x, target_sum_half))

    # Symmetric initialization
    h_half = h_full_initial[:half_n].copy()
    h_full = create_full_h(h_half)
    h_full = project(h_full, target_sum_half * 2)
    h_half = h_full[:half_n]
    best_c5 = objective(h_half)

    bounds = [(0.0, 1.0)] * half_n

    # Global exploration
    de_start_time = time.time()
    while time.time() - start_time < budget_s - 300:
        # Run differential evolution for maxiter=10
        result_de = differential_evolution(
            objective_de, 
            bounds, 
            popsize=15, 
            strategy='best1bin',
            mutation=(0.5, 1.0),
            recombination=0.8,
            tol=1e-6,
            maxiter=5,
            seed=seed,
            updating='immediate',
            polish=False,
            workers=1
        )
        if result_de.success:
            h_candidate = project(result_de.x, target_sum_half)
            c5_candidate = objective(h_candidate)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                h_half = h_candidate.copy()

    # Local refinement
    while time.time() - start_time < budget_s - 20:
        def objective_local(x):
            return objective(project(x, target_sum_half))
        
        perturbation = np.random.normal(0, 0.05, half_n)
        perturbed = np.clip(project(h_half + perturbation, target_sum_half), 0, 1)
        
        result = minimize(
            fun=objective_local,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result.success:
            new_h = project(result.x, target_sum_half)
            new_c5 = objective(new_h)
            if new_c5 < best_c5:
                h_half = new_h
                best_c5 = new_c5

    h_full = create_full_h(h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    return (h_full.tolist(), best_c5, n_points)
