            h_full[i] = h_half[i]
            h_full[n_points - 1 - i] = h_half[i]
        return h_full

    # Normalize half to target sum and [0,1]
    def normalize_half(h_half):
        h_half = np.copy(h_half)
        original_sum = np.sum(h_half)
        h_half = np.clip(h_half, 0, 1)
        new_sum = np.sum(h_half)
        if new_sum != 0:
            h_half = h_half * (target_sum_half / new_sum)
        return h_half

    # Initialize population with symmetric initial_h_values and random symmetric vectors
    population_size = 40
    population = []
    
    # Prepare symmetric and normalized initial solution
    h_half = np.zeros(half_n)
    for i in range(half_n):
        h_half[i] = (h_full_initial[i] + h_full_initial[n_points - 1 - i]) / 2
    h_half = normalize_half(h_half)
    population.append(h_half)

    seed_vals = np.random.SeedSequence(seed).generate_state(1)[0]
    rng = np.random.Generator(np.random.PCG64(seed_vals))
    for _ in range(population_size - 1):
        h_half = rng.random(half_n)
        population.append(normalize_half(h_half))

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s:
        # Evaluate all candidates
        fitness = []
        for h_half in population:
            h_full = create_full_h(h_half)
            corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
            c5 = np.max(corr)
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_half.copy()

        # Tournament selection for parents
        num_offspring = population_size // 2
        parents = []
        tournament_size = 3
        
        # Generate parent pairs for offspring
        while len(parents) < 2 * num_offspring:
            idxs = rng.choice(len(population), tournament_size, replace=False)
            winner_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            parents.append(population[winner_idx])
        
        # Crossover and mutation to create offspring
        offspring = []
        for i in range(0, 2 * num_offspring, 2):
            if i + 1 >= len(parents):
                break
            p1, p2 = parents[i], parents[i+1]
            if half_n > 1:
                cross_point = rng.integers(1, half_n)
                c1 = np.concatenate([p1[:cross_point], p2[cross_point:]])
                c2 = np.concatenate([p2[:cross_point], p1[cross_point:]])
            else:
                c1, c2 = p1.copy(), p2.copy()
            
            # Apply mutation
            for child in [c1, c2]:
                if rng.random() < 0.1:  # 10% mutation rate
                    num_mutate = max(1, int(0.05 * half_n))
                    indices = rng.choice(half_n, num_mutate, replace=False)
                    noise = rng.normal(0, 0.05, len(indices))
                    child[indices] += noise
                child = normalize_half(child)
                offspring.append(child)
        
        # Replace bottom half of population with top offspring
        combined = population + offspring
        combined_fitness = []
        for h in combined:
            h_full = create_full_h(h)
            corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
            combined_fitness.append(np.max(corr))
        
        # Select best candidates
        indices = np.argsort(combined_fitness)[:population_size]
        population = [combined[i] for i in indices]
    
    # Final check of constraints
    h_full = create_full_h(best_h)
    assert abs(np.sum(h_full) - n_points/2) < 1e-6, "Integral constraint violated"
    assert np.all((-1e-6 <= h_full) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
