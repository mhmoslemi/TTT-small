    for _ in range(m):
        perturbation = np.random.normal(0, 0.05, n_points)
        h = h_initial + perturbation
        h = repair(h, target_sum)
        particles.append(h)
        velocities.append(np.zeros(n_points))
        p_best.append(h.copy())
        obj = evaluate_objective(h)
        p_best_obj.append(obj)

    g_best = p_best[np.argmin(p_best_obj)]
    g_best_obj = min(p_best_obj)

    start_time = time.time()
    iteration = 0
    while (time.time() - start_time) < budget_s:
        iteration += 1
        time_elapsed = time.time() - start_time
        w = w_init * (1 - (time_elapsed / budget_s))

        for i in range(m):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_cog = c1 * r1 * (p_best[i] - particles[i])
            vel_sco = c2 * r2 * (g_best - particles[i])
            velocities[i] = w * velocities[i] + vel_cog + vel_sco
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            particles[i] += velocities[i]
            particles[i] = repair(particles[i], target_sum)
            obj_new = evaluate_objective(particles[i])
            if obj_new < p_best_obj[i]:
                p_best[i] = particles[i].copy()
                p_best_obj[i] = obj_new
                if obj_new < g_best_obj:
                    g_best = particles[i].copy()
                    g_best_obj = obj_new

    best_h = g_best
    best_c5 = g_best_obj
    assert abs(np.sum(best_h) - target_sum) < 1e-6, f"Sum constraint violated: {np.sum(best_h)}"
    return (best_h.tolist(), best_c5, n_points)
