import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0
    dx = 2.0 / n_points

    # Symmetrize and normalize the initial half
    h_full = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full[i] = h_full[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = h_full.sum()
    if abs(current_sum - n_points / 2.0) > 1e-6 and current_sum != 0:
        h_full = h_full * (n_points / 2.0 / current_sum)
    h_full = np.clip(h_full, 0, 1)
    initial_h_half = h_full[:half_n]

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_h_half(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode="full")
        return np.max(corr) * dx

    # PSO parameters
    SWARM_SIZE = 50
    inertia_start = 0.9
    inertia_end = 0.4
    COGNITIVE = 2.0
    SOCIAL = 2.0
    MAX_VELOCITY = 0.1

    np.random.seed(seed)
    # Initialize particles
    positions = [initial_h_half.copy() for _ in range(SWARM_SIZE)]
    velocities = [np.random.rand(half_n) * 0.01 for _ in range(SWARM_SIZE)]

    for i in range(SWARM_SIZE):
        perturb = np.random.randn(half_n) * 0.05
        positions[i] = np.clip(positions[i] + perturb, 0, 1)
        s = positions[i].sum()
        if s != 0:
            positions[i] = (positions[i] / s) * target_sum

    # Evaluate initial personal bests
    personal_best_positions = [p.copy() for p in positions]
    personal_best_scores = [evaluate_h_half(p) for p in positions]
    current_global_idx = np.argmin(personal_best_scores)
    global_best_position = personal_best_positions[current_global_idx].copy()

    while time.time() - start_time < budget_s - 10:
