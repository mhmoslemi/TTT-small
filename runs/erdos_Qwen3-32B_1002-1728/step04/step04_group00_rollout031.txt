import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    # Create initial h_half from initial_h_values
    h_half_initial = np.zeros(half_n)
    for i in range(half_n):
        h_half_initial[i] = (h_full_initial[i] + h_full_initial[n_points - 1 - i]) / 2
    current_sum = np.sum(h_half_initial)
    if current_sum != 0:
        scale = target_sum_half / current_sum
        h_half_initial *= scale
    h_half_initial = np.clip(h_half_initial, 0, 1)

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    np.random.seed(seed)
    n_particles = 100
    particles = [h_half_initial.copy()]
    velocities = [np.zeros(half_n)]
    for _ in range(1, n_particles):
        h = np.random.rand(half_n)
        total = h.sum()
        if total != 0:
            h = h * target_sum_half / total
        h = np.clip(h, 0, 1)
        particles.append(h)
        velocities.append(np.random.rand(half_n) * 0.1)

    pbest = [p.copy() for p in particles]
    pbest_c5 = np.zeros(n_particles)
    for i in range(n_particles):
        h_full = create_full_h(particles[i])
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        pbest_c5[i] = np.max(corr) * dx
        particles[i] = pbest[i] if pbest_c5[i] < np.inf else particles[i]

    gbest_idx = np.argmin(pbest_c5)
    best_c5 = pbest_c5[gbest_idx]
    gbest_pos = pbest[gbest_idx].copy()

    c1, c2 = 2.0, 2.0

        for i in range(n_particles):

            new_pos = particles[i] + velocities[i]

            total = new_pos.sum()
            if total != 0:
                new_pos = (new_pos / total) * target_sum_half
            new_pos = np.clip(new_pos, 0, 1)
            particles[i] = new_pos

            h_full = create_full_h(new_pos)
            corr = np.correlate(h_full, 1 - h_full, mode='full')

            if current_c5 < pbest_c5[i]:
