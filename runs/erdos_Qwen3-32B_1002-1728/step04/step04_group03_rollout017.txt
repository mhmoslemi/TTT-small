import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_quaternary():
        base = np.array([0, 3, 3, 0, 1, 2, 2, 1])
        pattern_repeats = n_points // len(base)
        remaining = n_points % len(base)
        quaternary = np.tile(base, pattern_repeats + 1)
        quaternary = quaternary[:n_points]
        h = quaternary / 3.0
        return h

    h_initial = create_quaternary()
    np.clip(h_initial, 0.0, 1.0, out=h_initial)

    def objective_fft(h_vars):
        h_full = np.array(h_vars, dtype=float)
        n = len(h_full)
        h_fft = np.fft.fft(h_full, n=2 * n)
        s_fft = np.fft.fft(1.0 - h_full, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    best_h = np.copy(h_initial)
    best_c5 = objective_fft(best_h)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        rng = np.random.RandomState(seed + n_attempt)
        perturbed = best_h + rng.uniform(-0.03, 0.03, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if not np.isclose(total, 0):
            perturbed = perturbed * (target_sum / total)
        perturbed = np.clip(perturbed, 0, 1)

        methods = ['L-BFGS-B', 'TNC', 'SLSQP']
        for method in methods:
            if time.time() - start_time >= budget_s - 30:
                break
            result = minimize(
                fun=objective_fft,
                x0=perturbed,
                method=method,
                bounds=bounds_per_var,
                constraints=constraints if method != 'L-BFGS-B' else (),
                options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                new_h = result.x.reshape(n_points)
                new_h = np.clip(new_h, 0, 1)
                scaled_sum = np.sum(new_h)
                if not np.isclose(scaled_sum, 0) and np.isclose(scaled_sum, target_sum, atol=1e-4):
                    new_c5 = result.fun
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                        break

    h_full = best_h
    assert np.isclose(np.sum(h_full), target_sum, atol=1e-6), "Sum constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Bounds violation"
    
    return (h_full.tolist(), best_c5, n_points)
