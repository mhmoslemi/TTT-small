def run(seed=42, budget_s=1100, **kwargs):

    for _ in range(pop_size):

        init_pop.append(h_perturbed)
    population = np.array(init_pop)

        current_min_idx = np.argmin(c5_list)

            best_c5 = c5_list[current_min_idx]
