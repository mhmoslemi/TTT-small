import numpy as np
import time

def create_full_h(y, n_points):
    h = np.zeros(n_points)
    half_n = n_points // 2
    for i in range(half_n):
        h[i] = y[i]
        h[n_points - 1 - i] = y[i]
    return h

def evaluate_candidate(y, n_points):
    h = create_full_h(y, n_points)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
