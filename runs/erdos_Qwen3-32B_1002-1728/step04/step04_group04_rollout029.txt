import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_full = h_vars.copy()
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # GA parameters
    np.random.seed(seed)
    pop_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    max_gens = 500

    # Initialize population
    population = []
    h_initial = h_full_initial.copy()
    h_initial_sum = h_initial.sum()
    if abs(h_initial_sum) > 1e-10:
        h_initial = (h_initial / h_initial_sum) * target_sum
    h_initial = np.clip(h_initial, 0, 1)
    population.append(h_initial)

    for _ in range(pop_size - 1):
        h_perturbed = h_full_initial.copy() + np.random.normal(0, 0.1, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_total = h_perturbed.sum()
        if abs(h_total) > 1e-10:
            h_perturbed = (h_perturbed / h_total) * target_sum
        h_perturbed = np.clip(h_perturbed, 0, 1)
        population.append(h_perturbed)

    population = np.array(population)
    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 20:
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        idx = np.argmin(fitness)
        if fitness[idx] < best_c5:
            best_c5 = fitness[idx]
            best_h = population[idx].copy()

        # Selection and crossover
        new_population = [population[idx].copy()]  # Elitism

        while len(new_population) < pop_size:
            # Tournament selection
            parent_indices = np.random.randint(0, len(population), 3)
            parent1 = population[np.argmin(fitness[parent_indices])]
            parent_indices = np.random.randint(0, len(population), 3)
            parent2 = population[np.argmin(fitness[parent_indices])]

            # Crossover
            if np.random.rand() < crossover_rate:
                cross_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
                child2 = np.concatenate([parent2[:cross_point], parent1[cross_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            # Mutation & Normalize
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    child += np.random.normal(0, 0.1, n_points)
                child = np.clip(child, 0, 1)
                total = child.sum()
                if abs(total) > 1e-10:
                    child = (child / total) * target_sum
                child = np.clip(child, 0, 1)
                new_population.append(child)
                if len(new_population) >= pop_size:
                    break

        population = np.array(new_population[:pop_size])

    # Final check for constraints
    assert best_h is not None
    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Out of bounds"
    return (h_full.tolist(), best_c5, n_points)
