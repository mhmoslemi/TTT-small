    target_sum_half = n_points / 4.0  # 2 * sum(h_half) = n_points / 2

    def make_full_h(h_half):
        return np.concatenate([h_half, h_half[::-1]])

    def evaluate(h_half):
        h_full = make_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    population_size = 40
    rng = np.random.default_rng(seed)
    
    # Symmetrize and normalize initial guess
    h_half_initial = (h_full_initial[:half_n] + h_full_initial[-half_n:][::-1]) / 2
    h_half_initial /= h_half_initial.sum() * 2 / n_points
    h_half_initial = np.clip(h_half_initial, 0, 1)
    
    # Initialize population with symmetric variants
    population = [h_half_initial]
    for _ in range(population_size - 1):
        h_half = rng.uniform(0, 1, size=half_n)
        h_half = np.clip(h_half, 0, 1)
        h_half *= target_sum_half / h_half.sum()
        population.append(h_half)

    best_c5 = np.inf
    best_h_half = None
    generation = 0

    while time.time() - start_time < budget_s - 10:
        # Evaluate population
        fitness = [evaluate(ph) for ph in population]
        current_idx = np.argmin(fitness)
        if fitness[current_idx] < best_c5:
            best_c5 = fitness[current_idx]
            best_h_half = population[current_idx]

        # Selection (tournament)
        def tournament():
            idxs = rng.integers(0, population_size, 3)
            return min(idxs, key=lambda i: fitness[i])

        parents_idx = [tournament() for _ in range(population_size)]
        parents = [population[i] for i in parents_idx]
        parents_fitness = [fitness[i] for i in parents_idx]

        # Crossover and mutation
        offspring = []
        for i in range(0, len(parents), 2):
            p1 = parents[i]
            p2 = parents[i+1] if i+1 < len(parents) else p1

            # One-point crossover
            if half_n > 1:
                cx_point = rng.integers(1, half_n)
                child = np.concatenate([p1[:cx_point], p2[cx_point:]])
            else:
                child = p1.copy()

            # Mutation
            if rng.random() < 0.1:
                mut_points = rng.integers(0, half_n, size=int(0.05*half_n))
                child[mut_points] += rng.normal(0, 0.05, mut_points.shape)
                child = np.clip(child, 0, 1)
                child_sum = child.sum()
                if child_sum != 0:
                    child *= target_sum_half / child_sum
                else:
                    child[:] = target_sum_half / half_n
                child = np.clip(child, 0, 1)

            # Add to offspring
            offspring.append(child)

        # Elitism (top 5%)
        elite_size = max(1, int(0.05 * population_size))
        elites = [population[i] for i in np.argsort(fitness)[:elite_size]]

        # Replace worst 50%
        pop_and_offspring = population + offspring
        combined = [(h, evaluate(h)) for h in pop_and_offspring]
        combined.sort(key=lambda x: x[1])
        new_population = [h for h, _ in combined[:population_size]]

        # Ensure elites are included
        population = elites + new_population
        population = population[:population_size]

        # Progress tracking
        generation += 1
        if generation % 20 == 0:
            pass  # Silent progress tracking

    h_full = make_full_h(best_h_half)
    return (h_full.tolist(), best_c5, n_points)
