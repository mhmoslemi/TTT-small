    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_sum = np.sum(h_full_initial)
    if h_sum != 0:
        h_full_initial *= (target_sum_half * 2.0 / h_sum)
    h_full_initial = np.clip(h_full_initial, 0, 1)
    h_half_initial = h_full_initial[:half_n]

    pop_size = 50
    np.random.seed(seed)
    population = [h_half_initial.copy()]
    for i in range(1, pop_size):
        noise = np.random.normal(0.0, 0.1, half_n)
        new_h = h_half_initial.copy() + noise
        new_h = np.clip(new_h, 0, 1)
        sum_new = np.sum(new_h)
        if sum_new != 0:
            new_h = new_h * (target_sum_half / sum_new)
        new_h = np.clip(new_h, 0, 1)
        population.append(new_h)

    best_c5 = float('inf')
    best_h = h_half_initial.copy()

    # GA loop
    max_generations = 0
    while time.time() - start_time < budget_s - 20:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()

        # Select top 50%
        sorted_indices = np.argsort(fitness)
        top_k = [population[i] for i in sorted_indices[:pop_size // 2]]

        # Generate new population
        new_population = []
        for _ in range(pop_size):
            parent1 = top_k[np.random.randint(len(top_k))]
            parent2 = top_k[np.random.randint(len(top_k))]
            child = (parent1 + parent2) / 2.0

            # Mutation
            if np.random.rand() < 0.2:
                noise = np.random.normal(0.0, 0.3 * (1 - (time.time() - start_time) / (budget_s - 20)) ** 1.5, half_n)
                child += noise

            # Normalize and clip
            child = np.clip(child, 0, 1)
            sum_child = np.sum(child)
            if sum_child != 0:
                child = (target_sum_half / sum_child) * child
            child = np.clip(child, 0, 1)
            new_population.append(child)

        population = new_population

    # Final validation
    final_h_full = create_full_h(best_h)
    sum_h = np.sum(final_h_full)
    if abs(sum_h - n_points / 2.0) > 1e-6:
        scale = (n_points / 2.0) / sum_h
        final_h_full *= scale
        final_h_full = np.clip(final_h_full, 0, 1)

    final_c5 = compute_c5(best_h)
    return (final_h_full.tolist(), final_c5, n_points)
