            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = build_h(x_opt)
            
            # Find cluster to split (largest segment)
            max_len = -1
            split_idx = 0
            for i, (s, e, _) in enumerate(clusters):
                l = e - s
                if l > max_len:
                    max_len = l
                    split_idx = i
            
            # Split cluster
            if max_len > 1:
                s, e, v = clusters[split_idx]
                mid = (s + e) // 2
                clusters[split_idx:e] = [(s, mid, v), (mid, e, v)]
    
    # Validate final solution
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (best_h.tolist(), float(best_c5), n_points)
