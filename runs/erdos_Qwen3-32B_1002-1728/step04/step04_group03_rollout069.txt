    h_full = create_full_h(best_h_half)
    assert np.isclose(np.sum(h_full), target_sum_full, atol=1e-6), "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
