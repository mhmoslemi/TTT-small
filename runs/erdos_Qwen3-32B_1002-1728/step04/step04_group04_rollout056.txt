import time

    h_initial = np.clip(h_initial, 0, 1)

    else:
        h_initial = np.zeros_like(h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    best_c5 = evaluate_c5(h_initial)
    best_h = h_initial.copy()

    # Bayesian Optimization setup
    X = [h_initial.copy()]
    y = [best_c5]
    X = np.array(X)
    y = np.array(y).reshape(-1, 1)
    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e1))
    gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=1, alpha=1e-6)

    np.random.seed(seed)

        time_left = budget_s - (time.time() - start_time)
