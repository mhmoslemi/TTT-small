import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def evaluate_candidate(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Symmetrize and normalize initial h
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j])/2
    total_sum = np.sum(h_full_sym)
    if total_sum != 0:
        h_full_sym *= (n_points / 2.0) / total_sum
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n].copy()

    # PSO Parameters
    SWARM_SIZE = 50
    MAX_VELOCITY = 0.1
    COGNITIVE = 2.0
    SOCIAL = 2.0

    np.random.seed(seed)
    # Initialize swarm
    positions = [initial_h_half.copy() + np.random.normal(0, 0.02, half_n) for _ in range(SWARM_SIZE)]
    velocities = [np.zeros(half_n) for _ in range(SWARM_SIZE)]
    
    # Enforce constraints
    for i in range(SWARM_SIZE):
        positions[i] = np.clip(positions[i], 0, 1)
        s = np.sum(positions[i])
        if s != 0:
            positions[i] *= target_sum / s
        else:
            positions[i].fill(target_sum / (half_n))

    # Evaluate initial fitness
    fitness = [evaluate_candidate(p) for p in positions]
    personal_best = positions.copy()
    personal_best_fitness = fitness.copy()
    global_best = personal_best[np.argmin(personal_best_fitness)].copy()
    best_c5 = np.min(personal_best_fitness)

    while time.time() - start_time < budget_s - 1:
        time_used = time.time() - start_time
        phi = min(1.0, time_used / budget_s)
        inertia = max(0.4, 0.9 - 0.5 * phi)

        for i in range(SWARM_SIZE):
            rand1 = np.random.rand()
            rand2 = np.random.rand()
            velocities[i] = (
                inertia * velocities[i] +
                COGNITIVE * rand1 * (personal_best[i] - positions[i]) +
                SOCIAL * rand2 * (global_best - positions[i])
            )
            velocities[i] = np.clip(velocities[i], -MAX_VELOCITY, MAX_VELOCITY)
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            s = np.sum(positions[i])
            if s != 0:
                positions[i] *= target_sum / s
            else:
                positions[i].fill(target_sum / half_n)

        # Evaluate and update bests
        for i in range(SWARM_SIZE):
            current_c5 = evaluate_candidate(positions[i])
            if current_c5 < personal_best_fitness[i]:
                personal_best[i] = positions[i].copy()
                personal_best_fitness[i] = current_c5
                if current_c5 < best_c5:
                    global_best = positions[i].copy()
                    best_c5 = current_c5

    h_full = create_full_h(global_best)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0+1e-6))
    assert abs(np.sum(h_full) - n_points/2.0) < 1e-6
    
    return (h_full.tolist(), best_c5, n_points)
