import time

def run(seed=42, budget_s=1100, **kwargs):

    target_sum_half = n_points / 4.0

        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def evaluate_candidate(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initial symmetric conversion and normalization
    h_full_sym = h_full_initial.copy()
