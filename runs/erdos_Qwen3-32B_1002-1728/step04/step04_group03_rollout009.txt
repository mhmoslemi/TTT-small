        cross_corr = np.fft.ifft(h_fft * s_fft).real
        max_corr = np.max(np.abs(cross_corr))

    def constraint_sum(h):

    result = minimize(
        fun=objective,
        x0=h_quan,
        method='SLSQP',  # Allows for constraints

        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}

    h_opt = result.x.reshape(n_points)
    h_opt = np.clip(h_opt, 0.0, 1.0)

    assert np.all(h_opt >= -1e-6) and np.all(h_opt <= 1.0 + 1e-6), "Bounds violated"
