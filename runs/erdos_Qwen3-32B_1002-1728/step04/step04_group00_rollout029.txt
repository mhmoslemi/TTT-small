def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = (n_points / 2.0) / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_initial = h_full_sym[:half_n]
    
    n_particles = 100
    c1, c2 = 2.0, 2.0
    w_max, w_min = 0.9, 0.4
    velocity_clamp = 0.5

        pos_clipped = np.clip(pos, 0, 1)
        pos_sum = np.sum(pos_clipped)

        return np.clip(pos_clipped, 0, 1)
    
    for i in range(n_particles):
        pos = h_half_initial + np.random.uniform(-0.1, 0.1, half_n)
        pos = fix_position(pos)
        positions[i] = pos

    for i in range(n_particles):
        h_full = np.zeros(n_points)
