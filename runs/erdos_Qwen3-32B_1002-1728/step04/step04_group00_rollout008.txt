import numpy as np
import time

def create_full_h(h_vars, n_points):
    half_n = len(h_vars)
    h_full = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        h_full[i] = h_full[j] = h_vars[i]
    return h_full

def compute_energy(h_half, n_points, dx):
    h_full = create_full_h(h_half, n_points)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Create symmetric initial h_half
    h_full = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full[i] + h_full[j]) / 2
        h_full[i] = h_full[j] = avg
    current_sum = h_full.sum()
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = (target_sum_half * 2.0) / current_sum
        h_full *= scale
    h_full = np.clip(h_full, 0, 1)
    h_half = h_full[:half_n]

    # Ensure sum constraint
    s = h_half.sum()
    if s != 0:
        scale = target_sum_half / s
        h_half *= scale
        h_half = np.clip(h_half, 0, 1)
    best_h_half = h_half.copy()
    best_energy = compute_energy(best_h_half, n_points, dx)

    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.9999
    T_min = 1e-5
    current_T = T0
    np.random.seed(seed)

    while time.time() - start_time < budget_s - 60:
        # Perturb with noise scaled by current_T
        perturbed = best_h_half + np.random.normal(0, 0.2, half_n) * current_T
        perturbed = np.clip(perturbed, 0, 1)
        s = perturbed.sum()
        if s != 0:
            scale = target_sum_half / s
            perturbed *= scale

        new_energy = compute_energy(perturbed, n_points, dx)
        delta = new_energy - best_energy

        # Accept based on Metropolis criterion
        if delta < 0 or np.random.rand() < np.exp(-delta / current_T):

        if current_T < T_min:
