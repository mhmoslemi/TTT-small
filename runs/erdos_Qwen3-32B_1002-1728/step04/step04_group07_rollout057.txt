        perturbed = best_h_half + np.random.uniform(-delta, delta, size=half_n)
        perturbed = np.clip(perturbed, 0, 1.1)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        h_full = create_full_h(perturbed)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        new_c5 = np.max(corr) * dx

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h_half = perturbed.copy()
            T = max(T * cooling_rate, 1e-8)
            delta = min(delta * 1.05, 0.5)
        else:
            dE = best_c5 - new_c5
            if np.random.rand() < np.exp(dE / T):
                best_c5 = new_c5
                best_h_half = perturbed.copy()
                delta = max(delta * 0.95, 0.05)
            T *= cooling_rate

    def objective_for_opt(x):
        perturbed = x.copy()
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)
        h_full = create_full_h(perturbed)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    result = minimize(
        fun=objective_for_opt,
        x0=best_h_half,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * half_n,
        options={'maxiter': 150, 'ftol': 1e-12, 'disp': False}
    )

    if result.success:
        new_h = result.x
        total = np.sum(new_h)
        if total != 0:
            new_h = (new_h / total) * target_sum_half
        new_h = np.clip(new_h, 0, 1)
        h_full_new = create_full_h(new_h)
        corr = np.correlate(h_full_new, 1 - h_full_new, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h_half = new_h

    h_full_final = create_full_h(best_h_half)
    assert abs(np.sum(h_full_final) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full_final >= -1e-6) & (h_full_final <= 1.0 + 1e-6)), "Value bounds violated"
    return (h_full_final.tolist(), best_c5, n_points)
