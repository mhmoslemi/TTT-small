            gbest_fitness = fitness
            gbest_position = particles[i].copy()

    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    velocity_limit = 0.5

    while time.time() - start_time < budget_s - 30:
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest_positions[i] - particles[i]) +
                             social * r2 * (gbest_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -velocity_limit, velocity_limit)
            particles[i] = correct_position(particles[i] + velocities[i], target_sum)

        for i in range(n_particles):
            fitness = compute_c5(particles[i])
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest_positions[i] = particles[i].copy()
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_position = particles[i].copy()

    h_full = gbest_position.copy()
    assert np.allclose(h_full.sum(), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), gbest_fitness, n_points)
