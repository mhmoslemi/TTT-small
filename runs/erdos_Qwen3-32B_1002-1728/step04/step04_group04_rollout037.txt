        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 30:
            alpha *= 0.9

        # Perturbation parameters
        num_perturb = max(2, int(0.05 * n_points))
        perturb_indices = np.random.choice(n_points, size=num_perturb, replace=False)
        delta = np.random.uniform(-0.2, 0.2, size=num_perturb)
        h_candidate = current_h.copy()
        h_candidate[perturb_indices] += delta
        h_candidate = np.clip(h_candidate, 0, 1)
        current_sum_candidate = np.sum(h_candidate)
        if abs(current_sum_candidate) > 1e-10:
            h_candidate = (h_candidate / current_sum_candidate) * target_sum
        h_candidate = np.clip(h_candidate, 0, 1)

        new_c5 = compute_c5(h_candidate)
        delta_energy = new_c5 - current_c5

        if delta_energy < 0:
            # Accept improvement
            current_h = h_candidate.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # Accept based on temperature
            if np.random.rand() < np.exp(-delta_energy / T):
                current_h = h_candidate.copy()
                current_c5 = new_c5

        T *= alpha

    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Out of bounds"
    return (h_full.tolist(), best_c5, n_points)
