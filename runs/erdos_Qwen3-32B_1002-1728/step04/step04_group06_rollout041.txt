import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    rng = np.random.Generator(np.random.PCG64(seed))
    population_size = 50
    mut_factor = 0.7
    cross_rate = 0.9
    local_iter = 5

    def create_full(h_half):
        return np.concatenate([h_half, h_half[::-1]])

    def normalize(h_half):
        s = h_half.sum()
        if s < 1e-12:
            return np.ones_like(h_half) * target_sum_half / half_n
        scale = target_sum_half / s
        return np.clip(h_half * scale, 0.0, 1.0)

    def evaluate(h_half):
        h = create_full(normalize(h_half))
        corr = np.correlate(h, 1.0 - h, 'full')
        return np.max(np.abs(corr)) * dx

    # Initialize population
    pop = np.array([h_initial[:half_n] + rng.normal(0, 0.03, half_n) for _ in range(population_size)])
    for i in range(population_size):
        pop[i] = normalize(pop[i])

    best_h = pop[0].copy()
    best_c5 = evaluate(best_h)

    # Differential Evolution phase
    while time.time() - start_time < budget_s * 0.8:
        for i in range(population_size):
            idxs = list(range(population_size))
            a, b, c = pop[[*rng.choice(idxs, 3, replace=False) if _ != i else None for _ in idxs]]
            mutant = normalize(pop[i] + mut_factor * (a - b))
            cross = rng.random(half_n) < cross_rate
            trial = np.where(cross, mutant, pop[i])
            trial = normalize(trial)
            trial_c5 = evaluate(trial)
            if trial_c5 < evaluate(pop[i]):
                pop[i] = trial
                if trial_c5 < best_c5:
                    best_h = trial
                    best_c5 = trial_c5
            if time.time() - start_time >= budget_s * 0.8:
                break

    # Local optimization phase
    fit = np.array([evaluate(ind) for ind in pop])
    top = pop[np.argsort(fit)[:local_iter]]

    for cand in top:
        if time.time() - start_time >= budget_s - 10:
            break
        # Objective function for local search
        def obj(x):
            return evaluate(normalize(x))

        # Constraints and bounds
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
        bounds = [(0.0, 1.0)] * half_n

        result = scipy.optimize.minimize(
            obj, cand, method='SLSQP',
            bounds=bounds, constraints=constraints,
            options={'maxiter': 30}
        )
        if result.success:
            val = evaluate(normalize(result.x))
            if val < best_c5:
                best_h = normalize(result.x)
                best_c5 = val

    # Final validation and packaging
    h_full = create_full(normalize(best_h))
    assert abs(h_full.sum() - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    return (h_full.tolist(), best_c5, n_points)
