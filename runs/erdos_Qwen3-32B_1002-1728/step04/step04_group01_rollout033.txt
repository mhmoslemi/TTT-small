        result.append((prev_start, prev_end, prev_val))
        return result

    # Main optimization loop
    np.random.seed(seed)
    while time.time() - start_time < budget_s - 30:
        # Select a cluster to split
        if not clusters:
            break
        cluster_idx = np.random.randint(len(clusters))
        (start, end, val) = clusters[cluster_idx]
        length = end - start + 1
        if length <= 1:
            continue  # Can't split further

        # Split at midpoint
        mid = (start + end) // 2
        left_len = mid - start + 1
        right_len = end - mid

        candidates = []
        # Generate candidate (left_val, right_val) pairs
        if right_len > 0:
            for left_val in np.linspace(0.0, 1.0, 5):
                right_val = (length * val - left_val * left_len) / right_len
                if 0 <= right_val <= 1:
                    candidates.append((left_val, right_val))
        if left_len > 0:
            for right_val in np.linspace(0.0, 1.0, 5):
                left_val = (length * val - right_val * right_len) / left_len
                if 0 <= left_val <= 1:
                    candidates.append((left_val, right_val))
        candidates.append((val, val))  # No change case

        best_split_c5 = best_c5
        best_clusters = clusters
        for left_val, right_val in set(candidates):
            # Construct new clusters after split
            new_clusters = []
            inserted = False
            for i, (s, e, v) in enumerate(clusters):
                if i == cluster_idx and not inserted:
                    new_clusters.append((start, mid, left_val))
                    new_clusters.append((mid + 1, end, right_val))
                    inserted = True
                elif i != cluster_idx:
                    new_clusters.append((s, e, v))
            merged_new = merge_clusters(new_clusters)
            # Build array and normalize
            h_new = clusters_to_array(merged_new)
            total_new = np.sum(h_new)
            if abs(total_new - target_sum) > 1e-6:
                h_new = h_new * (target_sum / total_new)
            h_new = np.clip(h_new, 0.0, 1.0)
            c5_new = eval_c5(h_new)
            if c5_new < best_split_c5:
                best_split_c5 = c5_new
                best_clusters = merged_new
                if c5_new < best_c5:
                    best_h[:] = h_new[:]
                    best_c5 = c5_new
        clusters = best_clusters

    return (best_h.tolist(), best_c5, n_points)
