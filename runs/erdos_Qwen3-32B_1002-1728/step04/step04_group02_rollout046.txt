def create_full_h(h_half, n_points):
    half_n = n_points // 2
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[half_n:] = h_half[::-1]
    return h_full

def compute_c5(h_full, dx):
    n = len(h_full)
    n_conv = 2 * n - 1
    h_fft = np.fft.fft(h_full, n_conv)
    one_minus_h_fft = np.fft.fft(1.0 - h_full, n_conv)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    return np.max(corr) * dx

def normalize_h_half(h_half, target_sum):
    current_sum = np.sum(h_half)
    if current_sum == 0:
        return h_half
    return h_half * (target_sum / current_sum)
