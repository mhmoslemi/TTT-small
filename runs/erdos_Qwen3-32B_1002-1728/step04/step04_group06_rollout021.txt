import numpy as np
import time
import scipy.optimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_full = n_points / 2.0
    target_sum_half = target_sum_full / 2.0
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    h_full_initial = create_symmetric_h_full(h_initial[:half_n])
    if abs(np.sum(h_full_initial) - target_sum_full) > 1e-6:
        scale = target_sum_full / np.sum(h_full_initial)
        h_full_initial = np.clip(h_full_initial * scale, 0.0, 1.0)

    def evaluate_candidate(h_half_cand):
        h_full = create_symmetric_h_full(h_half_cand)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        corr_first = corr[:n_points]
        return np.max(corr_first) * dx

    population_size = 50
    F = 0.7
    CR = 0.9
    rng = np.random.default_rng(seed)

    population = np.zeros((population_size, half_n))
    population[0] = h_initial[:half_n].copy()
    for i in range(1, population_size):
        perturb = rng.normal(0.0, 0.1, half_n)
        candidate = h_initial[:half_n] + perturb
        population[i] = np.clip(candidate, 0.0, 1.0)
        current_sum = np.sum(population[i])
        if current_sum > 1e-6:
            population[i] *= target_sum_half / current_sum

    best_c5 = evaluate_candidate(population[0])
    best_h_half = population[0].copy()

    while time.time() - start_time < budget_s - 60:
        for idx in range(population_size):
            a, b, c = population[rng.choice(population_size, 3, replace=False)]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0.0, 1.0)
            cross_mask = rng.random(half_n) < CR
            cross_mask[rng.integers(0, half_n)] = True
            trial = np.where(cross_mask, mutant, population[idx])
            current_sum = np.sum(trial)
            trial = np.clip(trial, 0.0, 1.0)
            if current_sum > 1e-6:
                trial = trial * (target_sum_half / current_sum)
            if evaluate_candidate(trial) < evaluate_candidate(population[idx]):
                population[idx] = trial
                if evaluate_candidate(trial) < best_c5:
                    best_c5 = evaluate_candidate(trial)
                    best_h_half = trial.copy()

    def refine_h(h_half):
        h_refined = h_half.copy()
        for _ in range(3):
            h_full = create_symmetric_h_full(h_refined)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            corr_first = corr[:n_points]
            k = np.argmax(corr_first)  # worst shift
            def objective(x):
                h_half_candidate = x.copy()
                h_full_candidate = create_symmetric_h_full(h_half_candidate)
                end_idx = n_points - k
                a = h_full_candidate[:end_idx]
                b = h_full_candidate[k:k + end_idx]
                return np.sum(a * (1 - b)) * dx
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]
            min_res = scipy.optimize.minimize(
                objective, h_refined, method="SLSQP", 
                bounds=[(0, 1)] * half_n, constraints=cons, tol=1e-8
            )
            if min_res.success:
                h_refined = np.clip(min_res.x, 0, 1)
                scale = target_sum_half / np.sum(h_refined)
                h_refined *= scale
        return h_refined

    best_h_half = refine_h(best_h_half)
    h_full = create_symmetric_h_full(best_h_half)
    assert abs(np.sum(h_full) - target_sum_full) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), evaluate_candidate(best_h_half), n_points)
