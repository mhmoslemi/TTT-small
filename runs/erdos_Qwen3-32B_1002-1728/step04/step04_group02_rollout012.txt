        if time.time() - start_time > budget_s - 60:  # Reserve 60s for finalization
            break
            
        # Generate perturbed candidate
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        # Enforce integral constraint
        perturbed_sum = np.sum(perturbed_h)
        if perturbed_sum != 0:
            perturbed_h = perturbed_h * (target_sum / perturbed_sum)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Run trial
        opt_h, opt_c5 = run_trial(perturbed_h)
        if opt_c5 < best_c5:
            best_h = opt_h
            best_c5 = opt_c5
            
    # Final validation and cleanup
    h_full = best_h
    h_full = np.clip(h_full, 0, 1)
    sum_h = np.sum(h_full)
    if abs(sum_h - target_sum) > 1e-6 and sum_h != 0:
        h_full = h_full * (target_sum / sum_h)
    h_full = np.clip(h_full, 0, 1)
    final_c5 = compute_c5_fft(h_full)
    
    return (h_full.tolist(), final_c5, n_points)
