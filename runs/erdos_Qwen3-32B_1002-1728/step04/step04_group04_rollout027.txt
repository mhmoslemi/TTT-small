        if np.random.rand() < 0.1 or learning_rate < min_lr:
            noise = np.random.normal(0, perturbation, n_points)
            h = best_h + noise
            h = np.clip(h, 0.0, 1.0)
            h_sum = np.sum(h)
            if abs(h_sum) > 1e-6:
                h = h * (target_sum / h_sum)
            h = np.clip(h, 0.0, 1.0)

        # Decay rates
        learning_rate = max(learning_rate * lr_decay, min_lr)
        perturbation *= perturbation_decay

    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    h_sum = np.sum(best_h)
    if abs(h_sum - target_sum) > 1e-6:
        best_h = best_h * (target_sum / h_sum)
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h.tolist(), best_c5, n_points)
