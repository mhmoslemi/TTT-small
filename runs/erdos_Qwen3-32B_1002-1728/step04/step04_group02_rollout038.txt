import time

def run(seed=42, budget_s=1100, **kwargs):

        raise ValueError("initial_h_values must match selected n_points")

    def create_perturbation(h_base, noise_std=0.1):
        h_pert = h_base.copy() + np.random.normal(0, noise_std, size=n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert *= (target_sum / np.sum(h_pert))
        return h_pert

    population.extend([create_perturbation(h_initial) for _ in range(49)])

        # Compute overlap using FFT-based convolution
        n = len(h)
        h_fft = np.fft.fft(h, 2*n - 1)
        one_minus_h_fft = np.fft.fft(1.0 - h, 2*n - 1)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * dx

        return h
