import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum of h_half should be this

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def compute_c5(h_full):
        n = len(h_full)
        h_fft = np.fft.fft(h_full, 2 * n - 1)
        one_minus_h_fft = np.fft.fft(1.0 - h_full, 2 * n - 1)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * dx

    # Initial symmetric setup
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (n_points / 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    best_h_half = h_full_sym[0:half_n].copy()
    best_c5 = compute_c5(create_full_h(best_h_half))

    # GA parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    tournament_size = 3
    elitism_count = 5
    stagnation_count = 0
    max_stagnation = 50

    # Initialize population
    np.random.seed(seed)
    population = [best_h_half.copy()]
    for _ in range(pop_size - 1):
        new_h_half = population[0].copy() + np.random.uniform(-0.05, 0.05, half_n)
        new_h_half = np.clip(new_h_half, 0, 1)
        current_sum = np.sum(new_h_half)
        new_h_half *= target_sum_half / current_sum if current_sum != 0 else 0.0
        new_h_half = np.clip(new_h_half, 0, 1)
        population.append(new_h_half)

    # Evaluate initial population
    for ind in population:
        h_full = create_full_h(ind)
        c5 = compute_c5(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_h_half[:] = ind.copy()

    # Main GA loop
    while time.time() - start_time < budget_s - 60:
        next_population = []
        # Elites
        population_with_fitness = [(compute_c5(create_full_h(ind)), ind) for ind in population]
        population_with_fitness.sort(key=lambda x: x[0])
        elites = [pf[1].copy() for pf in population_with_fitness[:elitism_count]]
        next_population.extend(elites)

        # Generate remaining individuals
        current_pop = [pf[1] for pf in population_with_fitness]
        while len(next_population) < pop_size:
            # Tournament selection
            idxs = np.random.choice(len(current_pop), tournament_size, replace=False)
            parent1 = min([current_pop[i] for i in idxs], key=lambda x: compute_c5(create_full_h(x)))
            idxs = np.random.choice(len(current_pop), tournament_size, replace=False)
            parent2 = min([current_pop[i] for i in idxs], key=lambda x: compute_c5(create_full_h(x)))

            # Crossover
            if np.random.rand() < crossover_rate:
                cross_point = np.random.randint(1, half_n)
                child1 = np.concatenate((parent1[:cross_point], parent2[cross_point:])).copy()
                child2 = np.concatenate((parent2[:cross_point], parent1[cross_point:])).copy()
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.1, half_n)
                    child += noise
                child = np.clip(child, 0, 1)
                current_sum = np.sum(child)
                if current_sum != 0:
                    child *= target_sum_half / current_sum
                child = np.clip(child, 0, 1)
                next_population.append(child)
                if len(next_population) >= pop_size:
                    break

        population = next_population

        # Update best solution
        current_best = min(compute_c5(create_full_h(ind)) for ind in population)
        if current_best < best_c5:
            best_c5 = current_best
            best_h_half[:] = [ind for ind in population if compute_c5(create_full_h(ind)) == best_c5][0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Mutation rate adjustment
        if stagnation_count >= max_stagnation:
            mutation_rate = min(mutation_rate * 1.5, 0.3)
            stagnation_count = 0

    # Final checks
    h_full = create_full_h(best_h_half)
    final_c5 = compute_c5(h_full)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), final_c5, n_points)
