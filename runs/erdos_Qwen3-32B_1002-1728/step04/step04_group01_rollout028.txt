            new_h = create_full_h(new_h_half)
            new_h = np.clip(new_h, 0, 1)
            s = np.sum(new_h)
            if s != 0:
                new_h = new_h * (n_points / 2.0 / s)
            new_h = np.clip(new_h, 0, 1)
            c5 = compute_c5(new_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = new_h.copy()
                h_half_initial = new_h_half.copy()

    best_h = np.clip(best_h, 0, 1)
    assert abs(np.sum(best_h) - n_points / 2.0) < 1e-6
    assert np.all((best_h >= -1e-8) & (best_h <= 1.0 + 1e-8))
    return (best_h.tolist(), best_c5, n_points)
