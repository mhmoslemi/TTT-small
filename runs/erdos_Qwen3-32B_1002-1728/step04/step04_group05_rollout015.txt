import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points/2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n].copy()

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_candidate(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    current_h_half = initial_h_half.copy()
    best_h_half = current_h_half.copy()
    current_c5 = evaluate_candidate(current_h_half)
    best_c5 = current_c5

    T = 1.0
    alpha = 0.999
    r = 0.01

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 10:
        new_h_half = current_h_half + np.random.normal(0, r, half_n)
        new_h_half = np.clip(new_h_half, 0, 1)
        s = np.sum(new_h_half)
        if s == 0:
            new_h_half = target_sum_half * np.random.rand(half_n)
            s = np.sum(new_h_half)
        scale = target_sum_half / s
        new_h_half *= scale
        new_h_half = np.clip(new_h_half, 0, 1)

        new_c5 = evaluate_candidate(new_h_half)
        if np.isnan(new_c5):
            continue

        delta = new_c5 - current_c5
        if delta < 0:
            current_h_half = new_h_half.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h_half = new_h_half.copy()
                best_c5 = new_c5
        else:
            if np.random.rand() < np.exp(-delta / T):
                current_h_half = new_h_half.copy()
                current_c5 = new_c5

        T *= alpha
        r *= 0.99

    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points/2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
