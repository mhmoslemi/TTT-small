import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Because 2 * sum(h_half) = n_points / 2.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective_fft(h_vars):
        h_full = create_full_h(h_vars)
        n = len(h_full)
        h_fft = np.fft.fft(h_full, n=2 * n)
        s_fft = np.fft.fft(1.0 - h_full, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Symmetric initial guess based on initial_h_values
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n].copy()

    best_h_half = h_half.copy()
    best_c5 = objective_fft(best_h_half)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 30:
        n_attempt += 1
        rng = np.random.RandomState(seed + n_attempt)
        perturbed = best_h_half + rng.uniform(-0.1, 0.1, size=half_n)  # Increased perturbation
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if not np.isclose(total, 0):
            perturbed = perturbed * (target_sum_half / total)
        perturbed = np.clip(perturbed, 0, 1)

        # Try multiple optimization methods to escape local optima
        methods = ['COBYLA', 'SLSQP', 'Nelder-Mead', 'L-BFGS-B']
        for method in methods:
            if time.time() - start_time >= budget_s - 30:
                break
            result = minimize(
                fun=objective_fft,
                x0=perturbed,
                method=method,
                bounds=bounds_per_var,
                constraints=cons if method in ['SLSQP', 'COBYLA'] else (),
                options={'maxiter': 150, 'ftol': 1e-10, 'disp': False}
            )
            if result.success:
                new_h_half = result.x.reshape(half_n)
                new_h_full = create_full_h(new_h_half)
                if np.all(new_h_full >= 0) and np.all(new_h_full <= 1.0 + 1e-6) and np.isclose(np.sum(new_h_full), n_points / 2.0, atol=1e-6):
                    new_c5 = result.fun
                    if new_c5 < best_c5:
                        best_h_half = new_h_half.copy()
                        best_c5 = new_c5
                        break  # Found better solution in this attempt
            elif method == 'Nelder-Mead' and result.message in ['Maximum number of iterations has been exceeded.']:
                # Partially accept even if not fully converged
                new_h_half = result.x.reshape(half_n)
                new_h_full = create_full_h(new_h_half)
                if np.all(new_h_full >= 0) and np.all(new_h_full <= 1.0 + 1e-6) and np.isclose(np.sum(new_h_full), n_points / 2.0, atol=1e-6):
                    new_c5 = result.fun
                    if new_c5 < best_c5:
                        best_h_half = new_h_half.copy()
                        best_c5 = new_c5
                        break

    # Final verification step
    h_full = create_full_h(best_h_half)
    assert np.isclose(np.sum(h_full) - n_points / 2.0, 0, atol=1e-6), "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
