        h_full[half_n:] = h_half[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    personal_best_positions = positions.copy()
    personal_best_scores = np.array([compute_c5(p) for p in positions])
    global_idx = np.argmin(personal_best_scores)
    global_pos = personal_best_positions[global_idx].copy()
    global_score = personal_best_scores[global_idx]

    # PSO loop
    for _ in range(max_pso_iters):
        if time.time() - start_time > budget_s - 120:  # reserve 120s for refinement and final steps
            break
        for i in range(n_particles):
            r1 = np.random.rand(half_n)
            r2 = np.random.rand(half_n)
            velocities[i] = w * velocities[i] + c1 * r1 * (personal_best_positions[i] - positions[i]) + c2 * r2 * (global_pos - positions[i])
            velocities[i] = np.clip(velocities[i], -0.5, 0.5)
            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            s = np.sum(positions[i])
            if s == 0:
                positions[i].fill(target_sum_half / half_n)
            else:
                positions[i] *= (target_sum_half / s)
            score = compute_c5(positions[i])
            if score < personal_best_scores[i]:
                personal_best_positions[i] = positions[i].copy()
                personal_best_scores[i] = score
                if score < global_score:
                    global_pos = positions[i].copy()
                    global_score = score

    # Now refine with SQP
    def objective(x):
        h_full = np.zeros(n_points)
        h_full[:half_n] = x
        h_full[half_n:] = x[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Run SQP from global best
    result_sqp = minimize(
        fun=objective,
        x0=global_pos,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8, 'disp': False}
    )
    best_h_half = result_sqp.x if result_sqp.success else global_pos

    # Final checks and construction
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    h_full_sum = np.sum(h_full)
    if abs(h_full_sum - n_points / 2.0) > 1e-6:
        h_full *= (n_points / 2.0 / h_full_sum)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx

    return (h_full.tolist(), final_c5, n_points)
