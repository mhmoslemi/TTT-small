import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    def evaluate_candidate(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Symmetrize and scale initial h
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j])/2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points/2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half*2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n]

    # SA parameters
    T = 1.0
    alpha = 0.999  # Cooling factor
    r = 0.05  # Initial perturbation radius
    r_decay = 0.99  # Perturbation decay rate

    best_h_half = initial_h_half.copy()
    best_c5 = evaluate_candidate(best_h_half)
    current_h_half = best_h_half.copy()
    current_c5 = best_c5

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 10:
        # Generate candidate by perturbing all elements
        candidate = current_h_half + np.random.normal(0, r, half_n)
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate)
        if candidate_sum != 0:
            candidate = (candidate / candidate_sum) * target_sum_half
        else:
            candidate = np.zeros_like(candidate)
        
        new_c5 = evaluate_candidate(candidate)
        
        # Acceptance criterion
        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h_half[:] = candidate
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h_half[:] = candidate
                best_c5 = new_c5

        # Update temperature and perturbation radius
        T *= alpha
        r *= r_decay

    h_full = create_full_h(best_h_half)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0+1e-6))

    return (h_full.tolist(), best_c5, n_points)
