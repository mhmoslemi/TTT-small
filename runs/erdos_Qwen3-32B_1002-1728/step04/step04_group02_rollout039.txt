import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5_const(h):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    def normalize(h):
        h = np.clip(h, 0.0, 1.0)
        total = np.sum(h)
        if abs(total) < 1e-9:
            return np.zeros_like(h)
        scale = target_sum / total
        return np.clip(h * scale, 0.0, 1.0)

    def perturb(h, rate=0.1, noise_std=0.1):
        h_new = h.copy()
        mask = np.random.rand(n_points) < rate
        noise = np.random.normal(0, noise_std, n_points)
        h_new[mask] += noise[mask]
        return normalize(h_new)

    pop_size = 50
    np.random.seed(seed)
    population = [h_initial]
    while len(population) < pop_size:
        base = population[np.random.randint(len(population))]
        candidate = perturb(base)
        population.append(candidate)
    population = np.array(population)

    best_h = population[np.argmin([compute_c5_const(h) for h in population])].copy()
    best_c5 = compute_c5_const(best_h)

    elite_ratio = 0.1
    elite_count = int(pop_size * elite_ratio)
    tournament_size = 3
    generations = 0

    while time.time() - start_time < budget_s - 30:
        fit_vals = np.array([compute_c5_const(h) for h in population])
        best_idx = np.argmin(fit_vals)
        if fit_vals[best_idx] < best_c5:
            best_h = population[best_idx].copy()
            best_c5 = fit_vals[best_idx]

        elite_indices = np.argsort(fit_vals)[:elite_count]
        elites = population[elite_indices]
        new_pop = [ind for ind in elites]

        for _ in range(pop_size - elite_count):
            p1 = population[np.argmin([fit_vals[i] for i in np.random.choice(len(population), tournament_size)])]
            p2 = population[np.argmin([fit_vals[i] for i in np.random.choice(len(population), tournament_size)])]
            cx_point = np.random.randint(1, n_points)
            child = np.concatenate((p1[:cx_point], p2[cx_point:]))
            child = perturb(child, rate=0.2 if generations % 25 == 0 else 0.1)
            new_pop.append(child)

        population = np.array(new_pop)
        generations += 1

    h_final = best_h
    assert abs(np.sum(h_final) - target_sum) < 1e-6
    assert np.allclose(h_final, np.clip(h_final, 0.0, 1.0))
    return (h_final.tolist(), best_c5, n_points)
