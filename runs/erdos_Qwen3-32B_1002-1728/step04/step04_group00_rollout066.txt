    initial_h = np.array(initial_h_values)
    n_points = len(initial_h)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # Initial symmetric h_half construction
    h_half_initial = (initial_h[:half_n] + initial_h[n_points - half_n:]) / 2.0
    sum_h_initial = np.sum(h_half_initial)
    if sum_h_initial != 0:
        h_half_initial = h_half_initial * (target_sum_half / sum_h_initial)
    h_half_initial = np.clip(h_half_initial, 0, 1)
    sum_h_initial = np.sum(h_half_initial)
    if sum_h_initial != 0:
        h_half_initial = h_half_initial * (target_sum_half / sum_h_initial)

    # Initialize PSO
    n_particles = 100
    rng = np.random.default_rng(seed)
    positions = np.tile(h_half_initial.reshape(1, -1), (n_particles, 1)) + 0.02 * rng.normal(size=(n_particles, half_n))
    velocities = 0.01 * rng.normal(size=positions.shape)

    # Normalize initial particles
    for i in range(n_particles):
        positions[i] = np.clip(positions[i], 0, 1)
        norm = np.sum(positions[i])
        if norm != 0:
            positions[i] *= (target_sum_half / norm)

    # PSO parameters
    pbest_positions = positions.copy()
    pbest_fitness = np.array([objective(pos, n_points, dx, half_n) for pos in positions])
    gbest_index = np.argmin(pbest_fitness)
    gbest_position = pbest_positions[gbest_index]
    gbest_fitness = pbest_fitness[gbest_index]
    w_max, w_min, c1, c2 = 0.9, 0.4, 2.0, 2.0

    while time.time() - start_time < budget_s - 30:
        current_fitness = np.array([objective(pos, n_points, dx, half_n) for pos in positions])

        improved_idx = current_fitness < pbest_fitness
        pbest_positions[improved_idx] = positions[improved_idx]
        pbest_fitness[improved_idx] = current_fitness[improved_idx]

        new_gbest_index = np.argmin(pbest_fitness)
        if pbest_fitness[new_gbest_index] < gbest_fitness:
            gbest_position = pbest_positions[new_gbest_index]
            gbest_fitness = pbest_fitness[new_gbest_index]

        # Inertia weight decay
        time_used = time.time() - start_time
        fraction = time_used / budget_s
        w = w_max - (w_max - w_min) * fraction
        w = np.clip(w, w_min, w_max)

        # Velocity update
        r1 = rng.random(size=positions.shape)
        r2 = rng.random(size=positions.shape)
        global_part = np.tile(gbest_position.reshape(1, -1), (n_particles, 1))
        velocities = w * velocities + c1 * r1 * (pbest_positions - positions) + c2 * r2 * (global_part - positions)
        velocities = np.clip(velocities, -0.12, 0.12)

        # Position update and normalization
        positions += velocities
        positions = np.clip(positions, 0, 1)
        for i in range(n_particles):
            norm = np.sum(positions[i])
            if norm != 0:
                positions[i] = (positions[i] / norm) * target_sum_half
            positions[i] = np.clip(positions[i], 0, 1)

    # Final solution validation
    h_full = create_full_h(gbest_position, n_points, half_n)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= 0) & (h_full <= 1.0))
    return (h_full.tolist(), gbest_fitness, n_points)
