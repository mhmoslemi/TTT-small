import numpy as np
from scipy.optimize import basinhopping, Lbfgsb
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum of left half

    # Objective function
    def objective(x):
        h_half = np.clip(x, 0, 1)
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        h_full = np.clip(h_full, 0, 1)
        current_sum = np.sum(h_full)
        if current_sum != 0:
            h_full = h_full * (n_points / 2.0) / current_sum
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initial guess from symmetry
    h_half_initial = h_full_initial[:half_n].copy()
    sum_init = np.sum(h_half_initial)
    if sum_init != 0:
        h_half_initial = h_half_initial * (target_sum_half / sum_init)

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    minimizer_kwargs = {
        "method": "L-BFGS-B",
        "bounds": bounds_per_var,
        "options": {"ftol": 1e-10, "maxiter": 50}
    }

    # Simulated Annealing
    best_h_half = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 120:  # 2 min padding
        result_bh = basinhopping(

            niter=100,
            T=1.0,

            minimizer_kwargs=minimizer_kwargs,

        if result_bh.fun < best_c5:

    if best_h_half is None:

    h_full = np.zeros(n_points)
