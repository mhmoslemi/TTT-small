        h_full = create_full_h(h_clipped)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return np.max(corr)

    # Symmetric initialization
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum_full = np.sum(h_full_sym)
    if abs(current_sum_full - (n_points / 2.0)) > 1e-6 and current_sum_full != 0:
        h_full_sym = h_full_sym * (n_points / 2.0 / current_sum_full)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]
    current_sum_half = np.sum(h_half)
    if current_sum_half != 0:
        h_half = h_half / current_sum_half * target_sum_half
    h_half = np.clip(h_half, 0, 1)
    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 20:
        h_perturbed = best_h_half + np.random.uniform(-0.35, 0.35, size=half_n)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_sum = np.sum(h_perturbed)
        if h_sum != 0:
            h_perturbed = h_perturbed / h_sum * target_sum_half
        h_perturbed = np.clip(h_perturbed, 0, 1)

        bounds_per_var = Bounds(np.zeros(half_n), np.ones(half_n))
        def objective_func(x):
            x_clipped = np.clip(x, 0, 1)
            x_sum = np.sum(x_clipped)
            if x_sum == 0:
                return float('inf')
            x_proc = x_clipped / x_sum * target_sum_half
            x_proc = np.clip(x_proc, 0, 1)
            h_full = create_full_h(x_proc)
            corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
            return np.max(corr)

        result = minimize(
            fun=objective_func,
            x0=h_perturbed,
            method='L-BFGS-B',
            bounds=bounds_per_var,
            options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 400}
        )

        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_sum_candidate = np.sum(h_candidate)
            if h_sum_candidate != 0:
                h_candidate = h_candidate / h_sum_candidate * target_sum_half
            h_candidate = np.clip(h_candidate, 0, 1)
            c5 = objective(h_candidate)
            if c5 < best_c5:
                best_h_half = h_candidate
                best_c5 = c5

    h_full = create_full_h(best_h_half)
    h_sum = np.sum(h_full)
    if abs(h_sum - (n_points / 2.0)) > 1e-6:
        h_full = h_full * (n_points / 2.0 / h_sum)
    h_full = np.clip(h_full, 0, 1)
    final_corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), np.max(final_corr), n_points)
