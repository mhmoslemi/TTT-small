import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Total integral constraint

    # Break symmetry in initial guess
    np.random.seed(seed)
    h = h_initial.copy().astype(np.float64)
    pert = np.random.normal(0, 0.1, size=n_points)
    h += pert
    h = np.clip(h, 0, 1)
    h *= (target_sum / np.sum(h))

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_constraints(h_flat):
        h = h_flat.reshape(n_points)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-8:
            h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
        return h

    # Simulated Annealing Phase (70% of time budget)
    sa_end = start_time + budget_s * 0.7
    best_h = h.copy()
    best_c5 = objective(best_h)
    
    temp = 0.5
    cooling_rate = 0.9995
    min_temp = 1e-6

    while time.time() < sa_end:
        # Perturb
        h_proposal = best_h.copy()
        idx = np.random.choice(n_points, size=5, replace=False)
        perturbation = np.random.uniform(-0.2, 0.2, size=5)
        h_proposal[idx] += perturbation
        h_proposal = repair_constraints(h_proposal)
        
        current_c5 = objective(h_proposal)
        delta = current_c5 - best_c5
        
        # Accept better or probabilistic worse
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            if current_c5 < best_c5:
                best_h = h_proposal.copy()
                best_c5 = current_c5
        
        # Cool down
        temp = max(temp * cooling_rate, min_temp)
    
    # Deterministic Refinement Phase
    bounds = [(0, 1)] * n_points
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result = minimize(
        fun=lambda x: objective(x),
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 300, 'ftol': 1e-8}
    )
    
    if result.success:
        refined_h = repair_constraints(result.x)
        final_c5 = objective(refined_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = refined_h

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
