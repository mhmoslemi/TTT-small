        pass  # Skip errors in refinement

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Constraint violation"
    assert np.allclose(np.clip(h_full, 0, 1), h_full), "Value bounds violation"
    return (h_full.tolist(), best_c5, n_points)
