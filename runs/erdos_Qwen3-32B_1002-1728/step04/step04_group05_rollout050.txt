import numpy as np
import time

def evaluate_candidate(y_half):
    half_n = len(y_half)
    n = half_n * 2
    h_full = np.zeros(n)
    for i in range(half_n):
        h_full[i] = h_full[n - 1 - i] = y_half[i]
    dx = 2.0 / n
    correlate = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(correlate) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0

    # Symmetrize and normalize initial y
    initial_y = h_full_initial[:half_n].copy()
    current_sum = np.sum(initial_y)
    if current_sum != 0:
        initial_y = initial_y * (target_sum_half / current_sum)
    initial_y = np.clip(initial_y, 0, 1)
    current_y = initial_y.copy()
    best_y = initial_y.copy()
    best_c5 = evaluate_candidate(best_y)

    # SA parameters
    T = 1.0
    alpha = 0.999
    initial_mutation_scale = 0.05
    mutation_scale = initial_mutation_scale
    np.random.seed(seed)

    while (time.time() - start_time) < (budget_s - 10):
        # Generate candidate
        new_y = current_y.copy()
        n_perturb = max(1, int(0.1 * half_n))
        perturb_indices = np.random.choice(half_n, size=n_perturb, replace=False)
        delta = np.random.normal(0, mutation_scale, size=n_perturb)
        new_y[perturb_indices] += delta
        new_y_clipped = np.clip(new_y, 0, 1)
        sum_clipped = np.sum(new_y_clipped)
        if sum_clipped == 0:
            continue
        scale = target_sum_half / sum_clipped
        scaled_y = new_y_clipped * scale

        # Check bounds
        if np.any(scaled_y > 1. + 1e-8) or np.any(scaled_y < -1e-8):
            continue
        c5_candidate = evaluate_candidate(scaled_y)
        delta_c5 = c5_candidate - best_c5

        # Acceptance criterion
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            current_y = scaled_y.copy()
            if c5_candidate < best_c5:
                best_y = current_y.copy()
                best_c5 = c5_candidate

        # Temperature and mutation schedule
        T *= alpha
        if T < 1e-6:
            break
        mutation_scale = max(mutation_scale * 0.999, 5e-4)

    # Construct final h
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = best_y[i]
    h_full = np.clip(h_full, 0, 1)
    assert np.all(h_full >= -1e-8) and np.all(h_full <= 1.0 + 1e-8)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
