            best_c5 = current_best
            best_individual = evaluated[0][0].copy()

        new_population = []
        elites = [ep[0].copy() for ep in evaluated[:2]]
        new_population.extend(elites)

        remaining_size = population_size - len(elites)
        for _ in range(remaining_size):
            p1 = tournament_select(evaluated)
            p2 = tournament_select(evaluated)
            cross_point = rs.randint(1, half_n-1)
            child = np.concatenate((p1[:cross_point], p2[cross_point:]))
            if rs.rand() < 0.1:
                n_mutate = max(1, int(0.05 * half_n))
                idxs = rs.choice(half_n, size=n_mutate, replace=False)
                child[idxs] += rs.normal(0, 0.05, size=n_mutate)
                child = np.clip(child, 0, 1.0)
                total = np.sum(child)
                if total != 0:
                    child *= (target_sum / total)
                child = np.clip(child, 0, 1.0)
            new_population.append(child)

        population = new_population

    final_h = create_full_h(best_individual)
    return (final_h.tolist(), best_c5, n_points)
