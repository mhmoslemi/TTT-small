    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Because full sum is n_points / 2

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}]

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if current_sum != 0 and abs(current_sum - (n_points / 2.0)) > 1e-6:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    n_attempt = 0

    decay_rate = 0.8
    noise_amplitude = 0.3

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        noise_scale = noise_amplitude * (1 - frac ** decay_rate)

        # Increase number of perturbations per iteration
        for _ in range(4):
            seed += 1
            rnd = np.random.RandomState(seed)
            perturbed = best_h_half + rnd.normal(0, noise_scale, half_n)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total != 0:
                perturbed = (perturbed / total) * target_sum_half
                perturbed = np.clip(perturbed, 0, 1)

            method = 'SLSQP' if n_attempt % 2 == 0 else 'L-BFGS-B'
            result = minimize(
                fun=objective,
                x0=perturbed,
                method=method,
                bounds=bounds_per_var,
                constraints=cons,
                options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
            )

            if result.success and result.fun < best_c5:
                new_h_half = np.clip(result.x, 0, 1)
                if abs(np.sum(new_h_half) - target_sum_half) < 1e-6:
                    new_h = create_full_h(new_h_half)
                    corr = np.correlate(new_h, 1 - new_h, mode='full')
                    new_c5 = np.max(corr) * dx
                    if new_c5 < best_c5:
                        best_h_half = new_h_half
                        best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, f"Sum constraint violation: {np.sum(h_full)}"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), f"Value out of bounds: {h_full}"

    return (h_full.tolist(), best_c5, n_points)
