        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_candidate.copy()
            best_c5 = c5_candidate

        T *= cooling_rate

    def objective(x):
        return evaluate_c5(x)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    result = minimize(
        fun=objective,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success:
        h_refined = result.x
        c5_refined = evaluate_c5(h_refined)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    return (best_h.tolist(), best_c5, n_points)
