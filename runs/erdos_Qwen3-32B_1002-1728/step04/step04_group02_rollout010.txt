import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def normalize(particle):
        p_clipped = np.clip(particle, 0, 1)
        s = np.sum(p_clipped)
        if s == 0:
            return np.full_like(p_clipped, 0.5)
        scale = target_sum / s
        p_scaled = p_clipped * scale
        return np.clip(p_scaled, 0, 1)

    def compute_c5(h):
        n = len(h)
        h_fft = np.fft.fft(h, 2 * n - 1)
        one_minus_h_fft = np.fft.fft(1.0 - h, 2 * n - 1)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real
        return np.max(corr) * dx

    np.random.seed(seed)
    N_PARTICLES = 50
    MAX_VELOCITY = 0.5
    particles = np.zeros((N_PARTICLES, n_points))
    velocities = np.zeros_like(particles)
    particles[0] = normalize(h_initial.copy())

        perturbed = h_initial + np.random.uniform(-0.1, 0.1, n_points)
        particles[i] = normalize(perturbed)

    pbest_positions = particles.copy()
    pbest_values = np.array([compute_c5(p) for p in particles])
    gbest_idx = np.argmin(pbest_values)
    gbest_position = particles[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    w_inertia = 0.7
    w_cognitive = 1.0
