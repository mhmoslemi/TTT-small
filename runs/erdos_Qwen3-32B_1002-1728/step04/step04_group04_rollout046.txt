                child_sum = child.sum()
                if child_sum > 1e-10:
                    child = (child / child_sum) * target_sum
                child = np.clip(child, 0, 1)
                new_population.append(child)
            if len(new_population) >= pop_size:
                break

        population = new_population[:pop_size]

    final_h = best_h
    final_h_sum = final_h.sum()
    if abs(final_h_sum - target_sum) > 1e-6:
        final_h = np.clip(final_h, 0, 1)
        final_h *= target_sum / final_h_sum

    final_c5 = compute_c5(final_h, dx)
    assert abs(final_h.sum() - target_sum) < 1e-6, "Constraint violation"
    return (final_h.tolist(), final_c5, n_points)
