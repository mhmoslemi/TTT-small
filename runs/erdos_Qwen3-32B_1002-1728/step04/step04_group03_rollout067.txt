            perturbation = np.random.uniform(-0.05, 0.05, n_points) + perturbation * 0.5

    # Final validation
    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.all(np.isclose(np.clip(best_h, 0, 1.0), best_h, atol=1e-5)), "Bounds violated"
    return (best_h.tolist(), best_c5, n_points)
