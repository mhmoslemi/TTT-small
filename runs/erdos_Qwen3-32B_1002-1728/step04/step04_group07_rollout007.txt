    if abs(current_sum_best - target_sum_half) > 1e-6 and current_sum_best > 1e-10:
        best_h_half = (best_h_half / current_sum_best) * target_sum_half
    best_h_half = np.clip(best_h_half, 0.0, 1.0)

    bounds = [(0.0, 1.0) for _ in range(half_n)]
    constraints = {'type': 'eq', 'fun': sum_constraint}

    try:
        result = minimize(objective, x0=best_h_half, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 200, 'ftol': 1e-10})
        if result.success and result.fun < best_c5:
            best_h_half = result.x.copy()
            best_c5 = result.fun
    except:
        pass

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
