    for i in range(half_n):
        j = n_points - 1 - i
        h_full_initial[i] = h_full_initial[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_initial)
    if abs(current_sum - (n_points / 2.0)) > 1e-6:
        h_full_initial *= (n_points / 2.0) / current_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    y_initial = h_full_initial[:half_n]

    # Setup constraints and bounds
    A = np.ones((1, half_n))
    linear_constraint = scipy.optimize.LinearConstraint(A, lb=[target_sum_y], ub=[target_sum_y])
    bounds = [(0, 1) for _ in range(half_n)]

    best_c5 = float('inf')
    best_y = y_initial.copy()

    # Multi-start optimization
    remaining_time = budget_s - 10
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        # Perturb initial guess
        current_y = y_initial.copy()
        noise = np.random.default_rng(seed).normal(0, 0.01, half_n)
        current_y += noise
        current_y = np.clip(current_y, 0, 1)
        total = np.sum(current_y)
        if total != 0:
            current_y *= (target_sum_y / total)
        current_y = np.clip(current_y, 0, 1)

        # Run SLSQP
        result = scipy.optimize.minimize(
            fun=objective_func,
            x0=current_y,
            args=(n_points, dx),
            bounds=bounds,
            constraints=linear_constraint,
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            c5_candidate = result.fun
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_y = result.x.copy()

    # Construct final result
    h_full = create_full_h(best_y, n_points)
    h_full = np.clip(h_full, 0, 1)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
