def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(initial_h)

    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def symmetric_h(h_half):
        h = np.zeros(n_points)
        h[:half_n] = h_half
        h[half_n:] = h_half[::-1]
        return h

    def evaluate(ind):
        h = symmetric_h(ind)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_h(ind):
        # Normalize to sum constraint and bounds
        total = np.sum(ind)
        if total == 0:
            return ind  # Avoid division by zero
        ind = (ind * target_sum_half) / total
        ind = np.clip(ind, 0, 1)
        return ind

    np.random.seed(seed)
    population_size = 40
    elite_pct = 0.05
    mutate_pct = 0.1
    mutate_strength = 0.05
    repair_clip = 0.05

    # Initialize population
    population = []

    population.append(repair_h(h_avg))
    # Generate M-1 symmetric random candidates
    for _ in range(population_size - 1):
        h_half = np.random.rand(half_n)
        population.append(repair_h(h_half))
    population = [repair_h(p) for p in population]

    best_c5 = np.inf
    best_h = None

    while time.time() - start_time < budget_s - 10:

        fitness = [evaluate(p) for p in population]

        offspring = []
        # Elitism: keep top 5%
