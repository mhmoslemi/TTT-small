        if time.time() - start_time > budget_s - 5:
            break

    h_full = create_full_h(best_y)
    assert np.all((h_full >= -1e-6) & (h_full <= 1 + 1e-6))
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
