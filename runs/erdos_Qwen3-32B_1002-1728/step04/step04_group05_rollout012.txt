import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0

    def create_full_h(y):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = y[i]
            h_full[n_points - 1 - i] = y[i]
        return h_full

    def evaluate_candidate(y):
        h_full = create_full_h(y)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * (2.0 / n_points)

    y_initial = h_full_initial[:half_n].copy()
    current_sum = np.sum(y_initial)
    if current_sum != 0:
        y_initial = (y_initial / current_sum) * target_sum
    y_initial = np.clip(y_initial, 0, 1)
    best_y = y_initial.copy()
    best_c5 = evaluate_candidate(best_y)

    learning_rate = 1e-3
    perturbation_mag = 0.01
    tabu_list = []
    MAX_TABU = 50
    NO_IMPROVEMENT_THRESHOLD = 100
    no_improvement = 0
    delta = 1e-5

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 10:
        current_y = best_y.copy()
        base_c5 = evaluate_candidate(current_y)

        grad = np.zeros(half_n)
        for i in range(half_n):
            y_perturbed = current_y.copy()
            y_perturbed[i] += delta
            y_perturbed = np.clip(y_perturbed, 0, 1)
            sum_perturbed = np.sum(y_perturbed)
            if sum_perturbed != 0:
                y_perturbed = (y_perturbed / sum_perturbed) * target_sum
            y_perturbed = np.clip(y_perturbed, 0, 1)
            c5_up = evaluate_candidate(y_perturbed)
            grad[i] = (c5_up - base_c5) / delta

        updated_y = current_y - learning_rate * grad
        updated_y = np.clip(updated_y, 0, 1)
        sum_updated = np.sum(updated_y)
        if sum_updated != 0:
            updated_y = (updated_y / sum_updated) * target_sum
        updated_y = np.clip(updated_y, 0, 1)
        new_c5 = evaluate_candidate(updated_y)

        if new_c5 < best_c5:
            best_c5 = new_c5
