import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_y = n_points / 4.0

    y_initial = h_initial[:half_n].copy()
    sum_y_initial = y_initial.sum()
    if sum_y_initial != 0:
        y_initial = y_initial * (target_sum_y / sum_y_initial)
    y_initial = np.clip(y_initial, 0, 1)

    def create_h_from_y(y_vec):
        h_full = np.zeros(n_points)
        h_full[:half_n] = y_vec
        h_full[half_n:] = y_vec[::-1]
        return h_full

    def objective_projected(y_vec):
        y_clipped = np.clip(np.array(y_vec), 0, 1)
        sum_clipped = y_clipped.sum()
        if sum_clipped < 1e-8:
            return 1e6
        y_normalized = y_clipped * (target_sum_y / sum_clipped)
        h_full = create_h_from_y(y_normalized)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr

    bounds = [(0, 1)] * half_n
    best_y = y_initial.copy()
    best_c5 = objective_projected(best_y)

    np.random.seed(seed)
    start_time = time.time()

    while time.time() - start_time < 0.9 * budget_s:
        perturbation = np.random.normal(0, 0.1, half_n)
        y_perturbed = best_y + perturbation
        y_perturbed = np.clip(y_perturbed, 0, 1)
        sum_perturbed = y_perturbed.sum()
        if sum_perturbed < 1e-8:
            y_perturbed = np.repeat(target_sum_y / half_n, half_n)
        else:
            y_perturbed = y_perturbed * (target_sum_y / sum_perturbed)
        result = minimize(
            objective_projected, 
            x0=y_perturbed, 
            method='L-BFGS-B', 
            bounds=bounds,
            options={'maxfun': 200}
        )
        if result.success:
            candidate_y = result.x
            candidate_c5 = result.fun
            if candidate_c5 < best_c5:
                best_y = candidate_y.copy()
                best_c5 = candidate_c5

    h_full = create_h_from_y(best_y)
    assert np.isclose(np.sum(h_full), n_points / 2.0, atol=1e-6)

    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), (np.min(h_full), np.max(h_full))
    return (h_full.tolist(), best_c5, n_points)
