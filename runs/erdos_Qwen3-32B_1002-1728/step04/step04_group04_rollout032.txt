            c5_new = objective(perturbed)
            if c5_new < best_c5:
                best_h = perturbed
                best_c5 = c5_new

    # Final tightening of solution using tighter tolerances
    if best_c5 > 1e-6:
        final_result = minimize(
            fun=objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_per_var,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-12, 'disp': False}
        )
        if final_result.success:
            h_final = final_result.x.copy()
            h_final = np.clip(h_final, 0, 1)
            total_final = np.sum(h_final)
            if abs(total_final - target_sum) < 1e-6:
                best_h = h_final
                best_c5 = objective(h_final)

    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Out of bounds"
    return (h_full.tolist(), best_c5, n_points)
