    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    # Symmetric initialization
    h_half_initial = h_full_initial[:half_n].copy()
    for i in range(half_n):
        j = n_points - 1 - i

    current_sum = np.sum(h_half_initial)
    if current_sum != 0:

    h_half_initial = np.clip(h_half_initial, 0, 1)
    h_full = create_full_h(h_half_initial)

    def de_objective(h_vars):

        if current_sum < 1e-8:

        h_scaled = np.clip(h_scaled, 0, 1)

        corr = np.correlate(h_full_de, 1 - h_full_de, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half_n
