def create_symmetric_h_full(h_half, n_points, half_n):
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[-half_n:] = h_half[::-1]
    return h_full

def normalize_h_half(h_half, target_sum):
    current_sum = np.sum(h_half)
    if current_sum == 0:
        h_half = np.full_like(h_half, target_sum / len(h_half))
    else:
        scale = target_sum / current_sum
        h_half = h_half * scale
    return np.clip(h_half, 0.0, 1.0)

def compute_c5(x, n_points, half_n, dx):
    h_full = create_symmetric_h_full(x, n_points, half_n)
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    max_corr = np.max(np.abs(corr))
    return max_corr * dx
