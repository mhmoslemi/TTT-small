import numpy as np
import time
import math
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        return h_full

    # Initialize symmetric h
    h_full = create_full_h(h_full_initial[:half_n])
    current_sum = h_full.sum()
    if current_sum != 0:
        h_full = (h_full / current_sum) * (n_points / 2.0)
    h_full = np.clip(h_full, 0, 1)
    best_h_half = h_full[:half_n].copy()
    best_c5 = np.max(np.correlate(h_full, 1 - h_full, mode='full')) * dx

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.996  # Slightly slower cooling
    T_min = 1e-5

    np.random.seed(seed)
    while time.time() - start_time < budget_s:
        if T < T_min:
            break

        # Generate perturbation
        noise = np.random.normal(0, T, size=half_n)
        new_h_half = best_h_half + noise
        new_h_half = np.clip(new_h_half, 0, 1)

        # Normalize sum
        current_sum_new = new_h_half.sum()
        if current_sum_new > 1e-10:  # Avoid division by zero
            scale = target_sum_half / current_sum_new
            new_h_half *= scale
        new_h_half = np.clip(new_h_half, 0, 1)

        # Create full h and compute new C5
        new_h_full = create_full_h(new_h_half)
        corr = np.correlate(new_h_full, 1 - new_h_full, mode='full')
        new_c5 = np.max(corr) * dx

        # Acceptance step
        delta_E = best_c5 - new_c5
        if delta_E < 0:
            accept = True  # Improvement
        else:
            accept = np.random.rand() < math.exp(delta_E / T)

        if accept:
            best_h_half[:] = new_h_half[:]

        # Cool down
        T *= alpha

    # Final checks and return
    h_full = create_full_h(best_h_half)
