import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def evaluate(individual):
        h_full = create_full_h(individual)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym = h_full_sym * scale
    initial_h_half = np.clip(h_full_sym, 0, 1)[:half_n]

    n_particles = 100
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    velocity_clamp = 0.1

    np.random.seed(seed)
    positions = []
    for _ in range(n_particles):
        p = initial_h_half + np.random.rand(half_n) * 0.1
        p = np.clip(p, 0, 1)
        p = (p / p.sum()) * target_sum_half
        positions.append(p)
    positions = np.array(positions)
    velocities = np.random.rand(n_particles, half_n) * 0.01

    pbest_positions = positions.copy()
    pbest_scores = np.array([evaluate(p) for p in positions])
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    while time.time() - start_time < budget_s:
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        w = max(w_min, w)

        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (pbest_positions[i] - positions[i]) +
                             c2 * r2 * (gbest_position - positions[i]))
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)

            new_position = np.clip(positions[i] + velocities[i], 0, 1)
            total = np.sum(new_position)
            if total != 0:
                new_position = (new_position / total) * target_sum_half
            else:
                new_position.fill(target_sum_half / half_n)
            positions[i] = new_position

            score = evaluate(new_position)
            if score < pbest_scores[i]:
                pbest_scores[i] = score
                pbest_positions[i] = positions[i].copy()
                if score < gbest_score:
                    gbest_score = score
                    gbest_position = new_position.copy()

    h_full = create_full_h(gbest_position)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), gbest_score, n_points)
