    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0

        if current_sum < 1e-9:
            h_half[:] = target_sum_half / half_n

    # Symmetric initialization
    h_full_initial = create_symmetric_h_full(h_full_initial[:half_n])

        corr = np.correlate(h_full, 1.0 - h_full, mode='full')

                    best_c5 = trial_c5

    def objective(x):
