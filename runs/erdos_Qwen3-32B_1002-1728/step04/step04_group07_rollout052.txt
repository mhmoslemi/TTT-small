import time

    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def create_full_h(h_sym):
        h = np.zeros(n_points)
        h[:half_n] = h_sym
        h[-half_n:] = h_sym[::-1]
        return h

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_constraints(h_sym):
        h_sym = np.clip(h_sym, 0, 1)
        current_sum = np.sum(h_sym)
        if not np.isclose(current_sum, 0):
            scale = target_sum / (2 * current_sum)
            h_sym = np.clip(h_sym * scale, 0, 1)
        if not np.isclose(np.sum(h_sym), target_sum / 2, atol=1e-5):
            h_sym = h_sym / np.sum(h_sym) * (target_sum / 2)
        return h_sym

    # Initialize from symmetric average

    current_sum = np.sum(h_initial[:half_n])

        h_sym_initial = repair_constraints(h_sym_initial)
    best_h = h_sym_initial.copy()

    # PSO parameters

    max_iter_pso = 200

    particles = np.stack([repair_constraints(p) for p in particles])
