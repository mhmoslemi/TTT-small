import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    # PSO parameters
    num_particles = 50
    inertia = 0.7
    personal_coeff = 1.49
    social_coeff = 1.49
    max_velocity = 0.5

    # Initialize particles around the given initial values
    np.random.seed(seed)
    h_half_initial = h_full_initial[:half_n].copy()
    positions = []
    velocities = []
    for _ in range(num_particles):
        perturbation = np.random.uniform(-0.1, 0.1, size=half_n)
        pos = np.clip(h_half_initial + perturbation, 0, 1)
        total = np.sum(pos)
        pos = pos * (target_sum_half / total) if total != 0 else 0.5 * np.ones(half_n)
        positions.append(pos)
        velocities.append(np.zeros_like(pos))

    positions = np.array(positions)
    velocities = np.array(velocities)

    # Compute personal best and global best
    def compute_c5(h_half):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half[i]
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr) * dx

    personal_best_positions = positions.copy()
    personal_best_scores = np.array([compute_c5(p) for p in positions])
    global_best_idx = np.argmin(personal_best_scores)
    global_best_pos = personal_best_positions[global_best_idx]
    global_best_score = personal_best_scores[global_best_idx]
    best_c5 = global_best_score
    best_h_half = global_best_pos.copy()

    while time.time() - start_time < budget_s - 60:
        for i in range(num_particles):
            if time.time() - start_time >= budget_s - 60:
                break
            r1, r2 = np.random.rand(2, half_n)
            velocities[i] = (
                inertia * velocities[i]
                + personal_coeff * r1 * (personal_best_positions[i] - positions[i])
                + social_coeff * r2 * (global_best_pos - positions[i])
            )
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            positions[i] += velocities[i]
            positions[i] = np.clip(positions[i], 0, 1)
            total = np.sum(positions[i])
            positions[i] = positions[i] * (target_sum_half / total) if total != 0 else 0.5 * np.ones(half_n)

            c5 = compute_c5(positions[i])
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best_positions[i] = positions[i].copy()
                if c5 < global_best_score:
                    global_best_pos = positions[i].copy()
                    global_best_score = c5
                    best_h_half[:] = global_best_pos
                    best_c5 = c5

    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = best_h_half[i]
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Bounds exceeded"
    return (h_full.tolist(), best_c5, n_points)
