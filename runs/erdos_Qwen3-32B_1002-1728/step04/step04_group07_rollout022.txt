        if total != 0:
            h_full = (h_full / total) * (n_points / 2.0)
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    lb = [0.0] * half_n
    ub = [1.0] * half_n

    with time.time() - start_time < budget_s:
        res = minimize(
            fun=local_objective,
            x0=best_h_half,
            method='L-BFGS-B',
            bounds=list(zip(lb, ub)),
            options={'maxiter': 200, 'ftol': 1e-10}
        )
    if res.success:
        refined_h_half = res.x
        refined_c5 = res.fun
        if refined_c5 < best_c5:
            best_h_half = refined_h_half
            best_c5 = refined_c5

    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 60:
        for _ in range(3):
            if time.time() - start_time >= budget_s:
                break
            perturbed = best_h_half + np.random.uniform(-0.1, 0.1, half_n)
            perturbed = np.clip(perturbed, 0.0, 1.0)
            res_perturb = minimize(
                fun=local_objective,
                x0=perturbed,
                method='L-BFGS-B',
                bounds=list(zip(lb, ub)),
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if res_perturb.success:
                new_h_half = res_perturb.x
                new_c5 = res_perturb.fun
                if new_c5 < best_c5:
                    best_h_half = new_h_half
                    best_c5 = new_c5

    h_full = create_full_h(best_h_half)
    total_full = np.sum(h_full)
    if total_full != 0:
        h_full = (h_full / total_full) * (n_points / 2.0)
    h_full = np.clip(h_full, 0, 1)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), f"Values out of bounds: {h_full}"
    return (h_full.tolist(), best_c5, n_points)
