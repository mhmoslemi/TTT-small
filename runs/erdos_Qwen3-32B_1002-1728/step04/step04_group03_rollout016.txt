from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_points = len(initial_h_values)
    
    # Quaternary sequence construction with precomputed properties
    base_pattern = np.array([0, 3, 3, 0, 1, 2, 2, 1], dtype=int)
    pattern_len = len(base_pattern)
    full_q = np.tile(base_pattern, (n_points + pattern_len - 1) // pattern_len)[:n_points]
    h_initial = full_q / 3.0
    
    # Ensure initial solution has proper properties
    h_initial = np.clip(h_initial, 0, 1)
    norm_factor = (n_points / 2) / np.sum(h_initial)
    h_initial *= norm_factor
    h_initial = np.clip(h_initial, 0, 1)
    
    dx = 2.0 / n_points
    
    # Optimization targets and constraints
    def objective(h_vars):
        h_var = np.array(h_vars)
        n = len(h_var)
        h_fft = np.fft.fft(h_var, n=2*n)
        s_fft = np.fft.fft(1.0 - h_var, n=2*n)
        corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(corr) * dx
    
    # Setup for optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0})
    
    best_h = h_initial.copy()
    best_value = objective(best_h)
    
    np.random.seed(seed)
    
    while time.time() - start_time < budget_s - 30:
        # Random perturbation strategy
        perturbed = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if not np.isclose(perturbed_sum, 0):
            perturbed = perturbed * (n_points / 2 / perturbed_sum)
            perturbed = np.clip(perturbed, 0, 1)
            
        # Optimization with restart strategy
       olvers = [('SLSQP', {'constraints': cons}), 
                 ('L-BFGS-B', {}), 
                 ('TNC', {})]
                 
        for method, method_kwargs in solvers:
            if time.time() - start_time >= budget_s - 30:
                break
                
            work_constraints = method_kwargs.pop('constraints', [])
            result = minimize(
                objective, 
                x0=perturbed, 
                method=method,
                bounds=bounds,
                constraints=work_constraints if work_constraints else None,
                options={'maxiter': 100, 'ftol': 1e-8, 'disp': False},
                **method_kwargs
            )
            
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_sum = np.sum(new_h)
                if not np.isclose(new_sum, n_points / 2, atol=1e-6):
                    new_h = new_h * (n_points / 2 / new_sum)
                    new_h = np.clip(new_h, 0, 1)
                new_value = objective(new_h)
                
                if new_value < best_value:
                    best_h = new_h.copy()
                    best_value = new_value
    
    # Final verification
    verification_h = best_h
    verification_h = np.clip(verification_h, 0, 1)
    sum_h = np.sum(verification_h)
    if not np.isclose(sum_h, n_points / 2, atol=1e-6):
        verification_h = verification_h * (n_points / 2 / sum_h)
        verification_h = np.clip(verification_h, 0, 1)
    
    return (verification_h.tolist(), best_value, n_points)
