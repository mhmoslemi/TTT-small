import numpy as np
import time

def compute_c5(h_full):
    n = len(h_full)
    dx = 2.0 / n
    h_fft = np.fft.fft(h_full, 2 * n - 1)
    one_minus_h_fft = np.fft.fft(1.0 - h_full, 2 * n - 1)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    return np.max(corr) * dx

    n_points = len(h_initial)

    inertia_w = 0.7
    cog_w = 1.0
    soc_w = 1.0
    max_vel = 0.5

    particles = []
    velocities = []
    pbest_pos = []
    pbest_cost = []
    for _ in range(n_particles):
        h = h_initial + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum) > 1e-10:
