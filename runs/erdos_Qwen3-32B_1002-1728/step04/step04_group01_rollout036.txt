import numpy as np
from scipy.optimize import minimize
import time

def evaluate_C5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h_initial has correct sum
    current_sum = h_initial.sum()
    if abs(current_sum - target_sum) > 1e-6:
        h_initial = h_initial / (current_sum + 1e-12) * target_sum
    h_initial = np.clip(h_initial, 0.0, 1.0)

    best_h = h_initial.copy()
    current_h = h_initial.copy()
    best_c5 = evaluate_C5(current_h, dx)

    np.random.seed(seed)
    T = 0.15  # Initial temperature
    cooling_rate = 0.999

    while time.time() - start_time < budget_s - 60:
        remaining_time = budget_s - (time.time() - start_time)
        perturb_count = min(n_points, max(5, int(n_points ** 0.5)))
        for _ in range(min(int(remaining_time * 10), 1000)):
            trial_h = current_h.copy()
            # Perturb a random subset
            indices = np.random.choice(n_points, size=min(perturb_count, n_points), replace=False)
            # Perturbation scales with current values for better mixing
            base_vals = trial_h[indices]
            perturbations = (np.random.rand(len(indices)) - 0.5) * (1.0 - base_vals) * 0.7 * (1.0 - T)
            trial_h[indices] += perturbations
            # Clip to bounds
            trial_h = np.clip(trial_h, 0.0, 1.0)
            s = trial_h.sum()
            if s != 0 and abs(s) > 1e-6:
                # Scale to maintain total sum constraint
                scale = target_sum / s
                trial_h = np.clip(trial_h * scale, 0.0, 1.0)
            # Enforce sum constraint exactly
            norm_scale = target_sum / (trial_h.sum() + 1e-12)
            trial_h = trial_h * norm_scale
            trial_h = np.clip(trial_h, 0.0, 1.0)

            # Evaluate and accept/reject
            new_c5 = evaluate_C5(trial_h, dx)
            delta = new_c5 - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = trial_h.copy()
                if new_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = new_c5

            # Update temperature
            T *= cooling_rate ** (1.0 / (5 * n_points))

        if time.time() - start_time >= budget_s - 60:
            break

    # Local optimization phase with SLSQP
    def objective_func(h_flat):
        return evaluate_C5(h_flat.reshape(n_points), dx)

    result = minimize(
        fun=objective_func,
        x0=current_h,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n_points,
        constraints={'type': 'eq', 'fun': lambda x: x.sum() - target_sum},
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    if result.success:
        refined_h = np.clip(result.x, 0.0, 1.0)
        refined_h = refined_h / (refined_h.sum() + 1e-12) * target_sum
        refined_h = np.clip(refined_h, 0.0, 1.0)
        new_c5 = evaluate_C5(refined_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = refined_h

    assert abs(best_h.sum() - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return (best_h.tolist(), best_c5, n_points)
