import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n): h_full[i] = h_full[n_points-1-i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def repair(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        scale = target_sum_half / max(1e-6, np.sum(x_clipped))
        return np.clip(x_clipped * scale, 0.0, 1.0)

    np.random.seed(seed)
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        h_full_sym[i] = h_full_sym[n_points-1-i] = (h_full_initial[i] + h_full_initial[n_points-1-i]) / 2
    h_half = h_full_sym[0:half_n] * (target_sum_half / np.sum(h_full_sym[0:half_n]))
    h_half = np.clip(h_half, 0.0, 1.0)

    # Particle Swarm Optimization
    n_p, w, c1, c2 = 30, 0.7, 1.5, 1.5
    positions = [repair(h_half + np.random.uniform(-0.1, 0.1, half_n)) for _ in range(n_p)]
    velocities = [np.random.uniform(-0.05, 0.05, half_n) for _ in range(n_p)]
    p_best, g_best, g_score = [0]*n_p, h_half.copy(), float('inf')

    for i, x in enumerate(positions):
        fo = objective(x)
        p_best[i] = (fo, x.copy())
        if fo < g_score:
            g_best = x.copy()
            g_score = fo

    t0 = time.time()
    while time.time() - start_time < budget_s - 50:  # Reserve time for final phase
        for i in range(n_p):

            velocities[i] = w * velocities[i] + c1 * r1 * (p_best[i][1] - positions[i]) + c2 * r2 * (g_best - positions[i])
