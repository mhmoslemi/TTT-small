                    options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
                )
            else:
                res = minimize(
                    fun=objective_fft,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons if method != 'L-BFGS-B' else (),
                    options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
                )
            if res.success:
                new_h_half = res.x
                new_h_full = create_full_h(new_h_half)
                if np.isclose(np.sum(new_h_full), n_points / 2.0, atol=1e-6) and np.all(new_h_full <= 1.0):
                    new_c5 = res.fun
                    if new_c5 < best_c5:
                        best_h_half = new_h_half.copy()
                        best_c5 = new_c5
                        break  # Skip remaining methods for this candidate

        # Occasional global search using differential evolution
        if time.time() - start_time < budget_s - 150 and half_n <= 100:
            def objective_de(x):
                return objective_fft(x)
            de_bounds = [(0.0, 1.0)] * half_n
            de_p0 = best_h_half if np.isclose(np.sum(best_h_half), target_sum_half, atol=0.5) else perturbed
            res_de = differential_evolution(
                objective_de,
                de_bounds,
                maxiter=10,
                popsize=15,
                init='sobol',
                strategy='best1bin',
                tol=1e-6,
                mutation=(0.5, 1.0),
                recombination=0.7,
                seed=rng.integers(1 << 30)
            )
            if res_de.success and res_de.fun < best_c5:
                best_h_half = res_de.x
                best_c5 = res_de.fun

    # Final assertion and return
    h_full = create_full_h(best_h_half)
    assert np.isclose(np.sum(h_full), n_points / 2.0, atol=1e-6), "Integral violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
