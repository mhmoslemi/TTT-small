import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from functools import partial

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = (n_points + 1) // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0
    max_gen = 500

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-(n_points - half_n):] = h_half[:(n_points - half_n)][::-1]  # Ensures symmetric
        return h_full

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        h_full = np.clip(h_full, 0, 1)
        corr = np.correlate(h_full, 1 - h_full, mode="full")
        valid_corr = corr[np.abs(np.arange(len(corr)) - (n_points - 1)) <= n_points - 1]
        return np.max(valid_corr) * dx

    def normalize(h_half):
        h_half = np.clip(h_half, 0, 1)
        current_sum = h_half.sum()
        if abs(current_sum - target_sum_half) > 1e-8 and current_sum != 0:
            h_half = h_half * (target_sum_half / current_sum)
        h_half = np.clip(h_half, 0, 1)
        return h_half

    def init_individual():
        if hasattr(h_initial, "__len__"):
            half_init = h_initial[:half_n]
        else:
            half_init = np.random.rand(half_n)
        half_init = normalize(half_init)
        return half_init

    def mutate(h_half, mutation_rate=0.2):
        noise = np.random.normal(0, mutation_rate, half_n)
        h_mutated = h_half + noise
        return normalize(h_mutated)

    def crossover(p1, p2):
        split = np.random.randint(1, half_n)
        child = np.concatenate([p1[:split], p2[split:]])
        return normalize(child)

    def tournament_selection(pop, k=3):
        indices = np.random.choice(len(pop), k, replace=False)
        best_idx = min(indices, key=lambda i: compute_c5(pop[i]))
        return pop[best_idx]

    def convex_refine(h_half):
        h_var = cp.Variable(half_n)
        objective = cp.Minimize(cp.max(
            cp.convex.kl_div(h_var, h_half)
        ))
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum_half
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT, maxiters=150)
        return normalize(h_var.value)

    np.random.seed(seed)

    pop = [init_individual() for _ in range(pop_size)]
    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)

    # GA Evolution Loop
    for gen in range(max_gen):
        if time.time() - start_time >= budget_s - 60:
            break

        new_pop = []

        # Elitism (preserve top 20%)

            parent2 = tournament_selection(pop)
            child = crossover(parent1, parent2)
            child = mutate(child, (0.1 if gen < 100 else 0.25))

    final_c5 = compute_c5(best_h)

        h_full.tolist(),
