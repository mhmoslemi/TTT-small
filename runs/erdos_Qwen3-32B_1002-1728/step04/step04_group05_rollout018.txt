import numpy as np
import time
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(y):
        y_proj = np.clip(y, 0, 1).copy()
        s = np.sum(y_proj)
        if s != 0:
            y_proj = y_proj * (target_sum / s)
        y_proj = np.clip(y_proj, 0, 1)
        h_full = np.zeros(n_points)
        h_full[:half_n] = y_proj
        h_full[half_n:] = y_proj[::-1]
        return h_full

    def evaluate_c5(y):
        h_full = create_full_h(y)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    initial_y = h_initial[:half_n].copy()
    best_y = initial_y.copy()
    best_c5 = evaluate_c5(best_y)

    np.random.seed(seed)
    MAX_RESTARTS = int(1.5 * (budget_s * 0.8) / n_points)
    MAX_ITER = 100
    restart_count = 0

    while time.time() - start_time < budget_s - 10:
        # Perturb initial_y for restart
        perturb = np.random.normal(0, 0.1, len(initial_y))
        current_y = np.clip(initial_y + perturb, 0, 1)
        s = np.sum(current_y)
        if s != 0:
            current_y = current_y * (target_sum / s)
        current_y = np.clip(current_y, 0, 1)

        # Run optimization
        res = minimize(evaluate_c5, current_y, method='L-BFGS-B', bounds=Bounds(0, 1), 
                      options={'maxiter': MAX_ITER, 'ftol': 1e-7, 'gtol': 1e-6})
        if res.success and res.fun < best_c5:
            best_y = res.x
            best_c5 = res.fun

        restart_count += 1
        if restart_count >= 3 * MAX_RESTARTS:
            break

    h_full = create_full_h(best_y)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), best_c5, n_points)
