import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 4.0  # sum of first half due to symmetry

    # Symmetricize and scale initial guess
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        h_full_sym[i] = h_full_sym[n_points - 1 - i] = (h_full_initial[i] + h_full_initial[n_points - 1 - i])/2
    current_sum = np.sum(h_full_sym)
    if np.abs(current_sum - n_points/2.0) > 1e-6:
        h_full_sym = h_full_sym * (n_points/2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_initial = h_full_sym[:half_n]

    def make_full_h(x):
        h = np.zeros(n_points)
        for i in range(half_n):
            h[i] = h[n_points - 1 - i] = x[i]
        return h

    def objective(x):
        h = make_full_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half_n
    constraints = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    best_x = h_half_initial.copy()
    best_val = objective(best_x)

    while time.time() - start_time < budget_s - 10:
        x0 = best_x.copy()
        perturbation = np.random.normal(0, 0.03, size=half_n)
        x0 += perturbation
        x0 = np.clip(x0, 0, 1)
        s = np.sum(x0)
        if s != 0:
            x0 = (x0 / s) * target_sum
        
        result = opt.minimize(objective, x0=x0, bounds=bounds, constraints=constraints, method='SLSQP',
                              options={'maxiter': 200, 'ftol': 1e-6})
        if result.success and result.fun < best_val:
            best_val = result.fun
            best_x = result.x.copy()
            
        if time.time() - start_time > budget_s - 5:
            break

    h_final = make_full_h(best_x)
    h_final = np.clip(h_final, 0, 1)
    assert abs(np.sum(h_final) - n_points/2.0) < 1e-6
    
    return (h_final.tolist(), best_val, n_points)
