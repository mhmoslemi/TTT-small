import numpy as np
import time
from numpy.random import default_rng
from scipy.optimize import minimize, Bounds, LinearConstraint

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def evaluate_candidate(h_half):
        h_full = create_symmetric_h_full(h_half)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr) * dx

    def repair_candidate(h_half):
        h_half = np.clip(h_half, 0.0, 1.0)
        current_sum = np.sum(h_half)
        if current_sum > 1e-6:
            h_half *= target_sum_half / current_sum
        return h_half

    population_size = 30
    rng = default_rng(seed)
    population = np.zeros((population_size, half_n))
    h_half_initial = h_full_initial[:half_n].copy()
    population[0] = repair_candidate(h_half_initial)
    for i in range(1, population_size):
        noise = rng.normal(0.0, 0.1, half_n)
        candidate = population[0] + noise
        population[i] = repair_candidate(candidate)

    best_c5 = evaluate_candidate(population[0])
    best_h_half = population[0].copy()

    F = 0.9
    CR = 0.9
    generations = 600

    for gen in range(generations):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(population_size):
            idxs = rng.choice(population_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            trial = np.copy(population[i])
            mask = (rng.random(half_n) < CR) | (rng.integers(0, half_n) == np.arange(half_n))
            trial = np.where(mask, mutant, population[i])
            trial = repair_candidate(trial)
            trial_c5 = evaluate_candidate(trial)
            if trial_c5 < evaluate_candidate(population[i]):
                population[i] = trial
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h_half = trial.copy()

        if gen % 10 == 0:
            top_indices = np.argsort([evaluate_candidate(p) for p in population])[:2]
            for idx in top_indices:
                if time.time() - start_time > budget_s - 10:
                    break
                candidate = population[idx].copy()
                def obj(x):
                    return evaluate_candidate(x)
                result = minimize(obj, x0=candidate, method='L-BFGS-B',
                                  bounds=Bounds(0.0, 1.0),
                                  constraints=LinearConstraint(np.ones(half_n), 
                                                              lb=[target_sum_half - 1e-6], 
                                                              ub=[target_sum_half + 1e-6]),
                                  options={'maxiter': 30})
                if result.success:
                    refined = repair_candidate(result.x)
                    population[idx] = refined
                    refined_c5 = evaluate_candidate(refined)
                    if refined_c5 < best_c5:
                        best_c5 = refined_c5
                        best_h_half = refined.copy()

    h_full = create_symmetric_h_full(best_h_half)
    h_full = repair_candidate(h_full)
    h_full = create_symmetric_h_full(repair_candidate(h_full[:half_n]))

    return (h_full.tolist(), best_c5, n_points)
