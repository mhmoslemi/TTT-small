from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    def create_full_h(h_vars):
        h_full = np.zeros(n_points, dtype=np.float64)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    # Symmetrize initial h
    h_full_sym = h_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_initial[i] + h_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[:half_n]

    best_h_half = h_half.copy()
    best_c5 = np.max(np.correlate(create_full_h(h_half), 1 - create_full_h(h_half), mode='full')) * dx

    np.random.seed(seed)

    def objective郭(x):
        x = np.clip(x, 0, 1)
        s = np.sum(x)
        if s < 1e-9:  # Avoid division by zero
            return float('inf')
        x = (x / s) * target_sum_half
        x = np.clip(x, 0, 1)
        h = create_full_h(x)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half_n

    # Run DE optimization
    try:
        pop_limit = max(50, int(half_n * 0.5))
        result_de = differential_evolution(
            objective郭,
            bounds,
            mutation=(0.5, 1.0),
            popsize=pop_limit,
            recombination=0.9,
            strategy='best1bin',
            polish=False,
            workers=1,
            updating='deferred',
            maxiter=30,
            tol=1e-7
        )
        if result_de.success:
            improved = result_de.x
            improved = np.clip(improved, 0, 1)
            s = np.sum(improved)
            if s < 1e-9:
                new_c5 = best_c5
            else:
                improved = (improved / s) * target_sum_half
                improved = np.clip(improved, 0, 1)
                h = create_full_h(improved)
                corr = np.correlate(h, 1 - h, mode='full')
                new_c5 = np.max(corr) * dx
            if new_c5 < best_c5:
                best_h_half = improved
                best_c5 = new_c5
    except Exception as e:

    # Time remaining check
    elapsed = time.time() - start_time
    if elapsed < 1000:  # Reserve 100s for cleanup
        try:
            from scipy.optimize import minimize
            # Local refinement after DE
            res_loc = minimize(
                fun=objective郭,
                x0=best_h_half,
                bounds=bounds,
                method='L-BFGS-B',
                options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8}
            )
            if res_loc.success:
                local_best = res_loc.x
                local_best = np.clip(local_best, 0, 1)
                s = np.sum(local_best)
                if s < 1e-9:
                    final_c5 = best_c5

                    final_corr = np.correlate(final_h, 1 - final_h, mode='full')
                    final_c5 = np.max(final_corr) * dx

                    best_c5 = final_c5
        except Exception as e:
            pass

    h_final = create_full_h(best_h_half)
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6)), "Value bounds violated"
    assert abs(np.sum(h_final) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    return (h_final.tolist(), best_c5, n_points)
