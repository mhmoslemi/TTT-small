Now, implementing the expected improvement (EI) acquisition function:

The EI is defined as the expected value of the improvement over the current best value. For a new candidate x, we have the predicted mean μ(x) and standard deviation σ(x) from the GP. The EI is computed as:

EI(x) = (μ(x) - y_best - ξ) * Φ((μ(x) - y_best - ξ) / σ(x)) + σ(x) * φ((μ(x) - y_best - ξ)/σ(x))

where Φ is the standard normal CDF and φ is the standard normal PDF. Here, ξ is a small parameter to balance exploration/exploitation. Typically, ξ is set to a small value like 0.0 or 0.01. EI = (y_best - mu) * Phi((y_best - mu) / sigma) + sigma * phi((y_best - mu)/sigma)

import numpy as np
from scipy.optimize import minimize
import time
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
from scipy.stats import norm

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vars):
        h_full = h_vars.copy()
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def project_to_feasible(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum == 0:
            h_clipped[:] = target_sum / n_points
        else:
            scale = target_sum / current_sum
            h_scaled = h_clipped * scale
            h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    # Normalize initial guess
    h_current = h_full_initial.copy()
    h_current = project_to_feasible(h_current)
    best_c5 = objective(h_current)
    best_h = h_current.copy()

    # Setup Gaussian Process
    kernel = RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e2)) * ConstantKernel(1.0)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-6, n_restarts_optimizer=0)
    X = h_current.reshape(1, -1)
    y = np.array([best_c5]).ravel()
    gp.fit(X, y)

    np.random.seed(seed)
    candidate_batch_size = 30  # Reduced for efficiency
    acquired_candidates = [X[0].copy()]

    while time.time() - start_time < budget_s - 60:
        # Generate candidate points in feasible space
        while len(acquired_candidates) < candidate_batch_size:
            raw_candidates = np.random.rand()
            cand = np.random.rand(n_points)
            cand = project_to_feasible(cand)
            # Avoid duplicates
            if not any(np.allclose(cand, a, atol=1e-2) for a in acquired_candidates):
                acquired_candidates.append(cand)
        candidates = np.array(acquired_candidates[-candidate_batch_size:])
        acquired_candidates = [acquired_candidates[0]]  # Reset after batch

        # GP prediction
        mu, sigma = gp.predict(candidates, return_std=True)
        y_best = best_c5
        imp = y_best - mu  # Since we minimize
        with np.errstate(divide='ignore'):
            Z = imp / (sigma + 1e-8)
            ei = imp * norm.cdf(Z) + sigma * norm.pdf(Z)
        ei[sigma == 0] = 0.0

        next_idx = np.argmax(ei)
        next_candidate = candidates[next_idx].copy()

        # Local optimization
        def local_constrained_opt(x0):
            bounds = [(0, 1) for _ in range(n_points)]
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
            result = minimize(
                objective, x0=project_to_feasible(x0),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 40, 'ftol': 1e-8}
            )
            new_h = result.x
            new_h = project_to_feasible(new_h)
            return new_h, objective(new_h)

        new_h, new_c5 = local_constrained_opt(next_candidate)

        # Update GP
        X = np.vstack((X, new_h.reshape(1, -1)))
        y = np.append(y, new_c5)
        gp.fit(X, y)

        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Out of bounds"
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
