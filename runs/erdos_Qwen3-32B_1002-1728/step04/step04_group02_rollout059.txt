            gbest_value = pbest_values[i]
            gbest_position = particles[i].copy()

    velocities = 0.1 * np.random.rand(n_particles, n_points)

    # PSO Main Loop
    while time.time() - start_time < budget_s - 60:
        for i in range(n_particles):
            # Velocity update
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest_positions[i] - particles[i]) +
                             social * r2 * (gbest_position - particles[i]))
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)

            # Position update
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            current_sum = np.sum(particles[i])
            if current_sum == 0:
                particles[i] = gbest_position.copy()
                current_sum = target_sum
            else:
                particles[i] = particles[i] * target_sum / current_sum
            particles[i] = np.clip(particles[i], 0, 1)

            # Update pbest and gbest
            c5_new = compute_c5(particles[i])
            if c5_new < pbest_values[i]:
                pbest_values[i] = c5_new
                pbest_positions[i] = particles[i].copy()
                if c5_new < gbest_value:
                    gbest_value = c5_new
                    gbest_position = particles[i].copy()

    h_full = gbest_position
    assert abs(np.sum(h_full) - target_sum) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), gbest_value, n_points)
