        raise ValueError("n_points must be divisible by n_low")
    group_size = n_points // n_low
    h_low = h_full_initial.reshape(n_low, group_size).mean(axis=1)
    current_sum = np.sum(h_low)
    h_low = h_low * (n_low / 2.0) / current_sum
    h_low = np.clip(h_low, 0.0, 1.0)

    # Low-resolution optimization
    best_c5_low = compute_c5(h_low)
    best_h_low = h_low.copy()

    np.random.seed(seed)
    trials_low = 5
    for trial in range(trials_low):
        if time.time() - start_time >= budget_s - 60:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_low)
        h_perturbed = best_h_low + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        if abs(current_sum - n_low / 2.0) > 1e-6 and current_sum != 0:
            h_perturbed = h_perturbed * (n_low / 2.0) / current_sum
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        def objective_low(x):
            return compute_c5(x)

        bounds_low = [(0.0, 1.0)] * n_low
        cons_low = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_low / 2.0})
        result_low = minimize(
            objective_low, h_perturbed, method='trust-constr',
            bounds=bounds_low, constraints=cons_low,
            options={'maxiter': 300, 'xtol': 1e-8, 'gtol': 1e-8}
        )
        if result_low.success:
            h_candidate = result_low.x
            if np.all((h_candidate >= 0.0) & (h_candidate <= 1.0)) and abs(np.sum(h_candidate) - n_low / 2.0) < 1e-6:
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < best_c5_low:
                    best_c5_low = c5_candidate
                    best_h_low = h_candidate

    # Upscale and high-resolution optimization
    h_high = upscale_h(best_h_low, n_low, n_points)
    best_h_high = h_high
    best_c5_high = compute_c5(h_high)

    trials_high = 3
    for trial in range(trials_high):
        if time.time() - start_time >= budget_s - 30:
            break
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = best_h_high + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        current_sum = np.sum(h_perturbed)
        if abs(current_sum - target_sum_high) > 1e-6 and current_sum != 0:
            h_perturbed = h_perturbed * (target_sum_high / current_sum)
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)

        def objective_high(x):
            return compute_c5(x)

        bounds_high = [(0.0, 1.0)] * n_points
        cons_high = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_high})
        result_high = minimize(
            objective_high, h_perturbed, method='trust-constr',
            bounds=bounds_high, constraints=cons_high,
            options={'maxiter': 300, 'xtol': 1e-8, 'gtol': 1e-8}
        )
        if result_high.success:
            h_candidate = result_high.x
            if np.all((h_candidate >= 0.0) & (h_candidate <= 1.0)) and abs(np.sum(h_candidate) - target_sum_high) < 1e-6:
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < best_c5_high:
                    best_c5_high = c5_candidate
                    best_h_high = h_candidate

    h_full = best_h_high
    assert abs(np.sum(h_full) - target_sum_high) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= 0.0) & (h_full <= 1.0)), "Value bounds violated"
    return (h_full.tolist(), best_c5_high, n_points)
