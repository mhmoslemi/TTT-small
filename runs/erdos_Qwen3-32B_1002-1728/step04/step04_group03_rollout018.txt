import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Convert to quaternary sequence and adjust sum
    q = np.clip(np.rint(3 * h_initial).astype(int), 0, 3)

    while diff != 0:

        for i in idxs:

                q[i] -= 1
                diff -= 1
                if diff == 0:
