import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_full = n_points / 2.0
    target_sum_half = (n_points / 2.0) / 2.0

    def create_full_h(h_half_candidate):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_half_candidate[i]
        return h_full

    def compute_c5(h_full):
        n = len(h_full)
        h_fft = np.fft.fft(h_full, 2 * n - 1)
        one_minus_h_fft = np.fft.fft(1.0 - h_full, 2 * n - 1)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = np.fft.ifft(corr_fft).real * dx
        return np.max(corr)

    np.random.seed(seed)
    population_size = 50
    population = []
    h_half_initial = h_full_initial[:half_n].copy()
    current_sum = np.sum(h_half_initial)
    if current_sum != 0:
        h_half_initial = h_half_initial * (target_sum_half / current_sum)
    h_half_initial = np.clip(h_half_initial, 0, 1)
    population.append(h_half_initial)

    for _ in range(population_size - 1):
        h_candidate = h_half_initial.copy() + np.random.normal(0, 0.1, half_n)
        h_candidate = np.clip(h_candidate, 0, 1)
        current_sum = np.sum(h_candidate)
        if current_sum != 0:
            h_candidate = h_candidate * (target_sum_half / current_sum)
        h_candidate = np.clip(h_candidate, 0, 1)
        population.append(h_candidate)

    population = np.array(population)
    fitness = []
    for h_half in population:
        h_full = create_full_h(h_half)
        fitness.append(compute_c5(h_full))
    fitness = np.array(fitness)
    best_idx = np.argmin(fitness)
    best_h_half = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    def tournament_selection(pop, fit, tournament_size=3):
        indices = np.random.choice(len(pop), tournament_size)
        tournament_fit = fit[indices]
        winner_idx = indices[np.argmin(tournament_fit)]
        return pop[winner_idx]

    while time.time() - start_time < budget_s - 60:
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            if np.random.rand() < 0.7:
                cp = np.random.randint(1, half_n)
                child1 = np.concatenate((parent1[:cp], parent2[cp:]))
                child2 = np.concatenate((parent2[:cp], parent1[cp:]))
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            def mutate(h, mr=0.1, std=0.1):
                mask = np.random.rand(len(h)) < mr
                h[mask] += np.random.normal(0, std, size=np.sum(mask))
                h = np.clip(h, 0, 1)
                s = max(np.sum(h), 1e-8)
                h = h * (target_sum_half / s)
                return np.clip(h, 0, 1)

            child1 = mutate(child1)
            child2 = mutate(child2)

            h_full1 = create_full_h(child1)
            f1 = compute_c5(h_full1)
            h_full2 = create_full_h(child2)
            f2 = compute_c5(h_full2)
            new_population.extend([(child1, f1), (child2, f2)])
            if f1 < best_c5:
                best_h_half = child1.copy()

                best_c5 = f2

            best_c5 = fitness[best_idx]

        constraints = [cp.sum(h_vars) == target_sum_half, h_vars <= 1]
        def c5_obj(_):
            h_full = create_full_h(_)

            return h_r, c5_r
