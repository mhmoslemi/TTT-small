        noise = 0.1 * rng.standard_normal(half_n)
        position = np.clip(h_half_initial + noise, 0, 1)
        position = repair(position, target_sum_half)
        velocity = 0.1 * rng.uniform(-1, 1, half_n)
        curr_val = objective(position, n_points, dx)
        particles.append({
            'position': position.copy(),
            'velocity': velocity.copy(),
            'p_best': position.copy(),
            'p_best_value': curr_val
        })

    p_best_values = np.array([p['p_best_value'] for p in particles])
    g_best_index = np.argmin(p_best_values)
    g_best = particles[g_best_index]['p_best'].copy()
    g_best_value = p_best_values[g_best_index]

    velocity_clamp = 0.2 * (target_sum_half / half_n)

    while time.time() - start_time < budget_s - 10:
        w = max(0.4, w * 0.99)
        iter_velocity_resets = 0
        for p in particles:
            r1 = rng.uniform()
            r2 = rng.uniform()
            cognitive = c1 * r1 * (p['p_best'] - p['position'])
            social = c2 * r2 * (g_best - p['position'])
            p['velocity'] = w * p['velocity'] + cognitive + social
            p['velocity'] = np.clip(
                p['velocity'],
                -velocity_clamp * 1.5,
                velocity_clamp * 1.5
            )
            p['position'] = repair(
                p['position'] + p['velocity'],
                target_sum_half
            )
            curr_val = objective(p['position'], n_points, dx)
            if curr_val < p['p_best_value']:
                p['p_best'] = p['position'].copy()
                p['p_best_value'] = curr_val
                if curr_val < g_best_value:
                    g_best = p['position'].copy()
                    g_best_value = curr_val

        if iter_velocity_resets >= 50:
            for p in particles:
                p['velocity'] = 0.1 * rng.uniform(-1, 1, half_n)
            iter_velocity_resets = 0

    h_full = create_full_h(g_best, n_points)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    
    return (h_full.tolist(), g_best_value, n_points)
