def run(seed=42, budget_s=1100, **kwargs):

    half_coarse = n_coarse // 2
    target_sum_coarse = n_coarse / 4.0

    def create_full_coarse(h_vars):
        h_full = np.zeros(n_coarse)
        for i in range(half_coarse):
            h_full[i] = h_full[n_coarse - 1 - i] = np.clip(h_vars[i], 0, 1)
        return h_full

    def objective_coarse(h_vars):
        h_full = create_full_coarse(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * (2.0 / n_coarse)
        sum_var = np.sum(h_vars)
        penalty = 1e4 * (sum_var - target_sum_coarse) ** 2
        return c5 + penalty

    bounds_coarse = [(0.0, 1.0) for _ in range(half_coarse)]
    h_half_coarse_initial = np.full(half_coarse, target_sum_coarse / half_coarse)
    remaining_budget = budget_s

    # Coarse grid optimization
    result_coarse = differential_evolution(
        func=objective_coarse,
        bounds=bounds_coarse,
        strategy='best1bin',
        maxiter=20,
        popsize=10,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        polish=True
    )
    best_h_half_coarse = np.clip(result_coarse.x, 0, 1)

        h_half_fine_coarse = np.clip(h_half_fine_coarse * scaling, 0, 1)

    # Fine grid setup
    half_fine = n_fine // 2
