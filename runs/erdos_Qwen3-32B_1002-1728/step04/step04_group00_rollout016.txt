            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective_de(x):
        x = np.clip(x, 0, 1)
        total = np.sum(x)
        if total == 0:
            return float('inf')
        scale = target_sum_half / total
        x = x * scale
        x = np.clip(x, 0, 1)
        h_full = create_full_h(x)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n
    strategy = 'best1bin'
    popsize = 10
    mutation = (0.5, 1.0)
    recombination = 0.9
    maxiter = 1000

    # Estimate time per iteration and adjust maxiter dynamically
    max_walltime = budget_s - 10  # leave buffer
    # Conservative estimate: assume 0.1s per function call
    estimated_calls = int(max_walltime / 0.02)
    maxiter = estimated_calls // (popsize * 2)

    result = differential_evolution(
        objective_de, bounds,
        strategy=strategy,
        popsize=popsize,
        mutation=mutation,
        recombination=recombination,
        seed=seed,
        maxiter=maxiter,
        disp=False,
        polish=False,
        workers=1
    )

    best_h_half = np.clip(result.x, 0, 1)
    best_h_half = best_h_half * (target_sum_half / np.sum(best_h_half))
    best_h_half = np.clip(best_h_half, 0, 1)

    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5 = np.max(corr) * dx

    return (h_full.tolist(), c5, n_points)
