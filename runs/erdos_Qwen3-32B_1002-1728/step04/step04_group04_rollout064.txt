            parent2 = population[p2_index].copy()

            if np.random.rand() < crossover_rate:
                cx_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
                child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()

            # Mutation
            for child in [child1, child2]:
                mutation = np.random.normal(0, 0.1, n_points)
                child += mutation * (np.random.rand(n_points) < mutation_rate)
                new_population.append(adjust_candidate(child, target_sum))

        # Replace worst half
        sorted_idx = np.argsort(fitness)
        for i in range(len(new_population)):
            insert_idx = sorted_idx[-i-1] if i < len(sorted_idx)//2 else i
            population[insert_idx] = new_population[i]

        if time.time() - start_time > budget_s - 20:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
