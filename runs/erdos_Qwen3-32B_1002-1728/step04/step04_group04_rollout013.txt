import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def enforce_constraints(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum < 1e-10:
            return h_clipped
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    np.random.seed(seed)
    pop_size = 100
    population = []
    # Use initial_h_values with constraint enforcement
    h_initial_adj = enforce_constraints(h_initial.copy())
    population.append(h_initial_adj)
    # Generate remaining population members
    for _ in range(pop_size -1):
        h = np.random.rand(n_points)
        h = enforce_constraints(h)
        population.append(h)
    population = np.array(population)

    best_h = np.array(population[0].copy())
    best_c5 = evaluate_fitness(best_h)

    while time.time() - start_time < budget_s - 1.0:
        # Evaluate population
        fitness = np.array([evaluate_fitness(ind) for ind in population])
        # Update best
        current_best_idx = np.argmin(fitness)
        current_best = population[current_best_idx]
        current_c5 = fitness[current_best_idx]
        if current_c5 < best_c5:
            best_h[:] = current_best.copy()
            best_c5 = current_c5

        new_population = [best_h.copy()]  # Elitism

        # Tournament selection and offspring generation
        while len(new_population) < pop_size:
            # Select parents via tournament
            tourn_size = 3
            parent1_idx = np.argmin([fitness[np.random.choice(pop_size)] for _ in range(tourn_size)])
            parent2_idx = np.argmin([fitness[np.random.choice(pop_size)] for _ in range(tourn_size)])

            p1, p2 = population[parent1_idx], population[parent2_idx]
            # Crossover with 70% chance
            if np.random.rand() < 0.7:
                cross_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([p1[:cross_point], p2[cross_point:]])
                child2 = np.concatenate([p2[:cross_point], p1[cross_point:]])
            else:
                child1, child2 = p1.copy(), p2.copy()
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < 0.2:
                    perturbation = np.random.normal(0, 0.15, n_points)
                    np.clip(child + perturbation, 0, 1, out=child)
                child[:] = enforce_constraints(child)
                new_population.append(child)
                if len(new_population) >= pop_size:
                    break
        population = np.array(new_population)

    # Final constraint checks
    best_h = enforce_constraints(best_h)
    actual_sum = np.sum(best_h)
    if abs(actual_sum - target_sum) > 1e-6:
        best_h = (best_h / actual_sum) * target_sum
        best_h = np.clip(best_h, 0, 1)
    return (best_h.tolist(), evaluate_fitness(best_h), n_points)
