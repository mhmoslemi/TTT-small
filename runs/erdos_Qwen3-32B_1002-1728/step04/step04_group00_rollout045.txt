import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0  # h_half sum must satisfy sum(h_full) = n_points / 2
    dx = 2.0 / n_points

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def evaluate_c5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize h with symmetric constraints
    current_h_half = np.zeros(half_n)
    current_h_full = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (current_h_full[i] + current_h_full[j]) / 2
        current_h_full[i] = current_h_full[j] = avg
    current_sum = np.sum(current_h_full)
    if not np.allclose(current_sum, n_points / 2.0, atol=1e-6):
        scale = (n_points / 2.0) / current_sum
        current_h_full *= scale
    current_h_full = np.clip(current_h_full, 0, 1)
    current_h_half = current_h_full[:half_n]
    current_c5 = evaluate_c5(current_h_half)
    best_c5 = current_c5
    best_h_half = current_h_half.copy()

    # Simulated Annealing
    np.random.seed(seed)
    T = 1.0  # Initial temperature
    alpha = 0.9993  # Cooling rate
    T_min = 1e-5

    while time.time() - start_time < budget_s - 20:
        if T < T_min:
            break

        # Perturb and scale
        perturbed = current_h_half + np.random.normal(0, T, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        current_sum_pert = np.sum(perturbed)
        if not np.isclose(current_sum_pert, 0):
            scale = target_sum_half / current_sum_pert
            perturbed *= scale
        perturbed = np.clip(perturbed, 0, 1)

        new_c5 = evaluate_c5(perturbed)
        delta = new_c5 - current_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h_half = perturbed.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h_half[:] = current_h_half[:]
        T *= alpha

    # Final validity checks
    best_h_full = create_full_h(best_h_half)
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)

    return (best_h_full.tolist(), best_c5, n_points)
