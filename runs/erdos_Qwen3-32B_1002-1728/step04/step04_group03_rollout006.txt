    best_c5 = evaluate_erdos_solution(h_current, dx)

    # Simulated Annealing Parameters
    initial_T = 1.0
    cooling_rate = 0.995
    T = initial_T

    # Annealing Loop
    while time.time() - start_time < budget_s * 0.8:
        perturb = T * np.random.randn(n_points) * 0.1
        h_proposal = np.clip(h_current + perturb, 0, 1)
        sum_prop = np.sum(h_proposal)
        if sum_prop != 0:
            h_proposal = np.clip(h_proposal * (target_sum / sum_prop), 0, 1)
        else:
            continue  # Skip invalid samples
        c5_proposal = evaluate_erdos_solution(h_proposal, dx)
        delta = best_c5 - c5_proposal
        if (c5_proposal < best_c5) or (np.random.rand() < np.exp(delta / T)):
            h_current = h_proposal.copy()
            if c5_proposal < best_c5:
                best_h = h_proposal.copy()
                best_c5 = c5_proposal
        T *= cooling_rate

    # Gradient Refinement
    def objective(h_vec):
        h = h_vec.reshape(n_points)
        return evaluate_erdos_solution(h, dx)

    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0, 1)] * n_points
    x0 = best_h.flatten()
    cons = [{'type': 'eq', 'fun': constraint_sum}]

    if time.time() - start_time < budget_s - 30:
        result = minimize(
            objective, x0=x0, method='SLSQP', bounds=bounds,
            constraints=cons, options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_opt = np.clip(result.x.reshape(n_points), 0, 1)
            h_opt_sum = np.sum(h_opt)
            if not np.isclose(h_opt_sum, target_sum):
                scaling = target_sum / h_opt_sum
                h_opt = np.clip(h_opt * scaling, 0, 1)
            c5_opt = evaluate_erdos_solution(h_opt, dx)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Final Verification
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6)
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_c5, n_points)
