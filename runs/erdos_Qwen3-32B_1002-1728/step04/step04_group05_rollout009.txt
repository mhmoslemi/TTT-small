    target_half_sum = n_points / 4.0

    initial_h_half = h_full_initial[:half_n].copy()
    h_full_sym = create_full_h(initial_h_half)
    current_sum = np.sum(h_full_sym)
    if np.abs(current_sum - n_points / 2.0) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_half_sum * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n]

    swarm_size = 50
    inertia_initial = 0.9
    inertia_final = 0.4
    cognitive = 2.0
    social = 2.0

    np.random.seed(seed)

    positions = [initial_h_half.copy() + np.random.normal(0, 0.05, half_n) for _ in range(swarm_size)]
    for i in range(swarm_size):
        positions[i] = np.clip(positions[i], 0, 1)
        s = np.sum(positions[i])
        if s != 0:
            positions[i] = (positions[i] / s) * target_half_sum

    velocities = [np.random.randn(half_n) * 0.01 for _ in range(swarm_size)]
    pbest_positions = [p.copy() for p in positions]
    pbest_scores = [evaluate_candidate(p, dx) for p in pbest_positions]
    gbest_idx = np.argmin(pbest_scores)
    gbest_position = pbest_positions[gbest_idx]
    best_c5 = pbest_scores[gbest_idx]

    while time.time() - start_time < budget_s - 10:
        elapsed = time.time() - start_time
        t_ratio = min(elapsed / (budget_s - 10), 1.0)
        inertia = inertia_initial - (inertia_initial - inertia_final) * t_ratio

        for i in range(swarm_size):
            pos = positions[i]
            vel = velocities[i]
            r1 = np.random.rand()
            r2 = np.random.rand()
            pbest_dir = pbest_positions[i] - pos
            gbest_dir = gbest_position - pos
            new_vel = inertia * vel + cognitive * r1 * pbest_dir + social * r2 * gbest_dir
            velocities[i] = new_vel

            new_pos = pos + new_vel
            new_pos = np.clip(new_pos, 0, 1)
            s = np.sum(new_pos)
            if s != 0:
                new_pos = (new_pos / s) * target_half_sum
            else:
                new_pos = np.ones(half_n) * (target_half_sum / half_n)
            positions[i] = new_pos

            current_c5 = evaluate_candidate(new_pos, dx)
            if current_c5 < pbest_scores[i]:
                pbest_positions[i] = new_pos.copy()
                pbest_scores[i] = current_c5
                if current_c5 < best_c5:
                    gbest_position = new_pos.copy()
                    best_c5 = current_c5

    h_full = create_full_h(gbest_position)
    h_full = np.clip(h_full, 0, 1)
    total_integral = np.sum(h_full)
    if abs(total_integral - n_points / 2.0) > 1e-6:
        h_full = (h_full / total_integral) * (n_points / 2.0)
        h_full = np.clip(h_full, 0, 1)

    return (h_full.tolist(), best_c5, n_points)
