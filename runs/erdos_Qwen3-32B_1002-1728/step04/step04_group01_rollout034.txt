import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Because 2 * sum(h_half) = n_points / 2.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def evaluate_C5(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr * dx)

    # Symmetric initialization
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - n_points / 2.0) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n].copy()
    best_h_half = h_half
    best_c5 = evaluate_C5(best_h_half)

    # Cluster management
    def find_clusters(values):
        clusters = []
        n = len(values)
        if n == 0:
            return clusters
        start = 0
        current_val = values[0]
        for i in range(1, n):
            if abs(values[i] - current_val) > 1e-6:
                clusters.append({'start': start, 'end': i - 1, 'value': current_val})
                start = i
                current_val = values[i]
        clusters.append({'start': start, 'end': n - 1, 'value': current_val})
        return clusters

    clusters = find_clusters(best_h_half)

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 30 and clusters:
        if not clusters:
            break
        cluster_idx = np.random.randint(len(clusters))
        cluster = clusters[cluster_idx]
        s = cluster['start']
        e = cluster['end']
        v_current = cluster['value']
        L = e - s + 1
        k = s + (e - s) // 2
        L1 = k - s + 1
        L2 = e - k

        S = L * v_current
        lower_v1 = max(0.0, (S - L2) / L1)
        upper_v1 = min(1.0, S / L1)
        v1_candidates = np.linspace(lower_v1, upper_v1, 5)
        best_split_c5 = np.inf
        best_v1 = v_current
        best_v2 = v_current

        for v1 in v1_candidates:
            v2 = (S - L1 * v1) / L2
            if not (0 <= v2 <= 1):
                continue
            # Apply to temp h_half
            temp_h = best_h_half.copy()
            temp_h[s:k + 1] = v1
            temp_h[k + 1:e + 1] = v2
            h_full = create_full_h(temp_h)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr * dx)
            if c5 < best_split_c5:
                best_split_c5 = c5
                best_v1 = v1
                best_v2 = v2

        if best_split_c5 < best_c5 and np.isfinite(best_split_c5):
            # Remove old cluster
            clusters.pop(cluster_idx)
            # Add new clusters
            clusters.append({'start': s, 'end': k, 'value': best_v1})
            clusters.append({'start': k + 1, 'end': e, 'value': best_v2})
            # Update best_h_half
            best_h_half[s:k + 1] = best_v1
            best_h_half[k + 1:e + 1] = best_v2
            best_c5 = best_split_c5

    # Final check against constraints
    if not np.all((best_h_half >= -1e-10) & (best_h_half <= 1.0 + 1e-10)):
        best_h_half = np.clip(best_h_half, 0, 1)
    if abs(np.sum(best_h_half) - target_sum_half) > 1e-6:
        scale = target_sum_half / max(1e-10, np.sum(best_h_half))
        best_h_half = np.clip(best_h_half * scale, 0, 1)

    h_full = create_full_h(best_h_half)

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
