        return h_full

        sum_h = np.sum(h_clipped)
        if sum_h == 0:
            h_scaled = np.ones_like(h_clipped) * target_sum_half / h_clipped.size
        else:
            h_scaled = h_clipped * (target_sum_half / sum_h)
        h_full = create_full_h(h_scaled)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0)] * half_n
