import numpy as np
import time
import scipy.optimize as opt
from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def normalize_h_half(h_half):
        h_half = np.clip(h_half, 0.0, 1.0)
        current_sum = np.sum(h_half)
        if abs(current_sum) > 1e-10:
            h_half = h_half * (target_sum_half / current_sum)
        else:
            h_half = np.full_like(h_half, target_sum_half / half_n)
        return np.clip(h_half, 0.0, 1.0)

    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        h_full_sym[i] = h_full_sym[n_points-1-i] = (h_full_initial[i] + h_full_initial[n_points-1-i])/2
    current_sum = np.sum(h_full_sym)
    if current_sum != 0:
        h_full_sym = h_full_sym * (n_points/2.0) / current_sum
    h_full_sym = np.clip(h_full_sym, 0.0, 1.0)
    h_half_initial = h_full_sym[:half_n].copy()

    def evaluate_candidate(h_half_candidate):
        h_full = create_symmetric_h_full(h_half_candidate)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    population_size = 50
    population = np.zeros((population_size, half_n))
    rng = default_rng(seed)
    half_initial = h_half_initial.copy()
    population[0] = half_initial
    for i in range(1, population_size):
        perturb = rng.normal(0.0, 0.1, half_n)
        candidate = half_initial + perturb
        population[i] = normalize_h_half(candidate)

    best_c5 = evaluate_candidate(population[0])
    best_h_half = population[0].copy()
    iteration = 0

    while time.time() - start_time < budget_s - 30:
        fitness_scores = np.array([evaluate_candidate(population[i]) for i in range(population_size)])
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_c5:
            best_c5 = fitness_scores[min_idx]
            best_h_half = population[min_idx].copy()

        def tournament_select():
            indices = rng.integers(0, population_size, 3)
            winner_idx = indices[np.argmin(fitness_scores[indices])]
            return population[winner_idx].copy()

        new_offspring = []
        for _ in range(population_size // 2):
            p1 = tournament_select()
            p2 = tournament_select()
            if rng.random() < 0.8:
                cx_point = rng.integers(1, half_n - 1)
                c1 = np.hstack([p1[:cx_point], p2[cx_point:]])
                c2 = np.hstack([p2[:cx_point], p1[cx_point:]])
            else:
                c1, c2 = p1.copy(), p2.copy()
            for child in [c1, c2]:
                if rng.random() < 0.1:
                    mutation = rng.normal(0.0, 0.05, half_n)
                    child += mutation
                child = normalize_h_half(child)
                new_offspring.append(child)
        new_offspring = np.array(new_offspring)
        population = np.vstack([population, new_offspring])
        all_fitness = np.array([evaluate_candidate(population[i]) for i in range(population.shape[0])])
        indices = np.argsort(all_fitness)
        population = population[indices[:population_size]]

        iteration += 1
        if iteration % 5 == 0:
            def local_func(x):
                h_full = create_symmetric_h_full(x)
                corr = np.correlate(h_full, 1.0 - h_full, mode='full')
                return np.max(np.abs(corr)) * dx
            bounds = [(0.0, 1.0)] * half_n
            constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
            if time.time() - start_time < budget_s - 60:
                result = opt.minimize(local_func, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=[constraint], options={'maxiter': 20, 'ftol': 1e-6})
                if result.success:
                    new_h = result.x
                    new_h = np.clip(new_h, 0.0, 1.0)
                    s = np.sum(new_h)
                    if abs(s - target_sum_half) > 1e-6:
                        new_h = new_h * (target_sum_half / s)
                    new_h = np.clip(new_h, 0.0, 1.0)
                    population[population_size - 1] = new_h

    def final_local(x):
        h_full = create_symmetric_h_full(x)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx
    bounds = [(0.0, 1.0)] * half_n
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
    if time.time() - start_time < budget_s - 60:
        result = opt.minimize(final_local, x0=best_h_half, method='SLSQP', bounds=bounds, constraints=[constraint], options={'maxiter': 50, 'ftol': 1e-8})
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0.0, 1.0)
            s = np.sum(final_h)
            if abs(s - target_sum_half) > 1e-6:
                final_h = final_h * (target_sum_half / s)
            best_h_half = final_h

    h_full = create_symmetric_h_full(best_h_half)
    h_full = np.clip(h_full, 0.0, 1.0)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    return (h_full.tolist(), np.max(np.abs(np.correlate(h_full, 1 - h_full, mode='full'))) * dx, n_points)
