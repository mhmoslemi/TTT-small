import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)

    # Initial h normalization
    current_sum = h.sum()
    if current_sum != 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1.0)
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_c5(best_h)
    current_c5 = best_c5

    start_time = time.time()
    T = 1.0  # Initial temperature
    alpha = 0.995  # Cooling rate

    while time.time() - start_time < budget_s - 30:
        # Perturb
        noise = np.random.normal(0, 0.1 * T, n_points)
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1.0)
        sum_perturbed = perturbed.sum()
        if sum_perturbed == 0:
            perturbed = current_h.copy()
        else:
            scale = target_sum / sum_perturbed
            perturbed = perturbed * scale
            perturbed = np.clip(perturbed, 0, 1.0)
        new_c5 = evaluate_c5(perturbed)
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = perturbed.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
        # Cool down
        T *= alpha
        # Enforce T minimum
        if T < 1e-8:
            T = 1e-8

    # Gradient refinement
    def objective(x):
        h_val = x.reshape(-1)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    result = minimize(
        fun=objective,
        x0=best_h,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    if result.success:
        refined_h = result.x.reshape(-1)
        refined_h = np.clip(refined_h, 0, 1.0)  # Just in case
        refined_c5 = evaluate_c5(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h

    # Final validation

import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    current_sum = h.sum()

    h = np.clip(h, 0, 1.0)
    current_h = h.copy()
    best_h = current_h.copy()

    current_c5 = best_c5

    start_time = time.time()
    T = 1.0
    alpha = 0.995

    while time.time() - start_time < budget_s - 30:
        noise = np.random.normal(0, 0.1 * T, n_points)
        perturbed = current_h + noise
        perturbed = np.clip(perturbed, 0, 1.0)
        sum_perturbed = perturbed.sum()
        if sum_perturbed > 1e-10:
            scale = target_sum / sum_perturbed

                current_h, current_c5 = perturbed.copy(), new_c5
                if current_c5 < best_c5:
                    best_h, best_c5 = current_h.copy(), current_c5
        # Cool down
        T *= alpha
        if T < 1e-8:
            T = 1e-8

    def objective(x):
        h_val = x.reshape(-1)
        corr = np.correlate(h_val, 1 - h_val, mode='full')
        return np.max(corr * dx)

    bounds = [(0.0, 1.0)] * n_points

    result = minimize(objective, x0=best_h, bounds=bounds, constraints=cons, method='SLSQP',
