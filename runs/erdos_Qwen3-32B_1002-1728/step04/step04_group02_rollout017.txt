            if current_sum != 0:
                particles[i] = particles[i] * (target_sum / current_sum)

        # Check time constraint
        if time.time() - start_time >= budget_s:
            break

    # Final checks
    h_best = gbest_position.copy()
    final_c5 = compute_c5(h_best, dx)

    assert np.allclose(np.sum(h_best), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((h_best >= -1e-6) & (h_best <= 1 + 1e-6)), "Value bounds violated"

    return (h_best.tolist(), final_c5, n_points)
