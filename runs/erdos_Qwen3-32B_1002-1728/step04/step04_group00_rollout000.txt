        h_half_initial = h_half_initial * (target_sum_half / sum_half)
    h_half_initial = np.clip(h_half_initial, 0, 1)

    best_c5 = np.inf
    best_h_best = h_half_initial.copy()

    while time.time() - start_time < budget_s - 20:
        np.random.seed(seed)
        seed += 1

        # Run differential evolution with limited time
        time_de_start = time.time()
        result_de = differential_evolution(func=objective_de, bounds=bounds,
                                           strategy='best1bin', maxiter=10, popsize=15,
                                           mutation=(0.5, 1), recombination=0.7,
                                           polish=False, init='sobol', atol=1e-3,
                                           disp=False, seed=seed)
        elapsed = time.time() - time_de_start
        if elapsed > budget_s - 20 - (time.time() - start_time):
            break

        h_half_candidate = result_de.x
        c5_candidate = result_de.fun

        # Refine with local optimization
        def objective_refine(x):
            current_sum = np.sum(x)
            if abs(current_sum) < 1e-6:
                x = np.zeros_like(x) + 1e-5
            else:
                x *= target_sum_half / current_sum
            x_clipped = np.clip(x, 0, 1)
            h_full = create_full_h(x_clipped)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            return np.max(corr) * dx

        result_refine = minimize(
            fun=objective_refine,
            x0=h_half_candidate,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result_refine.success and result_refine.fun < c5_candidate:
            h_half_candidate = result_refine.x
            c5_candidate = result_refine.fun

        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h_best = h_half_candidate.copy()

    def finalize_h(h铪):
        scale = target_sum_half / np.sum(h_half) if np.sum(h_half) != 0 else 1.0
        h_scaled = h_half * scale
        h_clipped = np.clip(h_scaled, 0, 1)
        return h_clipped

    h_best = finalize_h(best_h_best)
    h_full = create_full_h(h_best)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), best_c5, n_points)
