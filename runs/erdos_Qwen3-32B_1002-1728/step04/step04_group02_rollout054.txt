                s = np.sum(particles[i])
                if s > 0:
                    particles[i] *= target_sum / s
                particles[i] = np.clip(particles[i], 0, 1)
                c5_new = compute_c5(particles[i])
                if c5_new < pbest_values[i]:
                    pbest_positions[i] = particles[i].copy()
                    pbest_values[i] = c5_new
                    if c5_new < gbest_value:
                        gbest_position = particles[i].copy()
                        gbest_value = c5_new

    h_final = gbest_position.copy()
    assert abs(np.sum(h_final) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6)), "Value bounds violated"
    final_c5 = compute_c5(h_final)
    return (h_final.tolist(), final_c5, n_points)
