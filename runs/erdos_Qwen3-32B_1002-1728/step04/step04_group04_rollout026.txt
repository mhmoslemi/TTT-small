            current_h = np.clip(current_h, 0, 1)

            # Check improvement
            new_c5 = compute_c5(current_h, dx)
            if new_c5 < current_c5:
                current_c5 = new_c5

            # Decay learning rate
            learning_rate *= decay_factor
            perturbation_scale *= perturb_decay

            best_c5 = current_c5
            best_h = current_h.copy()

    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial guess
    h_current = h_full_initial.copy()
    current_sum = np.sum(h_current)
    if abs(current_sum) < 1e-10:
        h_current = np.zeros_like(h_current)
    else:
        h_current = (h_current / current_sum) * target_sum
    h_current = np.clip(h_current, 0, 1)
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)

    np.random.seed(seed)

    def compute_c5(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_gradient(h, k_shift, n):
        grad = np.zeros(n)
        for i in range(n):
            j = i + k_shift
            if 0 <= j < n:
                grad[i] += 1 - h[j]
                grad[j] -= h[i]
        return grad

    learning_rate = 0.01
    perturbation_scale = 0.15
    perturb_decay = 0.95
    grad_decay = 0.97

    while time.time() - start_time < budget_s - 10:
        perturbed = best_h + np.random.normal(0, perturbation_scale, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if abs(total) > 1e-10:
            perturbed = (perturbed / total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
        current_h = perturbed.copy()
        current_c5 = compute_c5(current_h, dx)

        for _ in range(300):
            if time.time() - start_time >= budget_s - 10:
                break
            corr = np.correlate(current_h, 1 - current_h, mode='full')
            max_corr = np.max(corr)
            k_index = np.argmax(corr)
            k_shift = k_index - (n_points - 1)

            grad = compute_gradient(current_h, k_shift, n_points)
            current_h -= learning_rate * grad

            # Projection
            current_h = np.clip(current_h, 0, 1)
            sum_h = np.sum(current_h)
            if abs(sum_h) > 1e-10:
                current_h = (current_h / sum_h) * target_sum
                current_h = np.clip(current_h, 0, 1)

            current_c5 = compute_c5(current_h, dx)
            learning_rate *= grad_decay

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        perturbation_scale *= 0.9

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Out of bounds"
    return (best_h.tolist(), best_c5, n_points)
