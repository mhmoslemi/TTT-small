import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_full_sum = n_points / 2.0
    target_half_sum = target_full_sum / 2.0

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize symmetric h_half
    h_full_sym = create_full_h(h_full_initial[:half_n])
    current_sum = np.sum(h_full_sym)
    if current_sum != target_full_sum and current_sum != 0:
        scale = target_full_sum / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[:half_n]

    best_h_half = h_half.copy()
    best_c5 = compute_c5(h_full_sym)

    # Simulated Annealing
    np.random.seed(seed)
    T = 1.0
    alpha = 0.995  # Adjusted cooling rate
    T_min = 1e-5

    while time.time() - start_time < budget_s and T > T_min:
        # Generate perturbation
        perturbed = best_h_half + np.random.normal(0, T, half_n)
        perturbed = np.clip(perturbed, 0, 1)
