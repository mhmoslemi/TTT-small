    target_sum = n_points / 4.0  # sum(h_half) = target_sum

    def create_full_h(h_vars):
        return np.concatenate([h_vars, h_vars[::-1]])

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(x) - target_sum

    bounds_per_var = [(0.0, 1.0) for _ in range(m)]
    cons = ({'type': 'eq', 'fun': constraint_sum})

    # Initialize with symmetric average of initial values
    h_half = (h_full_initial[:m] + np.flip(h_full_initial[m:])) / 2
    total = np.sum(h_half)
    if abs(total) > 1e-8:
        h_half = (h_half / total) * target_sum
    h_half = np.clip(h_half, 0, 1)

    best_c5 = objective(h_half)
    best_h = h_half.copy()

    np.random.seed(seed)
    t_ref = time.time()

    while time.time() - start_time < budget_s - 60:
        frac = min(1.0, (time.time() - start_time) / budget_s)
        noise_scale = 0.3 * (1.0 - frac**1.2)

        # Perturbation with normal + stepwise mixture
        perturbed = h_half.copy()
        normal_pert = np.random.randn(m) * noise_scale
        step_pert = np.random.choice([-0.1, 0.1], size=m)
        perturbed = np.clip(perturbed + normal_pert + step_pert, 0, 1)

        total_pert = np.sum(perturbed)
        if total_pert > 1e-8:
            perturbed = (perturbed / total_pert) * target_sum

        # For local refinement, mix with best solution
        refine_coeff = max(0.2, np.random.rand())  # 20-100% mixing
        perturbed = np.clip(perturbed * refine_coeff + best_h * (1 - refine_coeff), 0, 1)
        perturbed = np.clip((perturbed / np.sum(perturbed)) * target_sum, 0, 1)

        # Alternate optimization algorithms and tolerances
        if best_c5 > 0.381:  # Initial phase: faster
            result = minimize(objective, perturbed, method='SLSQP',
                             bounds=bounds_per_var, constraints=cons,
                             options={'maxiter': 30, 'ftol': 1e-5})
        else:  # Post-improvement phase: higher precision
            result = minimize(objective, perturbed, method='L-BFGS-B',
                             bounds=bounds_per_var, constraints=cons,
                             options={'maxiter': 40, 'ftol': 1e-7})

        current_c5 = best_c5
        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_h_full = create_full_h(new_h)
            corr = np.correlate(new_h_full, 1 - new_h_full, 'full')
            new_c5 = np.max(corr) * dx
            if new_c5 < current_c5 - 1e-6:  # Accept only substantial improvements
                best_c5 = new_c5
                best_h = new_h

        # Rare global restart with Fourier-inspired sampling
        if np.random.rand() < 0.05 and best_c5 > 0.3809:
            freq_freq = np.random.rand(m) + freq_freq
            h_prop = np.fft.irfft(np.random.normal(size=n_points))[:m]
            h_prop = np.clip(h_prop, 0, 1)
            h_prop = (h_prop / np.sum(h_prop)) * target_sum
            h_prop = np.clip(h_prop, 0, 1)
            result = minimize(objective, h_prop, method='L-BFGS-B',
                            bounds=bounds_per_var, constraints=cons,
                            options={'maxiter': 40, 'ftol': 1e-7})
            if result.success:
                new_h = np.clip(result.x, 0, 1)
                new_h_full = create_full_h(new_h)
                corr = np.correlate(new_h_full, 1 - new_h_full, 'full')
                new_c5 = np.max(corr) * dx
                if new_c5 < current_c5 - 1e-6:
                    best_c5 = new_c5
                    best_h = new_h

    h_full = create_full_h(best_h)
    assert np.allclose(np.sum(h_full), n_points / 2.0, atol=1e-6)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), best_c5, n_points)
