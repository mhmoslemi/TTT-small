import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def project(x):
        sum_x = np.sum(x)
        if sum_x == 0:
            return x.copy()
        x_scaled = x * target_sum_half / sum_x
        x_proj = np.clip(x_scaled, 0.0, 1.0)
        return x_proj

    def objective(x):
        x_proj = project(x)
        h_full = create_full_h(x_proj)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]

    h_half_initial = (h_full_initial[:half_n] + h_full_initial[-half_n:][::-1]) / 2
    h_half_initial = project(h_half_initial)
    best_x = h_half_initial.copy()
    best_c5 = objective(best_x)

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 20:

        perturbed = np.clip(perturbed, 0, 1)
        perturbed = project(perturbed)
        result = minimize(
            fun=objective,
            x0=perturbed,

                best_c5 = new_c5
