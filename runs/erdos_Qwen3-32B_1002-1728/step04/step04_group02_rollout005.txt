import time
from numpy.fft import fft, ifft

def compute_c5(h_full):
    n = len(h_full)
    dx = 2.0 / n
    n_full = 2 * n - 1
    h_fft = fft(h_full, n_full)
    g = 1.0 - h_full
    g_fft = fft(g, n_full)
    corr_fft = h_fft * g_fft
    corr = ifft(corr_fft).real
    max_corr = np.max(corr)
    return max_corr * dx

def tournament_selection(population, c5_list, tournament_size):
    indices = np.random.choice(len(population), tournament_size)
    candidates_c5 = [c5_list[i] for i in indices]
    best_idx = indices[np.argmin(candidates_c5)]
    return population[best_idx]

def crossover(parent1, parent2, crossover_rate):
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    split_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    return child1, child2

def mutate(individual, mutation_rate, target_sum):
    mutation = np.random.rand(len(individual)) < mutation_rate
    delta = np.random.normal(0, 0.1, size=len(individual))
    individual = individual.copy()
    individual[mutation] += delta[mutation]
    individual = np.clip(individual, 0, 1)
    current_sum = individual.sum()
    if current_sum != 0:
        individual = individual * (target_sum / current_sum)
    else:
        individual = np.random.rand(len(individual)) * 0.5
        individual = np.clip(individual, 0, 1)
        individual = individual * (target_sum / individual.sum())
    return individual
