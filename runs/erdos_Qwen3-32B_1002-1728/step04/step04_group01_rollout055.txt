import numpy as np
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h to meet constraints
    h_full = h_full_initial.copy()
    current_sum = np.sum(h_full)
    if not np.allclose(current_sum, target_sum):
        h_full = (h_full / current_sum) * target_sum
    h_full = np.clip(h_full, 0, 1)
    current_best = h_full.copy()
    best_c5 = evaluate_c5(current_best, dx)

    # Simulated Annealing Phase
    np.random.seed(seed)
    sa_time_limit = budget_s - 120  # Reserve 2 mins for refinement
    while time.time() - start_time < sa_time_limit:
        # Perturb current_best with decreasing noise
        time_remaining = sa_time_limit - (time.time() - start_time)
        noise_std = max(0.01, 0.2 * (time_remaining / (sa_time_limit)))
        perturbed = current_best + np.random.normal(0, noise_std, n_points)
        perturbed = np.clip(perturbed, 0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:

        # Early escape if stagnation detected
        if time.time() - start_time > sa_time_limit - 300:
            break

    # Deterministic refinement phase
    def obj(x):
        x = np.reshape(x, (n_points,))
