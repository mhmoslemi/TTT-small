        for i in range(pop_size):
            # Select 3 distinct random indices
            idxs = np.sort(rng.choice(pop_size, 3, replace=False))
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]

            # Mutation: DE/rand/1
            mutant = a + F * (b - c)
            mutant = normalize(mutant, target_sum)

            # Crossover (binomial)
            cross_point = rng.integers(0, n_points)
            trial = np.empty(n_points, dtype=np.float64)
            for k in range(n_points):
                if rng.random() < CR or k == cross_point:
                    trial[k] = mutant[k]
                else:
                    trial[k] = population[i][k]

            # Clip and normalize for safety
            trial = normalize(trial, target_sum)
            trial_score = evaluate_candidate(trial, dx)

            # Selection
            if trial_score < fitness[i]:
                population[i] = trial.copy()
                fitness[i] = trial_score
                if trial_score < best_c5:
                    best_c5 = trial_score
                    best_h = trial.copy()
        generation += 1

    # Final validation
    assert abs(population[best_idx].sum() - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((population[best_idx] >= -1e-6) & (population[best_idx] <= 1.0 + 1e-6)), "Value bounds violated"

    return (best_h.tolist(), best_c5, n_points)
