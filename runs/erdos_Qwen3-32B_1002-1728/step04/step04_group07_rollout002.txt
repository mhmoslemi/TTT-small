import numpy as np
import time
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_var):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_var
        h_full[-half_n:] = h_var[::-1]
        return h_full

    def compute_c5(h_full):
        correlations = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(correlations) * dx

    def feasibility_repair(positions):
        repaired = np.clip(positions, 0, 1)
        non_zeros = np.sum(repaired, axis=1)
        mask = (non_zeros > 1e-8)
        repaired[mask] = repaired[mask] * (target_sum_half / non_zeros[mask, None])
        return repaired

    # Initialize symmetric and normalized version
    h_sym = (h_initial[:half_n] + h_initial[-half_n:][::-1]) / 2
    h_sym = h_sym / np.clip(np.sum(h_sym), 1e-6, np.inf) * target_sum_half
    h_sym = np.clip(h_sym, 0, 1)

    # Particle Swarm Optimization
    swarmsize = 30
    max_iter = 40
    w = 0.7
    c1, c2 = 1.5, 1.5

    np.random.seed(seed)
    max_psotime = 400
    particles_pos = feasibility_repair(
        h_sym[None, :] + np.random.uniform(-0.2, 0.2, (swarmsize, half_n)))
    particles_vel = np.random.uniform(-0.05, 0.05, (swarmsize, half_n))
    particles_pbest = particles_pos.copy()
    pbest_scores = np.array([compute_c5(create_full_h(p)) for p in particles_pos])
    gbest_idx = np.argmin(pbest_scores)
    gbest_pos = particles_pos[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]

    t1 = time.time()
    for iter_p in range(max_iter):
        if time.time() - start_time > budget_s - 30:
            break
        for i in range(swarmsize):
            r1, r2 = np.random.rand(2)
            particles_vel[i] = (w * particles_vel[i] +
                               c1 * r1 * (particles_pbest[i] - particles_pos[i]) +
                               c2 * r2 * (gbest_pos - particles_pos[i]))
            particles_pos[i] = np.clip(particles_pos[i] + particles_vel[i], 0, 1)
            sum_p = np.sum(particles_pos[i])
            if sum_p > 1e-8:
                particles_pos[i] = particles_pos[i] * (target_sum_half / sum_p)
            particles_pos[i] = np.clip(particles_pos[i], 0, 1)
        # Evaluate and update pbest/gbest
        current_scores = np.array([compute_c5(create_full_h(p)) for p in particles_pos])
        improved = current_scores < pbest_scores
        particles_pbest[improved] = particles_pos[improved].copy()
        pbest_scores[improved] = current_scores[improved]
        new_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[new_gbest_idx] < gbest_score:
            gbest_pos = particles_pbest[new_gbest_idx].copy()
            gbest_score = pbest_scores[new_gbest_idx]
    t1 = time.time() - t1

    # Final optimization with CVXPY (SQP)
    h_opt = cp.Variable(half_n)
    constraints = [0 <= h_opt, h_opt <= 1, cp.sum(h_opt) == target_sum_half]
    corr_kernel = create_full_h(h_opt)
    correlations = np.correlate(corr_kernel, 1 - corr_kernel, mode='full')
    max_overlap = cp.Maximize(cp.max(correlations) * dx)
    problem = cp.Problem(max_overlap, constraints)

    assert np.all(np.abs(final_h) <= 1.0 + 1e-6)
    
    return (final_h.tolist(), final_c5, n_points)
