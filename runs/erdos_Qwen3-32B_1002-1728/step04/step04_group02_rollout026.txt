import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    n_particles = 50

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    np.random.seed(seed)
    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    positions[0] = h_initial.copy()

    for i in range(1, n_particles):
        perturbed = h_initial + np.random.uniform(-0.1, 0.1, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = perturbed.sum()
        if perturbed_sum != 0:
            perturbed = (perturbed / perturbed_sum) * (n_points / 2.0)
        positions[i] = perturbed

    pbest_positions = positions.copy()
    pbest_values = np.array([compute_c5(pos) for pos in positions])
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    best_c5 = pbest_values[gbest_idx]

    w = 0.7
    c1 = 1.0
    c2 = 1.0
    velocity_clamp = 0.5

    no_improvement = 0
    max_no_improvement = 200
    prev_best_c5 = best_c5

    while time.time() - start_time < budget_s - 30:
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            cognitive = c1 * r1 * (pbest_positions[i] - positions[i])
            social = c2 * r2 * (gbest_position - positions[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -velocity_clamp, velocity_clamp)
            new_position = positions[i] + velocities[i]
            new_position = np.clip(new_position, 0, 1)
            new_sum = new_position.sum()
            if new_sum != 0:
                new_position = (new_position / new_sum) * (n_points / 2.0)
            positions[i] = new_position
            new_value = compute_c5(new_position)
            if new_value < pbest_values[i]:
                pbest_positions[i] = new_position.copy()
                pbest_values[i] = new_value
                if new_value < best_c5:
                    gbest_position = new_position.copy()
                    best_c5 = new_value
