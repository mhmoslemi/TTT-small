            cognitive = c1 * r1 * (pbest_positions[i] - positions[i])
            social = c2 * r2 * (gbest_position - positions[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            positions[i] += velocities[i]
            positions[i] = process_position(positions[i])
            current_values[i] = evaluate(positions[i])
            if current_values[i] < pbest_values[i]:
                pbest_values[i] = current_values[i]
                pbest_positions[i] = positions[i].copy()

        current_gbest_index = np.argmin(pbest_values)
        if pbest_values[current_gbest_index] < gbest_value:
            gbest_value = pbest_values[current_gbest_index]
            gbest_position = pbest_positions[current_gbest_index].copy()

    best_h_full = create_full_h(gbest_position)
    assert abs(np.sum(best_h_full) - n_points / 2) < 1e-6
    assert np.all(best_h_full >= -1e-6) and np.all(best_h_full <= 1 + 1e-6)
    return (best_h_full.tolist(), gbest_value, n_points)
