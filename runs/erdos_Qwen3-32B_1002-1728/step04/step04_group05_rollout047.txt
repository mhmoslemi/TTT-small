        for i in range(half_n):
            j = n_points - 1 - i
            h_full[i] = h_full[j] = y[i]
        return h_full

    def evaluate_candidate(y):
        h_full = create_full_h(y)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (n_points / 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_y = h_full_sym[:half_n].copy()

    def project(y):
        y_proj = np.clip(y, 0, 1)
        s = np.sum(y_proj)
        if s == 0:
            y_proj[0] = 0.01
            s = 0.01
        y_proj = y_proj * (target_sum_half / s)
        return y_proj

    def perturb(y, mag):
        delta = np.random.randn(len(y)) * mag
        new_y = y + delta
        return project(new_y)

    def compute_gradient(y):
        grad = np.zeros_like(y)
        eps = 1e-5
        for i in range(len(y)):
            y_plus = y.copy()
            y_plus[i] += eps
            y_minus = y.copy()
            y_minus[i] -= eps
            grad[i] = (evaluate_candidate(y_plus) - evaluate_candidate(y_minus)) / (2 * eps)
        return grad

    np.random.seed(seed)
    current_y = initial_y.copy().astype('float64')
    best_y = current_y.copy()
    best_c5 = evaluate_candidate(best_y)
    learning_rate = 1e-3
    perturbation_mag = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    tabu_hashes = set()
    iteration = 0

    while time.time() - start_time < budget_s - 10:
        if iteration % 100 == 0:
            time_left = budget_s - 10 - (time.time() - start_time)
            if time_left < 1:
                break

        grad = compute_gradient(current_y)
        current_y -= learning_rate * grad
        current_y = project(current_y)
        current_c5 = evaluate_candidate(current_y)

        if current_c5 < best_c5:
            best_y = current_y.copy()
            best_c5 = current_c5
            no_improvement_count = 0
            h = hash(current_y.tobytes())
            if h not in tabu_hashes:
                tabu_hashes.add(h)
                if len(tabu_hashes) > 50:
                    tabu_hashes.pop()
        else:
            no_improvement_count += 1

        if no_improvement_count > max_no_improvement:
            perturbation_mag = min(perturbation_mag * 1.5, 0.5)
            current_y = perturb(current_y, perturbation_mag)
            no_improvement_count = 0

        learning_rate = max(learning_rate * 0.999, 1e-6)
        iteration += 1

    h_full = create_full_h(best_y)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0+1e-6))
    assert abs(np.sum(h_full) - n_points/2.0) < 1e-6
    
    return (h_full.tolist(), best_c5, n_points)
