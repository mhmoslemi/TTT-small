    # Normalize initial guess to meet constraint
    h_initial_sum = np.sum(h_initial)
    if h_initial_sum > 1e-10:
        h_initial = (h_initial / h_initial_sum) * target_sum
        h_initial = np.clip(h_initial, 0, 1)

    # Precompute all shifts and their A_k matrices
    matrices = []
    for k in range(-n_points + 1, n_points):
        A = np.zeros((n_points, n_points))
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                A[i, j] = 1.0
        matrices.append(A)

    # Define optimization variables
    h = cp.Variable(n_points)
    t = cp.Variable()
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    # Constraints: sum(h) = target_sum, 0 ≤ h ≤ 1
    constraints = [
        cp.sum(h) == target_sum,
        h >= 0,
        h <= 1
    ]
    
    # Add quadratic constraints
    for A in matrices:
        quadratic_expr = cp.quad_form(h, A)
        rhs = target_sum - t / dx
        constraints.append(quadratic_expr <= rhs)

    # Create problem
    problem = cp.Problem(objective, constraints)

    # Set up warm start
    h.value = h_initial
    t.value = evaluate_erdos_solution(h_initial)

    # Try to solve with SCIP within time budget
    if time.time() - start_time < budget_s:
        try:
            problem.solve(solver=cp.SCIP, verbose=False, 
                         warm_start=True,
                         maximum_iterations=500,
                         time_limit=budget_s - (time.time() - start_time) - 10)
        except Exception:
            pass

    # Ensure solution meets constraints
    if h.value is None:
        best_h = h_initial.copy()
    else:
        best_h = np.clip(h.value, 0, 1)
    total_h = np.sum(best_h)
    if abs(total_h - target_sum) > 1e-6:
        best_h = (best_h / total_h) * target_sum
        best_h = np.clip(best_h, 0, 1)

    # Evaluate final C5 bound
    c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), c5, n_points)
