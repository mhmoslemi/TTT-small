from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def repair_individual(individual, target_sum):
        individual = np.clip(individual, 0, 1)
        sum_current = np.sum(individual)
        if sum_current == 0:
            return individual * 0.0
        scaled = individual * (target_sum / sum_current)
        return np.clip(scaled, 0, 1)

    def generate_random_h():
        h = np.random.rand(n_points) * 0.5  # To reduce variance?
        h = h * (target_sum / h.sum()) if h.sum() > 0 else np.zeros(n_points)
        return repair_individual(h, target_sum)

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    h_full_initial = repair_individual(h_full_initial, target_sum)
    population = [h_full_initial]
    for _ in range(49):
        population.append(generate_random_h())

    best_h = h_full_initial.copy()
    best_c5 = compute_c5(best_h)

    tournament_size = 3
    mutation_rate = 0.1
    population_size = 50

    while time.time() - start_time < budget_s - 60:
        # Evaluate all population once
        fitnesses = [compute_c5(ind) for ind in population]
        # Select elites (top 5)
        elite_indices = np.argsort(fitnesses)[:5]
        elites = [population[i].copy() for i in elite_indices]
        current_best_c5 = min(fitnesses)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[np.argmin(fitnesses)].copy()

        # Create new population
        new_population = elites.copy()

        while len(new_population) < population_size:
            # Select two parents via tournament selection
            def select_parent():
                indices = np.random.choice(len(population), tournament_size)
                best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
                return population[best_idx].copy()

            parent1 = select_parent()
            parent2 = select_parent()

            # One-point crossover
            split = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:split], parent2[split:]])

            # Mutation
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, mutation_rate, n_points)
                child += noise

            # Repair the child
            child = repair_individual(child, target_sum)

            # Local refinement: minimize the c5 value
            def refine_obj(x):
                h = np.clip(x, 0, 1).flatten()
                h_repaired = repair_individual(h, target_sum)
                return compute_c5(h_repaired)

            # Initial guess is child
            x0 = child.copy()
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            # Constraints: sum(x) = target_sum
            cons = ({
                'type': 'eq',
                'fun': lambda x: np.sum(x) - target_sum
            },)

            result = minimize(refine_obj, x0, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 30, 'ftol': 1e-6})
            if result.success:
                refined_h = repair_individual(result.x, target_sum)
            else:
                refined_h = child.copy()

            new_population.append(refined_h)

        population = new_population

    h_full = best_h
    c5_bound = compute_c5(h_full)

    assert abs(np.sum(h_full) - target_sum) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), c5_bound, n_points)
