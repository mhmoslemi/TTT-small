        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    current_pop_c5 = np.array([evaluate(h) for h in population])
    best_mask = np.argmin(current_pop_c5)
    best_c5 = current_pop_c5[best_mask]
    best_h_half = population[best_mask].copy()

    while time.time() - start_time < budget_s - 30:
        for i in range(population_size):
            indices = rng.choice(population_size, 3, replace=False)
            a, b, c = population[indices]
            mutant = a + F * (b - c)
            trial = population[i].copy()
            crossover_points = rng.random(half_n) < CR
            trial[crossover_points] = mutant[crossover_points]
            trial_sum = trial.sum()
            if trial_sum != 0:
                trial = target_sum_half / trial_sum * trial
            trial = np.clip(trial, 0.0, 1.0)
            c5 = evaluate(trial)
            if c5 < best_c5:
                best_c5 = c5
                best_h_half[:] = trial
            if c5 < evaluate(population[i]):
                population[i] = trial
        if (time.time() - start_time) < budget_s - 30:
            current_pop_c5 = np.array([evaluate(h) for h in population])
            sorted_indices = np.argsort(current_pop_c5)
            for i in sorted_indices[:5]:
                h = population[i].copy()
                def obj(x):
                    x = np.clip(x, 0, 1)
                    h_full = np.zeros(n_points)
                    h_full[:half_n] = x
                    h_full[-half_n:] = x[::-1]
                    corr = np.correlate(h_full, 1 - h_full, mode='full')
                    return np.max(np.abs(corr)) * dx
                bounds = [(0, 1)] * half_n
                constraints = ({'type': 'eq', 'fun': lambda x: x.sum() - target_sum_half})
                result = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 25})
                if result.success and result.fun < best_c5:
                    optimized = np.clip(result.x, 0, 1)
                    population[i] = optimized
                    best_c5 = result.fun
                    best_h_half[:] = optimized

    h_full = np.zeros(n_points)
    h_full[:half_n] = best_h_half
    h_full[-half_n:] = best_h_half[::-1]
    h_full = np.clip(h_full, 0, 1)
    assert abs(h_full.sum() - n_points / 2) < 1e-6, "Integral constraint violated"
    return (h_full.tolist(), best_c5, n_points)
