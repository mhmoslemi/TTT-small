import numpy as np
from scipy.optimize import basinhopping, Bounds, minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        h_clipped = np.clip(h, 0, 1)
        h_sum = h_clipped.sum()
        if h_sum == 0:
            return np.inf
        h_norm = h_clipped * (target_sum / h_sum)
        n = n_points
        h_fft = fft(h_norm, n=2*n)
        s_fft = fft(1 - h_norm, n=2*n)
        cross_corr = ifft(h_fft * s_fft.conj()).real
        return np.max(cross_corr) * dx

    bounds = Bounds(0.0, 1.0)
    h_flat = h_initial.flatten()
    h_sum = h_initial.sum()
    h_flat = np.clip(h_flat * (target_sum / h_sum), 0, 1)

    def callback(x, f, context):
        return time.time() - start_time > budget_s - 10

    result_bh = basinhopping(
        objective, 
        h_flat, 
        niter=30, 
        T=2.0, 
        stepsize=0.2, 
        minimizer_kwargs={
            'method': 'L-BFGS-B',
            'bounds': bounds
        },
        callback=callback, 
        seed=seed
    )

    best_h = result_bh.x.reshape(n_points)
    best_val = result_bh.fun

    # Post-refinement with projection
    def project(x):
        x = np.clip(x, 0, 1).copy()
        x_sum = x.sum()
        return x * (target_sum / x_sum) if x_sum > 1e-8 else x

    result_refine = minimize(
        objective, 
        best_h, 
        method='SLSQP', 
        bounds=bounds,
        jac='3-point',
        options={'maxiter': 50, 'ftol': 1e-7}
    )
    if result_refine.success and result_refine.fun < best_val:
        best_h = project(result_refine.x)
        best_val = objective(best_h)

    # Final validation
    assert np.allclose(best_h.sum(), target_sum, atol=1e-6)
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    
    return (best_h.tolist(), best_val, n_points)
