            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            particles[i] += velocities[i]
            particles[i] = repair(particles[i])
            new_corr = np.correlate(particles[i], 1 - particles[i], mode='full') * dx
            new_c5 = np.max(new_corr)
            if new_c5 < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = new_c5
                if new_c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = new_c5
        w = max(0.4, 0.9 - (time.time() - start_time) / (budget_s * 0.9))

    assert abs(np.sum(global_best) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((global_best >= -1e-6) & (global_best <= 1 + 1e-6)), "Value bounds violated"

    return (global_best.tolist(), global_best_c5, n_points)
