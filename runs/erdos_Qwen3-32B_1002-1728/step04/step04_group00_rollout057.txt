def create_full_h(h_half, n_points):
    half_n = len(h_half)
    h_full = np.zeros(n_points)
    h_full[:half_n] = h_half
    h_full[n_points - half_n:] = h_half[::-1]
    return h_full

def evaluate_individual(h_half, n_points, dx):
    h_full = create_full_h(h_half, n_points)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
