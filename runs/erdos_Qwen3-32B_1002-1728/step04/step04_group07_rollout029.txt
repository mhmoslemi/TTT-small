def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(h_half) == n_points / 4 exactly

        h_half = np.array(h_half_candidate)
        sum_h = np.sum(h_half)
