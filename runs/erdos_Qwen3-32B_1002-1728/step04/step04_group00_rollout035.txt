            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - (target_sum / 2.0)}]

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - target_sum) > 1e-6 and current_sum != 0:
        scale = target_sum / current_sum
        h_full_sym = h_full_sym * scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    n_attempt = 0

    while time.time() - start_time < budget_s - 60:
        n_attempt += 1
        frac_time = (budget_s - (budget_s - (time.time() - start_time))) / budget_s
        noise_scale = 0.2 * (1 - frac_time ** 0.3)  # Slower decay

        for _ in range(2):  # Increased to 2 perturbations
            seed = (seed + 1) * 42
            rng = np.random.RandomState(seed)
            perturbed = best_h_half + rng.normal(0, noise_scale, half_n)
            perturbed = np.clip(perturbed, 0, 1)
            total = np.sum(perturbed)
            if total != 0:
                perturbed = (perturbed / total) * (target_sum / 2.0)
            perturbed = np.clip(perturbed, 0, 1)

            methods = ['Nelder-Mead', 'L-BFGS-B'] if n_attempt % 2 == 0 else ['SLSQP', 'BFGS']
            for method in methods:
                result = minimize(
                    fun=objective,
                    x0=perturbed,
                    method=method,
                    bounds=bounds_per_var,
                    constraints=cons,
                    options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
                )
                if result.success and result.fun < best_c5:
                    new_h_half = np.clip(result.x, 0, 1)
                    h_full = create_full_h(new_h_half)
                    if abs(np.sum(h_full) - target_sum) < 1e-6:
                        corr = np.correlate(h_full, 1 - h_full, mode='full')
                        new_c5 = np.max(corr) * dx
                        if new_c5 < best_c5:
                            best_h_half = new_h_half
                            best_c5 = new_c5
                            break  # Break early if improvement found

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - target_sum) < 1e-6, f"Violation in integral constraint: {abs(np.sum(h_full) - target_sum)}"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "h out of bounds"
    return (h_full.tolist(), best_c5, n_points)
