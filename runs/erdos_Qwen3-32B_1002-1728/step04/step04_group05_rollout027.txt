    def create_full_h(y):
        h = np.zeros(n_points, dtype=np.float64)
        h[:half_n] = y
        h[half_n:] = y[::-1]
        return h

    def normalize_y(y):
        s = np.sum(y)
        if s == 0:
            return y
        y = (y / s) * target_sum
        y = np.clip(y, 0.0, 1.0)
        return y

    def evaluate_c5(y):
        h = create_full_h(y)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)

    # Symmetrize initial
    y_initial = h_full_initial[:half_n].copy()
    y_initial = normalize_y(y_initial)
    best_y = y_initial.copy()
    best_c5 = evaluate_c5(y_initial)

    # Setup optimization parameters
    np.random.seed(seed)
    learning_rate = 1e-3
    perturbation_mag = 0.02
    tabu_states = dict()
    MAX_TABU = 100

    def state_key(y, precision=1e-4):
        rounded = np.round(y / precision).astype(int)
        return tuple(rounded)

    def compute_gradient(y, eps=1e-6):
        g = np.zeros_like(y)
        base = evaluate_c5(y)
        for i in range(len(y)):
            y_up = y.copy()
            y_up[i] += eps
            y_up = normalize_y(y_up)
            up = evaluate_c5(y_up)

            y_down = y.copy()
            y_down[i] -= eps
            y_down = normalize_y(y_down)
            down = evaluate_c5(y_down)

            g[i] = (up - down) / (2 * eps)
        return g

    iteration = 0
    no_improvement = 0
    MAX_NO_IMPROVE = 200
    while time.time() - start_time < budget_s - 10:

        # Compute gradient and take step
