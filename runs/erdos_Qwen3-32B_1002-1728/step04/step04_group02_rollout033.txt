        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    h_full_sym *= target_sum_full / np.sum(h_full_sym)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_initial = h_full_sym[:half_n]

    population_size = 50
    population = [h_half_initial.copy()]
    for _ in range(population_size - 1):
        new_h = h_half_initial.copy()
        perturbation = np.random.normal(0, 0.1, size=half_n)
        new_h = np.clip(new_h + perturbation, 0, 1)
        new_h *= target_sum_half / np.sum(new_h)
        population.append(new_h)

    def compute_c5(h_half):
        h_full = create_full_h(h_half)
        pad_length = 2 * n_points - 1
        h_fft = fft.fft(h_full, pad_length)
        one_minus_h_fft = fft.fft(1.0 - h_full, pad_length)
        corr_fft = h_fft * np.conj(one_minus_h_fft)
        corr = fft.ifft(corr_fft).real
        return np.max(corr) * dx

    current_c5s = np.array([compute_c5(h) for h in population])
    best_index = np.argmin(current_c5s)
    best_h_half = population[best_index].copy()
    best_c5 = current_c5s[best_index]

    np.random.seed(seed)
    mutation_rate = 0.1
    crossover_rate = 0.7
    tournament_size = 3
    elite_frac = 0.1
    elites = int(population_size * elite_frac)

    while time.time() - start_time < budget_s - 30:
        def tournament():
            indices = np.random.randint(0, len(population), tournament_size)
            fittest = min(indices, key=lambda i: current_c5s[i])
            return population[fittest].copy()

        new_population = []
        elite_indices = np.argsort(current_c5s)[:elites]
        for idx in elite_indices:
            new_population.append(population[idx].copy())

        while len(new_population) < population_size:
            parent1 = tournament()
            parent2 = tournament()
            child = parent1.copy()
            if np.random.rand() < crossover_rate:
                cross = np.random.randint(1, half_n)
                child[:cross] = parent1[:cross]
                child[cross:] = parent2[cross:]
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.choice(half_n, size=int(half_n * 0.1), replace=False)
                for i in mut_idx:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.1), 0, 1)
            child *= target_sum_half / np.sum(child)

            def refine_child(x):
                h_full = create_full_h(x)
                corr = np.fft.ifft(np.fft.fft(h_full) * np.conj(np.fft.fft(1.0 - h_full))).real
                return np.max(corr) * dx

            bounds = [(0, 1)] * half_n
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
            res = minimize(refine_child, x0=child, bounds=bounds, constraints=cons,
                          method='SLSQP', options={'maxiter': 20})
            if res.success:
                optimized = res.x.clip(0, 1) * (target_sum_half / np.sum(res.x.clip(0, 1)))
            else:
                optimized = child
            new_population.append(optimized)

        population = new_population
        current_c5s = np.array([compute_c5(h) for h in population])
        best_index = np.argmin(current_c5s)
        if current_c5s[best_index] < best_c5:
            best_c5 = current_c5s[best_index]
            best_h_half = population[best_index].copy()

    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - target_sum_full) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    return (h_full.tolist(), best_c5, n_points)
