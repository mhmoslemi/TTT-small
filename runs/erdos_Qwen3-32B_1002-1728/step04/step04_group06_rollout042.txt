            # DE mutation: select 3 random indices
            idxs = rng.integers(0, population_size, 3)
            a, b, c = population[idxs]
            # Mutate
            mutant = a + F * (b - c)
            # Crossover
            cross_points = rng.random(half_n) < CR
            if not np.any(cross_points):
                cross_points[0] = True  # Ensure at least one parameter is taken from mutant
            trial = np.where(cross_points, mutant, population[j])
            # Evaluate trial
            trial_normalized = normalize_h_half(trial)
            trial_c5 = evaluate_candidate(trial_normalized)
            # Selection
            if trial_c5 < evaluate_candidate(population[j]):
                population[j] = trial_normalized

        generations += 1
        # Evaluate current population
        fitness_scores = np.array([evaluate_candidate(population[i]) for i in range(population_size)])
        min_idx = np.argmin(fitness_scores)
        current_best_c5 = fitness_scores[min_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h_half = population[min_idx].copy()

        # Run QP refinement on top 5% of candidates every 10 generations
        if generations % 10 == 0:
            top_idx = np.argsort(fitness_scores)[:int(population_size * 0.05)]
            for idx in top_idx:
                h_candidate = population[idx].copy()
                # Compute current overlap for all k to find k_max
                h_full = create_symmetric_h_full(h_candidate)
                corr = np.correlate(h_full, 1.0 - h_full, mode='full')

                k_max = k_values[max_k_idx]
                # Define objective function for this k_max
                def objective_qp(x):

                    h_full_qp = create_symmetric_h_full(h_half)
                    i_list = np.arange(n_points)
                    i_k_list = (i_list + k_max) % n_points
                    overlap = np.sum(h_full_qp[i_list] * (1.0 - h_full_qp[i_k_list]))
                    return overlap * dx  # align with C5

                # Bounds and constraint
                target_sum = target_sum_half
                bounds = [(0.0, 1.0) for _ in range(half_n)]
                constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
                # Initial guess
                x0_qp = h_candidate
                # Run optimize
                result = scipy.optimize.minimize(objective_qp, x0_qp, method='SLSQP', bounds=bounds, constraints=constraints)
                if result.success:

                        refined_h = np.clip(refined_h, 0.0, 1.0)

    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)

    target_sum_half = n_points / 4.0  # sum(h_half) = target_sum_half
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def normalize_h_half(h_half):
        current_sum = np.sum(h_half)
        if current_sum == 0:
            h_half = np.full_like(h_half, target_sum_half / half_n)
        else:
            scale = target_sum_half / current_sum
            h_half = h_half * scale
        return np.clip(h_half, 0.0, 1.0)

    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (n_points / 2.0) / current_sum
    h_full_sym = np.clip(h_full_sym, 0.0, 1.0)
    h_half_initial = h_full_sym[:half_n].copy()

    def evaluate_candidate(h_half_candidate):
        h_full = create_symmetric_h_full(h_half_candidate)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    population_size = 50
    rng = default_rng(seed)
    population = np.zeros((population_size, half_n))
    population[0] = h_half_initial.copy()
    for i in range(1, population_size):
        perturb = rng.normal(0.0, 0.1, half_n)
        candidate = h_half_initial + perturb
        population[i] = normalize_h_half(candidate)

    best_c5 = evaluate_candidate(population[0])
    best_h_half = population[0].copy()

    F = 0.7  # DE mutation factor
    CR = 0.9  # DE crossover rate

    while time.time() - start_time < budget_s - 60:

            idxs = rng.integers(0, population_size, 3)
            a, b, c = population[idxs]
            mutant = a + F * (b - c)
            cross_points = rng.random(half_n) < CR
            if not np.any(cross_points):
                cross_points[0] = True
            trial = np.where(cross_points, mutant, population[j])
            # Selection
            trial_normalized = normalize_h_half(trial)
            if evaluate_candidate(trial_normalized) < evaluate_candidate(population[j]):
                population[j] = trial_normalized

        # Evaluate and track best
        fitness_scores = np.array([evaluate_candidate(population[i]) for i in range(population_size)])
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_c5:
            best_c5 = fitness_scores[min_idx]
            best_h_half = population[min_idx].copy()

        # QP refinement on top 5% every 25 generations
        if np.random.rand() < 0.05 and time.time() - start_time < budget_s - 60:
            for idx in np.argsort(fitness_scores)[:5]:
                h_candidate = population[idx].copy()
                h_full = create_symmetric_h_full(h_candidate)
                corr = np.correlate(h_full, 1-h_full, mode='full')
                max_k = np.argmax(np.abs(corr))
                def objective_qp(x):
                    h_half_qp = x
                    h_full_qp = create_symmetric_h_full(h_half_qp)
                    overlap = np.sum(h_full_qp[(np.arange(n_points) + max_k) % n_points] * 
                                   (1 - h_full_qp[np.arange(n_points) * 0 - 18]))
                    return np.sum(h_full_qp * (1 - h_full_qp[(np.arange(n_points) + max_k) % n_points])) * dx
                # Constraint and bounds
                cons = [{'type':'eq','fun':lambda x: np.sum(x) - target_sum_half}]
                result = scipy.optimize.minimize(objective_qp, h_candidate, bounds=[(0,1)]*half_n, 
                                                 constraints=cons, method='SLSQP')
                if result.success:

                    refined_h = np.clip(refined_h, 0, 1)
                    # Normalize
