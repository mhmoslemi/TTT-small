import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    
    # Enforce symmetry and normalization
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_initial[i] = h_full_initial[j] = avg
    current_sum = np.sum(h_full_initial)
    if current_sum != 0:
        h_full_initial *= (n_points / 2.0) / current_sum
    h_full_initial = np.clip(h_full_initial, 0, 1)
    y_initial = h_full_initial[:half_n]

    def create_h(y):
        h = np.zeros(n_points)
        for i in range(half_n):
            h[i] = h[n_points - 1 - i] = y[i]
        return h

    def objective(y):
        y = np.clip(y, 0, 1)
        target_sum = n_points / 4.0
        current_sum = y.sum()
        if current_sum == 0:
            return float('inf')
        y = (y * target_sum / current_sum).clip(0, 1)
        h = create_h(y)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1)] * half_n
    best_score = float('inf')
    best_y = y_initial.copy()

    np.random.seed(seed)
    
    # Adaptive restart strategy
    n_restarts = 0
    while time.time() - start_time < budget_s - 20:

        if n_restarts > 1:
            # Perturb from best solution found so far

            current_y = best_y + perturb
        else:
            current_y = y_initial.copy() + np.random.normal(0, 0.02, size=half_n)
        
        # Run optimization with early exit
        stop = {'stop': False}
        def callback(x):
            if time.time() - start_time > budget_s - 10:
                stop['stop'] = True
            return stop['stop']
        
        result = minimize(
            objective,
            current_y,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-6},
            callback=callback
