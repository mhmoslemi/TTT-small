import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Sum of first half

    def create_full_h(y):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = y[i]
        return h_full

    # Symmetrize and normalize initial h
    h_full_sym = h_full_initial.copy()

        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = h_full_sym.sum()
    if current_sum != 0:
        h_full_sym *= (n_points / 2.0) / current_sum
    h_full_sym = np.clip(h_full_sym, 0, 1)
    y_initial = h_full_sym[:half_n].copy()

    # Objective function to minimize
    def objective(y_flat):
        y = y_flat.reshape(half_n)
        y = np.clip(y, 0, 1)
        current_sum_y = y.sum()
        if current_sum_y == 0:
            scaled_y = y.copy()
        else:
            scaling = target_sum_half / current_sum_y
            scaled_y = y * scaling
            scaled_y = np.clip(scaled_y, 0, 1)
        h_full = create_full_h(scaled_y)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Bounds for each y element
    bounds = [(0, 1) for _ in range(half_n)]

    np.random.seed(seed)
    best_obj = np.inf
    best_y = y_initial.copy()

    # Adaptive number of restarts based on time
    max_restart_attempts = 25
    restarts = 0
    while time.time() - start_time < budget_s - 10 and restarts < max_restart_attempts:

        # Generate perturbed initial y

        y_perturbed = y_initial.copy() + perturbation

        # Run optimization

                       x0=y_perturbed,
                       method='L-BFGS-B', 
                       bounds=bounds,
                       options={'ftol': 1e-8, 'maxiter': 100})
        
        if res.fun < best_obj:
            best_obj = res.fun
            best_y = res.x.copy()
