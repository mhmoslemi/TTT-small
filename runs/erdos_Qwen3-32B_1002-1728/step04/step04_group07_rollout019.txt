import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Symmetric initialization
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]

    # Particle Swarm Optimization
    np.random.seed(seed)
    n_particles = 30
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter_pso = 1000

    objective_values = np.array([objective(p) for p in positions.copy()])
    personal_best = positions.copy()
    personal_best_values = objective_values.copy()
    gbest_idx = np.argmin(personal_best_values)
    gbest = personal_best[gbest_idx].copy()
    gbest_val = personal_best_values[gbest_idx]

    def repair_position(p):
        total = np.sum(p)
        if abs(total - target_sum_half) > 1e-8 and total != 0:
            p = p * (target_sum_half / total)
        p = np.clip(p, 0, 1)
        return p

    # Initialize to valid positions
    for i in range(n_particles):
        positions[i] = repair_position(positions[i])
    objective_values = np.array([objective(p) for p in positions])
    personal_best = positions.copy()
    personal_best_values = objective_values.copy()
    gbest_idx = np.argmin(personal_best_values)
    gbest = personal_best[gbest_idx].copy()
    gbest_val = personal_best_values[gbest_idx]

            ov = objective(positions[i])
            if ov < personal_best_values[i]:

        new_gbest_idx = np.argmin(personal_best_values)
        if personal_best_values[new_gbest_idx] < gbest_val:
