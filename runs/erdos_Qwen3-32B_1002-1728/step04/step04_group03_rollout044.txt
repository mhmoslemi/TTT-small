import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Generate quaternary base pattern
    base_pattern = np.array([0, 3, 3, 0, 1, 2, 2, 1])
    repeats = n_points // len(base_pattern)
    remainder = n_points % len(base_pattern)
    h_quat = np.tile(base_pattern, repeats + 1)[:n_points] / 3.0
    
    def objective_fft(h_vars):
        h_full = np.clip(h_vars, 0, 1)
        n = len(h_full)
        h_fft = np.fft.fft(h_full, n=2 * n)
        s_fft = np.fft.fft(1.0 - h_full, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    methods = ['SLSQP', 'L-BFGS-B', 'TNC']
    best_h = h_quat.copy()
    best_c5 = objective_fft(best_h)

    np.random.seed(seed)
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 30:
        # Perturb current best
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed = best_h + perturb
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if not np.isclose(total, 0):
            perturbed = perturbed * (target_sum / total)
        perturbed = np.clip(perturbed, 0, 1)

        for method in methods:
            if time.time() - start_time >= budget_s - 30:
                break
            result = minimize(
                fun=objective_fft,
                x0=perturbed,
                method=method,
                bounds=bounds,
                constraints=cons if method != 'L-BFGS-B' else (),
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            if result.success:
                new_h = result.x
                if np.all((new_h >= 0) & (new_h <= 1)) and np.isclose(np.sum(new_h), target_sum, atol=1e-6):
                    new_c5 = result.fun
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5

    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
