    def objective(y_half):
        y = np.array(y_half)
        sum_y = np.sum(y)
        if sum_y == 0:

            h_full = create_full_h(np.zeros(half_n))
        else:
            normalized = y * (target_sum / sum_y)
            y_clipped = np.clip(normalized, 0, 1)
            h_full = create_full_h(y_clipped)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Symmetrize initial_h_values and extract initial y_half
    h_full_sym = initial_h_values.copy()

        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_sym[i] + h_full_sym[j]) / 2
    current_sum = np.sum(h_full_sym)

    h_full_sym = np.clip(h_full_sym, 0, 1)

    # Run optimization
    bounds = [(0, 1)] * half_n  # Each element of y is bounded by [0,1]
    result = minimize(objective, x0=initial_y_half, method='L-BFGS-B', bounds=bounds, options={'maxiter': 1000})
    best_y = result.x

    # Post processing
    # build full h from best_y
    sum_y = np.sum(best_y)
