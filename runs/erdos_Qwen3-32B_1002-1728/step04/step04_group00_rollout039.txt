def run(seed=42, budget_s=1100, **kwargs):

    h_half_init = np.clip(h_half_init, 0, 1)

    population_size = 50

    mutation_decay = 0.998
    np.random.seed(seed)
    random.seed(seed)

    # Initialize population around h_half_init
    population = [h_half_init.copy() + np.random.normal(0, 0.1, half_n) for _ in range(population_size)]
