import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    # Normalize initial guess
    current_sum = np.sum(h_initial)
    if abs(current_sum) < 1e-10:
        current_h = np.zeros(n_points)
    else:
        current_h = (h_initial / current_sum) * target_sum
    current_h = np.clip(current_h, 0, 1)
    current_sum_new = np.sum(current_h)
    if abs(current_sum_new - target_sum) > 1e-6:
        current_h = (current_h / current_sum_new) * target_sum
        current_h = np.clip(current_h, 0, 1)

    # Evaluate initial c5
    best_c5 = evaluate_erdos_solution(current_h)
    best_h = current_h.copy()

    T = 1.0
    alpha = 0.999
    n_perturb = max(1, int(0.05 * n_points))

    while time.time() - start_time < budget_s - 10:
        perturb_indices = np.random.choice(n_points, size=n_perturb, replace=False)
        h_candidate = current_h.copy()
        delta = np.random.uniform(-0.2, 0.2, size=len(perturb_indices))
        h_candidate[perturb_indices] += delta
        h_candidate = np.clip(h_candidate, 0, 1)
        candidate_sum = np.sum(h_candidate)

        if abs(candidate_sum) > 1e-10:
            h_candidate = (h_candidate / candidate_sum) * target_sum
            h_candidate = np.clip(h_candidate, 0, 1)
        else:
            h_candidate = np.zeros_like(h_candidate)

        c5_candidate = evaluate_erdos_solution(h_candidate)
        delta_c5 = c5_candidate - best_c5

        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            current_h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = current_h.copy()

        T *= alpha

    # Final constraint enforcement
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-6:
        best_h = (best_h / final_sum) * target_sum
        best_h = np.clip(best_h, 0, 1)

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    assert np.isfinite(best_c5)
    return (best_h.tolist(), best_c5, n_points)
