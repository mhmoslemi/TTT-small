        h_full[-half_n:] = h_vars[::-1]
        return h_full

    def objective(h_vars):

        return np.max(corr) * dx

        h_vars = np.clip(h_vars, 0, 1)
        total = np.sum(h_vars)

            h_vars = h_vars * (target_sum_half / total)
        return h_vars
