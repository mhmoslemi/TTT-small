    current_y = y_initial.copy()

    # Evaluate initial c5
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_y
    h_full[half_n:] = best_y[::-1]
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    best_c5 = np.max(corr) * dx
    current_c5 = best_c5

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.999
    min_T = 1e-6
    r = 0.05
    r_decay = 0.995

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 10:
        # Generate perturbed candidate y
        y_prime = current_y + np.random.normal(0, r, size=half_n)
        y_clamped = np.clip(y_prime, 0, 1)
        sum_y = np.sum(y_clamped)
        if abs(sum_y) < 1e-6:
            continue

        scale_factor = target_sum_y / sum_y
        y_scaled = y_clamped * scale_factor
        y_scaled = np.clip(y_scaled, 0, 1)

        # Build full h and compute C₅
        h_full = np.zeros(n_points)
        h_full[:half_n] = y_scaled
        h_full[half_n:] = y_scaled[::-1]
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        new_c5 = np.max(corr) * dx

        # Acceptance criterion
        delta = new_c5 - current_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_y = y_scaled.copy()
            current_c5 = new_c5
            if delta < 0 and new_c5 < best_c5:
                best_c5 = new_c5
                best_y = current_y.copy()

        T = max(T * alpha, min_T)
        r *= r_decay

    # Final validation
    h_full = np.zeros(n_points)
    h_full[:half_n] = best_y
    h_full[half_n:] = best_y[::-1]
    assert np.all((h_full >= -1e-6) & (h_full <= 1 + 1e-6)) and abs(np.sum(h_full) - n_points / 2.0) < 1e-6

    return (h_full.tolist(), best_c5, n_points)
