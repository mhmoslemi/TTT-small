import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values).copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial projection to feasible region
    h = np.clip(h, 0, 1)
    h_sum = h.sum()
    if h_sum != 0:
        h = h / h_sum * target_sum
    h = np.clip(h, 0, 1)
    h_sum = h.sum()

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx

    np.random.seed(seed)
    lr = 0.01  # Initial learning rate
    perturb_scale = 0.1
    decay_factor = 0.99
    perturb_prob = 0.1

    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    while time.time() - start_time < budget_s - 30:
        # Compute current correlation and shift
        cm = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cm)
        s = max_idx - (n_points - 1)  # Convert index to shift

        # Compute gradient for the dominant shift
        grad = np.zeros(n_points)
        for m in range(n_points):
            if m + s < n_points:
                grad[m] += (1 - h[m + s])
            if m - s >= 0:
                grad[m] -= h[m - s]

        # Update step
        h = h - lr * grad

        # Projection to feasible region
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        if h_sum != 0:
            h = h / h_sum * target_sum
        h = np.clip(h, 0, 1)

        # Acceptance criterion
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        # Anneal learning rate and perturb
        lr *= decay_factor
        if np.random.rand() < perturb_prob:
            h += np.random.normal(0, perturb_scale, n_points)
            h = np.clip(h, 0, 1)
            h_sum = h.sum()
            if h_sum != 0:
                h = h / h_sum * target_sum
            h = np.clip(h, 0, 1)

    # Final constraint check
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    return (best_h.tolist(), best_c5, n_points)
