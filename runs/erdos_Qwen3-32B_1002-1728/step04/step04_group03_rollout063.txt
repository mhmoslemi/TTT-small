from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def quaternionize(values):
        levels = np.array([0, 1/3, 2/3, 1.0])
        return np.take_along_axis(
            levels,
            np.argmin(np.abs(np.subtract.outer(values, levels)), axis=-1),
            axis=0
        )

    h_quat = quaternionize(h_initial)
    current_sum = h_quat.sum()
    if not np.isclose(current_sum, target_sum, atol=1e-6):
        diff = target_sum - current_sum
        h_quat = h_quat.copy()
        for _ in range(100):
            if np.abs(diff) < 1e-6:
                break
            for i in range(n_points):
                if np.abs(diff) < 1e-6:
                    break
                val = h_quat[i]
                level_idx = int(np.where(np.array([0, 1/3, 2/3, 1.0]) == val)[0])
                if diff > 0 and level_idx < 3:
                    increase = (level_idx + 1, level_idx)
                    delta = np.array([0, 1/3, 2/3, 1.0])[increase[0]] - val
                    if delta <= diff:
                        h_quat[i] += delta
                        diff -= delta
                    else:
                        h_quat[i] += diff
                        diff = 0
                elif diff < 0 and level_idx > 0:
                    decrease = (level_idx - 1, level_idx)
                    delta = val - np.array([0, 1/3, 2/3, 1.0])[decrease[0]]
                    if delta >= -diff:
                        h_quat[i] -= delta
                        diff += delta
                    else:
                        h_quat[i] += diff
                        diff = 0
        if np.abs(diff) > 1e-6:
            scale = target_sum / h_quat.sum()
            h_quat = h_quat * scale
            h_quat = np.clip(h_quat, 0, 1.0)

    def objective(x):
        x = np.clip(x, 0, 1.0)
        n = len(x)
        x_fft = np.fft.fft(x, 2*n)
        s_fft = np.fft.fft(1 - x, 2*n)
        corr = np.fft.ifft(x_fft * np.conj(s_fft)).real
        return np.max(corr) * dx

    def constrain_sum(x):
        return x.sum() - target_sum

    cons = {'type': 'eq', 'fun': constrain_sum}
    bounds = [(0, 1)] * n_points
    h_opt = np.clip(h_quat, 0, 1.0).copy()

    best_c5 = float('inf')
    best_h = h_opt.copy()

    np.random.seed(seed)
    while time.time() - start_time < budget_s - 10:
        perturbed = h_opt + np.random.uniform(-0.05, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1.0)
        scale = target_sum / perturbed.sum()
        perturbed = (perturbed * scale).clip(0, 1.0)

        result = minimize(
            objective,
            perturbed,
            bounds=bounds,
            constraints=[cons],
            method="SLSQP",
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            x = result.x.clip(0, 1.0)
            if abs(x.sum() - target_sum) < 1e-6:
                c5 = result.fun
                if c5 < best_c5:
                    best_c5 = c5
                    best_h[:] = x

    assert abs(best_h.sum() - target_sum) < 1e-6, "Constraint violated"
    return (best_h.tolist(), best_c5, n_points)
