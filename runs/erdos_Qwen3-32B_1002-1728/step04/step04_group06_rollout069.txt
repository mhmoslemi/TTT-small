import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values)
    n = len(h)
    target_sum = n / 2.0

    # Normalize initial h to satisfy constraints
    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        h = h * (target_sum / current_sum)

    def compute_overlaps(h):
        corr = np.correlate(h, h, mode='full')
        overlaps = np.array([corr[n - 1 + k] for k in range(1, n)])
        return overlaps

    def compute_gradient(h, k):
        grad = np.zeros(n)
        for i in range(n):
            term1 = h[i + k] if (i + k < n) else 0
            term2 = h[i - k] if (i - k >= 0) else 0
            grad[i] = term1 + term2
        return grad

    def project(h, target_sum):
        n = len(h)
        lo, hi = -5, 5
        for _ in range(100):
            lam = (lo + hi) / 2
            temp = h - lam
            clamped = np.clip(temp, 0.0, 1.0)
            s = np.sum(clamped)
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        lam = (lo + hi) / 2
        h_proj = np.clip(h - lam, 0.0, 1.0)
        return h_proj

    best_h = h.copy()

    lr = 0.02
