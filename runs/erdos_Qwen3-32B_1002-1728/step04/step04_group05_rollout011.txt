                    p_mag *= 1.5
                    retries += 1

        # Learning rate decay even when no improvement
        lr *= 0.9999

    h_final = create_full_h(best_y)
    assert np.allclose(np.sum(h_final), n_points / 2.0, atol=1e-6)
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6))
    return (h_final.tolist(), best_c5, n_points)
