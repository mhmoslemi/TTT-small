import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def normalize_h(h):
        h_clipped = np.clip(h, 0, 1)
        current_sum = h_clipped.sum()
        if current_sum < 1e-8:
            return np.zeros_like(h)
        return h_clipped * (target_sum / current_sum)

    np.random.seed(seed)
    pop_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5
    pop = np.zeros((pop_size, n_points))
    pop[0] = normalize_h(h_initial)
    for i in range(1, pop_size):
        pert = h_initial + np.random.normal(0, 0.1, n_points)
        pop[i] = normalize_h(pert)

    best_h = pop[0].copy()
    best_c5 = compute_c5(best_h)

    while time.time() - start_time < budget_s:
        # Evaluate population
        c5_values = np.array([compute_c5(h) for h in pop])
        current_best_idx = np.argmin(c5_values)
        if c5_values[current_best_idx] < best_c5:
            best_h = pop[current_best_idx].copy()
            best_c5 = c5_values[current_best_idx]

        # Selection - tournament
        parent_indices = []
        for _ in range(pop_size // 2):
            candidates = np.random.randint(0, pop_size, tournament_size)
            parent_idx = candidates[np.argmin(c5_values[candidates])]
            parent_indices.append(parent_idx)

        parents = pop[parent_indices]
        next_pop = []

        # Crossover and mutation
        for i in range(0, len(parents), 2):
            if i + 1 >= len(parents):
                next_pop.append(parents[i].copy())
                continue

            if np.random.rand() < crossover_rate:
                cx_point = np.random.randint(1, n_points - 1)
                child1 = np.vstack((parents[i][:cx_point], parents[i+1][cx_point:])).ravel()
                child2 = np.vstack((parents[i+1][:cx_point], parents[i][cx_point:])).ravel()
            else:
                child1, child2 = parents[i].copy(), parents[i+1].copy()

            for child in [child1, child2]:
                if np.random.rand() < mutation_rate:
                    noise = np.random.normal(0, 0.1, n_points)
                    child += noise
                child_h = normalize_h(child)
                next_pop.append(child_h)

        # Truncate population
        next_pop = np.array(next_pop[:pop_size])
        pop = next_pop

    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
