    np.random.seed(seed)

    p_best = []
    p_val = []

    p_best = np.array(p_best)
    p_val = np.array(p_val)

    g_best = p_best[g_best_idx].copy()

    w_start, w_end, c1, c2 = 0.9, 0.4, 2.5, 2.5
