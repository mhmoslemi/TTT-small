    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0
    
    # Create symmetric h_full from initial values
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        scale = (n_points / 2.0) / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[:half_n].copy()
    h_half_sum = np.sum(h_half)
    if abs(h_half_sum - target_sum_half) > 1e-6 and h_half_sum != 0:
        h_half = (h_half / h_half_sum) * target_sum_half
    h_half = np.clip(h_half, 0, 1)
    
    best_h_half = h_half.copy()
    current_h_half = best_h_half.copy()
    
    current_c5 = compute_c5(create_full_h(current_h_half, n_points, half_n), dx)
    best_c5 = current_c5
    
    T0 = 1.0
    T = T0
    alpha = 0.999
    T_min = 1e-5
    base_seed = seed
    
    while time.time() - start_time < budget_s - 60 and T > T_min:
        # Generate perturbed h_half
        rng = np.random.RandomState(base_seed)
        base_seed += 1
        
        noise = rng.normal(0, T, size=half_n)
        perturbed = current_h_half + noise
        # Clip between 0 and 1
        perturbed = np.clip(perturbed, 0, 1)
        # Compute sum and scale
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed *= target_sum_half / perturbed_sum
        perturbed = np.clip(perturbed, 0, 1)
        
        # Create h_full and compute new_c5
        h_full_candidate = create_full_h(perturbed, n_points, half_n)
        new_c5 = compute_c5(h_full_candidate, dx)
        
        # Acceptance decision
        delta_E = new_c5 - current_c5
        if delta_E < 0 or np.exp(-delta_E / T) > rng.rand():
            current_h_half = perturbed.copy()
            current_c5 = new_c5
            # Update best if necessary
            if current_c5 < best_c5:
                best_h_half = current_h_half.copy()
                best_c5 = current_c5
        
        # Cool the temperature
        T *= alpha
    
    h_full = create_full_h(best_h_half, n_points, half_n)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    
    return (h_full.tolist(), best_c5, n_points)
