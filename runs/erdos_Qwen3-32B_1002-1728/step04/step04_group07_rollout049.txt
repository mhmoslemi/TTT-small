import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_sym):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_sym[i]
        return h_full

    def evaluate_c5(h_sym):
        h_full = create_full_h(h_sym)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initial symmetric solution
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    total = np.sum(h_full_sym)
    if abs(total - n_points / 2.0) > 1e-6 and total != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / total)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[:half_n]

    # PSO setup
    np.random.seed(seed)
    n_particles = 30
    w, c1, c2 = 0.7, 1.5, 1.5
    v_clip = 0.2

    particles = h_half + np.random.uniform(-0.1, 0.1, (n_particles, half_n))
    particles = np.clip(particles, 0, 1)
    for i in range(n_particles):
        t = particles[i].sum()
        if t > 1e-8: particles[i] *= target_sum_half / t
        particles[i] = np.clip(particles[i], 0, 1)

    velocities = np.random.uniform(-0.1, 0.1, size=(n_particles, half_n))
    p_best = particles.copy()
    p_best_scores = np.array([evaluate_c5(p) for p in p_best])
    g_best_idx = p_best_scores.argmin()
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]

    # PSO iterations
    iterations = 0
    while time.time() - start_time < budget_s * 0.5:
        r1 = np.random.rand(n_particles, half_n)
        r2 = np.random.rand(n_particles, half_n)
        velocities = w * velocities + c1 * r1 * (p_best - particles) + c2 * r2 * (g_best - particles)
        velocities = np.clip(velocities, -v_clip, v_clip)
        particles += velocities
        particles = np.clip(particles, 0, 1)
        for i in range(n_particles):
            t = particles[i].sum()
            if t > 1e-8: particles[i] *= target_sum_half / t
            particles[i] = np.clip(particles[i], 0, 1)
        scores = np.array([evaluate_c5(p) for p in particles])
        idx_update = scores < p_best_scores
        p_best[idx_update] = particles[idx_update]
        p_best_scores[idx_update] = scores[idx_update]
        if np.min(scores) < g_best_score:
            g_best[:] = particles[scores.argmin()]
            g_best_score = np.min(scores)
        iterations += 1

    # Local refinement
    bounds_per_var = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    result = minimize(
        fun=lambda x: evaluate_c5(x),
        x0=g_best,
        method='SLSQP',
        bounds=bounds_per_var,
        constraints=cons,
        options={'maxiter': 300, 'tol': 1e-8}
    )

    # Final validation
    final_h = result.x if result.success else g_best
    h_full = create_full_h(final_h)
    final_c5 = evaluate_c5(final_h)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), final_c5, n_points)
