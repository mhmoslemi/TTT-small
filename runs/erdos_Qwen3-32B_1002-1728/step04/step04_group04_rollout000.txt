import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize the initial solution to satisfy the sum constraint
    h_current = h_full_initial.copy().astype(float)
    current_sum = h_current.sum()
    if current_sum != 0:
        h_current = h_current / current_sum * target_sum
    h_current = np.clip(h_current, 0, 1)
    best_h = h_current.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    np.random.seed(seed)
    T = 1.0
    alpha = 0.999  # Cooling rate
    while time.time() - start_time < budget_s - 10:
        # Perturbation
        num_perturb = max(1, int(0.05 * n_points))
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-T, T, size=num_perturb)
        h_candidate = h_current.copy()
        h_candidate[perturb_indices] += delta
        h_candidate = np.clip(h_candidate, 0, 1)

        # Adjust sum to target_sum
        total = h_candidate.sum()
        if abs(total - target_sum) > 1e-6:
            scale = target_sum / total
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)

        # Compute new_c5
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr) * dx

        # Acceptance
        delta_c = new_c5 - best_c5
        if delta_c < 0:
            h_current = h_candidate.copy()
            best_c5 = new_c5

        else:
            if np.random.rand() < np.exp(-delta_c / T):

        # Cool down

    h_full = best_h

import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    if current_sum != 0:
        h_current = (h_current / current_sum) * target_sum

    best_h = h_current.copy()
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx

    np.random.seed(seed)
    T = 1.0
    alpha = 0.999  # cooling rate

    while time.time() - start_time < budget_s - 10:
        # Perturbation
        num_perturb = max(1, int(0.05 * n_points))
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-T, T, size=num_perturb)
        h_candidate = h_current.copy()
        h_candidate[perturb_indices] += delta
        h_candidate = np.clip(h_candidate, 0, 1)
        total = h_candidate.sum()

        # Adjust sum to target
        if abs(total - target_sum) > 1e-6:
            scale = target_sum / total
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)

        # Evaluate new solution
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr) * dx

        delta_c = new_c5 - best_c5
