import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial symmetric and scaled h_half
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    h_full_sym_sum = np.sum(h_full_sym)
    if h_full_sym_sum != 0:
        scale = target_sum / h_full_sym_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_initial = h_full_sym[0:half_n].copy()

    # Initialize population
    population_size = 50
    population = [h_half_initial.copy()]
    np.random.seed(seed)
    for _ in range(population_size - 1):
        perturbed = h_half_initial + np.random.normal(0, 0.1, half_n)
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_h_half = h_half_initial.copy()
    best_c5 = float('inf')

    while time.time() - start_time < budget_s - 60:
        # Evaluate current population for best
        for h_half in population:
            h_full = np.zeros(n_points)
            for i in range(half_n):
                h_full[i] = h_full[n_points - 1 - i] = h_half[i]
            h_sum = np.sum(h_full)
            if h_sum != 0:
                scale = target_sum / h_sum
                h_full *= scale
            h_full = np.clip(h_full, 0, 1)
            h_sum_clip = np.sum(h_full)
            if h_sum_clip != 0:
                scale_clip = target_sum / h_sum_clip
                h_full *= scale_clip
                h_full = np.clip(h_full, 0, 1)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h_half = h_half.copy()

        # Prepare population with C5 scores for selection
        population_c5 = []
        for indiv in population:
            h_full = np.zeros(n_points)
            for i in range(half_n):
                h_full[i] = h_full[n_points - 1 - i] = indiv[i]
            h_sum = np.sum(h_full)
            if h_sum != 0:
                scale = target_sum / h_sum
                h_full *= scale
            h_full = np.clip(h_full, 0, 1)
            corr = np.correlate(h_full, 1 - h_full, mode='full')
            c5 = np.max(corr) * dx
            population_c5.append((c5, indiv))
        # Sort by C5 and select top half
        population_c5.sort(key=lambda x: x[0])
        selected = [indiv for c5, indiv in population_c5[:(len(population) + 1) // 2]]

        # Generate new population via crossover and mutation
        new_population = []
        while len(new_population) < population_size and time.time() - start_time < budget_s - 30:
            parent1, parent2 = random.sample(selected, 2)
            cross_point = random.randint(1, half_n - 1)
            child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            # Apply mutation
            if random.random() < 0.2:
                noise = np.random.normal(0, 0.3, half_n)
                child += noise
                child = np.clip(child, 0, 1)
            # Process child to full h and back to h_half
            h_full = np.zeros(n_points)
            for i in range(half_n):
                h_full[i] = h_full[n_points - 1 - i] = child[i]
            h_sum = np.sum(h_full)
            if h_sum != 0:
                scale = target_sum / h_sum
                h_full *= scale
            h_full = np.clip(h_full, 0, 1)
            new_child = h_full[:half_n].copy()
            new_population.append(new_child)
        population = new_population

    # Final processing of best solution
    h_full = np.zeros(n_points)
    for i in range(half_n):
        h_full[i] = h_full[n_points - 1 - i] = best_h_half[i]
    h_sum = np.sum(h_full)
    if h_sum != 0:
        scale = target_sum / h_sum
        h_full *= scale
    h_full = np.clip(h_full, 0, 1)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    final_c5 = np.max(corr) * dx
    h_values = h_full.tolist()

    assert abs(np.sum(h_full) - target_sum) < 1e-6, f"Sum constraint violated by {abs(np.sum(h_full) - target_sum)}"
    assert all(0 <= x <= 1 for x in h_values), "Value out of [0,1] range"

    return (h_values, final_c5, n_points)
