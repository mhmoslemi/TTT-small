        corr = np.correlate(h_full, 1 - h_full, mode='full')
        c5 = np.max(corr) * dx

    def project(x):

        x = x / x.sum() * target_sum
        x = np.clip(x, 0, 1)
        return x

    # Differential evolution with strong sum constraint
