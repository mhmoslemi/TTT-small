        x_clipped = np.clip(x, 0.0, 1.0)
        current_sum = np.sum(x_clipped)
        if current_sum == 0.0:
            return np.ones_like(x_clipped) * target_sum / len(x_clipped)
        scale = target_sum / current_sum
        x_scaled = x_clipped * scale
        x_scaled_clipped = np.clip(x_scaled, 0.0, 1.0)
        final_sum = np.sum(x_scaled_clipped)
        if abs(final_sum - target_sum) > 1e-6:
            indices = np.where(x_scaled_clipped < 1.0)[0]
            if len(indices) > 0:
                x_scaled_clipped[indices[-1]] += target_sum - final_sum
        return x_scaled_clipped
