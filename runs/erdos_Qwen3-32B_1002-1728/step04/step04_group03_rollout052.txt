import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # Sum of half variables for symmetric construction

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        cross_corr = np.correlate(h_full, 1.0 - h_full, mode="full")
        return np.max(cross_corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half},)

    # Create symmetric initial guess from initial_h_values
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n].copy()

    best_h_half = h_half.copy()
    best_c5 = objective(best_h_half)

    np.random.seed(seed)
    perturbation_magnitude = 0.1  # Larger perturbation for exploration

    while time.time() - start_time < budget_s - 20:
        # Generate a new perturbation
        rng = np.random.RandomState(int(time.time() % seed))
        perturbed = best_h_half + rng.uniform(-perturbation_magnitude, perturbation_magnitude, size=half_n)
        perturbed = np.clip(perturbed, 0, 1)
        total = np.sum(perturbed)
        if not np.isclose(total, 0):
            perturbed = perturbed * (target_sum_half / total)
        perturbed = np.clip(perturbed, 0, 1)

        # Optimization with SLSQP only (supports equality constraints)
        result = minimize(
