import time

    target_half_sum = target_sum / 2.0
    dx = 2.0 / n_points

    def create_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def compute_c5(h_half):

        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        s = np.sum(h_half)
        if abs(s - target_half_sum) > 1e-6 and s != 0:
            scale = target_half_sum / s
            h_half = np.clip(h_half * scale, 0.0, 1.0)
        return h_half

    h_full_sym = np.clip(h_full_sym * scale, 0.0, 1.0)

    max_generations = 100000  # Will break early based on time
