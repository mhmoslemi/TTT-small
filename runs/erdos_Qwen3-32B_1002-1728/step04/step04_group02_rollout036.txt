            if current_c5 < personal_best_fitness[particle_idx]:
                personal_best_fitness[particle_idx] = current_c5
                personal_best_positions[particle_idx] = particles[particle_idx].copy()

        # Update global best
        current_global_idx = np.argmin(fitness)
        if fitness[current_global_idx] < global_best_fitness:
            global_best_position = particles[current_global_idx].copy()
            global_best_fitness = fitness[current_global_idx]
            best_h[:] = global_best_position
            best_c5 = global_best_fitness

    # Final validation
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"

    return (best_h.tolist(), best_c5, n_points)
