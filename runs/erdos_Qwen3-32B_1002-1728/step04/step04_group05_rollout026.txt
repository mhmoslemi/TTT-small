import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2

        corr = np.correlate(h_full, 1 - h_full, mode='full')

    np.random.seed(seed)
    SWARM_SIZE = 50
    particles = [h_half_initial.copy() for _ in range(SWARM_SIZE)]
