import numpy as np
import time

def compute_c5(h_full):
    n = len(h_full)
    h_fft = np.fft.fft(h_full, 2 * n - 1)
    one_minus_h_fft = np.fft.fft(1.0 - h_full, 2 * n - 1)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    dx = 2.0 / n
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    np.random.seed(seed)

    n_particles = 50
    positions = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    pbest_pos = np.zeros((n_particles, n_points))
    pbest_c5 = np.full(n_particles, np.inf)

    def init_particle():
        p = h_initial.copy() + np.random.uniform(-0.1, 0.1, size=n_points)
        p = np.clip(p, 0, 1)
        p_sum = np.sum(p)
        if p_sum != 0:
            p *= target_sum / p_sum
        else:
            p = 0.5 * np.ones(n_points)
        return p

    for i in range(n_particles):
        positions[i] = init_particle()
        pbest_pos[i] = positions[i].copy()
        c5 = compute_c5(positions[i])
        pbest_c5[i] = c5

    best_idx = np.argmin(pbest_c5)
    gbest_pos = positions[best_idx].copy()
    best_c5 = pbest_c5[best_idx]

    w = 0.7
    c1 = 1.0
    c2 = 1.0
    max_velocity = 0.5

    while time.time() - start_time < budget_s - 5:
