import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    # Initial symmetric h_half
    h_full_sym = np.zeros(n_points)
    for i in range(half_n):
        j = n_points - 1 - i
        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - 1.0) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (1.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0.0, 1.0)
    h_half_initial = h_full_sym[:half_n].copy()

    # Normalize sum of h_half to 0.5
    current_sum_half = np.sum(h_half_initial)
    if current_sum_half != 0.0:
        h_half_initial = h_half_initial * (0.5 / current_sum_half)
    h_half_initial = np.clip(h_half_initial, 0.0, 1.0)

    def evaluate_candidate(h_half_candidate):
        h_full = create_symmetric_h_full(h_half_candidate)
        corr_values = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr_values) * dx

    population_size = 50
    rng = default_rng(seed)
    population = np.zeros((population_size, half_n))
    population[0] = h_half_initial.copy()
    for i in range(1, population_size):
        noise = rng.normal(0, 0.1, half_n)
        perturbed = h_half_initial + noise
        perturbed = np.clip(perturbed, 0.0, 1.0)
        sum_perturbed = np.sum(perturbed)

        population[i] = perturbed

    max_generations = 500

            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
