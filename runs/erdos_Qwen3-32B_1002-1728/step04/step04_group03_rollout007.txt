    dx = 2.0 / len(h)
    cross_correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(cross_correlation) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_full_current = h_full_initial.copy()
    current_sum = np.sum(h_full_current)
    if not np.isclose(current_sum, target_sum):
        scaling = target_sum / current_sum
        h_full_current = np.clip(h_full_current * scaling, 0, 1)
    best_h = h_full_current.copy()
    best_c5 = compute_c5(best_h)

    initial_T = 1.0
    cooling_rate = 0.995
    T = initial_T
    np.random.seed(seed)

    # Simulated Annealing
    while time.time() - start_time < budget_s - 60:
        perturbation_scale = T * 0.1
        noise = np.random.randn(n_points) * perturbation_scale
        new_h = np.clip(h_full_current + noise, 0, 1)
        current_sum = np.sum(new_h)
        if current_sum > 1e-6:
            scaling = target_sum / current_sum
            new_h = np.clip(new_h * scaling, 0, 1)
        else:
            new_h = best_h.copy()
        new_c5 = compute_c5(new_h)
        delta = new_c5 - best_c5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h_full_current = new_h
            if new_c5 < best_c5:
                best_h = h_full_current.copy()
                best_c5 = new_c5
        T *= cooling_rate

    # Gradient refinement
    def objective(x):
        h = np.clip(x, 0, 1)
        current_sum = np.sum(h)
        if current_sum < 1e-6:
            return float('inf')
        h = np.clip(h * (target_sum / current_sum), 0, 1)
        return compute_c5(h)

    bounds = [(0.0, 1.0)] * n_points
    for method in ['L-BFGS-B', 'SLSQP']:
        if time.time() - start_time >= budget_s - 30:
            break
        x0 = best_h.copy()
        result = minimize(
            fun=objective,
            x0=x0,
            method=method,
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_opt = np.clip(result.x, 0, 1)
            current_sum = np.sum(h_opt)
            if current_sum > 1e-6:
                h_opt = h_opt * (target_sum / current_sum)
                h_opt = np.clip(h_opt, 0, 1)
                new_c5 = compute_c5(h_opt)
                if new_c5 < best_c5:
                    best_h = h_opt
                    best_c5 = new_c5

    assert np.allclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint failed"
    assert np.all((best_h >= -1e-6) & (best_h <= 1 + 1e-6)), "Bounds constraint failed"
    return (best_h.tolist(), best_c5, n_points)
