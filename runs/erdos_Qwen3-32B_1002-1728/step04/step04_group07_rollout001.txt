import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Symmetric initialization
    h_full_sym = np.copy(h_full_initial)
    for i in range(half_n):
        avg = (h_full_initial[i] + h_full_initial[n_points - 1 - i]) / 2.0
        h_full_sym[i] = h_full_sym[n_points - 1 - i] = avg
    current_sum = np.sum(h_full_sym)
    h_full_sym = np.clip(h_full_sym * (n_points / 2.0 / current_sum), 0, 1) if current_sum != 0 else np.zeros(n_points)
    h_half = h_full_sym[:half_n]

    # Particle Swarm Optimization (PSO) setup
    np.random.seed(seed)
    particles_pos = np.random.uniform(0, 1, (30, half_n))
    particles_vel = np.zeros_like(particles_pos)
    particles_pbest = np.copy(particles_pos)
    pbest_scores = np.array([objective(p) for p in particles_pos])
    gbest_idx = np.argmin(pbest_scores)
    gbest_pos = np.copy(particles_pbest[gbest_idx])
    gbest_score = pbest_scores[gbest_idx]

    w = 0.7
    c1, c2 = 1.5, 1.5

    # PSO execution
    while time.time() - start_time < budget_s - 300:
        for i in range(30):
            r1, r2 = np.random.rand(2)
            particles_vel[i] = (w * particles_vel[i] +
                               c1 * r1 * (particles_pbest[i] - particles_pos[i]) +
                               c2 * r2 * (gbest_pos - particles_pos[i]))
            particles_pos[i] = np.clip(particles_pos[i] + particles_vel[i], 0, 1)
            
            # Normalize to maintain sum constraint
            total = particles_pos[i].sum()
            if not np.isclose(total, 0):
                particles_pos[i] = (particles_pos[i] / total) * target_sum_half
            particles_pos[i] = np.clip(particles_pos[i], 0, 1)
            
            current_score = objective(particles_pos[i])
            if current_score < pbest_scores[i]:
                particles_pbest[i] = np.copy(particles_pos[i])
                pbest_scores[i] = current_score

        new_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[new_gbest_idx] < gbest_score:
            gbest_score = pbest_scores[new_gbest_idx]
            gbest_pos = particles_pbest[new_gbest_idx].copy()

    # Refinement with SLSQP
    bounds = [(0.0, 1.0)] * half_n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    
    result = minimize(
        fun=objective,
        x0=gbest_pos,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-10}
    )

    best_h_half = result.x if result.success else gbest_pos
    best_c5 = result.fun if result.success else gbest_score
    h_full = create_full_h(best_h_half)

    # Final checks
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
