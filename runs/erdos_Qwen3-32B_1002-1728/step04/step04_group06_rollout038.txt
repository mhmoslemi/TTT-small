import numpy as np
import time
from scipy.optimize import differential_evolution, minimize
import warnings

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def normalize_h_half(h_half):
        current_sum = np.sum(h_half)
        if current_sum == 0:
            h_half = np.full_like(h_half, target_sum_half / half_n)
        else:
            scale = target_sum_half / current_sum
            h_half = h_half * scale
        return np.clip(h_half, 0.0, 1.0)

    h_half_initial = h_full_initial[:half_n].copy()
    h_half_initial = normalize_h_half(h_half_initial)

    def objective(h_half_candidate):
        h_half = normalize_h_half(h_half_candidate)
        h_full = create_symmetric_h_full(h_half)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(np.abs(corr)) * dx

    bounds = [(0.0, 1.0) for _ in range(half_n)]

    best_h_half = h_half_initial.copy()
    best_c5 = objective(best_h_half)

    # Differential Evolution phase
    timeout = 800  # 800 seconds for DE
    de_start = time.time()
    try:
        de_result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            tol=1e-6,
            seed=seed,
            mutation=(0.5, 1.0),
            recombination=0.9,
            polish=False,
            init='sobol',
            workers=-1,
            updating='deferred'
        )
        if de_result.success:
            best_h_half = de_result.x
            best_c5 = de_result.fun
    except:
        pass
    finally:
        elapsed_de = time.time() - de_start
        if elapsed_de > timeout:
            best_h_half = h_half_initial
            best_c5 = objective(best_h_half)

    # Local refinement phase
    try:
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
        local_result = minimize(
            objective,
            x0=best_h_half,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100}
        )
        if local_result.success and local_result.fun < best_c5:
            best_h_half = local_result.x
    except:
        pass

    h_full = create_symmetric_h_full(normalize_h_half(best_h_half))
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    return (h_full.tolist(), best_c5, n_points)
