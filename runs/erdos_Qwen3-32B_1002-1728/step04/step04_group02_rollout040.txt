                child = np.concatenate((parent1[:cross_point], parent2[cross_point:]))
            else:
                if np.random.rand() < 0.5:
                    child = parent1.copy()
                else:
                    child = parent2.copy()

            if np.random.rand() < mutation_rate:
                mutated = child + np.random.normal(0, 0.05, n_points)
                child = make_feasible(mutated)
            else:
                child = make_feasible(child)
            new_population.append(child)

        population = new_population[:population_size]
        fitness = [compute_c5(ind) for ind in population]
        current_best_idx = np.argmin(fitness)
        current_best = fitness[current_best_idx]

        if current_best < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = current_best
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        if stagnation_counter > 50:
            mutation_rate = 0.3
            stagnation_counter = 0

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Integral constraint violated"
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Value bounds violated"
    return (best_h.tolist(), best_c5, n_points)
