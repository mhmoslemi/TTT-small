import numpy as np
from scipy.optimize import differential_evolution
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objectiveDE(x):
        h = np.clip(x, 0, 1)
        total = h.sum()
        if np.isclose(total, 0.0):
            return float('inf')
        h_scaled = h * (target_sum / total)
        h_scaled = np.clip(h_scaled, 0, 1)
        corr = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result_de = differential_evolution(
        objectiveDE,
        bounds=bounds,
        popsize=8,
        mutation=(0.5, 1.0),
        recombination=0.9,
        strategy='best1bin',
        tol=1e-6,
        maxiter=100,
        polish=False,
        seed=seed,
        disp=False
    )

    elapsed = time.time() - start_time
    if elapsed >= 0.95 * budget_s:
        # Fallback to initial construction to meet deadline
        h_best = np.clip(h_full_initial / h_full_initial.sum() * target_sum, 0, 1)
        corr = np.correlate(h_best, 1 - h_best, mode='full')
        best_c5 = np.max(corr) * dx
        return (h_best.tolist(), best_c5, n_points)

    h_best_x = result_de.x
    h_best = np.clip(h_best_x, 0, 1)
    total = h_best.sum()
    if np.isclose(total, 0.0):
        h_best = np.zeros_like(h_best)
    else:
        h_best = h_best * (target_sum / total)
        h_best = np.clip(h_best, 0, 1)

    assert np.all((h_best >= -1e-6) & (h_best <= 1.0 + 1e-6)), "Out of bounds"
    assert abs(np.sum(h_best) - target_sum) < 1e-6, "Constraint violation"
    
    final_corr = np.correlate(h_best, 1 - h_best, mode='full')
    best_c5 = np.max(final_corr) * dx
    return (h_best.tolist(), best_c5, n_points)
