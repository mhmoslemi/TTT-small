    c1 = 1.5
    c2 = 1.5

    np.random.seed(seed)
    particles_x = [h_half_init + np.random.uniform(-0.1, 0.1, half_n) for _ in range(num_particles)]
    for p in particles_x:
        p[:] = repair_position(p).copy()
    particles_v = [np.random.uniform(-0.05, 0.05, half_n) for _ in range(num_particles)]
    particles_pbest = [p.copy() for p in particles_x]
    pbest_objs = [objective(p) for p in particles_x]
    gbest_idx = np.argmin(pbest_objs)
    gbest = particles_pbest[gbest_idx].copy()
    best_c5 = pbest_objs[gbest_idx]

    # PSO iteration
    particle_clock = time.time()
    while time.time() - start_time < budget_s - 300 and time.time() - particle_clock < max_pso_time:
        for i in range(num_particles):
            x_i = particles_x[i]
            v_i = particles_v[i]
            pbest_i = particles_pbest[i]
            r1 = np.random.rand(half_n)
            r2 = np.random.rand(half_n)
            v_i_new = w * v_i + c1 * r1 * (pbest_i - x_i) + c2 * r2 * (gbest - x_i)
            x_i_new = x_i + v_i_new
            x_i_new = repair_position(x_i_new)
            obj_new = objective(x_i_new)
            # Update particles
            particles_x[i] = x_i_new.copy()
            particles_v[i] = v_i_new.copy()
            # Update pbest
            if obj_new < pbest_objs[i]:
                particles_pbest[i] = x_i_new.copy()
                pbest_objs[i] = obj_new
                if obj_new < best_c5:
                    gbest[:] = x_i_new.copy()
                    best_c5 = obj_new

    # Local optimization with SLSQP
    bounds = [(0.0, 1.0)] * half_n
    eq_const = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half}
    result = minimize(
        fun=objective,
        x0=gbest,
        method='SLSQP',
        bounds=bounds,
        constraints=eq_const,
        options={'maxiter': 300, 'ftol': 1e-10}
    )
    if result.success:
        h_half = result.x.copy()
        c5_refined = objective(h_half)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            gbest[:] = h_half.copy()

    h_full = create_full_h(gbest)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
