import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial guess to meet sum constraint
    h_current = h_initial.copy()
    current_sum = h_current.sum()
    if current_sum > 1e-10:
        h_current = (h_current / current_sum) * target_sum
    h_current = np.clip(h_current, 0, 1)
    
    # Objective function
    def compute_c5(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()
    
    # SA parameters
    temp = 1.0
    alpha = 0.9992  # slower cooling for deeper search
    np.random.seed(seed)
    
    while time.time() - start_time < budget_s:
        # Generate perturbation
        perturbed = h_current.copy()
        perturb_size = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=perturb_size, replace=False)
        delta = np.random.normal(0, 0.15, size=perturb_size)  # Gaussian perturbations
        
        # Apply perturbations and clip
        perturbed[indices] += delta
        perturbed = np.clip(perturbed, 0, 1)
        
        # Restore sum constraint
        current_total = perturbed.sum()
        if abs(current_total - target_sum) > 1e-6:
            perturbed = (perturbed / current_total) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
        
        # Evaluate candidate
        c_candidate = compute_c5(perturbed)
        c_current = compute_c5(h_current)
        delta_energy = c_candidate - c_current
        
        # Metropolis criterion
        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / temp):
            h_current[:] = perturbed
            if c_candidate < best_c5:
                best_c5 = c_candidate
                best_h[:] = h_current
        
        # Annealing step
        temp *= alpha
    
    # Final validation
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
