        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_half_initial = (h_full_initial[:half_n] + h_full_initial[-1:-half_n-1:-1]) / 2.0
    current_sum = h_half_initial.sum()
    if current_sum != 0:
        h_half_initial = h_half_initial * (target_sum / current_sum)
    h_half_initial = np.clip(h_half_initial, 0, 1)

    # PSO parameters
    np.random.seed(seed)
    n_particles = 100
    max_velocity = 0.5
    c1, c2 = 2.0, 2.0

    # Initialize particles
    noise = 0.1 * np.random.randn(n_particles, half_n)

    sums = raw_positions.sum(axis=1, keepdims=True)

    pbest_positions = positions.copy()

    best_c5 = pbest_values.min()
    gbest = pbest_positions[pbest_values.argmin()].copy()

    while time.time() - start_time < budget_s - 60:
