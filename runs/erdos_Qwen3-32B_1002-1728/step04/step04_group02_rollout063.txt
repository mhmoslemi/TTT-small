    sum_h = np.sum(h_full)
    if abs(sum_h - n_points_initial / 2) > 1e-6:
        h_full *= (n_points_initial / 2) / sum_h
    h_full = np.clip(h_full, 0, 1)

    assert np.allclose(np.sum(h_full), n_points_initial / 2, atol=1e-6)
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))
    return (h_full.tolist(), best_c5, n_points_initial)
