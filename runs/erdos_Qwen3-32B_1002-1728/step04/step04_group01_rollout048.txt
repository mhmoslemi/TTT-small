from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def objective(h_vars):
        h_full = create_full_h(h_vars)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def fourier_init(h_current):
        h_fft = np.fft.fft(h_current)
        h_fft_new = np.ones_like(h_fft) * 0.5
        h_candidate = np.fft.ifft(h_fft_new).real
        h_candidate = np.clip(h_candidate, 0, 1)
        total = np.sum(h_candidate)
        if total != 0:
            scale = target_sum / total
            h_candidate *= scale
            h_candidate = np.clip(h_candidate, 0, 1)
        # Enforce symmetry
        h_candidate[half_n:] = h_candidate[n_points - half_n - 1::-1]
        return h_candidate

    # Generate Fourier-based initialization
    h_fourier = fourier_init(h_full_initial)
    c5_fourier = objective(h_fourier)
    c5_initial = objective(h_full_initial)
    best_h = h_fourier if c5_fourier < c5_initial else h_full_initial

    # Apply perturbation and optimization
    np.random.seed(seed)
    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})

    # Extract half variables from best_h
    best_h_half = create_full_h(h_best)[:half_n] if n_points % 2 == 0 else h_best[:half_n + 1]

    while time.time() - start_time < budget_s - 30:
        seed_val = int((time.time() - start_time) * 1000) % (2**31 - 1)
        np.random.seed(seed_val)
        perturbed = best_h_half + 0.1 * np.random.randn(half_n)
        perturbed = np.clip(perturbed, 0, 2)
        total = np.sum(perturbed)
        if total != 0:
            perturbed = (perturbed / total) * target_sum_half
        perturbed = np.clip(perturbed, 0, 1)

        solver_choice = int((seed_val * 13) % 3)
        if solver_choice == 0:
            res = minimize(objective, perturbed, method='COBYLA', bounds=bounds_per_var, constraints=cons, options={'maxiter': 100})
        elif solver_choice == 1:
            res = minimize(objective, perturbed, method='SLSQP', bounds=bounds_per_var, constraints=cons, options={'maxiter': 150})
        else:
            res = minimize(objective, perturbed, method='Nelder-Mead', bounds=bounds_per_var, options={'maxiter': 100})

        if res.success:
            new_h_half = res.x
            h_full = create_full_h(new_h_half)
            new_c5 = objective(new_h_half)
            if new_c5 < best_c5:
                best_h_half = new_h_half
