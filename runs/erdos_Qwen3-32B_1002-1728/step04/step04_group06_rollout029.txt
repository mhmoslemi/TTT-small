import scipy.optimize
from numpy.random import default_rng

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum = n_points / 4.0
    dx = 2.0 / n_points

    def create_symmetric_h_full(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[-half_n:] = h_half[::-1]
        return h_full

    def normalize_h_half(h_half):
        current_sum = np.sum(h_half)
        if np.abs(current_sum) < 1e-10:
            h_half = np.full_like(h_half, target_sum / half_n)
        else:
            scale = target_sum / current_sum
            h_half = h_half * scale
        return np.clip(h_half, 0.0, 1.0)

    def evaluate_candidate(h_half):
        h_full = create_symmetric_h_full(h_half)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        return np.max(corr) * dx

    rng = default_rng(seed)
    population_size = 50
    F = 0.7
    CR = 0.9
    population = np.zeros((population_size, half_n))
    h_half_initial = h_full_initial[:half_n].copy()
    population[0] = normalize_h_half(h_half_initial)
    for i in range(1, population_size):
        perturb = rng.normal(0.0, 0.1, half_n)
        candidate = h_half_initial + perturb
        population[i] = normalize_h_half(candidate)

    best_c5 = float('inf')
    best_h_half = population[0].copy()

    while time.time() - start_time < budget_s - 30:
        new_population = []

            candidates = [j for j in range(population_size) if j != i]
            a, b, c = population[rng.choice(candidates, 3, replace=False)]
            mutant = a + F * (b - c)

            for j in range(half_n):
                if rng.random() < CR:
                    trial[j] = mutant[j]
            trial = normalize_h_half(trial)
