    n_points = len(h_full_initial)

    half_n = n_points // 2

    dx = 2.0 / n_points

    target_sum_half = n_points / 4.0  # sum(h_half) = n_points/4

    def create_full_h(h_vars):

        h_full = np.zeros(n_points)

        for i in range(half_n):

            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]

        return h_full

    def objective_pso(x):

        h_half = x

        h_full = create_full_h(h_half)

        corr = np.correlate(h_full, 1 - h_full, mode='full')

    # Initial symmetric h_half

    h_full_sym = h_full_initial.copy().copy()

    for i in range(half_n):

        h_full_sym[i] = h_full_sym[j] = (h_full_initial[i] + h_full_initial[j]) / 2

    current_sum = np.sum(h_full_sym)

    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:

    w = 0.7

    c1 = 1.5

    c2 = 1.5

    pbest_positions = positions.copy()
