import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    n_half = n_points // 2
    sum_target = n_points / 2.0
    m_pulses = 3  # Number of symmetric pulses

    np.random.seed(seed)
    
    # Initialize pulse parameters (center, width, amplitude)
    params_initial = []
    for i in range(m_pulses):
        c = np.random.uniform(0.2, 0.8)
        w = np.random.uniform(0.1, 0.3)
        a = np.random.uniform(0.2, 0.6)
        params_initial.append((c, w, a))

    # Convert pulse parameters to flat array
    def params_to_array(params_list):
        c, w, a = zip(*params_list)
        return np.array(c + w + a)

    # Convert flat array to pulse parameters
    def array_to_params(arr):
        c = arr[:m_pulses]
        w = arr[m_pulses:2*m_pulses]
        a = arr[2*m_pulses:]
        return list(zip(c, w, a))

    # Generate h from pulses
    def generate_h(params_list):
        h_half = np.zeros(n_half)
        x_half = np.arange(n_half) * dx
        for c, w, a in params_list:
            x_start = c - w / 2
            x_end = c + w / 2
            mask = (x_half >= x_start) & (x_half <= x_end)
            h_half[mask] += a
        h_full = np.zeros(n_points)
        h_full[:n_half] = h_half
        h_full[n_points-n_half:] = h_half
        sum_h = np.sum(h_full)
        if sum_h == 0:
            return np.zeros(n_points)
        h_scaled = h_full * (sum_target / sum_h)
        h_clipped = np.clip(h_scaled, 0.0, 1.0)
        sum_clipped = np.sum(h_clipped)
        if sum_clipped == 0:
            return h_clipped
        h_rescaled = h_clipped * (sum_target / sum_clipped)
        return np.clip(h_rescaled, 0.0, 1.0)

    # Objective function
    def objective(arr):
        params = array_to_params(arr)
        h = generate_h(params)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Bounds for pulse parameters
    bounds = []
    for _ in range(m_pulses):
        bounds.append((0.1*dx, 1.0 - 0.1*dx))  # center
        bounds.append((0.05, 0.5))           # width
        bounds.append((0.0, 1.0))            # amplitude

    # Initial guess from random pulses
    x0 = params_to_array(params_initial)
    x0 = np.clip(x0, [b[0] for b in bounds], [b[1] for b in bounds])

    # Simulated Annealing with temperature schedule
    current_x = x0.copy()
    current_val = objective(current_x)
    best_x = current_x.copy()
    best_val = current_val
    temp = 0.1
    alpha = 0.999

    while time.time() - start_time < budget_s - 200:
        perturbation = np.random.normal(0, 0.1, size=current_x.shape)
        new_x = current_x + perturbation
        new_x = np.clip(new_x, [b[0] for b in bounds], [b[1] for b in bounds])
        new_val = objective(new_x)

        # Acceptance probability
        if new_val < current_val or np.exp((current_val - new_val) / (temp + 1e-6)) > np.random.rand():
            current_x, current_val = new_x, new_val
            if new_val < best_val:
                best_x, best_val = new_x, new_val

        temp *= alpha

    # Local refinement
    result = minimize(
        fun=objective,
        x0=best_x,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 100, 'gtol': 1e-8}
    )
    if result.success:
        final_x = result.x
        final_val = result.fun
    else:
        final_x = best_x
        final_val = best_val

    # Generate final h
    best_params = array_to_params(final_x)
    h_final = generate_h(best_params)
    assert abs(np.sum(h_final) - sum_target) < 1e-6, "Integral constraint violated"
    assert np.all((h_final >= -1e-6) & (h_final <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_final.tolist(), final_val, n_points)
