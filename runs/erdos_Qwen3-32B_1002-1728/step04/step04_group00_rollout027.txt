import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    def compute_c5(h_full):
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = h_full_sym.sum()
    if abs(current_sum - (n_points / 2.0)) > 1e-6:
        scale = target_sum_half * 2.0 / current_sum
        h_full_sym *= scale
        h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half_current = h_full_sym[:half_n]

    best_h_half = h_half_current.copy()
    best_c5 = compute_c5(create_full_h(best_h_half))

    t_initial = 1.0
    alpha = 0.9995
    t_min = 1e-5
    temp = t_initial
    np.random.seed(int(seed))
    epsilon = 1e-6

    while time.time() - start_time < budget_s - 10:
        temp = max(temp * alpha, t_min)

        noise = np.random.normal(0, temp, half_n)
        h_proposal = best_h_half + noise
        h_proposal = np.clip(h_proposal, 0, 1)

        total = h_proposal.sum()
