def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)

    if current_sum != 0:
        h_initial = h_initial * (target_sum / current_sum)
    h_initial = np.clip(h_initial, 0.0, 1.0)
    
    pop_size = 50
    F = 0.7
    CR = 0.9
    rng = default_rng(seed)
    pop = np.zeros((pop_size, n_points))
    pop[0] = h_initial.copy()
    
    for i in range(1, pop_size):
        perturb = rng.normal(0, 0.05, n_points)
        candidate = h_initial + perturb
        candidate = np.clip(candidate, 0.0, 1.0)
        candidate_sum = candidate.sum()
        if candidate_sum == 0:
            candidate = np.full(n_points, target_sum / n_points)
        else:
            candidate = candidate * (target_sum / candidate_sum)
        candidate = np.clip(candidate, 0.0, 1.0)
        pop[i] = candidate
    
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(np.abs(corr)) * (2.0 / n_points)
    
    best_c5 = float('inf')
    best_h = None
    
    ngen = 0
    while time.time() - start_time < budget_s - 30:
        for i in range(pop_size):
            if time.time() - start_time >= budget_s - 30:
                break
            idxs = list(range(pop_size))
            rng.shuffle(idxs)
            r1, r2, r3 = idxs[:3]
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = pop[i].copy()
            crossover_mask = rng.random(n_points) < CR
            if np.random.rand() < CR:
                crossover_mask[0] = True
            trial[crossover_mask] = mutant[crossover_mask]
            trial_clip = np.clip(trial, 0.0, 1.0)
            trial_sum = trial_clip.sum()
            if trial_sum == 0:
                trial_final = np.full(n_points, target_sum / n_points)
            else:
