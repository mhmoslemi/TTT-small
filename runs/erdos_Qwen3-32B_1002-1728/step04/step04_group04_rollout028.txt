import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Normalize initial h
    h = h_full_initial.copy()
    current_sum = h.sum()
    if abs(current_sum) < 1e-10:
        h = np.zeros_like(h)
    else:
        h = (h / current_sum) * target_sum
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    np.random.seed(seed)
    lr = 0.01
    update_steps = 0

    while time.time() - start_time < budget_s - 5:
        corr = np.correlate(h, 1 - h, mode='full')
        max_index = np.argmax(corr)
        k_star = max_index - (n_points - 1)

        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            shifted_i = i + k_star
            if 0 <= shifted_i < n_points:
                grad[i] += 1 - h[shifted_i]
            shifted_i = i - k_star
            if 0 <= shifted_i < n_points:
                grad[i] += -h[shifted_i]
        grad *= dx

        h = h - lr * grad

        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 1e-6:
            h = (target_sum / current_sum) * h
            h = np.clip(h, 0, 1)
        else:
            h = np.zeros_like(h) * target_sum
            h = np.clip(h, 0, 1)

        # Update best solution
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

        update_steps += 1

        # Perturb and adjust learning rate every 100 steps
        if update_steps % 100 == 0:
            h += np.random.normal(0, 0.2, size=n_points)
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 1e-6:
                h = (target_sum / current_sum) * h
                h = np.clip(h, 0, 1)
            lr *= 0.95
            # Early escape if near timeout
            if time.time() - start_time > budget_s - 30:
                break

    # Ensure final projection
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 1e-6:
        h = (target_sum / current_sum) * h
        h = np.clip(h, 0, 1)

    assert abs(np.sum(h) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h >= -1e-6) & (h <= 1.0 + 1e-6)), "Out of bounds"
    return (h.tolist(), best_c5, n_points)
