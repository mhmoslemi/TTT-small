import numpy as np
import time

def compute_c5(h_full):
    n = len(h_full)
    h_fft = np.fft.fft(h_full, 2 * n - 1)
    one_minus_h_fft = np.fft.fft(1.0 - h_full, 2 * n - 1)
    corr_fft = h_fft * np.conj(one_minus_h_fft)
    corr = np.fft.ifft(corr_fft).real
    return np.max(corr) * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points

    # Initialize population with initial_h_values and perturbations
    population = []
    h = h_initial.astype(float)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * (n_points / 2.0 / h_sum)
    h = np.clip(h, 0, 1)
    population.append(h)

    np.random.seed(seed)
    for _ in range(49):
        perturbed = h_initial.astype(float)
        noise = np.random.normal(0, 0.1, size=n_points)
        perturbed += noise
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum != 0:
            perturbed = perturbed * (n_points / 2.0 / perturbed_sum)
        perturbed = np.clip(perturbed, 0, 1)
        population.append(perturbed)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stagnation_counter = 0
    mutation_rate = 0.1

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        current_best_index = np.argmin(fitness)
        current_best_value = fitness[current_best_index]

        # Update global best
        if current_best_value < best_c5:
            best_h = population[current_best_index].copy()
            best_c5 = current_best_value
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            if stagnation_counter > 50:
                mutation_rate = 0.3
            else:
                mutation_rate = 0.1

        # Create new population with elitism
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:5]]
        new_population = elites.copy()

        # Generate remaining population via crossover and mutation
        while len(new_population) < 50:
            # Tournament selection for parents
            tournament_size = 3
            tournament1 = np.random.choice(len(population), tournament_size)
            parent1_idx = sorted_indices[tournament1[np.argmin([fitness[i] for i in tournament1])]]
            tournament2 = np.random.choice(len(population), tournament_size)
            parent2_idx = sorted_indices[tournament2[np.argmin([fitness[i] for i in tournament2])]]

            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]

            # Crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])

            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.1)
                    child[i] = max(0.0, min(1.0, child[i]))
            child_sum = np.sum(child)
            if child_sum != 0:
                child = child * (n_points / 2.0 / child_sum)
            else:
                child = np.ones(n_points) * (n_points / 2.0) / n_points

    return (best_h.tolist(), best_c5, n_points)
