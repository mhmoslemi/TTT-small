    while time.time() - start_time < budget_s - 30:
        fitness = np.array([compute_c5(create_full_h(ind)) for ind in population])
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h_half = population[best_idx].copy()

        new_population = []
        for _ in range(population_size):
            candidates = np.random.randint(0, population_size, 3)
            fit_vals = [fitness[i] for i in candidates]
            winner_idx = np.argmin(fit_vals)
            parent = population[candidates[winner_idx]].copy()

            child = parent.copy()
            if np.random.rand() < 0.7:
                parent2 = population[np.random.randint(0, population_size)]
                cross_point = np.random.randint(1, half_n-1)
                child = np.zeros_like(parent)
                child[:cross_point] = parent[:cross_point]
                child[cross_point:] = parent2[cross_point:]
            
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.1, size=half_n)
                child += noise
                child = np.clip(child, 0, 1)
                current_sum = child.sum()
                if current_sum != 0:
                    child *= target_sum_half / current_sum
                child = np.clip(child, 0, 1)
            
            new_population.append(child)

        for idx in range(len(new_population)):
            h_half = new_population[idx].copy()
            bounds = [(0, 1)] * half_n
            cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
            res = minimize(
                fun=lambda x: compute_c5(create_full_h(x)), 
                x0=h_half, 
                bounds=bounds, 
                constraints=cons,
                method='SLSQP', 
                options={'maxiter': 30}
            )
            if res.success:
                new_population[idx] = res.x
    
        population = np.array(new_population)
    
    h_full = create_full_h(best_h_half)
    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    return (h_full.tolist(), best_c5, n_points)
