from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def objective(h_vec):
        h = h_vec.reshape(n_points)
        n = len(h)
        h_fft = np.fft.fft(h, n=2 * n)
        s_fft = np.fft.fft(1.0 - h, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    best_c5 = np.inf

    np.random.seed(seed)
