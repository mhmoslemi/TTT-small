import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(h_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_half
        h_full[half_n:] = h_half[::-1]
        return h_full

    def evaluate_candidate(h_half):
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Symmetrize and normalize initial h
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        h_full_sym[i] = h_full_sym[n_points - 1 - i] = (h_full_values[i] + h_full_values[n_points - 1 - i]) / 2
    current_sum = np.sum(h_full_sym)
    if abs(current_sum - (n_points / 2.0)) > 1e-6 and current_sum != 0:
        h_full_sym = h_full_sym * (target_sum_half * 2.0 / current_sum)
    h_full_sym = np.clip(h_full_sym, 0, 1)
    initial_h_half = h_full_sym[:half_n]

    # PSO parameters
    SWARM_SIZE = 50
    MAX_ITER = 10000  # Conservative upper bound
    COGNITIVE = 2.0
    SOCIAL = 2.0
    MAX_VELOCITY = 0.1

    np.random.seed(seed)

    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_scores = []
    gbest_index = 0
    gbest_score = float('inf')

    for _ in range(SWARM_SIZE):
        # Perturb initial with small random values
        h_half = initial_h_half.copy() + np.random.normal(0, 0.01, half_n)
        h_half = np.clip(h_half, 0, 1)
        h_half = h_half / np.sum(h_half) * target_sum_half if np.sum(h_half) != 0 else initial_h_half.copy()
        h_half = np.clip(h_half, 0, 1)

        score = evaluate_candidate(h_half)
        particles.append(h_half)
        velocities.append(np.zeros(half_n))
        pbest_positions.append(h_half.copy())
        pbest_scores.append(score)

    gbest_position = pbest_positions[0].copy()
    gbest_score = pbest_scores[0]

    # Identify the global best
    for i in range(SWARM_SIZE):
        if pbest_scores[i] < gbest_score:
            gbest_position = pbest_positions[i].copy()
            gbest_score = pbest_scores[i]

    while time.time() - start_time < budget_s - 10:
        inertia = 0.9 - (0.5 * ((time.time() - start_time) / budget_s))
        inertia = np.clip(inertia, 0.4, 0.9)

        for i in range(SWARM_SIZE):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_cognitive = COGNITIVE * r1 * (pbest_positions[i] - particles[i])
            vel_social = SOCIAL * r2 * (gbest_position - particles[i])
            velocities[i] = inertia * velocities[i] + vel_cognitive + vel_social
            velocities[i] = np.clip(velocities[i], -MAX_VELOCITY, MAX_VELOCITY)

            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)

            # Normalize sum
            current_sum = np.sum(particles[i])
            if current_sum != 0:
                particles[i] = (particles[i] / current_sum) * target_sum_half

            # Evaluate new position
            score = evaluate_candidate(particles[i])
