        corr = np.correlate(h_full, 1 - h_full, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:

            best_c5 = new_c5

        return np.sum(x) - target_sum_half

    while time.time() - start_time < budget_s - 30:  # Leave time to finalize

            fun=objective_local,
            x0=best_h_half,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(half_n)],

def run(seed=42, budget_s=1100, **kwargs):

    n_points = len(h_full_initial)

    target_sum_half = target_sum_full / 2.0  # Since full sum is 2 * sum of half for symmetric case

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = h_vars[i]
        return h_full

    # Preprocess initial h_half
    h_full_sym = h_full_initial.copy()
    for i in range(half_n):
        j = n_points - 1 - i
        avg = (h_full_initial[i] + h_full_initial[j]) / 2
        h_full_sym[i] = h_full_sym[j] = avg
    current_sum = np.sum(h_full_sym)
    if current_sum != 0:
        scale = target_sum_full / current_sum
        h_full_sym *= scale
    h_full_sym = np.clip(h_full_sym, 0, 1)
    h_half = h_full_sym[0:half_n]
    h_half_initial = h_half.copy()
    total = np.sum(h_half_initial)
    if total != 0:
        h_half_initial = h_half_initial * (target_sum_half / total)
        h_half_initial = np.clip(h_half_initial, 0, 1)

    best_h_half = h_half_initial.copy()
    best_c5 = 0
    h_full = create_full_h(best_h_half)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    best_c5 = np.max(corr) * dx

    # Differential Evolution Objective Function
    def objective_de(h_vars):
        h_clip = np.clip(h_vars, 0, 1)
        sum_h = np.sum(h_clip)
        if sum_h == 0 or np.isnan(sum_h):
            return 1e6
        scaled_h = h_clip * (target_sum_half / sum_h)
        if np.any(scaled_h < 0) or np.any(scaled_h > 1):
            return 1e6
        h_f = create_full_h(scaled_h)
        corr = np.correlate(h_f, 1 - h_f, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(half_n)]
    np.random.seed(seed)

    # Run DE with time bounding
    while time.time() - start_time < budget_s - 60:
        try:
            result_de = differential_evolution(
                objective_de, bounds,
                strategy='best1bin',
                maxiter=5,
                popsize=15,
                mutation=(0.5, 0.8),
                recombination=0.7,
                seed=np.random.randint(0, 10000),
                tol=1e-5,
                disp=False
            )
        except:
            result_de = {'x': best_h_half, 'fun': best_c5}

        new_h_half = result_de.x
        new_h_half_clip = np.clip(new_h_half, 0, 1)
        sum_h = np.sum(new_h_half_clip)
        if sum_h == 0:
            continue

        new_h_half_scaled = new_h_half_clip * (target_sum_half / sum_h)
        new_h_half_scaled = np.clip(new_h_half_scaled, 0, 1)
        h_f = create_full_h(new_h_half_scaled)
        corr = np.correlate(h_f, 1 - h_f, mode='full')
        new_c5 = np.max(corr) * dx

        if new_c5 < best_c5:
            best_h_half = new_h_half_scaled
            best_c5 = new_c5

        if time.time() - start_time >= budget_s - 30:
            break

    # Local Optimization
    def objective_local(x):
        h_clip = np.clip(x, 0, 1)
        total = np.sum(h_clip)
        if total == 0:
            return 1e6
        scaled = h_clip * (target_sum_half / total)
        h_full = create_full_h(scaled)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

        result_local = minimize(
            fun=objective_local,
            x0=best_h_half,
            method='L-BFGS-B',
            bounds=[(0, 1)] * half_n,
            options={'maxiter': 80, 'ftol': 1e-8, 'gtol': 1e-6}

            if total_opt != 0:

                x_opt_scaled = np.clip(x_opt_scaled, 0, 1)
                h_f_opt = create_full_h(x_opt_scaled)
                corr = np.correlate(h_f_opt, 1 - h_f_opt, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_h_half = x_opt_scaled
                    best_c5 = new_c5

            break

    assert abs(np.sum(h_full) - target_sum_full) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1 + 1e-6))
