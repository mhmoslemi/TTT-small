        if abs(new_sum - target_sum_half) < 1e-6:
            return scaled_clipped
        adjusted = scaled_clipped.copy()
        delta = target_sum_half - new_sum
        adjusted[-1] += delta
        adjusted = np.clip(adjusted, 0, 1)
        return adjusted

    def objective(h_vars):
        h_half = adjust_h_half(h_vars)
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds_per_var = [(0.0, 1.0) for _ in range(half_n)]

    # Initial guess based on symmetric adjustment and normalization
    initial_h_half = (h_full_initial[:half_n] + h_full_initial[-half_n:][::-1]) / 2.0
    total = np.sum(initial_h_half)
    if total != 0:
        initial_h_half = initial_h_half * (target_sum_half / total)
    initial_h_half = np.clip(initial_h_half, 0, 1)

    # Global exploration using differential evolution
    de_result = differential_evolution(
        objective,
        bounds_per_var,
        seed=seed,
        mutation=(0.5, 1),
        popsize=15,
        maxiter=50,
        tol=1e-6,
        strategy='best1bin',
        polish=False
    )

    best_h_half = de_result.x
    best_c5 = de_result.fun

    # Refine with L-BFGS-B
    def local_objective(x):
        h_half = adjust_h_half(x)
        h_full = create_full_h(h_half)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    lbfgs_result = minimize(
        local_objective,
        x0=best_h_half,
        method='L-BFGS-B',
        bounds=bounds_per_var,
        options={'maxiter': 600, 'gtol': 1e-8, 'ftol': 1e-10}
    )

    refined_h_half = lbfgs_result.x
    refined_c5 = lbfgs_result.fun

    if refined_c5 < best_c5:
        best_h_half = refined_h_half
        best_c5 = refined_c5

    # Further multi-perturbation refinement
    while (time.time() - start_time < budget_s - 20):
        perturbed = best_h_half + np.random.uniform(-0.2, 0.2, size=half_n)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = adjust_h_half(perturbed)

        perturbed_result = minimize(
            local_objective,
            x0=perturbed,
            method='L-BFGS-B',
            bounds=bounds_per_var,
            options={'maxiter': 300, 'gtol': 1e-8, 'ftol': 1e-10}
        )

        if perturbed_result.success:
            c5_new = perturbed_result.fun
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h_half = perturbed_result.x

    h_full = create_full_h(adjust_h_half(best_h_half))

    assert abs(np.sum(h_full) - n_points / 2.0) < 1e-6, "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"

    return (h_full.tolist(), best_c5, n_points)
