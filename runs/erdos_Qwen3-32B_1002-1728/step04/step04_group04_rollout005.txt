            if abs(total_new - target_sum) > 1e-6:
                h_new = (h_new / total_new) * target_sum
                h_new = np.clip(h_new, 0, 1)
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        elif abs(np.sum(perturbed) - target_sum) < 1e-6:
            c5_new = objective(perturbed)
            if c5_new < best_c5:
                best_h = perturbed.copy()
                best_c5 = c5_new

        # Adaptive perturbation scaling with diversity kicks
        if np.random.rand() < 1/50:  # 2% diversity boost
            perturbation_scale = min(perturbation_scale * 2, 0.5)
        else:
            perturbation_scale *= perturbation_decay_rate
        perturbation_scale = max(perturbation_scale, 0.005)

    h_full = best_h
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Out of bounds"
    return (h_full.tolist(), best_c5, n_points)
