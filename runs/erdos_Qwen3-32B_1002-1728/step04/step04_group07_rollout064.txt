def run(seed=42, budget_s=1100, **kwargs):

    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0

    def create_full_h(left_half):
        h_full = np.zeros(n_points)
        h_full[:half_n] = left_half
        h_full[half_n:] = left_half[::-1]
        h_full = np.clip(h_full, 0, 1)
        return h_full

    def rescale(left_vars):
        left_vars = np.clip(left_vars, 0, 1)
        total = np.sum(left_vars)
        if total == 0:
            left_vars = np.zeros_like(left_vars)
            total = 1e-10
        left_vars = left_vars * (target_sum_half / total)
        left_vars = np.clip(left_vars, 0, 1)
        return left_vars

    left_half_initial = h_full_initial[:half_n]
    left_half_initial = rescale(left_half_initial)
    best_solution = left_half_initial.copy()
    best_c5 = np.inf

    def de_objective(left_vars):
        left_scaled = rescale(left_vars)
        h_full = create_full_h(left_scaled)
        corr = np.correlate(h_full, 1 - h_full, mode='full') * dx
        return float(np.max(corr))

    bounds = [(0.0, 1.0)] * half_n
    de_time_limit = 650  # 650s for global search

    # Differential Evolution global search
    if time.time() - start_time < de_time_limit:
        result_de = differential_evolution(
            de_objective, bounds, 
            mutation=(0.5, 1.0), 
            recombination=0.9, 
            tol=1e-6, 
            maxiter=100,

            strategy='best1bin',
            polynomial_order=2,
            workers=1,
            seed=seed
        )
        if result_de.success:
            best_solution = result_de.x
            best_c5 = result_de.fun

    def wrapped_obj(x):
        return de_objective(x)

    cons_slsqp = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum_half})
    if time.time() - start_time < budget_s - 10:

        if result_local.success:

            best_c5 = de_objective(best_solution)

    # Final processing
    h_full = create_full_h(best_solution)
