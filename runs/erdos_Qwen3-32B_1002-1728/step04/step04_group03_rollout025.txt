    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

    # Clip and evaluate function safely
    def bounded_objective(x):
        return objective(np.clip(x.reshape(n_points), 0, 1))

    # Optimize from the structured initial guess
    x0 = h_quaternary.reshape(n_points)
    result = minimize(
        bounded_objective,
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    # Final validation and return of best result
    h_full = np.clip(result.x.reshape(n_points), 0, 1)
    best_c5 = bounded_objective(h_full)

    assert np.isclose(np.sum(h_full), target_sum, atol=1e-6), "Constraint violation"
    assert np.all(np.isfinite(h_full)), "NaN/Inf values present"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Bounds violation"

    return (h_full.tolist(), best_c5, n_points)
