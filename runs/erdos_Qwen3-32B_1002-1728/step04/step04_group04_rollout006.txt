    h_current = h_initial.copy()
    current_sum = np.sum(h_current)
    h_current = (h_current / current_sum) * target_sum if abs(current_sum) > 1e-10 else np.zeros_like(h_current)
    h_current = np.clip(h_current, 0, 1)
    best_h = h_current.copy()
    best_c5 = objective(best_h)

    X = [best_h.copy()]
    y = [best_c5]

    kernel = RBF(length_scale=2.0) + WhiteKernel(noise_level=1e-10)
    gp = GaussianProcessRegressor(kernel=kernel, alpha=1e-4, n_restarts_optimizer=0, normalize_y=True)

    np.random.seed(seed)
    perturbation_scale = 0.2

    while time.time() - start_time < budget_s - 30:
        num_candidates = 100
        candidates = []
        for _ in range(num_candidates):
            perturbed = best_h + np.random.normal(0, perturbation_scale, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed_sum = np.sum(perturbed)
            if perturbed_sum != 0:
                perturbed = (perturbed / perturbed_sum) * target_sum
            perturbed = np.clip(perturbed, 0, 1)
            candidates.append(perturbed)
        candidates = np.array(candidates)

        if candidates.size > 0:
            y_pred, y_std = gp.predict(candidates, return_std=True)
            improvement = (y_pred.ravel() - best_c5)
            with np.errstate(divide='ignore', invalid='ignore'):
                Z = improvement / (y_std + 1e-6)
                EI = (improvement * norm.cdf(Z)) + (y_std * norm.pdf(Z))
                EI[improvement <= 0] = 0.0

            best_candidate_idx = np.argmax(EI)
            selected_candidate = candidates[best_candidate_idx]

            c5_candidate = objective(selected_candidate)
            selected_candidate = np.clip(selected_candidate * (target_sum / np.sum(selected_candidate)), 0, 1)

            X.append(selected_candidate.copy())
            y.append(c5_candidate)

            if c5_candidate < best_c5 and np.all(selected_candidate >= 0) and np.all(selected_candidate <= 1.0):
                best_h = selected_candidate.copy()
                best_c5 = c5_candidate

            perturbation_scale *= 0.99

        try:
            X_array = np.array(X).reshape(-1, n_points)
            y_array = np.array(y).reshape(-1, 1)
            if X_array.size > 0:
                gp.fit(X_array, y_array)
        except Exception:
            pass

    h_full = best_h
    return (h_full.tolist(), best_c5, n_points)
