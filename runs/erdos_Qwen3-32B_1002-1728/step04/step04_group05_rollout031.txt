def create_full_h(y, n_points):
    """Construct symmetric h from first half y"""
    half_n = len(y)
    h_full = np.zeros(n_points)
    h_full[:half_n] = y
    h_full[-half_n:] = y[::-1]
    return h_full

def evaluate_candidate(y, target_sum_half, dx, n_points):
    """Evaluate C5 for a symmetric y"""
    h_full = create_full_h(y, n_points)
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
