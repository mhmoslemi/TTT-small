from scipy.optimize import differential_evolution

def run(seed=42, budget_s=1100, **kwargs):

    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # sum(left half) = n_points/4

    def create_full_h(left_vars):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = left_vars[i]
        return h_full

    # Initial symmetric normalization
    initial_left = h_initial[:half_n].copy()
    initial_sum = np.sum(initial_left)
    if initial_sum != 0:
        initial_left = initial_left * (target_sum_half / initial_sum)
    initial_left = np.clip(initial_left, 0, 1)

    def de_objective(x_vars):
        x = np.array(x_vars)
        s = np.sum(x)
        if s == 0:
            return np.inf
        x = x * (target_sum_half / s)
        x = np.clip(x, 0, 1)
        h_full = create_full_h(x)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    bounds = [(0.0, 1.0)] * half_n

    # Global exploration with DE
    result_de = differential_evolution(
        de_objective, bounds,
        strategy='best1bin', maxiter=50, popsize=15,
        mutation=(0.5, 1), recombination=0.9,
        seed=seed, init='sobol', polish=False, workers=1
    )

    best_left = result_de.x
    s = np.sum(best_left)
    best_left = best_left * (target_sum_half / s)
    best_left = np.clip(best_left, 0, 1)
    h_full_de = create_full_h(best_left)
    current_c5 = np.max(np.correlate(h_full_de, 1 - h_full_de, 'full')) * dx

    # Local refinement with CVXPY
    h_var = cp.Variable(n_points)
    t_var = cp.Variable()
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]

    # Add all possible shift constraints for full correlation
    for k in range(-n_points + 1, n_points):
        overlap_expr = 0.0
        for i in range(n_points):
            i_shifted = i + k
            if 0 <= i_shifted < n_points:

        overlap_expr *= dx

    problem = cp.Problem(objective, constraints)
