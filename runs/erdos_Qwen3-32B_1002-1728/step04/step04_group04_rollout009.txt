        grad[idx] += 1 - h[idx + k]
        # Compute term2: -h[i -k]
        valid_minus = (np.arange(n) - k) < n
        valid_minus &= (np.arange(n) - k) >= 0
        idx_minus = np.arange(n)[valid_minus]
        grad[idx_minus] -= h[idx_minus - k]
        return grad * dx

    def project(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if abs(current_sum) < 1e-10:
            return h_clipped.copy()
        h_scaled = h_clipped * (target_sum / current_sum)
        h_proj = np.clip(h_scaled, 0, 1)
        current_sum = np.sum(h_proj)
        if abs(current_sum - target_sum) > 1e-6:
            non_clamped = (h_proj > 0) & (h_proj < 1)
            if np.sum(non_clamped) > 0:
                needed = target_sum - current_sum
                delta = needed / np.sum(non_clamped)
                h_proj[non_clamped] += delta
                h_proj = np.clip(h_proj, 0, 1)
        return h_proj

    # Normalize initial guess
    h_current = h_full_initial.copy()
    h_current = project(h_current, target_sum)
    best_h = h_current.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    np.random.seed(seed)
    learning_rate = 0.05  # Initial learning rate

    while time.time() - start_time < budget_s:

        # Evaluate current C5
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        idx_max = np.argmax(corr)
        k = idx_max - (n_points - 1)
        gradient = compute_gradient(best_h, k)

        # Perform gradient step
        h_new = best_h - learning_rate * gradient
        h_proj = project(h_new, target_sum)

        # Check if the new solution is better
        corr_new = np.correlate(h_proj, 1 - h_proj, mode='full')
        c5_new = np.max(corr_new) * dx

        # Perturb and restart occasionally
        if np.random.rand() < 0.05:
            perturbation = np.random.normal(0, 0.1, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = project(h_perturbed, target_sum)
            corr_p = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_p = np.max(corr_p) * dx
            if c5_p < best_c5:
                best_h = h_perturbed
                best_c5 = c5_p

        # Update best solution
        if c5_new < best_c5:
            best_h = h_proj
            best_c5 = c5_new

        # Decrease learning rate
        learning_rate *= 0.995
        if time.time() - start_time > budget_s - 30:  # Last 30 seconds: small steps
            learning_rate *= 0.8

        # Enforce time constraint
        if time.time() - start_time > budget_s:
            break

    # Final checks
    assert abs(np.sum(best_h) - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
