import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    h_current = h_full_initial.copy().astype(float)
    current_sum = np.sum(h_current)
    if abs(current_sum) < 1e-10:
        h_current[:] = 0.0
    else:
        h_current[:] = (h_current / current_sum) * target_sum
    h_current = np.clip(h_current, 0, 1)
    best_h = h_current.copy()
    best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx

    T = 1.0
    alpha = 0.999
    min_perturb = 3
    max_perturb = 7

    while time.time() - start_time < budget_s:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 60:
            alpha = 0.98
        elif time_left < 120:
            alpha = 0.99

        current_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx

        n_perturb = np.random.randint(min_perturb, min(max_perturb, n_points) + 1)
        perturb_indices = np.random.choice(n_points, n_perturb, replace=False)
        h_candidate = h_current.copy()
        delta = np.random.uniform(-0.2, 0.2, size=n_perturb)
        h_candidate[perturb_indices] += delta
        h_candidate = np.clip(h_candidate, 0, 1)

        current_total = np.sum(h_candidate)
        if abs(current_total) > 1e-10:
            h_candidate = (h_candidate / current_total) * target_sum
        else:
            h_candidate[:] = 0.0
        h_candidate = np.clip(h_candidate, 0, 1)

        candidate_c5 = np.max(np.correlate(h_candidate, 1 - h_candidate, mode='full')) * dx
        delta_c5 = candidate_c5 - current_c5

        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h_current[:] = h_candidate[:]
            if candidate_c5 < best_c5:
                best_h[:] = h_candidate[:]
                best_c5 = candidate_c5

        T *= alpha

        if T < 1e-4 or time.time() - start_time + 30 > budget_s:
            break

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violation"
    assert np.all((best_h >= 0) & (best_h <= 1)), "Invalid range"
    return (best_h.tolist(), best_c5, n_points)
