import numpy as np
import time

def perturb_y(y, mutation_strength, target_sum_y):
    random_direction = np.random.normal(0, 1, y.shape) - np.mean(y)
    perturbation = mutation_strength * random_direction
    y_perturbed = y + perturbation
    y_clamped = np.clip(y_perturbed, 0.0, 1.0)
    sum_clamped = np.sum(y_clamped)
    if sum_clamped < 1e-10:
        return y_clamped.copy()
    scaling_factor = target_sum_y / sum_clamped
    y_scaled = y_clamped * scaling_factor
    y_final = np.clip(y_scaled, 0.0, 1.0)
    return y_final.copy()

def evaluate_candidate(y, dx, n_points):
    h_full = np.zeros(n_points)
    for i in range(len(y)):
        h_full[i] = h_full[n_points - 1 - i] = y[i]
    corr = np.correlate(h_full, 1 - h_full, mode='full')
    return np.max(corr) * dx
