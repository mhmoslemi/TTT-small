def compute_c5(h_full, dx):
    # compute max convolution
    n = len(h_full)
    h_fft = np.fft.fft(h_full, 2*n - 1)
    one_h_fft = np.fft.fft(1.0 - h_full, 2*n - 1)

    corr_fft = h_fft * np.conj(one_h_fft)
    corr = np.fft.ifft(corr_fft).real
    # in mode 'full', the correlation shifts are varying. The maximum value is the maximum of corr[0], ..., corr[2n-2]
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Normalize initial h
    h_initial = np.clip(h_initial, 0, 1)
    h_initial_sum = h_initial.sum()
    if not np.allclose(h_initial_sum, target_sum, atol=1e-6) and h_initial_sum != 0:
        h_initial = h_initial * target_sum / h_initial_sum
    h_initial = np.clip(h_initial, 0, 1)

    # Initialize population
    population = []
    population.append(h_initial.copy())
    np.random.seed(seed)
    for _ in range(49):
        perturbation = np.random.normal(0, 0.05, n_points)
        indiv = h_initial + perturbation
        indiv = np.clip(indiv, 0, 1)
        indiv_sum = indiv.sum()
        if indiv_sum != 0:
            indiv = indiv * target_sum / indiv_sum
        indiv = np.clip(indiv, 0, 1)
        population.append(indiv)
    population = np.array(population)

    # Fitness function
    def evaluate(individual):
        n = len(individual)
        h_fft = np.fft.fft(individual, 2 * n - 1)
        one_h_fft = np.fft.fft(1.0 - individual, 2 * n - 1)
        corr = np.fft.ifft(h_fft * np.conj(one_h_fft)).real
        return np.max(corr) * dx

    # Evaluate initial population
    fitness = np.array([evaluate(indiv) for indiv in population])
    best_idx = np.argmin(fitness)
    best_indiv = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    # GA parameters
    pop_size = 50
    tournament_size = 3
    num_elites = 5
    mu_rate = 0.1
    cross_rate = 0.7
    stagnation = 0
    no_improve_threshold = 50

    while time.time() - start_time < budget_s - 10:
        # Tournament selection
        parents = []
        for _ in range(pop_size):
            conts = np.random.choice(pop_size, tournament_size, replace=False)
            best_cont = conts[np.argmin(fitness[conts])]
            parents.append(population[best_cont])
        parents = np.array(parents)

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if i + 1 < pop_size and np.random.rand() < cross_rate:
                p1, p2 = parents[i], parents[i+1]
                cp = np.random.randint(1, n_points-1)
                o1 = np.concatenate((p1[:cp], p2[cp:]))

                s = vec.sum()
                if s != 0:
                    vec *= target_sum / s
                vec = np.clip(vec, 0, 1)
            offspring.extend([o1, o2])

        for child in offspring:

                    child[i] += noise
                    child[i] = np.clip(child[i], 0, 1)
            # Renormalize
            s = child.sum()
            if s != 0:
                child *= target_sum / s
            child = np.clip(child, 0, 1)

        # Evaluate offspring
