            particles[i] = normalize(particles[i])
            new_fitness = eval_fitness(particles[i])
            if new_fitness < personal_best_fitness[i]:
                personal_best[i] = particles[i].copy()
                personal_best_fitness[i] = new_fitness
                if new_fitness < global_best_fitness:
                    global_best[:] = particles[i].copy()
                    global_best_fitness = new_fitness
        iteration += 1

    h_full = global_best
    c5_bound = global_best_fitness
    assert abs(np.sum(h_full) - target_sum) < 1e-6, "Sum constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Bounds violated"
    return (h_full.tolist(), c5_bound, n_points)
