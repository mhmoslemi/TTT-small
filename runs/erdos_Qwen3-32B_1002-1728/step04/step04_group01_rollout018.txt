import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values).copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    # Normalize initial guess
    h_initial = np.clip(h_initial * target_sum / h_initial.sum(), 0.0, 1.0)

    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h, n_points)

    np.random.seed(seed)
    temp = 1.0
    temp_min = 1e-4
    cooling_rate = 0.995
    perturb_sigma = 0.08

    # Simulated Annealing Phase
    while time.time() - start_time < budget_s - 90:
        h_perturbed = best_h + np.random.normal(0, perturb_sigma, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed *= target_sum / h_perturbed.sum()  # Rescale to constraint
        h_perturbed = np.clip(h_perturbed, 0, 1)  # Final clipping

        current_c5 = evaluate_c5(best_h, n_points)
        new_c5 = evaluate_c5(h_perturbed, n_points)
        delta_c = current_c5 - new_c5

        if delta_c > 0 or np.exp(delta_c / max(temp, 1e-6)) > np.random.rand():
            if evaluate_c5(h_perturbed, n_points) < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = evaluate_c5(best_h, n_points)

        temp *= cooling_rate

    # Local Refinement Phase
    def obj_func(x):
        return evaluate_c5(x.reshape(n_points), n_points)

    bounds = [(0.0, 1.0)] * n_points
    constraint = {'type': 'eq', 'fun': lambda x: x.sum() - target_sum}

    result = minimize(
        obj_func,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 50, 'ftol': 1e-8}

    if result.success:

        final_c5 = evaluate_c5(final_h, n_points)
        if final_c5 < best_c5:
            best_h, best_c5 = final_h, final_c5

    assert abs(best_h.sum() - target_sum) < 1e-6
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))

    return best_h.tolist(), best_c5, n_points
