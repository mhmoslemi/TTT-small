```python
import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.7
    tournament_size = 5

    def normalize(h):
        h = np.clip(h, 0, 1).copy()
        total = h.sum()
        if abs(total) < 1e-10:
            return np.zeros_like(h)
        return (h / total) * target_sum

    def generate_individual(base, noise_scale):
        perturbed = base + np.random.normal(0, noise_scale, n_points)
        return normalize(perturbed)

    def objective(h):
        h_norm = normalize(h)
        corr = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(corr) * dx

    # Initialize population
    population = [h_initial.copy()]
    for _ in range(1, pop_size):
        population.append(generate_individual(h_initial, 0.1))
    population = np.array(population)

    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        idx = np.argmin(fitness)
        current_best = population[idx]
        current_c5 = fitness[idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()

        # Tournament selection
        def select_parent():
            indices = np.random.choice(pop_size, tournament_size)
            return population[indices[np.argmin(fitness[indices])]]

        new_population = [best_h]  # Elitism

        while len(new_population) < pop_size:
            parent1 = select_parent()
            parent2 = select_parent()

            # Crossover
            if random.random() < crossover_rate:
                cross = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:cross], parent2[cross:]])
                child2 = np.concatenate([parent2[:cross], parent1[cross:]])
            else:
                child1, child2 = parent1, parent2

            # Mutation
            for child in [child1, child2]:
                if random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child = normalize(child + noise)
                new_population.append(normalize(child))
            if len(new_population) >= pop_size:
                break

        population = np.array(new_population[:pop_size])

    # Final checks and return
    h_final = normalize(best_h)
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr_final) * dx
    return (h_final.tolist(), final_c5, n_points)
```
