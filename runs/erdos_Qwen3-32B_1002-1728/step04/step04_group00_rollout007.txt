        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < math.exp(-delta / T):
            # Accept into current
            current_h[:] = new_h_half
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h[:] = new_h_half
                best_c5 = new_c5

        # Cool
        T *= alpha
        if T < T_min:
            break

    best_h_full = make_symmetric(best_h)
    assert abs(np.sum(best_h_full) - n_points / 2) < 1e-6
    assert np.all((best_h_full >= -1e-6) & (best_h_full <= 1 + 1e-6))

    return (best_h_full.tolist(), best_c5, n_points)
