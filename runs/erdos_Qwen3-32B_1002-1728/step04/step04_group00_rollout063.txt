import numpy as np
import time
import math
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum_half = n_points / 4.0  # since sum(h_full) = n_points / 2

    def create_full_h(h_half):
        return np.concatenate((h_half, h_half[::-1]))

    h_half_initial = h_full_initial[:half_n].copy()
    sum_h_half = np.sum(h_half_initial)
    if sum_h_half != 0:
        scale_factor = target_sum_half / sum_h_half
        h_half_initial = h_half_initial * scale_factor
        h_half_initial = np.clip(h_half_initial, 0, 1)

            h_half_initial = h_half_initial * (target_sum_half / sum_clipped)
            h_half_initial = np.clip(h_half_initial, 0, 1)

    random.seed(seed)
