import numpy as np
from scipy.optimize import differential_evolution
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def create_full_h(h_vars):
        h_full = np.zeros(n_points)
        h_full[:half_n] = h_vars
        h_full[half_n:] = h_vars[::-1]
        return h_full

    def objective(x):
        h_full = create_full_h(x)
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    def constraint_sum(x):
        return np.sum(create_full_h(x)) - target_sum

    # Symmetrize initial h
    h_left = h_initial[:half_n].copy()
    total = np.sum(h_left)
    if total > 1e-6:
        h_left = (h_left / total) * (target_sum / 2)
    h_left = np.clip(h_left, 0, 1)

    bounds = [(0, 1)] * half_n
    max_gen = int((budget_s * 0.5) // 10)  # Reserve budget for CVXPY

    # Global optimization with DE
    de_result = differential_evolution(
        func=objective,
        bounds=bounds,
        strategy='best1bin',
        maxiter=max_gen,
        polish=False,
        recombination=0.9,
        mutation=(0.5, 1.0),
        tol=1e-4,
        popsize=50,
        constraints=({'type': 'eq', 'fun': constraint_sum},),
        seed=seed
    )
    
    if time.time() - start_time > budget_s - 300:
        best_h_vals = h_left
    else:
        best_h_vals = de_result.x

    # Local refinement with Convex Optimization
    h_vars = cp.Variable(half_n)
    obj = cp.Minimize(objective(h_vars))
    constraints = [0 <= h_vars, h_vars <= 1, constraint_sum(h_vars) == 0]
    cvx_prob = cp.Problem(obj, constraints)
    
    if time.time() - start_time < budget_s - 10:
        cvx_prob.solve(solver=cp.SCIP, max_seconds=300 - 10, verbose=False)
        best_h_vals = h_vars.value.flatten()

    h_full = create_full_h(np.clip(best_h_vals, 0, 1))
    final_corr = np.correlate(h_full, 1 - h_full, mode='full')
    c5_bound = np.max(final_corr) * dx

    return (h_full.tolist(), c5_bound, n_points)
