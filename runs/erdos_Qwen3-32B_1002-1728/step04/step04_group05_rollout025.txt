import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)
    half_n = n_points // 2
    target_sum_half = n_points / 4.0
    dx = 2.0 / n_points

    def create_full_h(y):
        h_full = np.zeros(n_points)
        for i in range(half_n):
            h_full[i] = h_full[n_points - 1 - i] = y[i]
        if n_points % 2 == 1:
            h_full[half_n] = h_full[half_n]
        return h_full

    def normalize_y(y):
        s = np.sum(y)
        if abs(s) < 1e-6:
            return np.zeros_like(y)
        return (y * target_sum_half) / s

    def evaluate(y):
        h_full = create_full_h(normalize_y(y))
        corr = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(corr) * dx

    # Initialize particles with symmetric constraint
    np.random.seed(seed)
    y_init = h_initial[:half_n].copy()
    y_init_sum = np.sum(y_init)
    if abs(y_init_sum) > 1e-6:
        y_init = y_init * (target_sum_half / y_init_sum)
    y_init = np.clip(y_init, 0, 1)

    # PSO parameters
    swarm_size = 50
    max_iter = 10000
    w_max, w_min = 0.9, 0.4
    c1, c2 = 2.0, 2.0
    particles = []
    velocities = []
    personal_best = []
    pbest_scores = []

    # Initialize swarm
    for _ in range(swarm_size):
        y = y_init + 0.05 * np.random.randn(half_n)
        y = np.clip(y, 0, 1)
        y = normalize_y(y)
        particles.append(y)
        velocities.append(0.01 * np.random.randn(half_n))
        personal_best.append(y.copy())
        pbest_scores.append(evaluate(y))

    gbest_idx = np.argmin(pbest_scores)
    gbest = personal_best[gbest_idx].copy()
    best_c5 = pbest_scores[gbest_idx]

    end_time = start_time + budget_s
    for iteration in range(max_iter):
        if time.time() > end_time - 5:
            break
        t_left = end_time - time.time()
        if t_left / budget_s < 0.01:  # Less than 1% time left, break
            break

        # Update inertia weight
        w = w_max - (w_max - w_min) * (iteration / max_iter)

        for i in range(swarm_size):
            y = particles[i]
            v = velocities[i]
            pbest = personal_best[i]

            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v +
                 c1 * r1 * (pbest - y) +
                 c2 * r2 * (gbest - y))
            velocities[i] = np.clip(v, -0.1, 0.1)

            # Update position
            y_new = y + velocities[i]
            y_new = np.clip(y_new, 0, 1)
            y_new = normalize_y(y_new)
            particles[i] = y_new

            # Evaluate and update personal best

            if score < pbest_scores[i]:
                personal_best[i] = y_new.copy()

                    best_c5 = score

        # Early stopping if converged
