import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = np.array(initial_h_values)
    n_points = len(h_init)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    h_current = h_init.astype(float)

    h_current = np.clip(h_current, 0, 1)
    best_h = h_current.copy()

    current_c5 = best_c5

    initial_T = 2.0
    alpha = 0.995
    T = initial_T
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        num_perturb = max(5, int(0.05 * n_points))
        idx = np.random.choice(n_points, num_perturb, replace=False)
        delta = np.random.uniform(-0.2, 0.2, num_perturb)
        h_candidate = h_current.copy()
        h_candidate[idx] += delta
        h_candidate = np.clip(h_candidate, 0, 1)
        total = h_candidate.sum()

            scale = target_sum / total
            h_candidate = np.clip(h_candidate * scale, 0, 1)

        c5_candidate = evaluate_erdos_solution(h_candidate)
        delta_energy = c5_candidate - current_c5

        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):

            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_h = h_current.copy()
                best_c5 = c5_candidate
        T *= alpha

    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Constraint violated"
