            options={'maxiter': 50, 'ftol': 1e-8, 'disp': False}
        )
        if result.success:
            new_h = result.x
            new_c5 = compute_c5(new_h)
            if new_c5 < best_c5:
                best_h[:] = new_h[:]
                best_c5 = new_c5

    # Final validation
    assert np.isclose(np.sum(best_h), desired_sum, atol=1e-6)
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6))
    return (best_h.tolist(), best_c5, n_points)
