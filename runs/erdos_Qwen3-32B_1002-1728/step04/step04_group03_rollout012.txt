import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h_full_initial = np.array(initial_h_values)
    n_points = len(h_full_initial)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # Generate initial h with four groups
    s_A = n_points // 4
    s_B = s_A
    s_C = s_A
    s_D = n_points - s_A * 3

    group0 = np.full(s_A, 0.0)
    group1 = np.full(s_B, 1.0 / 3.0)
    group2 = np.full(s_C, 2.0 / 3.0)
    if s_D > 0:
        sum_first_three = (0.0 * s_A + (1/3.0) * s_B + (2/3.0) * s_C)
        val_D = (target_sum - sum_first_three) / s_D
        groupD = np.full(s_D, val_D)
        h0 = np.concatenate([group0, group1, group2, groupD])
    else:
        h0 = np.concatenate([group0, group1, group2])
    h0 = np.clip(h0, 0.0, 1.0)
    current_sum = np.sum(h0)
    if not np.isclose(current_sum, target_sum):
        h0 = h0 * (target_sum / (current_sum + 1e-10)) if current_sum != 0 else h0
    h0 = np.clip(h0, 0.0, 1.0)

    def objective_fft(h_vars):
        h_full = np.array(h_vars).flatten()
        n = len(h_full)
        h_fft = np.fft.fft(h_full, n=2 * n)
        s_fft = np.fft.fft(1.0 - h_full, n=2 * n)
        cross_corr = np.fft.ifft(h_fft * np.conj(s_fft)).real
        return np.max(cross_corr) * dx

    def constraint_sum(h):
        return np.sum(h) - target_sum

    cons = ({'type': 'eq', 'fun': constraint_sum},)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    best_h = h0.copy()
    best_c5 = objective_fft(best_h)

    np.random.seed(seed)

    while time.time() - start_time < budget_s - 30:
        perturbed = np.random.normal(loc=0.02, scale=0.1, size=n_points) + best_h
        perturbed = np.clip(perturbed, 0.0, 1.0)
        perturbed_sum = np.sum(perturbed)
        if perturbed_sum > 1e-8:
            perturbed = perturbed * (target_sum / perturbed_sum)
        perturbed = np.clip(perturbed, 0.0, 1.0)

        for method in ['SLSQP', 'L-BFGS-B', 'TNC', 'SLSQP']:
            if time.time() - start_time >= budget_s - 30:
                break
            result = minimize(
                fun=objective_fft,
                x0=perturbed,
                method=method,
                bounds=bounds,
                constraints=cons if method == 'SLSQP' else [],
                options={'maxiter': 150, 'ftol': 1e-8, 'disp': False}
            )
            if result.success:
                new_h = result.x
                new_h = np.clip(new_h, 0.0, 1.0)
                if np.all(new_h >= 0) and np.all(new_h <= 1.0 + 1e-6) and np.isclose(np.sum(new_h), target_sum, atol=1e-6):
                    new_c5 = result.fun
                    if new_c5 < best_c5:
                        best_h = new_h
                        best_c5 = new_c5
                        break

    h_full = best_h
    assert np.isclose(np.sum(h_full), target_sum, atol=1e-6), "Integral constraint violated"
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6)), "Value bounds violated"
    
    return (h_full.tolist(), best_c5, n_points)
