def repair_position(h, target_sum, n_points):
    h_clipped = np.clip(h, 0.0, 1.0)
    total = np.sum(h_clipped)
    if total == 0:
        return np.full_like(h_clipped, target_sum / n_points)
    return h_clipped * (target_sum / total)

    velocities = np.zeros((n_particles, n_points))

    g_best_obj = p_best_obj[g_best_idx]
