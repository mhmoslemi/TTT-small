            new_h_half = new_h_half * (target_sum_high / new_h_sum)
        result = minimize(
            fun=lambda x: compute_c5(create_full_h(x), dx),
            x0=new_h_half,
            bounds=bounds_high,
            constraints=cons_high,
            method='trust-constr',
            options={'maxiter': 400, 'gtol': 1e-8}
        )
        if result.success:
            x_opt = np.clip(result.x, 0, 1)
            h_full_opt = create_full_h(x_opt)
            c5_new = compute_c5(h_full_opt, dx)
            if c5_new < best_c5:
                best_h_half[:] = x_opt
                best_c5 = c5_new

    # Final validation and return
    h_full = create_full_h(best_h_half)
    h_full = np.clip(h_full, 0, 1)
    h_full = h_full * (n_points / 2.0 / h_full.sum())
    h_full = np.clip(h_full, 0, 1)

    assert abs(h_full.sum() - n_points / 2.0) < 1e-6
    assert np.all((h_full >= -1e-6) & (h_full <= 1.0 + 1e-6))

    return (h_full.tolist(), best_c5, n_points)
