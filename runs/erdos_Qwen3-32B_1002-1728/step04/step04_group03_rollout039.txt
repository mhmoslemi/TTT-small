    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_sum = np.sum(final_h)
        if not np.isclose(final_sum, target_sum, atol=1e-6):
            final_h = np.clip(final_h * (target_sum / final_sum), 0, 1)
        final_c5 = compute_c5(final_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    # Final validation
    assert np.all((best_h >= -1e-6) & (best_h <= 1.0 + 1e-6)), "Violation of bounds"
    assert np.isclose(np.sum(best_h), target_sum, atol=1e-6), "Integral constraint violated"
    
    return (best_h.tolist(), best_c5, n_points)
