import numpy as np
import time

def run(seed=42, budget_s=1100):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = max(np.correlate(h, 1 - h, mode="full") * dx)
    
    start_time = time.time()
    learning_rate = 0.01
    patience = 100
    no_improvement = 0
    
    while time.time() - start_time < budget_s:
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        m_max = np.argmax(overlap)
        k_shift = m_max - (n_points - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]
        grad *= dx
        
        h -= learning_rate * grad
        
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if current_sum > 1e-8:
            h = h_clipped * (n_points / 2) / current_sum
        else:
            h = h_clipped
        
        current_c5 = max(np.correlate(h, 1 - h, mode="full") * dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learning_rate = 0.01
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > patience:
                learning_rate *= 0.9
                no_improvement = 0
    
    return (best_h, best_c5, n_points)
