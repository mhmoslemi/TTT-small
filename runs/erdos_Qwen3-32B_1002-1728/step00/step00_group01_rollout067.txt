import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)

    def compute_c5(h_array):
        return np.max(np.correlate(h_array, 1 - h_array, mode='full')) * dx

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    np.random.seed(seed)
    start_time = time.time()

    alpha = 0.999

    while time.time() - start_time < budget_s:
        i, j = -1, -1
        delta_max = 0.0
        for _ in range(100):
            i = np.random.randint(n_points)
            j = np.random.randint(n_points)
            if i != j:
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max > 1e-10:
                    break

        if delta_max <= 1e-10:
            T *= alpha
            continue

        delta = np.random.uniform(0, delta_max)
        old_h = h.copy()
        old_c5 = compute_c5(h)
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
