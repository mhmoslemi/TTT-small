import numpy as np
import time

def repair_sum(h, required_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return h_clipped
    scale = required_sum / current_sum
    h_scaled = h_clipped * scale
    h_scaled = np.clip(h_scaled, 0.0, 1.0)
    return h_scaled
