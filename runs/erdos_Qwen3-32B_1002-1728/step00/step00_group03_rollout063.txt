import numpy as np
import time
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    start_time = time.time()
    
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    target_sum = n_points / 2.0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    current_h = h.copy()
    
    alpha = 0.1
    epsilon = 0.05
    dx = 2.0 / n_points
    
    while time.time() - start_time < budget_s - 10:
        # Compute overlaps for all positive lags
        overlaps = []
        for k in range(1, n_points):
            valid = n_points - k
            if valid <= 0:
                overlaps.append(0.0)
                continue
            overlap = np.sum(current_h[:valid] * (1 - current_h[k:k + valid])) * dx
            overlaps.append(overlap)
        if not overlaps:
            break
        max_overlap = max(overlaps)
        k_max = np.argmax(overlaps) + 1  # first k is 1
        
        # Compute gradient for k_max
        valid_len = n_points - k_max
        gradient = np.zeros(n_points)
        gradient[:valid_len] = (1 - current_h[k_max:k_max + valid_len]) * dx
        
        # Update step
        new_h = current_h - alpha * gradient
        new_h += epsilon * np.random.randn(n_points)
        
        # Projection onto [0, 1]
        new_h = np.clip(new_h, 0.0, 1.0)
        
        # Enforce sum constraint
        current_sum = np.sum(new_h)
        if abs(current_sum - target_sum) > 1e-8:
            scale = target_sum / current_sum
            new_h *= scale
            # Clip again after scaling
            new_h = np.clip(new_h, 0.0, 1.0)
        
        # Evaluate new H
        new_overlaps = []
        for k in range(1, n_points):
            valid = n_points - k
            if valid <= 0:
                new_overlaps.append(0.0)
                continue
            current_overlap = np.sum(new_h[:valid] * (1 - new_h[k:k + valid])) * dx
            new_overlaps.append(current_overlap)
        new_max_overlap = max(new_overlaps)
        
        if new_max_overlap < best_c5:
            current_h = new_h.copy()
            best_h = new_h.copy()
            best_c5 = new_max_overlap
            alpha *= 0.95
            epsilon = 0.05
        else:
            current_h = best_h.copy()
            epsilon *= 1.05
        
        # Check time
        if time.time() - start_time > budget_s - 5:
            break
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, n_points)
