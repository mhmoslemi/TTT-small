def compute_c5(h, dx):
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    return np.max(corr * dx)

def project_feasible(perturbed_h, target_sum):
    h = np.clip(perturbed_h, 0, 1).copy()
    s = h.sum()
    delta = target_sum - s

        return h
    if delta > 0:
        indices = np.where(h < 1.0)[0]

            h = np.clip(perturbed_h * (target_sum / max(1e-8, s)), 0, 1)
            return h

            return h
        subtract_amount = delta / len(indices)
        h[indices] -= subtract_amount
