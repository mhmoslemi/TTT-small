import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h_init = initial_h_values.copy()
    n_coarse = len(h_init)
    dx_coarse = 2.0 / n_coarse

    # Project and optimize on coarse grid
    def compute_correlation(h, dx, n):
        return np.correlate(h, 1 - h, mode='full') * dx

    def find_max_k(corr, n):
        l_max = np.argmax(corr)
        k = l_max - (n - 1)
        return corr[l_max], k

    def compute_gradient(h, k, dx, n):
        grad = np.zeros(n)
        for m in range(n):
            term1 = dx * (1 - h[m + k]) if m + k < n else 0.0
            term2 = -dx * h[m - k] if m - k >= 0 else 0.0
            grad[m] = term1 + term2
        return grad

    def project_simplex_box(h, target_sum):
        h_clipped = np.clip(h, 0, 1)
        current = h_clipped.sum()
        if np.isclose(current, target_sum):
            return h_clipped
        delta = target_sum - current
        idxs = np.argsort(h_clipped)
        if delta > 0:
            for i in range(len(idxs)):
                if h_clipped[idxs[i]] < 1:
                    add = min(delta, (1 - h_clipped[idxs[i]]))
                    h_clipped[idxs[i]] += add
                    delta -= add
                    if delta <= 1e-8:
                        break
        else:
            for i in range(len(idxs)-1, -1, -1):
                if h_clipped[idxs[i]] > 0:
                    sub = min(-delta, h_clipped[idxs[i]])
                    h_clipped[idxs[i]] -= sub
                    delta += sub
                    if delta <= 1e-8:
                        break
        return h_clipped

    start = time()
    max_iter = 1000
    h_coarse = project_simplex_box(h_init, n_coarse / 2)
    for iter in range(max_iter):
        if time() - start > budget_s:
            break
        corr = compute_correlation(h_coarse, dx_coarse, n_coarse)
        max_val, k = find_max_k(corr, n_coarse)
        grad = compute_gradient(h_coarse, k, dx_coarse, n_coarse)
        h_coarse = project_simplex_box(h_coarse - 0.1 * grad, n_coarse / 2)

    # Upsample to fine grid
    n_fine = 4 * n_coarse
    h_fine = np.repeat(h_coarse, 4)
    h_fine = project_simplex_box(h_fine, n_fine / 2)
    dx_fine = 2.0 / n_fine

    for iter in range(1000):
        if time() - start > budget_s:

        max_val, k = find_max_k(corr, n_fine)
        grad = compute_gradient(h_fine, k, dx_fine, n_fine)
        h_fine = project_simplex_box(h_fine - 0.05 * grad, n_fine / 2)

    # Final evaluation
    corr_final = compute_correlation(h_fine, dx_fine, n_fine)
    c5_bound = np.max(corr_final)
    return h_fine, c5_bound, n_fine
