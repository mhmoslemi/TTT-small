def repair(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return h_clipped
    scale_factor = target_sum / current_sum
    h_scaled = h_clipped * scale_factor
    h_scaled_clipped = np.clip(h_scaled, 0, 1)
    return h_scaled_clipped

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr * dx)

    inertia = 0.8
    c1 = 2.0
    c2 = 2.0

    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []
    global_best = None
    global_best_c5 = float('inf')

    # Initialize first particle from initial_h_values
    particle = repair(initial, target_sum)
    c5 = compute_c5(particle, dx)
    particles.append(particle)
    velocities.append(np.zeros_like(particle))
    personal_best.append(particle.copy())
    personal_best_c5.append(c5)
    if c5 < global_best_c5:
        global_best = particle.copy()
        global_best_c5 = c5

    # Initialize remaining particles
    for _ in range(num_particles - 1):
        perturbed = initial + np.random.normal(0, 0.1, size=n_points)
        repaired = repair(perturbed, target_sum)
        c5 = compute_c5(repaired, dx)
        particles.append(repaired)
        velocities.append(np.zeros_like(repaired))
        personal_best.append(repaired.copy())
        personal_best_c5.append(c5)
        if c5 < global_best_c5:
            global_best = repaired.copy()
            global_best_c5 = c5

    start_time = time.time()

    # Optimization loop
    while time.time() - start_time < budget_s:
        for i in range(num_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = (inertia * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))
            new_pos = particles[i] + velocities[i]
            new_pos = repair(new_pos, target_sum)
            new_c5 = compute_c5(new_pos, dx)

            if new_c5 < personal_best_c5[i]:
                personal_best[i] = new_pos.copy()
                personal_best_c5[i] = new_c5
            if new_c5 < global_best_c5:
                global_best = new_pos.copy()
                global_best_c5 = new_c5
            particles[i] = new_pos

        if time.time() - start_time >= budget_s:
            break

    return (global_best, global_best_c5, n_points)
