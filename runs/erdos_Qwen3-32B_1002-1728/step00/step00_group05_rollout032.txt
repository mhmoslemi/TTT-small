    patience = 50

        h_new *= (n_points / 2) / np.sum(h_new)
        
        c5 = evaluate_erdos_solution(h_new)

            best_h = h_new.copy()

            no_improvement = 0
            print(f"Iter {i}: C₅={best_c5:.6f} (best so far)")
        else:
            no_improvement += 1

            # Restart with previous best
            h = best_h.copy()
            learning_rate *= lr_decay
            no_improvement = 0
            print(f"Iter {i}: Restarting with learning rate={learning_rate:.5f}")

    if best_h is not None and not np.isnan(best_c5):

def evaluate_erdos_solution(h):
    if len(h) == 0:
        return np.inf
        
    # Ensure values are in [0, 1]
    h_clipped = np.clip(h, 0, 1).copy()
    
    # Normalize to meet constraint sum(h) == n_points / 2
    h_clipped *= (len(h) / 2) / np.sum(h_clipped)
    
    # Compute overlap integral
    ones = np.ones(len(h))
    c = correlate(h_clipped, 1 - h_clipped, mode="full")
    dx = 2.0 / len(h)
    c5 = np.max(c * dx)
    
    return c5


def compute_gradient(h, eps=1e-5):
    gradient = np.zeros_like(h)
    
    for i in range(len(h)):
        # Forward perturbation
        h_forward = h.copy()
        h_forward[i] += eps
        h_forward = np.clip(h_forward, 0, 1)
        h_forward *= (len(h) / 2) / np.sum(h_forward)
        c_forward = evaluate_erdos_solution(h_forward)
        
        # Backward perturbation
        h_backward = h.copy()
        h_backward[i] -= eps
        h_backward = np.clip(h_backward, 0, 1)
        h_backward *= (len(h) / 2) / np.sum(h_backward)
        c_backward = evaluate_erdos_solution(h_backward)
        
        gradient[i] = (c_forward - c_backward) / (2 * eps)
        
    return gradient
