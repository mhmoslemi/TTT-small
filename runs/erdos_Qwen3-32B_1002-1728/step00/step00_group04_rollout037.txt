def project(h_unprojected, target_sum, n_points):
    h = cp.Variable(n_points)
    constraints = [
        0 <= h,
        h <= 1,
        cp.sum(h) == target_sum
    ]
    objective = cp.Minimize(cp.sum_squares(h - h_unprojected))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    if prob.status != cp.OPTIMAL:
        # Fallback: clip to bounds and scale sum
        h_projected = np.clip(h_unprojected, 0, 1)

            scaling_factor = target_sum / current_sum
            h_projected *= scaling_factor
            h_projected = np.clip(h_projected, 0, 1)
        return h_projected
    return h.value

    return np.max(c_values)
