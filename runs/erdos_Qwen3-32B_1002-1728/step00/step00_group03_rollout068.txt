    n_points = len(initial_h)
    target_sum = n_points / 2.0
    assert np.isclose(np.sum(initial_h), target_sum, atol=1e-8), "Initial h sum mismatch"
    h = initial_h.copy()
    best_h = h.copy()
    dx = 2.0 / n_points

    def compute_c5(current_h):
        one_minus_h = 1.0 - current_h
        corr = np.correlate(current_h, one_minus_h, mode='full')
        return np.max(corr) * dx

    current_c5 = compute_c5(h)
    best_c5 = current_c5

    T = 1.0
    alpha = 0.99
    no_improvement = 0
    start_time = time.time()
    timeout = budget_s

    while True:
        current_time = time.time()
        if current_time - start_time > timeout - 10:
            break

        i = np.random.randint(n_points)
        j = np.random.randint(n_points)
        if i == j:
            continue

        h_i, h_j = h[i], h[j]
        max_add = min(1.0 - h_i, h_j)
        min_add = max(-h_i, h_j - 1.0)
        if max_add < min_add:
            continue

        if np.isclose(max_add, min_add):
            continue

        eps = np.random.uniform(min_add, max_add)
        new_i = h_i + eps
        new_j = h_j - eps

        old_i, old_j = h[i], h[j]
        h[i], h[j] = new_i, new_j

        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5

        if delta < 0:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            if np.random.rand() < np.exp(-delta / T):
                current_c5 = new_c5
                no_improvement += 1
            else:
                h[i], h[j] = old_i, old_j
                no_improvement += 1

        if no_improvement >= 100:
            alpha *= 0.95
            no_improvement = 0
        T *= alpha

    return (best_h.tolist(), best_c5, n_points)
