import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    n_points = 45
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h / h.sum() * n_points / 2
    dx = 2.0 / n_points

    def compute_c5(h):
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)

    best_h = h.copy()
    best_C5 = compute_c5(best_h)
    current_h = h.copy()
    current_C5 = best_C5

    T = 1.0
    alpha = 0.95
    patience = 20
    no_improve = 0

    def project_feasible(h):
        h = np.clip(h, 0, 1)
        sum_h = h.sum()
        if sum_h == 0:
            return (np.ones_like(h) * 0.5) if not 'initial_h_values' in globals() else initial_h_values.copy()
        scale = (n_points / 2) / sum_h
        return h * scale

    def perturb(current_h):
        h = current_h.copy()
        r = np.random.rand()
        if r < 0.5:
            i = np.random.randint(n_points)
            h[i] = np.random.rand()
        elif r < 0.8:
            n = len(h)
            l = np.random.randint(2, 11)
            start = np.random.randint(n)
            idxs = [(start + k) % n for k in range(l)]
            shift = np.random.choice([-1, 1])
            original = [h[i] for i in idxs]
            if shift == -1:
                new_vals = [original[-1]] + original[:-1]
            else:
                new_vals = original[1:] + [original[0]]
            for idx, val in zip(idxs, new_vals):
                h[idx] = val
        else:
            i, j = np.random.choice(n_points, size=2, replace=False)
            h[i], h[j] = h[j], h[i]
        return project_feasible(h)

    while time.time() - start_time < budget_s:
        new_h = perturb(current_h)
        new_C5 = compute_c5(new_h)
        delta = new_C5 - current_C5
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            current_h = new_h
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = current_C5
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve >= patience:
            T *= alpha
            no_improve = 0

    return (best_h.tolist(), best_C5, n_points)
