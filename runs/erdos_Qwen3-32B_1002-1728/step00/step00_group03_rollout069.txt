import numpy as np
import time
import random

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        n_points = 70
        h = np.zeros(n_points, dtype=np.float64)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.0
        # Normalize sum to meet constraint
        h *= (n_points / 2.0) / h.sum()

    n_points = len(h)
    dx = 2.0 / n_points

    def compute_c5(h_vec):
        correlations = np.correlate(h_vec, 1 - h_vec, mode='full')
        return correlations.max() * dx

    best_h = h.copy()
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_c5 = current_c5

    T = 1.0
    alpha = 0.99
    no_improvement_count = 0
    start_time = time.time()

    while time.time() - start_time < budget_s - 10:
        for _ in range(10):
            if time.time() - start_time >= budget_s - 10:
                break

            i, j = np.random.choice(n_points, 2, replace=False)
            if i == j:
                continue
            a = current_h[i]
            b = current_h[j]

            direction = random.random() < 0.5
            if direction:
                max_eps = min(a, 1 - b)
                if max_eps <= 1e-10:
                    continue
                eps = random.random() * max_eps
                new_a, new_b = a - eps, b + eps
            else:
                max_eps = min(b, 1 - a)
                if max_eps <= 1e-10:
                    continue
                eps = random.random() * max_eps
                new_a, new_b = a + eps, b - eps

            new_h = current_h.copy()
            new_h[i] = new_a
            new_h[j] = new_b
            new_c5 = compute_c5(new_h)
            delta = new_c5 - current_c5

            if delta < 0:
                current_h = new_h
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                prob = np.exp(-delta / T)
                if random.random() < prob:
                    current_h = new_h
                    current_c5 = new_c5
                    no_improvement_count += 1

            if no_improvement_count > 100:
                alpha *= 0.95
                no_improvement_count = 0
            T *= alpha

    return (best_h, best_c5, n_points)
