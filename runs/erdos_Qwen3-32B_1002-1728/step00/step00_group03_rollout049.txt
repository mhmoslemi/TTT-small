import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if not np.all(h >= 0) or not np.all(h <= 1):
        raise ValueError("Initial h contains values outside [0, 1]")
    if abs(np.sum(h) - target_sum) > 1e-6:
        raise ValueError("Initial h sum is not equal to n_points / 2")

    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    current_c5 = np.max(overlaps)
    best_h = h.copy()
    best_c5 = current_c5

    T = 1.0
    alpha = 0.99
    iterations_without_improvement = 0
    max_iterations_without_improvement = 100
    cooling_adjustment_factor = 0.95

    while time.time() - start_time < budget_s - 0.1:
        while True:
            i, j = np.random.choice(n_points, size=2, replace=False)
            delta_upper = min(1.0 - h[i], h[j])
            delta_lower = -min(1.0 - h[j], h[i])
            if delta_upper > delta_lower:
                break

        delta = np.random.uniform(delta_lower, delta_upper)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)

        overlaps = np.correlate(new_h, 1 - new_h, mode='full') * dx
        new_c5 = np.max(overlaps)

        delta_energy = new_c5 - current_c5

        if delta_energy < 0 or np.random.rand() < np.exp(-delta_energy / T):
            h = new_h.copy()
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                iterations_without_improvement = 0
            else:
                iterations_without_improvement += 1
        else:
            iterations_without_improvement += 1

        if iterations_without_improvement > max_iterations_without_improvement:
            alpha *= cooling_adjustment_factor
            iterations_without_improvement = 0

        T *= alpha

    return best_h, best_c5, n_points
