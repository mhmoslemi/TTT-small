    step_size = 0.1
    decay_rate = 0.5
    decay_steps = 50
    max_iter = int(1e5)

        if time.time() - start_time > budget_s - 1:
            break

        h1 = 1 - h
        cross_corr = np.correlate(h, h1, mode='full') * dx
