        gradient[i + k] += -h[i]
    return gradient

def local_optimize(h, n_points, steps=20, alpha=0.005):
    current = h.copy()
    for _ in range(steps):
        max_c = -np.inf
        best_k = -1
        for k in range(1, n_points):
            valid = n_points - k
            c = np.sum(current[:valid] * (1 - current[k:k + valid]))
            if c > max_c:
                max_c = c
                best_k = k
        grad = compute_gradient(current, best_k)
        current = current - alpha * grad
        current = project_to_feasible(current, n_points)
    return current

def crossover(parent1, parent2, n_points):
    c1, c2 = np.sort(np.random.choice(n_points, size=2, replace=False))
    child = np.zeros(n_points)
    child[:c1] = parent1[:c1]
    child[c1:c2] = parent2[c1:c2]
    child[c2:] = parent1[c2:]
    return project_to_feasible(child, n_points)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    population_size = 20
    population = []

    if 'initial_h_values' in globals():
        h_init = project_to_feasible(initial_h_values, n_points)
        population.append(h_init)

    while len(population) < population_size:
        population.append(generate_random_h(n_points))

    best_h = None
    best_c5 = float('inf')

    while time.time() - start_time < budget_s:
        population = sorted(population, key=lambda h: evaluate_erdos_solution(h, n_points))

        if evaluate_erdos_solution(population[0], n_points) < best_c5:
            best_c5 = evaluate_erdos_solution(population[0], n_points)
            best_h = population[0].copy()

        top = int(population_size * 0.3)
        for i in range(top):
            if time.time() - start_time >= budget_s:
                break
            population[i] = local_optimize(population[i], n_points)

        selected = population[:int(population_size * 0.5)]
        new_pop = []
        while len(new_pop) < population_size - len(population[:int(population_size * 0.5)]) and time.time() - start_time < budget_s:
            p1, p2 = np.random.choice(selected, 2)
            child = crossover(p1, p2, n_points)
            if np.random.rand() < 0.1:
                noise = np.random.normal(0, 0.1, n_points)
                child += noise
            child = project_to_feasible(child, n_points)
            new_pop.append(child)
        population = selected + new_pop

    return (best_h, best_c5, n_points)
