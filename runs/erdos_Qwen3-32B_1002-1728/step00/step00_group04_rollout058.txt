    current_sum = np.sum(h_clipped)
    if np.abs(current_sum - target_sum) < 1e-8:

# Helper function to project to feasible region
def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if np.abs(current_sum - target_sum) < 1e-8:
        return h_clipped
    n = len(h)
    h_var = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h_var - h_clipped))
    constraints = [cp.sum(h_var) == target_sum, h_var >= 0, h_var <= 1]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except:
        pass
    if prob.status == cp.OPTIMAL:
        projected_h = np.clip(h_var.value, 0, 1)
        if np.abs(np.sum(projected_h) - target_sum) < 1e-6:
            return projected_h
    return h_clipped
