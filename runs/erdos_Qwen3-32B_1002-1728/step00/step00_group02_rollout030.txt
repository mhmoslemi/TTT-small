    no_improvement = 0

    while (time.time() - start_time) < budget_s:
        current_h = best_h.copy()
        overlaps = evaluate_erdos_solution(current_h)
        current_overlaps = overlaps[0]
        current_c5 = current_overlaps.max()
        k_max = current_overlaps.argmax()

        dx = 2.0 / n_points
        h_array = current_h
        contributions = []
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                contrib = h_array[i] * (1 - h_array[j]) * dx * dx
            else:
                contrib = 0.0
            contributions.append((i, contrib))
        contributions.sort(key=lambda x: x[1], reverse=True)
        top_fraction = 0.2
        top_n = max(1, int(len(contributions) * top_fraction))
        top_indices = [i for i, _ in contributions[:top_n]]

        step = -learning_rate * dx
        for i in top_indices:
            j = i + k_max
            if 0 <= j < n_points:
                grad = (1 - h_array[j])
                h_array[i] += step * grad

        h_array = project_feasible(h_array, n_points)

        new_overlaps = evaluate_erdos_solution(h_array)
        new_c5 = new_overlaps[0].max()
        state_hash = tuple(np.round(h_array, 12).tolist())

        if state_hash not in tabu_list:
            if new_c5 < best_c5 - 1e-6:
                best_c5 = new_c5
                best_h = h_array.copy()
                tabu_list.insert(0, state_hash)
                tabu_list = tabu_list[:tabu_size]
                no_improvement = 0
            else:
                no_improvement += 1
        else:
            no_improvement += 1

        if no_improvement > max_no_improvement:
            new_n = n_points * 2
            new_h = np.zeros(new_n)
            for i in range(n_points):
                val = best_h[i]
                i1 = int(i * (new_n / n_points))
                i2 = i1 + 1
                if i2 < new_n:
                    new_h[i1] += val / 2
                    new_h[i2] += val / 2
                else:
                    new_h[i1] += val
            new_h = project_feasible(new_h, new_n)
            best_h = new_h
            n_points = new_n
            best_c5 = evaluate_erdos_solution(best_h)[0].max()
            tabu_list = []
            no_improvement = 0
            learning_rate = 0.1

        if time.time() - start_time > budget_s:
            break

    best_h = project_feasible(best_h, n_points)
    final_c5 = evaluate_erdos_solution(best_h)[0].max()
    return (best_h.tolist(), final_c5, n_points)
