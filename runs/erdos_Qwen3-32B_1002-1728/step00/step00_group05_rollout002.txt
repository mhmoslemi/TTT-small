import numpy as np
import scipy.optimize
from scipy.interpolate import interp1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

    best_c5 = compute_c5(h_current)
    best_h = h_current.copy()
    best_n = n
    max_refine_steps = 4

    for _ in range(max_refine_steps):
        if time.time() - start_time > budget_s - 60:
            break
        new_n = 2 * n
        x = np.linspace(0, 2, n, endpoint=False)
        new_x = np.linspace(0, 2, new_n, endpoint=False)
        f = interp1d(x, h_current, kind='linear', fill_value='extrapolate')
        new_h = f(new_x)
        target_sum = new_n / 2.0
        current_sum = np.sum(new_h)
        new_h *= target_sum / current_sum
        new_h = np.clip(new_h, 0, 1)
        constraints = scipy.optimize.LinearConstraint(np.ones((1, new_n)), lb=[target_sum], ub=[target_sum])
        result = scipy.optimize.minimize(

    return (best_h, best_c5, best_n)
