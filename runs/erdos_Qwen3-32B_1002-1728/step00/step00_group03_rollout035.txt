        dx = 2.0 / n

        if lag > 0:
            k = lag
            for i in range(n - k):
                j = i + k
                gradient[i] += (1 - h[j]) * dx
                gradient[j] += -h[i] * dx
        elif lag < 0:
            k_abs = -lag
            for i in range(k_abs, n):
                j = i - k_abs
                gradient[i] += (1 - h[j]) * dx
                gradient[j] += -h[i] * dx
        else:
            gradient[:] = (1 - 2 * h) * dx

        alpha = 0.1
        h = h - alpha * gradient
        h = project(h, n / 2.0)
        current_c5 = evaluate_erdos_solution(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h

        if time.time() - start_time > budget_s - 1:
            break

        if np.random.rand() < 0.01:  # Random perturbation
            perturbation = 0.05 * np.random.randn(n)
            h += perturbation
            h = project(h, n / 2.0)

    return (best_h.tolist(), best_c5, n)
