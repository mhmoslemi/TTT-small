import numpy as np
import cvxpy as cp
import time

def project(x, n_points):
    T = n_points / 2.0
    y = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum_squares(y - x))
    constraints = [cp.sum(y) == T, y >= 0, y <= 1]
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except:
        prob.solve()
    return np.clip(y.value, 0, 1)

def compute_C5(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    velocities = [np.zeros(n_points) for _ in particles]
