import cvxpy as cp
import time
import scipy

def project_to_feasible(h, n_points):
    h_clipped = np.clip(h, 0., 1.)
    h_var = cp.Variable(n_points)
    constraints = [h_var >= 0., h_var <= 1., cp.sum(h_var) == n_points / 2.]
    objective = cp.Minimize(cp.sum_squares(h_var - h_clipped))
    problem = cp.Problem(objective, constraints)
    try:
        problem.solve(solver=cp.ECOS)
    except:
        try:
            problem.solve(solver=cp.CVXOPT)
        except:
            problem.solve()
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return h_var.value
    else:
        scaled = h_clipped
        sum_clipped = np.sum(scaled)
        if abs(sum_clipped) < 1e-9:
            return 0.5 * np.ones(n_points)
        scale = (n_points / 2.0) / sum_clipped
        scaled = h_clipped * scale
        scaled = np.clip(scaled, 0., 1.)
        return scaled
