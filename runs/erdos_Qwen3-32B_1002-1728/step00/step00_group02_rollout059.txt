import numpy as np
from scipy.optimize import Bounds, LinearConstraint, minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    dx = 2.0 / n_points

    def objective(h):
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    sum_constraint = LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)
    bounds = Bounds(0.0, 1.0)
    x0 = h_initial.copy()
    
    start_time = time.time()
    best_c5 = None
    best_h = h_initial.copy()
    
    # Use up to 5 restarts with time budget
    while True:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Perform optimization
        result = minimize(
            objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=sum_constraint,
            options={'maxiter': 200, 'ftol': 1e-6}
        )
        
        curr_h = result.x.copy()
        # Ensure solution is feasible (clamp and fix sum constraint)
        curr_h = np.clip(curr_h, 0, 1)
        offset = np.sum(curr_h) - n_points/2
        curr_h[0] -= offset  # Fix first element to maintain sum constraint
        
        corr_values = np.correlate(curr_h, 1 - curr_h, mode='full') * dx
        curr_c5 = np.max(corr_values)
        
        if best_c5 is None or curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = curr_h.copy()
        
        x0 = best_h.copy()  # Continue from best solution for next iteration
        x0 += np.random.normal(0, 0.01, n_points)  # Perturb solution to escape local minima
        x0 = np.clip(x0, 0, 1)
        x0[0] -= np.sum(x0) - n_points/2  # Reset sum constraint after perturbation
    
    # Final validation
    corr_values = np.correlate(best_h, 1 - best_h, mode='full') * dx
    c5_bound = np.max(corr_values)
    
    return best_h, c5_bound, n_points
