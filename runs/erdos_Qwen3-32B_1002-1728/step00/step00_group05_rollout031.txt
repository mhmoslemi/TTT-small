    h_best = h.copy()

    n_points = n_coarse
    h_current = h.copy()
    max_iterations = 200
    for it in range(max_iterations):
        if time.time() - start_time > max_time - 5:
            break
        overlaps = evaluate_erdos_solution(h_current)
        k_star = np.argmax(overlaps)
        dx = 2.0 / n_points
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k_star < n_points:
                term1 = 1.0 - h_current[j + k_star]
            else:
                term1 = 0.0
            if j - k_star >= 0:
                term2 = h_current[j - k_star]
            else:
                term2 = 0.0
            grad[j] = dx * (term1 - term2)
        step_size = 0.01
        h_current = np.clip(h_current - step_size * grad, 0, 1)
        h_current = h_current * (n_points / 2) / h_current.sum()
        c5 = compute_c5(h_current)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_current.copy()
        if it % 50 == 0:
            print(f"Coarse Iter {it}: C5 = {c5}")

    # Upsample to fine grid
    h_fine = np.repeat(h_best, 4)
    n_points = len(h_fine)
    h_current = h_fine.copy()
    max_iterations = 1000
    for it in range(max_iterations):
        if time.time() - start_time > max_time - 5:
            break
        overlaps = evaluate_erdos_solution(h_current)
        k_star = np.argmax(overlaps)
        dx = 2.0 / n_points
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k_star < n_points:
                term1 = 1.0 - h_current[j + k_star]
            else:
                term1 = 0.0
            if j - k_star >= 0:
                term2 = h_current[j - k_star]
            else:
                term2 = 0.0
            grad[j] = dx * (term1 - term2)
        step_size = 0.005
        h_current = np.clip(h_current - step_size * grad, 0, 1)
        h_current = h_current * (n_points / 2) / h_current.sum()
        c5 = compute_c5(h_current)
        if c5 < c5_best:
            c5_best = c5
            h_best = h_current.copy()
        if it % 200 == 0:
            print(f"Fine Iter {it}: C5 = {c5}")
    return h_best, c5_best, len(h_best)
