from scipy.signal import correlate
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()

        h = initial_h_values.copy()
    except NameError:
        n_points = 70
        h = np.zeros(n_points)

    def callback(x, *__):

        if c5 < best_c5:

    result = minimize(

        bounds=bounds,
        constraints=cons,
        callback=callback,
        options={'maxiter': 100, 'ftol': 1e-6, 'disp': False}
    )

    best_h = np.clip(best_h, 0, 1)
