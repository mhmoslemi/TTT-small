def optimize_gradient_descent(h, n_points, budget_s, target_c5=0.380876):

    start_time = time.time()

    dx = 2.0 / n_points

    target_sum = n_points / 2.0

    max_iterations = 1000  # can be adjusted

    lr = 0.05

    for iteration in range(max_iterations):

        # Check time

        if time.time() - start_time > budget_s:

            break

        # Compute overlaps

        corrs = np.correlate(h, 1 - h, mode='full') * dx

        max_c5 = np.max(corrs)

        if max_c5 <= target_c5:

            break  # early stopping

        # Find the k (lag) corresponding to max_c5

        max_index = np.argmax(corrs)

        lag = max_index - (n_points - 1)

        # Compute gradient for this lag

        grad = np.zeros(n_points)

        for i in range(n_points):

            if i + lag < n_points:

                grad[i] += (1 - h[i + lag])

            if i - lag >= 0:

                grad[i] -= h[i - lag]

        # Gradient step with learning rate

        h_prev = h.copy()

        h -= lr * grad

        # Project into feasible set

        h = np.clip(h, 0, 1)

        # Normalize sum

        current_sum = np.sum(h)

        if np.isclose(current_sum, target_sum):

            pass

        else:

            h *= target_sum / current_sum

            h = np.clip(h, 0, 1)

        # Check for NaNs or infs

        if np.any(np.isnan(h)) or np.any(np.isinf(h)):

            h = h_prev.copy()

            lr *= 0.5

        # reducing learning rate over time

        lr *= 0.995

    return h, max_c5

    h_fine = np.repeat(h_coarse, n_fine // n_coarse)
    h_fine = np.clip(h_fine, 0, 1)
    target_sum_fine = n_fine / 2.0
    current_sum_fine = np.sum(h_fine)
    if not np.isclose(current_sum_fine, target_sum_fine, 1e-5):
        scaling = target_sum_fine / current_sum_fine
        h_fine *= scaling
        h_fine = np.clip(h_fine, 0, 1)

    # Fine grid optimization
    max_iterations_fine = 1000
    for iter in range(max_iterations_fine):
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n_fine
        corrs = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        max_c5 = np.max(corrs)
        if max_c5 <= 0.380876:
            break

        max_index = np.argmax(corrs)
        lag = max_index - (n_fine - 1)
        grad = compute_gradient_vectorized(h_fine, lag, n_fine)

        lr_fine = 0.01 * np.exp(-iter / 500)
        h_prev_fine = h_fine.copy()
        h_fine -= lr_fine * grad

        h_fine = np.clip(h_fine, 0, 1)
        current_sum = np.sum(h_fine)
        target_sum_fine = n_fine / 2.0
        if not np.isclose(current_sum, target_sum_fine, 1e-5):
            scaling = target_sum_fine / current_sum
            h_fine *= scaling
            h_fine = np.clip(h_fine, 0, 1)

        if np.any(np.isnan(h_fine)) or np.any(np.isinf(h_fine)):
            h_fine = h_prev_fine

    # Final evaluation
    dx_final = 2.0 / n_fine
    corrs_final = np.correlate(h_fine, 1 - h_fine, mode='full') * dx_final
    final_c5 = np.max(corrs_final)

    return h_fine, final_c5, n_fine
