import numpy as np
from time import monotonic

def compute_c5(h_candidate, n_points):
    max_overlap = 0.0
    dx = 2.0 / n_points
    for k in range(1, n_points):
        overlap = np.sum(h_candidate[0 : n_points - k] * (1.0 - h_candidate[k : n_points])) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    current_h = h_initial.copy()
    current_sum = np.sum(current_h)
    if not np.allclose(current_sum, target_sum, atol=1e-8):
        current_h = current_h / current_sum * target_sum
    current_c5 = compute_c5(current_h, n_points)

    best_h = current_h.copy()
    best_c5 = current_c5

    T_initial = 1.0
    T = T_initial
    alpha = 0.99
    sigma = 0.1 * T

    start_time = monotonic()

    while monotonic() - start_time < budget_s:
        perturbation = np.random.normal(0, sigma, size=n_points)
        new_h = current_h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_sum = np.sum(new_h)
        if new_sum == 0:
            new_h = np.full_like(new_h, target_sum / n_points)
        else:

        new_c5 = compute_c5(new_h, n_points)
