            delta = new_C5 - best_C5
            if np.random.rand() < np.exp(-delta / T):
                best_C5 = new_C5
                best_h = h_proj.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Cool down if no improvement
        if no_improvement_count >= 30:
            T = max(0.1, T * alpha)
            no_improvement_count = 0
    
    # Final projection and validation
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2.0) / best_h.sum()
    final_C5 = compute_C5(best_h, dx, n_points)
    return (best_h.tolist(), final_C5, n_points)
