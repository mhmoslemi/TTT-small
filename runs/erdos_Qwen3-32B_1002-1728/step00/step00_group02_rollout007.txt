import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64)
    best_h = h.copy()
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vec):
        overlaps = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        max_val = np.max(overlaps)
        max_index = np.argmax(overlaps)
        return max_val, overlaps, max_index

    best_c5, _, _ = compute_c5(h)

    learning_rate = 0.01
    max_iter = 1000
    patience = 100
    adjust_interval = 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        current_c5, overlaps, max_index = compute_c5(h)
        k_star = max_index - n_points + 1

        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_star
            if 0 <= j < n_points:
                gradient[i] = (1 - h[j]) * dx
            else:
                gradient[i] = 0.0

        for i in range(n_points):
            grad = gradient[i]
            if grad > 0:
                h[i] -= learning_rate
            elif grad < 0:
                h[i] += learning_rate
            h[i] = np.clip(h[i], 0.0, 1.0)

        if iteration % adjust_interval == 0:
            current_sum = np.sum(h)
            if abs(current_sum - target_sum) > 1e-6:
                valid_idx = np.where((h > 0) & (h < 1))[0]
                if valid_idx.size > 0:
                    delta = (target_sum - current_sum) / valid_idx.size
                    h[valid_idx] += delta
                    h = np.clip(h, 0.0, 1.0)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = h[:]

        if iteration % patience == 0:
            h = best_h.copy()
            learning_rate *= 0.95

    elapsed = time.time() - start_time
    if elapsed < budget_s:
        h_values = best_h.tolist()
        c5_bound = best_c5
    else:
        h_values = best_h.tolist()
        c5_bound = best_c5

    return h_values, c5_bound, n_points
