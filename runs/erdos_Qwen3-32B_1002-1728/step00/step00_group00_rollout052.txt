            h[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
        return h

    bounds = [(0.0, 1.0)] * n_points
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    options = {'maxiter': 50, 'ftol': 1e-6, 'disp': False}

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    tries = 0
    seed_gen = np.random.Generator(np.random.PCG64(seed))
    while time.time() - start_time < budget_s - 1e-6:
        if tries >= 15:
            break
        rand_seed = seed_gen.integers(1 << 30)
        h_initial = generate_perturbation(best_h, seed=rand_seed)
        res = minimize(
            compute_c5, h_initial, method='SLSQP',
            bounds=bounds, constraints=constraints, options=options
        )
        if res.success:
            new_c5 = compute_c5(res.x)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = res.x.copy()
        tries += 1

    return best_h, best_c5, n_points
