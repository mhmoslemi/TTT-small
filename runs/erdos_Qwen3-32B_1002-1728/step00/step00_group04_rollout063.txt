import numpy as np
import time
import initial_h_values

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values.initial_h_values)
    n = len(h)
    dx = 2.0 / n
    best_h = np.copy(h)
    best_c5 = compute_c5(h, n, dx)
    start_time = time.perf_counter()
    
    while time.perf_counter() - start_time < budget_s - 1:
        # Compute current overlaps to determine k_max
        correlate_result = np.correlate(h, 1 - h, mode='full')
        relevant_overlaps = correlate_result[n : 2 * n - 1] * dx
        
        if len(relevant_overlaps) == 0:
            break
        
        idx_max = np.argmax(relevant_overlaps)
        k_max = idx_max + 1
        
        # Coordinate-wise optimization
        for i in range(n):
            term1 = 0.0
            if (i + k_max) < n:
                term1 = 1.0 - h[i + k_max]
            term2 = 0.0
            if i >= k_max:
                term2 = -h[i - k_max]
            A = term1 + term2
            if A > 0:
                h[i] = 0.0
            elif A < 0:
                h[i] = 1.0
        
        # Mass redistribution
        target_sum = n / 2.0
        delta = (target_sum - h.sum()) / n
        h += delta
        h = np.clip(h, 0.0, 1.0)

            perturb_indices = np.random.choice(n, size=int(0.05 * n), replace=False)

        # Update best solution
        new_c5 = compute_c5(h, n, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5

    return best_h.tolist(), best_c5, n
