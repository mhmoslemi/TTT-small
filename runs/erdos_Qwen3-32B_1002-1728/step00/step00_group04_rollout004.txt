    start_time = time.time()

    try:
        initial_h = initial_h_values.copy()
        n_points = len(initial_h)
    except NameError:
        n_points = 42
        h = beta.rvs(2, 2, size=n_points)
        h_proj = project_to_feasible(h, n_points)
        initial_h = h_proj

    m = 30
    particles = [initial_h.copy()] if 'initial_h_values' in globals() else []
    while len(particles) < m:
        h = beta.rvs(2, 2, size=n_points)
        h_proj = project_to_feasible(h, n_points)
        particles.append(h_proj)
    particles = np.array(particles)

    velocities = np.random.rand(m, n_points) * 0.1

    pbest = particles.copy()
    pbest_C5 = np.array([compute_C5(p) for p in particles])
    gbest_idx = np.argmin(pbest_C5)
    gbest = particles[gbest_idx].copy()
    gbest_C5 = pbest_C5[gbest_idx]

    omega, c1, c2 = 0.7, 1.5, 1.5
    omega_decay = 0.001

    while time.time() - start_time < budget_s:
        for i in range(m):
            r1, r2 = np.random.rand(), np.random.rand()
            vel_cognitive = c1 * r1 * (pbest[i] - particles[i])
            vel_social = c2 * r2 * (gbest - particles[i])
            velocities[i] = omega * velocities[i] + vel_cognitive + vel_social
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = project_to_feasible(particles[i], n_points)
            current_C5 = compute_C5(particles[i])
            if current_C5 < pbest_C5[i]:
                pbest[i] = particles[i].copy()
                pbest_C5[i] = current_C5
                if current_C5 < gbest_C5:
                    gbest = particles[i].copy()
                    gbest_C5 = current_C5
        omega = max(0.4, omega - omega_decay)

    assert np.all(gbest >= 0) and np.all(gbest <= 1)
    assert np.isclose(np.sum(gbest), n_points / 2, atol=1e-6)
    return gbest.tolist(), gbest_C5, n_points
