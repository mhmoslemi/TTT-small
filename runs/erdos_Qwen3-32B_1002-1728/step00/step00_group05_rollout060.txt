        h = h_result
    
    return best_h, best_c5, best_n

def interpolate_to_resolution(old_h, old_n, new_n):
    x_old = np.linspace(0, 2, old_n)
    f = interpolate.interp1d(x_old, old_h, kind='linear', fill_value="extrapolate")
    x_new = np.linspace(0, 2, new_n)
    new_h = f(x_new)
    return np.clip(new_h, 0.0, 1.0)
