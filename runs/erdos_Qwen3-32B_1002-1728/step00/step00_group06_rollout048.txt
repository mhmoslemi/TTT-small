import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    best_h = h.copy()
    dx = 2.0 / n_points
    
    alpha = 0.05
    start_time = time.time()
    best_c5 = np.inf
    iterations = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s and iterations < 10000:
        # Compute overlaps using numpy.correlate
        h_flipped = 1.0 - h
        overlaps = np.correlate(h, h_flipped, mode='full') * dx
        current_c5 = np.max(overlaps)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Reduce learning rate if no improvement
        if no_improvement >= 100:
            alpha *= 0.5
            no_improvement = 0
        
        # Find the shift with maximum overlap
        max_idx = np.argmax(overlaps)
        k_star = max_idx - (n_points - 1)
        
        # Build shifted_h with zero-padding
        if k_star >= 0:
            padded = np.zeros(n_points + k_star)
            padded[k_star:] = h
            shifted_h = padded[:n_points]
        else:
            padded = np.zeros(n_points - k_star)
            padded[:n_points + k_star] = h
            shifted_h = padded[-k_star:]
        
        # Compute contributions
        contributions = h * (1.0 - shifted_h)
        
        # Update h
        h -= alpha * contributions
        
        # Clip to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        
        # Normalize
        total = h.sum()
        if total > 1e-8:
            h *= (n_points / 2.0) / total
        else:
            h[:] = 0.5 * np.ones(n_points)
        
        iterations += 1
    
    return best_h, best_c5, n_points
