            if total < 1e-10:
                h_clipped[0] = n_points / 2.0
                h_clipped[1:] = 0.0
            else:
                scale = (n_points / 2.0) / total
                h_clipped *= scale
            particles[i] = h_clipped
        
        new_scores = np.array([evaluate(p) for p in particles])
        improved = new_scores < pbest_scores
        pbest_scores[improved] = new_scores[improved]
        pbest_positions[improved] = particles[improved]
        
        new_gbest_idx = np.argmin(pbest_scores)
        if pbest_scores[new_gbest_idx] < gbest_score:
            gbest_score = pbest_scores[new_gbest_idx]
            gbest_position = particles[new_gbest_idx].copy()
        
        if time.time() - start_time > budget_s:
            break
    
    return (tuple(gbest_position), gbest_score, n_points)
