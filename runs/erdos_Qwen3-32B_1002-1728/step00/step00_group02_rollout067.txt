def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    import evaluate_erdos_solution

    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_single_h(h, dx)

    # SA parameters
    temp = 1.0
    alpha = 0.95
    max_iterations_per_temp = 100
    stagnation_counter = 0
    max_stagnation = 20

    while (time.time() - start_time) < budget_s:
        for _ in range(max_iterations_per_temp):
            # Perturb h
            perturbation_type = np.random.choice(['single', 'segment', 'swap'], p=[0.5, 0.3, 0.2])
            new_h = h.copy()

            if perturbation_type == 'single':
                idx = np.random.randint(n_points)
                new_val = np.random.uniform(0, 1)
                new_h[idx] = new_val
            elif perturbation_type == 'segment':
                l = np.random.randint(2, 11)  # 2-10 inclusive
                i = np.random.randint(n_points)
                j = (i + l) % n_points
                if i <= j:
                    segment = new_h[i:j]
                else:
                    segment = np.concatenate((new_h[i:], new_h[:j]))
                # Shift direction: -1 left, +1 right
                direction = np.random.choice([-1, 1], p=[0.5, 0.5])
                segment = np.roll(segment, direction)
                if i <= j:
                    new_h[i:j] = segment.copy()
                else:
                    # Handle wrap-around
                    new_h[i:] = segment[:n_points - i]
                    new_h[:j] = segment[n_points - i:]

            elif perturbation_type == 'swap':
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]

            # Project to feasible region
            new_h_clipped = np.clip(new_h, 0, 1)
            sum_h = np.sum(new_h_clipped)
            if sum_h == 0:
                new_h_clipped = initial_h_values.copy()
            else:
                scale = (n_points / 2) / sum_h
                new_h_clipped *= scale

            # Compute new_C5
            new_C5 = evaluate_single_h(new_h_clipped, dx)

            # Acceptance
            old_C5 = evaluate_single_h(h, dx)
            delta = new_C5 - old_C5
            if delta < 0 or np.exp(-delta / temp) > np.random.rand():
                h = new_h_clipped
                if new_C5 < best_C5:
                    best_h = h.copy()
                    best_C5 = new_C5
                    stagnation_counter = 0
                else:
                    stagnation_counter += 1
            else:
                stagnation_counter += 1

            if stagnation_counter >= max_stagnation:
                temp *= alpha
                stagnation_counter = 0

        # End of temp loop
        # Check time again
        if temp < 1e-6:
            break

    # Return
    return (best_h.tolist(), best_C5, n_points)
