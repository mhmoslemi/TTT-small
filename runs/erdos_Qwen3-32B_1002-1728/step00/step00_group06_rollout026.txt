import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time()
    learning_rate = 0.05
    max_iterations = int(1e6)  # Conservative upper limit
    iteration = 0
    tolerance = 1e-6
    prev_c5 = np.inf
    
    while time() - start_time < budget_s and iteration < max_iterations:
        iteration += 1
        
        overlap_result = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(overlap_result)
        max_idx = np.argmax(overlap_result)
        k = max_idx - (n_points - 1)
        k_used = int(k % n_points)
        
        i_values = np.arange(n_points)
        j_plus = (i_values + k_used) % n_points
        j_minus = (i_values - k_used) % n_points
        grad = (1 - h[j_plus]) - h[j_minus]
        
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if sum_h > 0:
            h *= (n_points / 2) / sum_h
            h = np.clip(h, 0.0, 1.0)
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if abs(current_c5 - prev_c5) < tolerance:
            break
        prev_c5 = current_c5
    
    final_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    return h, final_c5, n_points
