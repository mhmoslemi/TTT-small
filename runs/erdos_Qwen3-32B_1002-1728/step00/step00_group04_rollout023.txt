def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h_values = initial_h_values
    n_points = len(h_values)
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    np.random.seed(seed)
    m = 30  # Number of particles
    c1 = 1.5
    c2 = 1.5
    omega = 0.7
    min_omega = 0.4
    
    particles = [h_values.copy()]
    velocities = [np.zeros_like(h_values)]
    
    for _ in range(m - 1):
        candidate = np.random.beta(2, 2, size=n_points)
        particles.append(project(candidate))
        velocities.append(np.zeros_like(candidate))
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Evaluate initial C5
    def evaluate_c5(h):
        from functools import partial
        import numpy as np
        return np.max(np.correlate(h, 1 - h, mode="full") * dx)
    
    c5_scores = np.array([evaluate_c5(p) for p in particles])
    personal_best_positions = particles.copy()
    personal_best_c5 = c5_scores.copy()
    
    global_best_idx = np.argmin(personal_best_c5)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    best_h = global_best_position.copy()
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        for i in range(m):
            r1 = np.random.rand()
            r2 = np.random.rand()
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best_positions[i] - particles[i]) + c2 * r2 * (global_best_position - particles[i])
            particles[i] += velocities[i]
            clipped = np.clip(particles[i], 0, 1)
            particles[i] = project(clipped)
            c5_scores[i] = evaluate_c5(particles[i])
            
            if c5_scores[i] < personal_best_c5[i]:
                personal_best_c5[i] = c5_scores[i]
                personal_best_positions[i] = particles[i].copy()
                if c5_scores[i] < global_best_c5:
                    global_best_c5 = c5_scores[i]
                    global_best_position = particles[i].copy()
                    best_h = particles[i].copy()
        
        if omega > min_omega:
            omega -= 0.001
    
    assert np.isclose(np.sum(best_h), target_sum), f"Sum of h is {np.sum(best_h)}, expected {target_sum}"
    assert np.all((best_h >= 0) & (best_h <= 1)), "Values out of bounds"
    
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points

def project(candidate):
    n_points = len(candidate)
    target_sum = n_points / 2.0
    h = cp.Variable(n_points)
    constraints = [cp.sum(h) == target_sum, h <= 1, h >= 0]
    objective = cp.Minimize(cp.sum_squares(h - candidate))
    problem = cp.Problem(objective, constraints)
    
    try:
        problem.solve(solver=cp.ECOS, max_iter=100)
        if problem.status == cp.OPTIMAL:
            return np.clip(np.array(h.value).flatten(), 0, 1)
    except Exception:
        pass
    
    # Fallback to simple rounding if projection fails
    residue = target_sum - np.sum(candidate)
    scaling = 2.0 / n_points
    rounding = np.round(candidate * scaling) / scaling
    rounding = np.clip(rounding, 0, 1)
    rounding += residue / n_points
    return np.clip(rounding, 0, 1)
