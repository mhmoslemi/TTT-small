    dx = 2.0 / n_points
    
    cons = [{'type': 'eq', 'fun': constraint_eq, 'args': (n_points,)}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        fun=compute_objective,
        x0=h0,
        args=(n_points,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    h_opt = np.clip(result.x if result.success else h0, 0, 1)
    c5_bound = evaluate_erdos_solution(h_opt)
    elapsed = time.time() - start_time
    
    if elapsed < budget_s * 0.9:
        perturbed_h = h_opt + np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        current_sum = np.sum(perturbed_h)
        scale_factor = (n_points / 2) / current_sum if current_sum > 1e-8 else 1.0
        perturbed_h = np.clip(perturbed_h * scale_factor, 0, 1)
        
        result2 = minimize(
            fun=compute_objective,
            x0=perturbed_h,
            args=(n_points,),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        if result2.success:
            h_opt_new = np.clip(result2.x, 0, 1)
            c5_new = evaluate_erdos_solution(h_opt_new)
            if c5_new < c5_bound:
                h_opt, c5_bound = h_opt_new, c5_new
    
    return h_opt, c5_bound, n_points
