    def compute_c5(h):
        dx = 2.0 / n_points
        h1 = 1 - h
        corrs = np.correlate(h, h1, mode='full')
        return np.max(corrs) * dx

    best_c5 = float('inf')
    best_h = None
    generations = 0

    while time.time() - start_time < budget_s:

        c5_list = []

import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    n_points = 80
    population_size = 50
    tournament_size = 5
    mutation_std = 0.05
    target_sum = n_points / 2

    global initial_h_values
    if 'initial_h_values' not in globals():
        raise RuntimeError("initial_h_values not available")

    def repair(h, target_sum=target_sum):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        if abs(current_sum - target_sum) < 1e-8:
            return h_clipped
        if current_sum == 0:
            return np.full_like(h, target_sum / n_points)
        elif current_sum == n_points:
            return np.full_like(h, 0.5)
        scale = target_sum / current_sum
        h_scaled = h_clipped * scale
        h_scaled = np.clip(h_scaled, 0, 1)
        current_sum = np.sum(h_scaled)
        remaining = target_sum - current_sum
        idx = np.where((h_scaled > 0) & (h_scaled < 1))[0]
        if len(idx) > 0:
            per_element = remaining / len(idx)
            h_scaled[idx] += per_element
            h_scaled = np.clip(h_scaled, 0, 1)
        return h_scaled

    def compute_c5(h, dx=2.0/n_points):
        h1 = 1 - h
        corrs = np.correlate(h, h1, mode='full')
        return np.max(corrs) * dx

    population = [initial_h_values.copy()]
    for _ in range(population_size - 1):
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, int(target_sum), replace=False)
        h[indices] = 1.0
        population.append(h)
    population = np.array(population)

    best_c5 = float('inf')
    best_h = None

    while time.time() - start_time < budget_s:
        c5_list = []
        for i in range(population_size):
            h = repair(population[i])
            c5 = compute_c5(h)
            c5_list.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        new_population = []
        for _ in range(population_size):
            # Parent 1
            parents1 = np.random.choice(population_size, tournament_size)
            c5_p1 = [c5_list[i] for i in parents1]
            winner1 = parents1[np.argmin(c5_p1)]
            # Parent 2
            parents2 = np.random.choice(population_size, tournament_size)
            c5_p2 = [c5_list[i] for i in parents2]
            winner2 = parents2[np.argmin(c5_p2)]
            p1 = population[winner1]
            p2 = population[winner2]

            # Crossover
            mask = np.random.rand(n_points) < 0.5
            child = np.where(mask, p1, p2)

            # Mutation
            noise = np.random.normal(0, mutation_std, child.shape)
            child += noise
            child = np.clip(child, 0, 1)

            # Repair
            child = repair(child)
            new_population.append(child)

        population = np.array(new_population)

    if best_h is not None:
        dx = 2.0 / n_points
        best_c5 = compute_c5(best_h, dx)
    else:
        best_c5 = 0.38085  # fallback to known bound
        best_h = initial_h_values.copy()

    return (best_h, best_c5, n_points)
