import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    initial_h = initial_h_values  # Pre-imported global variable
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Convert h to z for optimization
    z_initial = 2.0 * initial_h - 1.0
    
    # Define optimization variables
    z = cp.Variable(n_points)
    c = cp.Variable()
    
    # Constraints: bounds and sum(z) = 0
    constraints = [
        z >= -1,
        z <= 1,
        cp.sum(z) == 0
    ]
    
    # Add cross-correlation constraints for all m shifts
    for m in range(n_points):
        terms = [z[i] * z[i + m] for i in range(n_points - m)]
        if terms:
            constraints.append(cp.sum(terms) >= c)
    
    problem = cp.Problem(cp.Maximize(c), constraints)
    
    # Track time and solve
    start_time = time.time()
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        is_optimal = problem.status == cp.OPTIMAL
    except:
        is_optimal = False
    
    # Use optimized z if successful, else fallback to initial
    if is_optimal:
        z_opt = z.value
    else:
        z_opt = z_initial
    
    # Convert back to h and normalize sum
    h_opt = (z_opt + 1) / 2
    h_sum = np.sum(h_opt)
    if not np.isclose(h_sum, n_points / 2):
        h_opt *= (n_points / 2) / h_sum
    
    # Compute C₅ using actual cross-correlation
    correlation = np.correlate(h_opt, 1 - h_opt, mode="full")
    c5_bound = np.max(correlation * dx)
    
    return h_opt.tolist(), c5_bound, n_points
