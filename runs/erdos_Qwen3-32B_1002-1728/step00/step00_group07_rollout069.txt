import numpy as np
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def repair_h(h, desired_sum):
    h_clipped = np.clip(h, 0.0, 1.0)
    current_sum = h_clipped.sum()
    if current_sum == 0:
        return 0.5 * np.ones_like(h)
    ratio = desired_sum / current_sum
    adjusted = h_clipped * ratio
    adjusted = np.clip(adjusted, 0.0, 1.0)
    delta = desired_sum - adjusted.sum()
    if abs(delta) > 1e-6:
        delta /= len(h)
        adjusted += delta
        adjusted = np.clip(adjusted, 0.0, 1.0)
    return adjusted

def generate_random_h(n_points):
    return repair_h(np.random.rand(n_points), n_points / 2)

def tournament_selection(pop, fitness, size=3):

    return pop[idxs[0]], pop[idxs[1]]

    mask = np.random.rand(len(p1)) > 0.5
