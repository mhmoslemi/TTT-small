def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # ∑h == n_points/2

    # Enforce initial constraint compliance
    h = h.clip(0, 1)
    current_sum = h.sum()
    if np.abs(current_sum - target_sum) > 1e-8:
        h *= target_sum / current_sum

    best_h = h.copy()
    best_c5 = None
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 1:
        # Compute overlap integral for all lags
        overlap = np.correlate(h, 1 - h, mode='full')
        current_c5_candidate = overlap.max() * dx
        current_c5_idx = overlap.argmax()
        lag = current_c5_idx - (n_points - 1)  # Map index to lag value

        # Update best known solution
        if best_c5 is None or current_c5_candidate < best_c5:
            best_c5 = current_c5_candidate
            best_h = h.copy()
            iteration_no_imp = 0
        else:
            iteration_no_imp += 1
            if iteration_no_imp > 300:
                break  # Early stopping

        # Compute gradient for most problematic lag
        grad = np.zeros(n_points, dtype=np.float64)
        for i in range(n_points):
            j = (i + lag) % n_points
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx

        # Step using learning rate schedule
        lr = 0.1 * (0.95 ** (iteration // 40))

        # Projection step: maintain [0,1] constraint

        # Enforce integral constraint

        h = h.clip(0, 1)

        iteration += 1
        if iteration % 100 == 0:
            total_time = time.time() - start_time
            print(f"Iter: {iteration} | Time: {total_time:.2f}s | C₅: {best_c5:.8f}")
