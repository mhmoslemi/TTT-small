    while time.time() - start_time < budget_s:
        inertia_weight = max(0.05, 0.7 - 0.001 * (time.time() - start_time))

        for i in range(m):
            # Velocity update
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = (inertia_weight * velocities[i] +
                             c1 * r1 * (personal_best[i] - particles[i]) +
                             c2 * r2 * (global_best - particles[i]))

            # Position update
            particles[i] += velocities[i]
            particles[i] = adjust_sum(particles[i], target_sum=n_points / 2)

            # Evaluate new position
            c5_current = compute_c5(particles[i])

            # Update personal best
            if c5_current < personal_best_c5[i]:
                personal_best[i] = particles[i].copy()
                personal_best_c5[i] = c5_current
                if c5_current < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = c5_current

        # Progress check every 50 iterations
        if (time.time() - start_time) > 10.0:
            pass  # Budget tracking handled by outer loop

    return (global_best, global_best_c5, n_points)
