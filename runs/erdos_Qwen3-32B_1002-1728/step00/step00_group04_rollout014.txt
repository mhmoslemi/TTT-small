def project(h, n_points):

    # Clip to [0, 1]

    h_clipped = np.clip(h, 0.0, 1.0)

    # Renormalize to have sum = n_points / 2

    current_sum = np.sum(h_clipped)

    if current_sum == 0:

        # All zeros, but this is impossible since sum must be n_points /2

        h_clipped[0] = n_points / 2

    else:

        scale = (n_points / 2.0) / current_sum

        h_clipped *= scale

    return h_clipped
