import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100, **kwargs):
    import time
    start_time = time.time()
    
    h_initial = initial_h_values
    n_points = len(h_initial)
    z_initial = 2 * h_initial - 1
    
    z = cp.Variable(n_points)
    c = cp.Variable()
    
    constraints = [
        z >= -1,
        z <= 1,
        cp.sum(z) == 0
    ]
    
    for m in range(n_points):
        terms = [z[i] * z[i + m] for i in range(n_points - m)]
        constraints.append(cp.sum(terms) >= c)
    
    objective = cp.Maximize(c)
    problem = cp.Problem(objective, constraints)
    
    best_z = None
    best_c = -np.inf
    
    solvers_list = [cp.PDLP, cp.CBC, cp.GLPK, cp.CVXOPT]
    for solver in solvers_list:
        if time.time() - start_time > budget_s - 30:
            break
        try:
            problem.solve(solver=solver, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE] and problem.value > best_c:
                best_c = problem.value
                best_z = z.value.copy()
        except Exception:
            continue
    
    if best_z is None:
        z_sol = z_initial
    else:
        z_sol = best_z
    
    h_sol = (np.clip(z_sol, -1, 1) + 1) / 2
    sum_h = np.sum(h_sol)
    correction = (n_points / 2 - sum_h) / n_points
    h_sol = np.clip(h_sol + correction, 0, 1)
    
    dx = 2.0 / n_points
    corr = np.correlate(h_sol, 1 - h_sol, mode='full') * dx
    c5_bound = np.max(corr)
    
    assert np.all(h_sol >= 0) and np.all(h_sol <= 1)
    assert abs(np.sum(h_sol) - n_points / 2) < 1e-6
    
    return h_sol, c5_bound, n_points
