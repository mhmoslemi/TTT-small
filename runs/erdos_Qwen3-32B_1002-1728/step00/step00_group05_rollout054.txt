import numpy as np
import scipy.optimize
import time

def evaluate_erdos_C5(h):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(result)

def objective_function(h_vector):
    h = h_vector.copy()
    return evaluate_erdos_C5(h)

def optimize_h(n_points, initial_h=None):
    if initial_h is not None:
        # Nearest-neighbor interpolation to new grid
        if len(initial_h) < n_points:

            h0 = np.repeat(initial_h, ratio)
            if len(h0) < n_points:
