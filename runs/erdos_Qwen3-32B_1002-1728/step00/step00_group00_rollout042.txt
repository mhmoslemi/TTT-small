def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_population = 50
    mutation_rate = 0.2
    n_points = 45  # Assume initial_h_values provided

    population = []
    for _ in range(n_population):
        h = initial_h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        h[i], h[j] = h[j], h[i]
        population.append(h)

    def evaluate(h):
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        return np.max(corr) * dx

    pop_scores = [(h, evaluate(h)) for h in population]
    pop_scores.sort(key=lambda x: x[1])
    best_h, best_c5 = pop_scores[0]
    start_time = time.time()

    def tournament():
        candidates = np.random.choice(len(pop_scores), size=3, replace=False)
        best_idx = min(candidates, key=lambda i: pop_scores[i][1])
        return pop_scores[best_idx][0].copy()

    while time.time() - start_time < budget_s:
        parent1 = tournament()
