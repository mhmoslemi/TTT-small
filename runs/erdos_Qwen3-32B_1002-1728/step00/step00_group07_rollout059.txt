import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    learning_rate = 0.01

    while time.time() - start_time < budget_s:
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        k_max_index = np.argmax(overlap)
        k = k_max_index - (n_points - 1)

        grad = np.zeros(n_points)
        for i in range(n_points):
            ip = i + k
            if 0 <= ip < n_points:
                grad[i] += 1 - h[ip]
            im = i - k
            if 0 <= im < n_points:
                grad[i] -= h[im]

        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        current_sum = h.sum()
        if current_sum != 0:
            h *= target_sum / current_sum

        # Recompute the overlap after the update
        overlap_new = np.correlate(h, 1 - h, mode='full') * dx
        new_c5 = np.max(overlap_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()

    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
