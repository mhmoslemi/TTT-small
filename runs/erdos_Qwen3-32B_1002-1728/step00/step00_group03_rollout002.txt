def compute_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        j = i + k
        if j < n:
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    return grad

def adjust_sum(h_clamped, target_sum, atol=1e-8):
    current_sum = np.sum(h_clamped)
    required = target_sum
    delta = required - current_sum
    if abs(delta) < atol:
        return h_clamped.copy()
    h = h_clamped.copy()
    n = len(h)
    if delta > 0:  # Need to increase sum
        # Sort indices in ascending order of h
        indices = np.argsort(h)
        for idx in indices:
            if h[idx] >= 1.0:
                continue
            available = 1.0 - h[idx]
            add = min(available, delta)
            h[idx] += add
            delta -= add
            if delta <= atol:
                break

        # Sort indices in descending order of h

                continue
            available = h[idx]
            sub = min(available, delta)
            h[idx] -= sub
            delta -= sub
            if delta <= atol:
                break
    return h
