def project_to_feasible(h_proposed, target_sum):
    n = len(h_proposed)
    h_clipped = np.clip(h_proposed, 0.0, 1.0)
    # Solve the projection problem: minimize ||h - h_clipped||^2 s.t. sum(h) = target_sum and 0 <= h[i] <=1
    h = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h - h_clipped))
    constraints = [cp.sum(h) == target_sum, h >= 0, h <= 1]
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.PDLP)
    return np.array(h.value).flatten()

import numpy as np

import copy

def project_to_feasible(h_proposed, target_sum):
    n = len(h_proposed)
    h_clipped = np.clip(h_proposed, 0.0, 1.0)
    h = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(h - h_clipped))
    constraints = [cp.sum(h) == target_sum, h >= 0, h <= 1]
