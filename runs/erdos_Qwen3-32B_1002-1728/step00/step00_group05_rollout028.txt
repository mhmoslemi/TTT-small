        h_tensor.assign(tf.clip_by_value(h_tensor, 0.0, 1.0))
        current_sum = tf.reduce_sum(h_tensor)
        scaling = (n_points / 2) / current_sum
        h_tensor.assign(h_tensor * scaling)
        
        if i % iterations_per_log == 0:
            current_h = h_tensor.numpy()
            current_c5 = evaluate_c5_local(current_h, dx)
            if current_c5 < best_c5_approx:
                best_h[:] = current_h.copy()
                best_c5_approx = current_c5
    
    # Final evaluation using official evaluator
    final_best_h = best_h
    c5_bound = evaluate_erdos_solution(h_values=final_best_h, n_points=n_points)
    
    return final_best_h.tolist(), c5_bound, n_points
