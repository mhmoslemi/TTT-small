    for k in range(total_shifts):
        shifted = [(i + k) % n_points for i in range(n_points)]
        terms = [h[i] * (1 - h[s_idx]) for i, s_idx in enumerate(shifted)]
        overlap_sum = sum(terms)
        objective_expr += overlap_sum ** 2

    objective = cp.Minimize(objective_expr)
    problem = cp.Problem(objective, constraints)

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return h_init, evaluate_erdos_solution(h_init), n_points

    try:
        problem.solve(solver=cp.ECOS, verbose=False)
    except:
        return h_init, evaluate_erdos_solution(h_init), n_points

    if problem.status not in {cp.OPTIMAL, cp.OPTIMAL_INACCURATE}:
        return h_init, evaluate_erdos_solution(h_init), n_points

    h_values = np.clip(h.value, 0, 1)
    c5_bound = evaluate_erdos_solution(h_values)
    return h_values, c5_bound, n_points
