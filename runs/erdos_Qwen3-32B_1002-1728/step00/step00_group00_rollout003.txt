import numpy as np
from scipy.optimize import minimize, Bounds

def run(seed=42, budget_s=1100, **kwargs):
    import time

    base_seed = seed
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_vec):
        h = h_vec.reshape(-1)
        ccorr = np.correlate(h, 1 - h, mode='full')
        return np.max(ccorr) * dx

    sum_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}
    bounds = Bounds(0.0, 1.0)

    best_h = h.copy()
    best_c5 = objective(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        np.random.seed(base_seed)
        perturbation = 0.1 * np.random.randn(n_points)
        h0 = best_h + perturbation
        h0 = np.clip(h0, 0, 1)
        current_sum = np.sum(h0)
        target_sum = n_points / 2.0
        h0 = h0 * (target_sum / current_sum) if current_sum != 0 else h0
        h0 = np.clip(h0, 0, 1)
        result = minimize(
            objective, h0, method='SLSQP', bounds=bounds,
            constraints=[sum_constraint], options={'maxiter': 150, 'ftol': 1e-6}
        )
        if result.success:
            candidate_h = result.x
            candidate_c5 = result.fun
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h
        base_seed += 1
        if base_seed > 2**32:  # Prevent seed cycling
            base_seed = 0

    return (best_h, best_c5, n_points)
