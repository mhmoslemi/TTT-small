        perturbed_h = np.clip(perturbed_h, 0, 1)
        current_sum = np.sum(perturbed_h)
        perturbed_h *= (n_points / 2) / current_sum
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        result2 = minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 300}
        )
        if result2.success and objective(result2.x) < objective(result.x):
            result = result2

    best_h = np.clip(result.x, 0, 1)
    c5_bound = objective(best_h)
    
    # Final projection onto feasible region
    best_h = np.clip(best_h, 0, 1)
    scale = (n_points / 2) / best_h.sum()
    best_h = np.clip(best_h * scale, 0, 1)

    return best_h, c5_bound, n_points
