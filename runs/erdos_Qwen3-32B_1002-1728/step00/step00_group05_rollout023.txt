import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial_h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    pop_size = 100
    n_select = 50
    max_gen = 1000
    mutation_rate = 0.1
    mutation_std = 0.1

    def norm_h(h):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s == 0:
            return h * target_sum
        return h * (target_sum / s)

    pop = [norm_h(np.random.uniform(0, 1, n_points)) for _ in range(pop_size - 1)]
    pop.append(norm_h(initial_h))
    pop = np.array(pop)
    best_c5 = float('inf')
    best_h = None

    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s:
            break

        c5_list = []
        for h in pop:
            cc = correlate(h, 1 - h, mode='full')
            c5 = np.max(cc * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            c5_list.append(c5)

        idx = np.argsort(c5_list)
        selected = pop[idx[:n_select]]

        new_pop = []
        while len(new_pop) < pop_size - n_select:
            p1, p2 = selected[np.random.choice(n_select, 2)]
            child = np.where(np.random.rand(n_points) < 0.5, p1, p2)
            if np.random.rand() < mutation_rate:
                mask = np.random.rand(n_points) < mutation_rate
                noise = np.random.normal(0, mutation_std, n_points)
                child += mask * noise
            child = norm_h(child)
            new_pop.append(child)

        pop = np.concatenate([selected, new_pop])

    return (best_h, best_c5, n_points)
