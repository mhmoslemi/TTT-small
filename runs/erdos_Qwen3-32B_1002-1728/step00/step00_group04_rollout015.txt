import numpy as np

import math

def project_to_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return np.ones(n_points) * 0.5
    target_sum = n_points / 2.0
    scaling = target_sum / current_sum
    h_scaled = h_clipped * scaling
    h_projected = np.clip(h_scaled, 0, 1)
    return h_projected

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1 - h, mode='full')
    relevant = overlap[n:]  # for shifts k=1 to n-1
    c5 = np.max(relevant * dx)
    return c5

def generate_individual(n_points):
    h_rand = np.random.beta(2, 2, n_points)
    return project_to_feasible(h_rand, n_points)

def compute_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n - k):
        grad[i] += (1 - h[i + k])
        grad[i + k] -= h[i]
    return grad

def local_optimize(h, n_points, learning_rate=0.01, steps=5):
    for _ in range(steps):
        dx = 2.0 / n_points
        overlap = np.correlate(h, 1 - h, mode='full')

def crossover(parent1, parent2):
    n = len(parent1)
    indices = np.sort(np.random.choice(n, 2, replace=False))
    a, b = indices
    child = np.zeros(n)
    child[:a] = parent1[:a]
    child[a:b] = parent2[a:b]
    child[b:] = parent1[b:]

    noise = np.random.randn(n) * 0.1  # sigma=0.1
    individual = individual.copy()
    individual[mask] += noise[mask]
    return project_to_feasible(individual, n)

import numpy as np
import time
import math

def project_to_feasible(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    if current_sum == 0:
        return np.ones(n_points) * 0.5
    target_sum = n_points / 2.0
    scaling = target_sum / current_sum
    h_scaled = h_clipped * scaling
    h_projected = np.clip(h_scaled, 0, 1)
    return h_projected

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1 - h, mode='full')
    relevant = overlap[n:]
    c5 = np.max(relevant * dx)
    return c5

def compute_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    for i in range(n - k):
        grad[i] += (1 - h[i + k])
        grad[i + k] -= h[i]
    return grad

def local_optimize(h, n_points, learning_rate=0.01, steps=20):
    for _ in range(steps):
        dx = 2.0 / n_points
        overlap = np.correlate(h, 1 - h, mode='full')
        relevant = overlap[n_points:]
        max_index = np.argmax(relevant)
        k = max_index + 1
        if k >= len(h):
            continue
        grad = compute_grad(h, k)
        h = h - learning_rate * grad
        h = project_to_feasible(h, n_points)
    return h

def crossover(parent1, parent2):
    n = len(parent1)
    indices = np.sort(np.random.choice(n, 2, replace=False))
    a, b = indices
    child = np.zeros(n)
    child[:a] = parent1[:a]
    child[a:b] = parent2[a:b]
    child[b:] = parent1[b:]
    return project_to_feasible(child, n)

    noise = np.random.randn(len(individual)) * 0.1
    individual = individual.copy()
    individual[mask] += noise[mask]
