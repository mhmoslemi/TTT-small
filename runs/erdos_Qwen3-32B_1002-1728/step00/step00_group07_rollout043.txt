import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()
    max_time = budget_s

    best_h = h.copy()
    best_C5 = None

    while time.time() - start_time < max_time:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        current_C5 = max_corr * dx

        if best_C5 is None or current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

        max_idx = np.argmax(corr)
        k_shift = max_idx - (n_points - 1)

        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + k_shift
            if 0 <= j < n_points:
                grad[i] += 1.0 - h[j]
                grad[j] -= h[i]
        grad *= dx

        learning_rate = 0.02
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)

        current_sum = np.sum(h)
        delta = target_sum - current_sum
        h += delta / n_points
        h = np.clip(h, 0, 1)

    return best_h.tolist(), best_C5, n_points
