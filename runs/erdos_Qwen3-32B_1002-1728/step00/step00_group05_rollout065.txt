import numpy as np
import time
from numpy import correlate

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    initial_h_values_global = initial_h_values
    n_coarse = 70
    n_fine = 280
    s_coarse = n_coarse / 2
    s_fine = n_fine / 2
    
    def evaluate_c5(h, n):
        dx = 2.0 / n
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def optimize(h, n, s, max_iter=500, lr=0.01, decay_rate=0.99):
        best_h = h.copy()
        best_c5 = evaluate_c5(h, n)
        start_time = time.time()
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            corr = np.correlate(h, 1 - h, mode='full')
            corr_vals = corr * (2.0 / n)
            max_val = np.max(corr_vals)
            max_k = np.argmax(corr_vals)
            k_max = max_k - (corr_vals.shape[0] - n)
            
            # Gradient calculation
            dx = 2.0 / n
            g = np.zeros(n)
            start_i = max(0, -k_max)
            end_i = min(n, n - k_max)
            for i in range(start_i, end_i):
                j = i + k_max
                if j < 0 or j >= n:
                    continue
                g[i] += (1 - h[j]) * dx
                g[j] -= h[i] * dx
            
            # Update h
            h -= lr * g
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            delta = s - sum_h
            h += delta / n
            h = np.clip(h, 0, 1)
            
            c5 = evaluate_c5(h, n)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            lr *= decay_rate
        return best_h, best_c5
    
    def upsample(h, new_n):
        repeat = new_n // len(h)
        return np.repeat(h, repeat).astype(np.float64)
    
    # Coarse grid optimization
    h_coarse = initial_h_values_global.copy()
    h_coarse_best, _ = optimize(h_coarse, n_coarse, s_coarse, max_iter=300, lr=0.05)
    
    # Fine grid optimization
    h_fine = upsample(h_coarse_best, n_fine) * s_fine / np.sum(h_coarse_best)
    h_fine = np.clip(h_fine, 0, 1)
    
    h_fine_best, c5_final = optimize(h_fine, n_fine, s_fine, max_iter=800, lr=0.008)
    return (h_fine_best.tolist(), c5_final, n_fine)
