        if current_sum > 1e-10:

        # Track best solution
        if current_c5 < best_c5:
            best_h[:] = h[:]
            best_c5 = current_c5
            no_improvement = 0
        else:
            no_improvement += 1

        # Learning rate decay
        if no_improvement >= patience_decay and learning_rate > 1e-7:
            learning_rate *= decay_factor
            no_improvement = 0

        # Early stopping
        if no_improvement >= max_no_improvement:
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(n_points, best_h)
    return best_h, final_c5, n_points
