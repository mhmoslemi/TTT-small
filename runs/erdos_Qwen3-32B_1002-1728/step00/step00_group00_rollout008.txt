import numpy as np
import time

def mutate_swap(h, n_points):
    h = h.copy()
    i, j = np.random.choice(n_points, size=2, replace=False)
    h[i], h[j] = h[j], h[i]
    return h

def crossover_uniform(parent1, parent2, n_points):
    child1 = np.zeros(n_points, dtype=np.float64)
    child2 = np.zeros(n_points, dtype=np.float64)
    for i in range(n_points):
        if np.random.rand() < 0.5:
            child1[i] = parent1[i]
            child2[i] = parent2[i]
        else:
            child1[i] = parent2[i]
            child2[i] = parent1[i]
    return child1, child2

def adjust_sum(candidate, T):
    candidate = candidate.copy()
    current_sum = np.sum(candidate)
    if np.isclose(current_sum, T):
        return candidate
    delta = T - current_sum

    if delta > 0:
        indices = np.where(candidate < 1)[0]
        indices.sort()
        for i in indices:
            add = min(delta, 1 - candidate[i])
            candidate[i] += add
            delta -= add
            if delta < 1e-10:
                break
    elif delta < 0:
        indices = np.where(candidate > 0)[0]
        indices.sort(kind='mergesort')[::-1]
        for i in indices:
            remove = min(-delta, candidate[i])
            candidate[i] -= remove
            delta += remove
            if delta > -1e-10:
                break
    return candidate

def perturb(h, T, noise_std=0.05):
    candidate = h + np.random.normal(0, noise_std, len(h))
    candidate = np.clip(candidate, 0, 1)
    return adjust_sum(candidate, T)

def selection_tournament(population, fitnesses, tournament_size, population_size):
    selected = []
    for _ in range(population_size):
        contestants = np.random.choice(population_size, size=tournament_size, replace=False)
        best_idx = contestants[np.argmax([fitnesses[i] for i in contestants])]
        selected.append(population[best_idx])
    return selected

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    global initial_h_values
    initial = initial_h_values
    n_points = len(initial)
    T = n_points / 2.0

    population_size = 50
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 3
    dx = 2.0 / n_points

    population = [initial.copy()]
    for _ in range(population_size - 1):
        candidate = perturb(initial, T)
        population.append(candidate)

    def evaluate_individual(h):
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr) * dx

    fitnesses = [1 / (evaluate_individual(h) + 1e-10) for h in population]

    best_idx = np.argmax(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = evaluate_individual(best_h)
    start_time = time.time()

    generation = 0
    while time.time() - start_time < budget_s:
        parents = selection_tournament(population, fitnesses, tournament_size, population_size)

        new_population = []
        for i in range(0, population_size, 2):
            parent1 = parents[i]
            parent2 = parents[(i + 1) % population_size]

            if np.random.rand() < crossover_rate:
                child1, child2 = crossover_uniform(parent1, parent2, n_points)
                child1 = adjust_sum(child1, T)
                child2 = adjust_sum(child2, T)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()

            if np.random.rand() < mutation_rate:
                child1 = mutate_swap(child1, n_points)
            if np.random.rand() < mutation_rate:
                child2 = mutate_swap(child2, n_points)

            new_population.extend([child1, child2])

        new_fitnesses = [1 / (evaluate_individual(h) + 1e-10) for h in new_population]
        combined_pop = population + new_population
