import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    required_sum = n_points / 2

    def project(h_vec):
        h_proj = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_proj)
        if not np.isclose(current_sum, required_sum):
            h_proj *= required_sum / current_sum
        return h_proj

    def compute_c5(h_vec):
        h_proj = project(h_vec)
        overlaps = np.correlate(h_proj, 1 - h_proj, mode='full') * dx
        return np.max(overlaps)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    T = 1.0
    cooling_rate = 0.9995
    mutation_scale = 0.1
    start_time = time.time()

    while time.time() - start_time < budget_s:
        candidate = best_h + np.random.normal(0, mutation_scale * T, n_points)
        candidate = project(candidate)
        current_c5 = compute_c5(candidate)
        delta = current_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = candidate.copy()
            best_c5 = current_c5
        T *= cooling_rate

    best_h = project(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
