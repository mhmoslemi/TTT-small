import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    max_time = 1100

    # Initialize with provided initial_h_values or a default
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        n = len(h)
        # Normalize to meet ∫h dx = 1
        h *= (n / 2) / h.sum()
        h = np.clip(h, 0, 1)
    else:
        # Simple default: a central pulse
        n = 70
        h = np.zeros(n)
        mid = n // 2
        h[mid-5:mid+5] = 0.5
        h *= (n / 2) / h.sum()

    best_h, best_c5, best_n = h.copy(), np.max(np.correlate(h, 1-h, 'full') * (2 / n)), n

    grid_sizes = [70, 280]
    for grid in grid_sizes:
        if grid == best_n:
            current_h = best_h.copy()
        else:
            # Upsample using linear interpolation
            old_n = best_n
            old_x = np.linspace(0, 2, old_n)
            new_x = np.linspace(0, 2, grid)
            current_h = np.interp(new_x, old_x, best_h)  # Interpolate for upsample
            current_h *= (grid / 2) / current_h.sum()     # Enforce sum constraint
            current_h = np.clip(current_h, 0, 1)

        # Objective function
        def objective(x):
            h = np.clip(x, 0, 1).flatten()
            corr = np.correlate(h, 1-h, mode='full') * (2 / grid)
            return np.max(corr)

        bounds = Bounds(0, 1)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - grid / 2}]
        result = minimize(objective, current_h, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-6, 'maxiter': 200})

        if result.success:

            new_c5 = result.fun
            if new_c5 < best_c5:
                best_h, best_c5, best_n = optimized_h, new_c5, grid

        # Early termination check
        if time.time() - start_time > max_time * 0.98:
            break

    return best_h, best_c5, best_n
