import numpy as np
import scipy.optimize as opt
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time()
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points

    def objective(h_flat):
        h_vals = h_flat.copy()
        cross_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(cross_corr * dx)

    def sum_constraint(h_flat):
        return np.sum(h_flat) - n_points / 2

    h_init = h.copy()
    bounds = [(0., 1.) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': sum_constraint}

    result = opt.minimize(
        objective,
        h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    h_opt = np.clip(result.x, 0, 1)
    total = h_opt.sum()
    if abs(total - n_points/2) > 1e-5:
        scale = (n_points/2) / total
        h_opt = np.clip(h_opt * scale, 0, 1)

    c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full') * dx)
    return (h_opt.tolist(), c5, n_points)
