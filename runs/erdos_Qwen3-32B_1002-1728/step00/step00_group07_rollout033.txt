def compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, 'full')) * dx

def repair_constraints(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    if current_sum == 0:
        return h
    scale = target_sum / current_sum
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    delta = target_sum - h.sum()
    while abs(delta) > 1e-6:
        non_ones = np.where(h < 1.0)[0]
        non_zeros = np.where(h > 0.0)[0]
        if delta > 0 and len(non_ones) > 0:
            idx = np.random.choice(non_ones, min(10, len(non_ones)))
            max_add = np.minimum(1.0 - h[idx], np.abs(delta))
            h[idx] += max_add
            delta += -max_add.sum()
        elif delta < 0 and len(non_zeros) > 0:
            idx = np.random.choice(non_zeros, min(10, len(non_zeros)))
            max_subtract = np.minimum(-delta, h[idx])
            h[idx] -= max_subtract
            delta += max_subtract.sum()
        else:
            break
    h = np.clip(h, 0.0, 1.0)
    return h
