import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    sum_h = np.sum(h)
    h = h * (n / 2) / sum_h
    
    learning_rate = 0.1
    decay_factor = 0.5
    decay_interval = 100
    max_no_improvement = 500
    best_c5 = np.inf
    best_h = h.copy()
    no_improvement = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 10:
        target = 1.0 - h
        corr = np.correlate(h, target, mode='full')
        max_val = np.max(corr)
        k_index = np.argmax(corr)
        lag = k_index - (n - 1)
        
        i_plus = (np.arange(n) + lag) % n
        i_minus = (np.arange(n) - lag) % n
        gradient = (1 - h[i_plus]) - h[i_minus]
        
        h -= learning_rate * gradient
        np.clip(h, 0.0, 1.0, out=h)
        sum_h = np.sum(h)
        h *= (n / 2) / sum_h
        
        current_c5 = max_val * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement > max_no_improvement:
            break
        
        if (time.time() - start_time) % decay_interval < 0.1:
            learning_rate *= decay_factor
    
    return best_h.tolist(), best_c5, n
