import numpy as np
import time

def hash_state(h, precision=5):
    return tuple(np.round(h, precision).tolist())

def project_feasible(h, n_points):
    target_sum = n_points / 2.0
    clipped = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(clipped)
    if current_sum == 0:
        # Avoid division by zero; this should not happen
        return np.zeros_like(h)
    scale = target_sum / current_sum
    return clipped * scale

def compute_overlaps(h, dx):
    return np.correlate(h, 1 - h, mode='full') * dx

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)

    n_points = len(h)

    # Check initial sum

    best_h = h.copy()
    best_c5 = np.max(compute_overlaps(h, dx))
    
    tabu_set = set()
    tabu_size = 50
    max_iterations_without_improvement = 50
    iterations_without_improvement = 0
    eta = 0.1  # Learning rate
    
    while time.time() - start_time < budget_s:
        # Current overlaps
        overlaps = compute_overlaps(h, dx)
        current_c5 = np.max(overlaps)
        k_star_index = np.argmax(overlaps)

def hash_state(h, precision=5):

def project_feasible(h, n_points):
