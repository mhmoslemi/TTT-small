def repair_individual(h, desired_sum):

    h = np.clip(h, 0.0, 1.0)
    current_sum = np.sum(h)
    delta = desired_sum - current_sum
    if abs(delta) < 1e-6 * desired_sum:
        return h  # Already meets the sum constraint
    # Determine whether to increase or decrease
    if delta > 0:

        # Iterate through h and add to elements not at max (1)
        indices = np.where(h < 1)[0]
