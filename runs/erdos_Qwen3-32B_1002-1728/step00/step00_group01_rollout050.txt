    current_sum = h.sum()
    diff = required_sum - current_sum
    if abs(diff) > 1e-8:
        for i in range(n_points):
            if 0 < h[i] < 1:
                h[i] += diff
                break

    start_time = time.time()
    best_h = h.copy()
    current_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    best_c5 = current_c5

    T = 1000.0
    alpha = 0.999

    while time.time() - start_time < budget_s:
        i, j = np.random.choice(n_points, 2, replace=False)
        max_delta = min(h[i], 1 - h[j])
        if max_delta <= 0:
            continue
        delta = np.random.uniform(0, max_delta)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_h = np.clip(new_h, 0.0, 1.0)
        corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr * dx)
        if new_c5 < current_c5:
            h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            prob = np.exp((current_c5 - new_c5) / max(T, 1e-6))
            if np.random.rand() < prob:
                h = new_h.copy()
                current_c5 = new_c5
        T *= alpha

    return (best_h.tolist(), best_c5, n_points)
