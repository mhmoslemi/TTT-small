            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'disp': False}
        )

        if result.success:
            current_c5 = compute_c5(result.x, new_n)
            if current_c5 < best_c5:
                best_h = result.x.copy()
                best_c5 = current_c5
                best_n_points = new_n
                h = result.x.copy()
                n_points = new_n

    return (best_h.tolist(), best_c5, best_n_points)
