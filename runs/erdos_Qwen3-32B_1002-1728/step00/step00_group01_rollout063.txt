    z = cp.Variable(n_points)
    c = cp.Variable()
    constraints = []
    constraints += [z >= -1, z <= 1]
    constraints += [cp.sum(z) == 0]
    for m in range(n_points):
        if m == 0:
            expr = cp.sum(z * z)
            constraints += [expr >= c * n_points]
        else:
            expr = cp.sum([z[i] * z[i + m] for i in range(n_points - m)])
            constraints += [expr >= c * (n_points - m)]
    objective = cp.Maximize(c)
    problem = cp.Problem(objective, constraints)
    z.value = z_initial
    start_time = time.time()
    problem.solve(solver=cp.ECOS, verbose=False)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s * 0.95:
        h_values = h.copy()
        dx = 2.0 / n_points
        h_adj = 1 - h_values
        corr = np.correlate(h_values, h_adj, mode='full')
        return h.tolist(), np.max(corr) * dx, n_points
    if problem.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_values = h.copy()
        dx = 2.0 / n_points
        h_adj = 1 - h_values
        corr = np.correlate(h_values, h_adj, mode='full')
        return h.tolist(), np.max(corr) * dx, n_points
    h_values = (z.value + 1) / 2.0
    h_values = np.clip(h_values, 0, 1)
    sum_h = np.sum(h_values)
    if not np.isclose(sum_h, n_points / 2):
        h_values += (n_points / 2 - sum_h) / n_points
        h_values = np.clip(h_values, 0, 1)
    dx = 2.0 / n_points
    h_adj = 1 - h_values
    corr = np.correlate(h_values, h_adj, mode='full')
    c5_bound = np.max(corr) * dx
    return h_values.tolist(), c5_bound, n_points
