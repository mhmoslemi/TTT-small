                noise = np.random.normal(0, sigma, n_points)
                child = np.clip(child + noise, 0, 1)
                # Re-project
                child = project_feasible(child, target_sum)
            offspring.append(child)
        
        # Replace population
        population = np.concatenate([selected, np.array(offspring)])
    
    # Validate best_h
    if best_h is not None:
        dx = 2.0 / n_points
        assert np.allclose(best_h.sum() * dx, 1.0), "Sum violation"
    return (best_h, best_c5, n_points)
