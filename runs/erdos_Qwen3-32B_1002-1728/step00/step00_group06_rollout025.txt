import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    start_time = time.time()

    learning_rate = 0.01
    max_iter = 100000  # Cap to ensure time safety

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # Evaluate current C5 and worst shift
        h_f = 1 - h
        corr = np.correlate(h, h_f, mode='full') * dx
        current_c5 = np.max(corr)

        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)

        if lag == 0:
            continue

        k_star = abs(int(lag)) if lag < 0 else int(lag)

        # Compute gradient for k_star
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_star
            if 0 <= j < n_points:
                grad[i] += (1 - h[j])
                grad[j] -= h[i]

        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-8:
            grad = grad / grad_norm

        # Update h with gradient descent
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)

        # Enforce the average constraint
        current_sum = h.sum()
        scaling_factor = target_sum / current_sum
        h *= scaling_factor
        h = np.clip(h, 0, 1)

    return (h.tolist(), float(current_c5), n_points)
