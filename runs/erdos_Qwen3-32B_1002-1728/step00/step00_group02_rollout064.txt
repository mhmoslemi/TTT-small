    import time

        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    def project_feasible(h, n_points):
        h_clipped = np.clip(h, 0, 1)
        current_sum = np.sum(h_clipped)
        target = n_points / 2.0
        if current_sum != 0:
            h_scaled = h_clipped * (target / current_sum)
        else:
            h_scaled = np.full_like(h, 0.5)
        return h_scaled

    best_c5 = compute_overlaps(best_h, dx)
    tabu_list = set()
    eta = 0.1
    iteration = 0
    max_no_improvement = 50
    no_improvement_count = 0
    tabu_max_size = 50
    start_time = time.time()

    while time.time() - start_time < budget_s:
        h = best_h.copy()
        overlap = np.correlate(h, 1 - h, mode="full")
        idx_max = np.argmax(overlap)
        k_shift = idx_max - (n_points - 1)

        n = len(h)
        if k_shift >= 0:
            valid_i = np.arange(n - k_shift)
        else:
            valid_i = np.arange(-k_shift, n)

        contributions = []
        for i in valid_i:
            j = i + k_shift
            contributions.append(h[i] * (1 - h[j]))
        contributions = np.array(contributions)
        top_indices = valid_i[np.argsort(-contributions)]
        top_n = max(1, len(top_indices) // 5)
        top_indices = top_indices[:top_n]

        for i in top_indices:
            j = i + k_shift
            grad = 1 - h[j]
            h[i] -= eta * grad

        current_c5 = compute_overlaps(h, 2.0 / len(h))

        h_rounded = tuple(np.round(h, decimals=6))
