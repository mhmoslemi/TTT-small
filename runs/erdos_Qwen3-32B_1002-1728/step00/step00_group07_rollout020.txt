import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    assert np.isclose(np.sum(h), n_points / 2), "Initial h does not satisfy sum constraint"

    def compute_c5(current_h):
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        return np.max(corr) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()

    T = 0.2
    cooling_rate = 0.999
    start_time = time.time()

    while time.time() - start_time < budget_s - 0.1:
        i, j = np.random.choice(n_points, size=2, replace=False)
        h_i = h[i]
        h_j = h[j]

        lower = max(-h_i, h_j - 1.0)
        upper = min(1.0 - h_i, h_j)

        if lower > upper:
            continue

        delta = np.random.uniform(lower, upper)
        old_i = h[i].copy()
        old_j = h[j].copy()
        h[i] += delta
        h[j] -= delta
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)

        new_c5 = compute_c5(h)

        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = h[:]
        else:
            delta_c5 = new_c5 - best_c5
            if delta_c5 != 0:
                prob = np.exp(-delta_c5 / T)
                if np.random.rand() >= prob:
                    h[i] = old_i
                    h[j] = old_j

        T *= cooling_rate

    best_h = np.clip(best_h, 0.0, 1.0)
    assert np.isclose(np.sum(best_h), n_points / 2), "Final h violates sum constraint"
    return (best_h.tolist(), best_c5, n_points)
