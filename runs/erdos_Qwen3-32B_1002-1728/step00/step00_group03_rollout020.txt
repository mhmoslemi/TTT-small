    h_clipped = np.clip(h, 0, 1)
    n = len(h_clipped)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, target_sum):
        return h_clipped
    delta = target_sum - current_sum
    indices = np.arange(n)
    if delta > 0:
        # Need to increase: find elements not at 1
        can_increase = indices[h_clipped < 1]
        if can_increase.size > 0:
            # Sort by current value
            increasing_order = can_increase[np.argsort(h_clipped[can_increase])]
            for i in increasing_order:
                possible = 1 - h_clipped[i]
                add = min(delta, possible)
                h_clipped[i] += add

    else:

        can_decrease = indices[h_clipped > 0]
        if can_decrease.size > 0:

            decreasing_order = can_decrease[np.argsort(-h_clipped[can_decrease])]

import numpy as np

def project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    n = len(h_clipped)
    current_sum = np.sum(h_clipped)
    if np.isclose(current_sum, target_sum):
        return h_clipped
    delta = target_sum - current_sum
    indices = np.arange(n)
    if delta > 0:
        # Need to increase: find elements not at 1
        can_increase = indices[h_clipped < 1]
        if can_increase.size > 0:
            # Sort by current value
            increasing_order = can_increase[np.argsort(h_clipped[can_increase])]
            for i in increasing_order:
                possible = 1 - h_clipped[i]
                add = min(delta, possible)
                h_clipped[i] += add
                delta -= add
                if delta <= 1e-10:  # Close enough
                    break
    else:
        # Need to decrease: find elements not at 0
        can_decrease = indices[h_clipped > 0]
        if can_decrease.size > 0:
            # Sort by current value in descending order
            decreasing_order = can_decrease[np.argsort(-h_clipped[can_decrease])]
            for i in decreasing_order:
                possible = h_clipped[i]
                subtract = min(-delta, possible)
                h_clipped[i] -= subtract
                delta += subtract
                if delta >= -1e-10:
                    break

    current_sum = np.sum(h_clipped)
    if not np.isclose(current_sum, target_sum):
        h_clipped = h_clipped * (target_sum / current_sum)
