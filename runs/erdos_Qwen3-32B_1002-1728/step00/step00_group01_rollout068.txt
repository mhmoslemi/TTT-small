                best_c5 = new_c5
        else:
            delta_energy = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta_energy / T):
                current_h = new_h.copy()
                current_c5 = new_c5

        T *= alpha

    return (best_h.tolist(), best_c5, n_points)
