import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    initial = initial_h_values.copy()
    n_points = len(initial)
    sum_h_target = n_points / 2.0
    dx = 2.0 / n_points

    pop_size = 50
    tournament_size_pct = 0.3
    mutation_strength = 0.1
    max_generations = 1500
    time_budget = budget_s

    def generate_candidate():
        h = np.random.rand(n_points)
        current_sum = h.sum()
        h_scaled = h * (sum_h_target / current_sum)
        h_clipped = np.clip(h_scaled, 0, 1)
        delta = sum_h_target - h_clipped.sum()
        h_clipped += delta / n_points
        h_clipped = np.clip(h_clipped, 0, 1)
        return h_clipped

    def evaluate(h):
        overlaps = np.correlate(h, 1 - h, mode='full')
        return np.max(overlaps) * dx

    population = [initial]
    population += [generate_candidate() for _ in range(pop_size - 1)]

    best_h = population[0].copy()
    best_c5 = evaluate(best_h)
    start_time = time.time()

    for generation in range(max_generations):
        if time.time() - start_time > time_budget:
            break

        # Tournament selection
        ts = int(pop_size * tournament_size_pct)
        parent_indices = [np.argmin([evaluate(population[i]) for i in np.random.choice(pop_size, ts, replace=False)]) for _ in range(pop_size)]

        # Crossover
        offspring = []
        for i in range(0, pop_size, 2):
            if i + 1 < pop_size:
                p1 = population[parent_indices[i]]
                p2 = population[parent_indices[i+1]]
            else:
                p1 = population[parent_indices[i]]
                p2 = p1.copy()
            alpha = np.random.uniform(0.4, 0.6)
            child1 = alpha * p1 + (1 - alpha) * p2
            child2 = alpha * p2 + (1 - alpha) * p1
            offspring.append(child1)
            offspring.append(child2)

        # Mutation
        epsilon = 1e-5
        for i in range(len(offspring)):
            m = (np.random.rand(n_points) - 0.5) * 2 * mutation_strength
            offspring[i] = np.clip(offspring[i] + m, 0, 1)
            delta = sum_h_target - offspring[i].sum()
            offspring[i] += delta / n_points
            offspring[i] = np.clip(offspring[i], 0, 1)
            offspring[i] = np.clip(offspring[i] + (sum_h_target - offspring[i].sum()) / n_points, 0, 1)

        # Merge and select
        population += offspring[:pop_size]
        scores = np.array([evaluate(h) for h in population])
        ranked = [(s, h) for s, h in zip(scores, population)]
        ranked.sort()
        population = [h for s, h in ranked[:pop_size]]
        if ranked[0][0] < best_c5:
            best_c5 = ranked[0][0]
            best_h[:] = ranked[0][1].copy()

    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h += (sum_h_target - best_h.sum()) / n_points
    best_h = np.clip(best_h, 0, 1)
    return best_h, best_c5, n_points
