        # Find worst-case lag
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)

        # Compute gradient for worst-case lag
        grad = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += 1.0 - h[j + k]
            if 0 <= j - k < n_points:
                grad[j] -= h[j - k]

        # Gradient update
        h -= alpha * grad

        # Project to feasible region
        x = cp.Variable(n_points)
        obj = cp.Minimize(cp.sum_squares(x - h))
        constraints = [x >= 0, x <= 1, cp.sum(x) == desired_sum]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=50, abstol=1e-6, reltol=1e-6)
            if x.value is not None:
                h = np.clip(x.value, 0, 1)
        except:
            pass

        # Ensure exact sum constraint
        h_sum = h.sum()
        if abs(h_sum - desired_sum) > 1e-6:
            delta = (desired_sum - h_sum) / n_points
            h += delta
            h = np.clip(h, 0, 1)

        # Learning rate schedule
        iter_count += 1
        if iter_count % 100 == 0:
            alpha *= 0.9

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    return best_h.tolist(), np.max(corr), n_points
