        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-6}
    )
    
    if result.success:
        h_opt = result.x.reshape(n_points)
        c5_bound = objective(h_opt, n_points, dx)
    else:
        h_opt = x0
        c5_bound = objective(x0, n_points, dx)
    
    return (h_opt.tolist(), c5_bound, n_points)
