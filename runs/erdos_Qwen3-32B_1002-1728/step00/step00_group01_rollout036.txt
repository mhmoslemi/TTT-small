import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy().astype(float)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points
    cons =({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})

    def compute_c5(h_vec):
        h_flat = h_vec.reshape(n_points)
        corr = np.correlate(h_flat, 1 - h_flat, mode='full') * dx
        return np.max(corr), np.argmax(corr)

    start_time = time.time()
    best_c5, best_h = compute_c5(h)

    for _ in range(20):
        if time.time() - start_time > budget_s * 0.95:
            break
        current_c5, max_idx = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        def objective(x):
            x = x.reshape(n_points)
            corr = np.correlate(x, 1 - x, mode='full') * dx
            return corr[max_idx]

        res = minimize(
            objective, 
            x0=h, 
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50}
        )
        if res.success and np.allclose(np.sum(res.x), target_sum):
            h = np.clip(res.x, 0, 1)

    final_c5, _ = compute_c5(h)
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if not np.allclose(h_sum, target_sum):
        delta = target_sum - h_sum
        h += delta / n_points
        h = np.clip(h, 0, 1)

    return h, final_c5, n_points
