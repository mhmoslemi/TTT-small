    start_time = time.time()
    budget_s = 1100

    n_points = len(initial_h)

        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr * dx)

    def objective(h_vec):
        return compute_c5(h_vec.reshape(n_points))

    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0

    x0 = initial_h.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}

    result = minimize(
        objective,
        x0=x0,
        args=(),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-6, 'disp': False}
    )

    if result.success:
        optimized_h = result.x.reshape(n_points)
    else:
        optimized_h = x0.copy()

    # Project to feasible set
    optimized_h /= np.sum(optimized_h)
