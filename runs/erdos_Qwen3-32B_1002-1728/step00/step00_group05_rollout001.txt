import numpy as np
import time
import scipy.optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    h0 = initial_h_values.copy().flatten()
    n_points = len(h0)
    dx = 2.0 / n_points
    rng = np.random.default_rng(seed)
    h0 += 0.01 * rng.normal(size=h0.shape)  # Small perturbation
    h0 = np.clip(h0, 0, 1)
    # Ensure sum constraint
    h0 /= (np.sum(h0) / (n_points / 2))
    h0 = np.clip(h0, 0, 1)
    
    best_h = h0.copy()
    best_c5 = float('inf')
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr * dx)
    
    def callback(x):
        nonlocal best_h, best_c5
        current_c5 = objective(x)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h[:] = x.copy()
    
    # Enforce sum(h) = n_points/2 constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint_sum})
    bounds = [(0, 1)] * n_points
    
    result = opt.minimize(
        objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        callback=callback,
        options={'maxiter': 500, 'ftol': 1e-6}
    )
    
    # Final validation and projection
    best_h = np.clip(best_h, 0, 1)
    current_sum = np.sum(best_h)
    adjustment = (n_points / 2 - current_sum) / n_points
    best_h += adjustment
    best_h = np.clip(best_h, 0, 1)
    
    # Compute final bound
    c5_bound = objective(best_h)
    
    return best_h, c5_bound, n_points
