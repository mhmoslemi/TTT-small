import numpy as np
from scipy.optimize import minimize
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    start_time = time()
    
    def compute_c5(h, dx):
        overlaps = np.correlate(h, 1 - h, mode='full')
        return np.max(overlaps) * dx

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    while time() - start_time < budget_s - 1:
        # Find worst overlap
        overlaps = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(overlaps)
        shift_k = max_idx - (n_points - 1)
        
        # Skip zero shift as it's fixed by sum constraint
        if shift_k == 0:
            break
        
        def objective(x_vec):
            x = x_vec.reshape(n_points)
            if shift_k > 0:
                val = np.sum(x[:n_points - shift_k] * (1 - x[shift_k:]))
            elif shift_k < 0:
                k_neg = -shift_k
                val = np.sum(x[k_neg:] * (1 - x[:n_points - k_neg]))
            else:
                return 0.0
            return -val * dx  # Minimize negative of overlap

        # Constraints: sum(h) = n_points / 2
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        # Bounds: [0,1]
        bounds = [(0.0, 1.0)] * n_points
        
        result = minimize(
            objective,
            x0=h,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-6}
        )

        if result.success:
            new_h = result.x.reshape(n_points)
            # Enforce bounds and sum
            new_h = np.clip(new_h, 0, 1)
            delta = (np.sum(new_h) - n_points / 2)
            if abs(delta) > 1e-8:
                new_h -= delta / n_points
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
            h = new_h
        
        # Early exit if time is nearly up
        if time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
