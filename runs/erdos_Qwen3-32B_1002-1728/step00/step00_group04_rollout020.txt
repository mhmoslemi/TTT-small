import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    h = initial_h_values.copy().astype(float)
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    learning_rate = 0.01
    lr_decay = 0.9
    no_improvement = 0
    improvement_threshold = 1e-6

    while time.time() - start_time < budget_s - 5:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        valid_overlaps = overlaps[n:]
        current_c5 = np.max(valid_overlaps)
        k_index = np.argmax(valid_overlaps)
        k = k_index + 1

        if abs(best_c5 - current_c5) > improvement_threshold:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            if no_improvement > 50:
                learning_rate *= lr_decay
                no_improvement = 0

        grad = np.zeros(n)
        for i in range(n - k):
            grad[i] += (1 - h[i + k]) * dx
            grad[i + k] -= h[i] * dx
        h_candidate = h - learning_rate * grad

        h_proj = cp.Variable(n)
        constraints = [
            cp.sum(h_proj) == target_sum,
            h_proj >= 0,
            h_proj <= 1
        ]
        problem = cp.Problem(
            cp.Minimize(0.5 * cp.sum_squares(h_proj - h_candidate)),
            constraints
        )
        problem.solve(solver=cp.ECOS, verbose=False)
        h = h_proj.value.astype(float)

    c5_bound = evaluate_erdos_solution(best_h)
    return best_h.tolist(), c5_bound, n
