import time
import numpy as np
import cvxpy as cp

def run(seed=42, budget_s=1100):
    h = initial_h_values.copy().astype(float)
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    learn_rate = 0.01

    # Define projection problem once
    param = cp.Parameter(n_points)
    h_proj = cp.Variable(n_points)
    objective = cp.Minimize(0.5 * cp.sum_squares(h_proj - param))
    constraints = [0 <= h_proj, h_proj <= 1, cp.sum(h_proj) == n_points / 2]
    projection_problem = cp.Problem(objective, constraints)

    iteration = 0
    while time.time() - start_time < budget_s:
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        valid_overlaps = overlaps[n_points-1 : 2*n_points-1]
        current_c5 = np.max(valid_overlaps)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            learn_rate = 0.01  # Reset learning rate on improvement
        else:
            if iteration % 100 == 0:
                learn_rate *= 0.95

        k_max = np.argmax(valid_overlaps)

        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]

        h_candidate = h - learn_rate * grad

        # Project
        param.value = h_candidate
        projection_problem.solve(solver=cp.ECOS, verbose=False)
        if projection_problem.status != cp.OPTIMAL:
            break
        h = h_proj.value.copy().astype(float)

        iteration += 1

    if best_h is None:
        best_h = h.copy()
    return (best_h, best_c5, n_points)
