import numpy as np
import cvxpy as cp
from time import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h_proposed, n_points):
    n = len(h_proposed)
    h = cp.Variable(n)
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    objective = cp.Minimize(cp.sum_squares(h - h_proposed))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        if prob.status != cp.OPTIMAL:
            return h_proposed.copy()
    except:
        return h_proposed.copy()
    return h.value

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time()
    n_points = 42
    m = 20  # number of particles

    # Initialize particles
    positions = []
    velocities = []
