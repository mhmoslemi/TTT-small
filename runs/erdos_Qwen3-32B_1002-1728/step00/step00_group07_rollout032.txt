def tournament_selection(pop, fitnesses, tournament_size):
    # 假设fitnesses是较小的当前C5值更好，所以选择fitness最小的
    # 随机选择tournament_size的索引，找出其中的最小值
    indices = np.random.choice(len(pop), tournament_size, replace=False)
    best_idx = indices[0]
    for idx in indices[1:]:
        if fitnesses[idx] < fitnesses[best_idx]:
            best_idx = idx
    return pop[best_idx]
