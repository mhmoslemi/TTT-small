import numpy as np
import time

def repair(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    current_sum = np.sum(h_clipped)
    delta = target_sum - current_sum
    if np.isclose(delta, 0):
        return h_clipped
    if delta > 0:
        available = 1.0 - h_clipped
        available_total = np.sum(available)
        if available_total > 1e-10:
            fraction = delta / available_total
            h_clipped += available * fraction
    else:
        available = h_clipped
        available_total = np.sum(available)
        if available_total > 1e-10:
            fraction = (-delta) / available_total
            h_clipped -= available * fraction
    h_clipped = np.clip(h_clipped, 0, 1)
    return h_clipped

def tournament_selection(fitnesses, tournament_size=5):
    indices = np.random.choice(len(fitnesses), tournament_size)
    selected_fitness = [fitnesses[i] for i in indices]
    best_idx = indices[np.argmin(selected_fitness)]
    return best_idx

        for h in population:
