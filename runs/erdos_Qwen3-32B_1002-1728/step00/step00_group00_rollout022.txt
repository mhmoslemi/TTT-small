def compute_c5(h, n_points, dx):
    one_minus_h = 1 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    return max(0.0, np.max(corr))  # Ensure non-negative

def repair_h(h, n_points, target_sum):
    h_clipped = np.clip(h, 0, 1)
    s = np.sum(h_clipped)
    if s == 0:
        return initial_h_values.copy()
    scaling_factor = target_sum / s
    h_scaled = h_clipped * scaling_factor
    h_scaled = np.clip(h_scaled, 0, 1)
    return h_scaled
