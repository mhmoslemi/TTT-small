def normalize(h):
    target_sum = len(h) / 2
    current_sum = h.sum()
    if current_sum == 0:
        return h.copy()
    scale1 = target_sum / current_sum
    h_scaled = h * scale1
    h_clamped = np.clip(h_scaled, 0, 1)
    new_sum = h_clamped.sum()
    if new_sum == 0:
        return h_clamped
    scale2 = target_sum / new_sum
    h_final = h_clamped * scale2
