import numpy as np

def repair_h(h, n_points):
    h_clipped = np.clip(h, 0, 1)
    target_sum = n_points / 2
    for _ in range(2):
        current_sum = h_clipped.sum()
        if np.allclose(current_sum, target_sum, atol=1e-8):
            return h_clipped
        delta = target_sum - current_sum
        h_clipped = np.clip(h_clipped + delta / n_points, 0, 1)
    return h_clipped

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1 - h, mode='full')
    return np.max(overlap) * dx
