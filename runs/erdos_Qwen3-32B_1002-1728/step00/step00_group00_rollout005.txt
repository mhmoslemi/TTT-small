def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    np.random.seed(seed)
    start_time = time.time()

    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    target_sum = n_points / 2.0

    def repair(h):
        h_clipped = np.clip(h, 0.0, 1.0)
        s = h_clipped.sum()
        if s == 0:
            return np.full_like(h, 0.5)
        scale = target_sum / s
        return np.clip(h_clipped * scale, 0.0, 1.0)

    def compute_objective(h):
        dx = 2.0 / n_points
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx

    n_particles = 50
    # Initialize particles near the initial solution
    positions = []
    velocities = []
    for _ in range(n_particles):
        # Perturb initial solution
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        new_h = h_initial + perturbation
        new_h = repair(new_h)
        positions.append(new_h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    positions = np.array(positions)
    velocities = np.array(velocities)

    # Initialize personal bests and global best
    personal_best_values = np.array([compute_objective(p) for p in positions])
    personal_best_positions = positions.copy()
    best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[best_idx].copy()
    global_best_value = personal_best_values[best_idx]

    # PSO parameters
    w = 0.8  # inertia
    c1 = 2.0  # cognitive
    c2 = 2.0  # social

    while time.time() - start_time < budget_s:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +

    return (global_best_position.tolist(), global_best_value, n_points)
