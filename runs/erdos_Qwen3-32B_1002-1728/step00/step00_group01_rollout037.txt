import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h_init = initial_h_values.copy()
    n_points = len(h_init)
    dx = 2.0 / n_points
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def perturb(h):
        h = h.copy().astype(float)
        h += np.random.normal(0, 0.1, size=h.shape)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target = n_points / 2.0
        h += (target - current_sum) / n_points
        h = np.clip(h, 0, 1)
        return h
    
    def optimize(h_initial):
        x0 = h_initial.ravel()
        bounds = scipy.optimize.Bounds(0.0, 1.0)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2.0}]
        result = scipy.optimize.minimize(
            objective, x0=x0, bounds=bounds, constraints=cons,
            method='SLSQP', options={'maxiter': 500, 'ftol': 1e-8}
        )
        if result.success:
            h_opt = result.x.reshape(n_points)
            val = objective(h_opt)
            return h_opt, val
        return h_initial, objective(h_initial.ravel())
    
    best_h, best_c5 = h_init, objective(h_init.ravel())
    start_time = time.time()
    
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        h_perturbed = perturb(h_init)
        h_current, c5 = optimize(h_perturbed)
        if c5 < best_c5:
            best_h, best_c5 = h_current, c5
        if time.time() - start_time >= budget_s:
            break
    
    return best_h, best_c5, n_points
