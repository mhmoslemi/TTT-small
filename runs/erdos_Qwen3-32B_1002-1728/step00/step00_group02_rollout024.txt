import numpy as np

import time

def run(seed=42, budget_s=1100, **kwargs):
    h_initial = np.array(initial_h_values)
    n_points = len(h_initial)

            k_abs = -k
            i_min = k_abs

            i_list = np.arange(i_min, i_max + 1)
            j_list = i_list + k
        else:
            i_min = 0
            i_max = n_points - 1 - k
            i_list = np.arange(i_min, i_max + 1)

    while (high - low) > 1e-6 and (time.time() - start_time) < budget_s:
        mid = (low + high) / 2
        h = cp.Variable(n_points)
        h.value = current_h.copy()

            constraints.append(sum_term >= rhs)
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:

            high = mid
            best_c5 = mid
            best_h = h.value.copy()

        else:
            low = mid
