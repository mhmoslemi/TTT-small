def optimize_step(h_initial, n_points, max_iter=100):
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    h_initial = h_initial.copy()
    h_initial_sum = np.sum(h_initial)
    if not np.isclose(h_initial_sum, target_sum, atol=1e-6):
        scaling_factor = target_sum / h_initial_sum
        h_initial *= scaling_factor
        h_initial = np.clip(h_initial, 0.0, 1.0)
    
    result = minimize(
        objective,
        h_initial.ravel(),
        args=(n_points,),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': 1e-8, 'disp': False}
    )
    return result.x.reshape(n_points), result.fun

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    h = initial_h_values.copy()
    n_points = len(h)
    best_c5 = compute_c5(h, n_points)
    best_h = h.copy()
    best_n = n_points
    
    MAX_GRID_POINTS = 1024
    while (time.time() - start_time) < budget_s:
        time_left = budget_s - (time.time() - start_time)
        max_iter = int(min(150, time_left / 10))  # Allocate approx 10s per iteration
        
        try:
            h_opt, c5_opt = optimize_step(best_h, n_points, max_iter)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass
        
        # Check grid refinement
        if n_points * 2 <= MAX_GRID_POINTS and time_left > 120:
            new_n = n_points * 2
            new_h = np.repeat(best_h, 2)
            target_sum = new_n / 2.0
            current_sum = np.sum(new_h)
            if not np.isclose(current_sum, target_sum, atol=1e-6):
                new_h = new_h * (target_sum / current_sum)
                new_h = np.clip(new_h, 0.0, 1.0)
            n_points = new_n
            best_h = new_h.copy()
        else:
            break
    
    return (best_h, best_c5, best_n)
