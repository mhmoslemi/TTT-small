        h_clipped = np.clip(h, 0, 1)
        s = h_clipped.sum()
        if abs(s - target) < 1e-8:
            return h_clipped
        delta = target - s
        indices = np.argsort(h_clipped) if delta < 0 else np.argsort(-h_clipped)
        for i in indices:
            if delta <= 0:
                break
            adjustment = min(abs(delta), 1 - h_clipped[i] if delta > 0 else h_clipped[i])
            h_clipped[i] += adjustment if delta > 0 else -adjustment
            delta -= adjustment
        return h_clipped

    particles[0] = project(h_init, target_sum)
    personal_best[0] = particles[0].copy()
    current_c5 = evaluate_erdos_solution(particles[0])
    personal_best_values[0] = current_c5
    if current_c5 < global_best_value:
        global_best = particles[0].copy()
        global_best_value = current_c5

    for i in range(1, m):
        particles[i] = project(np.random.rand(n_points), target_sum)
        personal_best[i] = particles[i].copy()
        current_c5 = evaluate_erdos_solution(particles[i])
        personal_best_values[i] = current_c5
        if current_c5 < global_best_value:
            global_best = particles[i].copy()
            global_best_value = current_c5
        velocities[i] = 0.1 * np.random.rand(n_points)

    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        for i in range(m):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] = project(particles[i] + velocities[i], target_sum)
            current_c5 = evaluate_erdos_solution(particles[i])
            if current_c5 < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_c5
            if current_c5 < global_best_value:
                global_best = particles[i].copy()
                global_best_value = current_c5
        omega = max(omega - omega_decay, 0.01)

    return (global_best.tolist(), global_best_value, n_points)
