import numpy as np
import time

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    current_h = h.copy()

    best_c5 = current_c5

    T_initial = 1000.0
    alpha = 0.999
    T = T_initial
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        new_h = current_h.copy()
        # Generate valid perturbation
        while True:
            i, j = np.random.randint(0, n_points, 2)

        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = evaluate_c5(new_h)
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h[:] = new_h[:]

        delta_c5 = current_c5 - new_c5
