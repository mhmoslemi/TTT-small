                grad[m] += (1 - h[m + k])
            if m - k >= 0:
                grad[m] -= h[m - k]
        h -= lr * grad
        h = np.clip(h, 0, 1)
        total_sum = np.sum(h)
        if total_sum > 1e-8:
            h *= (n / 2.0) / total_sum
        else:
            h[:] = 0.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    n_points = 42
    try:
        h_init = initial_h_values.copy()
        h_init = np.clip(h_init, 0, 1)
        sum_h = np.sum(h_init)
        if sum_h > 1e-8:
            h_init = h_init * (n_points / 2.0) / sum_h
        else:
            h_init[:] = 0.0
    except:
        h_init = np.random.beta(2, 2, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2.0) / np.sum(h_init)

    population_size = 20
    population = [h_init]
    for _ in range(1, population_size):
        h = np.random.beta(2, 2, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h)
    population = np.array(population)

    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    while time.time() - start_time < budget_s:
        fitness = np.array([-evaluate_erdos_solution(p) for p in population])
        top_indices = np.argsort(fitness)[-int(population_size * 0.5):]
        parents = population[top_indices]

        offspring = []
        while len(offspring) < population_size - len(parents):
            p1 = parents[np.random.randint(len(parents))]
            p2 = parents[np.random.randint(len(parents))]
            points = np.sort(np.random.choice(n_points - 1, 2, replace=False))
            child = np.copy(p1)
            child[points[0]:points[1]] = p2[points[0]:points[1]]
            if np.random.rand() < 0.05:
                child += np.random.normal(0, 0.1, n_points)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2.0) / np.sum(child)
            offspring.append(child)
        offspring = np.array(offspring)

        top_for_local = parents[np.argsort(fitness[top_indices])[-int(population_size * 0.2):]]
        for h in top_for_local:
            h_opt = local_optimize(h, 0.01, 10)
            offspring = np.append(offspring, [h_opt], axis=0)

        new_pop = np.vstack([parents, offspring])
        new_fit = [-evaluate_erdos_solution(n) for n in new_pop]
        selected = new_pop[np.argsort(new_fit)[-population_size:]]
        population = selected

        current_best = min([evaluate_erdos_solution(h) for h in population])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = selected[np.argmin([evaluate_erdos_solution(h) for h in selected])].copy()

    return (best_h.tolist(), best_c5, n_points)
