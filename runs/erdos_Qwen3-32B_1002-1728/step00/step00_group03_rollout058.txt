import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    # Use provided initial_h_values
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    T = n_points / 2.0  # Integral constraint
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    start_time = time.time()
    
    def objective_with_penalty(h, weight=1.0):
        cross_corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(cross_corr)
        sum_h = np.sum(h)
        return c5 + weight * (sum_h - T)**2
    
    def compute_gradient(h_vec, k_shift, n):
        grad = np.zeros(n)
        for i in range(n):
            j = i + k_shift
            if 0 <= j < n:
                grad[i] += 1 - h_vec[j]
                grad[j] -= h_vec[i]
        return grad
    
    while time.time() - start_time < budget_s - 1:
        h_flat = h.flatten()
        cross_corr = np.correlate(h_flat, 1 - h_flat, mode='full') * dx
        idx_max = np.argmax(cross_corr)
        k_shift = idx_max - (n_points - 1)
        
        gradient = compute_gradient(h_flat, k_shift, n_points)
        grad_penalty = 2 * (np.sum(h_flat) - T)
        grad_total = gradient + grad_penalty * (np.ones_like(h_flat))
        
        # Conjugate gradient step with bound constraints
        def neg_grad(x):
            return -grad_total + 0.1 * (np.sum(x) - T)  # Weak regularization
        
        res = minimize(
            objective_with_penalty, 
            h_flat, 
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            jac=neg_grad,
            options={'maxiter': 2, 'ftol': 1e-3, 'gtol': 5e-1}
        )
