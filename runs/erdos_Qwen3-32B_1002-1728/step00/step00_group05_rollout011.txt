        if np.isclose(current_sum, target_sum):
            return h
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum == 0:
            return np.ones(n_points) * target_sum / n_points
        if current_sum == n_points:
            return np.zeros(n_points)
        scale = target_sum / current_sum
        h_scaled = h * scale
        for _ in range(10):
            h_scaled = np.clip(h_scaled, 0, 1)
            current_sum = np.sum(h_scaled)
            if np.isclose(current_sum, target_sum):
                break
            scale = target_sum / current_sum
            h_scaled *= scale
        return h_scaled

    # Coarse grid optimization (n=70)
    start_time = time.time()
    max_coarse_iterations = 2000
    best_c5 = compute_c5(best_h, n_points)

    for i in range(max_coarse_iterations):
        if time.time() - start_time > budget_s * 0.8:
            break
        max_c, max_k = evaluate_overlap_and_max_k(best_h, n_points)
        grad = compute_gradient(best_h, max_k, n_points)
        alpha = 0.01
        h_new = best_h - alpha * grad
        h_new = project_to_sum(h_new, n_points)
        h_new = np.clip(h_new, 0, 1)
        new_c5 = compute_c5(h_new, n_points)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5

    # Upsample to n=280
    n_points_fine = 280
    h_fine = np.kron(best_h, np.ones(4))
    h_fine = project_to_sum(h_fine, n_points_fine)
    best_fine_h = h_fine.copy()
    best_c5_fine = compute_c5(best_fine_h, n_points_fine)

    # Fine grid optimization
    max_fine_iterations = 3000
    for i in range(max_fine_iterations):
        if time.time() - start_time > budget_s:
            break
        max_c, max_k = evaluate_overlap_and_max_k(best_fine_h, n_points_fine)
        grad = compute_gradient(best_fine_h, max_k, n_points_fine)
        alpha = 0.005
        h_new = best_fine_h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_sum(h_new, n_points_fine)
        new_c = compute_c5(h_new, n_points_fine)
        if new_c < best_c5_fine:
            best_fine_h = h_new.copy()
            best_c5_fine = new_c

    return best_fine_h, best_c5_fine, n_points_fine
