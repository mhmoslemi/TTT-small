import numpy as np
from scipy.optimize import minimize
import time

def run():
    start_time = time.time()
    budget_s = 1100
    h = initial_h_values.copy()
    n_points = len(h)
    target_sum = n_points / 2.0

    def objective(h_flat):
        one_minus_h = 1.0 - h_flat
        corr = np.correlate(h_flat, one_minus_h, mode='full')
        dx = 2.0 / n_points
        return np.max(corr * dx)

    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum})
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    result = minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 500, 'disp': False}
    )

    optimized_h = np.clip(result.x, 0.0, 1.0)
    optimized_h_sum = np.sum(optimized_h)
    if not np.isclose(optimized_h_sum, target_sum, atol=1e-6):
        idx = np.argmin(optimized_h > 1e-8)
        correction = target_sum - optimized_h_sum
        optimized_h[idx] = np.clip(optimized_h[idx] + correction, 0.0, 1.0)

    dx = 2.0 / n_points
    one_minus = 1.0 - optimized_h
    corr = np.correlate(optimized_h, one_minus, mode='full')
    c5_bound = np.max(corr * dx)

    return (optimized_h.tolist(), c5_bound, n_points)
