    bounds = [(0, 1) for _ in range(n_points)]
    sum_constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]

    while time.time() - start_time < budget_s:
        k_max = None
        max_c5 = 0
        for k in range(1, n_points):
            c5 = compute_overlap(current_h, k)
            if c5 > max_c5:
                max_c5 = c5
                k_max = k
        if k_max is None:
            break

        def objective(x):
            shifted = np.roll(x, -k_max)
            return dx * np.sum(x * (1 - shifted))

        result = minimize(objective, x0=current_h, method='SLSQP', bounds=bounds, constraints=sum_constraint)

        if result.success:
            new_h = result.x
            new_h = np.clip(new_h, 0, 1)
            new_sum = np.sum(new_h)
            if abs(new_sum - n_points / 2) > 1e-5:
                new_h = new_h * (n_points / 2) / new_sum
            new_c5 = compute_max_c5(new_h)
            if new_c5 < best_c5:
                best_h[:] = new_h[:]
                best_c5 = new_c5
                current_h[:] = new_h[:]
    return best_h, best_c5, n_points
