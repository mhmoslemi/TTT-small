import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy().astype(np.float64)

    best_h = h.copy()

    start_time = time.time()

    while time.time() - start_time < budget_s:
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        c5_values = corr[-n:] * dx
        current_c5 = np.max(c5_values)
        k_current = np.argmax(c5_values)

        # Greedy zero-out for current worst k
        for i in range(n - k_current):
            h[i] = 0.0

        # Redistribute mass
        total = h.sum()
        alpha = (n / 2 - total) / n
        h += alpha
        h = np.clip(h, 0.0, 1.0)

        # Update best solution
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        c5_values = corr[-n:] * dx
        current_c5 = np.max(c5_values)
        if current_c5 < best_c5:

            h = np.clip(h, 0.0, 1.0)
