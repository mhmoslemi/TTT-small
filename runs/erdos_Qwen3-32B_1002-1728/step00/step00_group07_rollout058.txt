import numpy as np
import time

def run(seed=42, budget_s=1100, **kwargs):
    h = np.array(initial_h_values, dtype=np.float64)
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    start_time = time.time()

    learning_rate = 0.01
    while time.time() - start_time < budget_s - 1.0:
        overlap_full = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(overlap_full)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        max_idx = np.argmax(overlap_full)
        k_shift = max_idx - (n_points - 1)

        gradient = np.zeros_like(h)
        for j in range(n_points):
            j_plus = j + k_shift
            if 0 <= j_plus < n_points:
                gradient[j] += (1.0 - h[j_plus])
            j_minus = j - k_shift
            if 0 <= j_minus < n_points:
                gradient[j] -= h[j_minus]
        gradient *= dx

        h -= learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        scale_factor = target_sum / current_sum
        h *= scale_factor

    return best_h, best_c5, n_points
