import numpy as np
from time import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    one_minus_h = 1 - best_h
    corr = correlate(best_h, one_minus_h, mode='full')
    best_C5 = np.max(corr) * dx
    
    alpha = 0.995
    temp = 1.0
    start_time = time()
    
    while time() - start_time < budget_s - 1:
        # Perturb h
        h_prop = best_h.copy()
        indices = np.random.choice(n_points, int(n_points * 0.1) + 2, replace=False)
        deltas = np.random.normal(0, 0.1 * temp, len(indices))
        h_prop[indices] += deltas
        
        # Normalize and clip
        h_prop = np.clip(h_prop, 0, 1)
        s = h_prop.sum()
        if s > 1e-6:
            h_prop *= (n_points / 2) / s
        h_prop = np.clip(h_prop, 0, 1)
        
        # Compute candidate C5
        one_minus = 1 - h_prop
        corr = correlate(h_prop, one_minus, mode='full')
        current_C5 = np.max(corr) * dx
        
        # Update best_h
        if current_C5 < best_C5 or np.random.rand() < np.exp(-(current_C5 - best_C5)/max(temp, 1e-6)):
            best_h[:] = h_prop

        # Cool down
        decay_interval = 100

            temp *= alpha
