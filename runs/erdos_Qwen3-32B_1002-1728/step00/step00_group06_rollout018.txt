    for iter_count in range(int(1e6)):
        # Check remaining time
        if time.time() - start_time > budget_s - 1e-6:
            break
        
        # Evaluate overlaps across all lags
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            if k == 0:
                contrib = np.sum(h * (1 - h))
            else:
                contrib = np.sum(h[:-k] * (1 - h[k:]))
            overlaps[k] = contrib
        
        overlaps *= dx
        current_c5 = overlaps.max()
        k_max = overlaps.argmax()
        
        # Compute gradient for worst-case overlap
        term1 = np.zeros(n_points)
        term1[:n_points - k_max] = 1 - h[k_max:]
        term2 = np.zeros(n_points)
        term2[k_max:] = h[:n_points - k_max]
        gradient = (term1 - term2) * dx
        
        # Gradient descent update
        h = h - alpha * gradient
        
        # Project onto valid domain
        h = np.clip(h, 0, 1)
        h_sum = h.sum()
        h = h * (target_sum / h_sum)
        
        # Update best solution
        current_c5_new = evaluate_erdos_solution(h)
        if current_c5_new < best_c5:
            best_c5 = current_c5_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Decay learning rate on plateau
            if no_improvement_count > no_improvement_limit:
                alpha *= 0.5
                no_improvement_count = 0
    
    return best_h.tolist(), best_c5, n_points
