import numpy as np

def compute_c5(h_vec):
    n = len(h_vec)
    dx = 2.0 / n
    one_minus_h = 1.0 - h_vec
    corr = np.correlate(h_vec, one_minus_h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    try:
        h = initial_h_values.copy()

        n_points = 47

        h_sum = h.sum()

            delta = (h.sum() - n_points/2) / n_points
            h -= delta / n_points

    best_h = h.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    T = 1000.0
    alpha = 0.9995
