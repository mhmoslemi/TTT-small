import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    n_coarse = 70
    n_fine = 280
    dx_coarse = 2.0 / n_coarse
    dx_fine = 2.0 / n_fine

    h_initial = np.array(initial_h_values)
    h_coarse = h_initial * (n_coarse / 2.0) / np.sum(h_initial)

    def objective(h, n, dx):
        h = np.clip(h, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return np.max(corr)

    def optimize(n, h0, max_iter, time_remaining):
        eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0}
        bounds = [(0.0, 1.0) for _ in range(n)]
        result = opt.minimize(
            lambda h: objective(h, n, 2.0 / n),
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=eq_cons,
            options={'maxiter': max_iter, 'ftol': 1e-5}
        )
        return result.x if result.success or result.nit > 0 else h0

    # Coarse optimization
    h_coarse_opt = h_coarse.copy()
    max_coarse_iter = 50
    for _ in range(2):  # 2 passes
        t0 = time.time()
        h_coarse_opt = optimize(n_coarse, h_coarse_opt, max_coarse_iter, budget_s - (time.time() - start_time))
        if time.time() - start_time > budget_s * 0.9:
            break

    # Fine grid optimization
    upsample_factor = int(n_fine / n_coarse)
    h_fine_initial = np.kron(h_coarse_opt, [1.0 / upsample_factor] * upsample_factor)
    h_fine_initial = h_fine_initial * (n_fine / 2.0) / np.sum(h_fine_initial)
    h_fine_opt = h_fine_initial.copy()
    max_fine_iter = 200
    for _ in range(2):
        t0 = time.time()
        h_fine_opt = optimize(n_fine, h_fine_opt, max_fine_iter, budget_s - (time.time() - start_time))
        if time.time() - start_time > budget_s * 0.99:
            break

    # Evaluate and return best solution
    final_c5_coarse = objective(h_coarse_opt, n_coarse, dx_coarse)
    final_c5_fine = objective(h_fine_opt, n_fine, dx_fine)
    best_h = h_fine_opt if final_c5_fine < final_c5_coarse else h_coarse_opt
    best_n = n_fine if final_c5_fine < final_c5_coarse else n_coarse
    return best_h, min(final_c5_coarse, final_c5_fine), best_n
