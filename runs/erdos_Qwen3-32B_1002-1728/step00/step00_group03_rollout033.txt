import numpy as np
import time
import math
from scipy.signal import correlate

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n_points = len(h)
    desired_sum = n_points / 2.0
    h = np.clip(h, 0, 1)
    current_sum = h.sum()

        diff = desired_sum - current_sum
        h[0] += diff
        h[0] = np.clip(h[0], 0, 1)
    dx = 2.0 / n_points

    def compute_c5(h):
        one_minus_h = 1.0 - h
        corr = correlate(h, one_minus_h, mode='full')
        return np.max(corr * dx)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5

    T = 1.0
    alpha = 0.99
    iter_no_improve = 0

    while time.time() - start_time < budget_s - 10:
        i, j = np.random.choice(n_points, size=2, replace=False)
        s = current_h[i] + current_h[j]
        new_i = np.random.uniform(max(0, s - 1), min(1, s))
        new_j = s - new_i
        new_h = current_h.copy()
        new_h[i], new_h[j] = new_i, new_j

        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5

        if delta < 0:
