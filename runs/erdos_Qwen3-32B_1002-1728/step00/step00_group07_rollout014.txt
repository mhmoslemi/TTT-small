    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def tournament_selection(population, fitness, size):
    idxs = np.random.choice(len(population), size)
    best_idx = idxs[np.argmin(fitness[idxs])]
    return population[best_idx].copy()

def run():
    n_points = 80
    dx = 2.0 / n_points
    desired_sum = n_points / 2
    pop_size = 50
    start = time.time()
    population = [np.random.rand(n_points) for _ in range(pop_size)]
    for h in population:
        h[:] = repair(h, desired_sum)

    best_c5 = np.inf
    best_h = None

    while time.time() - start < 1100:
        fitness = np.array([evaluate_c5(h, dx) for h in population])
        current_best = population[np.argmin(fitness)]
        if fitness.min() < best_c5:
            best_c5 = fitness.min()
            best_h = current_best.copy()

        offspring = []
        for _ in range(pop_size // 2):
            p1 = tournament_selection(population, fitness, 5)
            p2 = tournament_selection(population, fitness, 5)
            mask = np.random.rand(n_points) < 0.5
            c1 = np.where(mask, p1, p2)
            c2 = np.where(~mask, p1, p2)
            if np.random.rand() < 0.1:
                c1 += np.random.normal(0, 0.05, n_points)
                c1 = repair(c1, desired_sum)
            if np.random.rand() < 0.1:
                c2 += np.random.normal(0, 0.05, n_points)
                c2 = repair(c2, desired_sum)
            offspring.extend([c1, c2])
        population = offspring

    return (best_h, best_c5, n_points)
