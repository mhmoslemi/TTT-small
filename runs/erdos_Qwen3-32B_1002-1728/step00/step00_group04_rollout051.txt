    def compute_gradient(h, k):
        grad = np.zeros(n_points)
        for i in range(n_points - k):
            grad[i] += 1 - h[i + k]
            grad[i + k] -= h[i]
        return grad
    
    # Local optimization with gradient descent
    def optimize(h):
        best_h = h.copy()
        best_val = evaluate(h)
        current_h = h.copy()
        lr = 0.01
        for _ in range(100):
            k_max = np.argmax(np.correlate(current_h, 1 - current_h, mode='full')[:n_points])
            grad = compute_gradient(current_h, k_max) + np.random.normal(0, 1e-3, n_points)  # tiny noise for pressure
            current_h = project_feasible(current_h - lr * grad)
            val = evaluate(current_h)
            if val < best_val:
                best_val, best_h = val, current_h.copy()
            else:
                lr *= 0.95  # decay learning rate
            if time.time() - start_time > budget_s - 5:
                break
        return best_h, best_val
    
    # Genetic operators
    def crossover(p1, p2):
        idxs = np.sort(np.random.choice(n_points, 2, replace=False))
        child = np.where(np.arange(n_points) < idxs[0], p1,
                        np.where(np.arange(n_points) < idxs[1], p2,
                               p1))
        return project_feasible(child)
    
    def mutate(h):
        h = h + np.random.normal(0, 0.1, n_points)
        return project_feasible(h)
    
    # Track best solution
    best_h, best_val = optimize(h_initial)[0], 1e9
    
    # Evolution loop
    while time.time() - start_time < budget_s:
        # Evaluate population
        fitness = [(evaluate(ind), ind) for ind in population]
        fitness.sort()
        for val, ind in fitness:
            if val < best_val:
                best_val, best_h = val, ind.copy()
        
        # Select top half
        top_half = [ind for val, ind in fitness[:10]]
        
        # Generate new population
        next_population = top_half.copy()
        
        # Crossover and mutation
        for _ in range(10):
            p1, p2 = np.random.choice(top_half, 2, replace=False)
            next_population.append(crossover(p1, p2))
            if np.random.rand() < 0.1:
                next_population.append(mutate(p1))
        
        # Local optimization on top
        for ind in top_half[:2]:
            optimized, val = optimize(ind)
            next_population.append(optimized)
        
        # Maintain diversity
        next_population.extend([generate_individual() for _ in range(5)])
        
        # Truncate to population size
        population = [ind for val, ind in sorted([(evaluate(ind), ind) for ind in next_population], 
                                               key=lambda x: x[0])[:20]]
    
    return best_h, best_val, n_points
