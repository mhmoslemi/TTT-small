import numpy as np
import time
import cvxpy as cp

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    target = 1.0 - h
    corr = np.correlate(h, target, mode='full')
    return np.max(corr) * dx

def project_cvxpy(h, target_sum):
    n = len(h)
    h_clipped = np.clip(h, 0, 1)
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - h_clipped))
    constraints = [cp.sum(x) == target_sum, 0 <= x, x <= 1]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
    return x.value if prob.status == cp.OPTIMAL else h_clipped

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    target_sum = n_points / 2

    try:
        h_initial = initial_h_values.copy()
    except NameError:
        h_initial = np.random.beta(2, 2, n_points)
        h_initial = h_initial / h_initial.sum() * target_sum
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = project_cvxpy(h_initial, target_sum)
    
    m = 30
    particles = [h_initial.copy()]
    for _ in range(m - 1):
        h = np.random.beta(2, 2, n_points)
        h = h / h.sum() * target_sum
        h = np.clip(h, 0, 1)
        h = project_cvxpy(h, target_sum)
        particles.append(h)
    
    velocities = [np.zeros(n_points) for _ in range(m)]
    personal_best = [p.copy() for p in particles]
    personal_best_values = [compute_c5(p, n_points) for p in particles]
    global_best_index = np.argmin(personal_best_values)
    global_best = particles[global_best_index].copy()
    best_c5 = personal_best_values[global_best_index]

    omega = 0.7
    c1 = 1.5
    c2 = 1.5
    omega_decay = 0.001

    while time.time() - start_time < budget_s:
        for i in range(m):
            r1, r2 = np.random.rand(2)
            velocities[i] = omega * velocities[i] + c1 * r1 * (personal_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            particles[i] += velocities[i]
            particles[i] = project_cvxpy(particles[i], target_sum)
            c5 = compute_c5(particles[i], n_points)
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best[i] = particles[i].copy()
                if c5 < best_c5:
                    best_c5 = c5
                    global_best = particles[i].copy()
        omega = max(0.1, omega - omega_decay)
    
    return (global_best, best_c5, n_points)
