import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
