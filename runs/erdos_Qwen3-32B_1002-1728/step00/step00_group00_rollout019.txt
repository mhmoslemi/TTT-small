import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    global initial_h_values
    h_init = np.array(initial_h_values).flatten()
    n_points = len(h_init)
    expected_sum = n_points / 2.0
    dx = 2.0 / n_points

    def project(h):
        h = np.clip(h, 0, 1)
        s = h.sum()
        correction = expected_sum - s
        h[-1] += correction
        h[-1] = np.clip(h[-1], 0, 1)
        return h

    best_h = project(h_init)
    best_c5 = evaluate_erdos_solution(best_h)

    def objective(h_flat):
        h = h_flat.reshape(n_points)
        dh = 1.0 - h
        corr = np.correlate(h, dh, mode='full')
        return np.max(corr) * dx

    while time.time() - start_time < budget_s - 10:
        perturbation = np.random.normal(0, 0.2, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        h_perturbed = project(h_perturbed)

        try:
            bounds_sq = [(0, 1)] * n_points
            result = minimize(
                objective,
                h_perturbed,
                method="SLSQP",
                bounds=bounds_sq,
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - expected_sum}
            )
            if result.success:
                h_opt = project(result.x)
                c5_current = evaluate_erdos_solution(h_opt)
                if c5_current < best_c5:
                    best_c5, best_h = c5_current, h_opt
            else:
                result = minimize(
                    objective,
                    h_perturbed,
                    method="Nelder-Mead",
                    bounds=bounds_sq
                )
                if result.success:
                    h_opt = project(result.x)
                    c5_current = evaluate_erdos_solution(h_opt)
                    if c5_current < best_c5:
                        best_c5, best_h = c5_current, h_opt
        except:
            pass

    return best_h.tolist(), best_c5, n_points
